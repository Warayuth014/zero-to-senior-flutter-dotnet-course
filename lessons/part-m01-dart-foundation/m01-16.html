6. StreamController: เมื่อเราเป็นคนป้อนเอง

final controller = StreamController<String>();

final subscription = controller.stream.listen(
  (event) => print('ผู้ฟังได้รับ: $event'),
  onError: (Object error) => print('จับ error: $error'),
  onDone: () => print('ปิด stream แล้ว'),
);

controller.add('เริ่มรอบตรวจนับ');
await controller.close();       // ผู้สร้างปิด
await subscription.cancel();    // ผู้ฟังยกเลิก

listen รับ callback ได้สามตัวตามที่เรียนใน D1.11 — ตัวแรกรับข้อมูล, onError รับข้อผิดพลาด, onDone ทำงานเมื่อ stream ปิด

สิ่งสำคัญคือ ข้อผิดพลาดใน stream ไม่ทำให้ stream จบ ถ้ามี onError ดักไว้ ข้อมูลตัวถัดไปจะยังส่งมาตามปกติ

8. ใครปิดอะไร

บทบาทความรับผิดชอบถ้าลืม
ผู้สร้าง StreamControllerเรียก close()ทรัพยากรค้าง และ onDone ไม่เคยทำงาน
ผู้เรียก listen()เรียก cancel()callback ยังทำงานหลังผู้ใช้ออกจากหน้าจอ

ในแอป Flutter จริง อาการของการลืมยกเลิกจะปรากฏเป็น “กดย้อนกลับแล้วแอปพัง” หรือ “ข้อมูลของหน้าจอเก่าโผล่มาทับหน้าจอใหม่” ซึ่งเป็นบั๊กที่หาสาเหตุยาก เพราะเกิดเฉพาะตอนสลับหน้าจอเร็ว ๆ

ตอนนี้ยังไม่มีหน้าจอให้จัดการ แต่ให้ติดนิสัยตั้งแต่วันนี้ว่า ทุกครั้งที่พิมพ์ .listen( ให้คิดทันทีว่าจะ cancel() ที่ไหน

event หายไปเฉย ๆ ใน broadcastส่งก่อนที่จะมีคนฟังฟังให้เรียบร้อยก่อนเริ่มส่ง

10. แบบฝึกหัด

สร้าง ScanBus ที่จำลองเครื่องสแกนของ PDA โดย

  1. ให้หลายส่วนของแอปฟังพร้อมกันได้
  2. ตัดช่องว่างและแปลงเป็นตัวพิมพ์ใหญ่ก่อนส่งออก
  3. ไม่ส่งค่าว่างออกไป
  4. ไม่ส่งค่าเดิมซ้ำติดกัน
  5. มี dispose() สำหรับปิดให้เรียบร้อย
ดูแนวคำตอบและเหตุผล
import 'dart:async';

class ScanBus {
  final _controller = StreamController<String>.broadcast();

  Stream<String> get scans => _controller.stream
      .map((raw) => raw.trim().toUpperCase())
      .where((code) => code.isNotEmpty)
      .distinct();

  void emit(String raw) {
    if (_controller.isClosed) return;
    _controller.add(raw);
  }

  Future<void> dispose() => _controller.close();
}

Future<void> main() async {
  final bus = ScanBus();

  final a = bus.scans.listen((code) => print('หน้าจอรายการ: $code'));
  final b = bus.scans.listen((code) => print('ตัวนับ: $code'));

  bus.emit(' pal-1001 ');
  bus.emit('PAL-1001');    // ซ้ำติดกัน -> ถูกตัด
  bus.emit('   ');         // ว่าง -> ถูกตัด
  bus.emit('pal-1002');

  await Future<void>.delayed(Duration.zero);
  await a.cancel();
  await b.cancel();
  await bus.dispose();
}

ทำไม _controller ต้อง private และเปิดเฉพาะ getter scans: ตามหลักจาก D1.7 — ถ้าเปิด controller ออกไปตรง ๆ ใครก็ปิดมันได้ หรือส่งค่าดิบที่ยังไม่ผ่านการกรองเข้ามาได้ การเปิดเฉพาะ Stream ทำให้คนนอกทำได้อย่างเดียวคือฟัง

ทำไมต้องเป็น broadcast: เพราะโจทย์ข้อ 1 บอกว่าต้องมีหลายส่วนฟังพร้อมกัน ถ้าใช้แบบปกติ ผู้ฟังคนที่สองจะถูกปฏิเสธทันที

จุดที่ต้องเข้าใจให้ลึก: .distinct() อยู่ใน getter จึงถูกสร้างใหม่ทุกครั้งที่มีคนเรียก scans แปลว่าผู้ฟังแต่ละคนมีตัวจำค่าล่าสุดของตัวเอง ในตัวอย่างนี้ผลออกมาเหมือนกันจึงไม่เห็นความต่าง แต่ถ้าผู้ฟังคนหนึ่งมาทีหลัง ค่าที่มันถือว่า “ซ้ำ” จะไม่ตรงกับอีกคน — ถ้าต้องการให้ทุกคนเห็นเหมือนกันเป๊ะ ต้องกรองก่อนเข้า controller แทน

ทำไมต้อง Future.delayed(Duration.zero) ก่อนยกเลิก: เพราะ event ที่ add เข้าไปถูกส่งแบบไม่ทันที ถ้ายกเลิกทันทีจะไม่มีใครได้รับอะไรเลย — เป็นตัวอย่างของ “โลกเปลี่ยนไประหว่างที่เรารอ” จาก D1.15

ลองต่อ: ย้าย .map, .where, .distinct() ไปทำก่อนเรียก _controller.add แทน แล้วเทียบว่าพฤติกรรมต่างกันอย่างไรเมื่อมีผู้ฟังเข้ามาทีหลัง

11. สรุปท้ายบท

  • Future ให้ค่าเดียว Stream ให้หลายค่าตามเวลา
  • async* คู่กับ yield ใช้ผลิต stream ทีละค่า
  • map/where/distinct ทำงานเหมือนกับที่ใช้บน collection
  • ข้อผิดพลาดใน stream ไม่ทำให้ stream จบ ถ้ามี onError ดักไว้
  • แบบผู้ฟังคนเดียวเป็นค่าเริ่มต้น ส่วน broadcast ทำให้ค่าที่ส่งก่อนมีคนฟังหายไป
  • ใครสร้างต้อง close() ใครฟังต้อง cancel()
  • distinct() กันเฉพาะค่าซ้ำที่ติดกัน
