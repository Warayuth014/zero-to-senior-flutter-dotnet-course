1. ปัญหาที่บทนี้แก้

สถานะของงานหนึ่งใบมีได้ไม่กี่แบบ: รอเริ่ม, กำลังทำ, เสร็จ, ยกเลิก วิธีที่คนใหม่มักเขียนคือใช้ข้อความ

String status = 'inProgress';

if (status == 'inprogress') { ... }   // สะกดไม่ตรง -> เงื่อนไขไม่เคยเป็นจริง
if (status == 'in_progress') { ... }  // อีกคนเขียนอีกแบบ
status = 'กำลังทำ';                    // ใครก็ใส่อะไรก็ได้

ปัญหาคือ compiler ช่วยอะไรไม่ได้เลย เพราะทุกอย่างเป็นข้อความที่ถูกต้องทั้งหมดในสายตาของมัน บั๊กจะไปโผล่ตอนผู้ใช้กดปุ่มแล้วไม่มีอะไรเกิดขึ้น

enum คือการบอก Dart ว่า “ค่าที่เป็นไปได้มีเท่านี้เท่านั้น” แล้วให้เครื่องมือช่วยตรวจแทนเรา

3. Mental model: รายชื่อที่ปิดแล้ว

enum คือรายชื่อที่ “ปิดรับสมัคร” แล้ว — เพิ่มไม่ได้ตอนโปรแกรมทำงาน มีได้เฉพาะที่เราเขียนไว้เท่านั้น ประโยชน์ที่ตามมาคือสามอย่าง

ประโยชน์เทียบกับใช้ข้อความ
สะกดผิดไม่ได้ข้อความสะกดผิดได้ตลอดเวลา
เครื่องมือเติมคำให้อัตโนมัติต้องจำเองว่ามีค่าอะไรบ้าง
เพิ่มสถานะใหม่แล้วรู้ทันทีว่าต้องไปแก้ตรงไหนไม่มีอะไรเตือน ต้องไล่หาเอง
ประโยชน์ข้อสามคือของจริง: วันที่ธุรกิจเพิ่มสถานะ “พักไว้ชั่วคราว” เข้ามา ถ้าใช้ enum analyzer จะขีดเส้นแดงทุกจุดที่ยังไม่รองรับให้ทันที ถ้าใช้ข้อความ คุณจะรู้ตอนลูกค้าโทรมา

4. enum แบบพื้นฐาน

enum PickStatus { pending, inProgress, done, cancelled }

สิ่งที่ enum ให้มาฟรีใช้ทำอะไร

5. switch ที่บังคับให้จัดการครบ

String messageFor(PickStatus status) => switch (status) {
  PickStatus.pending => 'รอเริ่มงาน',
  PickStatus.inProgress => 'กำลังหยิบ',
  PickStatus.done => 'เสร็จแล้ว',
  PickStatus.cancelled => 'ยกเลิก',
};

ลองลบบรรทัดใดบรรทัดหนึ่งออกแล้วรัน flutter analyze — จะได้ error ทันทีว่ายังจัดการไม่ครบ นี่คือคุณสมบัติที่มีค่าที่สุดของ enum

สิ่งที่ทำลายประโยชน์นี้: การเติม _ => 'อื่น ๆ' ปิดท้าย เพราะมันบอก Dart ว่า “ที่เหลือฉันจัดการหมดแล้ว” ทำให้ไม่มีอะไรเตือนเมื่อเพิ่มสถานะใหม่ ใช้ _ เฉพาะเมื่อคุณตั้งใจให้ค่าที่ไม่รู้จักถูกจัดการแบบเดียวกันจริง ๆ

6. Enhanced enum: enum ที่พกข้อมูลติดตัว

นี่คือจุดที่ enum ของ Dart ทรงพลังกว่าหลายภาษา — แต่ละค่ามี field และ method ของตัวเองได้

enum TaskType {
  putaway('PUTAWAY', 'จัดเก็บ', true),
  retrieve('RETRIEVE', 'หยิบออก', true),
  ret('RETURN', 'คืนของ', false);

  const TaskType(this.wireValue, this.label, this.needsLocation);

  final String wireValue;      // ค่าที่ API ใช้
  final String label;          // ข้อความที่ผู้ใช้เห็น
  final bool needsLocation;    // กฎของงานชนิดนี้
}

สังเกตสามอย่าง

  1. รายการค่าต้องอยู่บนสุด และปิดท้ายด้วย เซมิโคลอน ก่อนจะเขียนส่วนอื่น
  2. constructor ต้องเป็น const เพราะค่า enum ทุกตัวเป็นค่าคงที่
  3. ชื่อ ret ใช้แทน return เพราะ return เป็นคำสงวนของภาษา
PUTAWAY   | จัดเก็บ   | ต้องระบุช่องเก็บ: true
RETRIEVE  | หยิบออก  | ต้องระบุช่องเก็บ: true
RETURN    | คืนของ   | ต้องระบุช่องเก็บ: false

ผลคือกฎ “งานจัดเก็บต้องระบุช่องเก็บ” ถูกเก็บไว้ที่เดียวคือในนิยามของ enum แทนที่จะกระจายเป็น if ตามหน้าจอต่าง ๆ

7. แยกค่าที่ API ใช้ ออกจากชื่อที่เราใช้

นี่เป็นแนวปฏิบัติที่สำคัญมากเมื่อต้องคุยกับ .NET API

static TaskType fromWire(String raw) {
  for (final type in TaskType.values) {
    if (type.wireValue == raw.toUpperCase()) return type;
  }
  throw ArgumentError('ไม่รู้จักงานชนิด "$raw"');
}
ใช้ .name ตรง ๆแยก wireValue
เปลี่ยนชื่อตัวแปรในโค้ดสัญญากับ server พังทันทีไม่กระทบ
API ส่งค่าตัวพิมพ์ใหญ่ต้องแปลงกระจายทุกที่แปลงที่เดียวใน fromWire
API ใช้ค่าเก่าที่ตั้งชื่อไม่สวยต้องทนตั้งชื่อตามตั้งชื่อในโค้ดตามใจได้

คำถามสำคัญ: เมื่อ API ส่งค่าที่เราไม่รู้จักมา ควรทำอย่างไร มีสองทางและต้องเลือกอย่างตั้งใจ

ทางเลือกผลที่ได้เหมาะเมื่อ
โยน error ทันทีรู้ปัญหาเร็ว ข้อมูลไม่เพี้ยนค่าที่เพิ่มใหม่ต้องอัปเดตแอปอยู่แล้ว
มีค่า unknown รองรับแอปเก่ายังใช้งานต่อได้server เพิ่มค่าได้โดยไม่ปล่อยแอปใหม่

ทั้งสองทางถูก แต่ที่ผิดคือการไม่ตัดสินใจแล้วปล่อยให้โปรแกรมพังแบบไม่ตั้งใจ

8. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
The type 'PickStatus' is not exhaustively matchedswitch ยังไม่ครอบคลุมทุกค่าเพิ่มกรณีที่ขาด — อย่ารีบเติม _
Expected to find ';' ใน enhanced enumลืมเซมิโคลอนหลังรายการค่าสุดท้ายปิดรายการด้วย ; ก่อนเขียน constructor
Const constructors can't have a bodyเขียนตรรกะใน constructor ของ enumย้ายไปเป็น getter หรือ method
ข้อมูลเก่าในฐานข้อมูลแปลความหมายผิดหลังอัปเดตแอปเก็บ .index ลงฐานข้อมูลเก็บ wireValue ที่เป็นข้อความแทน
แอปพังเมื่อ server เพิ่มสถานะใหม่ไม่ได้ตัดสินใจเรื่องค่าที่ไม่รู้จักเลือกระหว่างโยน error กับมี unknown แล้วเขียนเทสต์ยืนยัน
ใช้ enum กับข้อมูลที่เพิ่มได้เรื่อย ๆ เช่นรายชื่อคลังข้อมูลนั้นไม่ใช่ชุดปิดใช้ข้อมูลจาก server เป็น list ธรรมดาแทน

9. แบบฝึกหัด

หน้าจอหนึ่งใช้ตัวแปรสามตัวเก็บสถานะการโหลด

bool isLoading = false;
bool hasError = false;
bool isDone = false;
  1. อธิบายว่าการเขียนแบบนี้เปิดช่องให้เกิดสถานะที่เป็นไปไม่ได้อย่างไร ยกตัวอย่างมาอย่างน้อยสองแบบ
  2. แปลงเป็น enum พร้อมข้อความภาษาไทยติดตัวแต่ละค่า
  3. เขียน switch ที่คืนข้อความสำหรับแสดงผล
ดูแนวคำตอบและเหตุผล

ข้อ 1 — สถานะที่เป็นไปไม่ได้: ตัวแปร bool สามตัวสร้างชุดค่าที่ต่างกันได้ 8 แบบ แต่มีเพียง 4 แบบที่มีความหมายจริง ตัวอย่างที่ผิดพลาดได้

  • isLoading = true พร้อม isDone = true — กำลังโหลดและเสร็จแล้วพร้อมกัน
  • hasError = true พร้อม isDone = true — สำเร็จและล้มเหลวพร้อมกัน
  • ทั้งสามเป็น false — หน้าจอจะแสดงอะไร ไม่มีใครตอบได้

อาการจริงที่ตามมาคือหน้าจอแสดงวงกลมหมุนทับข้อความ error หรือค้างอยู่ที่หน้าว่างเปล่า

enum LoadState {
  initial('เริ่มต้น'),
  loading('กำลังโหลด...'),
  success('โหลดสำเร็จ'),
  empty('ไม่พบข้อมูล'),
  failure('โหลดไม่สำเร็จ');

  const LoadState(this.message);
  final String message;

  bool get canRetry => this == failure;
}

String display(LoadState state) => switch (state) {
  LoadState.initial => 'กดปุ่มเพื่อเริ่ม',
  LoadState.loading => state.message,
  LoadState.success => state.message,
  LoadState.empty => 'ยังไม่มีงานในกะนี้',
  LoadState.failure => '${state.message} — ลองอีกครั้ง',
};

ทำไมต้องแยก empty ออกจาก success: เพราะ “โหลดสำเร็จแต่ไม่มีข้อมูล” กับ “โหลดสำเร็จและมีข้อมูล” ต้องแสดงหน้าจอคนละแบบ ถ้ารวมกันเป็นสถานะเดียว โค้ดหน้าจอจะต้องกลับไปเช็ค list.isEmpty อีกที ซึ่งเป็นการกระจายตรรกะออกไปอีกจุด

ผลลัพธ์ที่ได้: จาก 8 ชุดค่าที่เป็นไปได้ เหลือ 5 ที่มีความหมายทั้งหมด ไม่มีทางเข้าสถานะที่ขัดแย้งกันเองได้อีก — นี่คือการ “ทำให้สถานะที่ผิดเขียนออกมาไม่ได้เลย” ซึ่งเป็นหลักคิดที่จะใช้ต่อไปทั้งคอร์ส

ลองต่อ: เพิ่มค่า refreshing เข้าไป แล้วรัน flutter analyze — จะเห็นว่ามันชี้จุดที่ต้องแก้ให้ทันที ลองเทียบกับกรณีที่ใช้ bool สามตัวว่าคุณจะรู้ได้อย่างไรว่าต้องแก้ตรงไหนบ้าง

10. สรุปท้ายบท

  • enum ใช้กับ “ชุดค่าที่ปิดแล้ว” — รู้ครบตั้งแต่ตอนเขียนโค้ด
  • ประโยชน์ที่ใหญ่ที่สุดคือ switch ที่เตือนเมื่อจัดการไม่ครบ
  • enhanced enum เก็บ label, ค่าสำหรับ API และกฎของแต่ละค่าไว้ในที่เดียว
  • อย่าเก็บ .index ลงฐานข้อมูล ให้ใช้ค่าข้อความที่กำหนดเอง
  • ตัดสินใจล่วงหน้าว่าจะทำอย่างไรกับค่าที่ไม่รู้จักจาก server
  • แทน bool หลายตัวด้วย enum เพื่อกำจัดสถานะที่ขัดแย้งกันเอง
