1. ปัญหาที่บทนี้แก้

ใน 2.6 เราเลือก ListView.builder ได้แล้ว แต่รายการในแอปจริงต้องตอบคำถามอีกสี่ข้อที่ตัวอย่างพื้นฐานไม่ได้ครอบคลุม

  1. ถ้าไม่มีข้อมูลเลย จะแสดงอะไร
  2. ระหว่างโหลดจะแสดงอะไร
  3. ตัวคั่นระหว่างแถวทำอย่างไรให้ถูก
  4. บนแท็บเล็ตควรแสดงกี่คอลัมน์

2. คำศัพท์ที่ต้องรู้ก่อน

คำความหมาย
itemCountจำนวนแถวทั้งหมด
itemExtentความสูงคงที่ของทุกแถว
delegateตัวกำหนดกติกาการจัดวางของตาราง
empty stateหน้าตาเมื่อไม่มีข้อมูล
paginationการโหลดข้อมูลทีละหน้า

3. Mental model: สี่สถานะที่ทุกรายการต้องมี

รายการหนึ่งรายการไม่ได้มีแค่ “มีข้อมูล” กับ “ไม่มีข้อมูล” แต่มีสี่สถานะที่ต้องแสดงต่างกัน

สถานะผู้ใช้ควรเห็นทำอะไรต่อได้
กำลังโหลดตัวบอกว่ากำลังทำงานรอ
มีข้อมูลรายการทำงาน
ว่าง“ยังไม่มีงานในกะนี้”รู้ว่าไม่ใช่ความผิดพลาด
ล้มเหลวข้อความ + ปุ่มลองใหม่ลองใหม่
ทบทวนจาก D1.10: นี่คือเหตุผลที่เราแนะนำให้ใช้ enum แทน bool หลายตัว — เพราะสี่สถานะนี้ต้องเกิดทีละหนึ่งเท่านั้น และการแสดง “ไม่มีงาน” ทั้งที่ระบบล่มเป็นบั๊กที่ทำให้พนักงานกลับบ้านทั้งที่มีของรออยู่ (D1.14 หัวข้อ 9)

4. ListView.builder ที่ครบถ้วน

Widget build(BuildContext context) {
  return switch (state) {
    LoadState.loading => const Center(child: CircularProgressIndicator()),
    LoadState.failure => ErrorView(onRetry: _load),
    LoadState.empty => const EmptyView(message: 'ยังไม่มีงานในกะนี้'),
    LoadState.success => ListView.builder(
      itemCount: lines.length,
      itemBuilder: (context, index) {
        final line = lines[index];
        return InboundLineCard(
          key: ValueKey<String>(line.palletCode),   // ตัวตนจากข้อมูล ไม่ใช่ตำแหน่ง
          line: line,
        );
      },
    ),
  };
}

สามสิ่งที่ต้องมีเสมอ

สิ่งที่ต้องมีเหตุผล
itemCountถ้าไม่ใส่ รายการจะเลื่อนไม่รู้จบและ itemBuilder จะถูกเรียกเกินข้อมูลที่มี
key จากรหัสจริงป้องกันข้อมูลสลับแถวเมื่อเรียงใหม่ (1.8)
แยกสถานะให้ครบสี่ผู้ใช้แยกออกว่า “ไม่มีของ” กับ “ระบบพัง” ต่างกัน

5. ตัวคั่นระหว่างแถว

// ถูก — ใช้ตัวที่ออกแบบมาเพื่อการนี้
ListView.separated(
  itemCount: lines.length,
  separatorBuilder: (context, index) => const Divider(height: 1),
  itemBuilder: (context, index) => LineCard(line: lines[index]),
)

// ผิด — แทรกตัวคั่นเข้าไปในข้อมูลแล้วคำนวณ index เอง
ListView.builder(
  itemCount: lines.length * 2 - 1,
  itemBuilder: (context, index) {
    if (index.isOdd) return const Divider();
    return LineCard(line: lines[index ~/ 2]);   // อ่านยากและพลาดง่าย
  },
)

แบบหลังใช้งานได้จริง แต่ทำให้การใส่ key ยุ่งยาก และเมื่อต้องเพิ่มความสามารถอย่างการลบแถว การคำนวณ index จะซับซ้อนขึ้นเรื่อย ๆ จนผิดพลาด

ทางเลือกที่ง่ายกว่าตัวคั่นคือใช้ Card ที่มีระยะขอบในตัว อย่างที่ lab ทำ — แต่ละใบแยกกันด้วยเงาและช่องว่าง โดยไม่ต้องมีเส้นคั่นเลย

6. GridView: เลือกจำนวนคอลัมน์

มีสองวิธีกำหนด และเลือกต่างกันตามเจตนา

// แบบที่ 1 — กำหนดจำนวนคอลัมน์ตรง ๆ
GridView.builder(
  gridDelegate: const SliverGridDelegateWithFixedCrossAxisCount(
    crossAxisCount: 2,
    mainAxisSpacing: 12,
    crossAxisSpacing: 12,
    mainAxisExtent: 120,      // ความสูงของแต่ละช่อง
  ),
  itemCount: menus.length,
  itemBuilder: (context, index) => MenuTile(menu: menus[index]),
)

// แบบที่ 2 — กำหนดความกว้างสูงสุด แล้วให้ Flutter หาจำนวนคอลัมน์เอง
gridDelegate: const SliverGridDelegateWithMaxCrossAxisExtent(
  maxCrossAxisExtent: 260,
  mainAxisSpacing: 12,
  crossAxisSpacing: 12,
),
กำหนดจำนวนคอลัมน์กำหนดความกว้างสูงสุด
ควบคุมได้แม่นยำมากน้อยกว่า
ปรับตามจอกว้างต้องเขียนเงื่อนไขเองปรับให้อัตโนมัติ
เหมาะกับเมนูหลักบน PDA ที่ต้องเห็นครบในหน้าเดียวรายการสินค้าบนแท็บเล็ตหรือเว็บ
อย่าใส่ทั้ง childAspectRatio และ mainAxisExtent พร้อมกัน เพราะทั้งคู่กำหนดความสูงคนละวิธี — ใช้ mainAxisExtent เมื่อรู้ความสูงที่ต้องการ และใช้ childAspectRatio เมื่อสัดส่วนสำคัญกว่า เช่นรูปภาพ

7. ระวังเรื่องขนาดตัวอักษร

เมื่อล็อกความสูงของช่องด้วย mainAxisExtent หรือ itemExtent ต้องทดสอบกับผู้ใช้ที่ตั้งขนาดตัวอักษรใหญ่ขึ้นด้วย

ข้อความไทยที่พอดีในความสูง 120 อาจล้นเมื่อผู้ใช้ตั้งขนาดตัวอักษรเป็น 130% ซึ่งพนักงานคลังที่อายุมากมักตั้งไว้ ผลคือแถบสีเหลืองดำโผล่ในทุกช่องของตาราง

ทางแก้คือเผื่อความสูงไว้ หรือใช้ maxLines กับ ellipsis ตามที่จะเรียนใน 2.11 — และทดสอบกับขนาดตัวอักษรที่ใหญ่ที่สุดที่รองรับเสมอ

8. โหลดทีละหน้า

เมื่อข้อมูลมีเป็นพัน การโหลดมาทั้งหมดครั้งเดียวจะช้าและกินหน่วยความจำ วิธีที่ใช้กันคือโหลดเพิ่มเมื่อผู้ใช้เลื่อนใกล้ท้ายรายการ

ListView.builder(
  itemCount: lines.length + (hasMore ? 1 : 0),   // เผื่อแถวสุดท้ายไว้แสดงสถานะ
  itemBuilder: (context, index) {
    if (index >= lines.length) {
      _loadMoreIfNeeded();                       // ต้องกันเรียกซ้ำ (D1.15)
      return const Center(child: CircularProgressIndicator());
    }
    return LineCard(
      key: ValueKey<String>(lines[index].palletCode),
      line: lines[index],
    );
  },
)

จุดที่ต้องระวังคือ itemBuilder ถูกเรียกได้หลายครั้งต่อการเลื่อนหนึ่งครั้ง จึงต้องกันการยิงซ้ำด้วยธงเหมือนที่ทำใน D1.15 หัวข้อ 7 ไม่งั้นจะยิง API สิบครั้งพร้อมกัน

สำหรับงานคลัง มักดีกว่าถ้าให้ผู้ใช้ค้นหาแทนการเลื่อนดูทั้งหมด — พนักงานที่ถือ PDA ไม่ได้อยากเลื่อนผ่านสองพันรายการเพื่อหาพาเลทใบเดียว

9. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
รายการเลื่อนไม่รู้จบลืมใส่ itemCountใส่เสมอ
RangeError ใน itemBuilderitemCount ไม่ตรงกับข้อมูลจริงตรวจว่าใช้ list ตัวเดียวกัน
ติ๊กถูกย้ายไปแถวอื่นหลังเรียงใหม่ใช้ index เป็นตัวตนใช้ ValueKey จากรหัสจริง และเก็บสถานะโดยอ้างรหัส
หน้าจอบอกว่าไม่มีข้อมูลทั้งที่ระบบล่มดักข้อผิดพลาดแล้วคืนรายการว่างแยกสี่สถานะให้ครบ
ช่องในตารางล้นเมื่อผู้ใช้ตั้งตัวอักษรใหญ่ล็อกความสูงตายตัวเผื่อความสูง หรือใช้ maxLines + ellipsis
ยิง API ซ้ำสิบครั้งตอนเลื่อนถึงท้ายไม่ได้กันการเรียกซ้ำใช้ธงกันตามแบบ D1.15
ใส่ทั้ง childAspectRatio และ mainAxisExtentกำหนดความสูงซ้อนกันเลือกอย่างเดียว

10. แบบฝึกหัด

ทำหน้าจอเมนูหลักของแอปคลัง ที่มีเมนู 6 รายการ โดย

  1. บน PDA กว้าง 320–480 ต้องเป็น 2 คอลัมน์ และเห็นครบทั้ง 6 รายการโดยไม่ต้องเลื่อน
  2. บนแท็บเล็ตกว้างกว่า 480 ให้ช่องกว้างไม่เกิน 260 และปรับจำนวนคอลัมน์เอง
  3. ปุ่มแต่ละช่องต้องสูงอย่างน้อย 48
ดูแนวคำตอบและเหตุผล
LayoutBuilder(
  builder: (context, constraints) {
    final isCompact = constraints.maxWidth < Breakpoints.compact;

    if (isCompact) {
      // ข้อ 1 — คำนวณความสูงช่องให้ 3 แถวพอดีพื้นที่ที่มี
      const rows = 3;
      const gap = 12.0;
      final tileHeight = (constraints.maxHeight - gap * (rows - 1)) / rows;

      return GridView.builder(
        gridDelegate: SliverGridDelegateWithFixedCrossAxisCount(
          crossAxisCount: 2,
          mainAxisSpacing: gap,
          crossAxisSpacing: gap,
          mainAxisExtent: tileHeight.clamp(Breakpoints.minTapTarget, 200),
        ),
        itemCount: menus.length,
        itemBuilder: (context, index) => MenuTile(
          key: ValueKey<String>(menus[index].id),
          menu: menus[index],
        ),
      );
    }

    // ข้อ 2 — ให้ Flutter หาจำนวนคอลัมน์จากความกว้างสูงสุดต่อช่อง
    return GridView.builder(
      gridDelegate: const SliverGridDelegateWithMaxCrossAxisExtent(
        maxCrossAxisExtent: 260,
        mainAxisSpacing: 12,
        crossAxisSpacing: 12,
        mainAxisExtent: 140,
      ),
      itemCount: menus.length,
      itemBuilder: (context, index) => MenuTile(
        key: ValueKey<String>(menus[index].id),
        menu: menus[index],
      ),
    );
  },
)

ทำไมบน PDA ต้องคำนวณความสูงเอง แทนที่จะใช้ค่าคงที่

เพราะโจทย์ข้อ 1 บอกว่า “ต้องเห็นครบโดยไม่ต้องเลื่อน” ซึ่งเป็นข้อกำหนดที่มาจากงานจริง — พนักงานที่ถือ PDA ในมือข้างหนึ่งและถือของอีกข้าง ไม่ควรต้องเลื่อนหาเมนูหลัก การคำนวณจากพื้นที่ที่ได้รับจริง (ตามกฎข้อ 1 ใน 2.1) ทำให้พอดีเสมอไม่ว่าจอจะสูงเท่าไร

ทำไมมี .clamp()

เพื่อกันกรณีสุดโต่งสองด้าน — ถ้าจอเตี้ยมาก (เช่นเมื่อแป้นพิมพ์ขึ้น) ช่องอาจคำนวณได้เตี้ยกว่า 48 จนกดไม่โดน และถ้าจอสูงมากช่องจะยืดจนดูประหลาด clamp จำกัดทั้งสองด้านในบรรทัดเดียว

ทำไมสองกรณีใช้ delegate คนละแบบ

จอเจตนาเครื่องมือที่ตรงกับเจตนา
PDA“ต้องเห็นครบ 6 ช่อง ใน 2 คอลัมน์”กำหนดจำนวนคอลัมน์ตรง ๆ
แท็บเล็ต“ช่องอย่าใหญ่เกินไป ที่เหลือแล้วแต่พื้นที่”กำหนดความกว้างสูงสุด

นี่คือประเด็นที่สำคัญที่สุดของข้อนี้ — เลือกเครื่องมือตาม เจตนา ไม่ใช่เลือกตัวใดตัวหนึ่งไปใช้ทั้งแอปเพราะคุ้นมือ

จุดที่คนมักพลาด: ใช้ MediaQuery อ่านความกว้างของทั้งเครื่องแทน LayoutBuilder ซึ่งจะผิดทันทีถ้าตารางนี้ถูกเอาไปวางในแผงด้านข้างที่กว้างแค่ครึ่งจอ — ความต่างของสองตัวนี้คือเรื่องของ 2.9 และ 2.10

ลองต่อ: เขียนเทสต์ที่ตั้งจอเป็น 360 แล้วตรวจว่าเห็นครบ 6 ช่องโดยไม่มี exception จากนั้นตั้งเป็น 1024 แล้วตรวจว่าช่องกว้างไม่เกิน 260 — ดูรูปแบบการตั้งขนาดจอได้จาก test/layout_basics_test.dart

11. สรุปท้ายบท

  • รายการต้องมีสี่สถานะ: กำลังโหลด มีข้อมูล ว่าง ล้มเหลว
  • ใส่ itemCount และ key จากรหัสจริงเสมอ
  • ใช้ ListView.separated แทนการคำนวณ index เอง
  • กำหนดจำนวนคอลัมน์เมื่อต้องควบคุมแม่นยำ กำหนดความกว้างสูงสุดเมื่อต้องการให้ปรับเอง
  • อย่าใส่ childAspectRatio คู่กับ mainAxisExtent
  • ทดสอบกับขนาดตัวอักษรใหญ่สุดที่รองรับ
  • โหลดทีละหน้าต้องกันการยิงซ้ำ

บทถัดไป: 2.8 จะแนะนำ sliver ซึ่งเป็นกลไกที่อยู่เบื้องหลัง ListView และเป็นทางที่ทำให้ส่วนหัวเลื่อนหายไปพร้อมรายการได้ตามที่เจอในแบบฝึกหัดของ 2.6
