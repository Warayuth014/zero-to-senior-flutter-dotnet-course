1. ปัญหาที่บทนี้แก้

เรารู้จัก setState มาตั้งแต่ 1.5 และใช้มาตลอด Part 3 แต่พอแอปโตขึ้นจะเริ่มเจอสามอาการ

  • ข้อมูลชิ้นเดียวต้องส่ง callback ผ่านห้าชั้นเพื่อให้ทุกคนอัปเดตพร้อมกัน
  • สลับหน้าจอแล้วข้อมูลที่โหลดไว้หายหมด ต้องโหลดใหม่
  • ทดสอบตรรกะทีต้องประกอบทั้งหน้าจอ

ทั้งสามอาการมีต้นเหตุเดียวกัน — ข้อมูลอยู่ผิดที่ บทนี้จะทำให้เห็นว่าเมื่อไหร่ setState เหมาะ และเมื่อไหร่ถึงเวลาย้ายออก

3. Mental model: ถามสามคำถาม

ก่อนตัดสินว่าข้อมูลชิ้นหนึ่งควรอยู่ที่ไหน ให้ถามตามลำดับนี้

คำถามถ้าตอบว่า...ควรอยู่ที่
1. ใครต้องใช้บ้างหน้าจอนี้คนเดียวsetState ในหน้าจอ
หลายหน้าจอชั้นที่อยู่เหนือขึ้นไป
2. ต้องอยู่รอดข้ามหน้าจอไหมไม่ต้องsetState
ต้องชั้นที่อายุยาวกว่า
3. มีตรรกะที่ควรทดสอบแยกไหมไม่มีsetState
มีแยกออกมาเป็น class ของตัวเอง
ประโยคเดียวที่ต้องจำ: setState ไม่ได้แย่ — มันเหมาะกับข้อมูลที่ เกิดและตายพร้อมหน้าจอ ปัญหาเกิดเมื่อเราใช้มันกับข้อมูลที่ควรอยู่ยาวกว่านั้น

4. ตัวอย่างที่ setState เหมาะ

class _PickScreenState extends State<PickScreen> {
  bool _showHelp = false;              // เปิดคำแนะนำ
  bool _sortByQuantity = false;        // เรียงตามจำนวน

  void _toggleHelp() => setState(() => _showHelp = !_showHelp);
}

ทั้งสองตัวผ่านเกณฑ์ครบสามข้อ — ใช้ในหน้าจอนี้คนเดียว ไม่ต้องอยู่รอดข้ามหน้าจอ และไม่มีตรรกะที่ต้องทดสอบ

ถ้าผู้ใช้ออกจากหน้าจอแล้วกลับมา คำแนะนำปิดอยู่เหมือนเดิม ซึ่งเป็นพฤติกรรมที่ถูกต้อง

5. ตัวอย่างที่ setState ไม่เหมาะ

class _PickScreenState extends State<PickScreen> {
  List<PickLine> _lines = [];          // มาจาก server
  bool _loading = false;
  String? _error;

  Future<void> _load() async {
    setState(() => _loading = true);
    final result = await api.fetchLines();
    if (!mounted) return;
    setState(() {
      _lines = result;
      _loading = false;
    });
  }
}

โค้ดนี้ทำงานได้และไม่ผิดอะไร แต่ตกเกณฑ์ทั้งสามข้อ

เกณฑ์ปัญหาที่จะเจอ
ใครใช้บ้างหน้าสรุปยอดก็ต้องใช้ ต้องโหลดซ้ำอีกชุด
อยู่รอดข้ามหน้าจอกดดูรายละเอียดแล้วกลับมา ต้องโหลดใหม่ทั้งหมด
ทดสอบแยกจะทดสอบว่าโหลดล้มเหลวแล้วขึ้นอะไร ต้องประกอบทั้งหน้าจอ

บทที่เหลือของ Part นี้คือการย้ายข้อมูลแบบนี้ออกไปทีละขั้น

6. กฎการใช้ setState ที่ต้องจำ

// ผิด — งานที่ต้องรอ อยู่ข้างใน
setState(() async {
  _lines = await api.fetchLines();
});

// ถูก — รอให้เสร็จก่อน แล้วค่อยแจ้ง
final result = await api.fetchLines();
if (!mounted) return;
setState(() => _lines = result);
กฎเหตุผลบทที่อธิบาย
ข้างใน setState ต้องเสร็จทันทีFlutter ต้องรู้ทันทีว่าแก้ค่าเสร็จแล้ว1.5
ห้ามเรียกใน buildจะวนไม่รู้จบ1.9
ตรวจ mounted หลัง awaitหน้าจออาจหายไปแล้ว1.6, 3.13
ห้ามเรียกใน disposeหน้าจอกำลังถูกทำลาย3.12

7. อย่าเรียกเมื่อค่าไม่เปลี่ยน

void _select(SortMode next) {
  if (_sortMode == next) return;      // ไม่มีอะไรเปลี่ยน ไม่ต้องแจ้ง
  setState(() => _sortMode = next);
}

เป็นการลดงานที่ไม่จำเป็นตามหลักใน 1.10 — แต่ระวังกับดักหนึ่งข้อ

การเทียบแบบนี้ใช้ได้กับค่าเดี่ยวอย่าง enum หรือ bool แต่ ใช้กับ list ไม่ได้ ถ้าเราแก้ข้างใน list ก้อนเดิม การเทียบจะบอกว่า “เหมือนเดิม” ทั้งที่เนื้อหาเปลี่ยนไปแล้ว — ทบทวน D1.5 หัวข้อ 7
// อันตราย — แก้ก้อนเดิม
_lines.add(newLine);

// ปลอดภัย — สร้างก้อนใหม่
_lines = [..._lines, newLine];

ใน lab ของ Part นี้เราใช้แบบหลังเสมอ

void _setLine(int index, PickLine line) {
  // สร้าง list ก้อนใหม่ ไม่แก้ก้อนเดิม (D1.5)
  final next = [..._lines];
  next[index] = line;
  _lines = next;
}

9. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
กดแล้วไม่มีอะไรเกิดขึ้นเปลี่ยนค่าโดยไม่เรียก setStateห่อการเปลี่ยนค่าไว้ข้างใน
เพิ่มลง list แล้วหน้าจอไม่อัปเดตแก้ก้อนเดิมแล้วมีการเทียบค่าสร้างก้อนใหม่ด้วย [...old, x]
setState() called during buildเรียกใน buildย้ายไป callback หรือ initState
setState() called after dispose()ไม่ตรวจ mountedตรวจหลัง await ทุกครั้ง
ต้องส่ง callback ผ่านหลายชั้นข้อมูลอยู่ต่ำเกินไปย้ายขึ้น (4.3)
สลับหน้าจอแล้วต้องโหลดใหม่ทุกครั้งข้อมูลผูกกับหน้าจอย้ายออกไปชั้นที่อายุยาวกว่า (4.5)

10. แบบฝึกหัด

หน้าจอหยิบสินค้ามีข้อมูลหกชิ้น จงจัดว่าแต่ละชิ้นควรอยู่ที่ไหน พร้อมเหตุผลตามเกณฑ์สามข้อ

ข้อมูลบริบท
เปิด/ปิดคำแนะนำการใช้งานผู้ใช้กดปุ่มเครื่องหมายคำถาม
รายการงานที่ต้องหยิบมาจาก server หน้าสรุปก็ใช้
ข้อความที่พิมพ์ในช่องค้นหาใช้กรองรายการในหน้านี้
ชื่อและสิทธิ์ของผู้ใช้ที่ล็อกอินเกือบทุกหน้าจอใช้
แถวที่กำลังส่งคำสั่งอยู่ใช้ปิดปุ่มของแถวนั้น
ตำแหน่งการเลื่อนของรายการอยากให้กลับมาแล้วอยู่ที่เดิม
ดูแนวคำตอบและเหตุผล
ข้อมูลอยู่ที่ไหนเหตุผล
เปิด/ปิดคำแนะนำsetStateผ่านทั้งสามเกณฑ์ — คนเดียวใช้ ไม่ต้องอยู่รอด ไม่มีตรรกะ
รายการงานstoreหลายหน้าจอใช้ และมีตรรกะการโหลดที่ต้องทดสอบ
ข้อความค้นหาsetStateใช้กรองในหน้านี้เท่านั้น และควรล้างเมื่อออกจากหน้าจอ
ผู้ใช้ที่ล็อกอินชั้นบนสุดของแอปอายุเท่ากับ session ไม่ใช่เท่าหน้าจอ
แถวที่กำลังส่งคำสั่งstoreเป็นสถานะของคำสั่ง ไม่ใช่ของหน้าจอ — ถ้าอยู่ในหน้าจอแล้วผู้ใช้ออกไป คำสั่งจะไม่มีใครติดตาม
ตำแหน่งการเลื่อนขึ้นกับความต้องการดูคำอธิบายด้านล่าง

ข้อที่ตัดสินยากที่สุด — แถวที่กำลังส่งคำสั่ง

ดูเผิน ๆ เหมือนเป็นเรื่องของหน้าจอ เพราะใช้แค่ปิดปุ่ม แต่ถามด้วยเกณฑ์ข้อ 2 จะได้คำตอบต่างออกไป — ถ้าผู้ใช้กดปุ่มแล้วออกจากหน้าจอทันที คำสั่งนั้นยังวิ่งอยู่ ถ้าสถานะอยู่ในหน้าจอที่ตายไปแล้ว จะไม่มีใครรู้ว่ามันจบหรือยัง

ถ้าธุรกิจต้องการเก็บที่ไหน
กลับมาแล้วเริ่มที่บนสุดไม่ต้องเก็บเลย
กลับมาแล้วอยู่ที่เดิมภายในรอบเดียวกันในหน้าจอ ผ่านตัวควบคุมการเลื่อน
อยู่ที่เดิมแม้ปิดแอปแล้วเปิดใหม่บันทึกลงเครื่อง

นี่เป็นตัวอย่างสำคัญว่า คำตอบมาจากความต้องการทางธุรกิจ ไม่ใช่จากกฎทางเทคนิค — คนที่ตอบว่า “เก็บใน store เสมอ” หรือ “เก็บใน setState เสมอ” ยังไม่ได้ถามคำถามที่ถูก

11. สรุปท้ายบท

  • setState เหมาะกับข้อมูลที่เกิดและตายพร้อมหน้าจอ
  • ถามสามคำถาม — ใครใช้ ต้องอยู่รอดไหม มีตรรกะต้องทดสอบไหม
  • ข้างใน setState ต้องทำงานเสร็จทันที ห้ามมีงานที่ต้องรอ
  • ตรวจ mounted หลัง await เสมอ
  • สร้าง list ก้อนใหม่แทนการแก้ก้อนเดิม
  • ถ้าต้องส่ง callback ผ่านหลายชั้น หรือข้อมูลหายเมื่อสลับหน้าจอ แปลว่าอยู่ผิดที่

บทถัดไป: 4.2 จะลดจำนวนข้อมูลที่ต้องเก็บลง ด้วยการไม่เก็บสิ่งที่คำนวณได้ — ซึ่งเป็นวิธีกำจัดบั๊กประเภท “ตัวเลขสองที่ไม่ตรงกัน” ตั้งแต่ต้นทาง
