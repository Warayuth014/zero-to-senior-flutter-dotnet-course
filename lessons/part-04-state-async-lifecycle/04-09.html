1. ปัญหาที่บทนี้แก้

โค้ดที่ดูปกติมาก

Future<void> _submit() async {
  final receipt = await repository.confirm(draft);   // ใช้เวลา 2 วินาที
  ScaffoldMessenger.of(context).showSnackBar(         // ระเบิดตรงนี้
    SnackBar(content: Text('บันทึกแล้ว ${receipt.id}')),
  );
}

ทำงานได้ตลอดตอนทดสอบ แล้ววันหนึ่งจะได้ข้อความนี้

FlutterError: Looking up a deactivated widget's ancestor is unsafe.

สาเหตุ — ระหว่างสองวินาทีนั้น ผู้ใช้กดปุ่มย้อนกลับ หน้าจอถูกถอดออกไปแล้ว แต่ _submit() ยังทำงานต่อ และไปเรียก context ของหน้าจอที่ไม่มีอยู่แล้ว

ข้อความมันอ่านยาก ชี้ไม่ตรงจุด และเกิดเฉพาะเมื่อจังหวะพอดี — จึงมักหลุดขึ้น production ไปก่อน

2. คำศัพท์ที่ต้องรู้ก่อน

คำความหมาย
async gapช่วงเวลาที่โค้ดหยุดรอที่ await
mountedค่า bool ที่บอกว่า State นี้ยังอยู่ในต้นไม้หรือไม่
deactivated widgetwidget ที่ถูกถอดออกจากต้นไม้แล้ว
UI lifetimeอายุของหน้าจอ
business operationงานทางธุรกิจ ซึ่งมีอายุของตัวเอง ไม่ผูกกับหน้าจอ

3. Mental model: หน้าจออาจไม่รอคุณ

0.0s  ผู้ใช้กดปุ่ม → _submit() เริ่ม
0.0s  await repository.confirm(...)     ← โค้ดหยุดตรงนี้
0.8s  ผู้ใช้กดย้อนกลับ → หน้าจอถูกถอด, dispose() ถูกเรียก
2.0s  server ตอบกลับ → โค้ดทำงานต่อจากบรรทัดถัดไป
2.0s  ใช้ context ของหน้าจอที่ไม่มีอยู่แล้ว → พัง
ประโยคเดียวที่ต้องจำ: ทุกบรรทัดที่อยู่หลัง await ต้องถือว่าหน้าจออาจหายไปแล้ว — ไม่ใช่บางครั้ง แต่ทุกครั้ง

5. context.mounted: เมื่อ context ไม่ใช่ของ State

บางที context ที่จะใช้ไม่ได้มาจาก State แต่มาจาก builder

showDialog(
  context: context,
  builder: (dialogContext) => AlertDialog(     // context คนละตัว
    actions: [
      TextButton(
        onPressed: () async {
          await store.markDone(lineId);
          if (!dialogContext.mounted) return;   // ตรวจตัวที่จะใช้จริง
          Navigator.of(dialogContext).pop();
        },
        child: const Text('ยืนยัน'),
      ),
    ],
  ),
)

จุดที่พลาดกันคือ ตรวจ mounted ของหน้าจอ แต่ไปใช้ dialogContext ซึ่งเป็นคนละอายุกัน — กล่องโต้ตอบอาจถูกปิดไปแล้วในขณะที่หน้าจอเบื้องหลังยังอยู่ครบ

context ที่ใช้ต้องตรวจ
context ของ Statemounted
context ที่รับมาจาก builderthatContext.mounted
กฎง่าย ๆ — ตรวจตัวที่จะใช้ ไม่ใช่ตัวที่อยู่ใกล้มือ

6. analyzer หาให้ได้ก่อนที่ผู้ใช้จะเจอ

Flutter มี lint ชื่อ use_build_context_synchronously ที่เตือนเมื่อมีการใช้ context ข้าม async gap

flutter analyze --no-pub
วิธีแก้เมื่อ analyzer เตือนเหมาะเมื่อ
ย้ายการอ่านค่ามาก่อน awaitแค่ต้องการ Theme.of หรือค่าอื่นที่อ่านได้ล่วงหน้า
ใส่ if (!mounted) return;ต้องใช้ context หลังงานเสร็จจริง ๆ
ย้ายงาน async ไปไว้ใน storeดีที่สุด ถ้าทำได้ (ดูหัวข้อ 8)
// ย้ายมาก่อน await — ไม่ต้องตรวจอะไรเลย
final messenger = ScaffoldMessenger.of(context);
final receipt = await repository.confirm(draft);
messenger.showSnackBar(SnackBar(content: Text('บันทึกแล้ว ${receipt.id}')));

วิธีนี้ใช้ได้เพราะ ScaffoldMessengerState ที่หยิบไว้ยังใช้งานได้แม้หน้าจอที่หยิบมันมาจะถูกปิดไปแล้ว — แต่ต้องรู้ตัวว่ามันจะแสดง SnackBar ขึ้นมาบนหน้าจอใหม่ที่ผู้ใช้กำลังดูอยู่ ซึ่งอาจไม่ใช่สิ่งที่ต้องการ

อย่าปิด lint ตัวนี้ทั้งโปรเจกต์เพราะรำคาญ — มันหาบั๊กที่หายากที่สุดชนิดหนึ่งได้ฟรี ๆ ถ้าจุดไหนตรวจสอบแล้วว่าปลอดภัยจริง ให้ปิดเฉพาะบรรทัดนั้นพร้อมคำอธิบาย

7. mounted ไม่ได้แก้ปัญหาที่คนคิดว่ามันแก้

นี่คือหัวใจของบทนี้ และเป็นสิ่งที่ทำให้บทนี้ต่อกับ 4.10

สิ่งที่ mounted ตอบได้สิ่งที่ตอบไม่ได้
“หน้าจอนี้ยังอยู่ไหม”“ผลลัพธ์นี้เป็นของคำขอล่าสุดหรือเปล่า”
“ใช้ context ได้ไหม”“ข้อมูลที่ฉันกำลังจะเขียนทับ ใหม่กว่าของฉันหรือเปล่า”
// mounted เป็น true ตลอด แต่ยังผิด
await load('PAL-0001');   // ช้า
await load('PAL-0002');   // เร็ว มาถึงก่อน
// ผลของ PAL-0001 มาทีหลัง แล้วทับ PAL-0002
// หน้าจอยัง mounted อยู่ตลอด — mounted จึงช่วยอะไรไม่ได้เลย

และในทางกลับกัน mounted == false ก็ไม่ได้แปลว่าควรทิ้งงานเสมอไป

await repository.markDone(lineId);   // server บันทึกแล้ว
if (!mounted) return;                 // ← ถ้า return ตรงนี้ ใครอัปเดตข้อมูล?

ถ้าผู้ใช้กดย้อนกลับหลังกดปิดงาน งานนั้น ก็ยังถูกปิดจริงที่ server การ return เฉย ๆ ทำให้ข้อมูลในแอปไม่ตรงกับความจริง — พอเข้าหน้าเดิมอีกครั้งจะยังเห็นว่ายังไม่เสร็จ

แยกให้ออก: อายุของหน้าจอ กับ อายุของงานทางธุรกิจ เป็นคนละเรื่อง mounted คุมได้แค่เรื่องแรก

8. ทางออกที่แท้จริง: อย่าให้ widget ถืองาน async

เปิด lib/state_basics/state_basics_app.dart แล้วมองหา mounted — จะไม่เจอเลยแม้แต่ที่เดียว

ไม่ใช่เพราะลืม แต่เพราะไม่มีจุดไหนที่ widget รอ await แล้วใช้ context ต่อ

ElevatedButton(
  onPressed: store.isBusy(line.id)
      ? null
      : () => store.markDone(line.id),   // สั่งแล้วจบ ไม่รอผล
  child: const Text('ปิดงาน'),
)

ผลลัพธ์ไม่ได้กลับมาที่ปุ่ม แต่ไปอยู่ใน store แล้ว store แจ้ง แล้ว ListenableBuilder วาดใหม่

ปุ่ม → store.markDone()
                │
                ├→ await repository ...
                └→ notifyListeners()
                        │
                        └→ ListenableBuilder วาดใหม่ (ถ้ายังอยู่)

widget ถืองาน asyncstore ถืองาน async
ต้องตรวจ mountedทุกจุดหลัง awaitไม่ต้อง
หน้าจอปิดกลางทางต้องเขียนโค้ดรับมืองานเดินต่อจนจบ ข้อมูลถูกต้อง
ทดสอบต้องมี widgetทดสอบ store ตรง ๆ ได้

นี่เป็นเหตุผลเชิงปฏิบัติอีกข้อของสิ่งที่ทำมาตั้งแต่ 4.5 — การย้ายสถานะออกจาก widget ไม่ได้ทำให้แค่ทดสอบง่ายขึ้น แต่ทำให้ปัญหาทั้งหมวดนี้หายไปเลย

9. store ไม่มี mounted แล้วใช้อะไรแทน

ChangeNotifier ไม่มี mounted เพราะมันไม่ได้อยู่ในต้นไม้ widget สิ่งที่มันมีคือ dispose()

@override
void dispose() {
  stopPolling();
  _generation++;      // ทำให้ผลของคำขอที่ยังค้างอยู่ ถูกทิ้งเมื่อกลับมา
  super.dispose();
}

บรรทัด _generation++ คือกลไกเดียวกับที่ใช้กันผลลัพธ์เก่าใน 4.10 — พอเลขรุ่นเปลี่ยน คำขอที่ยิงไปก่อนหน้าจะตรวจเจอว่าตัวเองล้าสมัยแล้วจึงไม่เขียนอะไรทับ

สิ่งที่ store ต้องทำใน dispose()

ต้องทำเพราะ
ยกเลิก Timerไม่งั้นมันเดินต่อตลอดอายุแอป (4.11)
ยกเลิกการ subscribe streamเหตุผลเดียวกัน
ทำให้ผลที่ค้างอยู่ถูกทิ้งไม่ให้เขียนทับสถานะของ store ที่ตายแล้ว

และสิ่งที่ ห้าม ทำคือข้ามงานเก็บกวาดใน finally เพราะคิดว่าไม่มีใครฟังแล้ว — เหตุผลอยู่ใน 4.8 หัวข้อ 8

10. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
Looking up a deactivated widget's ancestorใช้ context หลัง await โดยไม่ตรวจif (!mounted) return;
setState() called after dispose()เรียก setState หลังหน้าจอปิดเหมือนข้างบน
ตรวจ mounted แล้วยังพังตรวจของ State แต่ใช้ context ของ dialogตรวจ dialogContext.mounted
ปิดงานแล้วแต่กลับเข้ามายังเห็นว่าไม่เสร็จreturn เมื่อ unmounted โดยไม่อัปเดตข้อมูลย้ายงานไปไว้ใน store

11. แบบฝึกหัด

class _ConfirmScreenState extends State<ConfirmScreen> {
  bool _submitting = false;
  final _focusNode = FocusNode();

  Future<void> _submit() async {
    setState(() => _submitting = true);

    final receipt = await widget.repository.confirm(widget.draft);

    setState(() => _submitting = false);
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(content: Text('บันทึกแล้ว ${receipt.id}')),
    );
    _focusNode.requestFocus();
    Navigator.of(context).pop(receipt);
  }
}

ข้อ 2 — แก้ด้วย mounted

Future<void> _submit() async {
  if (_submitting) return;                  // กันกดซ้ำ (4.8)
  setState(() => _submitting = true);

  Receipt? receipt;
  try {
    receipt = await widget.repository.confirm(widget.draft);
  } on ReceiptException catch (error) {
    if (!mounted) return;
    setState(() => _submitting = false);
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(content: Text(error.message)),
    );
    return;
  }

  if (!mounted) return;                     // ด่านเดียวคุมทุกบรรทัดข้างล่าง
  setState(() => _submitting = false);
  ScaffoldMessenger.of(context).showSnackBar(
    SnackBar(content: Text('บันทึกแล้ว ${receipt.id}')),
  );
  Navigator.of(context).pop(receipt);
}

สังเกตว่า _focusNode.requestFocus() ถูกตัดออก — ถ้ากำลังจะ pop อยู่แล้ว การขอโฟกัสไม่มีความหมาย และเป็นตัวอย่างของโค้ดที่เขียนไว้โดยไม่ได้ตั้งใจจริง

ข้อ 3 — ปัญหาที่ยังเหลือ

ข้อ 2 กันไม่ให้แอปพังได้จริง แต่ ไม่ได้แก้ปัญหาจริง — มันแค่เงียบ

ลองไล่ตามลำดับเวลา

0.0s  กดยืนยัน → ยิงคำขอไป server
0.8s  ผู้ใช้กดย้อนกลับ → หน้าจอปิด
2.0s  server ตอบว่า "บันทึกสำเร็จ ใบรับ RC-771"
2.0s  if (!mounted) return;   ← ทิ้งใบรับไปเฉย ๆ
ผลที่ตามมาความเสียหาย
ผู้ใช้ไม่รู้ว่าบันทึกสำเร็จอาจยืนยันซ้ำอีกรอบ ได้ใบรับสองใบ
หน้ารายการยังแสดงว่ายังไม่ยืนยันข้อมูลในแอปไม่ตรงกับ server

class ReceiptStore extends ChangeNotifier {
  Future<void> confirm(ReceiptDraft draft) async {
    if (_submitting) return;
    _submitting = true;
    _errorMessage = null;
    notifyListeners();

    try {
      _lastReceipt = await repository.confirm(draft);   // ผลอยู่ใน store
      _confirmedDraftIds.add(draft.id);                 // ข้อมูลถูกต้องเสมอ
    } on ReceiptException catch (error) {
      _errorMessage = error.message;
    } finally {
      _submitting = false;
      notifyListeners();      // ถ้าไม่มีใครฟังก็ไม่เป็นไร ไม่พัง
    }
  }
}

  child: const Text('ยืนยัน'),

ส่วน Navigator.pop ที่หายไป — ให้หน้าจอฟัง store แล้วปิดตัวเองเมื่อยืนยันสำเร็จ ซึ่งเป็นผลข้างเคียงที่ต้องทำนอก build ตามรูปแบบใน 4.6 หัวข้อ 8 และจะทำเต็มรูปแบบใน Part 5

บทเรียนที่ควรจำ: mounted เป็นเครื่องมือกันแอปพัง ไม่ใช่เครื่องมือออกแบบ ถ้าพบว่าต้องโรย if (!mounted) return; เต็มไฟล์ นั่นคือสัญญาณว่างาน async อยู่ผิดที่

12. สรุปท้ายบท

  • ทุกบรรทัดหลัง await ต้องถือว่าหน้าจออาจหายไปแล้ว
  • mounted ต้องตรวจก่อน setState, Navigator, ScaffoldMessenger, Theme และ FocusNode
  • ตรวจ mounted ของ context ตัวที่จะใช้จริง ไม่ใช่ตัวที่อยู่ใกล้มือ
  • use_build_context_synchronously หาจุดเหล่านี้ให้ฟรี อย่าปิดทั้งโปรเจกต์
  • mounted ตอบได้แค่ว่าหน้าจอยังอยู่ไหม ไม่ได้ตอบว่าผลลัพธ์นี้ใหม่ที่สุดหรือเปล่า
  • mounted == false ไม่ได้แปลว่าควรทิ้งงานทางธุรกิจ
  • ทางแก้ที่ดีที่สุดคือย้ายงาน async ไปไว้ใน store แล้วปัญหาทั้งหมวดนี้หายไป
  • store ไม่มี mounted — ใช้ dispose() ยกเลิกตัวจับเวลาและทำให้ผลที่ค้างอยู่ถูกทิ้ง

บทถัดไป: 4.10 จะแก้ปัญหาที่ mounted ตอบไม่ได้ — ผลของคำขอเก่าที่มาถึงช้ากว่าคำขอใหม่ แล้วเขียนทับข้อมูลล่าสุด
