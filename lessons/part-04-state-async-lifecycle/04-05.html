1. ปัญหาที่บทนี้แก้

4.4 จบด้วยข้อจำกัดที่ชัดเจน — หน้าจอที่โหลดข้อมูลจาก server มีอย่างน้อยสามสิ่งที่เปลี่ยนพร้อมกัน และ ValueNotifier เก็บได้ค่าเดียว

ChangeNotifier แก้ปัญหานี้ด้วยการให้เราเก็บอะไรก็ได้ แล้วบอกเองว่าเมื่อไหร่ควรแจ้ง — ซึ่งทั้งเป็นข้อดีและข้อควรระวัง

2. คำศัพท์ที่ต้องรู้ก่อน

คำความหมาย
ChangeNotifierคลาสที่มีความสามารถ “แจ้งคนที่ฟังอยู่”
notifyListeners()คำสั่งแจ้ง เราเป็นคนเรียกเอง
storeคำเรียกทั่วไปของที่เก็บสถานะพร้อมตรรกะ
commandmethod ที่เปลี่ยนสถานะ เช่น load(), markDone()
repositoryชั้นที่รู้วิธีดึงข้อมูล — store ไม่ต้องรู้ว่ามาจากไหน

3. Mental model: กล่องที่เราคุมเองทุกอย่าง

ValueNotifierChangeNotifier
เก็บได้ค่าเดียวกี่ค่าก็ได้
ใครแจ้งมันแจ้งให้เองเมื่อค่าเปลี่ยนเราต้องเรียกเอง
มีที่วางตรรกะไม่ค่อยมีมี — เป็น class เต็ม
ความเสี่ยงลืมสร้างก้อนใหม่ลืมเรียก notifyListeners
ความเสี่ยงหลักของ ChangeNotifier: ถ้าเปลี่ยนค่าแล้วลืมเรียก notifyListeners() หน้าจอจะไม่อัปเดต และ ไม่มี error อะไรเตือนเลย — ทางป้องกันคือทำให้ทุกการเปลี่ยนแปลงต้องผ่าน method ที่เราควบคุมได้ ตามหลัก D1.7

4. ดูของจริงใน lab

class PickStore extends ChangeNotifier {
  PickStore({required this.repository, String Function()? createCommandId})
    : _createCommandId = createCommandId ?? (() => ...);

  final PickRepository repository;          // รับสัญญา ไม่ใช่ตัวจริง

  List<PickLine> _lines = const [];         // ทุก field เป็น private
  LoadState _state = LoadState.idle;
  String? _errorMessage;

  List<PickLine> get lines => List.unmodifiable(_lines);   // อ่านได้ แก้ไม่ได้
  LoadState get state => _state;
  String? get errorMessage => _errorMessage;
}

สามอย่างที่ทำให้ store นี้แข็งแรง

สิ่งที่ทำป้องกันอะไรมาจากบท

คืน List.unmodifiableคนนอกแก้ list ข้ามกฎD1.7, 4.3
รับ PickRepository เข้ามาผูกกับ server จนทดสอบไม่ได้D1.8

5. รูปแบบของคำสั่งที่โหลดข้อมูล

Future<void> load() async {
  final generation = ++_generation;
  _state = LoadState.loading;
  _errorMessage = null;
  notifyListeners();                    // แจ้งครั้งที่ 1 — เริ่มโหลด

  try {
    final result = await repository.fetchLines();
    if (generation != _generation) return;

    _lines = result;
    _state = result.isEmpty ? LoadState.empty : LoadState.ready;
  } on PickException catch (error) {
    if (generation != _generation) return;
    _state = LoadState.failed;
    _errorMessage = error.message;
  } finally {
    if (generation == _generation) notifyListeners();   // แจ้งครั้งที่ 2 — จบ
  }
}

โครงนี้จะเจอซ้ำในทุก store ที่เขียนต่อจากนี้

  1. ตั้งสถานะกำลังโหลด แล้วแจ้ง — ผู้ใช้เห็นตัวหมุนทันที
  2. ทำงานที่ต้องรอ
  3. ตั้งผลลัพธ์ ไม่ว่าจะสำเร็จหรือล้มเหลว
  4. แจ้งอีกครั้งใน finally — จึงแจ้งเสมอไม่ว่าเกิดอะไร

ส่วนเรื่อง generation คือการกันผลลัพธ์เก่าจาก 3.13 ซึ่งจะอธิบายเต็มอีกครั้งใน 4.10

6. เปลี่ยนให้ครบก่อนแจ้ง

// ผิด — แจ้งกลางทาง หน้าจอเห็นสถานะที่ไม่สอดคล้อง
_lines = newLines;
notifyListeners();              // ตรงนี้ state ยังเป็น loading อยู่
_state = LoadState.ready;
notifyListeners();

// ถูก — เปลี่ยนให้ครบแล้วแจ้งครั้งเดียว
_lines = newLines;
_state = LoadState.ready;
notifyListeners();

เหตุผลตรงกับที่เตือนใน 1.13 หัวข้อ 5 — การแจ้งกลางทางทำให้หน้าจอวาดในจังหวะที่ข้อมูลยังไม่สอดคล้องกัน ผู้ใช้จะเห็นภาพแวบหนึ่งที่ผิด เช่นรายการขึ้นมาแล้วแต่ยังมีตัวหมุนอยู่

ข้อยกเว้นที่ตั้งใจคือใน lab ตอนที่ผลไม่แน่ชัด

_errorMessage = '${error.message} กำลังตรวจสอบสถานะล่าสุด';
notifyListeners();              // แจ้งก่อน เพื่อให้ผู้ใช้เห็นว่ากำลังตรวจสอบ
await load();                   // แล้วค่อยไปโหลดใหม่

ตรงนี้แจ้งกลางทางโดยตั้งใจ เพราะขั้นตอนตรวจสอบใช้เวลา และผู้ใช้ควรรู้ว่าระบบกำลังทำอะไรอยู่

8. เก็บกวาดใน dispose

@override
void dispose() {
  stopPolling();
  _generation++;        // ทำให้ผลที่ยังค้างอยู่ถูกทิ้งทั้งหมด
  super.dispose();
}

ChangeNotifier มี dispose มาให้แล้ว เราเพิ่มการเก็บกวาดของเราก่อนเรียกของแม่ ตามลำดับที่เรียนใน 3.12

บรรทัด _generation++ เป็นเทคนิคที่ใช้แทนการตรวจ mounted — เพราะ store ไม่มี mounted การเพิ่มรุ่นทำให้งานที่ยังค้างอยู่ทั้งหมดถูกทิ้งเมื่อผลกลับมา

ถ้าเรียก notifyListeners() หลัง dispose จะได้ error ทันที ซึ่งดีเพราะบอกว่ามีงานค้างที่ยังไม่ถูกจัดการ — อย่าแก้ด้วยการดักเงียบ ๆ ให้ไปหาว่างานไหนยังไม่หยุด

9. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
เปลี่ยนข้อมูลแล้วหน้าจอไม่อัปเดตลืม notifyListeners()ทำ field เป็น private แล้วเปลี่ยนผ่าน method เท่านั้น
คนนอกแก้ list ได้โดยตรงคืน list ตัวจริงออกไปคืน List.unmodifiable
หน้าจอกระพริบเห็นสถานะกลางแจ้งหลายครั้งต่อการกระทำเดียวเปลี่ยนให้ครบก่อนแล้วแจ้งครั้งเดียว
ธงกำลังโหลดค้างเมื่อเกิดข้อผิดพลาดตั้งค่ากลับไว้ท้าย tryย้ายไป finally
ทดสอบ store ไม่ได้ ต้องมี serverstore สร้าง client เองรับ repository เข้ามาผ่าน constructor
A ChangeNotifier was used after being disposedงานที่ค้างอยู่แจ้งหลัง store ตายหยุดงานทั้งหมดใน dispose
store ตัวเดียวเก็บทุกอย่างของทั้งแอปไม่ได้แบ่งตามส่วนงานแยก store ตามส่วนงาน

class _CountScreenState extends State<CountScreen> {
  List<CountLine> _lines = [];

  Future<void> _load() async {
    setState(() => _loading = true);

ข้อ 1 — ปัญหาของโค้ดเดิม

ปัญหาผลที่เกิด
ไม่ตรวจ mountedพังเมื่อออกจากหน้าจอระหว่างโหลด (3.13)
ไม่แยกสถานะว่างออกจากสำเร็จไม่มีข้อมูลกับล้มเหลวดูเหมือนกัน (4.2)
ใช้ e.toString() เป็นข้อความผู้ใช้เห็นข้อความทางเทคนิค (D1.14)
ผูกกับ api โดยตรงทดสอบต้องมี server
เรียก setState สามครั้งวาดใหม่สามรอบต่อการโหลดหนึ่งครั้ง
// สัญญา — ไม่รู้ว่าข้อมูลมาจากไหน
abstract interface class CountRepository {
  Future<List<CountLine>> fetchLines();
}

class CountException implements Exception {
  const CountException(this.message);
  final String message;
}

enum CountLoadState { idle, loading, ready, empty, failed }

class CountStore extends ChangeNotifier {
  CountStore({required this.repository});

  final CountRepository repository;

  List<CountLine> _lines = const [];
  CountLoadState _state = CountLoadState.idle;
  String? _errorMessage;

  List<CountLine> get lines => List.unmodifiable(_lines);
  CountLoadState get state => _state;
  String? get errorMessage => _errorMessage;

  int get totalQuantity =>
      _lines.fold(0, (sum, line) => sum + line.quantity);   // คำนวณ (4.2)

  Future<void> load() async {
    if (_state == CountLoadState.loading) return;   // กันเรียกซ้อน

    _state = CountLoadState.loading;
    _errorMessage = null;
    notifyListeners();

    try {
      final result = await repository.fetchLines();
      _lines = result;
      _state = result.isEmpty
          ? CountLoadState.empty
          : CountLoadState.ready;
    } on CountException catch (error) {
      _state = CountLoadState.failed;
      _errorMessage = error.message;        // ข้อความที่เตรียมไว้ให้ผู้ใช้
    } finally {
      notifyListeners();                     // แจ้งเสมอ
    }
  }
}
class FakeCountRepository implements CountRepository {
  FakeCountRepository({this.lines = const []});

  List<CountLine> lines;
  CountException? error;
  Completer<List<CountLine>>? gate;
  int fetchCount = 0;

  @override
  Future<List<CountLine>> fetchLines() async {
    fetchCount++;
    final pending = gate;
    if (pending != null) return pending.future;
    if (error != null) throw error!;
    return lines;
  }
}

void main() {
  test('ไม่มีข้อมูลได้สถานะ empty ไม่ใช่ ready', () async {
    final store = CountStore(repository: FakeCountRepository());
    await store.load();
    expect(store.state, CountLoadState.empty);
  });

  test('ล้มเหลวแล้วได้ข้อความที่เตรียมไว้ ไม่ใช่ข้อความทางเทคนิค', () async {
    final repo = FakeCountRepository()
      ..error = const CountException('เชื่อมต่อไม่ได้');
    final store = CountStore(repository: repo);

    await store.load();

    expect(store.state, CountLoadState.failed);
    expect(store.errorMessage, 'เชื่อมต่อไม่ได้');
  });

  test('เรียกโหลดซ้อนระหว่างที่ยังไม่เสร็จ ไม่ยิงซ้ำ', () async {
    final repo = FakeCountRepository();
    final gate = Completer<List<CountLine>>();
    repo.gate = gate;
    final store = CountStore(repository: repo);

    final first = store.load();
    await store.load();
    await store.load();

    gate.complete(const []);
    await first;

    expect(repo.fetchCount, 1);
  });
}

ทำไมสามกรณีนี้ทดสอบไม่ได้ด้วยโค้ดเดิม

กรณีทำไมเดิมทำไม่ได้
สถานะว่างโค้ดเดิมไม่มีสถานะนี้เลย
ข้อความผิดพลาดต้องทำให้ server ล้มเหลวจริง ซึ่งควบคุมไม่ได้
เรียกซ้อนต้องควบคุมเวลาให้คำขอค้าง ซึ่งทำกับ server จริงไม่ได้

จุดที่ควรสังเกตที่สุด — เทสต์ทั้งสามไม่มีคำว่า widget, tester หรือ pumpWidget เลยแม้แต่คำเดียว ซึ่งหมายความว่าตรรกะการโหลดแยกออกจากหน้าจอได้จริง ตามหลักที่ใช้มาตั้งแต่ D1.20

ลองต่อ: เพิ่ม getter totalQuantity แล้วเขียนเทสต์ — จะเห็นว่าไม่ต้องแก้ load() เลย เพราะเป็นค่าที่คำนวณได้ตามหลัก 4.2

11. สรุปท้ายบท

  • ChangeNotifier เก็บได้หลายค่าและมีที่วางตรรกะ
  • เราเป็นคนเรียก notifyListeners() เอง — ลืมแล้วไม่มีอะไรเตือน
  • ทำ field เป็น private แล้วเปลี่ยนผ่าน method เท่านั้น
  • คืนข้อมูลออกไปแบบแก้ไม่ได้
  • เปลี่ยนให้ครบก่อนแล้วแจ้งครั้งเดียว
  • รับ repository เข้ามาผ่าน constructor เพื่อให้ทดสอบได้
  • แจ้งใน finally เพื่อให้แจ้งเสมอไม่ว่าเกิดอะไร
  • หยุดงานทั้งหมดใน dispose

บทถัดไป: 4.6 จะแก้ผลข้างเคียงของ store — เมื่อแจ้งหนึ่งครั้ง ทุกคนที่ฟังอยู่จะวาดใหม่หมด ซึ่งอาจมากเกินจำเป็น
