1. ปัญหาที่บทนี้แก้

นี่คือโค้ดที่เกือบทุกคนเขียนในครั้งแรก

bool _loading = false;
List<PickLine> _lines = [];

// ในหน้าจอ
if (_loading) return const CircularProgressIndicator();
return ListView(children: _lines.map(...).toList());

มันทำงานได้จนกว่าจะเจอสถานการณ์จริงข้อใดข้อหนึ่ง

สถานการณ์สิ่งที่ผู้ใช้เห็นสิ่งที่ควรเห็น
server ตอบสำเร็จ แต่ไม่มีงานเลยจอว่างเปล่า ไม่มีอะไรบอก“วันนี้ยังไม่มีงาน”
เน็ตหลุด โหลดไม่สำเร็จจอว่างเปล่าเหมือนกันข้อความผิดพลาด + ปุ่มลองใหม่
ยังไม่เริ่มโหลดเลยจอว่างเปล่าอีกอะไรก็ได้ที่ไม่ทำให้เข้าใจผิด

สามสถานการณ์ที่ต่างกันสิ้นเชิง แต่ผู้ใช้เห็นเหมือนกันหมด เพราะ เราเก็บได้แค่สองสถานะจาก boolean ตัวเดียว

2. คำศัพท์ที่ต้องรู้ก่อน

คำความหมาย
loading stateสถานะระหว่างรอผลลัพธ์
empty stateโหลดสำเร็จ แต่ไม่มีข้อมูล — เป็นความสำเร็จ ไม่ใช่ความผิดพลาด
error stateโหลดไม่สำเร็จ
initial loadการโหลดครั้งแรก ยังไม่มีข้อมูลเดิมให้แสดง
refreshการโหลดซ้ำ มีข้อมูลเดิมอยู่แล้ว
stale dataข้อมูลเก่าที่ยังแสดงอยู่ ทั้งที่อาจไม่ตรงกับปัจจุบัน
exhaustiveครอบคลุมทุกกรณี ไม่มีทางตกหล่น

3. Mental model: สถานะคือคำถามว่า “ตอนนี้ควรวาดอะไร”

อย่าคิดว่าสถานะคือ “ตัวแปรที่ต้องเก็บ” ให้คิดว่าเป็น คำตอบเดียวที่ตัดสินว่าจอนี้ควรแสดงอะไร

สถานะเดียว  →  หน้าตาเดียว

idle     →  ยังไม่เริ่ม
loading  →  ตัวหมุน
ready    →  รายการ
empty    →  “ยังไม่มีงาน”
failed   →  ข้อความผิดพลาด + ปุ่มลองใหม่
ประโยคเดียวที่ต้องจำ: ถ้าตอบไม่ได้ว่าตอนนี้อยู่สถานะไหนโดยดูค่าเดียว แสดงว่าสถานะยังออกแบบไม่เสร็จ

4. ทำไม boolean หลายตัวถึงไม่ใช่คำตอบ

ทางแก้แรกที่คนคิดถึงคือเพิ่ม boolean

bool _loading = false;
bool _failed = false;
bool _empty = false;

ลองนับดูว่า boolean สามตัวเก็บได้กี่แบบ — 2 × 2 × 2 = 8 แบบ แต่แบบที่มีความหมายจริงมีแค่ 4

loadingfailedemptyหมายความว่าอะไร
truefalsefalseกำลังโหลด ✓
falsetruefalseล้มเหลว ✓
falsefalsetrueไม่มีข้อมูล ✓
falsefalsefalseมีข้อมูล ✓
truetruefalseกำลังโหลดและล้มเหลวพร้อมกัน? ✗
truefalsetrueกำลังโหลดแต่ไม่มีข้อมูล? ✗
falsetruetrueล้มเหลวและว่างพร้อมกัน? ✗
truetruetrueทุกอย่างพร้อมกัน? ✗

สี่แบบล่างคือ สถานะที่ไม่ควรมีอยู่ได้เลย แต่ชนิดข้อมูลอนุญาตให้เกิดขึ้น และเมื่อเกิดขึ้นจริง (เช่นลืมเซ็ต _failed = false ก่อนโหลดใหม่) หน้าจอจะแสดงอะไรก็ไม่รู้

นี่คือหลักการเดียวกับ 4.2 — ถ้าค่าที่เป็นไปไม่ได้สร้างไม่ได้ตั้งแต่แรก ก็ไม่ต้องเขียนโค้ดกันมัน

5. enum: หนึ่งค่า หนึ่งความหมาย

เปิด lib/state_basics/pick_store.dart

enum LoadState { idle, loading, ready, empty, failed }

ค่าเดียว ห้าความหมาย และ ไม่มีทางเป็นสองอย่างพร้อมกัน เพราะภาษาบังคับไว้แล้ว

สถานะเกิดเมื่อไหร่
idleสร้าง store แล้วแต่ยังไม่เรียก load()
loadingเริ่ม load() แล้ว ยังไม่ได้ผล
readyได้ผลแล้ว และมีอย่างน้อยหนึ่งรายการ
emptyได้ผลแล้ว แต่รายการว่าง
failedเกิดข้อผิดพลาด

จุดสำคัญคือบรรทัดนี้ใน load()

_state = result.isEmpty ? LoadState.empty : LoadState.ready;

ทั้ง empty และ ready อยู่ในเส้นทางความสำเร็จทั้งคู่ — แยกกันแค่เพื่อให้หน้าจอพูดกับผู้ใช้ต่างกัน ไม่ใช่เพราะอันหนึ่งผิดพลาด

6. switch ที่ครบทุกทาง

ใน state_basics_app.dart

body: ListenableBuilder(
  listenable: widget.store,
  builder: (context, child) => switch (widget.store.state) {
    LoadState.idle || LoadState.loading => const Center(
      key: Key('loading-view'),
      child: CircularProgressIndicator(),
    ),
    LoadState.empty => const Center(
      key: Key('empty-view'),
      child: Text('วันนี้ยังไม่มีงานหยิบสินค้า'),
    ),
    LoadState.failed => _FailedView(store: widget.store),
    LoadState.ready => _LineListView(store: widget.store),
  },
),

สิ่งที่ควรสังเกต

สิ่งที่เห็นทำไมสำคัญ
ไม่มี default:ทำให้ compiler เตือนเมื่อเพิ่มค่าใหม่ใน enum แล้วลืมจัดการ
idle || loadingสองสถานะแสดงเหมือนกันได้ แต่ยังแยกเก็บไว้
ทุกทางคืน widgetไม่มีทางที่จอจะว่างเปล่าโดยไม่ตั้งใจ
ลองเอง: เพิ่ม offline เข้าไปใน LoadState แล้วรัน flutter analyze — จะได้ error ทันทีว่า switch ไม่ครบ นี่คือประโยชน์ที่ได้จากการไม่ใส่ default:

7. เมื่อสถานะต้องพกข้อมูลติดตัวมาด้วย

enum เก็บได้แค่ “อยู่สถานะไหน” แต่บางสถานะมีข้อมูลของตัวเอง

สถานะข้อมูลที่ควรมากับมัน
readyรายการงาน
failedข้อความผิดพลาด และลองใหม่ได้หรือไม่

ใน lab เราแก้ด้วยการเก็บแยกไว้ข้าง ๆ

LoadState _state = LoadState.idle;
List<PickLine> _lines = const [];
String? _errorMessage;

ใช้ได้ดีสำหรับหน้าจอขนาดนี้ แต่มีจุดอ่อนที่ต้องรู้ตัว — _errorMessage เป็น String? ซึ่งแปลว่าตอนอยู่ failed มันก็ยัง อาจจะ เป็น null ได้ในสายตาของภาษา ต้องอาศัยวินัยของเราเองที่จะเซ็ตให้ครบ

ทางแก้ที่ปิดช่องนั้นได้จริงคือ sealed class ที่เคยใช้ใน 3.14

sealed class PickViewState { const PickViewState(); }

class PickLoading extends PickViewState { const PickLoading(); }
class PickEmpty extends PickViewState { const PickEmpty(); }

class PickReady extends PickViewState {
  const PickReady(this.lines);
  final List<PickLine> lines;      // มีแน่นอน ไม่ต้องเช็ก null
}

class PickFailed extends PickViewState {
  const PickFailed(this.message, {this.canRetry = true});
  final String message;             // มีแน่นอน
  final bool canRetry;
}
return switch (state) {
  PickLoading() => const LoadingView(),
  PickEmpty() => const EmptyView(),
  PickReady(:final lines) => LineListView(lines: lines),
  PickFailed(:final message, :final canRetry) => FailedView(message, canRetry),
};
enum + field แยกsealed class
อ่านง่ายง่ายกว่าต้องเขียนคลาสหลายตัว
ข้อมูลผูกกับสถานะไม่ผูก ต้องระวังเองผูกแน่น ภาษาบังคับ
เหมาะกับสถานะเรียบง่าย ข้อมูลน้อยสถานะที่แต่ละแบบพกข้อมูลต่างกัน

เราจะกลับมาเรื่องนี้เต็ม ๆ ใน 4.15

8. โหลดครั้งแรก ไม่เหมือนโหลดซ้ำ

ปัญหาที่จะเจอทันทีที่เพิ่มการโหลดซ้ำ (4.11)

Future<void> load() async {
  _state = LoadState.loading;    // ← ทำให้รายการที่มีอยู่หายไปทั้งจอ
  notifyListeners();
  ...
}

ถ้าโหลดซ้ำทุก 10 วินาที ผู้ใช้จะเห็นรายการหายไปแล้วกลับมาทุก 10 วินาที — และถ้ากำลังเลื่อนอยู่ ตำแหน่งจะเด้งกลับด้วย

โหลดครั้งแรกโหลดซ้ำ
มีข้อมูลเดิมไหมไม่มีมี
ควรแสดงตัวหมุนเต็มจอรายการเดิม + ตัวบอกเล็ก ๆ
ถ้าล้มเหลวหน้าผิดพลาดเต็มจอรายการเดิม + แถบเตือน
// แนวทาง: เข้าสู่ loading เฉพาะเมื่อยังไม่มีอะไรให้แสดง
if (_lines.isEmpty) {
  _state = LoadState.loading;
  notifyListeners();
}

หลักคิดคือ อย่าเอาของที่ใช้งานได้อยู่ออกจากจอ เพียงเพราะกำลังจะไปหาของใหม่

9. ล้มเหลวตอนโหลดซ้ำ: ล้างรายการหรือไม่

คนหน้างานยืนอยู่หน้าแร็ค มีรายการหยิบ 12 บรรทัดบนจอ แล้วเน็ตหลุด

ทางเลือกผลที่ตามมา
ล้างรายการ แสดงหน้าผิดพลาดทำงานต่อไม่ได้เลย ทั้งที่ข้อมูลเมื่อ 10 วินาทีก่อนยังใช้ได้
เก็บรายการเดิม + แถบเตือนทำงานต่อได้ และรู้ว่าข้อมูลอาจไม่สด

โดยทั่วไปเลือกทางที่สอง แต่ต้องแยกให้ออกว่าอะไร “อ่าน” และอะไร “เขียน”

การกระทำตอนออฟไลน์
ดูรายการ เลื่อนดูทำได้ ข้อมูลเก่ายังมีประโยชน์
กดปิดงานต้องบอกให้ชัดว่าส่งไม่ได้ ไม่ใช่ทำเป็นว่าสำเร็จ
ข้อมูลเก่าที่ไม่บอกว่าเก่า อันตรายกว่าไม่มีข้อมูล — ถ้าเก็บ stale data ไว้ ต้องแสดงให้เห็นเสมอว่าอัปเดตล่าสุดเมื่อไหร่

10. หลายแหล่ง: อย่าให้ของไม่สำคัญล้มทั้งจอ

หน้าสรุปงานหนึ่งหน้าอาจโหลดสามอย่างพร้อมกัน

ข้อมูลสำคัญแค่ไหนถ้าล้มเหลว
รายการงานขาดไม่ได้หน้าใช้ไม่ได้ ต้องแสดงผิดพลาด
ยอดคงเหลือคลังมีก็ดีซ่อนส่วนนั้น ที่เหลือใช้ต่อได้
ประกาศจากระบบไม่จำเป็นซ่อนไปเงียบ ๆ

ถ้าใช้สถานะก้อนเดียวคุมทั้งหน้า พอประกาศโหลดไม่ได้ ทั้งหน้าก็ขึ้นผิดพลาด — ทั้งที่รายการงานมาครบแล้ว

วิธีแก้คือ ให้แต่ละแหล่งมีสถานะของตัวเอง แล้วหน้าจอตัดสินว่าอันไหนถึงขั้นบล็อกทั้งหน้า

LoadState _tasksState = LoadState.idle;     // สำคัญ
LoadState _summaryState = LoadState.idle;   // เสริม
ถ้าเลือกซ่อนความล้มเหลวของส่วนเสริม ต้องยังบันทึก log ไว้เสมอ ไม่งั้นจะกลายเป็นปัญหาที่ไม่มีใครรู้ว่ามีอยู่

11. ปุ่มลองใหม่ไม่ได้เหมาะกับทุกความผิดพลาด

ชนิดความผิดพลาดลองใหม่ช่วยไหมควรทำอะไร
หมดเวลารอ / เน็ตหลุดช่วยปุ่มลองใหม่
server ขัดข้อง (500)อาจช่วยปุ่มลองใหม่ แต่เว้นระยะ
ยังไม่ล็อกอิน (401)ไม่ช่วยพาไปหน้าเข้าสู่ระบบ
ข้อมูลถูกคนอื่นแก้ไปแล้ว (409)ไม่ช่วยโหลดข้อมูลใหม่แล้วให้ตัดสินใจอีกครั้ง
กรอกข้อมูลผิดไม่ช่วยพากลับไปแก้ช่องที่ผิด

การกดลองใหม่กับ 401 คือการทำสิ่งเดิมซ้ำและได้ผลเดิม ซึ่งทำให้ผู้ใช้รู้สึกว่าแอปพัง ทั้งที่แอปแค่บอกไม่ถูกว่าต้องทำอะไร

รายละเอียดการแปลงรหัสตอบกลับเป็นการกระทำจะอยู่ใน Part 8

12. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
หมุนไม่หยุดทั้งที่ server ตอบแล้วif (items.isEmpty) spinner — เดาสถานะจากข้อมูลแยกสถานะออกจากข้อมูล
โหลดไม่สำเร็จแล้วจอว่าง ไม่มีอะไรบอกไม่มีสถานะ failedเพิ่มสถานะและหน้าจอของมัน
ลองใหม่แล้วยังขึ้นข้อความผิดพลาดเดิมไม่ล้าง _errorMessage ตอนเริ่มโหลดเคลียร์ทุกครั้งที่เข้า loading
รายการกระพริบทุกครั้งที่โหลดซ้ำเข้า loading ทั้งที่มีข้อมูลอยู่เข้า loading เฉพาะเมื่อยังว่าง
เพิ่มสถานะใหม่แล้วบางจอไม่รองรับมี default: ใน switchตัด default: ให้ compiler ช่วยหา
ส่วนเสริมล้มแล้วทั้งหน้าใช้ไม่ได้สถานะก้อนเดียวคุมทุกแหล่งแยกสถานะต่อแหล่ง
ข้อมูลบนจอไม่ตรงความจริงแต่ไม่มีใครรู้เก็บ stale data โดยไม่บอกแสดงเวลาที่อัปเดตล่าสุด

13. แบบฝึกหัด

หน้าจอหนึ่งใช้โค้ดนี้

bool _loading = false;
String? _error;
List<PickLine> _lines = [];

Future<void> load() async {
  _loading = true;
  notifyListeners();
  try {
    _lines = await repository.fetchLines();
  } catch (e) {
    _error = 'โหลดไม่สำเร็จ';
  }
  _loading = false;
  notifyListeners();
}

Widget build() {
  if (_loading) return const CircularProgressIndicator();
  if (_error != null) return Text(_error!);
  return LineList(lines: _lines);
}
  1. ระบุข้อบกพร่องให้ได้อย่างน้อยสามข้อ (มีมากกว่านั้น)

  3. ตอบว่าถ้าโหลดซ้ำแล้วล้มเหลว ควรล้างรายการเดิมไหม เพราะอะไร

ข้อ 1 — ข้อบกพร่อง

#ข้อบกพร่องผลที่ผู้ใช้เจอ
1ไม่มีสถานะ emptyไม่มีงานจริง กับ ยังโหลดไม่เสร็จ หน้าตาเหมือนกัน
2ไม่ล้าง _error ตอนเริ่มโหลดกดลองใหม่แล้วสำเร็จ แต่ยังขึ้นข้อความผิดพลาดค้างอยู่
3_loading = false ไม่ได้อยู่ใน finallyถ้ามี exception ชนิดที่ catch ไม่ครอบ จะหมุนค้างถาวร (4.8)
4catch (e) จับทุกอย่างบั๊กในโค้ดเราเองก็กลายเป็น “โหลดไม่สำเร็จ” หาสาเหตุไม่เจอ (4.8)
5ล้มเหลวแล้ว _lines ยังค้างของเดิมลำดับ if ทำให้ซ่อนไว้ แต่สถานะจริงขัดกัน
6ไม่แยกโหลดครั้งแรกกับโหลดซ้ำรายการกระพริบทุกครั้งที่รีเฟรช
7ไม่มีปุ่มลองใหม่ล้มเหลวแล้วตัน ต้องปิดแอป

ข้อ 2 — เขียนใหม่

enum LoadState { idle, loading, ready, empty, failed }

LoadState _state = LoadState.idle;
List<PickLine> _lines = const [];
String? _errorMessage;

LoadState get state => _state;
List<PickLine> get lines => List.unmodifiable(_lines);
String? get errorMessage => _errorMessage;

Future<void> load() async {
  _errorMessage = null;                 // แก้ข้อ 2
  if (_lines.isEmpty) {                 // แก้ข้อ 6
    _state = LoadState.loading;
  }
  notifyListeners();
  try {
    final result = await repository.fetchLines();
    _lines = result;
    _state = result.isEmpty             // แก้ข้อ 1
        ? LoadState.empty
        : LoadState.ready;
  } on PickException catch (error) {    // แก้ข้อ 4
    _state = LoadState.failed;
    _errorMessage = error.message;
  } finally {                           // แก้ข้อ 3
    notifyListeners();
  }
}
Widget build(BuildContext context) => switch (store.state) {
  LoadState.idle || LoadState.loading => const LoadingView(),
  LoadState.empty => const EmptyView('วันนี้ยังไม่มีงาน'),
  LoadState.failed => FailedView(                     // แก้ข้อ 7
    message: store.errorMessage ?? 'โหลดไม่สำเร็จ',
    onRetry: store.load,
  ),
  LoadState.ready => LineList(lines: store.lines),
};

สังเกตว่าเมื่อสถานะครบแล้ว หน้าจอไม่ต้องเดาอะไรเลย — มันแค่แปลสถานะเป็นหน้าตา ซึ่งทำให้ทดสอบง่ายด้วย เพราะตั้งสถานะแล้วตรวจว่าเห็นอะไรได้ตรง ๆ

ข้อ 3 — โหลดซ้ำแล้วล้มเหลว

ไม่ควรล้าง โดยมีเงื่อนไข

เหตุผลที่ไม่ล้าง — คนหน้างานถือ PDA ยืนอยู่หน้าแร็ค รายการหยิบ 12 บรรทัดที่โหลดมาเมื่อ 10 วินาทีก่อนยังใช้ทำงานได้ การล้างทิ้งเพราะรีเฟรชไม่ผ่านคือการหยุดงานเขาโดยไม่จำเป็น

เงื่อนไขที่ต้องมี

ต้องทำเพราะ
แสดงแถบเตือนว่ารีเฟรชไม่สำเร็จไม่ให้เข้าใจผิดว่าข้อมูลสด
แสดงเวลาที่อัปเดตล่าสุดให้ตัดสินเองได้ว่าเก่าเกินไปหรือยัง
การกดปิดงานต้องบอกชัดถ้าส่งไม่ได้อ่านข้อมูลเก่าได้ แต่เขียนต้องถึง server จริง

ข้อยกเว้นที่ควรล้าง — เมื่อความล้มเหลวบอกว่าข้อมูลเดิม ผิดแน่นอน เช่น 401 (สิทธิ์หมดอายุ ไม่ควรให้ดูต่อ) หรือ 409 พร้อมข้อมูลว่ามีคนแก้ไปแล้ว

สรุปเป็นหลักเดียว — เก็บข้อมูลเก่าไว้เมื่อมันยังน่าจะจริง และล้างเมื่อรู้ว่ามันไม่จริงแล้ว

14. สรุปท้ายบท

  • boolean ตัวเดียวเก็บได้สองสถานะ แต่ของจริงมีอย่างน้อยห้า
  • boolean หลายตัวเปิดช่องให้เกิดสถานะที่ไม่ควรมีอยู่
  • enum ทำให้เป็นได้ทีละหนึ่งสถานะเท่านั้น
  • empty เป็นความสำเร็จ ไม่ใช่ความผิดพลาด
  • switch ที่ไม่มี default: ทำให้ compiler ช่วยหาจุดที่ลืม
  • เมื่อแต่ละสถานะพกข้อมูลต่างกัน ให้ใช้ sealed class (4.15)
  • โหลดครั้งแรกกับโหลดซ้ำต้องแสดงต่างกัน
  • เก็บข้อมูลเก่าไว้เมื่อยังน่าจะจริง และต้องบอกเสมอว่าเก่า
  • แยกสถานะต่อแหล่ง อย่าให้ส่วนเสริมล้มทั้งหน้า
  • ปุ่มลองใหม่เหมาะกับความผิดพลาดบางชนิดเท่านั้น

บทถัดไป: 4.8 จะเจาะบรรทัด finally ที่โผล่มาในบทนี้ — ทำไมมันคือสิ่งเดียวที่กันหน้าจอค้างหมุนได้จริง
