1. ปัญหาที่บทนี้แก้

4.3 แก้ปัญหาข้อมูลกระจายด้วยการย้ายขึ้นไปที่แม่ร่วม แต่ยังมีข้อจำกัดที่แก้ไม่ได้ — ข้อมูลยังผูกกับหน้าจอ เมื่อหน้าจอถูกทำลาย ข้อมูลหายไปด้วย

ValueNotifier คือก้าวแรกที่ทำให้ข้อมูลอยู่นอกหน้าจอได้ โดยยังเรียบง่ายมาก

2. คำศัพท์ที่ต้องรู้ก่อน

คำความหมาย
ValueNotifier<T>กล่องเก็บค่าหนึ่งค่าที่แจ้งเมื่อค่าเปลี่ยน
ValueListenable<T>สัญญาว่า “อ่านค่าได้และฟังได้” แต่เปลี่ยนค่าไม่ได้
.valueค่าปัจจุบัน อ่านและเขียนได้
equalityการเทียบว่าค่าใหม่ต่างจากค่าเดิมหรือไม่

3. Mental model: กล่องที่ประกาศเมื่อของข้างในเปลี่ยน

final counter = ValueNotifier<int>(0);

counter.value = 1;     // เปลี่ยนค่า → แจ้งทุกคนที่ฟังอยู่
counter.value = 1;     // ค่าเท่าเดิม → ไม่แจ้ง
ประโยคเดียวที่ต้องจำ: มันแจ้งเมื่อ กำหนดค่าใหม่ที่ไม่เท่าค่าเดิม — คำว่า “กำหนดค่าใหม่” สำคัญมาก เพราะการแก้ข้างในของเดิมไม่นับ

4. รูปแบบพื้นฐาน

class ThemeController extends ValueNotifier<ThemeMode> {
  ThemeController() : super(ThemeMode.system);

  void toggle() {
    value = value == ThemeMode.dark ? ThemeMode.light : ThemeMode.dark;
  }
}

การสืบทอดแบบนี้ให้ประโยชน์สองอย่างตามที่เรียนใน D1.8 — ได้ความสามารถของ ValueNotifier มาฟรี และเพิ่ม method ที่สื่อเจตนาอย่าง toggle() แทนที่จะให้คนนอกไปคำนวณเอง

ฝั่งหน้าจออ่านผ่าน builder ตามที่เรียนใน 1.13

ValueListenableBuilder<ThemeMode>(
  valueListenable: themeController,
  builder: (context, mode, child) => MaterialApp(
    themeMode: mode,
    home: child,
  ),
  child: const HomeScreen(),        // ไม่ขึ้นกับ mode จึงสร้างครั้งเดียว
)

5. กับดักใหญ่: แก้ข้างในไม่นับว่าเปลี่ยน

final lines = ValueNotifier<List<PickLine>>([]);

lines.value.add(newLine);        // ไม่แจ้งใครเลย — หน้าจอไม่อัปเดต
lines.value = [...lines.value, newLine];   // แจ้ง เพราะเป็นก้อนใหม่

เหตุผลตรงกับ D1.5 หัวข้อ 7 — ValueNotifier เทียบว่า “เป็นก้อนเดียวกันหรือเปล่า” เมื่อเราแก้ข้างในก้อนเดิม มันจึงเห็นว่าไม่เปลี่ยน

นี่คือเหตุผลที่ lab สร้าง list ก้อนใหม่เสมอ

void _setLine(int index, PickLine line) {
  // สร้าง list ก้อนใหม่ ไม่แก้ก้อนเดิม (D1.5)
  final next = [..._lines];
  next[index] = line;
  _lines = next;
}
อาการนี้หายากเป็นพิเศษเพราะ ไม่มี error อะไรเลย โค้ดดูถูกทุกบรรทัด ข้อมูลก็เปลี่ยนจริง แต่หน้าจอไม่ขยับ — และถ้ามีอะไรอื่นทำให้หน้าจอวาดใหม่พอดี มันจะแสดงถูกบ้างผิดบ้าง

6. ปิดไม่ให้คนนอกเปลี่ยนค่า

class ScanCounter {
  final ValueNotifier<int> _count = ValueNotifier<int>(0);

  /// เปิดให้อ่านและฟังได้ แต่เปลี่ยนค่าไม่ได้
  ValueListenable<int> get count => _count;

  void increment() => _count.value++;

  void reset() => _count.value = 0;

  void dispose() => _count.dispose();
}

การคืนเป็น ValueListenable แทน ValueNotifier ทำให้คนนอกเรียก .value = ... ไม่ได้ — เป็นหลักเดียวกับการคืน List.unmodifiable ใน D1.7

ประโยชน์คือกฎทั้งหมดถูกบังคับผ่าน increment() และ reset() เท่านั้น ไม่มีทางที่ใครจะข้ามไปตั้งค่าเองได้

7. เมื่อไหร่ที่มันเล็กเกินไป

ValueNotifier เก็บได้ค่าเดียว ซึ่งพอสำหรับหลายกรณี แต่หน้าจอที่โหลดข้อมูลจาก server มีอย่างน้อยสามสิ่งที่เปลี่ยนพร้อมกัน

final lines = ValueNotifier<List<PickLine>>([]);
final state = ValueNotifier<LoadState>(LoadState.idle);
final error = ValueNotifier<String?>(null);

ปัญหาผลที่เกิด

8. ใครสร้าง คนนั้นคืน

class _CounterScreenState extends State<CounterScreen> {
  final _tapCount = ValueNotifier<int>(0);      // เราสร้าง

  @override
  void dispose() {
    _tapCount.dispose();                         // เราคืน (1.6, 3.12)
    super.dispose();
  }
}
สร้างที่ไหนใครคืน
ใน State ของหน้าจอหน้าจอนั้น ใน dispose
ในชั้นที่อยู่ยาวกว่าหน้าจอชั้นนั้น — หน้าจอห้ามคืน
รับมาจากภายนอกผ่าน constructorผู้ที่ส่งมา

ValueListenableBuilder เลิกฟังให้เองเมื่อถูกถอดออก แต่ ไม่ได้คืนตัว notifier ให้ — เป็นเรื่องเดียวกับที่ 1.13 หัวข้อ 7 เตือนไว้

9. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
เปลี่ยนข้อมูลใน list แล้วหน้าจอไม่อัปเดตแก้ข้างในก้อนเดิมกำหนดก้อนใหม่ value = [...value, x]
หน้าจอวาดใหม่ทั้งที่ค่าเท่าเดิมกำหนดค่าที่เทียบว่าไม่เท่ากันแม้เนื้อหาเหมือนใช้ชนิดที่เทียบด้วยค่าได้ หรือเทียบเองก่อนตั้ง
คนนอกตั้งค่าข้ามกฎได้เปิด ValueNotifier ออกไปตรง ๆคืนเป็น ValueListenable
หน่วยความจำรั่วลืม disposeคืนใน dispose ของผู้สร้าง
used after being disposedคืนตัวที่รับมาจากภายนอกคืนเฉพาะที่สร้างเอง
ต้องซ้อน builder หลายชั้นใช้ ValueNotifier หลายตัวกับข้อมูลที่เปลี่ยนพร้อมกันรวมเป็นค่าเดียว หรือใช้ ChangeNotifier (4.5)
เห็นสถานะกลางที่ไม่ถูกต้องแวบหนึ่งเปลี่ยนหลายค่าแยกกันเปลี่ยนให้ครบก่อนแล้วแจ้งครั้งเดียว

10. แบบฝึกหัด

เขียน ScanCounter ที่นับจำนวนการยิงบาร์โค้ดในกะปัจจุบัน โดย

  1. เก็บจำนวนที่สำเร็จและจำนวนที่ถูกปฏิเสธ
  2. คนนอกอ่านและฟังได้ แต่ตั้งค่าเองไม่ได้
  3. มีค่าที่คำนวณได้คือจำนวนรวมและอัตราสำเร็จ
  4. รีเซ็ตได้เมื่อเริ่มกะใหม่
  5. ทดสอบได้โดยไม่ต้องมีหน้าจอ

แล้วตอบว่าจะใช้ ValueNotifier กี่ตัว เพราะอะไร

ดูแนวคำตอบและเหตุผล

คำตอบเรื่องจำนวน notifier — ใช้ตัวเดียว

เพราะทั้งสองตัวเลขเปลี่ยนจากเหตุการณ์เดียวกันและถูกอ่านพร้อมกันเสมอ ถ้าแยกเป็นสองตัว หน้าจอจะวาดใหม่สองรอบต่อหนึ่งการยิง และอาจเห็นช่วงที่ตัวเลขไม่สอดคล้องกัน

/// ค่าที่แก้ไม่ได้ เปลี่ยนทีเดียวทั้งก้อน
class ScanTally {
  const ScanTally({this.accepted = 0, this.rejected = 0});

  final int accepted;
  final int rejected;

  // ค่าที่คำนวณได้ ไม่เก็บซ้ำ (4.2)
  int get total => accepted + rejected;

  double get successRate => total == 0 ? 0 : accepted / total;

  ScanTally copyWith({int? accepted, int? rejected}) => ScanTally(
    accepted: accepted ?? this.accepted,
    rejected: rejected ?? this.rejected,
  );

  // ต้องมี เพื่อให้ ValueNotifier รู้ว่าค่าเปลี่ยนจริงหรือไม่
  @override
  bool operator ==(Object other) =>
      other is ScanTally &&
      other.accepted == accepted &&
      other.rejected == rejected;

  @override
  int get hashCode => Object.hash(accepted, rejected);
}

class ScanCounter {
  final ValueNotifier<ScanTally> _tally =
      ValueNotifier<ScanTally>(const ScanTally());

  /// เปิดให้อ่านและฟัง แต่ตั้งค่าไม่ได้
  ValueListenable<ScanTally> get tally => _tally;

  void recordAccepted() {
    _tally.value = _tally.value.copyWith(
      accepted: _tally.value.accepted + 1,
    );
  }

  void recordRejected() {
    _tally.value = _tally.value.copyWith(
      rejected: _tally.value.rejected + 1,
    );
  }

  void resetForNewShift() => _tally.value = const ScanTally();

  void dispose() => _tally.dispose();
}
test('นับแยกสำเร็จกับถูกปฏิเสธ และคำนวณอัตราให้', () {
  final counter = ScanCounter();
  addTearDown(counter.dispose);

  counter.recordAccepted();
  counter.recordAccepted();
  counter.recordRejected();

  expect(counter.tally.value.accepted, 2);
  expect(counter.tally.value.total, 3);
  expect(counter.tally.value.successRate, closeTo(0.667, 0.001));
});

test('รีเซ็ตแล้วเริ่มนับใหม่', () {
  final counter = ScanCounter();
  addTearDown(counter.dispose);

  counter.recordAccepted();
  counter.resetForNewShift();

  expect(counter.tally.value.total, 0);
});

test('แจ้งเมื่อค่าเปลี่ยนจริงเท่านั้น', () {
  final counter = ScanCounter();
  addTearDown(counter.dispose);

  var notifications = 0;
  counter.tally.addListener(() => notifications++);

  counter.recordAccepted();
  counter.resetForNewShift();
  counter.resetForNewShift();     // ค่าเท่าเดิมแล้ว ไม่ควรแจ้ง

  expect(notifications, 2);
});

ทำไมต้องเขียน operator ==

เพราะ ValueNotifier ใช้การเทียบเพื่อตัดสินว่าจะแจ้งหรือไม่ ถ้าไม่เขียน class ที่เราสร้างเองจะเทียบด้วยตัวตนตามที่เรียนใน D1.6 — ผลคือทุกครั้งที่ copyWith สร้างก้อนใหม่ มันจะถือว่าเปลี่ยนเสมอ แม้ตัวเลขจะเท่าเดิม

เทสต์ตัวสุดท้ายจับข้อนี้ — resetForNewShift() ครั้งที่สองตั้งค่าที่เท่ากับค่าปัจจุบัน จึงไม่ควรแจ้ง ถ้าลบ operator == ออก เทสต์นี้จะได้ 3 แทนที่จะเป็น 2

ข้อสังเกต: การเขียน operator == และ hashCode ด้วยมือทุกครั้งน่าเบื่อและลืมง่าย — Part 30 จะสอนให้เครื่องมือสร้างให้อัตโนมัติ ตอนนี้เขียนเองก่อนเพื่อให้เห็นว่ามันจำเป็นเพราะอะไร

ทำไมใช้ addTearDown ในเทสต์

เพื่อให้ dispose ถูกเรียกแม้เทสต์จะล้มเหลวกลางคัน — เป็นหลักเดียวกับ finally ใน D1.15

ลองต่อ: เปลี่ยนเป็น ValueNotifier<int> สองตัวแยกกัน แล้วเขียนเทสต์นับจำนวนครั้งที่หน้าจอถูกแจ้ง — จะเห็นว่าการยิงหนึ่งครั้งทำให้เกิดการแจ้งหนึ่งครั้งเหมือนกัน แต่เมื่อต้องเปลี่ยนทั้งสองค่าพร้อมกัน (เช่นย้ายหนึ่งรายการจากปฏิเสธไปสำเร็จ) จะได้สองครั้ง และมีช่วงที่ยอดรวมผิด

11. สรุปท้ายบท

  • ValueNotifier ทำให้ข้อมูลอยู่นอกหน้าจอได้ โดยยังเรียบง่าย
  • มันแจ้งเมื่อกำหนดค่าใหม่ที่ไม่เท่าค่าเดิม
  • แก้ข้างในก้อนเดิมไม่นับว่าเปลี่ยน ต้องสร้างก้อนใหม่
  • คืนเป็น ValueListenable เพื่อไม่ให้คนนอกตั้งค่าข้ามกฎ
  • class ที่เราสร้างเองต้องเขียน operator == ถ้าจะใช้เป็นค่าใน notifier
  • ใครสร้างคนนั้นคืน builder ไม่คืนให้
  • เมื่อมีหลายค่าที่เปลี่ยนพร้อมกัน ให้รวมเป็นค่าเดียวหรือใช้เครื่องมือที่ใหญ่กว่า

บทถัดไป: 4.5 จะแนะนำ ChangeNotifier ซึ่งเก็บได้หลายค่าและมีที่วางตรรกะ — เป็นเครื่องมือที่ lab ของ Part นี้ใช้จริง
