1. ปัญหาที่บทนี้แก้

สมมติหน้าจอมีสองส่วนที่ต้องใช้จำนวนเดียวกัน — ปุ่มเพิ่มลด และช่องกรอกตัวเลข ถ้าแต่ละส่วนเก็บค่าของตัวเอง จะเกิดอาการที่ผู้ใช้เห็นทันที

ผู้ใช้กดปุ่มเพิ่ม  →  ปุ่มแสดง 5   ช่องกรอกยังแสดง 4
ผู้ใช้พิมพ์ 12     →  ช่องแสดง 12  ปุ่มยังคิดว่าเป็น 5

นี่คือปัญหา “แหล่งความจริงสองที่” ซึ่งแก้ด้วยการย้ายข้อมูลขึ้นไปอยู่ที่แม่ร่วม

3. Mental model: หาแม่ร่วมที่ต่ำที่สุด

          PickScreen          ← ข้อมูลควรอยู่ที่นี่
         /          \
   QuantityStepper  QuantityField
   (ต้องใช้)         (ต้องใช้)
ประโยคเดียวที่ต้องจำ: ย้ายขึ้นไปให้ ต่ำที่สุดเท่าที่ยังครอบทุกคนที่ต้องใช้ — ไม่ใช่ยกขึ้นไปบนสุดของแอปทุกครั้ง เพราะยิ่งสูง ยิ่งมีคนที่ไม่เกี่ยวข้องถูกวาดใหม่

นี่คือกฎเดียวกับที่ 1.5 หัวข้อ 7 บอกไว้ และเป็นกฎเดียวกับการวาง builder ใน 1.13 — ต่างแค่มิติ

4. รูปแบบที่ใช้จริง

class _PickScreenState extends State<PickScreen> {
  int _quantity = 1;                          // แหล่งความจริงเดียว

  void _setQuantity(int next) {
    final clamped = next.clamp(1, 999);       // กฎอยู่ที่แม่ที่เดียว
    if (clamped == _quantity) return;
    setState(() => _quantity = clamped);
  }

  @override
  Widget build(BuildContext context) {
    return Column(
      children: [
        QuantityStepper(
          value: _quantity,                    // ข้อมูลลง
          onChanged: _setQuantity,             // เหตุการณ์ขึ้น
        ),
        QuantityField(
          value: _quantity,
          onChanged: _setQuantity,
        ),
      ],
    );
  }
}

สังเกตว่าทั้งสองส่วนใช้ _setQuantity ตัวเดียวกัน จึงเป็นไปไม่ได้ที่กฎจะไม่ตรงกัน — ตรงกับหลักใน 3.4 ที่ว่าทุกทางต้องไปที่ method เดียว

และลูกทั้งสองเป็น StatelessWidget ได้ เพราะไม่ต้องจำอะไร

5. ลูกต้องไม่เก็บสำเนาไว้เอง

// ผิด — ลูกเก็บสำเนา
class _QuantityFieldState extends State<QuantityField> {
  late int _local = widget.value;      // คัดลอกครั้งเดียวตอนเกิด

  // เมื่อแม่เปลี่ยนค่า _local ไม่รู้เรื่อง
}

ทบทวนจาก 1.6 — initState ทำงานครั้งเดียว การคัดลอกค่าจากแม่ตอนนั้นทำให้ค่าค้างไปตลอด

ถ้าจำเป็นต้องเก็บสำเนาจริง ๆ (เช่นช่องกรอกที่ต้องมีตัวควบคุมของตัวเอง) ต้องซิงก์เมื่อแม่ส่งค่าใหม่

@override
void didUpdateWidget(covariant QuantityField oldWidget) {
  super.didUpdateWidget(oldWidget);
  if (oldWidget.value != widget.value) {          // เทียบก่อนเสมอ (1.9)
    _controller.text = widget.value.toString();
  }
}
อย่าซิงก์โดยไม่เทียบก่อน — เพราะ didUpdateWidget ถูกเรียกทุกครั้งที่แม่วาดใหม่ ถ้าตั้งค่าทุกรอบ เคอร์เซอร์จะเด้งไปต้นบรรทัดขณะที่ผู้ใช้กำลังพิมพ์

6. เมื่อการส่งผ่านหลายชั้นเริ่มเป็นปัญหา

PickScreen (มีข้อมูล)
└── Column
    └── Card              ← ไม่ได้ใช้ แต่ต้องรับแล้วส่งต่อ
        └── Padding       ← ไม่ได้ใช้
            └── Row       ← ไม่ได้ใช้
                └── QuantityField   ← ใช้จริง
ระยะห่างทำอย่างไร
1–2 ชั้นส่งผ่าน constructor ตามปกติ
3 ชั้นขึ้นไป แต่ชั้นกลางเป็นแค่การจัดวางยังส่งผ่านได้ เพราะชั้นกลางไม่ใช่ widget ที่เราเขียนเอง
3 ชั้นขึ้นไป และชั้นกลางเป็น widget ของเราที่ไม่ได้ใช้เริ่มมีปัญหา — ดู 4.5

ข้อสังเกตสำคัญ — ชั้นกลางอย่าง Column หรือ Padding ไม่ได้รับข้อมูลผ่าน constructor เพราะเราเขียน QuantityField ไว้ข้างในโดยตรง ปัญหาการส่งผ่านหลายชั้นเกิดเฉพาะเมื่อเราแตก widget ของตัวเองออกมาเป็นชั้น ๆ แล้วต้องส่งต่อทีละชั้น

7. ส่งข้อมูลที่แก้ไม่ได้ลงไป

// เสี่ยง — ลูกแก้ได้เอง แม่ไม่รู้
LineList(lines: _lines)

// ปลอดภัย — ลูกแก้ไม่ได้
LineList(lines: List.unmodifiable(_lines))

ใน lab เราคืนแบบแก้ไม่ได้จาก store เสมอ

List<PickLine> get lines => List.unmodifiable(_lines);

และเทสต์ยืนยันว่าคนนอกแก้ไม่ได้จริง

test('รายการที่คืนออกไปแก้ไม่ได้', () async {
  await store.load();
  expect(
    () => store.lines.add(const PickLine(id: 'X', palletCode: 'X', quantity: 1)),

เหตุผลตรงกับ D1.7 — ถ้าลูกแก้ข้อมูลได้โดยตรง กฎทั้งหมดที่แม่เขียนไว้จะถูกข้าม และไม่มีใครแจ้งให้หน้าจอวาดใหม่

8. ยกขึ้นเท่าที่จำเป็น

ข้อมูลยกไปถึงไหนถ้ายกสูงเกินไปจะเกิดอะไร
โหมดแสดงผลของการ์ดใบเดียวอยู่ในการ์ดนั้นทั้งหน้าจอวาดใหม่เมื่อสลับโหมด
รายการที่ถูกเลือกในหน้านี้แม่ร่วมของรายการกับรายละเอียดหน้าจออื่นถูกวาดใหม่โดยไม่จำเป็น
ผู้ใช้ที่ล็อกอินชั้นบนสุดของแอปถูกต้องแล้ว
การยกทุกอย่างขึ้นไปไว้ที่เดียวทั้งแอปทำให้เกิดสิ่งที่เรียกว่า “ที่เก็บทุกอย่าง” — เมื่อค่าใด ๆ เปลี่ยน ทุกคนที่ฟังอยู่จะถูกแจ้งหมด แม้จะไม่เกี่ยวข้อง ผลคือแอปช้าลงและหาสาเหตุของการวาดใหม่ยากมาก

9. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
สองส่วนแสดงค่าไม่ตรงกันแต่ละส่วนเก็บสำเนาของตัวเองย้ายขึ้นไปที่แม่ร่วม
แม่เปลี่ยนค่าแล้วลูกไม่ตามลูกคัดลอกไว้ใน initStateอ่านจาก widget.value หรือซิงก์ใน didUpdateWidget
เคอร์เซอร์เด้งขณะพิมพ์ซิงก์ทุกครั้งโดยไม่เทียบก่อนเทียบค่าเก่ากับใหม่ก่อนตั้ง
กฎการตรวจไม่ตรงกันระหว่างสองทางเขียนตรรกะสองชุดให้ทุกทางเรียก method เดียว
ลูกแก้ข้อมูลแล้วหน้าจอไม่อัปเดตส่ง list ที่แก้ได้ลงไปส่งแบบแก้ไม่ได้
เปลี่ยนค่าเล็ก ๆ แล้วทั้งแอปวาดใหม่ยกขึ้นสูงเกินไปย้ายลงมาให้ต่ำที่สุดที่ยังครอบทุกคน

10. แบบฝึกหัด

หน้าจอหนึ่งมีโครงสร้างแบบนี้

PickScreen
├── AppBar          — แสดงชื่อหน้าจอ ไม่ใช้ข้อมูลอะไร
├── FilterBar       — เลือกกรองเฉพาะที่ยังไม่เสร็จ
├── LineList        — แสดงรายการตามตัวกรอง
└── SummaryFooter   — แสดงจำนวนที่กรองแล้ว
  1. ตัวกรองควรอยู่ที่ไหน
  2. รายการที่กรองแล้วควรเก็บเป็นตัวแปรหรือคำนวณ
  3. เขียนโครงโค้ด
  4. ถ้าวันหนึ่งหน้าจออื่นต้องใช้ตัวกรองเดียวกันด้วย จะทำอย่างไร
ดูแนวคำตอบและเหตุผล

ข้อ 1

อยู่ที่ PickScreen เพราะเป็นแม่ร่วมที่ต่ำที่สุดของสามส่วนที่ต้องใช้ — FilterBar เปลี่ยนมัน ส่วน LineList กับ SummaryFooter อ่านมัน ส่วน AppBar ไม่เกี่ยว จึงไม่ต้องรับอะไร

ข้อ 2

คำนวณ ตามหลักใน 4.2 — เก็บแค่ตัวกรองกับรายการทั้งหมด แล้วรายการที่กรองแล้วเป็น getter

class _PickScreenState extends State<PickScreen> {
  bool _showOnlyRemaining = false;         // เก็บ

  // คำนวณ — ไม่มีทางไม่ตรงกับตัวกรอง
  List<PickLine> get _visibleLines => _showOnlyRemaining
      ? widget.store.lines.where((line) => !line.done).toList()
      : widget.store.lines;

  void _setFilter(bool value) {
    if (value == _showOnlyRemaining) return;
    setState(() => _showOnlyRemaining = value);
  }

  @override
  Widget build(BuildContext context) {
    final visible = _visibleLines;           // คำนวณครั้งเดียวต่อการวาด

    return Scaffold(
      appBar: AppBar(title: const Text('หยิบสินค้า')),   // ไม่รับอะไร
      body: Column(
        children: [
          FilterBar(
            showOnlyRemaining: _showOnlyRemaining,
            onChanged: _setFilter,
          ),
          Expanded(child: LineList(lines: visible)),
          SummaryFooter(visibleCount: visible.length),
        ],
      ),
    );
  }
}

ทำไมเก็บ visible ลงตัวแปรใน build

เพราะ _visibleLines สร้าง list ใหม่ทุกครั้งที่เรียก ถ้าเรียกสองครั้ง (ครั้งหนึ่งให้ LineList อีกครั้งให้ SummaryFooter) จะกรองสองรอบโดยไม่จำเป็น — การเก็บลงตัวแปรในเครื่องแก้ได้ทันทีโดยไม่ต้องทำอะไรซับซ้อน

ข้อ 4 — ถ้าหน้าจออื่นต้องใช้ตัวกรองเดียวกัน

ให้กลับไปถามสามคำถามจาก 4.1 ใหม่ — คำตอบข้อแรกเปลี่ยนจาก “หน้าจอนี้คนเดียว” เป็น “หลายหน้าจอ” ดังนั้นตัวกรองต้องย้ายออกจาก setState

แต่ต้องถามต่ออีกข้อก่อนตัดสิน — สองหน้าจอนั้นต้องเห็นตัวกรองเดียวกันจริงไหม หรือแค่มีตัวกรองแบบเดียวกัน

ความต้องการทำอย่างไร
ตั้งกรองในหน้าหนึ่ง แล้วอีกหน้าต้องเห็นตามด้วยย้ายไปที่ store
แต่ละหน้ามีตัวกรองของตัวเอง แค่หน้าตาเหมือนกันคงไว้ที่ setState ของแต่ละหน้า แล้วแชร์เฉพาะ widget

ข้อนี้เป็นกับดักที่พบบ่อย — เห็นว่าสองหน้าจอมีตัวกรองเหมือนกัน เลยรีบย้ายไปที่เก็บกลาง ผลคือผู้ใช้ตั้งกรองในหน้าหนึ่งแล้วอีกหน้าเปลี่ยนตามโดยไม่ได้ตั้งใจ ซึ่งเป็นพฤติกรรมที่ผู้ใช้ไม่คาดคิด

ลองต่อ: ใน lab ให้เพิ่มตัวกรองตามแนวคำตอบนี้ลงใน PickScreen โดยยังไม่แตะ PickStore — จะเห็นว่าทำได้โดยไม่กระทบเทสต์ของ store เลยแม้แต่ตัวเดียว ซึ่งเป็นหลักฐานว่าเราวางข้อมูลถูกที่

11. สรุปท้ายบท

  • เมื่อหลายส่วนต้องใช้ข้อมูลเดียวกัน ให้ย้ายขึ้นไปที่แม่ร่วมที่ต่ำที่สุด
  • ข้อมูลไหลลงผ่าน constructor เหตุการณ์ไหลขึ้นผ่าน callback
  • ลูกต้องไม่เก็บสำเนา — ถ้าจำเป็นต้องซิงก์ใน didUpdateWidget พร้อมเทียบก่อน
  • ให้ทุกทางที่เปลี่ยนค่าเรียก method เดียว กฎจะได้ไม่แตกต่างกัน
  • ส่งข้อมูลแบบแก้ไม่ได้ลงไป
  • ยกขึ้นเท่าที่จำเป็น อย่ายกทุกอย่างไปไว้ที่เดียวทั้งแอป
  • ถามก่อนว่า “ต้องเห็นค่าเดียวกัน” หรือ “แค่มีแบบเดียวกัน”

บทถัดไป: 4.4 จะแนะนำเครื่องมือแรกที่ทำให้ข้อมูลอยู่นอกหน้าจอได้ — ValueNotifier ซึ่งเราเจอผ่าน ๆ มาแล้วใน 1.13
