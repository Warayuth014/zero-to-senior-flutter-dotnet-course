4. Load + generation

int _generation = 0;
Future<void> load({bool refresh = false}) async {
  final token = ++_generation;
  _emit(snapshot.loading(refresh: refresh));
  try {
    final tasks = await repository.fetchOpen();
    if (token != _generation) return;
    _emit(snapshot.data(List.unmodifiable(tasks)));
  } catch (e) {
    if (token != _generation) return;
    _emit(snapshot.failure(mapError(e), keepData: true));
  }
}

5. Idempotent confirm

Future<void> confirm(String taskId) async {
  if (_confirmingTaskId != null) return;
  final operationId = _pendingId ?? ids.next();
  _confirmingTaskId = taskId;
  _pendingId = operationId;
  _publishPending();
  try {
    await repository.confirm(taskId, operationId: operationId);
    _pendingId = null;
    await load(refresh: true);
  } on TimeoutException {
    _publishUnknown(); // keep operationId for retry/reconcile
  } finally {
    _confirmingTaskId = null;
    _publishIfAlive();
  }
}
