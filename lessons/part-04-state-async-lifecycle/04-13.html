1. ปัญหาที่บทนี้แก้

คนหน้างานหยิบของ 40 บรรทัด กดปุ่มปิดงานทีละบรรทัด ถ้าแต่ละครั้งต้องรอเซิร์ฟเวอร์ 600 มิลลิวินาที

40 บรรทัด × 0.6 วินาที = 24 วินาทีที่ยืนรอเฉย ๆ ต่อหนึ่ง order

และช่วงที่รอนั้นไม่มีอะไรเปลี่ยนบนจอเลย ผู้ใช้จึงมักกดซ้ำเพราะคิดว่ากดไม่ติด

ทางแก้ที่ตรงที่สุดคือ อัปเดตหน้าจอทันทีเหมือนสำเร็จแล้ว แล้วค่อยยิงคำขอไปเบื้องหลัง ซึ่งเรียกว่า optimistic UI

แต่ทันทีที่ทำ จะมีคำถามใหม่ที่ต้องตอบให้ได้ — ถ้ามันไม่สำเร็จล่ะ

4. ดูของจริงใน lab

เปิด lib/state_basics/pick_store.dart

Future<void> markDone(String lineId) async {
  if (_busyLineIds.contains(lineId)) return;
  final index = _lines.indexWhere((line) => line.id == lineId);
  if (index == -1) return;
  final before = _lines[index];              // 1. เก็บสำเนาไว้ก่อน
  if (before.done) return;

  final commandId = _pendingCommandIds.putIfAbsent(lineId, _createCommandId);
  _busyLineIds.add(lineId);
  _errorMessage = null;
  _setLine(index, before.copyWith(done: true));   // 2. โกหกไว้ก่อน
  notifyListeners();

  try {
    await repository.markDone(lineId, commandId: commandId);
    _pendingCommandIds.remove(lineId);             // 3. สำเร็จ ลบร่องรอย
  } on PickException catch (error) {
    if (error.outcomeUnknown) {                    // 4. ไม่รู้ผล
      _errorMessage = '${error.message} กำลังตรวจสอบสถานะล่าสุด';
      notifyListeners();
      await load();                                //    ถามความจริงใหม่
      return;
    }
    _rollback(lineId, before);                     // 5. รู้แน่ว่าล้มเหลว
    _pendingCommandIds.remove(lineId);
    _errorMessage = error.message;
  } finally {
    _busyLineIds.remove(lineId);
    notifyListeners();
  }
}

class PickLine {
  const PickLine({required this.id, required this.palletCode,
                  required this.quantity, this.done = false});
  final String id;              // final ทุกตัว
  final bool done;

  PickLine copyWith({bool? done}) => PickLine(   // เปลี่ยน = สร้างตัวใหม่
    id: id, palletCode: palletCode, quantity: quantity, done: done ?? this.done);

ถ้าเป็นวัตถุที่แก้ค่าในตัวเองได้ สำเนาจะไร้ความหมาย

// อันตราย ถ้า PickLine แก้ค่าในตัวเองได้
final before = _lines[index];    // ชี้ไปที่วัตถุตัวเดียวกัน
_lines[index].done = true;       // แก้ตัวเดียวกันนั้น

และด้วยเหตุผลเดียวกัน _setLine จึงสร้าง list ใหม่ทุกครั้ง

void _setLine(int index, PickLine line) {
  final next = [..._lines];    // สำเนาใหม่
  next[index] = line;
  _lines = next;
}

7. ทางเลือกที่มักดีกว่าทั้งสองแบบ

optimistic บอกว่า “เสร็จแล้ว” pessimistic บอกว่า “ยังไม่มีอะไรเกิดขึ้น” ทั้งคู่ไม่ตรงความจริง เพราะความจริงคือ “กำลังส่งอยู่”

┌──────────────────────────────┐
│ ○ PAL-0003  12 ชิ้น          │
│ ✓ PAL-0004  8 ชิ้น           │
│ ⟳ PAL-0005  20 ชิ้น  กำลังส่ง │  ← pending: ตรงกับความจริงที่สุด
└──────────────────────────────┘
ข้อดีข้อเสีย
ไม่เคยแสดงข้อมูลที่ผิดต้องมีสถานะเพิ่มและหน้าตาเพิ่ม
ผู้ใช้กดต่อได้เลย ไม่ต้องรอตัวเลขสรุปต้องตัดสินว่านับอันที่รออยู่ด้วยไหม
ถ้าล้มเหลว ไม่มีอะไร “ย้อน” ให้ตกใจ

ใน lab มี isBusy(lineId) อยู่แล้ว ซึ่งเป็นข้อมูลที่พอจะทำแบบนี้ได้ — เพียงแต่หน้าจอเลือกใช้มันแค่ปิดปุ่ม ไม่ได้แสดงเป็นสถานะแยก

ถ้าไม่แน่ใจว่าควรใช้แบบไหน ให้เริ่มที่ pending ก่อน — มันบอกความจริง ซึ่งเป็นสิ่งที่อธิบายให้ผู้ใช้เข้าใจได้ง่ายที่สุดเสมอ

8. กรณีที่ยากที่สุด: ไม่รู้ว่าสำเร็จหรือไม่

นี่คือจุดที่แยก optimistic ที่เขียนถูก ออกจากที่เขียนพอผ่าน

ยิงคำขอไป ─────────→ server รับ → บันทึกสำเร็จ ─────X (ตอบกลับหาย)
                                                     ↑
                                           เน็ตหลุดตรงนี้พอดี

ฝั่งแอปได้แค่ “หมดเวลารอ” ซึ่งแปลได้สองอย่างที่ต่างกันสิ้นเชิง

ความจริงที่เป็นไปได้ถ้าย้อนกลับ
คำขอไปไม่ถึงเลยถูกต้อง
ถึงแล้ว บันทึกแล้ว แต่คำตอบหายผิด — จอบอกว่ายังไม่เสร็จ ทั้งที่เสร็จแล้ว

ถ้าย้อนกลับในกรณีที่สอง ผู้ใช้จะกดใหม่ แล้วงานเดียวถูกบันทึกสองครั้ง

class PickException implements Exception {
  const PickException(this.message, {this.outcomeUnknown = false});
  final String message;

  /// true เมื่อไม่รู้ว่า server ทำสำเร็จหรือยัง เช่นหมดเวลารอ
  final bool outcomeUnknown;
}

11. บอกผู้ใช้ให้ตรงกับสิ่งที่เกิดจริง

สิ่งที่เกิดควรบอกว่าห้ามบอกว่า
ย้อนกลับแล้วเพราะถูกปฏิเสธ“ปิดงานไม่สำเร็จ: บรรทัดนี้ถูกยกเลิกแล้ว”“เกิดข้อผิดพลาด”
ไม่รู้ผล กำลังตรวจสอบ“กำลังตรวจสอบสถานะล่าสุด”“ปิดงานไม่สำเร็จ”
สำเร็จไม่ต้องบอกอะไร เครื่องหมายถูกก็พอเด้งข้อความทุกครั้ง

สิ่งที่แย่ที่สุดคือย้อนกลับเงียบ ๆ โดยไม่บอกอะไร — ผู้ใช้เห็นเครื่องหมายถูกหายไปเอง จะสรุปว่าแอปเสีย และจากนั้นจะไม่เชื่อสิ่งที่จอแสดงอีกเลย

ส่วนปุ่ม “เลิกทำ” ให้มีได้เฉพาะเมื่อ เซิร์ฟเวอร์รองรับการยกเลิกจริง ไม่ใช่แค่แก้ค่าในเครื่องกลับ

12. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ย้อนกลับแล้วได้ค่าเดียวกับที่เปลี่ยนไปเก็บตัวชี้ไปยังวัตถุที่แก้ได้ใช้วัตถุที่แก้ไม่ได้ + copyWith
ย้อนกลับไปทับบรรทัดอื่นใช้ index เก่าหลัง awaitหาใหม่ด้วย id
งานถูกบันทึกสองครั้งย้อนกลับตอนที่ไม่รู้ผล แล้วผู้ใช้กดใหม่แยก outcomeUnknown แล้วโหลดใหม่แทน
กดบรรทัดหนึ่งแล้วทั้งจอกดไม่ได้ใช้ bool เดียวคุมทั้งหน้าใช้ Set ต่อรายการ
เครื่องหมายถูกหายไปเองโดยไม่มีคำอธิบายย้อนกลับเงียบตั้งข้อความและแสดงให้เห็น
สต๊อกไม่ตรงกับความจริงใช้ optimistic กับสิ่งที่ย้อนไม่ได้เปลี่ยนเป็น pessimistic

Future<void> adjust(String lineId, int delta) async {

    await repository.adjust(lineId, delta);
  } catch (e) {
    _lines[index].counted -= delta;
    notifyListeners();
  }
}

#ข้อบกพร่องผลที่เกิด
1ไม่ตรวจ index == -1RangeError เมื่อบรรทัดหายไป
2แก้ค่าในวัตถุเดิมไม่มีสำเนาให้ย้อน และ widget อาจไม่วาดใหม่ (4.4)
3ใช้ index เก่าหลัง awaitย้อนไปทับบรรทัดอื่น
4catch (e) จับทุกอย่างบั๊กถูกกลืน (4.8)
5ไม่แยกกรณีไม่รู้ผลปรับจำนวนซ้ำสองรอบ
6ไม่กันกดซ้ำกดรัว ๆ ได้คำขอซ้อนกันหลายอัน
7ย้อนแบบ -= delta ไม่ใช่คืนค่าเดิมถ้ามีคำขออื่นแทรก จะได้ค่าที่ไม่เคยมีอยู่จริง
8ไม่บอกอะไรผู้ใช้เมื่อล้มเหลวตัวเลขเปลี่ยนเองโดยไม่มีคำอธิบาย
9ไม่มี finallyถ้ามี exception นอก catch จะค้าง

ข้อ 7 ละเอียดที่สุด — -= delta เป็นการย้อน การเปลี่ยนแปลง ไม่ใช่การคืน ค่า ลองไล่ดู

counted = 10
กด +1 → counted = 11, ยิงคำขอ A
กด +1 → counted = 12, ยิงคำขอ B
A ล้มเหลว → counted -= 1 → 11
B สำเร็จ  → เซิร์ฟเวอร์มีค่า 11 แต่จอแสดง 11 ที่ได้มาจากทางที่ผิด
           ถ้า B ล้มเหลวด้วย → counted = 10 ซึ่งบังเอิญถูก
           แต่ถ้า A ล้มเหลวหลัง B สำเร็จ ลำดับจะเพี้ยนทันที

ข้อ 2 — เขียนใหม่

Future<void> adjust(String lineId, int delta) async {
  if (_busyLineIds.contains(lineId)) return;        // แก้ข้อ 6

  final index = _lines.indexWhere((l) => l.id == lineId);
  if (index == -1) return;                          // แก้ข้อ 1
  final before = _lines[index];                     // แก้ข้อ 2, 7

  final commandId = _pendingCommandIds.putIfAbsent(lineId, _createCommandId);
  _busyLineIds.add(lineId);
  _errorMessage = null;
  _setLine(index, before.copyWith(counted: before.counted + delta));
  notifyListeners();

  try {
    await repository.adjust(lineId, delta, commandId: commandId);
    _pendingCommandIds.remove(lineId);
  } on CountException catch (error) {               // แก้ข้อ 4
    if (error.outcomeUnknown) {                     // แก้ข้อ 5
      _errorMessage = '${error.message} กำลังตรวจสอบยอดล่าสุด';
      notifyListeners();
      await load();
      return;
    }
    _rollback(lineId, before);                      // แก้ข้อ 3, 7
    _pendingCommandIds.remove(lineId);
    _errorMessage = 'ปรับยอดไม่สำเร็จ: ${error.message}';   // แก้ข้อ 8
  } finally {                                       // แก้ข้อ 9
    _busyLineIds.remove(lineId);
    notifyListeners();
  }
}

โดยที่ CountLine ต้องเปลี่ยนเป็นวัตถุที่แก้ไม่ได้

class CountLine {
  const CountLine({required this.id, required this.counted});
  final String id;
  final int counted;
  CountLine copyWith({int? counted}) =>
      CountLine(id: id, counted: counted ?? this.counted);
}

สังเกตว่า _rollback คืน before ทั้งก้อน ไม่ได้ลบ delta ออก — จึงได้ค่าที่เคยมีอยู่จริงเสมอ

ข้อ 3 — ควรใช้แบบไหน

คำตอบขึ้นกับว่าปุ่มนี้ทำอะไรจริง ๆ ซึ่งต้องแยกให้ออกก่อน

ถ้าปุ่มนี้คือเลือกเหตุผล
ปรับตัวเลขในใบนับที่ยังไม่ยืนยันoptimistic ในเครื่อง ไม่ยิงเลยยังเป็นแค่ร่าง ส่งทีเดียวตอนกดยืนยัน
ปรับยอดในใบนับที่บันทึกทีละครั้งpendingตัวเลขต้องถูกต้อง แต่ต้องกดต่อได้เร็ว
ปรับยอดสต๊อกจริงในระบบpessimisticกระทบสต๊อก และมีคนอื่นแก้พร้อมกันได้

คำตอบที่ควรให้คือแบบแรก — การนับสต๊อกโดยธรรมชาติเป็นงานที่คนเดินนับไปเรื่อย ๆ แล้วยืนยันทั้งใบตอนจบ การยิงเซิร์ฟเวอร์ทุกครั้งที่กดบวกลบเป็นการออกแบบที่สร้างปัญหาให้ตัวเองโดยไม่จำเป็น

ยิงทุกครั้งที่กดเก็บในเครื่องแล้วส่งทีเดียว
ต้องจัดการ race, rollback, idempotency ทุกปุ่มกดกี่ครั้งก็ไม่มีคำขอ ไม่มี race
เน็ตหลุดแล้วนับต่อไม่ได้เน็ตหลุดก็นับต่อได้ ส่งตอนมีสัญญาณ
คำขอเป็นร้อยครั้งต่อใบคำขอเดียวต่อใบ
ยกเลิกการนับกลางคันแล้วข้อมูลค้างยกเลิกได้สะอาด ไม่มีอะไรถูกส่งไป

แต่ต้องแลกกับสองอย่าง — ต้องเก็บร่างไว้ในเครื่องไม่ให้หายเมื่อแอปถูกปิด และตอนยืนยันต้องรับมือกับกรณีที่มีคนอื่นแก้ยอดไปแล้วระหว่างที่กำลังนับ

บทเรียนที่ควรจำ: คำถาม “optimistic หรือ pessimistic” มักมาจากการที่เราตั้งขอบเขตของคำสั่งเล็กเกินไป — ถ้าเปลี่ยนหนึ่งการกระทำของผู้ใช้ให้ตรงกับหนึ่งคำสั่งที่มีความหมายทางธุรกิจ คำถามนี้จะง่ายลงมาก

ลองต่อ: ใน test/pick_store_test.dart มีเทสต์ที่ทำให้ repository ล้มเหลวแล้วตรวจว่า done กลับเป็น false ลองแก้ _rollback ให้ใช้ index ที่หาไว้ตอนต้นแทนการหาใหม่ แล้วเพิ่มเทสต์ที่ให้ load() เปลี่ยนลำดับรายการระหว่างที่คำขอค้างอยู่ — จะเห็นว่าย้อนไปผิดบรรทัดจริง

14. สรุปท้ายบท

  • optimistic คือการโกหกชั่วคราว — ทำได้เมื่อคำโกหกนั้นแก้กลับง่าย
  • เก็บสำเนาของค่าเดิมไว้ก่อนเปลี่ยนเสมอ และวัตถุต้องแก้ไม่ได้ สำเนาจึงมีความหมาย
  • ย้อนกลับด้วยการคืนค่าเดิมทั้งก้อน ไม่ใช่การลบส่วนต่างออก
  • หาตำแหน่งใหม่ด้วย id เสมอ อย่าใช้ index เก่าหลัง await
  • เมื่อไม่รู้ผล ห้ามเดา — ให้โหลดใหม่แล้วถามความจริงจากเซิร์ฟเวอร์
  • ล็อกต่อรายการด้วย Set ไม่ใช่ล็อกทั้งจอด้วย bool
  • pending มักเป็นทางเลือกที่ดีที่สุด เพราะบอกความจริง
  • สิ่งที่กระทบสต๊อกหรือย้อนไม่ได้ในโลกจริง ต้องเป็น pessimistic
  • ห้ามย้อนกลับเงียบ ๆ ต้องบอกเหตุผลเสมอ
