1. ทำไมต้องมีทางสำรอง

12.1 สรุปไว้ว่าเรียลไทม์หลุดเงียบ — และนี่คือรายการเต็มของสิ่งที่ทำให้แอปไม่ได้รับเหตุการณ์โดยที่ไม่มีอะไรบอก

สาเหตุสถานะที่แอปเห็น
ต่อไม่ติดตั้งแต่แรกdisconnected — รู้
สายหลุดกลางทางreconnecting — รู้
ลองต่อใหม่จนครบแล้วเลิกdisconnected — รู้
ต่ออยู่แต่ไม่ได้เข้ากลุ่มconnected — ไม่รู้
ชื่อเหตุการณ์ไม่ตรงกับเซิร์ฟเวอร์connected — ไม่รู้
เหตุการณ์แปลงไม่ได้ทุกอันconnected — ไม่รู้
ตัวกลางในเครือข่ายตัด socket โดยไม่แจ้งconnected — ไม่รู้

สี่แถวล่างคือเหตุผลที่ทางสำรองไม่ควรพึ่งสถานะอย่างเดียว และทำไม 12.10 ถึงแนะนำตาข่ายกันพลาดที่โหลดใหม่เป็นระยะแม้ตอนที่ทุกอย่างดูปกติ

2. สลับอัตโนมัติ

void _onStatus(RealtimeStatus value) {
  final previous = _status;
  _status = value;

  switch (value) {
    case RealtimeStatus.connected:
      _stopPolling();
      if (previous == RealtimeStatus.reconnecting ||
          previous == RealtimeStatus.disconnected) {
        unawaited(load());
      }
    case RealtimeStatus.reconnecting:
    case RealtimeStatus.disconnected:
      // ระหว่างที่เรียลไทม์ใช้ไม่ได้ ให้ถามซ้ำเป็นระยะแทน
      _startPolling();
    case RealtimeStatus.connecting:
      break;
  }

  notifyListeners();
}
สถานะทำอะไรเหตุผล
connectedหยุดถามซ้ำ + โหลดอุดช่วงที่ขาดไม่จ่ายค่าเครือข่ายสองต่อ
reconnectingเริ่มถามซ้ำอาจกลับมาใน 2 วินาที หรือไม่กลับมาเลย — ไม่ต้องรอให้รู้
disconnectedเริ่มถามซ้ำทางเดียวที่เหลือ
connectingไม่ทำอะไรเป็นสถานะชั่วคราวตอนเริ่ม ซึ่ง start() โหลดให้อยู่แล้ว

การเริ่มถามซ้ำตั้งแต่ reconnecting เป็นการตัดสินใจที่มีเหตุผล — ถ้ารอจนถึง disconnected จะเสียเวลาไปราว 42 วินาทีตามค่าปริยาย ซึ่งนานเกินไปสำหรับข้อมูลที่ต้องการเรียลไทม์ตั้งแต่แรก

3. กันการเริ่มซ้ำ

void _startPolling() {
  if (_pollTimer != null) return;      // มีอยู่แล้ว ไม่สร้างใหม่
  _pollTimer = Timer.periodic(pollInterval, (_) => unawaited(load()));
}

void _stopPolling() {
  _pollTimer?.cancel();
  _pollTimer = null;                   // ตั้งเป็น null ด้วย ไม่ใช่แค่ cancel
}

สองบรรทัดนี้สำคัญกว่าที่ดู

ถ้าไม่มี if (_pollTimer != null) return;สถานะเปลี่ยนจาก reconnecting เป็น disconnected จะสร้างตัวจับเวลาตัวที่สอง ทับตัวแรกที่ยังเดินอยู่ — ยิงคำขอสองเท่า และตัวแรกยกเลิกไม่ได้อีกเลยเพราะไม่มีใครถืออ้างอิงไว้
ถ้าไม่ตั้ง _pollTimer = nullisPolling จะยังบอกว่ากำลังถามซ้ำอยู่ ทั้งที่ยกเลิกไปแล้ว

ตัวจับเวลาที่ถูกทับแบบนี้เรียกว่า orphaned timer และเป็นการรั่วชนิดที่หายากที่สุด เพราะไม่มีทางเข้าถึงเพื่อยกเลิกได้อีกเลย — วิธีเดียวที่จะหยุดคือปิดแอป

4. เทสต์ที่พิสูจน์ทั้งวงจร

test('หลุดแล้วเริ่มถามซ้ำ ต่อกลับได้แล้วหยุด', () {
  fakeAsync((async) {
    repository.result = [task('T-1')];
    unawaited(store.start());
    async.flushMicrotasks();
    expect(store.isPolling, isFalse, reason: 'ต่อได้อยู่ ไม่ต้องถามซ้ำ');

    inner.simulateDrop();
    async.flushMicrotasks();
    expect(store.isPolling, isTrue);

    final callsBefore = repository.fetchCalls;
    async.elapse(const Duration(seconds: 45));
    expect(repository.fetchCalls, callsBefore + 3);

    inner.simulateReconnected();
    async.flushMicrotasks();
    expect(store.isPolling, isFalse);

    // หลังหยุดแล้วต้องไม่มีคำขอเพิ่มจากตัวจับเวลาอีก
    // (+1 มาจากการโหลดอุดช่วงที่ขาด)
    final callsAfterReconnect = repository.fetchCalls;
    async.elapse(const Duration(minutes: 5));
    expect(repository.fetchCalls, callsAfterReconnect);

    store.dispose();
    async.flushTimers();
  });
});
ส่วนพิสูจน์อะไร
expect(isPolling, isFalse) ตอนแรกไม่จ่ายค่าเครือข่ายสองต่อตอนปกติ
elapse(45s) ได้ 3 ครั้งถามซ้ำจริงตามช่วงเวลา 15 วินาที
elapse(5 นาที) ไม่เพิ่มหยุดจริง ไม่ใช่แค่ธงเปลี่ยน

ส่วนสุดท้ายคือส่วนที่จับ orphaned timer ได้ — ถ้ามีตัวจับเวลาตัวที่สองค้างอยู่ ห้านาทีจะทำให้จำนวนคำขอเพิ่มขึ้น 20 ครั้ง

fakeAsync ทำให้ทดสอบเรื่องเวลาได้จริงในเสี้ยววินาที — ถ้าเขียนด้วย Future.delayed จริง เทสต์นี้จะใช้เวลาเกือบหกนาที ซึ่งไม่มีใครยอมรันใน CI

5. เลือกช่วงเวลาอย่างไร

ตอนเป็นทางสำรอง เกณฑ์ต่างจากตอนที่ถามซ้ำเป็นวิธีหลัก

ถามซ้ำเป็นวิธีหลักถามซ้ำเป็นทางสำรอง
ทำงานนานแค่ไหนตลอดเวลาเฉพาะตอนเรียลไทม์ล้ม
ต้นทุนรวมสูงและคงที่ต่ำ เพราะเกิดไม่บ่อย
ควรตั้งเท่าไหร่ห่างที่สุดที่ยอมรับได้ถี่กว่านั้นได้ เพราะจ่ายไม่นาน

lab ตั้งไว้ที่ 15 วินาที ซึ่งถี่กว่าที่จะยอมรับได้ถ้าเป็นวิธีหลัก แต่เหมาะสำหรับช่วงที่เรียลไทม์ล้ม

/// ความถี่ของการถามซ้ำ เมื่อเรียลไทม์ใช้ไม่ได้
///
/// ไม่ได้ใช้ตอนที่เรียลไทม์ทำงานปกติ — การมีทั้งสองอย่างพร้อมกันคือ
/// การจ่ายค่าเครือข่ายสองต่อเพื่อข้อมูลชุดเดียว
final Duration pollInterval;

ถ้าเรียลไทม์ล้มนานมาก อาจถอยห่างขึ้นเรื่อย ๆ แบบเดียวกับ 7.12

// ล้มนานแล้ว ลดความถี่ลงเพื่อประหยัดแบต แต่มีเพดาน
void _scheduleNextPoll() {
  _pollTimer = Timer(_currentInterval, () {
    unawaited(load());
    _currentInterval = Duration(
      seconds: (_currentInterval.inSeconds * 2).clamp(15, 120),
    );
    _scheduleNextPoll();
  });
}

/// กำลังถามซ้ำเป็นระยะอยู่ไหม — ใช้แสดงให้ผู้ใช้รู้ว่าข้อมูลอาจช้า
bool get isPolling => _pollTimer != null;
ควรบอกไม่ควรบอก
“ข้อมูลอาจช้าถึง 15 วินาที”“การเชื่อมต่อเรียลไทม์ล้มเหลว”
เวลาที่อัปเดตล่าสุดชื่อเทคโนโลยีที่ผู้ใช้ไม่รู้จัก
ปุ่มโหลดใหม่ทันทีปุ่ม “เชื่อมต่อใหม่” ที่ไม่รู้ว่าทำอะไร

หลักจาก 10.12 ใช้ที่นี่เหมือนกัน — ผู้ใช้ต้องการรู้ว่าข้อมูลที่เห็นเชื่อได้แค่ไหน ไม่ใช่ว่าเทคโนโลยีไหนกำลังทำงานอยู่

และอย่าขวางการทำงาน — ทางสำรองที่ทำงานอยู่แปลว่าข้อมูลช้า ไม่ได้แปลว่าทำอะไรไม่ได้

7. เรื่องที่โปรเจกต์จริงยังไม่ได้ทำ

Future<void> connect() async {
  ...
  try {
    ...
    await _connection!.start();
    dev.log('[SignalR] Connected to $hubUrl');
  } catch (e) {
    dev.log('[SignalR] Connect error: $e');
  }
}

catch บันทึกไว้แล้วจบ — คนเรียกไม่มีทางรู้ว่าล้มเหลว เพราะ connect() คืน Future<void> ที่สำเร็จเสมอ

ผลที่ตามมาบนเครื่องจริง

ลำดับเกิดอะไร
1PDA อยู่นอกรัศมี Wi-Fi ตอนเปิดจอ
2start() โยน ถูกจับและบันทึกลง log ที่ไม่มีใครดู
3จอโหลดสถานะครั้งแรกผ่าน HTTP ได้ (ถ้าเน็ตกลับมาพอดี) หรือไม่ได้
4ไม่มีเหตุการณ์ใด ๆ ตลอดกาล และไม่มีทางสำรอง
5ผู้ใช้เห็นหน้าจอที่ดูปกติทุกอย่าง แต่ตัวเลขไม่เคยเปลี่ยน

วิธีแก้ที่เล็กที่สุด คือให้ connect() บอกผลกลับไป

Future<bool> connect() async {
  ...
  try {
    await _connection!.start();
    return true;
  } catch (e) {
    dev.log('[SignalR] Connect error: $e');
    return false;      // คนเรียกตัดสินใจเองว่าจะทำอย่างไรต่อ
  }
}
// ในจอ
Future<void> _initSignalR() async {
  final ok = await _signalR.connect();
  if (!ok && mounted) _startFallbackPolling();
}

8. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ยิงคำขอสองเท่าไม่หยุดถามซ้ำตอนต่อกลับได้หยุดใน connected
ยิงถี่ขึ้นเรื่อย ๆ จนแอปช้าสร้างตัวจับเวลาซ้อนกันตรวจก่อนสร้าง
ตัวจับเวลาหยุดไม่ได้ทับตัวเก่าจนไม่มีอ้างอิงตรวจก่อนสร้าง และตั้งเป็น null ตอนยกเลิก
ต่อไม่ติดแล้วเงียบทั้งวันกลืนข้อผิดพลาดตอนต่อคืนผลให้คนเรียกตัดสินใจ
ผู้ใช้ไม่รู้ว่าข้อมูลช้าไม่มีอะไรบอกแสดงเวลาที่อัปเดตล่าสุด
ถามซ้ำถี่มากทั้งที่ล้มมาสองชั่วโมงช่วงเวลาคงที่ถอยห่างขึ้นแบบมีเพดาน

9. แบบฝึกหัด

ทีมพบว่าเรียลไทม์ในคลังหลุดบ่อยมาก จนทางสำรองทำงานเกือบตลอดเวลา

  1. ควรทำอย่างไร
  2. จะรู้ได้อย่างไรว่าตัดสินใจถูก
แนวคำตอบ

1. อย่าเพิ่งตัดสินใจ — วัดก่อน เพราะ “หลุดบ่อย” เป็นความรู้สึก ไม่ใช่ข้อมูล

// เก็บสถิติไว้ก่อนตัดสินใจ
class RealtimeMetrics {
  Duration totalConnected = Duration.zero;
  Duration totalDisconnected = Duration.zero;
  int dropCount = 0;
  int eventsReceived = 0;
  int pollsIssued = 0;
}

แล้วดูตัวเลขจริง

ถ้าพบว่าควรทำ
ต่ออยู่ได้ 30% ของเวลาถอดเรียลไทม์ทิ้ง ใช้ถามซ้ำอย่างเดียว — จ่ายความซับซ้อนทั้ง Part นี้เพื่อประโยชน์หนึ่งในสาม ไม่คุ้ม
ต่ออยู่ได้ 95% แต่หลุดบ่อยเป็นช่วงสั้น ๆเก็บไว้ ปรับ retryDelays ให้ต่อกลับเร็วขึ้น
หลุดเฉพาะบางพื้นที่ของคลังเป็นปัญหา Wi-Fi ไม่ใช่ปัญหาโค้ด — ส่งข้อมูลนี้ให้ทีมไอที มีค่ามากกว่าการแก้แอป
หลุดทุกครั้งที่หน้าจอดับเป็นพฤติกรรมของระบบปฏิบัติการ ต่อใหม่ตอนกลับมา (12.4)

แถวที่สามมีค่าที่สุด — ข้อมูลว่า “PDA หลุดตรงโซน C ทุกครั้ง” เป็นสิ่งที่ไม่มีใครในทีมไอทีรู้ และแอปเป็นที่เดียวที่วัดได้

2. ตัดสินจากสิ่งที่ผู้ใช้ได้รับจริง ไม่ใช่จากความสมบูรณ์ของโค้ด

วัดบอกอะไร
เวลาเฉลี่ยจากตอนเซิร์ฟเวอร์เปลี่ยน ถึงตอนหน้าจอเปลี่ยนตัวเลขจริงที่ผู้ใช้รู้สึก
จำนวนคำขอ HTTP ต่อเครื่องต่อวันต้นทุนที่จ่ายจริง
สัดส่วนของเหตุการณ์ที่ได้รับ เทียบกับที่เซิร์ฟเวอร์ส่งเรียลไทม์ทำงานจริงแค่ไหน

ถ้าตัวแรกดีขึ้นน้อยมากเมื่อเทียบกับตอนใช้ถามซ้ำอย่างเดียว แปลว่าเรียลไทม์ไม่ได้ทำงานให้เท่าที่จ่ายไป

สิ่งที่ยากที่สุดในข้อนี้ไม่ใช่เทคนิค — คือการยอมถอดของที่ทีมลงแรงสร้างไปแล้ว การมีตัวเลขทำให้การตัดสินใจนั้นเป็นเรื่องของข้อเท็จจริง ไม่ใช่เรื่องของใครถูกใครผิด

10. สรุป

  • เรียลไทม์ล้มได้หลายแบบ และสี่แบบในนั้นสถานะยังบอกว่าปกติ
  • สลับไปถามซ้ำตั้งแต่ reconnecting ไม่ต้องรอถึง disconnected
  • ตรวจก่อนสร้างตัวจับเวลา ไม่งั้นจะได้ตัวที่ยกเลิกไม่ได้
  • ช่วงเวลาของทางสำรองถี่กว่าวิธีหลักได้ เพราะจ่ายไม่นาน
  • บอกผู้ใช้ว่าข้อมูลช้าแค่ไหน ไม่ใช่บอกว่าเทคโนโลยีไหนล้ม
  • อย่ากลืนข้อผิดพลาดตอนต่อ — คืนผลให้คนเรียกตัดสินใจ
