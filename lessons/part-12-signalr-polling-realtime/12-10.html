1. สองแหล่งที่ต้องอยู่ด้วยกัน

เหตุการณ์โหลดผ่าน HTTP
ตอบคำถามว่ามีอะไรเปลี่ยนตอนนี้เป็นอย่างไร
ครบไหมไม่ครบ — พลาดได้ (12.2)ครบ ณ วินาทีที่ถาม
เร็วไหมเกือบทันทีต้องรอคำตอบ
เชื่อได้แค่ไหนเชื่อได้ถ้ามันมาถึงเชื่อได้เสมอ แต่เก่าทันทีที่ได้รับ
แถวสุดท้ายคือสิ่งที่คนมองข้าม — คำตอบของ HTTP เก่าตั้งแต่วินาทีที่เซิร์ฟเวอร์ถ่ายข้อมูลออกมา ไม่ใช่ตอนที่แอปได้รับ ระหว่างนั้นอาจมีอะไรเปลี่ยนไปแล้ว

นี่คือเหตุผลที่ทั้งสองแหล่งต้องมีตัวเทียบว่าอันไหนใหม่กว่า และตัวเทียบนั้นต้องมาจากเซิร์ฟเวอร์ ไม่ใช่จากเวลาที่แอปได้รับ

2. เหตุการณ์ไม่เรียงลำดับ

ความเข้าใจผิดที่พบบ่อยที่สุดคือ “เหตุการณ์มาถึงตามลำดับที่เกิด”

เซิร์ฟเวอร์ส่ง:     TaskDispatched(v2)  →  TaskCompleted(v3)
แอปได้รับ:          TaskCompleted(v3)  →  TaskDispatched(v2)

สาเหตุที่เป็นไปได้

สาเหตุรายละเอียด
สายหลุดกลางทางอันแรกหาย อันที่สองมาถึงหลังต่อกลับ
เซิร์ฟเวอร์ส่งจากหลาย instanceระบบที่มี backplane ไม่รับประกันลำดับข้ามเครื่อง
การโหลดใหม่แทรกกลางคำตอบของ HTTP ที่ถ่ายไว้ก่อนหน้า มาถึงหลังเหตุการณ์ (12.7 §3)

สาเหตุที่สามเกิดบ่อยที่สุดและไม่ต้องมีอะไรผิดปกติเลย

3. เลขรุ่นเป็นคำตอบ

/// เลขรุ่นที่เซิร์ฟเวอร์เพิ่มขึ้นทุกครั้งที่งานนี้เปลี่ยน
///
/// เป็นตัวตัดสินว่าข้อมูลไหนใหม่กว่า — จะมาจากเหตุการณ์หรือจากการ
/// โหลดผ่าน HTTP ก็ตัดสินด้วยตัวเลขนี้เหมือนกัน
final int version;
void _applyIfNewer(String taskId, int version, ...) {
  final existing = _tasks[taskId];
  if (existing == null) { ... }
  if (version <= existing.version) return; // เก่ากว่าหรือซ้ำ ทิ้ง
  _tasks[taskId] = update(existing);
}

เงื่อนไข <= ไม่ใช่ < เพราะรุ่นเดียวกันแปลว่าเหตุการณ์ซ้ำ ซึ่งเกิดได้เมื่อเซิร์ฟเวอร์ส่งซ้ำหรือเมื่อได้รับทั้งจากเหตุการณ์และจากการโหลด

test('เหตุการณ์เก่าที่มาถึงช้า ต้องไม่ทับของใหม่', () async {
  repository.result = [task('T-1', version: 5)];
  await store.start();

  // เหตุการณ์รุ่น 3 มาถึงหลังจากที่เรารู้รุ่น 5 แล้ว
  inner.emit(const TaskCompleted(taskId: 'T-1', version: 3));

  expect(store.tasks.single.status, LiveTaskStatus.waiting);
  expect(store.tasks.single.version, 5);
});
ทางเลือกอื่นแทนเลขรุ่นปัญหา
เวลาที่เซิร์ฟเวอร์บันทึกใช้ได้ ถ้าความละเอียดพอและนาฬิกาของเซิร์ฟเวอร์ทุกตัวตรงกัน
เวลาที่แอปได้รับใช้ไม่ได้ — เป็นตัวเลขที่บอกลำดับการมาถึง ไม่ใช่ลำดับการเกิด
ไม่มีเลย ใช้อันหลังสุดเสมอคือปัญหาที่กำลังแก้อยู่

4. เขียนทับทั้งชุด หรือผสาน

ตอนโหลดผ่าน HTTP มีสองทางเลือก

// lab เลือกทางนี้ — เขียนทับทั้งชุด
_tasks
  ..clear()
  ..addEntries(result.map((task) => MapEntry(task.id, task)));
เขียนทับทั้งชุดผสานทีละรายการ
รายการที่หายไปจากคำตอบถูกลบ — ถูกต้องต้องตัดสินใจว่าลบไหม
เหตุการณ์ที่มาถึงระหว่างรอคำตอบถูกทับรอดถ้ารุ่นใหม่กว่า
ความซับซ้อนสามบรรทัดต้องเทียบทีละรายการ

แถวแรกเป็นเหตุผลที่ lab เลือกเขียนทับ — งานที่หายไปจากรายการงานค้าง แปลว่ามันไม่ค้างแล้ว การผสานจะทำให้มันอยู่ตลอดไป

แถวที่สองคือราคาที่จ่าย และมีขอบเขตจำกัด — หน้าต่างที่เสี่ยงคือช่วงระหว่างที่เซิร์ฟเวอร์ถ่ายข้อมูลกับตอนที่แอปได้รับ ซึ่งเป็นหลักร้อยมิลลิวินาที และครั้งถัดไปที่มีเหตุการณ์เกี่ยวกับงานนั้นก็จะแก้ให้ถูกเอง

ถ้ารับความเสี่ยงนี้ไม่ได้ ให้ผสานโดยเทียบรุ่น พร้อมจัดการรายการที่หายไปให้ชัด

Future<void> load() async {
  final result = await repository.fetchOpen();
  final incoming = {for (final task in result) task.id: task};

  // ลบสิ่งที่ไม่อยู่ในคำตอบแล้ว — มันไม่ค้างแล้ว
  _tasks.removeWhere((id, _) => !incoming.containsKey(id));

  // ผสานทีละรายการ ให้รุ่นที่ใหม่กว่าชนะ
  for (final task in result) {
    final existing = _tasks[task.id];
    if (existing == null || task.version > existing.version) {
      _tasks[task.id] = task;
    }
  }
}
ยังมีช่องโหว่อยู่ — งานที่เพิ่งถูกสร้างจากเหตุการณ์ระหว่างรอคำตอบ จะถูกลบโดยบรรทัด removeWhere เพราะไม่อยู่ในคำตอบที่ถ่ายไว้ก่อนหน้า การแก้ให้ครบต้องใช้เลขรุ่นระดับทั้งชุด ซึ่งฝั่งเซิร์ฟเวอร์ต้องรองรับ

เลือกทางที่ง่ายก่อนเสมอ แล้วยกระดับเมื่อวัดได้ว่าปัญหาเกิดจริง — ตามหลักใน 9.12

5. เหตุการณ์ที่พูดถึงสิ่งที่เราไม่รู้จัก

case TaskCompleted(:final taskId, :final version):
  _applyIfNewer(
    taskId, version,
    (task) => task.copyWith(status: LiveTaskStatus.done, version: version),
    // งานที่เสร็จแล้วและเราไม่เคยรู้จัก ไม่ต้องเพิ่มเข้ามา —
    // ลิสต์นี้แสดงงานค้าง การเพิ่มงานที่เสร็จแล้วเข้าไปไม่มีประโยชน์
    orElse: null,
  );

เทียบกับ TaskDispatched ที่ตรงกันข้าม

case TaskDispatched(:final taskId, :final assignee, :final version):
  _applyIfNewer(
    taskId, version,
    (task) => task.copyWith(...),
    orElse: () => LiveTask(id: taskId, ...),   // งานใหม่ ต้องโผล่ในลิสต์
  );

ทั้งสองกรณีต้องตัดสินใจอย่างมีเหตุผล ไม่ใช่เขียนเหมือนกันไปหมด — และเหตุผลมาจากคำถามเดียวว่า ลิสต์นี้แสดงอะไร

6. เมื่อไหร่ควรโหลดใหม่ทั้งที่มีเหตุการณ์

จังหวะเพราะ
เปิดจอยังไม่มีข้อมูลอะไรเลย
ต่อกลับหลังหลุดมีช่วงที่ขาดไป (12.4)
กลับมาจากพื้นหลังระบบอาจตัดการเชื่อมต่อโดยไม่แจ้ง
ผู้ใช้ดึงลงเพื่อรีเฟรชเขาไม่เชื่อว่าข้อมูลทันสมัย — ควรให้เขาถูกต้องได้
ได้รับเหตุการณ์ที่แปลงไม่ได้หลายครั้งสัญญาอาจเปลี่ยน เหตุการณ์เชื่อไม่ได้แล้ว
ทุก ๆ ช่วงเวลาหนึ่ง แม้เรียลไทม์ปกติเป็นตาข่ายกันพลาดสำหรับเหตุการณ์ที่หายโดยไม่รู้ตัว

แถวสุดท้ายเป็นการยอมรับความจริง — ไม่มีระบบเรียลไทม์ไหนที่ไม่เคยพลาด การโหลดใหม่ทุกห้านาทีเป็นราคาที่ถูกมากเมื่อเทียบกับการที่หน้าจอผิดไปทั้งวันโดยไม่มีใครรู้

// ตาข่ายกันพลาด ห่างพอที่จะไม่กระทบต้นทุน แต่ถี่พอที่จะจำกัดความเสียหาย
_safetyNet = Timer.periodic(const Duration(minutes: 5), (_) {
  if (status == RealtimeStatus.connected) unawaited(load());
});

7. เหตุการณ์ที่บอกให้โหลด แทนที่จะบอกข้อมูล

มีรูปแบบที่สามที่ใช้ได้ดีในหลายกรณี

// เซิร์ฟเวอร์ส่งแค่ว่า "มีอะไรเปลี่ยนในกลุ่มนี้" ไม่ส่งข้อมูล
await _hub.Clients.All.SendAsync("TasksChanged", new { zone = "A1" });
case TasksChanged(:final zone):
  if (zone == _currentZone) unawaited(load());   // แล้วค่อยไปถามเอง
ได้เสีย
ไม่ต้องกังวลเรื่องลำดับหรือรุ่นเลยยิง HTTP ทุกครั้งที่มีอะไรเปลี่ยน
สัญญาของเหตุการณ์เล็กมาก แทบไม่มีอะไรให้ผิดช้ากว่าเล็กน้อย
ข้อมูลถูกต้องเสมอเพราะมาจากแหล่งเดียวถ้ามีเหตุการณ์ถี่มาก จะยิงถี่ตาม

เหมาะกับข้อมูลที่เปลี่ยนไม่บ่อยแต่ต้องถูกต้องแน่นอน และเป็นทางที่ควรเริ่มก่อนถ้าไม่แน่ใจ — เพราะเขียนง่ายกว่ามากและยกระดับเป็นแบบส่งข้อมูลมาด้วยทีหลังได้

ถ้าเหตุการณ์ถี่ ให้รวบก่อนยิง

Timer? _debounce;

void _scheduleReload() {
  _debounce?.cancel();
  // เหตุการณ์สิบอันใน 300ms ยิงคำขอเดียว
  _debounce = Timer(const Duration(milliseconds: 300), () => unawaited(load()));
}

8. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
งานที่เสร็จแล้วกลับมาค้างคำตอบเก่ามาถึงทีหลังเทียบเลขรุ่น
สถานะกระพริบไปมาเหตุการณ์กับการโหลดสลับกันทับให้ตัวที่รุ่นใหม่กว่าชนะเสมอ
งานที่เสร็จแล้วโผล่ในลิสต์งานค้างสร้างรายการใหม่จากเหตุการณ์ที่ไม่ควรสร้างตัดสินใจแยกตามชนิดของเหตุการณ์
ข้อมูลผิดทั้งวันโดยไม่มีใครรู้ไม่มีการโหลดใหม่เป็นระยะตาข่ายกันพลาดทุกไม่กี่นาที
ยิง HTTP ถี่มากตอนมีเหตุการณ์รัวโหลดใหม่ทุกเหตุการณ์รวบด้วย debounce
เชื่อเวลาที่แอปได้รับเป็นตัวตัดสินสับสนระหว่างลำดับการมาถึงกับลำดับการเกิดใช้เลขรุ่นจากเซิร์ฟเวอร์

9. แบบฝึกหัด

ทีมหลังบ้านบอกว่าเพิ่ม version ให้ทุกเหตุการณ์ไม่ได้ เพราะตารางเดิมไม่มีคอลัมน์นั้นและแก้ไม่ทันรุ่นนี้

  1. มีทางอื่นไหม
  2. ทางไหนที่ควรเลือก
แนวคำตอบ

1. สี่ทาง เรียงจากดีไปแย่

ทางใช้ได้ไหม
ใช้เวลาที่เซิร์ฟเวอร์บันทึก (updatedAt) แทนได้ ถ้ามีคอลัมน์นี้อยู่แล้ว และความละเอียดถึงมิลลิวินาที
เปลี่ยนเป็นเหตุการณ์ที่บอกให้โหลด (§7)ได้ และแก้ปัญหาทั้งหมดโดยไม่ต้องแตะตาราง
ให้เหตุการณ์ส่งสถานะเต็มมาด้วย แล้วเชื่ออันหลังสุดยังผิดได้ตอนคำตอบเก่ามาถึงทีหลัง
ไม่ทำอะไร หวังว่าจะไม่เกิดจะเกิด และจะแก้ยากเพราะทำซ้ำไม่ได้

2. เลือกทางที่สอง — เปลี่ยนเป็นเหตุการณ์ที่บอกให้โหลด

เหตุผลที่ชนะทางแรก แม้ทางแรกจะดูตรงกว่า

  • ไม่ต้องแตะฝั่งเซิร์ฟเวอร์เลยนอกจากลดข้อมูลที่ส่ง ซึ่งเป็นการแก้ที่เล็กกว่าการเพิ่มคอลัมน์
  • ไม่ต้องพึ่งความแม่นของนาฬิกาเซิร์ฟเวอร์ ซึ่งเป็นสมมติฐานที่พังเงียบเมื่อมีเซิร์ฟเวอร์หลายตัว
  • สัญญาเล็กลงมาก จึงมีอะไรให้ผิดน้อยลง

ราคาที่จ่ายคือคำขอ HTTP เพิ่มขึ้น ซึ่งจัดการได้ด้วย debounce และยังน้อยกว่าการถามซ้ำทุกห้าวินาทีอยู่ดี

สิ่งที่ควรทำควบคู่: เปิดตั๋วงานสำหรับการเพิ่ม version ไว้ พร้อมเขียนเหตุผลว่าทำไมถึงต้องมีในอนาคต — เพราะเมื่อระบบโตขึ้นและเหตุการณ์ถี่ขึ้น การโหลดใหม่ทุกครั้งจะกลายเป็นต้นทุนจริง และตอนนั้นจะได้ไม่ต้องเถียงกันใหม่ตั้งแต่ต้น

บทเรียนที่ใหญ่กว่า: เมื่อฝั่งหนึ่งทำสิ่งที่ถูกต้องที่สุดไม่ได้ ให้หาทางที่ถูกต้องที่ทำได้ตอนนี้ ไม่ใช่ทำสิ่งที่รู้อยู่แล้วว่าผิดแล้วหวังว่าจะไม่มีใครเจอ

10. สรุป

  • เหตุการณ์บอกว่ามีอะไรเปลี่ยน HTTP บอกว่าตอนนี้เป็นอย่างไร ต้องมีทั้งคู่
  • เหตุการณ์ไม่เรียงลำดับ และคำตอบของ HTTP เก่าตั้งแต่ตอนที่เซิร์ฟเวอร์ถ่ายข้อมูล
  • ตัวตัดสินว่าอันไหนใหม่กว่าต้องมาจากเซิร์ฟเวอร์ ไม่ใช่เวลาที่แอปได้รับ
