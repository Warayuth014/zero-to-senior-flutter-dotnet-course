1. ปัญหา

ในโปรเจกต์จริง จอ putaway กับจอ putaway prework ต่างก็ฟังเหตุการณ์จาก hub เดียวกัน และผู้ใช้เปิดจอที่สองซ้อนจอแรกได้

ถ้าแต่ละจอต่อของตัวเองผล
สอง socket ค้างไว้กินแบตสองเท่า
เซิร์ฟเวอร์เห็นสอง connection จากเครื่องเดียวส่งเหตุการณ์เดียวกันสองรอบ
ปิดจอบน แล้วตัดการเชื่อมต่อจอล่างเงียบทั้งที่ยังเปิดอยู่

แถวสุดท้ายคือสิ่งที่ต้องแก้ให้ได้ — ไม่ใช่แค่ประหยัด แต่คือความถูกต้อง

2. การนับผู้ใช้งาน

class SharedRealtimeClient {
  SharedRealtimeClient(this._inner);

  final RealtimeClient _inner;
  int _users = 0;

  /// ขอใช้การเชื่อมต่อ — ต่อจริงเฉพาะคนแรก
  Future<void> acquire() async {
    _users++;
    if (_users == 1) await _inner.connect();
  }

  /// เลิกใช้ — ตัดจริงเฉพาะเมื่อไม่เหลือใครแล้ว
  Future<void> release() async {
    if (_users == 0) return;
    _users--;
    if (_users == 0) await _inner.disconnect();
  }
}

แนวคิดตรงไปตรงมา และเป็นแนวเดียวกับที่ wmsapp ทำอยู่แล้ว

Future<void> connect() async {
  _refCount++;
  if (_connection != null &&
      _connection!.state != HubConnectionState.Disconnected) {
    return; // already connected
  }
  ...
}

Future<void> disconnect() async {
  _refCount--;
  if (_refCount <= 0) {
    _refCount = 0;
    if (_connection != null) {
      await _connection!.stop();
      _connection = null;
    }
  }
}

สังเกต if (_refCount <= 0) { _refCount = 0; ... } — เขียนกันค่าติดลบไว้แล้ว ซึ่งเป็นรายละเอียดที่คนมักลืม

3. ทำไมค่าติดลบถึงอันตราย

test('ปล่อยเกินจำนวน ต้องไม่ทำให้ตัวนับติดลบ', () async {
  await shared.acquire();
  await shared.release();
  await shared.release(); // เรียกเกิน — เป็นบั๊กที่พบบ่อย
  await shared.release();

  expect(shared.userCount, 0);

  // ถ้าตัวนับติดลบ การขอใช้ครั้งถัดไปจะไม่ต่อจริง แล้วจอจะเงียบ
  await shared.acquire();
  expect(inner.connectCalls, 2);
});

ถ้าปล่อยให้ติดลบเป็น -2 แล้วมีจอใหม่เรียก acquire() ตัวนับจะเป็น -1 ซึ่งไม่เท่ากับ 1 จึงไม่ต่อจริง — จอนั้นจะเปิดขึ้นมาแล้วเงียบตลอดกาล โดยไม่มีข้อผิดพลาดใด ๆ

การเรียก release() เกินเกิดง่ายกว่าที่คิด — จอที่ถูกทิ้งสองครั้ง โค้ดที่เรียกทั้งใน dispose และใน callback ของการออกจากระบบ หรือการรีเฟรชที่สร้าง store ใหม่โดยไม่ทิ้งตัวเก่า

4. กฎการจับคู่

ทุก acquire() ต้องมี release() คู่กันเป๊ะ ๆ หนึ่งครั้ง วิธีที่ทำให้ผิดยากคือให้ทั้งคู่อยู่ในคลาสเดียวกัน

Future<void> start() async {
  _eventSub = client.events.listen(_onEvent);
  _statusSub = client.status.listen(_onStatus);
  await client.acquire();          // ← ที่นี่
  await load();
}

@override
void dispose() {
  _eventSub?.cancel();
  _statusSub?.cancel();
  _stopPolling();
  unawaited(client.release());     // ← และที่นี่
  super.dispose();
}

อ่านสองเมธอดนี้คู่กันแล้วเห็นทันทีว่าจับคู่ครบ — ต่างจากการที่จอหนึ่งเรียก acquire แล้วอีกที่หนึ่งเรียก release

ถ้า start() ล้มเหลวกลางทางล่ะ

Future<void> start() async {
  _eventSub = client.events.listen(_onEvent);
  _statusSub = client.status.listen(_onStatus);
  await client.acquire();
  // ถ้า load() โยน acquire ก็ทำไปแล้ว และ dispose จะ release ให้
  // — จับคู่ยังครบ เพราะ dispose ถูกเรียกเสมอเมื่อจอถูกทิ้ง
  await load();
}

ที่นี่ปลอดภัยเพราะ load() จับข้อผิดพลาดเองอยู่แล้ว แต่ถ้ามีบรรทัดที่โยนได้อยู่ระหว่าง acquire กับจุดที่จอถือว่าเริ่มแล้ว ต้องคิดให้รอบคอบ

5. ควรตัดการเชื่อมต่อทันทีที่ไม่มีคนใช้ไหม

คำตอบไม่ชัดเจนเท่าที่คิด

ตัดทันทีรอสักครู่ก่อนตัด
ประหยัดทันทีผู้ใช้ที่สลับจอไปมาไม่ต้องรอต่อใหม่ทุกครั้ง
ผู้ใช้ที่กดผิดแล้วกดกลับ ต้องรอต่อใหม่ถือ socket ไว้เกินจำเป็นสักพัก

การต่อ SignalR ใหม่ใช้เวลาราวครึ่งวินาทีถึงสองวินาที ซึ่งผู้ใช้รู้สึกได้ ถ้าจอในฟีเจอร์เดียวกันสลับไปมาบ่อย การหน่วงก่อนตัดคุ้ม

Timer? _idleTimer;

Future<void> release() async {
  if (_users == 0) return;
  _users--;
  if (_users > 0) return;

  // รอสักครู่ เผื่อจอถัดไปในฟีเจอร์เดียวกันกำลังจะเปิด
  _idleTimer?.cancel();
  _idleTimer = Timer(const Duration(seconds: 10), () {
    if (_users == 0) unawaited(_inner.disconnect());
  });
}

Future<void> acquire() async {
  _idleTimer?.cancel();          // มีคนมาใช้แล้ว ยกเลิกการตัด
  _idleTimer = null;
  _users++;
  if (_users == 1 && _inner.currentStatus != RealtimeStatus.connected) {
    await _inner.connect();
  }
}
ระวังตรงนี้ — ตัวจับเวลาที่เพิ่มเข้ามาก็ต้องถูกยกเลิกให้ถูกจังหวะเหมือนกัน (12.8) และเงื่อนไข _users == 1 ต้องเพิ่มการตรวจสถานะจริง ไม่งั้นจะต่อซ้ำตอนที่ยังต่ออยู่

อย่าเพิ่มความซับซ้อนนี้ตั้งแต่แรก — เพิ่มเมื่อวัดได้ว่าผู้ใช้สลับจอบ่อยจริงและการรอทำให้ช้า ตามหลักใน 9.12

6. hub หลายตัว

โปรเจกต์จริงมีสอง hub และแต่ละตัวมีการนับของตัวเอง

final putawayHub = SharedRealtimeClient(HubClient(path: '/hubs/putaway', ...));
final sortingHub = SharedRealtimeClient(HubClient(path: '/hubs/sorting', ...));
คำถามคำตอบ
ควรรวมเป็น hub เดียวไหมเป็นการตัดสินใจฝั่งเซิร์ฟเวอร์ — แยกทำให้แต่ละฟีเจอร์ปรับอิสระได้
ควรมีตัวจัดการกลางที่ถือทุก hub ไหมเมื่อมี hub ตัวที่สาม (12.6 §8)
สลับ profile แล้วต้องทำอะไรตัดและต่อใหม่ทุก hub เพราะ URL เปลี่ยน (12.3 §7)

แถวสุดท้ายคือจุดที่ต้องมีตัวจัดการกลาง — ถ้าแต่ละ hub ถูกถือคนละที่ จะไม่มีใครรับผิดชอบการต่อใหม่ทั้งชุดตอนสลับเซิร์ฟเวอร์

7. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ปิดจอบนแล้วจอล่างเงียบตัดการเชื่อมต่อโดยไม่นับผู้ใช้นับผู้ใช้งาน
จอใหม่เปิดแล้วเงียบตลอดกาลตัวนับติดลบกันที่ศูนย์
มีสอง connection จากเครื่องเดียวแต่ละจอสร้างของตัวเองใช้ตัวที่แชร์กัน
ต่อซ้ำทั้งที่ยังต่ออยู่ตรวจแค่ตัวนับ ไม่ได้ตรวจสถานะจริงตรวจทั้งสองอย่าง
สลับเซิร์ฟเวอร์แล้วยังฟัง hub เก่าไม่มีใครรับผิดชอบการต่อใหม่มีตัวจัดการกลาง

8. แบบฝึกหัด

ผู้ใช้ออกจากระบบแล้วเข้าใหม่ด้วยบัญชีอื่น การเชื่อมต่อ SignalR ควรเป็นอย่างไร

  1. ต้องตัดและต่อใหม่ไหม
  2. ใครควรเป็นคนสั่ง
แนวคำตอบ

1. ต้อง ด้วยเหตุผลสามข้อ

เหตุผลรายละเอียด
token ที่ผูกกับ connection เป็นของคนเก่าส่งไปตอน accessTokenFactory ถูกเรียกครั้งแรก (12.3)
กลุ่มที่เข้าไว้เป็นของคนเก่าคนใหม่อาจดูแลสถานีคนละชุด
เหตุการณ์ที่กรองตามผู้ใช้จะยังส่งของคนเก่าถ้า hub ใช้ Clients.User(userId)

ข้อแรกสำคัญที่สุดในแง่ความปลอดภัย — เป็นรูปแบบเดียวกับที่ 8.8 ยืนยันว่าออกจากระบบต้องล้างทุกอย่างที่ผูกกับผู้ใช้

2. ไม่ใช่ SessionStore และไม่ใช่ SharedRealtimeClient — ให้ shell ที่ถือทั้งสองตัวเป็นคนต่อ ตามหลักใน 9.15 ที่ห้ามการพึ่งพาที่วนกลับ

// ใน shell ที่ถือทั้งสองอย่าง
sessionStore.addListener(() {
  if (sessionStore.status == AuthStatus.signedOut) {
    unawaited(hubs.disconnectAll());
  }
});

แต่วิธีที่สะอาดกว่าและใช้ได้จริงมากกว่าคือผูกอายุของการเชื่อมต่อไว้กับ shell ที่มีเฉพาะตอนเข้าสู่ระบบแล้ว — ออกจากระบบทำให้ shell ถูกทิ้ง store ถูกทิ้ง และ release() ถูกเรียกตามลำดับโดยอัตโนมัติ (12.8 §6)

AuthStatus.signedIn => WarehouseShell(...)   // สร้าง hub + store ข้างใน
AuthStatus.signedOut => LoginScreen(...)     // shell ถูกทิ้งทั้งก้อน

ไม่ต้องเขียนโค้ดจัดการการออกจากระบบเลยแม้แต่บรรทัดเดียว และเป็นไปไม่ได้ที่จะลืม — วิธีที่ดีที่สุดในการไม่ลืมทำอะไร คือทำให้ไม่มีอะไรต้องจำ

9. สรุป

  • นับผู้ใช้งาน ต่อจริงเฉพาะคนแรก ตัดจริงเฉพาะคนสุดท้าย
  • กันตัวนับติดลบ ไม่งั้นจอถัดไปจะเงียบตลอดกาล
  • ให้ acquire กับ release อยู่ในคลาสเดียวกัน จะได้เห็นว่าจับคู่ครบ
  • การหน่วงก่อนตัดคุ้มเมื่อผู้ใช้สลับจอบ่อย แต่อย่าเพิ่มตั้งแต่แรก
  • ผูกอายุการเชื่อมต่อไว้กับ shell ดีกว่าเขียนโค้ดจัดการการออกจากระบบ

บทหน้า: เหตุการณ์กับการโหลดใหม่ — อันไหนเชื่อได้แค่ไหน
