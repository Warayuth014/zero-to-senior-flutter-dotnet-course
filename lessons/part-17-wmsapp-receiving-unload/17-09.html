1. เจ็ดคลาสของโดเมน

คลาสมาจาก endpointใช้ทำอะไร

2. สามระดับของการป้องกันในโดเมนนี้

เอาหลักจาก 16.8 มาใช้กับไฟล์นี้จริง

factory POItem.fromJson(Map<String, dynamic> json) => POItem(
  id: json['id'],                              // ← ไม่ป้องกัน
  partId: json['partId'],                      // ← ไม่ป้องกัน
  owner: json['owner'],                        // ← ไม่ป้องกัน
  brand: json['brand'],                        // ← ไม่ป้องกัน
  itemDesc: json['itemDesc'],                  // ← ไม่ป้องกัน
  imageUrl: json['imageUrl'],                  // nullable อยู่แล้ว
  serialRequire: json['serialRequire'] ?? false,
  qtyOrdered: json['qtyOrdered'],              // ← ไม่ป้องกัน
  qtyReceived: json['qtyReceived'],            // ← ไม่ป้องกัน
  qtyRemaining: json['qtyRemaining'] ?? 0,
  status: json['status'],                      // ← ไม่ป้องกัน
  condition: json['condition'] ?? 'FG',
  lotNumber: json['lotNumber'],
  expiredDate: json['expiredDate'],
);
ฟิลด์ที่ไม่ป้องกันถ้าไม่มีค่าจะเกิดอะไร
id, partIdTypeError — สมเหตุสมผล ไม่มีรหัสก็ไม่มีความหมาย
owner, brand, itemDescTypeError — ไม่สมเหตุสมผล ข้อมูลนี้เป็นแค่คำอธิบาย
qtyOrdered, qtyReceivedTypeError — ก้ำกึ่ง
statusTypeError — ก้ำกึ่ง
แถวที่สองคือจุดที่เปราะที่สุด — สินค้าที่ยังไม่ได้กรอกยี่ห้อหรือคำอธิบายจะทำให้ทั้ง PO เปิดไม่ได้ ทั้งที่ข้อมูลนั้นแค่ใช้แสดงบนจอ
ผู้ใช้เห็นแค่ “เกิดข้อผิดพลาด: type 'Null' is not a subtype of type 'String'” และไม่มีใครรู้ว่าสาเหตุคือสินค้าตัวเดียวใน 30 ตัวที่ไม่มียี่ห้อ

แก้ได้ทันทีโดยไม่กระทบใคร

owner: json['owner'] ?? '',
brand: json['brand'] ?? '',
itemDesc: json['itemDesc'] ?? '',

ค่าว่างบนหน้าจอ ดีกว่าทั้งใบเปิดไม่ได้ — และผู้ใช้เห็นได้ทันทีว่าสินค้าตัวไหนข้อมูลไม่ครบ

เทียบกับคลาสอื่นในไฟล์เดียวกันที่ป้องกันมากกว่า

factory PartLine.fromJson(Map<String, dynamic> json) => PartLine(
  lineId: json['lineId'],
  poId: json['poId'],
  condition: json['condition'] ?? 'FG',        // ← มีค่าเริ่มต้น
  lots: (json['lots'] as List? ?? [])          // ← กัน null
      .map((l) => PartLot.fromJson(l))
      .toList(),
);

ความไม่สม่ำเสมอนี้เป็นร่องรอยว่าไฟล์ถูกเขียนต่างเวลากัน

3. ค่าเริ่มต้นที่ตัดสินใจแทนผู้ใช้

condition: json['condition'] ?? 'FG',

ถ้าเซิร์ฟเวอร์ไม่ส่งสภาพมา แอปเดาว่าเป็น FG (ของดี)

ถ้าเดาถูกถ้าเดาผิด
ทำงานปกติของ PW ถูกรับเข้าเป็น FG

ผลของการเดาผิด: ของที่ต้องแปรรูปก่อนถูกส่งไปขายเลย และของสองสภาพถูกผูกลงพาเลทเดียวกันได้ ทั้งที่กฎห้าม

เทียบกับค่าเริ่มต้นตัวอื่นในไฟล์เดียวกัน
ค่าเริ่มต้นเดาผิดแล้วเป็นอย่างไรปลอดภัยไหม
serialRequire ?? falseของที่ต้องสแกน S/N ถูกรับด้วยจำนวนไม่
qtyReceived ?? 0คิดว่ายังไม่เคยรับก้ำกึ่ง
condition ?? 'FG'ของเสียถูกรับเป็นของดีไม่
autoClosed ?? falseไม่ปิด session ทั้งที่ควรปิดปลอดภัย
สองข้อแรกคือค่าเริ่มต้นที่ผ่อนปรน — เดาผิดแล้วปล่อยของผ่านไป ส่วนข้อสุดท้ายเป็นค่าเริ่มต้นที่เข้มงวด — เดาผิดแล้วผู้ใช้แค่ต้องทำเพิ่ม

หลักที่ควรใช้: เมื่อไม่รู้ ให้เดาไปทางที่ผู้ใช้ต้องทำงานเพิ่ม ไม่ใช่ทางที่ระบบปล่อยผ่าน

// ปลอดภัยกว่า — ไม่รู้สภาพ = ไม่ให้รับ ต้องไปแก้ข้อมูลก่อน
final String? condition;
...
condition: json['condition'],       // ไม่เดา
...
bool get canReceive => condition != null;

แต่ในทางปฏิบัติต้องดูก่อนว่าฝั่ง .NET มีทางที่จะไม่ส่ง condition มาจริงไหม — ถ้าคอลัมน์นั้นเป็น NOT NULL ในฐานข้อมูล ค่าเริ่มต้นนี้ไม่มีวันถูกใช้และไม่ต้องแตะ

นี่คือคำถามที่ต้องถามทีมหลังบ้าน ไม่ใช่เดาเอง

4. ฟิลด์ที่มีเฉพาะบางกรณี

class ReceiptLineResponse {
  // ── ข้อมูลของรายการที่รับ (มีเสมอ) ──
  final int lineId;
  final String partId;
  ...
  final String poItemStatus;
  final String message;

  // ── ผลของการผูก Pallet (ถ้า scan-part ส่ง palletId มาด้วย) ──
  final String? palletId;
  final bool autoClosed;
  final String? poStatus;
  final String? closeMessage;
  final String? palletError;
}

ห้าฟิลด์ล่างมีค่าเฉพาะเมื่อคำขอส่ง palletId ไปด้วย — คอมเมนต์บอกไว้ชัดเจน ซึ่งช่วยมาก

แต่ชนิดข้อมูลไม่ได้บังคับความสัมพันธ์

ชุดค่าที่เป็นไปได้ตามชนิดเกิดจริงไหม
palletId มีค่า, palletError เป็น nullผูกสำเร็จ
palletId เป็น null, palletError มีค่าผูกไม่สำเร็จ
ทั้งสองเป็น nullไม่ได้ส่งพาเลทไป
ทั้งสองมีค่าไม่ควรเกิด แต่ชนิดไม่ได้ห้าม

โค้ดที่ใช้จึงต้องเลือกว่าจะเชื่ออันไหนก่อน

if (line.palletError != null) {
  // ถือว่าล้มเหลว แม้ palletId จะมีค่า
} else {
  _storeAssignedLine(line, line.palletId!);   // ← ใช้ ! เพราะเชื่อว่ามี
}

เครื่องหมาย ! ตรงนี้คือการยืนยันว่า “ถ้าไม่มี error แปลว่าต้องมี palletId” ซึ่งเป็นข้อตกลงที่ไม่ได้เขียนไว้ที่ไหน — ถ้าฝั่ง .NET เปลี่ยนแล้วส่งทั้งสองเป็น null แอปจะพังตรงนี้

เขียนให้ปลอดภัยได้โดยไม่เปลี่ยนโมเดล

final palletId = line.palletId;
if (line.palletError != null || palletId == null) {
  // ไม่มีพาเลท = ยังต้องผูก ไม่ว่าจะด้วยเหตุผลใด
  setState(() => _resumedPendingLines.insert(0, line));
} else {
  _storeAssignedLine(line, palletId);
}

5. ฝั่ง API — สี่เมธอด

extension ReceivingApi on ApiService {
  Future<ApiResult<POResponse>> getPO(String poId) async {
    final response = await _get('/receiving/po/$poId');
    if (!response.success) return ApiResult.error(response.error);   // ← ทิ้ง statusCode
    return ApiResult.success(POResponse.fromJson(response.data!));
  }

ทั้งหกจุดในไฟล์นี้ทิ้ง statusCode ตามที่ 16.7 นับไว้ — โดเมนนี้ไม่มีจุดไหนส่งต่อเลย

ผลที่ตามมาชัดเจนที่สุดตรงการค้นหา PO

เซิร์ฟเวอร์ตอบควรบอกผู้ใช้ตอนนี้บอก
404“ไม่พบ PO นี้ — ตรวจว่าสแกนถูกใบ”ข้อความจากเซิร์ฟเวอร์
500“ระบบมีปัญหา แจ้งทีมไอที”ข้อความจากเซิร์ฟเวอร์
ไม่มีเน็ต“เดินไปที่ที่มีสัญญาณ”“ไม่สามารถเชื่อมต่อ server ได้”

สามกรณีนี้ผู้ใช้ต้องทำคนละอย่าง แต่ทั้งหมดขึ้นเป็นกล่องแดงหน้าตาเหมือนกัน

showErrorDialog(context, message: result.error!);

แก้ขั้นแรกได้ด้วยการส่ง statusCode ต่อ แล้วจอเลือกข้อความเอง

if (result.isNotFound) {
  showErrorDialog(context,
    title: 'ไม่พบ PO',
    message: 'ไม่มี PO "$poId" ในระบบ\nตรวจว่าสแกนถูกใบ หรือ PO ยังไม่ถูกสร้าง');
  return;
}
showErrorDialog(context, message: result.error!);

6. เมธอดที่ประกอบ path ต่างกันสามแบบ

ในไฟล์เดียวมีทั้งสามแบบที่ 16.7 อธิบาย

// แบบต่อสตริง
final response = await _get('/receiving/po/$poId');

// แบบใช้ Uri — สองเมธอดที่ยิง path เดียวกัน
final path = Uri(
  path: '/receiving/validate-serial',
  queryParameters: {'partId': partId, 'serialNo': serialNo},
).toString();

final path = Uri(
  path: '/receiving/validate-serial',
  queryParameters: {'partId': partId},
).toString();

สองอันหลังใช้ Uri ซึ่งปลอดภัย — สำคัญมากเพราะ S/N อาจมีอักขระพิเศษ

S/Nต่อสตริงตรง ๆผ่าน Uri
SN-4471ทำงานได้ทำงานได้
SN 4471 (มีช่องว่าง)path พังเข้ารหัสเป็น SN%204471
SN/4471กลายเป็นคนละ pathเข้ารหัสถูก
SN#4471ทุกอย่างหลัง # หายไปเข้ารหัสถูก

ส่วน getPO ที่ต่อสตริงตรง ๆ ยังปลอดภัยตราบใดที่รหัส PO เป็นตัวอักษรกับตัวเลขและขีดกลาง — แต่ไม่มีอะไรรับประกันไว้

// ปลอดภัยกว่าและยาวขึ้นแค่นิดเดียว
final response = await _get('/receiving/po/${Uri.encodeComponent(poId)}');

อีกจุดที่ทำถูกคือการแปลงรายการ

Future<ApiResult<List<PendingPalletLine>>> getPendingPalletLines() async {
  final response = await _get('/receiving/pending-pallet-lines');
  if (!response.success) return ApiResult.error(response.error);

  final list = (response.data!['lines'] as List)      // ← ไม่กัน null
      .map((item) => PendingPalletLine.fromJson(item))
      .toList();
  return ApiResult.success(list);
}

ตรงนี้ไม่กัน null — ถ้าเซิร์ฟเวอร์ไม่ส่ง lines มา (เช่นตอนไม่มีของค้าง) จะพัง

final list = (response.data!['lines'] as List? ?? [])

เทียบกับ upload_api.dart ที่ใช้ items และก็ไม่กันเหมือนกัน ส่วน picking_api.dart กันบางเมธอด — ความไม่สม่ำเสมอเดียวกับที่ 16.7 ชี้ไว้

7. ข้อผิดพลาดที่พบบ่อย

เข้าใจผิดที่จริง
ฟิลด์ที่เป็นแค่คำอธิบายไม่ทำให้พังbrand ที่หายไปทำให้ทั้ง PO เปิดไม่ได้
ค่าเริ่มต้นปลอดภัยเสมอcondition ?? 'FG' ปล่อยของเสียผ่าน
ฟิลด์ nullable สองตัวเป็นอิสระต่อกันpalletId กับ palletError ผูกกันแต่ชนิดไม่ได้บังคับ
ต่อ path ด้วยสตริงพอS/N ที่มี # ทำให้พารามิเตอร์หายทั้งหมด
ทุก endpoint ส่ง array มาเสมอบางตัวอาจไม่ส่งเมื่อไม่มีข้อมูล

8. แบบฝึกหัด

เขียนชุดเทสต์ให้โมเดลของโดเมนนี้ โดยไม่แก้โค้ดเดิม

  1. เทสต์อะไรได้บ้าง และแต่ละข้อจับปัญหาอะไร
  2. เขียนเทสต์ที่จับปัญหาซึ่งบทนี้ชี้ไว้ อย่างน้อยสามข้อ
  3. ถ้าเทสต์ล้ม แต่โปรเจกต์นี้ทำงานอยู่ในคลังปกติ จะรายงานอย่างไร
แนวคำตอบ

ข้อ 1

เทสต์จับอะไร
JSON ครบทุกฟิลด์ → แปลงถูกตาข่ายพื้นฐาน
ฟิลด์ที่เป็นคำอธิบายหาย → ?บันทึกว่าตอนนี้พัง

import 'package:flutter_test/flutter_test.dart';

/// JSON ที่ครบทุกฟิลด์ — ใช้เป็นฐาน แล้วแต่ละเทสต์เอาออกทีละอย่าง
Map<String, dynamic> goodItem() => {
  'id': 101,
  'partId': 'PT-9821',
  'owner': 'ACME',

  'itemDesc': 'มอเตอร์ 3 เฟส',

  'qtyReceived': 5,
  'qtyRemaining': 15,

  'condition': 'FG',

void main() {
  group('POItem.fromJson', () {
    test('JSON ครบ แปลงได้ถูกต้อง', () {
      final item = POItem.fromJson(goodItem());
      expect(item.partId, 'PT-9821');
      expect(item.serialRequire, isTrue);

    });

    test('ไม่มี brand → ทั้งใบเปิดไม่ได้', () {
      // บันทึกพฤติกรรมปัจจุบัน: ฟิลด์ที่เป็นแค่คำอธิบายทำให้ทั้ง PO พัง
      // ถ้าวันหนึ่งแก้เป็น ?? '' เทสต์นี้จะล้ม แล้วให้เปลี่ยนเป็น
      //   expect(POItem.fromJson(json).brand, isEmpty);
      final json = goodItem()..remove('brand');
      expect(() => POItem.fromJson(json), throwsA(isA<TypeError>()));
    });

    test('ไม่มี condition → เดาว่าเป็น FG', () {

    });

    test('ไม่มี serialRequire → เดาว่าไม่ต้องสแกน S/N', () {
      final json = goodItem()..remove('serialRequire');
      expect(POItem.fromJson(json).serialRequire, isFalse);
    });
  });

  group('ReceiptLineResponse.fromJson', () {
    test('ผูก Pallet ไม่สำเร็จ ต้องอ่าน palletError ได้', () {
      final line = ReceiptLineResponse.fromJson({
        'lineId': 5,
        'partId': 'PT-9821',
        'owner': 'ACME',
        'brand': 'BOSCH',
        'itemDesc': 'มอเตอร์',
        'qtyOrdered': 20,
        'qtyReceived': 3,
        'condition': 'FG',
        'poItemStatus': 'PARTIAL',
        'message': 'ok',
        'palletError': 'Pallet PLT-001 มีสินค้า PW อยู่แล้ว',
      });

      // สัญญาที่ ScanPartScreen พึ่งพา (17.7)
      expect(line.palletError, isNotNull);
      expect(line.palletId, isNull);
    });
  });

  group('PartLinesResponse.fromJson', () {
    test('รายการซ้อนสามชั้นแปลงครบ', () {
      final response = PartLinesResponse.fromJson({
        'partId': 'PT-9821',
        'serialRequire': true,
        'lines': [
          {
            'lineId': 1,
            'poId': 'PO-2026-001',
            'condition': 'FG',
            'lots': [
              {'id': 101, 'lotNumber': 'LOT-A', 'qtyOrdered': 20, 'qtyReceived': 5},
              {'id': 102, 'lotNumber': 'LOT-B', 'qtyOrdered': 10, 'qtyReceived': 0},
            ],
          },
        ],
      });

      expect(response.lines, hasLength(1));
      expect(response.lines.first.lots, hasLength(2));
      expect(response.lines.first.lots[1].lotNumber, 'LOT-B');
    });

    test('ไม่มี lots → ได้รายการว่าง ไม่พัง', () {
      final response = PartLinesResponse.fromJson({
        'partId': 'PT-9821',
        'lines': [
          {'lineId': 1, 'poId': 'PO-1', 'condition': 'FG'},
        ],
      });
      expect(response.lines.first.lots, isEmpty);
    });
  });
}

สังเกตเทสต์ที่สองใน group แรก — เทสต์ที่ยืนยันว่าตอนนี้พัง ไม่ใช่เทสต์ที่ยืนยันว่าถูก

เทสต์แบบนี้มีค่าเพราะบันทึกพฤติกรรมปัจจุบันไว้ ทำให้เห็นทันทีเมื่อมีคนแก้ และคอมเมนต์บอกว่าถ้าล้มควรทำอะไร

ข้อ 3 — รายงานอย่างไร

ผิดที่สุดคือรายงานว่า “โมเดลมีบั๊ก 3 ข้อ” เพราะแอปทำงานอยู่จริงในคลังทุกวัน

ที่ควรรายงานคือเงื่อนไขที่จะทำให้พัง พร้อมความน่าจะเป็น

ข้อจะเกิดเมื่อตรวจได้จาก
brand หายฐานข้อมูลยอมให้คอลัมน์นี้ว่างถามทีม .NET หรือดู schema
condition หายเหมือนกันเหมือนกัน
serialRequire หายเหมือนกันเหมือนกัน

ถ้าคอลัมน์เหล่านี้เป็น NOT NULL ทั้งหมด ทั้งสามข้อไม่มีวันเกิด และควรบันทึกไว้ในเทสต์ว่า “ป้องกันไว้เพราะฝั่ง .NET รับประกันแล้ว” แทนที่จะแก้โค้ด

ถ้ามีคอลัมน์ไหน NULL ได้ นั่นคืองานจริงที่ต้องแก้ และเทสต์ที่เขียนไว้จะกลายเป็นตัวยืนยันว่าแก้แล้ว

ข้อสรุปที่สำคัญ: เทสต์บอกได้ว่า “ถ้าข้อมูลเป็นแบบนี้ จะเกิดอะไร” แต่บอกไม่ได้ว่า “ข้อมูลจะเป็นแบบนั้นไหม” — คำตอบครึ่งหลังอยู่ที่ฝั่งเซิร์ฟเวอร์ และต้องไปถาม ไม่ใช่เดาจากโค้ดฝั่งแอป

9. สรุป

  • เจ็ดคลาส และสามคลาสกลางซ้อนกันสามชั้นสะท้อนความจริงว่า Part หนึ่งมีหลายสภาพหลายล็อต
  • ฟิลด์ที่เป็นแค่คำอธิบาย (owner, brand, itemDesc) ไม่ป้องกัน ทำให้ทั้ง PO เปิดไม่ได้
  • ค่าเริ่มต้นแบ่งได้สองแบบ — ผ่อนปรน (อันตราย) กับเข้มงวด (ปลอดภัย)
  • condition ?? 'FG' และ serialRequire ?? false เป็นแบบผ่อนปรนทั้งคู่
  • เมื่อไม่รู้ ให้เดาไปทางที่ผู้ใช้ต้องทำเพิ่ม ไม่ใช่ทางที่ระบบปล่อยผ่าน
  • palletId กับ palletError ผูกกันแต่ชนิดไม่ได้บังคับ — ! เป็นข้อตกลงที่ไม่ได้เขียนไว้
  • ทั้งหกจุดในโดเมนนี้ทิ้ง statusCode — แยก 404 จาก 500 ไม่ได้
  • สอง endpoint ใช้ Uri ถูกต้อง แต่ getPO ต่อสตริงตรง ๆ
  • เทสต์ที่ยืนยันว่า “ตอนนี้พัง” มีค่าพอ ๆ กับเทสต์ที่ยืนยันว่าถูก

บทหน้า: ข้ามไปฝั่ง Unload — จอ Load to Basket และกล่องสองขั้นตอนที่ซับซ้อนที่สุดในโดเมน
