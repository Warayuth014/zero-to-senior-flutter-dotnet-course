2. ฟิลด์ที่คำนวณเองสองที่

// unload_models.dart
class UnloadItem {
  final List<String> serialNumbers;
  ...
  bool get serialRequire => serialNumbers.isNotEmpty;
}
// basket_models.dart
class UnloadedItem {
  final List<String> serialNumbers;
  final int qtyRemaining;
  ...
  bool get isDone => qtyRemaining <= 0;
  bool get serialRequire => serialNumbers.isNotEmpty;
}

ทั้งสองคลาสคำนวณ serialRequire เอง ต่างจาก receiving ที่รับมาจากเซิร์ฟเวอร์

โดเมนserialRequire มาจากผลเมื่อข้อมูลไม่ครบ
receivingฟิลด์จริง json['serialRequire'] ?? falseเดาว่าไม่ต้อง
unload / basketคำนวณจาก serialNumbers.isNotEmptyเดาว่าไม่ต้อง

ทั้งสองแบบล้มไปทาง “ไม่ต้องสแกน” เมื่อข้อมูลขาด ซึ่งเป็นค่าเริ่มต้นแบบผ่อนปรนตามที่ 17.9 อธิบาย

แต่มีความต่างที่สำคัญ

// receiving — สองเรื่องแยกกัน
serialRequire: true          // "ของนี้ต้องสแกน S/N"
serialNumbers: []            // "แต่ยังไม่รู้ว่ามีตัวไหนบ้าง"   ← เป็นไปได้

// unload — เรื่องเดียวกัน
serialNumbers: []            // "ไม่มี S/N" = "ไม่ต้องสแกน"     ← แยกไม่ได้
สถานการณ์receiving จับได้ไหมunload จับได้ไหม
ของต้องมี S/N และเซิร์ฟเวอร์ส่งรายชื่อมาครบได้ได้
ของไม่ต้องมี S/Nได้ได้
ของต้องมี S/N แต่เซิร์ฟเวอร์ไม่ส่งรายชื่อมาได้ — serialRequire ยังเป็น true แล้วผู้ใช้สแกนไม่ผ่านไม่ได้ — ผ่านไปด้วยจำนวนเลย

แถวสุดท้ายคือช่องโหว่จริง — ของที่ควรถูกติดตามทีละชิ้น จะถูกถ่ายลงด้วยจำนวนโดยไม่มีใครรู้

แก้ได้โดยขอให้เซิร์ฟเวอร์ส่งฟิลด์มาตรง ๆ

class UnloadItem {
  final bool serialRequire;                     // จากเซิร์ฟเวอร์
  final List<String> serialNumbers;

  factory UnloadItem.fromJson(Map<String, dynamic> json) => UnloadItem(
    // ถ้าเซิร์ฟเวอร์ยังไม่ส่งฟิลด์นี้ ให้เดาจากรายชื่อไปก่อน
    serialRequire: json['serialRequire'] ?? (json['serialNumbers'] as List?)?.isNotEmpty ?? false,
    ...
  );

  /// รายชื่อไม่ครบทั้งที่ต้องสแกน = ข้อมูลผิด ต้องหยุด
  bool get serialDataIncomplete => serialRequire && serialNumbers.isEmpty;
}

เขียนแบบนี้ทำงานได้ทั้งก่อนและหลังฝั่ง .NET เพิ่มฟิลด์ และ serialDataIncomplete ทำให้กรณีที่สามมองเห็นได้แทนที่จะเงียบ

3. isDone ที่ทนกว่าที่จำเป็น

bool get isDone => qtyRemaining <= 0;

ใช้ <= ไม่ใช่ == — เผื่อกรณีที่ค่าติดลบ

qtyRemaining== 0<= 0
5ยังไม่เสร็จยังไม่เสร็จ
0เสร็จเสร็จ
-2ยังไม่เสร็จเสร็จ

ค่าติดลบเกิดได้ถ้ามีการโหลดเกิน หรือข้อมูลจากสองแหล่งไม่ตรงกัน — <= ทำให้รายการนั้นถูกซ่อนแทนที่จะค้างอยู่ตลอดไป

เป็นการเขียนที่ป้องกันไว้โดยไม่ต้องอธิบาย และค่าใช้จ่ายเป็นศูนย์

แต่ควรระวังว่าการซ่อนไม่ใช่การแก้ — ถ้ามีค่าติดลบจริง ควรมีใครสักคนรู้

bool get isDone => qtyRemaining <= 0;

/// จำนวนติดลบ = ข้อมูลไม่สอดคล้อง ควรแจ้งทีมหลังบ้าน
bool get hasInvalidQty => qtyRemaining < 0;

4. ระดับการป้องกันที่ต่างกันสามไฟล์

เทียบการอ่านฟิลด์คำอธิบายในสามโดเมน

// receiving_models.dart — POItem
owner: json['owner'],              // พังถ้าไม่มี
brand: json['brand'],              // พังถ้าไม่มี
itemDesc: json['itemDesc'],        // พังถ้าไม่มี

// unload_models.dart — UnloadItem
owner: json['owner'],              // พังถ้าไม่มี
brand: json['brand'],              // พังถ้าไม่มี
itemDesc: json['itemDesc'],        // พังถ้าไม่มี

// basket_models.dart — UnloadedItem
owner: json['owner'] ?? '',        // ← กันไว้
brand: json['brand'] ?? '',        // ← กันไว้
itemDesc: json['itemDesc'] ?? '',  // ← กันไว้

ไฟล์ที่สามกันไว้ ส่วนสองไฟล์แรกไม่กัน — ทั้งที่เป็นข้อมูลชนิดเดียวกันของสินค้าตัวเดียวกัน

นี่คือหลักฐานว่าไฟล์ถูกเขียนต่างเวลากันและไม่มีข้อตกลงเขียนไว้
คนเขียน basket_models.dart รู้ว่าควรกัน — แต่ไม่มีใครกลับไปแก้สองไฟล์แรก เพราะไม่มีอะไรบอกว่ามีความไม่สม่ำเสมออยู่
เทสต์ตามที่ 17.9 เสนอจะทำให้เห็นทันที เพราะเทสต์เดียวกันจะผ่านที่หนึ่งและล้มที่อีกที่

ส่วนการอ่านรายการก็ต่างกันอีก

// unload_models.dart — UnloadSession
items: (json['items'] as List).map((i) => UnloadItem.fromJson(i)).toList(),
//                        ↑ ไม่กัน null

// basket_models.dart — UnloadedItemsResponse
items: (json['items'] as List).map((e) => UnloadedItem.fromJson(e)).toList(),
//                        ↑ ไม่กันเหมือนกัน

// แต่ฟิลด์อื่นในไฟล์เดียวกันกัน
confirmedLineIds: (json['confirmedLineIds'] as List? ?? []).map((e) => e as int).toList(),
serialNumbers: (json['serialNumbers'] as List?)?.map((e) => e as String).toList() ?? [],

รูปแบบที่เจอคือ ฟิลด์หลักไม่กัน ฟิลด์เสริมกัน — สมเหตุสมผลถ้าคิดว่าฟิลด์หลักต้องมีเสมอ แต่ผลคือคำตอบที่ผิดรูปทำให้พังโดยไม่มีข้อความที่ช่วยอะไร

if (!response.success) return ApiResult.error(response.error);

ตรงกับตารางใน 16.7 — basket_api.dart อยู่ในกลุ่มที่ทำครบ ส่วน unload_api.dart อยู่ในกลุ่มที่ทิ้ง

อีกจุดที่ทั้งสองไฟล์ทำเหมือนกัน

// unload_api.dart
final body = <String, dynamic>{
  ...
  // Part ไม่ require serial ต้องส่ง null/[] เสมอ (ฝั่ง backend รองรับทั้งคู่)
  'serialNumbers': serialNumbers ?? [],

if (qtyUnloaded != null) body['qtyUnloaded'] = qtyUnloaded;
// basket_api.dart
final response = await _post('/basket/load', {
  ...
  // Part ไม่ require serial ต้องส่ง null/[] เสมอ (ฝั่ง backend รองรับทั้งคู่)
  'serialNumbers': serialNumbers ?? [],
});

คอมเมนต์เดียวกันคำต่อคำในสามไฟล์ (รวม receiving_api.dart) — เป็นข้อตกลงกับฝั่ง .NET ที่ถูกบันทึกไว้ในที่ที่คนอ่านโค้ดจะเห็น

ทำถูกทำไม
บันทึกข้อตกลงไว้ในคอมเมนต์คนที่มาแก้ทีหลังจะไม่เปลี่ยนเป็น if (serialNumbers != null)
ส่ง [] แทน nullฝั่ง .NET ไม่ต้องแยกสองกรณี

สังเกตความต่างของการใส่ฟิลด์ที่อาจไม่มี

// unload — ใส่เฉพาะเมื่อมี
if (qtyUnloaded != null) body['qtyUnloaded'] = qtyUnloaded;

// basket — ใส่เสมอ แม้เป็น null
'lotNumber': lotNumber,

สองแบบนี้ต่างกันฝั่งเซิร์ฟเวอร์ — แบบแรกฟิลด์ไม่ปรากฏใน JSON เลย แบบที่สองปรากฏเป็น null

ถ้า .NET ใช้ไม่ส่งฟิลด์ส่งเป็น null
ค่าเริ่มต้นของ propertyได้ค่าเริ่มต้นได้ null
[Required]ไม่ผ่าน validationไม่ผ่าน validation
ตรวจ == nullเหมือนกันเหมือนกัน

ในกรณีทั่วไปได้ผลเหมือนกัน แต่ถ้า .NET ใช้ JsonPatchDocument หรือมีตรรกะที่แยก “ไม่ส่ง” จาก “ส่งค่าว่าง” จะต่างกันมาก — เป็นเรื่องที่ต้องยืนยันกับทีมหลังบ้าน ไม่ใช่เดา

8. แบบฝึกหัด

ผู้ใช้แจ้งว่า “สินค้า PT-5502 ที่มี S/N พอไป Load เข้า Basket มันไม่ให้สแกน S/N เลย ให้กรอกจำนวนอย่างเดียว”

  1. ไล่หาสาเหตุจากโค้ดที่บทนี้อ่านมา
  2. ยืนยันสมมติฐานอย่างไรโดยไม่ต้องแก้โค้ด
  3. แก้อย่างไรให้ปัญหานี้มองเห็นได้ในอนาคต
แนวคำตอบ

ข้อ 1 — ไล่หาสาเหตุ

เริ่มจากจุดที่ตัดสินใจว่าจะแสดงช่อง S/N ไหม

// load_to_basket_screen.dart
final requiresSerial = item.serialRequire;
var showBasketStep = !requiresSerial;      // ← true = ข้ามขั้น S/N ไปเลย

ตามต่อไปที่โมเดล

// basket_models.dart
bool get serialRequire => serialNumbers.isNotEmpty;

แปลว่า serialNumbers ที่ได้มาว่างเปล่า

serialNumbers: (json['serialNumbers'] as List?)?.map((e) => e as String).toList() ?? [],
สาเหตุที่เป็นไปได้ตรวจอย่างไร
เซิร์ฟเวอร์ไม่ส่ง serialNumbers มาเลยดู JSON จริง
ส่งมาเป็น []ดู JSON จริง
ส่งมาแต่ชื่อคีย์ต่าง (เช่น SerialNumbers)ดู JSON จริง — และ 16.8 เตือนเรื่องนี้ไว้
S/N ทั้งหมดถูก Load ไปแล้วดู qtyRemaining

สาเหตุที่สี่น่าสนใจ — ถ้า S/N ที่เหลือถูกโหลดไปหมดแล้ว รายการนั้นควรมี qtyRemaining == 0 และถูกกรองออกตั้งแต่แรก

.where((i) => i.partId == partId && !i.isDone)

จึงตัดสาเหตุที่สี่ออกได้ เหลือสามข้อแรกที่ทั้งหมดเป็นเรื่องของ JSON

ข้อ 2 — ยืนยันโดยไม่แก้โค้ด

# ยิง endpoint ตรงจากเครื่องที่ต่อวงเดียวกัน
curl http://192.168.1.50:5000/api/basket/unloaded-items

แล้วดูรายการของ PT-5502

ถ้าเห็นสรุปได้ว่า
ไม่มีคีย์ serialNumbers เลยฝั่ง .NET ไม่ได้ส่ง — งานอยู่หลังบ้าน
"serialNumbers": []ส่งแต่ว่าง — ต้องดูว่าทำไม
"SerialNumbers": [...]ชื่อคีย์เป็น PascalCase — endpoint นี้ไม่ผ่านตัวแปลงมาตรฐาน

กรณีที่สามเป็นสิ่งที่ 16.8 เตือนไว้ และจะทำให้ทุกฟิลด์ของรายการนั้นเป็นค่าเริ่มต้นหมด ไม่ใช่แค่ S/N — ตรวจได้โดยดูว่า owner กับ brand ว่างด้วยไหม

ถ้าว่างทั้งหมด ยืนยันได้ทันทีโดยไม่ต้องดู JSON ด้วยซ้ำ

ข้อ 3 — ทำให้มองเห็นได้

ปัญหานี้เงียบเพราะแอปไม่มีทางแยก “ไม่ต้องมี S/N” จาก “ควรมีแต่ไม่ได้มา”

class UnloadedItem {
  final bool? serialRequire;                  // จากเซิร์ฟเวอร์ — null = ไม่ได้ส่งมา
  final List<String> serialNumbers;

  factory UnloadedItem.fromJson(Map<String, dynamic> json) => UnloadedItem(
    serialRequire: json['serialRequire'] as bool?,
    serialNumbers: (json['serialNumbers'] as List?)
            ?.map((e) => e as String).toList() ?? [],
    ...
  );

  /// ต้องสแกน S/N ไหม — ใช้ค่าจากเซิร์ฟเวอร์ก่อน ถ้าไม่มีค่อยเดา
  bool get needsSerial => serialRequire ?? serialNumbers.isNotEmpty;

  /// บอกว่าต้องสแกน แต่ไม่มีรายชื่อมาให้ = ข้อมูลไม่ครบ
  bool get serialDataMissing => serialRequire == true && serialNumbers.isEmpty;
}

แล้วหยุดผู้ใช้ก่อนที่จะทำผิด

void _onScanPart() {
  ...
  final item = items.first;

  if (item.serialDataMissing) {
    showErrorDialog(
      context,

ข้อควรระวังก่อนทำ: ถ้าฝั่ง .NET ยังไม่ส่ง serialRequire มาเลย serialDataMissing จะเป็น false เสมอ และไม่มีอะไรเปลี่ยน

การแก้นี้จึงต้องทำสองฝั่ง — และการทำฝั่งแอปก่อนก็ยังคุ้ม เพราะพอฝั่ง .NET เพิ่มฟิลด์เมื่อไร แอปจะจับได้ทันทีโดยไม่ต้องปล่อยรุ่นใหม่

ข้อสรุปที่ใหญ่กว่า: ฟิลด์ที่คำนวณจากข้อมูลอื่น (serialRequire => serialNumbers.isNotEmpty) ประหยัดฝั่งเซิร์ฟเวอร์ แต่แลกมาด้วยการที่ข้อมูลขาดกับข้อมูลว่างแยกกันไม่ออก — ใช้ได้กับค่าที่ไม่สำคัญ แต่ไม่ควรใช้กับกฎที่ตัดสินว่าผู้ใช้ต้องทำอะไร

9. สรุป

  • เจ็ดคลาส 218 บรรทัด และมีสองคลาสที่ไม่มีใครเรียก
  • UnloadItem กับ UnloadedItem ต่างกันแค่ ed แต่คนละขั้นของ flow
  • serialRequire ถูกคำนวณจาก serialNumbers.isNotEmpty — แยกข้อมูลขาดจากไม่ต้องมีไม่ได้
  • receiving รับฟิลด์นี้มาตรง ๆ จึงจับกรณีข้อมูลขาดได้
  • isDone ใช้ <= เผื่อค่าติดลบ — ป้องกันไว้โดยไม่มีค่าใช้จ่าย
  • สามไฟล์โมเดลป้องกันฟิลด์คำอธิบายสามระดับต่างกัน
  • basket_api ส่ง statusCode ต่อ ส่วน unload_api ทิ้งทั้งสามเมธอด
  • คอมเมนต์เรื่อง serialNumbers ?? [] ปรากฏเหมือนกันสามไฟล์ — ข้อตกลงกับ .NET ที่บันทึกไว้ดี
  • “ไม่ส่งฟิลด์” กับ “ส่งเป็น null” อาจต่างกันฝั่งเซิร์ฟเวอร์
