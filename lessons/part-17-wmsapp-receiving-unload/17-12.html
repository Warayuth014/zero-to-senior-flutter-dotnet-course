4. สแกน Part แล้วเลือกให้อัตโนมัติ

// สแกน Part ID — แสดงทุก Lot ของ Part นั้นในลิสต์เสมออยู่แล้ว (ไม่ popup เลือก lot)
// ถ้ามีหลาย Lot ที่ยัง PENDING เลือก lot แรกที่เจอไปก่อน
// ผู้ใช้แตะการ์ด lot อื่นในลิสต์เพื่อสลับได้เอง
void _scanPart() {
  final partId = _partController.text.trim().toUpperCase();
  if (partId.isEmpty) return;

  final matches = _session?.items.where((i) => i.partId == partId).toList() ?? [];

  if (matches.isEmpty) {
    showErrorDialog(context, message: 'Part $partId ไม่อยู่ใน Pallet นี้');
    _partController.clear();
    return;
  }

  final pending = matches.where((i) => _partStatus[i.lineId] != 'CONFIRMED').toList();

  if (pending.isEmpty) {
    showWarningSnackbar(context, 'Part $partId ยืนยันไปแล้วทุก Lot');
    _partController.clear();
    return;
  }

  setState(() {
    _scannedLineId = pending.first.lineId;
    _scannedSerials.clear();
    _partController.clear();
  });
}

คอมเมนต์สามบรรทัดบนหัวอธิบายการตัดสินใจครบ — ทำไมไม่ popup และผู้ใช้จะสลับล็อตอย่างไร

สถานการณ์ผลลัพธ์วิธีบอกผู้ใช้
ไม่มี Part นี้บนพาเลทไม่เลือกอะไรกล่องแดง
มีแต่ยืนยันครบทุกล็อตแล้วไม่เลือกอะไรแถบเหลือง
มีล็อตที่ยังไม่ยืนยันเลือกล็อตแรกรายการนั้นถูกไฮไลต์

แถวกลางใช้แถบเหลืองไม่ใช่กล่องแดง เพราะไม่ใช่ความผิดพลาด — ผู้ใช้แค่สแกนของที่ทำไปแล้ว ซึ่งเกิดบ่อยเมื่อทำงานเร็ว

5. firstWhere ที่ไม่มีทางออก

UnloadItem? _currentScannedItem() {
  final lineId = _scannedLineId;
  if (lineId == null || _session == null) return null;
  return _session!.items.firstWhere((i) => i.lineId == lineId);
}

ตรวจ null สองตัวแล้ว แต่ firstWhere ไม่มี orElse

แต่ _selectLine() ที่ผู้ใช้แตะการ์ดเอง

7. ข้อผิดพลาดที่พบบ่อย

เข้าใจผิดที่จริง
getter ที่วนรายการช้าเกินไปที่ขนาดนี้ไม่รู้สึก และไม่มีทางไม่ตรง

สองฟิลด์ที่เปลี่ยนพร้อมกันเสมอต้องมีทั้งคู่ตัวหนึ่งควรเป็น getter

8. แบบฝึกหัด

ทีมอยากให้แผ่นนี้บอกได้ว่า “เหลืออีกกี่ชิ้น” ไม่ใช่แค่ “เหลืออีกกี่รายการ”

  1. ตอนนี้แถบความคืบหน้านับอะไร และต่างจากจำนวนชิ้นอย่างไร
  2. เพิ่มการนับชิ้นโดยไม่สร้างแหล่งความจริงใหม่
  3. ควรแสดงทั้งสองตัวเลขไหม หรือแทนที่
แนวคำตอบ

ข้อ 1

int get _confirmedCount => _partStatus.values.where((s) => s == 'CONFIRMED').length;
int get _totalCount => _partStatus.length;

นับรายการ (line) ไม่ใช่ชิ้น (qty)

พาเลทมีแถบบอกว่าความจริง
รายการ A: 2 ชิ้น (ยืนยันแล้ว)1 จาก 3 = 33%2 จาก 112 ชิ้น = 1.8%
รายการ B: 100 ชิ้น
รายการ C: 10 ชิ้น

ตัวเลข 33% ทำให้ผู้ใช้คิดว่าทำไปหนึ่งในสามแล้ว ทั้งที่ยังไม่ถึง 2%

บนพาเลทที่จำนวนชิ้นต่อรายการต่างกันมาก ตัวเลขนี้ทำให้ประเมินเวลาผิด

ข้อ 2 — เพิ่มโดยไม่สร้างแหล่งใหม่

ข้อมูลจำนวนชิ้นมีอยู่แล้วใน _session!.items ไม่ต้องเก็บเพิ่ม

/// จำนวนชิ้นทั้งหมดบนพาเลท — คำนวณจาก session เสมอ
int get _totalQty =>
    _session?.items.fold<int>(0, (sum, i) => sum + i.qty) ?? 0;

/// จำนวนชิ้นที่ยืนยันแล้ว — นับเฉพาะรายการที่สถานะ CONFIRMED
int get _confirmedQty =>
    _session?.items
        .where((i) => _partStatus[i.lineId] == 'CONFIRMED')
        .fold<int>(0, (sum, i) => sum + i.qty) ??
    0;

ทั้งสองอ่านจาก _session กับ _partStatus ซึ่งเป็นแหล่งเดิม — ไม่มีอะไรใหม่ให้ซิงก์

มีกับดักที่ต้องรู้: _reloadSession() ทำให้ items เปลี่ยนหลังยืนยันบางส่วน (17.11)
ถ้าผู้ใช้ยืนยัน 3 จาก 10 ชิ้น เซิร์ฟเวอร์จะสร้างรายการใหม่สำหรับ 7 ชิ้นที่เหลือ แล้วรายการเดิมกลายเป็น 3 ชิ้นที่ CONFIRMED
_totalQty จึงยังเท่าเดิม (3 + 7 = 10) และ _confirmedQty ได้ 3 — ถูกต้องพอดี เพราะคำนวณจากข้อมูลสด
ถ้าเก็บ _totalQty ไว้ตอนเปิด session แล้วไม่อัปเดต ตัวเลขจะเพี้ยนทันทีที่มีการยืนยันบางส่วน — นี่คือเหตุผลที่การคำนวณชนะการเก็บอีกครั้ง

ข้อ 3 — แสดงทั้งสองหรือแทนที่

ทางเลือกข้อดีข้อเสีย
แทนที่ด้วยจำนวนชิ้นตัวเลขเดียว ไม่สับสนเสียข้อมูลว่าเหลือกี่รายการให้เดินไปหยิบ
แสดงทั้งสองครบตัวเลขเยอะบนจอเล็ก
แถบใช้ชิ้น ข้อความบอกรายการสมดุลต้องออกแบบให้อ่านง่าย
UnloadProgressBar(
  // แถบใช้จำนวนชิ้น — สะท้อนงานที่เหลือจริง
  confirmedCount: _confirmedQty,
  totalCount: _totalQty,
  allConfirmed: _allConfirmed,
  // ข้อความรองบอกจำนวนรายการ — บอกว่าต้องเดินไปหยิบอีกกี่จุด
  subtitle: 'เหลืออีก ${_totalCount - _confirmedCount} รายการ',
)

เหตุผลที่เลือกทางที่สาม: สองตัวเลขตอบคำถามคนละข้อ

คำถามของผู้ใช้ตอบด้วย
“อีกนานไหมกว่าจะเสร็จ”จำนวนชิ้น
“ต้องเดินไปหยิบอีกกี่จุด”จำนวนรายการ

แถบยาว ๆ เหมาะกับคำถามแรกเพราะอ่านจากมุมตาได้ ส่วนคำถามที่สองเป็นตัวเลขที่อ่านตอนวางแผน

สิ่งที่ต้องตรวจก่อนปล่อย: _allConfirmed ยังคำนวณจากรายการอยู่

bool get _allConfirmed => _totalCount > 0 && _confirmedCount == _totalCount;

ซึ่งถูกต้องและไม่ควรเปลี่ยน — “ครบทุกรายการ” กับ “ครบทุกชิ้น” ต้องเกิดพร้อมกันอยู่แล้ว และการนับรายการเชื่อถือได้กว่าเพราะไม่ขึ้นกับว่าเซิร์ฟเวอร์แตกรายการอย่างไร

9. สรุป

  • สาม getter คำนวณจาก _partStatus ตัวเดียว — ไม่มีทางไม่ตรงกัน
  • totalCount == 0 ถูกดักก่อนหาร ไม่งั้นได้ NaN
  • หน้าจอประกอบจากเงื่อนไขหกชั้น เพิ่มส่วนใหม่ง่ายแต่ต้องอ่านทุกบรรทัด
  • _sessionOpen เป็นข้อมูลซ้ำที่ควรเป็น getter ของ _session != null
  • มีปุ่มยกเลิกการเลือกจริง ซึ่ง ScanPartScreen ขาดไป
  • ปุ่มยืนยันต้องมี S/N ครบจำนวน เข้มกว่า LoadToBasketScreen
  • สแกน Part ที่ยืนยันครบแล้วใช้แถบเหลือง ไม่ใช่กล่องแดง
  • _currentScannedItem() ประกาศคืน nullable แต่ใช้ firstWhere ที่โยนได้
  • ความคืบหน้านับรายการ ไม่ใช่ชิ้น — ทำให้ประเมินเวลาผิดเมื่อจำนวนต่อรายการต่างกันมาก
