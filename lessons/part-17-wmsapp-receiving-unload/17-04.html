2. ตัวชี้ขั้นตอนที่ไม่รับค่า

class PoStepIndicator extends StatelessWidget {
  const PoStepIndicator({super.key});

  @override
  Widget build(BuildContext context) {
    return const Row(
      children: [
        _StepDot(number: 1, label: 'สแกน PO', active: true),
        _StepLine(),
        _StepDot(number: 2, label: 'รับสินค้า', active: false),
      ],
    );
  }
}

ทั้งก้อนเป็น const — Flutter สร้างครั้งเดียวแล้วใช้ตัวเดิมตลอด ไม่ว่า build() ของจอจะทำงานกี่ครั้ง

สังเกตสองคลาสข้างล่างที่ขึ้นต้นด้วยขีดล่าง

class _StepDot extends StatelessWidget { ... }
class _StepLine extends StatelessWidget { ... }
ผลของขีดล่าง
ใช้ได้เฉพาะในไฟล์นี้ — ไม่มีใครเผลอเอาไปใช้ที่อื่น
เปลี่ยนหรือลบได้โดยไม่ต้องค้นทั้งโปรเจกต์
อ่านไฟล์เดียวแล้วเข้าใจครบ ไม่ต้องเปิดไฟล์อื่น

เป็นรูปแบบที่ควรทำตาม — widget ย่อยที่ใช้ที่เดียวควรอยู่ในไฟล์เดียวกันและเป็น private

ข้อจำกัดที่ต้องรู้: active: true ถูกฝังไว้ ถ้าวันหนึ่งอยากให้ตัวชี้บอกขั้นตอนจริง ต้องรับค่าเข้ามาและจะเสียความเป็น const ไป

// ถ้าจะทำให้รู้ขั้นตอน
class PoStepIndicator extends StatelessWidget {
  final int currentStep;
  const PoStepIndicator({super.key, required this.currentStep});

  @override
  Widget build(BuildContext context) => Row(     // const หายไป
    children: [
      _StepDot(number: 1, label: 'สแกน PO', active: currentStep >= 1),
      const _StepLine(),
      _StepDot(number: 2, label: 'รับสินค้า', active: currentStep >= 2),
    ],
  );
}

ราคาที่จ่ายเล็กมากในจอนี้ แต่หลักการที่ได้คือ ทุกค่าที่รับเข้ามาแลกมาด้วยการเสีย const จึงควรรับเฉพาะค่าที่ต้องเปลี่ยนจริง

3. รายการสินค้าที่ไม่ใช่ ListView

return WmsCard(
  child: Column(
    crossAxisAlignment: CrossAxisAlignment.start,
    children: [
      const Text('รายการสินค้า', ...),
      const SizedBox(height: 12),
      ...items.map(
        (item) => Container(
          margin: const EdgeInsets.only(bottom: 8),
          ...
        ),
      ),
    ],
  ),
);

...items.map(...) กระจายทุกรายการเข้าไปใน Column — ทุกแถวถูกสร้างและวาดพร้อมกันทั้งหมด ไม่ว่าจะเห็นบนจอหรือไม่

Column + spreadListView.builder
สร้าง widgetทุกแถวเสมอเฉพาะที่เห็นบนจอ
เลื่อนได้เองไม่ได้ ต้องมี SingleChildScrollView ครอบได้
ซ้อนใน Column อื่นได้เลยต้องกำหนดความสูงหรือใช้ shrinkWrap
เหมาะกับรายการสั้นที่รู้ว่าไม่ยาวรายการที่ยาวได้

ในจอนี้เลือก Column เพราะรายการสินค้าอยู่กลางหน้าที่มีของอื่นอยู่บนล่าง และทั้งหน้าถูกครอบด้วย SingleChildScrollView อยู่แล้ว

คำถามคือ PO หนึ่งใบมีสินค้ากี่รายการ

จำนวนรายการผล
5–20ทำงานได้ดี
50เริ่มหน่วงตอนเปิด
200ค้างชัดเจน — และรูป 200 ใบถูกโหลดพร้อมกัน
ปัญหานี้ทบกับ 16.10 — แต่ละแถวมี PartThumbnail ที่ไม่มี cacheWidth ที่ 4.2 MB ต่อรูป
PO ที่มี 50 รายการจะพยายามคลายรูป 50 ใบพร้อมกัน = 210 MB ซึ่งเป็นตัวเลขเดียวกับที่ 16.10 บอกว่าระบบเริ่มปิดแอปทิ้ง
ต่างจาก ListView.builder ที่โหลดเฉพาะแถวที่เห็น ตรงนี้โหลดทั้งหมดแม้ผู้ใช้จะไม่เลื่อนลงไปดูเลย

แก้ได้สองระดับ

// ระดับที่ 1 — แก้ที่ PartThumbnail (16.10) ลดจาก 4.2 MB เหลือ 50 KB ต่อรูป
// 50 รายการ → 2.5 MB ซึ่งรับได้

// ระดับที่ 2 — ถ้ารายการยาวจริง เปลี่ยนเป็น ListView
ListView.builder(
  shrinkWrap: true,                              // ให้สูงเท่าเนื้อหา
  physics: const NeverScrollableScrollPhysics(), // ให้ตัวนอกเลื่อนแทน
  itemCount: items.length,
  itemBuilder: (_, i) => _buildItemRow(items[i]),
)

shrinkWrap: true ทำให้ ListView วัดความสูงจากเนื้อหาทั้งหมด ซึ่งยังสร้างทุกแถวอยู่ดี — ได้แค่ความเป็นระเบียบ ไม่ได้ประโยชน์เรื่องประสิทธิภาพ

ถ้าต้องการประโยชน์จริง ต้องให้รายการมีความสูงจำกัดและเลื่อนเอง

SizedBox(
  height: 400,                    // จำกัดความสูง → สร้างเฉพาะที่เห็น
  child: ListView.builder(
    itemCount: items.length,
    itemExtent: 76,               // บอกความสูงต่อแถว → ข้ามการวัด (14.13)
    itemBuilder: (_, i) => _buildItemRow(items[i]),
  ),
)

แต่แลกมาด้วยรายการที่มีแถบเลื่อนของตัวเองซ้อนอยู่ในหน้าที่เลื่อนได้ ซึ่งใช้งานยากบนหน้าจอเล็ก — เป็นการแลกที่ต้องคุยกับผู้ใช้จริงก่อนตัดสิน

5. RichText กับ Text ต่างกันอย่างไร

RichText(
  text: TextSpan(
    style: TextStyle(fontSize: 11, color: AppTheme.textGrey(context)),
    children: [
      const TextSpan(text: 'Batch No', style: TextStyle(fontWeight: FontWeight.bold)),
      TextSpan(text: ' : ${item.lotNumber!}'),
    ],
  ),
),

ใช้เมื่อต้องการให้ข้อความบรรทัดเดียวมีหลายรูปแบบ — ที่นี่คือหัวข้อตัวหนากับค่าตัวปกติ

ทำแบบนี้ผลที่ต่าง
RichTextตัดบรรทัดเป็นก้อนเดียว หัวข้อกับค่าอยู่ติดกันเสมอ
Row ที่มีสอง Textตัดบรรทัดแยกกัน อาจหลุดคนละบรรทัดเมื่อพื้นที่แคบ
กับดักของ RichText ที่ต้องรู้: ไม่รับ style จากธีมโดยอัตโนมัติเหมือน Text — ถ้าไม่ระบุ style ที่ TextSpan ชั้นนอก จะได้ฟอนต์เริ่มต้นของระบบแทน Sarabun ซึ่งบน PDA จะเห็นสระไทยเพี้ยนทันที
ในโค้ดนี้ระบุ style ไว้ที่ชั้นนอกแล้ว จึงปลอดภัย — แต่ระบุแค่ขนาดกับสี ส่วนฟอนต์ตกทอดมาจาก DefaultTextStyle ของธีม
ถ้าจะให้แน่นอน ใช้ Text.rich(...) แทน ซึ่งรับ style จากธีมเหมือน Text ทุกประการ
// เขียนแบบนี้ปลอดภัยกว่าและสั้นกว่า
Text.rich(
  TextSpan(children: [
    const TextSpan(text: 'Batch No', style: TextStyle(fontWeight: FontWeight.bold)),
    TextSpan(text: ' : ${item.lotNumber!}'),
  ]),
  style: TextStyle(fontSize: 11, color: AppTheme.textGrey(context)),
)

7. แบบฝึกหัด

ผู้ใช้แจ้งว่า “PO ใบใหญ่ ๆ เปิดแล้วแอปค้างไปสิบวินาที บางทีเด้งออก”

  1. อธิบายสาเหตุโดยแยกเป็นสองส่วน
  2. เรียงลำดับการแก้ตามอัตราส่วนผลลัพธ์ต่อความเสี่ยง
  3. ถ้าแก้ข้อแรกแล้วยังไม่พอ ต้องคุยอะไรกับผู้ใช้ก่อนแก้ข้อถัดไป
แนวคำตอบ

ข้อ 1 — สองส่วนที่ทบกัน

ส่วนสาเหตุอาการ
เวลาColumn สร้างทุกแถวพร้อมกันค้างตอนเปิด
หน่วยความจำPartThumbnail ไม่มี cacheWidthเด้งออก

สังเกตว่าอาการสองอย่างมาจากคนละสาเหตุ — แก้อย่างเดียวจะได้ผลครึ่งเดียว

คำนวณ PO ที่มี 60 รายการ

ตอนนี้
widget ที่สร้าง60 แถว × ~12 widget = ~720
รูปที่ขอโหลด60 ใบพร้อมกัน
หน่วยความจำจากรูป60 × 4.2 MB = 252 MB

ข้อ 2 — ลำดับการแก้

ลำดับแก้อะไรผลความเสี่ยง
1ใส่ cacheWidth ที่ PartThumbnail252 MB → 3 MBต่ำมาก — บรรทัดเดียว หน้าตาไม่เปลี่ยน
2เอา FutureBuilder ออกลดงานต่อ buildต่ำ — แก้สองไฟล์
3เปลี่ยนเป็น ListView ที่จำกัดความสูงสร้างเฉพาะที่เห็นสูง — หน้าตาเปลี่ยน

ข้อ 1 ให้ผลมากที่สุดและเสี่ยงน้อยที่สุด — ทำก่อนเสมอ และแก้ที่เดียวได้ผลกับทั้ง 12 จุดที่ใช้ PartThumbnail

ข้อ 3 อยู่ท้ายสุดไม่ใช่เพราะยาก แต่เพราะเปลี่ยนวิธีที่ผู้ใช้ใช้งาน

ข้อ 3 — ต้องคุยอะไรก่อน

คำถามที่ต้องได้คำตอบก่อนแก้

คำถามทำไมต้องรู้
PO ใบใหญ่ที่สุดมีกี่รายการถ้าสูงสุด 30 ข้อ 1–2 อาจพอ ไม่ต้องแตะข้อ 3
ผู้ใช้เลื่อนดูทั้งรายการก่อนกดเริ่มไหมถ้าดูทั้งหมด รายการที่มีแถบเลื่อนซ้อนจะน่ารำคาญ
ต้องเห็นทั้งใบพร้อมกันเพื่อตัดสินใจอะไรไหมถ้าต้อง การจำกัดความสูงจะขัดการทำงาน

ถ้าคำตอบคือ “ผู้ใช้แค่ดูผ่าน ๆ แล้วกดเริ่มรับ” การจำกัดความสูงไม่กระทบอะไร

ถ้าคำตอบคือ “หัวหน้ากะต้องตรวจทั้งใบก่อนอนุมัติ” ต้องหาทางอื่น เช่นแสดงเป็นรายการเต็มแต่โหลดรูปเฉพาะที่เห็น

// ทางสายกลาง — คงเลย์เอาต์เดิม แต่ไม่โหลดรูปที่ยังไม่เห็น
// (แสดงไอคอนก่อน แล้วโหลดรูปเมื่อแถวเข้ามาในจอ)
VisibilityDetector(
  key: ValueKey(item.partId),
  onVisibilityChanged: (info) {
    if (info.visibleFraction > 0) setState(() => _visible.add(item.partId));
  },
  child: _visible.contains(item.partId)
      ? PartThumbnail(imageUrl: item.imageUrl)
      : const SizedBox(width: 40, height: 40),
)

วิธีนี้ต้องเพิ่มไลบรารี ซึ่งขัดกับสิ่งที่ 16.11 ชี้ว่าโปรเจกต์นี้ตั้งใจใช้ไลบรารีน้อย — ต้องชั่งน้ำหนักและคุยกับทีม ไม่ใช่ตัดสินใจเอง

ข้อสรุป: ปัญหาประสิทธิภาพส่วนใหญ่แก้ได้ด้วยการเปลี่ยนที่ผู้ใช้ไม่เห็น — ทำให้ครบก่อน แล้วค่อยพิจารณาการเปลี่ยนที่ผู้ใช้เห็น และการเปลี่ยนแบบหลังต้องถามผู้ใช้ก่อน ไม่ใช่ตัดสินจากตัวเลขอย่างเดียว

8. สรุป

  • สี่ widget ย่อยเป็น StatelessWidget ทั้งหมด และไม่มีตัวไหนรู้จัก ApiService
  • ตัวชี้ขั้นตอนทั้งก้อนเป็น const — สร้างครั้งเดียวตลอดอายุแอป
  • widget ย่อยที่ใช้ที่เดียวควรเป็น private และอยู่ในไฟล์เดียวกัน
  • ทุกค่าที่รับเข้ามาแลกมาด้วยการเสีย const
  • ...items.map() ใน Column สร้างทุกแถวเสมอ — รวมถึงโหลดรูปทุกใบพร้อมกัน
  • ปัญหานี้ทบกับ PartThumbnail ที่ไม่มี cacheWidth
  • shrinkWrap: true ไม่ได้ทำให้ประหยัดขึ้น
  • RichText ไม่รับ style จากธีม ควรใช้ Text.rich
  • ชื่อฟิลด์ในโค้ด (lotNumber) กับคำที่ผู้ใช้เห็น (Batch No) ไม่ตรงกัน
