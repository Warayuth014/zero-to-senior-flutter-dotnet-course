3. Map ของตัวควบคุมที่ต้องจัดการเอง

// key = UnloadItem.lineId (ไม่ใช่ partId — 1 Part อาจมีหลาย Lot บน pallet เดียวกัน)
final Map<int, String> _partStatus = {};

คอมเมนต์อธิบายเหตุผลของคีย์ไว้ ซึ่งสำคัญมาก — ถ้าใช้ partId เป็นคีย์ สินค้าเดียวกันคนละล็อตจะทับกัน

แต่ละรายการมีช่องกรอกจำนวนของตัวเอง จึงต้องมีตัวควบคุมหนึ่งตัวต่อรายการ

void _buildQtyControllers() {
  _disposeQtyControllers();            // ← คืนของเก่าก่อนเสมอ
  if (_session == null) return;

  for (final item in _session!.items) {
    _qtyCtrl[item.lineId] = TextEditingController(text: '${item.qty}');
  }
}

void _disposeQtyControllers() {
  for (final controller in _qtyCtrl.values) {
    controller.dispose();
  }
  _qtyCtrl.clear();
}
เรียก _buildQtyControllers() ที่ทำไม
หลังเปิด sessionสร้างช่องให้ทุกรายการบนพาเลท
หลัง _reloadSession()รายการเปลี่ยน lineId เปลี่ยน

และถูกคืนสามที่

@override
void dispose() {

  _disposeQtyControllers();       // ← 1. ตอนแผ่นถูกปิด
  super.dispose();
}

void _buildQtyControllers() {
  _disposeQtyControllers();       // ← 2. ก่อนสร้างชุดใหม่
  ...
}

void _resetForNextPallet() {
  _disposeQtyControllers();       // ← 3. ตอนเริ่มพาเลทใหม่
  ...
}
นี่คือการจัดการทรัพยากรที่ถูกที่สุดในทั้งสองโดเมน — ครบทุกทางออก และมีฟังก์ชันเดียวที่ทุกที่เรียก
ต่างจาก _pickPallet (17.5), _openAssignDialog (17.8) และ _showBasketDialog (17.10) ที่ไม่คืนเลย
คนเขียนคนเดียวกันทำถูกที่นี่และพลาดที่นั่น — ความต่างคือตัวควบคุมที่นี่เป็นฟิลด์ของ State จึงอยู่ในสายตาของ dispose()

จุดที่ยังพลาดได้

final qtyText = _qtyCtrl[item.lineId]?.text.trim() ?? '';

?. และ ?? '' กันกรณีที่ไม่มีตัวควบคุมของ lineId นั้น ซึ่งเกิดได้ถ้า _reloadSession() ทำงานแล้ว lineId เปลี่ยน — จบด้วย qty = 0 แล้วถูกปฏิเสธที่ด่านถัดไป

if (qty <= 0) {
  showErrorDialog(context, message: 'กรุณาระบุจำนวนที่ต้องการ Unload');
  return;
}

ผู้ใช้จะเห็นข้อความที่ไม่ตรงกับสาเหตุ — เขากรอกจำนวนไว้แล้ว แต่ตัวควบคุมหายไป

4. โหลดใหม่แทนการแก้เอง

// โหลด session ปัจจุบันซ้ำจาก server — ใช้หลัง confirm สำเร็จ เพื่อให้ LineId/qty ตรงกับจริงเสมอ
// (ถ้ามีเศษเหลือ backend จะสร้าง UnloadLine ใหม่ที่มี LineId ต่างจากเดิม
//  แก้ปัญหาโดยโหลดใหม่แทนการ patch เอง)
Future<void> _reloadSession() async {
  final session = _session;
  if (session == null) return;

  final result = await ApiService().openUnloadSession(
    palletId: session.palletId,
    operatorId: widget.userId,
  );
  if (!mounted || !result.success) return;

  final fresh = result.data!;
  setState(() {
    _session = fresh;
    _sessionId = fresh.sessionId;
    _partStatus.clear();
    for (final item in fresh.items) {
      _partStatus[item.lineId] =
          fresh.confirmedLineIds.contains(item.lineId) ? 'CONFIRMED' : 'PENDING';
    }
  });
  _buildQtyControllers();
}

คอมเมนต์อธิบายการตัดสินใจไว้ครบ — ทำไมถึงโหลดใหม่แทนที่จะแก้ข้อมูลในเครื่อง

วิธีเมื่อเซิร์ฟเวอร์แตกรายการใหม่
แก้ในเครื่องต้องเดาว่า lineId ใหม่คืออะไร — เดาไม่ได้
โหลดใหม่ได้ของจริงเสมอ

สังเกตว่าการยืนยันหยิบของบางส่วนทำให้เซิร์ฟเวอร์สร้างรายการใหม่สำหรับส่วนที่เหลือ — ข้อมูลในเครื่องจึงล้าสมัยทันทีหลังยืนยัน

นี่เป็นตัวอย่างของหลักที่ 12.4 สอนไว้: เมื่อไม่แน่ใจว่าสถานะในเครื่องยังตรงไหม ให้โหลดใหม่ทั้งชุด — ถูกกว่าและปลอดภัยกว่าการพยายามซิงก์ทีละส่วน

ราคาที่จ่ายคือหนึ่งคำขอเพิ่มต่อการยืนยันหนึ่งครั้ง ซึ่งยอมรับได้เพราะผู้ใช้ยืนยันไม่กี่สิบครั้งต่อพาเลท

สังเกตอีกอย่าง — openUnloadSession ถูกใช้ทั้งเปิดใหม่และโหลดซ้ำ

// เปิด session
final result = await ApiService().openUnloadSession(palletId: palletId, ...);

// โหลดซ้ำ — เมธอดเดียวกัน
final result = await ApiService().openUnloadSession(palletId: session.palletId, ...);

แปลว่า endpoint นี้เป็นแบบเรียกซ้ำได้โดยไม่เปลี่ยนผล — เรียกสิบครั้งกับพาเลทเดียวกันได้ session เดิม ซึ่งเป็นคุณสมบัติที่ทำให้การโหลดซ้ำปลอดภัย

5. การป้องกันการปิดแผ่น

bool get _canDismiss => !_loading && !_returning;

ปิดแผ่นไม่ได้ระหว่างที่มีคำขอค้างอยู่ หรือระหว่างที่กำลังคืนพาเลท

สถานะปิดได้ไหมเหตุผล
ว่าง ยังไม่สแกนพาเลทได้ไม่มีอะไรเสีย
เปิด session แล้ว ยืนยันไปบางส่วนได้สถานะอยู่ที่เซิร์ฟเวอร์แล้ว
กำลังยิงคำขอไม่ได้ผลลัพธ์ยังไม่รู้
กำลังคืนพาเลทไม่ได้มีแอนิเมชัน 5 วินาทีที่ต้องดูจนจบ
แถวที่สองต่างจาก ScanPartScreen อย่างสำคัญ — ที่นี่ปิดแผ่นกลางทางได้โดยไม่เสียงาน เพราะทุกการยืนยันถูกส่งไปเซิร์ฟเวอร์แล้ว ผู้ใช้สแกนพาเลทเดิมกลับมาได้ session เดิม
ต่างจาก S/N ที่สแกนค้างใน ScanPartScreen ซึ่งอยู่แต่ในเครื่องและหายเมื่อออกจากจอ
ความต่างอยู่ที่ว่าสถานะอยู่ฝั่งไหน ไม่ใช่ที่โค้ดฝั่งแอปเขียนดีกว่ากัน

แต่ยังมีสถานะที่หายได้ — S/N ที่สแกนไว้ในแผ่นนี้

final List<String> _scannedSerials = [];

ถ้าผู้ใช้สแกน S/N ไป 8 ตัวแล้วปัดแผ่นลง จะหายเหมือนกัน และ _canDismiss ไม่ได้ตรวจเรื่องนี้

เหตุผลที่เป็นไปได้สมเหตุสมผลไหม
ให้ผู้ใช้เห็นว่าพาเลทกำลังถูกส่งกลับใช่ — เป็นการกระทำทางกายภาพที่ต้องรอจริง
ให้เวลาผู้ใช้ถอยออกจากพาเลทใช่ ถ้าโฟล์คลิฟท์อัตโนมัติมารับ
ให้แอนิเมชันดูสวยไม่ — 5 วินาทีต่อพาเลทคือเวลาจริงของผู้ใช้

แต่การคืนพาเลทเป็นคำสั่งที่ไม่รู้ผลเมื่อเน็ตหลุด — คำสั่งอาจไปถึงแล้ว โฟล์คลิฟท์อาจกำลังมา แต่แอปบอกว่าล้มเหลว

7. ข้อผิดพลาดที่พบบ่อย

เข้าใจผิดที่จริง
แผ่นเลื่อนใช้แทนจอได้ทุกกรณีปิดโดยไม่ตั้งใจง่ายกว่ามาก
Map ของตัวควบคุมคืนพร้อม Stateต้องวนคืนเองทุกตัว
แก้ข้อมูลในเครื่องเร็วกว่าโหลดใหม่ที่นี่โหลดใหม่ถูกกว่าเพราะ lineId เปลี่ยน
ปิดแผ่นกลางทางแล้วเสียงานที่นี่ไม่เสีย เพราะสถานะอยู่ที่เซิร์ฟเวอร์

ปัญหาแรกที่เจอ: StationInfo มี Color และ IconData ซึ่งส่งผ่าน JSON ตรง ๆ ไม่ได้

StationInfo(
  id: 'UNL-1',
  label: 'Unload 1',
  color: const Color(0xFF1976D2),     // ส่งเป็นสตริง '#1976D2' แล้วแปลง
  icon: MdiIcons.trayArrowUp,          // ส่งไม่ได้ — ต้องแมปจากชื่อ
)
// แมปชื่อไอคอนที่รองรับ — ที่ไม่รู้จักใช้ตัวเริ่มต้น
IconData _iconFor(String? name) => switch (name) {
  'trayArrowUp' => MdiIcons.trayArrowUp,
  'trayArrowDown' => MdiIcons.trayArrowDown,
  _ => MdiIcons.viewGridOutline,
};

factory StationInfo.fromJson(Map<String, dynamic> json) => StationInfo(
  id: json['id'],
  label: json['label'] ?? json['id'],
  color: Color(int.parse('FF${json['color'] ?? '757575'}', radix: 16)),
  icon: _iconFor(json['icon']),
);

สังเกตว่าไอคอนยังต้องอยู่ในโค้ดอยู่ดี — เพิ่มไอคอนแบบใหม่ยังต้องปล่อยแอปใหม่ ได้แค่ไม่ต้องปล่อยเมื่อเพิ่มสถานีที่ใช้ไอคอนเดิม

ข้อ 2 — โหลดไม่สำเร็จ

นี่คือคำถามที่สำคัญที่สุด เพราะเปลี่ยนจอจาก “เปิดได้เสมอ” เป็น “ขึ้นกับเน็ต”

ทางเลือกผล
แสดงข้อผิดพลาด จอว่างผู้ใช้ทำงานไม่ได้เลย — แย่กว่าเดิมมาก
ใช้รายชื่อที่ฝังไว้เป็นตัวสำรองทำงานได้เสมอ แต่ต้องดูแลสองที่
จำรายชื่อล่าสุดที่โหลดสำเร็จทำงานได้ถ้าเคยโหลดสำเร็จ
// ทางที่สาม — เก็บลง SharedPreferences หลังโหลดสำเร็จ
Future<void> _loadStations() async {
  final result = await _api.getUnloadStations();
  if (!mounted) return;

  if (result.success) {
    setState(() => _stations = result.data!);
    unawaited(_cacheStations(result.data!));
    return;
  }

  // โหลดไม่ได้ → ใช้ของที่จำไว้ ผู้ใช้ยังทำงานต่อได้
  final cached = await _loadCachedStations();
  if (!mounted) return;
  setState(() => _stations = cached);
  if (cached.isEmpty) {
    showWarningSnackbar(context, 'โหลดรายชื่อสถานีไม่ได้ — สแกนบาร์โค้ดสถานีแทนได้');
  }
}

สังเกตข้อความสุดท้าย — บอกทางออกที่ยังใช้ได้ เพราะการสแกนบาร์โค้ดต้องเทียบกับรายชื่อ ซึ่งถ้าไม่มีรายชื่อก็สแกนไม่ได้เหมือนกัน

ตรงนี้เป็นข้อบกพร่องของแผน — ต้องให้เซิร์ฟเวอร์ตรวจแทน หรือยอมรับว่าไม่มีรายชื่อก็เข้าสถานีไม่ได้

ข้อ 3 — คุ้มไหม

คำถามถ้าตอบว่าสรุป
สถานีเปลี่ยนบ่อยแค่ไหนปีละครั้งไม่คุ้ม
เดือนละครั้งคุ้ม
ปล่อยแอปใหม่ยากแค่ไหนต้องเดินไปติดตั้งทีละเครื่อง 20 เครื่องคุ้มขึ้นมาก
มีสถานีที่ปิดชั่วคราวไหมมี ต้องซ่อมบ่อยคุ้ม — ซ่อนสถานีที่ปิดได้

คำตอบที่ตรงที่สุดสำหรับกรณีนี้: ถ้าโจทย์คือ “เพิ่ม UNL-4” อย่างเดียว การเพิ่มสามบรรทัดในโค้ดแล้วปล่อยแอปใหม่ ใช้เวลาน้อยกว่างานทั้งหมดข้างบนมาก

การย้ายไปเซิร์ฟเวอร์คุ้มก็ต่อเมื่อจะมีการเปลี่ยนอีกหลายครั้ง หรือต้องซ่อนสถานีที่ปิดชั่วคราว ซึ่งเป็นความสามารถที่ต่างออกไปและควรเป็นเหตุผลหลัก ไม่ใช่ผลพลอยได้

ตรงกับหลักที่ 9.12 สอน — ย้ายของออกจากโค้ดเมื่อมีเหตุผลที่ชัดเจน ไม่ใช่เพราะ “ฝังในโค้ดไม่ดี”

9. สรุป

  • สามสถานีถูกฝังในโค้ด — เปิดจอได้ทันทีแต่เพิ่มสถานีต้องปล่อยแอปใหม่
  • ข้อความผิดพลาดพิมพ์รายชื่อสถานีซ้ำ แทนที่จะสร้างจากรายการจริง
  • แผ่นเลื่อนต้องตั้งสามค่าเมื่อลูกวาดรูปทรงเอง
  • viewInsets.bottom + viewPadding.bottom กันแป้นพิมพ์บังช่องกรอก
  • Map ของตัวควบคุมถูกคืนครบสามทางออกผ่านฟังก์ชันเดียว — ดีที่สุดในสองโดเมนนี้
  • ตัวควบคุมที่เป็นฟิลด์ของ State ถูกคืน ส่วนที่สร้างในเมธอดไม่ถูกคืน
  • โหลด session ใหม่แทนการแก้ในเครื่อง เพราะ lineId เปลี่ยนหลังยืนยันบางส่วน
  • ปิดแผ่นกลางทางไม่เสียงาน เพราะสถานะอยู่ที่เซิร์ฟเวอร์ — แต่ S/N ที่สแกนค้างยังหาย
  • Future.wait กับตัวหน่วง 5 วินาทีบังคับให้แอนิเมชันครบเวลา
