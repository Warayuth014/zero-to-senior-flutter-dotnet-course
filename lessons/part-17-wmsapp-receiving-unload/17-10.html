2. สแกนแล้วเลือกให้อัตโนมัติ

void _onScanPart() {
  final partId = _scanCtrl.text.trim().toUpperCase();
  _scanCtrl.clear();                    // ← ล้างก่อนตรวจ
  if (partId.isEmpty) return;

  final items = _data?.items
      .where((i) => i.partId == partId && !i.isDone)
      .toList() ?? [];

  if (items.isEmpty) {
    showErrorDialog(context, message: 'ไม่พบ Part "$partId" ที่รอ Load เข้า Basket');
    _scanFocus.requestFocus();
    return;
  }

  setState(() => _selectedItem = items.first);
  _showBasketDialog(items.first);
}

_scanCtrl.clear() อยู่บรรทัดที่สอง — ล้างทันทีไม่ว่าผลจะเป็นอย่างไร

ล้างตอนไหนผล
ทันที (โค้ดนี้)สแกนถัดไปไม่ต่อท้ายค่าเดิมแน่นอน
เฉพาะเมื่อสำเร็จสแกนผิดแล้วยิงใหม่ จะได้ค่าติดกันสองอัน

ตรงกับที่ 3.9 สอน — ช่องสแกนต้องว่างเสมอหลังอ่านค่าออกมาแล้ว

ตัวกรองมีสองเงื่อนไข

.where((i) => i.partId == partId && !i.isDone)

isDone เป็น getter ที่คำนวณจากจำนวน

bool get isDone => qtyRemaining <= 0;

ใช้ <= ไม่ใช่ == — เผื่อกรณีที่ตัวเลขติดลบจากข้อมูลที่ผิด ซึ่งเป็นการเขียนที่ทนกว่า

จุดที่ต้องระวัง: items.first เลือกตัวแรกที่เจอโดยไม่ถามผู้ใช้
สินค้าตัวเดียวกันคนละล็อตจะมีหลายรายการ (ข้อมูลถูกจัดกลุ่มตาม Part+Lot) ถ้าผู้ใช้สแกน PT-9821 ที่มีสองล็อตค้างอยู่ ระบบเลือกล็อตแรกให้เลย
ต่างจาก ScanPartScreen ที่เด้ง popup ให้เลือกล็อต — สองจอในโปรเจกต์เดียวกันจัดการเรื่องเดียวกันคนละแบบอีกครั้ง

ที่นี่อาจยอมรับได้ เพราะของที่ unload แล้วอยู่ตรงหน้าผู้ใช้จริง ๆ และผู้ใช้แตะการ์ดในรายการเพื่อเลือกล็อตอื่นได้ — แต่ถ้าผู้ใช้ไม่รู้ว่ามีล็อตอื่น ก็จะไม่ไปแตะ

5. การตรวจ S/N ที่ต่างจาก receiving

void addSerial() {
  final sn = serialCtrl.text.trim().toUpperCase();
  serialCtrl.clear();
  if (sn.isEmpty) return;

  if (!item.serialNumbers.contains(sn)) {
    showErrorDialog(context, message: 'S/N "$sn" ไม่ใช่ของรายการนี้ หรือถูก Load ไปแล้ว');
    serialFocus.requestFocus();
    return;
  }
  if (scannedSerials.contains(sn)) {
    showErrorDialog(context, message: 'สแกน S/N "$sn" ซ้ำ');
    serialFocus.requestFocus();
    return;
  }
  if (scannedSerials.length >= item.qtyRemaining) {
    showErrorDialog(context, message: 'สแกนครบจำนวนที่เหลือแล้ว (${item.qtyRemaining})');
    return;
  }

  setDialogState(() => scannedSerials.add(sn));
  serialFocus.requestFocus();
}

ไม่มีการยิงเน็ตเลย — ต่างจาก _addSerial() ใน 17.6 ที่ต้องถามเซิร์ฟเวอร์

Receiving (17.6)Basket (ที่นี่)
รายชื่อ S/N ที่ถูกต้องไม่รู้ ต้องถามทีละตัวรู้แล้ว — มากับ item.serialNumbers
ความเร็วรอเน็ตทุกตัวทันที
ทำงานตอนเน็ตหลุดไม่ได้ได้ จนถึงตอนกดยืนยัน

เกิดจากการออกแบบฝั่งเซิร์ฟเวอร์ที่ส่งรายชื่อ S/N มาพร้อมข้อมูลรายการ

serialNumbers: (json['serialNumbers'] as List?)?.map((e) => e as String).toList() ?? [],

สิ่งที่ได้คือ flow ที่สแกนเร็วมากและไม่พึ่งเน็ตระหว่างทาง — เหมาะกับคลังที่สัญญาณไม่ทั่วถึง

แต่มีจุดที่ต้องระวัง

bool get serialRequire => serialNumbers.isNotEmpty;

serialRequire ไม่ได้มาจากเซิร์ฟเวอร์ แต่เดาจากว่ามีรายชื่อ S/N ส่งมาไหม

ถ้าเซิร์ฟเวอร์serialRequireผล
ส่ง S/N มาครบtrueถูก
ไม่ส่ง เพราะสินค้าไม่ต้องมี S/Nfalseถูก
ไม่ส่ง เพราะข้อมูลผิดพลาดfalseของที่ต้องสแกน S/N ถูกโหลดด้วยจำนวน

เทียบกับฝั่ง receiving ที่ serialRequire เป็นฟิลด์จริงจากเซิร์ฟเวอร์

// receiving_models.dart — เป็นฟิลด์จริง
serialRequire: json['serialRequire'] ?? false,

// basket_models.dart — คำนวณเอา
bool get serialRequire => serialNumbers.isNotEmpty;

8. แบบฝึกหัด

ผู้ใช้แจ้งว่า “สแกน S/N ไปเกือบครบ 20 ตัว แล้วเผลอกดข้างนอกกล่อง ทุกอย่างหายหมด ต้องเริ่มใหม่”

  1. ทำไมถึงหาย และเทียบกับ ScanPartScreen ว่าต่างกันอย่างไร
  2. เสนอทางแก้สามระดับ พร้อมข้อดีข้อเสีย
  3. เลือกหนึ่งแล้วเขียนโค้ด

  // สแกน S/N ไปหลายตัวแล้วกดข้างนอกโดยไม่ตั้งใจ = เสียงานทั้งหมด
  barrierDismissible: false,

          return;
        }
        final leave = await showConfirmDialog(

              'ยกเลิกแล้วต้องสแกนใหม่ทั้งหมด',
          confirmLabel: 'ยกเลิกการโหลด',
          cancelLabel: 'สแกนต่อ',
          isDanger: true,
        );
        if (leave && ctx.mounted) Navigator.pop(ctx, false);
      }

      return PopScope(
        // ปุ่มย้อนกลับของเครื่องก็ต้องผ่าน cancel() เหมือนกัน
        canPop: scannedSerials.isEmpty,
        onPopInvokedWithResult: (didPop, _) {
          if (!didPop) cancel();
        },
        child: AlertDialog(
          ...
          actions: [
            TextButton(onPressed: cancel, child: const Text('ยกเลิก')),
            ...
          ],
        ),
      );
    },
  ),

9. สรุป

  • จอ 470 บรรทัดแต่มี state แค่หกตัว — สถานะส่วนใหญ่อยู่ในกล่อง
  • _scanCtrl.clear() ทันทีหลังอ่านค่า ไม่รอผลลัพธ์
  • สแกน Part แล้วได้ล็อตแรกที่เจอ ไม่ถามผู้ใช้ ต่างจาก ScanPartScreen
  • กล่องมีสองขั้นตอนคุมด้วย showBasketStep ตัวเดียว
  • ปุ่ม “ถัดไป” ตรวจว่ามี S/N อย่างน้อยหนึ่ง ไม่ได้ตรวจว่าครบ
  • การตรวจ S/N ไม่ยิงเน็ตเลย เพราะรายชื่อมาพร้อมข้อมูลรายการ
  • serialRequire เป็น getter ที่เดาจากว่ามีรายชื่อ S/N ไหม
  • สถานะที่เป็น local ไม่มีทางค้างผิด แต่กู้ไม่ได้เมื่อกล่องปิด
  • ทรัพยากรสี่ตัวในกล่องไม่ถูกคืน
  • สร้าง SnackBar เองแทนที่จะใช้ของกลาง — เสียการสั่นไป
