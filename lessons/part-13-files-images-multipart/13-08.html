1. ทำไมต้องมีความคืบหน้า

คำขอ JSON ปกติเสร็จใน 200 มิลลิวินาที วงหมุนธรรมดาก็พอ แต่การอัปโหลดรูปบนเน็ตคลังใช้เวลา 3–30 วินาที

ไม่มีความคืบหน้ามี
ผู้ใช้ไม่รู้ว่าค้างหรือกำลังทำงานเห็นว่าขยับ
กดซ้ำเพราะคิดว่าไม่ติดรอ
ปิดแอปกลางคันรู้ว่าเหลืออีกนิดเดียว
แถวที่สองคือความเสียหายจริง — ผู้ใช้ที่กดซ้ำระหว่างที่ยังส่งอยู่ อาจได้รูปซ้ำถ้าไม่มีรหัสคำสั่ง (13.4) และเสียเน็ตไปเปล่าแน่นอน

2. MultipartRequest ไม่บอกความคืบหน้า

ต้องนับเอง โดยห่อ stream ของคำขอไว้

/// `MultipartRequest` ไม่มีการรายงานความคืบหน้าให้ จึงต้องห่อ stream ของ
/// ตัวมันเองแล้วนับไบต์ที่ไหลผ่าน
Future<http.StreamedResponse?> _send(
  http.MultipartRequest request, {
  void Function(int sent, int total)? onProgress,
  Future<void>? cancel,
}) async {
  final total = request.contentLength;
  var sent = 0;

  final counted = request.finalize().transform(
    StreamTransformer<List<int>, List<int>>.fromHandlers(
      handleData: (chunk, sink) {
        sent += chunk.length;
        onProgress?.call(sent, total);
        sink.add(chunk);        // ส่งต่อไปตามเดิม
      },
    ),
  );
  ...
}
ส่วนหน้าที่
request.finalize()ปิดคำขอแล้วคืน stream ของไบต์ที่จะส่ง
request.contentLengthขนาดรวม รู้ล่วงหน้าเพราะไฟล์อยู่ในหน่วยความจำแล้ว
StreamTransformerให้เรานับได้ระหว่างที่ข้อมูลไหลผ่าน โดยไม่เปลี่ยนข้อมูล
sink.add(chunk)ต้องส่งต่อ ไม่งั้นข้อมูลจะหายไปเฉย ๆ
finalize() เรียกได้ครั้งเดียว — เรียกซ้ำจะได้ StateError: Can't finalize a finalized Request ซึ่งเป็นเหตุผลที่การลองใหม่ต้องสร้าง MultipartRequest ตัวใหม่ ไม่ใช่ส่งตัวเดิมซ้ำ

3. ความคืบหน้าที่บอกความจริงแค่ครึ่งเดียว

สิ่งที่นับได้คือ “ไบต์ที่เขียนลง socket แล้ว” ซึ่งไม่เท่ากับ “เซิร์ฟเวอร์ได้รับแล้ว”

100% บนหน้าจอ  =  ส่งครบแล้ว
                   ยังไม่รู้ว่าเซิร์ฟเวอร์รับครบไหม
                   ยังไม่รู้ว่าเซิร์ฟเวอร์บันทึกสำเร็จไหม

ผลคือหลังจากถึง 100% ผู้ใช้ยังต้องรออีกพักหนึ่งกว่าจะได้คำตอบ ซึ่งดูเหมือนค้าง

ทางแก้รายละเอียด
เปลี่ยนข้อความตอนถึง 100%“กำลังบันทึก…” แทนที่จะแสดง 100% ค้างไว้
ให้แถบเปลี่ยนเป็นแบบไม่มีที่สิ้นสุดสื่อว่ายังทำงานอยู่แต่ไม่รู้ว่านานแค่ไหน
หยุดที่ 95%อย่าทำ — เป็นการโกหก และผู้ใช้จะเรียนรู้ว่าตัวเลขเชื่อไม่ได้
Text(
  store.progress == 1.0
      ? 'กำลังบันทึกที่เซิร์ฟเวอร์…'
      : 'กำลังส่ง ${(store.progress! * 100).round()}%',
)

สิ่งที่ต้องเข้าใจคือ การยกเลิกไม่รับประกันว่าเซิร์ฟเวอร์ไม่ได้รับ — ถ้าปิดตอนส่งไป 99% เซิร์ฟเวอร์อาจได้ครบพอดีและบันทึกไปแล้ว

ผลตามมาสองข้อ

ข้อทำอย่างไร
อย่าบอกผู้ใช้ว่า “ยกเลิกแล้ว ไฟล์ไม่ถูกส่ง”บอกว่า “หยุดส่งแล้ว” และให้เขาตรวจได้
รหัสคำสั่งยังต้องเดิมถ้าเขาส่งใหม่ จะได้ไม่มีสองใบ

lab จึงกลับไปที่สถานะ ready ไม่ใช่ล้าง ทำให้กดส่งใหม่ได้ทันทีด้วยรหัสเดิม

case UploadCancelled():
  _stage = UploadStage.ready;
  _message = null;

5. ความล้มเหลวสี่แบบ ต่างกันสี่ทาง

sealed class UploadOutcome { }
class UploadSucceeded extends UploadOutcome { final String url; }
class UploadRejected extends UploadOutcome { final String message; }
class UploadFailed extends UploadOutcome { ... final bool outcomeUnknown; }
class UploadCancelled extends UploadOutcome { }
ผลผู้ใช้ต้องทำอะไรสถานะที่ store ตั้ง
สำเร็จไม่ต้องทำอะไรdone
เซิร์ฟเวอร์ปฏิเสธ (413, 400)เลือกรูปใหม่ — กดส่งซ้ำไม่ช่วยidle พร้อมล้างรูป
ล้มเหลวชั่วคราวกดส่งซ้ำได้เลยfailed เก็บรูปไว้
ยกเลิกกดส่งใหม่ได้ready

แถวที่สองคือจุดที่ต่างจากความล้มเหลวทั่วไป และเป็นสิ่งที่ทำผิดกันบ่อย

case UploadRejected(:final message):
  // ผู้ใช้แก้ได้ แต่ต้องเลือกรูปใหม่ ไม่ใช่กดส่งซ้ำ
  _stage = UploadStage.idle;
  _image = null;
  _message = message;

ถ้าไม่ล้างรูปทิ้ง ผู้ใช้จะเห็นปุ่ม “ลองใหม่” แล้วกดซ้ำ ๆ กับรูปที่เซิร์ฟเวอร์ปฏิเสธไปแล้ว ได้ผลเดิมทุกครั้ง

6. การอัปโหลดที่ไม่รู้ผล

} on http.ClientException {
  return const UploadFailed(
    ApiFailureKind.network,
    'การเชื่อมต่อขาดระหว่างอัปโหลด',
    outcomeUnknown: true,
  );
}

ทุกความล้มเหลวของการอัปโหลดที่เกิดจากเครือข่าย เป็นกรณีที่ไม่รู้ผลเสมอ — ต่างจาก GET ที่รู้แน่ว่าไม่มีอะไรเกิดขึ้น (7.10)

case UploadFailed(:final message, :final outcomeUnknown):
  _stage = UploadStage.failed;
  _message = outcomeUnknown
      ? '$message — ตรวจว่ารูปขึ้นแล้วหรือยังก่อนส่งซ้ำ'
      : message;

ข้อความนี้อาจดูน่ารำคาญ แต่มันตรงกับความจริง และมีทางทำให้ดีขึ้นได้

ทำได้ผล
โหลดข้อมูลใหม่แล้วดูว่ามีรูปหรือยังตอบได้แน่นอน ไม่ต้องให้ผู้ใช้เดา
ส่งซ้ำด้วยรหัสเดิมปลอดภัยเพราะเซิร์ฟเวอร์จะไม่เก็บซ้ำ

ทางที่สองคือเหตุผลที่ Idempotency-Key คุ้มค่ามากในฟีเจอร์นี้ — เมื่อมีรหัสแล้ว ผู้ใช้กดส่งซ้ำได้อย่างสบายใจ และข้อความเตือนก็เปลี่ยนเป็นแค่ “ส่งไม่สำเร็จ ลองใหม่” ได้

8. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ข้อมูลที่ส่งไปหายบางส่วนลืม sink.add(chunk) ในตัวนับส่งต่อทุกก้อน
Can't finalize a finalized Requestส่ง MultipartRequest ตัวเดิมซ้ำสร้างใหม่ทุกครั้ง
ค้างที่ 100% นานยังรอเซิร์ฟเวอร์บันทึกเปลี่ยนข้อความตอนถึง 100%
ผู้ใช้กดลองใหม่กับรูปที่ถูกปฏิเสธไม่ได้ล้างรูปทิ้งแยก Rejected ออกจาก Failed
ได้รูปซ้ำในระบบกดซ้ำได้ระหว่างส่ง หรือรหัสคำสั่งเปลี่ยนกันกดซ้ำ และใช้รหัสเดิมตอนลองใหม่
ยกเลิกแล้วบอกว่าไฟล์ไม่ถูกส่งเข้าใจผิดว่าการยกเลิกรับประกันได้บอกว่า “หยุดส่งแล้ว” และให้ตรวจได้

9. แบบฝึกหัด

ผู้ใช้ถ่ายรูปแล้วกดส่ง ระหว่างที่ส่งอยู่เขากดปุ่มย้อนกลับออกจากจอ

  1. ควรเกิดอะไรขึ้น
  2. ต้องระวังอะไรในโค้ด
แนวคำตอบ

1. มีสามทางเลือกที่ต่างกันมาก

ทางเหมาะเมื่อ
ยกเลิกการส่งรูปนี้เป็นของจอนี้เท่านั้น และผู้ใช้ตั้งใจทิ้ง
เตือนก่อนออกผู้ใช้อาจกดพลาด และการส่งใกล้เสร็จแล้ว
ส่งต่อในพื้นหลังรูปเป็นส่วนหนึ่งของงานที่ใหญ่กว่าจอนี้

สำหรับงานคลัง ทางที่สามมักถูกที่สุด — ผู้ใช้ถ่ายรูปหลักฐานแล้วต้องรีบไปทำงานต่อ การบังคับให้ยืนรอหน้าจอสามสิบวินาทีคือการเสียเวลาที่ไม่จำเป็น

แต่ต้องมีที่ให้เห็นว่ายังส่งไม่เสร็จ เช่นแถบเล็ก ๆ บนหัวจอ ไม่งั้นผู้ใช้จะไม่มีทางรู้ว่าอะไรล้มเหลว

2. สิ่งที่ต้องระวัง ขึ้นกับทางที่เลือก

ถ้ายกเลิก — dispose ต้องยกเลิกจริง ไม่ใช่แค่ทิ้ง store

@override
void dispose() {
  // ยกเลิกการอัปโหลดที่ค้างอยู่ ไม่ให้ callback มาถึง store ที่ถูกทิ้งแล้ว
  if (_cancel case final completer? when !completer.isCompleted) {
    completer.complete();
  }
  super.dispose();
}

ถ้าลืม จะได้ A ChangeNotifier was used after being disposed ตอน onProgress ถูกเรียก (12.8)

ถ้าส่งต่อในพื้นหลัง — store ต้องไม่ผูกกับจอ

ต้องทำเพราะ
ย้าย store ขึ้นไปที่ shellจอถูกทิ้งแล้ว store ต้องยังอยู่ (9.8)
อย่าให้ store แตะ BuildContextจอนั้นไม่มีอยู่แล้ว (9.7)
มีที่แสดงผลรวมของงานที่ค้างผู้ใช้ต้องรู้ว่าอะไรสำเร็จอะไรไม่
ล้างเมื่อออกจากระบบรูปของคนก่อนต้องไม่ถูกส่งในชื่อคนใหม่ (8.8)

แถวสุดท้ายเป็นข้อที่คิดไม่ถึงและสำคัญ — งานที่ค้างอยู่ในคิวตอนกะเปลี่ยน จะถูกส่งด้วย token ของคนใหม่ถ้าไม่ระวัง ซึ่งทำให้ข้อมูลผู้ถ่ายผิด (13.4 แบบฝึกหัด)

วิธีที่ปลอดภัยที่สุดคือล้างคิวตอนออกจากระบบ พร้อมบอกผู้ใช้ว่ามีอะไรค้างอยู่กี่รายการ ให้เขาตัดสินใจว่าจะรอส่งให้เสร็จก่อนหรือไม่

10. สรุป

  • ต้องนับไบต์เอง โดยห่อ stream ของคำขอ และอย่าลืมส่งต่อข้อมูล
  • 100% แปลว่าส่งครบ ไม่ได้แปลว่าเซิร์ฟเวอร์บันทึกแล้ว
  • การยกเลิกไม่รับประกันว่าเซิร์ฟเวอร์ไม่ได้รับ
  • แยกการถูกปฏิเสธออกจากความล้มเหลวชั่วคราว เพราะพาไปคนละทาง
  • ความล้มเหลวจากเครือข่ายตอนอัปโหลด เป็นกรณีไม่รู้ผลเสมอ
  • รหัสคำสั่งทำให้การกดส่งซ้ำปลอดภัย

บทหน้า: ตรวจไฟล์ก่อนส่ง และเส้นแบ่งระหว่างความสะดวกกับความปลอดภัย
