1. ปัญหา

import 'dart:io';

final result = await _api.uploadPartImage(
  partId: part['partId'],
  imageFile: File(picked.path),      // ← บรรทัดนี้
);

โค้ดนี้ทำงานได้ดีบน Android และ iOS แต่คอมไพล์ไม่ผ่านบนเว็บ เพราะ dart:io ไม่มีอยู่บนแพลตฟอร์มนั้น

แพลตฟอร์มFilepicked.path
Android / iOSใช้ได้ที่อยู่จริงในระบบไฟล์
Windows / macOS / Linuxใช้ได้ที่อยู่จริง
Webไม่มีเป็น blob URL ที่เปิดด้วย File ไม่ได้
ถ้าโปรเจกต์ลงเฉพาะ PDA Android อย่างเดียว การใช้ File ก็ไม่ผิด — แต่ต้องรู้ตัวว่านี่คือการตัดสินใจ ไม่ใช่ค่าปริยาย และวันที่มีคนอยากรันบน Chrome เพื่อทดสอบ จะติดตรงนี้

2. ทางที่ทำงานได้ทุกที่

/// เก็บเป็น**ไบต์** ไม่ใช่ที่อยู่ไฟล์ เพราะบนเว็บไม่มีระบบไฟล์ให้อ่าน
/// และ `dart:io` ใช้ไม่ได้เลย — โมเดลที่พูดถึงไบต์ทำงานได้ทุกแพลตฟอร์ม
class PickedImage {
  final Uint8List bytes;
  final String fileName;
  final String mimeType;
  final int? width;
  final int? height;
}

image_picker คืน XFile ซึ่งออกแบบมาเพื่อเรื่องนี้โดยเฉพาะ

final picked = await ImagePicker().pickImage(source: source, ...);
if (picked == null) return null;

return PickedImage(
  bytes: await picked.readAsBytes(),   // ← ใช้ได้ทุกแพลตฟอร์ม
  fileName: picked.name,
  mimeType: picked.mimeType ?? 'image/jpeg',
);
XFileบนมือถือบนเว็บ
.pathที่อยู่จริงblob URL ที่ใช้กับ File ไม่ได้
.readAsBytes()อ่านไฟล์อ่านจาก blob
.nameชื่อไฟล์ชื่อไฟล์
.mimeTypeอาจเป็น nullมักมีค่า

3. ราคาของการเก็บเป็นไบต์

เก็บที่อยู่ไฟล์เก็บไบต์
หน่วยความจำไม่กิน จนกว่าจะอ่านกินเท่าขนาดไฟล์ตลอดเวลาที่ถืออยู่
ทำงานบนเว็บไม่ได้ได้
ส่งหลายไฟล์พร้อมกันไม่กินเพิ่มกินรวมกันทั้งหมด
ทดสอบต้องมีไฟล์จริงสร้าง Uint8List ในเทสต์ได้เลย

แถวแรกกับแถวสามคือราคาจริง — รูปที่ย่อแล้วราว 300 KB ไม่มีปัญหา แต่ถ้าถ่ายสิบใบแล้วถือไว้พร้อมกันคือ 3 MB ซึ่งยังไหว ส่วนวิดีโอ 50 MB คือคนละเรื่อง

ประเภทไฟล์ควรเก็บเป็น
รูปที่ย่อแล้วไบต์ — ง่ายและปลอดภัย
รูปเต็มความละเอียดหลายใบที่อยู่ไฟล์ อ่านตอนจะส่ง
วิดีโอหรือไฟล์ใหญ่ที่อยู่ไฟล์เสมอ และส่งแบบ stream

แถวสุดท้ายสำคัญ — การอ่านวิดีโอ 200 MB เข้าหน่วยความจำจะทำให้แอปถูกระบบปิดทันทีบน PDA ที่แรมน้อย

4. แล้วถ้าต้องรองรับทั้งสองแบบ

sealed class UploadPayload {
  const UploadPayload();
}

/// ไฟล์เล็ก อ่านเข้าหน่วยความจำแล้ว
class BytesPayload extends UploadPayload {
  const BytesPayload(this.bytes);
  final Uint8List bytes;
}

/// ไฟล์ใหญ่ อ่านตอนส่งเป็น stream — ใช้ได้เฉพาะแพลตฟอร์มที่มีระบบไฟล์
class FilePayload extends UploadPayload {
  const FilePayload(this.path, this.length);
  final String path;
  final int length;
}
final file = switch (payload) {
  BytesPayload(:final bytes) =>
      http.MultipartFile.fromBytes('file', bytes, filename: name),
  FilePayload(:final path, :final length) =>
      http.MultipartFile('file', File(path).openRead(), length, filename: name),
};

แต่อย่าเพิ่งทำถ้ายังไม่มีไฟล์ใหญ่ในระบบ — lab เลือกทางไบต์อย่างเดียวเพราะรูปที่ย่อแล้วเล็กพอ ตามหลักใน 9.12

5. แยกโค้ดตามแพลตฟอร์มเมื่อจำเป็นจริง

ถ้าต้องใช้ dart:io จริงในบางที่ Dart มีวิธีแยกให้

// file_saver.dart — ไฟล์ที่คนอื่น import
export 'file_saver_stub.dart'
    if (dart.library.io) 'file_saver_io.dart'
    if (dart.library.js_interop) 'file_saver_web.dart';
ไฟล์ถูกเลือกเมื่อ
file_saver_io.dartมี dart:io — มือถือและเดสก์ท็อป
file_saver_web.dartคอมไพล์เป็น JavaScript
file_saver_stub.dartที่เหลือ — โยนบอกว่าไม่รองรับ

ทั้งสามไฟล์ต้องมี API หน้าตาเหมือนกันเป๊ะ ไม่งั้นจะคอมไพล์ไม่ผ่านบนบางแพลตฟอร์มเท่านั้น ซึ่งเป็นข้อผิดพลาดที่หาเจอตอน build ครั้งสุดท้าย

วิธีนี้มีราคาสูงกว่าที่คิด — ต้องดูแลสามไฟล์ ทดสอบสามแพลตฟอร์ม และ IDE ช่วยได้ไม่เต็มที่ ใช้เมื่อไม่มีทางอื่นจริง ๆ ส่วนกรณีรูปภาพ การเก็บเป็นไบต์แก้ปัญหาได้โดยไม่ต้องแยกเลย

6. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
คอมไพล์ไม่ผ่านบนเว็บimport 'dart:io'เก็บเป็นไบต์ หรือแยกโค้ดตามแพลตฟอร์ม
ไฟล์หายก่อนอัปโหลดระบบล้างโฟลเดอร์ชั่วคราวหลังเลือกรูปอ่านไบต์ทันทีที่ได้ อย่ารอ
แอปถูกระบบปิดอ่านไฟล์ใหญ่เข้าหน่วยความจำส่งเป็น stream จากที่อยู่ไฟล์
เทสต์ต้องมีไฟล์จริงในเครื่องโมเดลพูดถึงที่อยู่ไฟล์โมเดลพูดถึงไบต์
mimeType เป็น nullบางแพลตฟอร์มไม่ได้ให้มาเดาจากนามสกุล แล้วตรวจจากเนื้อไฟล์ (13.9)

แถวที่สองเกิดจริงและหายาก — ระบบปฏิบัติการล้างโฟลเดอร์ชั่วคราวได้ทุกเมื่อ ไฟล์ที่ pickImage คืนมาจึงไม่รับประกันว่าจะอยู่จนถึงตอนที่กดส่ง โดยเฉพาะถ้าผู้ใช้เลือกรูปแล้วสลับไปแอปอื่นสักพัก

7. แบบฝึกหัด

ทีมอยากให้ทดสอบแอปบน Chrome ได้ เพื่อให้คนที่ไม่มี PDA ช่วยทดสอบได้ด้วย ตอนนี้ฟีเจอร์อัปโหลดใช้ File อยู่

  1. ต้องแก้อะไรบ้าง
  2. มีอะไรที่จะยังใช้ไม่ได้บนเว็บอยู่ดี
แนวคำตอบ

1. จุดที่ต้องแก้

ไฟล์แก้อะไร
ตัวห่อ pickerคืน PickedImage ที่มีไบต์ แทน File
_uploadFileMultipartFile.fromBytes แทน fromPath
จอที่แสดงตัวอย่างImage.memory(bytes) แทน Image.file(file)
import 'dart:io' ทุกที่ลบออก

งานไม่มากเพราะการเปลี่ยนจาก path เป็น bytes เป็นการเปลี่ยนที่ขอบเดียว — และหลังจากนั้นโค้ดจะทดสอบง่ายขึ้นด้วย

2. สิ่งที่ยังใช้ไม่ได้บนเว็บ และต้องบอกทีมล่วงหน้า

ฟีเจอร์สถานะบนเว็บ
เครื่องสแกนบาร์โค้ดของ PDAไม่มี — เป็นฮาร์ดแวร์ของเครื่อง
กล้องใช้ได้ แต่ต้องเป็น HTTPS หรือ localhost
การเก็บข้อมูลลงเครื่องใช้ได้ แต่เก็บใน browser storage ซึ่งผู้ใช้ล้างได้
การเรียก API ข้ามโดเมนติด CORS — เซิร์ฟเวอร์ต้องอนุญาต

แถวสุดท้ายเป็นเรื่องที่มักทำให้เสียเวลาครึ่งวันโดยไม่คาดคิด — บนมือถือไม่มีเรื่อง CORS เลย แต่บนเว็บ เบราว์เซอร์จะบล็อกคำขอทั้งหมดจนกว่าฝั่ง .NET จะเพิ่มการอนุญาต

คำถามที่ควรถามก่อนลงมือ: เป้าหมายจริงคือ “ให้คนที่ไม่มี PDA ช่วยทดสอบ” ซึ่งอาจแก้ได้ด้วยการรัน Android emulator บนคอมของเขา ซึ่งได้ผลที่ตรงกับของจริงมากกว่า และไม่ต้องแก้โค้ดเลย

ถ้าเป้าหมายคือ “อยากให้แอปนี้ใช้บนเว็บได้จริง” นั่นเป็นคนละเรื่องและต้องคุยกันอีกแบบ — เพราะ PDA กับเบราว์เซอร์มีวิธีทำงานต่างกันมาก ตามที่ Part 2 พูดถึงเรื่องการออกแบบสำหรับการสแกน

8. สรุป

  • dart:io ไม่มีบนเว็บ การใช้ File คือการตัดสินใจ ไม่ใช่ค่าปริยาย
  • เก็บเป็นไบต์ทำงานได้ทุกแพลตฟอร์มและทดสอบง่าย
  • แลกด้วยหน่วยความจำ ซึ่งไม่มีปัญหากับรูปที่ย่อแล้ว แต่มีกับไฟล์ใหญ่
  • ไฟล์ชั่วคราวหายได้ อ่านไบต์ทันทีที่ได้มา
  • แยกโค้ดตามแพลตฟอร์มมีราคา ใช้เมื่อไม่มีทางอื่น
