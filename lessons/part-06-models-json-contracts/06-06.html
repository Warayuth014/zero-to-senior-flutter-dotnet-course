1. ปัญหาที่บทนี้แก้

เซิร์ฟเวอร์ส่งเวลามาแบบนี้

"createdAt": "2026-08-15T10:00:00"

คำถามที่ตอบไม่ได้ — 10 โมงของที่ไหน

ถ้าตีความว่าเวลาไทยคือ
UTC17:00
เวลาไทยอยู่แล้ว10:00
เวลาของเซิร์ฟเวอร์ที่ตั้งเป็น UTC+809:00

ต่างกันเจ็ดชั่วโมง ซึ่งบนงานคลังแปลว่า “ของถึงตอนเช้า” กับ “ของถึงตอนเย็น”

สิ่งที่ทำให้บั๊กนี้ร้ายคือ มันดูถูกต้องเสมอบนเครื่องพัฒนา เพราะเครื่องพัฒนา เซิร์ฟเวอร์ทดสอบ และคนทดสอบ อยู่เขตเวลาเดียวกันหมด

2. คำศัพท์ที่ต้องรู้ก่อน

คำความหมาย
UTCเวลามาตรฐานที่ทั้งโลกใช้อ้างอิง
offsetระยะห่างจาก UTC เช่น +07:00
ISO-8601รูปแบบมาตรฐานของการเขียนเวลา
Zตัวอักษรที่แปลว่า UTC
local timeเวลาตามเขตเวลาของเครื่อง
date-onlyวันที่ล้วน ไม่มีเวลา

3. Mental model: เวลาเดียว หลายวิธีเขียน

ช่วงเวลาเดียวกัน เขียนได้หลายแบบ

"2026-08-15T03:00:00Z"        ← UTC
"2026-08-15T10:00:00+07:00"   ← เวลาไทย ชี้จุดเดียวกัน
"2026-08-15T11:00:00+08:00"   ← เวลาสิงคโปร์ ชี้จุดเดียวกัน

"2026-08-15T10:00:00"         ← ชี้จุดไหนก็ไม่รู้
ประโยคเดียวที่ต้องจำ: เวลาที่ไม่มีเขตเวลา ไม่ใช่เวลา — มันคือข้อความที่หน้าตาเหมือนเวลา และต้องปฏิเสธตั้งแต่ขอบ

4. ปฏิเสธเวลาที่ตีความไม่ได้

เปิด lib/json_basics/json_read.dart

/// เวลาต้องมี Z หรือ offset เสมอ
///
/// "2026-08-15T10:00:00" ตีความไม่ได้ว่าเป็นเวลาที่ไหน — คลังที่อยู่คนละ
/// เขตเวลาจะอ่านได้คนละความหมาย จึงต้องปฏิเสธตั้งแต่ขอบ
final RegExp _hasZone = RegExp(r'(Z|[+-]\d{2}:?\d{2})$');

DateTime requireUtcTime(Map<String, dynamic> json, String key, {
  String path = '',
}) {
  final raw = requireString(json, key, path: path);
  if (!_hasZone.hasMatch(raw)) {
    throw ContractException(
      _at(path, key),
      'เวลาต้องระบุเขตเวลาด้วย Z หรือ offset แต่ได้ "$raw"',
    );
  }
  final parsed = DateTime.tryParse(raw);
  if (parsed == null) {
    throw ContractException(_at(path, key), 'รูปแบบเวลาไม่ถูกต้อง: "$raw"');
  }
  return parsed.toUtc();
}

ทำไมต้องตรวจเองทั้งที่ DateTime.parse รับได้

DateTime.parse('2026-08-15T10:00:00')
// ได้ DateTime ที่ isUtc เป็น false โดยไม่บ่นอะไรเลย
// แล้ว toUtc() จะแปลงตามเขตเวลาของเครื่องที่รันอยู่

ผลคือ ผู้ใช้คนละเครื่องจะเห็นเวลาต่างกัน จากข้อมูลก้อนเดียวกัน ซึ่งเป็นบั๊กที่ทำซ้ำไม่ได้ถ้าไม่ได้ตั้งเขตเวลาเครื่องให้ต่างกัน

5. แปลงเป็น UTC ทันทีที่รับเข้ามา

return parsed.toUtc();

ไม่ว่าเซิร์ฟเวอร์จะส่ง Z หรือ +07:00 มา ในแอปจะเป็น UTC เสมอ

test('แปลง offset เป็น UTC ให้เสมอ', () {
  final dto = TaskDto.fromJson(
    taskJson({'createdAt': '2026-08-15T10:00:00+07:00'}),
  );

  expect(dto.createdAt.isUtc, isTrue);
  expect(dto.createdAt, DateTime.utc(2026, 8, 15, 3));
});
ได้อะไรจากการเก็บเป็น UTC เสมอ
เปรียบเทียบเวลาสองค่าได้ตรง ๆ โดยไม่ต้องคิดเรื่องเขตเวลา
เรียงลำดับได้ถูกต้องเสมอ
คำนวณระยะเวลาได้ถูก แม้ข้ามช่วงเปลี่ยนเวลาตามฤดูกาล
บันทึก log แล้วเทียบกับฝั่งเซิร์ฟเวอร์ได้ทันที

ในไฟล์ตัวอย่างของ lab มีทั้งสองแบบปนกันโดยตั้งใจ

"createdAt": "2026-08-15T03:00:00Z"
"createdAt": "2026-08-15T03:05:00+07:00"

เพื่อยืนยันว่าตัวแปลงรับได้ทั้งคู่ และให้ผลเป็น UTC เหมือนกัน

6. แปลงเป็นเวลาท้องถิ่นตอนแสดงเท่านั้น

// ในข้อมูล — UTC เสมอ

// ในหน้าจอ — แปลงตอนแสดง
Text(formatTime(task.createdAt.toLocal()))
ชั้นใช้เวลาแบบไหน
รับจาก JSONแปลงเป็น UTC ทันที
เก็บใน storeUTC
เปรียบเทียบและคำนวณUTC
แสดงบนหน้าจอแปลงเป็นเวลาท้องถิ่น
ส่งกลับขึ้นเซิร์ฟเวอร์UTC พร้อม Z
// ตอนส่ง
'completedAt': completedAt.toUtc().toIso8601String(),
// ได้ "2026-08-15T03:00:00.000Z"
อย่าบวกหรือลบชั่วโมงเอง — โค้ดแบบ time.add(const Duration(hours: 7)) จะพังทันทีที่แอปถูกใช้ในเขตเวลาอื่น และจะพังสองเท่าถ้ามีคนเผลอทำซ้ำอีกชั้นในโค้ดคนละที่

7. วันที่ล้วนไม่ใช่เวลา

บาง field เป็นวันที่จริง ๆ ไม่มีเวลา เช่นวันหมดอายุของสินค้า

"expiryDate": "2026-12-31"

ถ้าเอาไปแปลงเป็นเวลา จะเกิดปัญหาทันที

DateTime.parse('2026-12-31').toUtc()
// ได้ 2026-12-31T00:00:00Z
// พอแสดงเป็นเวลาไทย กลายเป็น 2026-12-31 07:00 ซึ่งยังโอเค

// แต่ถ้าเซิร์ฟเวอร์อยู่ฝั่งตะวันตกและส่ง 2026-12-31T00:00:00-05:00
// แปลงเป็นเวลาไทยจะกลายเป็น 2026-12-31 12:00 ก็ยังโอเค
// ปัญหาคือถ้าเป็น 2026-12-31T00:00:00+13:00 → เวลาไทยคือ 2026-12-30
// วันหมดอายุเลื่อนไปหนึ่งวัน
ทางแก้วิธี
เก็บเป็นข้อความfinal String expiryDate; แล้วแสดงตรง ๆ
เก็บเป็นสามตัวเลขปี เดือน วัน แยกกัน
ใช้ชนิดของตัวเองคลาส PlainDate ที่ไม่มีเวลาและเขตเวลา

สิ่งที่ห้ามทำคือ ยัดวันที่ล้วนให้กลายเป็นเวลาเที่ยงคืน UTC แล้วหวังว่าจะไม่มีใครแปลงกลับ

เกณฑ์ตัดสิน — ถามว่า ค่านี้เปลี่ยนตามที่คนดูอยู่ไหม

ข้อมูลเปลี่ยนตามผู้ดูเก็บเป็น
เวลาที่ปิดงานเปลี่ยน — คนละเขตเวลาเห็นคนละเวลาเวลาพร้อมเขตเวลา
วันหมดอายุที่พิมพ์บนกล่องไม่เปลี่ยน — เขียนว่า 31 ธ.ค. ก็คือ 31 ธ.ค.วันที่ล้วน

8. อย่าเรียก DateTime.now() กระจายทั่วแอป

// เลี่ยง — ทดสอบไม่ได้
bool get isOverdue => DateTime.now().isAfter(dueAt);

ปัญหาคือเทสต์จะให้ผลต่างกันตามวันที่รัน และทดสอบกรณี “เลยกำหนดพอดี” ไม่ได้เลย

// ดีกว่า — รับเวลาปัจจุบันเข้ามา
bool isOverdueAt(DateTime now) => now.isAfter(dueAt);
test('งานที่เลยกำหนดแล้ว', () {
  final task = ...;   // dueAt = 2026-08-15T03:00:00Z
  expect(task.isOverdueAt(DateTime.utc(2026, 8, 15, 4)), isTrue);
  expect(task.isOverdueAt(DateTime.utc(2026, 8, 15, 2)), isFalse);
});

เป็นหลักเดียวกับการฉีด repository ใน 4.5 — สิ่งที่ควบคุมไม่ได้ ให้รับเข้ามาแทนที่จะไปหยิบเอง

9. ความละเอียดที่ต้องตกลงกัน

ฝั่งเก็บละเอียดแค่ไหน
Dart DateTimeไมโครวินาที
.NET DateTimeOffsetละเอียดกว่านั้นอีก
SQL Server datetime แบบเก่าประมาณ 3 มิลลิวินาที

ผลคือเวลาที่ส่งขึ้นไปกับที่อ่านกลับมา อาจไม่เท่ากันเป๊ะ

ส่งขึ้น  2026-08-15T03:00:00.123456Z
อ่านกลับ 2026-08-15T03:00:00.123000Z    ← ปัดตอนบันทึกลงฐานข้อมูล

ถ้ามีโค้ดที่เทียบเวลาแบบ == เพื่อดูว่าข้อมูลเปลี่ยนหรือยัง จะได้ผลว่าเปลี่ยนตลอดเวลา

ทางแก้คือตกลงความละเอียดกันไว้ แล้วปัดให้ตรงกันตั้งแต่ก่อนส่ง — หรือเลิกใช้เวลาเป็นตัวเทียบว่าข้อมูลเปลี่ยนหรือไม่ ซึ่งมักเป็นทางที่ดีกว่า

10. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
เวลาเพี้ยนไป 7 ชั่วโมงเซิร์ฟเวอร์ส่งเวลาไม่มีเขตเวลาปฏิเสธที่ขอบ แล้วแจ้งหลังบ้าน
เวลาเพี้ยน 14 ชั่วโมงมีคนบวกเองสองที่อย่าบวกเอง ใช้ toLocal()
ผู้ใช้คนละเครื่องเห็นเวลาต่างกันเก็บเป็นเวลาท้องถิ่นเก็บเป็น UTC เสมอ
วันหมดอายุเลื่อนไปหนึ่งวันยัดวันที่ล้วนเป็นเวลาเก็บวันที่ล้วนแยกจากเวลา
เทสต์ผ่านวันนี้ พรุ่งนี้แดงเรียก DateTime.now() ในตรรกะรับเวลาปัจจุบันเข้ามา
ระบบคิดว่าข้อมูลเปลี่ยนตลอดเวลาเทียบเวลาที่ความละเอียดไม่ตรงกันตกลงความละเอียด หรือเลิกใช้เวลาเป็นตัวเทียบ
เรียงลำดับผิดเมื่อข้ามเขตเวลาเรียงด้วยข้อความแทนที่จะเป็นเวลาแปลงเป็น UTC ก่อนเรียง

11. แบบฝึกหัด

ทีมหลังบ้านส่งสัญญาของใบรับเข้ามาให้ตรวจ

{
  "orderNo": "IN-2026-0812",
  "createdAt": "2026-08-15T10:30:00",
  "expectedDate": "2026-08-16",
  "receivedAt": "2026-08-16T08:15:00+07:00",
  "expiryDate": "2027-02-28T00:00:00Z"
}
  1. ระบุปัญหาของแต่ละ field
  2. เขียนสัญญาที่ควรเป็น พร้อมเหตุผลที่จะอธิบายให้ทีมหลังบ้านฟัง
  3. ตอบว่าถ้าทีมหลังบ้านแก้ไม่ได้ในตอนนี้ ฝั่งแอปควรทำอย่างไร
ดูแนวคำตอบและเหตุผล

ข้อ 1 — ปัญหาของแต่ละ field

fieldปัญหาความเสียหาย
createdAtไม่มีเขตเวลาตีความได้หลายแบบ ต่างกันได้ถึง 26 ชั่วโมง
expectedDateเป็นวันที่ล้วน — ถูกแล้วไม่มี ถ้าฝั่งแอปไม่เอาไปแปลงเป็นเวลา
receivedAtถูกต้องไม่มี
expiryDateวันที่ล้วนถูกยัดให้เป็นเวลาวันหมดอายุอาจเลื่อนหนึ่งวัน

สังเกตว่าสองปัญหาเป็นคนละแบบและต้องแก้คนละทาง

createdAtexpiryDate
ผิดอย่างไรเป็นเวลาแต่ขาดข้อมูลเป็นวันที่แต่แกล้งเป็นเวลา
แก้อย่างไรเติมเขตเวลาถอดเวลาออก

ข้อ 2 — สัญญาที่ควรเป็น

{
  "orderNo": "IN-2026-0812",
  "createdAt": "2026-08-15T03:30:00Z",
  "expectedDate": "2026-08-16",
  "receivedAt": "2026-08-16T01:15:00Z",
  "expiryDate": "2027-02-28"
}

เหตุผลที่จะอธิบายให้ทีมหลังบ้าน

สำหรับ createdAt

ตอนนี้แอปได้ "2026-08-15T10:30:00" ซึ่งไม่มีอะไรบอกว่าเป็นเวลาที่ไหน แอปต้องเดา และถ้าเดาผิดเวลาจะเพี้ยนเจ็ดชั่วโมง ปัญหาคือมันดูถูกต้องบนเครื่องทดสอบเพราะเราอยู่เขตเวลาเดียวกัน แต่จะเพี้ยนทันทีถ้าเซิร์ฟเวอร์ย้ายไปคลาวด์ที่ตั้งเป็น UTC

ขอเป็น DateTimeOffset แทน DateTime ฝั่ง .NET แล้ว serialize จะได้ offset ติดมาเอง

สำหรับ expiryDate

วันหมดอายุที่พิมพ์บนกล่องคือ 28 ก.พ. ไม่ว่าใครจะดูจากที่ไหน แต่พอส่งเป็น "2027-02-28T00:00:00Z" มันกลายเป็นช่วงเวลาที่แปลงตามเขตเวลาได้ ถ้ามีสาขาที่ UTC+13 จะเห็นเป็นวันที่ 27 ก.พ.

ขอเป็นข้อความ "2027-02-28" ตรง ๆ

สังเกตว่าทั้งสองข้อ อธิบายด้วยผลที่จะเกิดกับธุรกิจ ไม่ใช่ด้วยศัพท์เทคนิค — “วันหมดอายุเลื่อนหนึ่งวัน” เข้าใจง่ายกว่า “ควรใช้ date-only contract”

ข้อ 3 — ถ้าหลังบ้านแก้ไม่ได้ตอนนี้

ทั้งสองกรณีทำได้ แต่ต้องทำอย่างมีเงื่อนไข

สำหรับ createdAt ที่ไม่มีเขตเวลา

/// รับเวลาที่ไม่มีเขตเวลา โดยตีความว่าเป็นเวลาไทย
///
/// ใช้ชั่วคราวกับ endpoint เก่าที่ยังส่ง DateTime แทน DateTimeOffset
/// (ยืนยันกับทีมหลังบ้าน 2026-08-15 ว่าเซิร์ฟเวอร์ตั้งเป็น Asia/Bangkok)
/// ต้องลบทิ้งเมื่อ endpoint เปลี่ยนเป็น DateTimeOffset แล้ว
DateTime requireLegacyBangkokTime(
  Map<String, dynamic> json,
  String key, {
  String path = '',
}) {
  final raw = requireString(json, key, path: path);
  if (_hasZone.hasMatch(raw)) return requireUtcTime(json, key, path: path);

  final naive = DateTime.tryParse(raw);
  if (naive == null) {
    throw ContractException(_at(path, key), 'รูปแบบเวลาไม่ถูกต้อง: "$raw"');
  }
  // ตีความว่าเป็นเวลาไทย แล้วแปลงเป็น UTC
  return DateTime.utc(
    naive.year, naive.month, naive.day,
    naive.hour, naive.minute, naive.second,
  ).subtract(const Duration(hours: 7));
}

บันทึกว่าใครยืนยันและเมื่อไหร่ถ้าสมมติฐานผิด จะรู้ว่าต้องไปถามใคร
เขียนไว้ว่าต้องลบเมื่อไหร่ไม่งั้นมันจะอยู่ตลอดไป

และต้องบันทึก log ทุกครั้งที่ใช้ทางนี้ เพื่อรู้ว่ายังมี endpoint ไหนเหลืออยู่

สำหรับ expiryDate ที่มีเวลาติดมา

/// ตัดส่วนเวลาทิ้ง เก็บเฉพาะวันที่ตามที่เซิร์ฟเวอร์เขียนมา
///
/// ไม่แปลงเขตเวลาเด็ดขาด เพราะ "28 ก.พ." คือ 28 ก.พ. ไม่ว่าใครดู
String requireDateOnly(Map<String, dynamic> json, String key, {
  String path = '',
}) {
  final raw = requireString(json, key, path: path);
  final datePart = raw.split('T').first;
  if (!RegExp(r'^\d{4}-\d{2}-\d{2}$').hasMatch(datePart)) {
    throw ContractException(_at(path, key), 'ต้องเป็นวันที่ yyyy-MM-dd: "$raw"');
  }
  return datePart;
}

วิธีนี้ปลอดภัยกว่าอันแรก เพราะ ไม่ได้เดาอะไรเลย — แค่ตัดส่วนที่ไม่ควรมีอยู่แต่แรกทิ้ง

ผิดแล้วรู้ตัวไหมไม่รู้ — เวลาเพี้ยนแต่ดูเหมือนปกติรู้ทันที ถ้ารูปแบบไม่ตรง

หลักที่ได้: ทางลัดที่ต้องเดาสมมติฐาน ต้องมีวันหมดอายุ ส่วนทางลัดที่แค่ทำความสะอาดข้อมูล อยู่ได้นานกว่า — และทั้งสองแบบต้องเขียนเหตุผลไว้ในโค้ด ไม่ใช่ในหัวคนเขียน

12. สรุปท้ายบท

  • เวลาที่ไม่มีเขตเวลา ไม่ใช่เวลา — ต้องปฏิเสธที่ขอบ
  • DateTime.parse รับเวลาที่ไม่มีเขตเวลาโดยไม่บ่น จึงต้องตรวจเอง
  • แปลงเป็น UTC ทันทีที่รับเข้ามา และเก็บเป็น UTC ตลอด
  • แปลงเป็นเวลาท้องถิ่นตอนแสดงเท่านั้น
  • อย่าบวกลบชั่วโมงเอง
  • วันที่ล้วนไม่ใช่เวลา — เกณฑ์คือค่านั้นเปลี่ยนตามที่คนดูอยู่ไหม
  • รับเวลาปัจจุบันเข้ามาแทนการเรียก DateTime.now() ในตรรกะ
  • ตกลงความละเอียดของเวลากับฝั่งหลังบ้าน
