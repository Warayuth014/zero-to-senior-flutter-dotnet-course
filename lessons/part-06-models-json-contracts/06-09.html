1. ปัญหาที่บทนี้แก้

status ที่ยังเป็นข้อความ จะกระจายไปทั่วแอป

if (task.status == 'PENDING') { ... }
if (task.status == 'pending') { ... }      // อีกที่หนึ่งเขียนตัวเล็ก
if (task.status != 'DONE') { ... }
Color get color => status == 'WORKING' ? Colors.blue : Colors.grey;
ปัญหาผลที่เกิด
พิมพ์ผิดแล้วไม่มีอะไรเตือน'PENDNG' จะไม่เท่ากับอะไรเลยตลอดกาล
ตัวพิมพ์ไม่ตรงกันเงื่อนไขเดียวกันให้ผลต่างกันคนละที่
เพิ่มสถานะใหม่แล้วไม่รู้ว่าต้องแก้ตรงไหนต้องค้นทั้งโปรเจกต์แล้วหวังว่าครบ
ไม่รู้ว่ามีสถานะอะไรบ้างต้องอ่านโค้ดทั้งหมดถึงจะรู้

คำความหมาย

3. Mental model: แปลครั้งเดียวที่ขอบ

"PENDING"  ─┐
"pending"   ├─ แปลที่เดียว ─→  TaskStatus.pending  ─→ ใช้ทั้งแอป
"Pending"  ─┘
ประโยคเดียวที่ต้องจำ: ข้อความสถานะควรมีอยู่แค่ในไฟล์เดียวคือไฟล์ที่แปลมัน — ถ้าเจอ 'PENDING' ที่ไฟล์อื่น แปลว่ามีอะไรรั่ว

4. enum ที่พกค่าบนสายมาด้วย

เปิด lib/json_basics/task_domain.dart

enum TaskStatus {
  pending('PENDING'),
  working('WORKING'),
  done('DONE'),
  unknown('');

  const TaskStatus(this.wireValue);

  /// ค่าที่ใช้คุยกับเซิร์ฟเวอร์ — แยกจากชื่อใน Dart โดยตั้งใจ
  /// ถ้าใช้ `name` ตรง ๆ วันที่มีคนเปลี่ยนชื่อ enum จะพังทั้งสัญญา
  final String wireValue;
}

ทำไมไม่ใช้ name ที่ Dart ให้มาฟรี

TaskStatus.pending.name    // ได้ 'pending'
ถ้าใช้ nameผลที่ตามมา
เปลี่ยนชื่อ pending เป็น waiting ให้ตรงกับภาษาธุรกิจสัญญากับเซิร์ฟเวอร์เปลี่ยนตามทันที โดยไม่มีอะไรเตือน
เซิร์ฟเวอร์ใช้ตัวใหญ่ แต่ Dart ต้องใช้ตัวเล็กต้องแปลงตัวพิมพ์ทุกครั้งอยู่ดี

การแยก wireValue ออกมาทำให้ ชื่อในโค้ดเปลี่ยนได้อย่างอิสระ โดยไม่กระทบสัญญา

5. ตัวแปลงที่รับชื่อเล่นได้

TaskStatus parseTaskStatus(String raw) => switch (raw.trim().toUpperCase()) {
  'PENDING' => TaskStatus.pending,
  'WORKING' => TaskStatus.working,
  // ระบบเก่าส่ง COMPLETED แทน DONE — รวมชื่อเล่นไว้ที่เดียว
  'DONE' || 'COMPLETED' => TaskStatus.done,
  _ => TaskStatus.unknown,
};
สิ่งที่ทำเหตุผล
trim()บางระบบมีช่องว่างติดมาจากฐานข้อมูล
toUpperCase()รับได้ทั้ง PENDING และ pending
'DONE' || 'COMPLETED'ระบบเก่าและใหม่ใช้คนละคำ แต่หมายถึงเรื่องเดียวกัน
_ => unknownค่าที่ไม่รู้จักต้องไม่ทำให้พัง — ดูหัวข้อ 6
test('แปลงค่าที่รู้จัก และรับชื่อเล่นของระบบเก่า', () {
  expect(parseTaskStatus('PENDING'), TaskStatus.pending);
  expect(parseTaskStatus('working'), TaskStatus.working);
  expect(parseTaskStatus('DONE'), TaskStatus.done);
  expect(parseTaskStatus('COMPLETED'), TaskStatus.done);
});

6. unknown คือสถานะที่ต้องมี

คำถามที่ต้องตอบ — เมื่อเซิร์ฟเวอร์ส่ง "PAUSED" ที่แอปรุ่นนี้ไม่รู้จัก ควรทำอย่างไร

ทางเลือกผลที่เกิด
โยนข้อผิดพลาดรายการทั้งหน้าโหลดไม่ได้ ทั้งที่มีงานเดียวที่สถานะใหม่
เดาเป็น pendingคนหน้างานไปทำงานที่ถูกพักไว้
ใช้ unknownเห็นงานนั้น แต่ทำอะไรกับมันไม่ได้

ทางที่สามถูกต้อง และเหตุผลอยู่ที่ธรรมชาติของ PDA — เครื่องในคลังอัปเดตแอปช้ากว่าเซิร์ฟเวอร์เสมอ

เซิร์ฟเวอร์อัปเดตวันจันทร์ → เพิ่มสถานะ PAUSED
PDA 40 เครื่องอัปเดตวันศุกร์ → สี่วันที่เจอค่าที่ไม่รู้จัก

ถ้าเลือกทางแรก คลังจะใช้งานไม่ได้สี่วัน

แต่ unknown ต้องมาพร้อมการจำกัดสิ่งที่ทำได้

/// สถานะที่ไม่รู้จัก ห้ามให้ทำงานต่อ เพราะไม่รู้ว่ามันแปลว่าอะไร
bool get canWorkOn => this == TaskStatus.pending || this == TaskStatus.working;
test('ค่าที่ไม่รู้จักต้องไม่ทำให้พัง แต่ต้องทำงานต่อไม่ได้', () {
  final status = parseTaskStatus('PAUSED');

  expect(status, TaskStatus.unknown);
  expect(status.canWorkOn, isFalse);
});

สังเกตว่า canWorkOn เขียนเป็น รายชื่อที่อนุญาต ไม่ใช่รายชื่อที่ห้าม

// อันตราย — สถานะใหม่ที่เพิ่มมาจะ "อนุญาต" โดยอัตโนมัติ
bool get canWorkOn => this != TaskStatus.done && this != TaskStatus.unknown;

7. อ่านสถานะที่ไม่รู้จัก ให้ผู้ใช้เข้าใจ

สิ่งที่ต้องทำเพราะ
แสดงว่า “สถานะไม่รู้จัก” ตรง ๆอย่าแสดงเป็นช่องว่าง — ผู้ใช้จะคิดว่าแอปโหลดไม่ครบ
ปิดปุ่มที่ทำงานกับงานนั้นไม่รู้ว่าทำได้หรือไม่
บอกให้อัปเดตแอปเป็นทางแก้จริงของปัญหานี้
บันทึกว่าเจอค่าอะไรทีมจะได้รู้ว่าต้องเพิ่มสถานะไหน
ListTile(
  title: Text(task.palletCode),
  subtitle: task.status == TaskStatus.unknown
      ? const Text('สถานะใหม่ที่แอปรุ่นนี้ยังไม่รองรับ — กรุณาอัปเดตแอป')
      : Text(statusLabel(task.status)),
  enabled: task.status.canWorkOn,
)

ข้อความนี้ทำสองอย่าง — บอกว่าไม่ใช่ความผิดของผู้ใช้ และบอกว่าต้องทำอะไรต่อ ตามหลักใน 5.10

8. ส่งกลับต้องใช้ wireValue

// ผิด — ส่งชื่อใน Dart ขึ้นไป
'status': task.status.name,        // ได้ 'pending' ซึ่งเซิร์ฟเวอร์อาจไม่รู้จัก

// ถูก
'status': task.status.wireValue,   // ได้ 'PENDING'

และต้องกันไม่ให้ส่ง unknown ขึ้นไป

if (task.status == TaskStatus.unknown) {
  throw StateError('ส่งสถานะที่ไม่รู้จักกลับขึ้นเซิร์ฟเวอร์ไม่ได้');
}

เพราะ unknown มี wireValue เป็นข้อความว่าง ซึ่งจะทำให้เซิร์ฟเวอร์ตีความผิดหรือปฏิเสธ — และที่แย่กว่าคือถ้าเซิร์ฟเวอร์รับข้อความว่างไปแล้วเซ็ตสถานะเป็นค่าตั้งต้น งานที่ถูกพักไว้จะกลับมาเป็นรอทำโดยไม่มีใครสั่ง

9. สถานะจากเซิร์ฟเวอร์ ไม่ใช่สถานะของหน้าจอ

สองอย่างนี้มักถูกยุบรวมกัน

TaskStatusสถานะของ workflow (4.15)
มาจากไหนเซิร์ฟเวอร์เกิดในแอปขณะทำงาน
ตัวอย่างpending, working, doneScanPallet, ScanLocation, Submitting
อยู่ได้นานแค่ไหนข้ามการเปิดปิดแอปหายเมื่อออกจากหน้าจอ
ใครเปลี่ยนเซิร์ฟเวอร์เท่านั้นการกระทำของผู้ใช้
งาน T-001 มี status = working  (เซิร์ฟเวอร์รู้ว่ามีคนกำลังทำ)
  └─ หน้าจอกำลังอยู่ขั้น ScanLocation  (แอปรู้ว่าทำถึงไหนแล้ว)

การยุบสองอย่างนี้เข้าด้วยกันทำให้เกิดคำถามที่ตอบไม่ได้ เช่น “ถ้าผู้ใช้ปิดแอปตอนอยู่ขั้นสแกนช่องเก็บ เซิร์ฟเวอร์ควรรู้ไหม” — คำตอบคือไม่ เพราะขั้นตอนในแอปไม่ใช่เรื่องของเซิร์ฟเวอร์

10. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
เงื่อนไขเดียวกันให้ผลต่างกันคนละที่เทียบข้อความ ตัวพิมพ์ไม่ตรงแปลงเป็น enum ที่ขอบ
พิมพ์ชื่อสถานะผิดแล้วไม่มีอะไรเตือนใช้ข้อความในเงื่อนไขใช้ enum
เพิ่มสถานะแล้วลืมแก้บางหน้าจอไม่ได้ใช้ switch ที่ครบทุกทางตัด default: ออก
แอปรุ่นเก่าพังทั้งคลังเมื่อเซิร์ฟเวอร์เพิ่มสถานะโยนข้อผิดพลาดเมื่อไม่รู้จักใช้ unknown
คนหน้างานทำงานที่ถูกพักไว้เดาสถานะที่ไม่รู้จักเป็นค่าอื่นห้ามเดา
สถานะใหม่ได้สิทธิ์ทำงานโดยอัตโนมัติเขียน canWorkOn เป็นรายชื่อที่ห้ามเขียนเป็นรายชื่อที่อนุญาต
เปลี่ยนชื่อ enum แล้วสัญญาพังใช้ name เป็นค่าบนสายแยก wireValue
งานที่พักไว้กลับมาเป็นรอทำเองส่ง unknown กลับขึ้นไปกันไว้ก่อนส่ง

11. แบบฝึกหัด

เซิร์ฟเวอร์ส่งสถานะของใบเบิกมาเป็นตัวเลข

รหัสความหมาย
10ร่าง
20รออนุมัติ
30อนุมัติแล้ว
40กำลังหยิบ
90ยกเลิก

และมีข้อกำหนดว่า ผู้ใช้กดเริ่มหยิบได้เฉพาะใบที่อนุมัติแล้ว และกดยกเลิกได้เฉพาะใบที่ยังไม่เริ่มหยิบ

  1. ออกแบบ enum และตัวแปลง
  2. เขียนกฎว่ากดอะไรได้บ้าง
  3. ตอบว่าถ้าเซิร์ฟเวอร์เพิ่มรหัส 50 ที่แปลว่า “หยิบครบแล้ว รอตรวจสอบ” แอปรุ่นเก่าจะเป็นอย่างไร และควรเป็นอย่างไร
ดูแนวคำตอบและเหตุผล

ข้อ 1 — enum และตัวแปลง

enum PickRequestStatus {
  draft(10),
  pendingApproval(20),
  approved(30),
  picking(40),
  cancelled(90),
  unknown(-1);

  const PickRequestStatus(this.wireValue);

  /// รหัสที่ใช้คุยกับเซิร์ฟเวอร์
  final int wireValue;
}

PickRequestStatus parsePickRequestStatus(int code) => switch (code) {
  10 => PickRequestStatus.draft,
  20 => PickRequestStatus.pendingApproval,
  30 => PickRequestStatus.approved,
  40 => PickRequestStatus.picking,
  90 => PickRequestStatus.cancelled,
  _ => PickRequestStatus.unknown,
};

ใช้ -1 เป็น wireValue ของ unknown เพราะเป็นค่าที่เซิร์ฟเวอร์ไม่มีทางใช้ ทำให้ถ้าเผลอส่งขึ้นไปจะถูกปฏิเสธทันที แทนที่จะถูกตีความเป็นอย่างอื่น

ข้อ 2 — กฎว่ากดอะไรได้

extension PickRequestStatusRules on PickRequestStatus {
  /// เริ่มหยิบได้เฉพาะใบที่อนุมัติแล้ว
  bool get canStartPicking => this == PickRequestStatus.approved;

  /// ยกเลิกได้เฉพาะใบที่ยังไม่เริ่มหยิบ
  bool get canCancel => switch (this) {
    PickRequestStatus.draft ||
    PickRequestStatus.pendingApproval ||
    PickRequestStatus.approved => true,
    PickRequestStatus.picking ||
    PickRequestStatus.cancelled ||
    PickRequestStatus.unknown => false,
  };
}

สังเกตว่า canCancel เขียนด้วย switch ที่ระบุครบทุกค่า แทนที่จะเขียนสั้น ๆ แบบนี้

// สั้นกว่า แต่อันตราย
bool get canCancel => this != PickRequestStatus.picking
    && this != PickRequestStatus.cancelled
    && this != PickRequestStatus.unknown;

ความต่างจะเห็นชัดในข้อ 3

ข้อ 3 — เมื่อเซิร์ฟเวอร์เพิ่มรหัส 50

แอปรุ่นเก่าจะเป็นอย่างไร

parsePickRequestStatus(50)  →  PickRequestStatus.unknown
สิ่งที่เกิดถูกต้องไหม
ไม่พัง โหลดรายการได้ปกติถูก
ใบนั้นแสดงว่าสถานะไม่รู้จักถูก
canStartPicking เป็น falseถูก
canCancel เป็น falseถูก แต่เพราะเขียนแบบ switch เท่านั้น

ถ้าเขียนแบบ != ผลจะต่างออกไป — unknown ถูกกันไว้แล้วก็จริง แต่ประเด็นอยู่ที่ตอนที่ทีมเพิ่มสถานะใหม่ในแอป

// วันที่ทีมเพิ่ม pendingInspection(50) เข้าไปใน enum
enum PickRequestStatus {
  ...
  picking(40),
  pendingInspection(50),    // ← เพิ่มใหม่
  cancelled(90),
  unknown(-1);
}
เขียนแบบเกิดอะไร
switch ครบทุกค่าflutter analyze แดงทันที — บังคับให้ตัดสินใจว่าสถานะใหม่ยกเลิกได้ไหม
!=คอมไพล์ผ่าน และ canCancel กลายเป็น true เงียบ ๆ

แถวล่างคือบั๊กจริง — ใบที่หยิบครบแล้วรอตรวจสอบ จะถูกยกเลิกได้ ทั้งที่ของถูกหยิบออกจากชั้นไปแล้ว

ควรเป็นอย่างไร

ต้องทำรายละเอียด
รุ่นเก่าต้องไม่พังทำได้แล้วด้วย unknown
รุ่นเก่าต้องไม่เดาทำได้แล้ว — ปิดทุกปุ่ม
รุ่นเก่าต้องบอกผู้ใช้“สถานะใหม่ กรุณาอัปเดตแอป”
ทีมต้องรู้ว่ามีคนเจอบันทึกรหัสที่ไม่รู้จักส่งกลับมา
ตอนเพิ่มในแอป compiler ต้องช่วยใช้ switch ครบทุกค่า

ข้อสุดท้ายเป็นข้อที่คนมองข้ามบ่อยที่สุด และเป็นข้อที่ป้องกันบั๊กที่แพงที่สุด

บทเรียนที่ควรจำ: unknown ป้องกันปัญหาที่เกิด ตอนนี้ จากเซิร์ฟเวอร์ที่ล้ำหน้ากว่า ส่วน switch ที่ครบทุกค่าป้องกันปัญหาที่จะเกิด ตอนที่เราตามทัน — ต้องมีทั้งคู่ เพราะกันคนละเวลา

12. สรุปท้ายบท

  • แปลงข้อความสถานะเป็น enum ที่ขอบ ที่เดียว
  • แยก wireValue ออกจากชื่อใน Dart เพื่อให้เปลี่ยนชื่อได้อิสระ
  • รวมชื่อเล่นของระบบเก่าไว้ในตัวแปลงเดียว
  • unknown จำเป็น เพราะ PDA อัปเดตช้ากว่าเซิร์ฟเวอร์เสมอ
  • ห้ามเดาสถานะที่ไม่รู้จักเป็นค่าอื่น
  • เขียนกฎเป็นรายชื่อที่อนุญาต ไม่ใช่รายชื่อที่ห้าม
  • ใช้ switch ที่ครบทุกค่า เพื่อให้ compiler เตือนตอนเพิ่มสถานะใหม่
  • ส่งกลับด้วย wireValue และห้ามส่ง unknown ขึ้นไป
  • สถานะจากเซิร์ฟเวอร์กับสถานะของหน้าจอเป็นคนละเรื่อง
