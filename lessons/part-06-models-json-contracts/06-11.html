1. ปัญหาที่บทนี้แก้

เรื่องนี้เคยเจอมาแล้วใน 4.13 แต่ตอนนั้นแก้เฉพาะจุด — ตอนนี้ต้องทำให้เป็นกฎ

class Task {
  String id;
  int quantity;
  TaskStatus status;
}
ปัญหาที่ตามมาตัวอย่าง
สำเนาที่เก็บไว้ย้อนกลับไม่ได้final before = task; แล้ว task.quantity = 5 — before เปลี่ยนตามไปด้วย
สองหน้าจอแก้วัตถุตัวเดียวกันหน้ารายละเอียดแก้ แล้วหน้ารายการเปลี่ยนตามทันทีทั้งที่ยังไม่บันทึก
หน้าจอไม่วาดใหม่แก้ค่าในวัตถุเดิม รายการยังชี้ไปที่วัตถุเดิม ไม่มีอะไรบอกว่าเปลี่ยน
ไล่หาไม่เจอว่าใครแก้ทุกที่ที่ถือวัตถุนี้แก้ได้หมด

แก้ค่าในตัวเดิม             สร้างตัวใหม่

4. final ทุก field

เปิด lib/json_basics/task_domain.dart

class Task {
  const Task({
    required this.id,
    required this.palletCode,
    required this.quantity,
    required this.status,
    required this.createdAt,
    this.note,
  });

  final String id;
  final String palletCode;
  final int quantity;
  final TaskStatus status;
  final DateTime createdAt;
  final String? note;
}

สังเกต const ที่ constructor — ทำให้สร้างเป็นค่าคงที่ได้เมื่อค่าที่ส่งเข้ามาเป็นค่าคงที่ทั้งหมด ซึ่งช่วยเรื่องประสิทธิภาพตามที่เรียนใน 1.8

final กับ List ไม่ได้ทำให้เนื้อในแก้ไม่ได้ — final List<String> codes ยังเรียก codes.add(...) ได้ ต้องใช้ List.unmodifiable หรือรับเป็น const เพิ่ม

5. copyWith และกับดักของมัน

Task copyWith({int? quantity, TaskStatus? status}) => Task(
  id: id,
  palletCode: palletCode,
  quantity: quantity ?? this.quantity,
  status: status ?? this.status,
  createdAt: createdAt,
  note: note,
);

ใช้ได้ดีจนกว่าจะเจอกรณีนี้ — ต้องการล้างหมายเหตุให้เป็น null

task.copyWith(note: null)     // ไม่มีผลอะไรเลย

เพราะ note ?? this.note แปลว่า “ถ้าไม่ส่งมาก็ใช้ของเดิม” และการส่ง null มา หน้าตาเหมือนกับการไม่ส่งทุกประการ

copyWith()              → note เป็น null → ใช้ของเดิม
copyWith(note: null)    → note เป็น null → ใช้ของเดิม  ← ตั้งใจคนละอย่าง แต่ได้ผลเดียวกัน

ทางแก้ใน lab คือแยกเป็นเมธอดของตัวเอง

/// ล้างหมายเหตุ — ต้องมีเมธอดแยก เพราะ copyWith ตั้งค่าเป็น null ไม่ได้
/// (`note ?? this.note` แปลว่า "ไม่ส่งมา = ใช้ของเดิม" เสมอ)
Task clearNote() => Task(
  id: id,
  palletCode: palletCode,
  quantity: quantity,
  status: status,
  createdAt: createdAt,
);
test('copyWith ตั้งค่าเป็น null ไม่ได้ จึงต้องมีเมธอดล้างแยก', () {
  final task = toTask(TaskDto.fromJson(taskJson({'note': 'วางชั้นล่าง'})));

  expect(task.copyWith().note, 'วางชั้นล่าง');
  expect(task.clearNote().note, isNull);
});

6. ทางเลือกอื่นสำหรับปัญหา null

วิธีข้อดีข้อเสีย
เมธอดแยก เช่น clearNote()อ่านง่ายที่สุด เจตนาชัดต้องเขียนเพิ่มทุก field ที่ล้างได้
ใช้ค่าพิเศษเป็นตัวแทน “ไม่ส่ง”ใช้ copyWith ตัวเดียวได้อ่านยาก ต้องเข้าใจกลไก
ให้เครื่องมือสร้างให้จัดการให้ครบโดยไม่ต้องเขียนต้องเพิ่มเครื่องมือ (Part 30)
// แบบใช้ค่าพิเศษ
const Object _keep = Object();

Task copyWith({Object? note = _keep}) => Task(
  ...
  note: identical(note, _keep) ? this.note : note as String?,
);

เขียนได้ แต่ทุกคนที่อ่านต้องเข้าใจว่า _keep คืออะไร — สำหรับ model ที่มี field ที่ล้างได้ไม่กี่ตัว เมธอดแยกอ่านง่ายกว่ามาก

เกณฑ์เลือก — ถ้ามี field ที่ล้างได้ไม่เกินสองสามตัว ใช้เมธอดแยก ถ้ามีเยอะ ใช้เครื่องมือสร้าง

7. เท่ากันเมื่อค่าเท่ากัน

โดยค่าเริ่มต้น Dart เทียบวัตถุด้วยความเป็นตัวเดียวกัน

final a = Task(id: 'T-001', ...);
final b = Task(id: 'T-001', ...);
a == b     // false — คนละวัตถุ แม้ค่าเหมือนกันทุกอย่าง

ซึ่งไม่ใช่สิ่งที่ต้องการสำหรับ model ที่แทนข้อมูล

@override
bool operator ==(Object other) =>
    identical(this, other) ||
    other is Task &&
        other.id == id &&
        other.palletCode == palletCode &&
        other.quantity == quantity &&
        other.status == status &&
        other.createdAt == createdAt &&
        other.note == note;

@override
int get hashCode =>
    Object.hash(id, palletCode, quantity, status, createdAt, note);
test('เท่ากันเมื่อค่าเท่ากัน ไม่ใช่เมื่อเป็นวัตถุตัวเดียวกัน', () {
  final a = toTask(TaskDto.fromJson(taskJson()));
  final b = toTask(TaskDto.fromJson(taskJson()));

  expect(a, equals(b));
  expect(a.hashCode, equals(b.hashCode));
  expect(identical(a, b), isFalse);
});
ได้อะไรจากการเทียบด้วยค่า
เขียนเทสต์ได้ตรงไปตรงมา — เทียบทั้งวัตถุแทนที่จะเทียบทีละ field
ใช้ใน Set และเป็นคีย์ของ Map ได้
ตรวจได้ว่าข้อมูลเปลี่ยนจริงไหม ก่อนสั่งวาดใหม่
== กับ hashCode ต้องเขียนคู่กันเสมอ — ถ้าเขียนแค่ == วัตถุที่เท่ากันจะมี hashCode ต่างกัน แล้ว Set จะเก็บซ้ำได้ทั้งที่ค่าเหมือนกัน

8. รายการก็ต้องแก้ไม่ได้

// ผิด — ข้างนอกแก้ได้
List<Task> get tasks => _tasks;

// ถูก
List<Task> get tasks => List.unmodifiable(_tasks);

เป็นเรื่องเดียวกับ 4.5 — และเหตุผลไม่ใช่แค่กันคนแก้ผิด แต่เพื่อให้ ทางเดียวที่ข้อมูลจะเปลี่ยนคือผ่านเมธอดของเจ้าของ

void _setLine(int index, PickLine line) {
  final next = [..._lines];    // สำเนาใหม่
  next[index] = line;
  _lines = next;               // ชี้ไปที่รายการใหม่
}

การสร้างรายการใหม่ทุกครั้งยังทำให้เทียบได้ว่าเปลี่ยนหรือยัง ซึ่งจำเป็นสำหรับการวาดใหม่แบบเลือกจุดใน 4.6

9. DTO ก็ควรแก้ไม่ได้เหมือนกัน

บางคนคิดว่า DTO เป็นแค่ทางผ่าน จึงไม่ต้องเข้มงวด — แต่กฎเดียวกันใช้ได้ผลดีกว่า

เหตุผลรายละเอียด
ป้องกันการแก้ระหว่างแปลงไม่มีใครไปแก้ DTO ระหว่างที่ mapper กำลังอ่านอยู่
ใช้ในเทสต์ได้ง่ายสร้างครั้งเดียวแล้วใช้ซ้ำได้ทุกเทสต์
กฎเดียวทั้งโปรเจกต์ไม่ต้องจำว่าคลาสไหนแก้ได้คลาสไหนแก้ไม่ได้

ใน lab ทั้ง TaskDto และ Task เป็น const constructor ที่มี final ทุก field เหมือนกัน

แต่ TaskDto ไม่ได้เขียน == เพราะไม่มีใครต้องเปรียบเทียบ DTO — สิ่งที่ต้องเปรียบเทียบคือข้อมูลของธุรกิจ ไม่ใช่ข้อมูลบนสาย

10. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ย้อนกลับแล้วได้ค่าเดียวกับที่เปลี่ยนไปวัตถุแก้ได้ สำเนาจึงชี้ไปที่เดิมfinal ทุก field
แก้ค่าแล้วหน้าจอไม่วาดใหม่แก้ในวัตถุเดิม ไม่มีอะไรบอกว่าเปลี่ยนสร้างวัตถุใหม่
copyWith(note: null) ไม่มีผลกับดักของ ??เมธอดแยกหรือค่าพิเศษ
วัตถุที่ค่าเหมือนกันไม่เท่ากันไม่ได้เขียน ==เขียน == คู่กับ hashCode
Set เก็บวัตถุซ้ำทั้งที่ค่าเหมือนกันเขียน == แต่ลืม hashCodeเขียนทั้งคู่
final List ยังถูกแก้ได้final กันแค่การชี้ใหม่List.unmodifiable
ไม่รู้ว่าใครแก้ข้อมูลทุกที่แก้ได้หมดให้แก้ผ่านเมธอดของเจ้าของเท่านั้น

11. แบบฝึกหัด

model ของใบรับเข้าที่กำลังทำอยู่

class InboundOrder {
  InboundOrder({
    required this.orderNo,
    required this.details,
    this.assignedTo,
    this.note,
  });

  final String orderNo;
  final List<InboundDetail> details;
  String? assignedTo;
  String? note;

  void assign(String userId) {
    assignedTo = userId;
  }
}
  1. หาข้อบกพร่องให้ได้อย่างน้อยสี่ข้อ
  2. เขียนใหม่ให้แก้ไม่ได้ พร้อมเมธอดที่จำเป็น
  3. ตอบว่า == ของ InboundOrder ควรเทียบ details ด้วยหรือไม่
ดูแนวคำตอบและเหตุผล

ข้อ 1 — ข้อบกพร่อง

#ข้อบกพร่องผลที่เกิด
1assignedTo และ note ไม่เป็น finalใครก็แก้ได้ ไล่หาไม่เจอว่าใครแก้
2มีเมธอด assign ที่แก้ตัวเองสำเนาที่เก็บไว้เปลี่ยนตาม
3details เป็น final แต่เนื้อในแก้ได้เรียก details.add(...) ได้
4ไม่มี == และ hashCodeเทียบไม่ได้ ใช้ใน Set ไม่ได้
5constructor ไม่เป็น constเสียโอกาสใช้เป็นค่าคงที่
6ไม่มีทางล้าง assignedToมอบหมายแล้วยกเลิกไม่ได้

ข้อ 2 — เขียนใหม่

class InboundOrder {
  InboundOrder({
    required this.orderNo,
    required List<InboundDetail> details,
    this.assignedTo,
    this.note,
  }) : details = List.unmodifiable(details);   // แก้ข้อ 3

  final String orderNo;
  final List<InboundDetail> details;
  final String? assignedTo;                    // แก้ข้อ 1
  final String? note;

  /// มอบหมายให้ผู้ใช้คนหนึ่ง — คืนใบใหม่ ไม่แก้ใบเดิม
  InboundOrder assignTo(String userId) => InboundOrder(   // แก้ข้อ 2
    orderNo: orderNo,
    details: details,
    assignedTo: userId,
    note: note,
  );

  /// ยกเลิกการมอบหมาย — ต้องแยกเพราะ copyWith ตั้ง null ไม่ได้
  InboundOrder unassign() => InboundOrder(     // แก้ข้อ 6
    orderNo: orderNo,
    details: details,
    note: note,
  );

  InboundOrder copyWith({String? note}) => InboundOrder(
    orderNo: orderNo,
    details: details,
    assignedTo: assignedTo,
    note: note ?? this.note,
  );

  @override
  bool operator ==(Object other) =>             // แก้ข้อ 4
      identical(this, other) ||
      other is InboundOrder &&
          other.orderNo == orderNo &&
          other.assignedTo == assignedTo &&
          other.note == note;

  @override
  int get hashCode => Object.hash(orderNo, assignedTo, note);
}

สังเกตว่า constructor ยังไม่เป็น const (ข้อ 5 ยังไม่ได้แก้) เพราะ List.unmodifiable(details) ใน initializer ทำให้เป็น const ไม่ได้ — ถ้าอยากได้ const ต้องให้ผู้เรียกส่ง list ที่แก้ไม่ได้มาเอง ซึ่งเป็นการย้ายภาระไปให้ผู้เรียก

สำหรับ model ที่สร้างจาก JSON เสมอ การไม่เป็น const ไม่เสียหายอะไร เพราะยังไงก็สร้างตอนรันอยู่แล้ว — จึงเลือกความปลอดภัยแทน

ข้อ 3 — == ควรเทียบ details ไหม

โค้ดข้างบน ไม่เทียบ และนั่นเป็นการตัดสินใจที่ต้องอธิบายได้

เหตุผลที่ไม่เทียบ

เหตุผลรายละเอียด
ราคาแพงใบที่มี 200 บรรทัด ต้องเทียบ 200 ครั้งทุกครั้งที่เทียบใบ
orderNo ระบุตัวตนอยู่แล้วสองใบที่เลขที่เดียวกันคือใบเดียวกัน
ถูกเรียกบ่อยทุกครั้งที่วาดรายการใหม่

เหตุผลที่ควรเทียบ

เหตุผลรายละเอียด
ตรวจว่าข้อมูลเปลี่ยนจริงไหมถ้าบรรทัดสินค้าเปลี่ยนแต่ == บอกว่าเท่ากัน หน้าจอจะไม่วาดใหม่
ความหมายตรงไปตรงมา“เท่ากัน” ควรแปลว่าเหมือนกันทุกอย่าง

คำตอบขึ้นกับว่าใช้ == ทำอะไร

ใช้ทำอะไรควรเทียบ details ไหม
หาว่าใบไหนคือใบเดียวกัน (ใน Set, ใน Map)ไม่ต้อง — orderNo พอ
ตัดสินว่าต้องวาดใหม่ไหมต้อง — ไม่งั้นบรรทัดที่เปลี่ยนจะไม่ถูกวาด
เขียนเทสต์เทียบผลลัพธ์ต้อง

ทางออกที่ดีที่สุดคือแยกสองคำถามออกจากกัน

/// เทียบว่าเป็นใบเดียวกันไหม — ใช้เมื่อต้องการหาหรือจับคู่
bool isSameOrder(InboundOrder other) => other.orderNo == orderNo;

/// == เทียบทุกอย่างรวม details — ใช้เมื่อต้องการรู้ว่าข้อมูลเปลี่ยนไหม
@override
bool operator ==(Object other) =>
    identical(this, other) ||
    other is InboundOrder &&
        other.orderNo == orderNo &&
        other.assignedTo == assignedTo &&
        other.note == note &&
        _listEquals(other.details, details);

@override
int get hashCode => Object.hash(
  orderNo, assignedTo, note, Object.hashAll(details),
);

ทำแบบนี้แล้ว == มีความหมายตรงไปตรงมาว่า “เหมือนกันทุกอย่าง” ซึ่งเป็นสิ่งที่คนอ่านโค้ดคาดหวัง ส่วนการหาตัวตนใช้เมธอดที่บอกเจตนาชัดเจน

บทเรียนที่ควรจำ: เมื่อ == ทำสองหน้าที่ที่ขัดกัน อย่าประนีประนอมด้วยการเทียบครึ่ง ๆ — ให้ == ทำหน้าที่เดียวตามความหมายมาตรฐาน แล้วเพิ่มเมธอดที่บอกเจตนาสำหรับอีกหน้าที่

12. สรุปท้ายบท

  • วัตถุที่แก้ไม่ได้ทำให้สำเนามีความหมาย และย้อนกลับได้จริง
  • final ทุก field และใช้ const constructor เมื่อทำได้
  • final List ไม่ได้ทำให้เนื้อในแก้ไม่ได้ ต้อง List.unmodifiable
  • copyWith ตั้งค่าเป็น null ไม่ได้ เพราะ ?? แยกไม่ออกจากการไม่ส่ง
  • ใช้เมธอดแยกเมื่อมี field ที่ล้างได้ไม่กี่ตัว
  • เขียน == คู่กับ hashCode เสมอ
  • DTO ก็ควรแก้ไม่ได้ แต่ไม่จำเป็นต้องเทียบค่า
  • ถ้า == ต้องทำสองหน้าที่ ให้แยกเป็นเมธอดที่บอกเจตนา
