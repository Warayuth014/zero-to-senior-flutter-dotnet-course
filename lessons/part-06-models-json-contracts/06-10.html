1. ปัญหาที่บทนี้แก้

เมื่อรายการมีเป็นพัน เซิร์ฟเวอร์จะไม่ส่งมาทั้งหมด แต่ส่งมาทีละหน้าพร้อมบอกว่ามีทั้งหมดเท่าไหร่

{
  "items": [ ... 20 รายการ ... ],
  "totalCount": 137
}

สองตัวเลขนี้มักถูกใช้ผิดตัว

โค้ดที่เขียนผลที่เกิด
Text('มีงาน ${items.length} รายการ')แสดง 20 ทั้งที่มีจริง 137
if (items.isEmpty) แสดง "ไม่มีงาน"ถูก แต่บังเอิญถูก
if (items.length < 20) หยุดโหลดพังเมื่อหน้าสุดท้ายมี 20 พอดี

แถวแรกทำให้คนหน้างานวางแผนงานผิด เพราะคิดว่าเหลืองานอีก 20 ชิ้น ทั้งที่จริงเหลือ 137

3. Mental model: สองตัวเลขที่ตอบคนละคำถาม

items.length  → "หน้านี้มีกี่รายการ"        ใช้วนแสดงผล
totalCount    → "ทั้งหมดมีกี่รายการ"        ใช้บอกผู้ใช้และคำนวณว่าโหลดครบหรือยัง
ประโยคเดียวที่ต้องจำ: ก่อนใช้ตัวเลขใดตัวเลขหนึ่ง ให้ถามว่ากำลังตอบคำถามข้อไหน — สองตัวนี้เท่ากันเฉพาะตอนที่ข้อมูลมีหน้าเดียว ซึ่งเป็นกรณีที่เจอตอนทดสอบพอดี

4. ซองที่ตรวจตัวเองได้

เปิด lib/json_basics/task_dto.dart

class TaskPageDto {
  const TaskPageDto({required this.items, required this.totalCount});

  factory TaskPageDto.fromJson(Map<String, dynamic> json) {
    final rawItems = requireList(json, 'items');
    final items = <TaskDto>[];
    for (var index = 0; index < rawItems.length; index++) {
      final path = 'items[$index]';
      items.add(TaskDto.fromJson(requireMap(rawItems[index], path), path: path));
    }

    final totalCount = requireInt(json, 'totalCount');
    if (totalCount < items.length) {
      throw const ContractException(
        'totalCount',
        'จำนวนทั้งหมดต้องไม่น้อยกว่าจำนวนในหน้านี้',
      );
    }

    return TaskPageDto(items: items, totalCount: totalCount);
  }

  final List<TaskDto> items;
  final int totalCount;

  bool get hasMore => items.length < totalCount;
}

7. ผูกจำนวนทั้งหมดกับเงื่อนไขการค้นหา

totalCount ไม่ใช่จำนวนงานทั้งหมดในระบบ แต่เป็นจำนวนที่ตรงกับเงื่อนไขที่ถามไป

ค้นหา ""       → totalCount = 137
ค้นหา "PAL-01" → totalCount = 8

ปัญหาเกิดเมื่อผู้ใช้เปลี่ยนเงื่อนไขระหว่างที่คำขอเก่ายังไม่กลับ

0.0s  ค้นหา "" → ยิงคำขอ A
0.3s  ค้นหา "PAL-01" → ยิงคำขอ B
0.8s  B กลับ → totalCount = 8, แสดง 8 รายการ
1.5s  A กลับ → totalCount = 137, ต่อท้ายอีก 20 รายการที่ไม่ตรงเงื่อนไข

นี่คือปัญหาเดียวกับ 4.10 และแก้ด้วยตัวนับรุ่นเหมือนกัน

Future<void> search(String query) async {

  _tasks = page.items;
  _totalCount = page.totalCount;

  notifyListeners();

และตอนโหลดหน้าถัดไป ต้องส่งเงื่อนไขเดิมไปด้วยเสมอ ไม่ใช่ส่งแค่ตำแหน่ง

ต่อกับ 4.7 โดยตรง — แต่ตอนนี้มีข้อมูลมากขึ้นให้แยกกรณี

สิ่งที่ได้แปลว่าควรแสดง
items: [], totalCount: 0, ไม่มีเงื่อนไขค้นหาวันนี้ไม่มีงาน“วันนี้ยังไม่มีงาน”
items: [], totalCount: 0, มีเงื่อนไขค้นหาไม่พบผลลัพธ์“ไม่พบงานที่ตรงกับ ...” + ปุ่มล้างเงื่อนไข
items: [] แต่ totalCount: 137ขอหน้าที่เกินไปแล้วไม่ควรเกิด — เป็นบั๊กของฝั่งเรา

สองแถวแรกต้องแสดงต่างกัน เพราะผู้ใช้ต้องทำต่างกัน — แถวแรกคือรอ ส่วนแถวสองคือแก้เงื่อนไข

9. ซองที่มีรูปแบบต่างกัน

แต่ละ endpoint อาจห่อรายการไม่เหมือนกัน

{"items": [...], "totalCount": 137}
{"data": [...], "total": 137}
{"rows": [...], "recordsTotal": 137}
{"result": {"list": [...], "count": 137}}

ทางที่ดีคือ แปลงให้เป็นรูปแบบเดียวที่ขอบ แล้วชั้นอื่นเห็นแบบเดียวกันหมด

class Page<T> {
  const Page({required this.items, required this.totalCount});
  final List<T> items;
  final int totalCount;
  bool get hasMore => items.length < totalCount;
}

แล้วแต่ละ endpoint มีตัวแปลงของตัวเอง แต่คืน Page<T> เหมือนกัน

ผลคือ store และหน้าจอไม่ต้องรู้ว่า endpoint ไหนใช้ชื่ออะไร — เป็นหลักเดียวกับ 6.7

10. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
แสดงจำนวนงานน้อยกว่าความจริงใช้ items.length แทน totalCountใช้ตัวเลขให้ตรงกับคำถาม
ยิงคำขอเกินหนึ่งครั้งเสมอเดาว่าหมดจากจำนวนที่ได้เทียบกับ totalCount
รายการซ้ำเมื่อเลื่อนลงต่อท้ายโดยไม่กรองกรองด้วย id
ผลค้นหาปนกับผลก่อนหน้าไม่มีตัวนับรุ่นตรวจรุ่นก่อนเขียนลง state (4.10)
โหลดหน้าถัดไปแล้วได้ข้อมูลไม่ตรงเงื่อนไขส่งแต่ตำแหน่ง ไม่ส่งเงื่อนไขส่งเงื่อนไขไปทุกครั้ง
“ไม่พบผลลัพธ์” กับ “ยังไม่มีงาน” เหมือนกันไม่แยกกรณีค้นหาดูว่ามีเงื่อนไขค้นหาอยู่หรือไม่
ทุก endpoint มีวิธีอ่านซองของตัวเองไม่ได้แปลงเป็นรูปแบบเดียวใช้ชนิดกลางอย่าง Page<T>

11. แบบฝึกหัด

หน้าจอค้นหางานมีรายการที่โหลดเพิ่มเมื่อเลื่อนถึงท้าย และมีช่องค้นหา

class SearchStore extends ChangeNotifier {
  List<Task> _tasks = [];
  int _totalCount = 0;
  bool _loading = false;

  Future<void> search(String query) async {
    _loading = true;
    notifyListeners();
    final page = await repository.fetchTasks(query: query, skip: 0);
    _tasks = page.items;
    _totalCount = page.totalCount;
    _loading = false;
    notifyListeners();
  }

  Future<void> loadMore() async {
    if (_tasks.length >= _totalCount) return;
    final page = await repository.fetchTasks(query: '', skip: _tasks.length);
    _tasks = [..._tasks, ...page.items];
    notifyListeners();
  }
}

ข้อ 2 — เขียนใหม่

class SearchStore extends ChangeNotifier {
  List<Task> _tasks = const [];
  int _totalCount = 0;
  String _query = '';                   // แก้ข้อ 7
  bool _loading = false;
  bool _loadingMore = false;
  int _generation = 0;
  String? _errorMessage;

  List<Task> get tasks => List.unmodifiable(_tasks);
  int get totalCount => _totalCount;
  bool get hasMore => _tasks.length < _totalCount;

  Future<void> search(String query) async {
    final generation = ++_generation;   // แก้ข้อ 2
    _loading = true;
    _errorMessage = null;
    notifyListeners();
    try {
      final page = await repository.fetchTasks(query: query, skip: 0);
      if (generation != _generation) return;
      _tasks = page.items;
      _totalCount = page.totalCount;
      _query = query;                   // จำเงื่อนไขที่ใช้จริง
    } on ContractException catch (error) {
      if (generation != _generation) return;
      _errorMessage = 'ข้อมูลจากเซิร์ฟเวอร์ไม่ถูกต้อง: ${error.path}';
    } finally {                          // แก้ข้อ 5
      if (generation == _generation) {
        _loading = false;
        notifyListeners();
      }
    }
  }

  Future<void> loadMore() async {
    if (_loadingMore || _loading || !hasMore) return;   // แก้ข้อ 4
    final generation = _generation;      // ไม่เพิ่ม — เกาะรุ่นของการค้นหาปัจจุบัน
    _loadingMore = true;
    notifyListeners();
    try {
      final page = await repository.fetchTasks(
        query: _query,                   // แก้ข้อ 1
        skip: _tasks.length,
      );
      if (generation != _generation) return;   // มีการค้นหาใหม่ระหว่างรอ

      // แก้ข้อ 3 — กรองซ้ำด้วย id
      final seenIds = {for (final task in _tasks) task.id};
      _tasks = [
        ..._tasks,
        for (final task in page.items)
          if (seenIds.add(task.id)) task,
      ];
      _totalCount = page.totalCount;     // แก้ข้อ 6
    } on ContractException catch (error) {
      if (generation != _generation) return;
      _errorMessage = 'โหลดเพิ่มไม่สำเร็จ: ${error.path}';
    } finally {
      if (generation == _generation) {
        _loadingMore = false;
        notifyListeners();
      }
    }
  }
}

ข้อ 3 — เลื่อนถึงท้ายพอดีตอนพิมพ์ค้นหาใหม่

ไล่ตามลำดับเวลากับโค้ดที่แก้แล้ว

0.0s  ผู้ใช้เลื่อนถึงท้าย → loadMore() เริ่ม
      generation = 5, query = "" , skip = 20
0.2s  ผู้ใช้พิมพ์ "PAL" → search("PAL") เริ่ม
      _generation กลายเป็น 6
0.7s  search กลับมา → generation 6 == 6 → เขียนลง state
      _tasks = ผลของ "PAL" 12 รายการ, _totalCount = 12
1.4s  loadMore กลับมา → generation 5 != 6 → return ทิ้งไป ✓

ถ้าไม่มีการตรวจรุ่นใน loadMore จะเกิดสิ่งนี้แทน

1.4s  loadMore เขียนต่อท้าย
      _tasks = ผลของ "PAL" 12 รายการ + งานที่ไม่ตรงเงื่อนไขอีก 20 รายการ
      _totalCount = 137 ทับของเดิม
      → ผู้ใช้ค้นหา "PAL" แต่เห็นงานที่ไม่มี PAL อยู่ด้วย
      → และหัวรายการบอกว่ามี 137 รายการ

อาการนี้ทำซ้ำยากมากเพราะต้องพิมพ์พอดีจังหวะที่เลื่อนถึงท้าย — จึงมักหลุดขึ้น production ไปก่อน

สิ่งที่ควรทำเพิ่มเพื่อลดโอกาสเกิด

วิธีช่วยอะไร
debounce ช่องค้นหา (4.10)ลดจำนวนคำขอ แต่ไม่ได้แก้ปัญหานี้
บล็อก loadMore ระหว่างที่ _loading เป็นจริงกันไม่ให้เริ่มโหลดเพิ่มระหว่างค้นหาใหม่
ตรวจรุ่นแก้ปัญหาจริง — กันผลที่ค้างอยู่แล้ว

สองข้อแรกลดโอกาส ข้อสุดท้ายกันได้จริง — เป็นข้อสรุปเดียวกับ 4.10 ว่า debounce เป็นเรื่องประสิทธิภาพ ส่วนตัวนับรุ่นเป็นเรื่องความถูกต้อง

12. สรุปท้ายบท

  • items.length ตอบว่าหน้านี้มีกี่รายการ ส่วน totalCount ตอบว่าทั้งหมดมีกี่รายการ
  • สองค่านี้เท่ากันเฉพาะตอนมีหน้าเดียว ซึ่งเป็นกรณีที่เจอตอนทดสอบพอดี
  • ให้ซองตรวจตัวเองว่าตัวเลขไม่ขัดแย้งกัน
  • อย่าเดาว่าโหลดครบจากจำนวนที่ได้ ให้เทียบกับ totalCount
  • กรองด้วย id ก่อนต่อท้าย เพราะข้อมูลเปลี่ยนได้ระหว่างโหลด
  • totalCount ผูกกับเงื่อนไขที่ถามไป จึงต้องส่งเงื่อนไขเดิมตอนโหลดหน้าถัดไป
  • ใช้ตัวนับรุ่นกันผลของคำขอเก่า และ loadMore ต้องเกาะรุ่นเดิม ไม่ใช่เพิ่มรุ่นใหม่
  • “ไม่มีงาน” กับ “ไม่พบผลลัพธ์” ต้องแสดงต่างกัน
  • แปลงซองของทุก endpoint ให้เป็นรูปแบบเดียวที่ขอบ
