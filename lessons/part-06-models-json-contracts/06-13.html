3. Mental model: เทสต์คือที่เก็บเหตุผล

โค้ด        บอกว่า "ทำอะไร"
เทสต์       บอกว่า "ทำไมถึงต้องทำแบบนั้น"
ประโยคเดียวที่ต้องจำ: ทุกครั้งที่เขียนโค้ดที่ดูแปลก ให้ถามว่ามีเทสต์ที่จะแดงไหมถ้ามีคนแก้ให้ดูปกติ — ถ้าไม่มี วันหนึ่งจะมีคนแก้จริง

ชื่อเทสต์ใน lab จึงเขียนเป็นประโยคที่บอกเหตุผล ไม่ใช่ชื่อเมธอด

test('รับทั้ง 12, 12.0 และ "12" เพราะ JSON ไม่แยกชนิด', ...);
test('ปฏิเสธเศษ ไม่ปัดให้เงียบ ๆ', ...);
test('ค่าที่ไม่รู้จักต้องไม่ทำให้พัง แต่ต้องทำงานต่อไม่ได้', ...);

4. ไฟล์ตัวอย่างจากของจริง

เปิด test/fixtures/tasks_page.json

{
  "items": [
    {"id": "T-001", ..., "quantity": 12, "createdAt": "2026-08-15T03:00:00Z"},
    {"id": "T-002", ..., "quantity": 8.0, "createdAt": "2026-08-15T03:05:00+07:00"},
    {"id": "T-003", ..., "quantity": "20", "createdAt": "2026-08-15T03:10:00Z"}
  ],
  "totalCount": 137
}

สังเกตว่าไฟล์เดียวครอบหลายเรื่องพร้อมกัน

แถวครอบอะไร
T-001กรณีปกติ มีหมายเหตุ เวลาเป็น Z
T-002จำนวนเป็น double, เวลามี offset, หมายเหตุเป็น null
T-003จำนวนเป็นข้อความ, สถานะใช้ชื่อเล่น COMPLETED, ไม่มีคีย์หมายเหตุ
totalCountไม่เท่ากับ items.length

การใส่ความหลากหลายไว้ในไฟล์เดียว ทำให้เทสต์เดียวจับได้หลายอย่าง

Map<String, dynamic> fixture(String name) =>
    jsonDecode(File('test/fixtures/$name').readAsStringSync())
        as Map<String, dynamic>;

5. ไฟล์ตัวอย่างมาจากไหน

แหล่งดีตรงไหนต้องระวัง
คัดลอกจาก response จริงตรงกับความจริงที่สุดต้องลบข้อมูลอ่อนไหวก่อน
เขียนเองตามเอกสารทำได้ก่อนที่ API จะเสร็จอาจไม่ตรงกับของจริง
ให้ฝั่งหลังบ้านส่งมาให้ทั้งสองฝั่งเห็นตรงกันต้องอัปเดตเมื่อสัญญาเปลี่ยน
ห้ามเก็บ token ชื่อลูกค้าจริง เบอร์โทร หรือราคาไว้ในไฟล์ตัวอย่าง — ไฟล์เหล่านี้อยู่ใน git และคนทั้งทีมเห็น ให้แทนด้วยค่าสมมติที่ รักษาหน้าตาและความยาวเดิม เช่นเปลี่ยนชื่อจริงเป็น “บริษัท ซัพพลาย จำกัด” แต่ยังเป็นภาษาไทยและยาวใกล้เคียงกัน

เหตุผลที่ต้องรักษาหน้าตาไว้ — ถ้าเปลี่ยนชื่อไทยยาว ๆ เป็น "test" จะไม่มีวันเจอบั๊กเรื่องข้อความยาวเกินหรือการเข้ารหัสภาษาไทย

6. ต้องมีทั้งไฟล์ที่ดีและที่เสีย

ไฟล์ที่ถูกต้องพิสูจน์ได้แค่ว่าอ่านได้ — ไม่ได้พิสูจน์ว่าจับความผิดพลาดได้

test('แถวที่พังต้องบอกว่าเป็นแถวที่เท่าไหร่', () {
  final json = {
    'items': [taskJson(), taskJson({'id': 'T-002'})..remove('quantity')],
    'totalCount': 2,
  };

  expect(
    () => TaskPageDto.fromJson(json),
    throwsA(isA<ContractException>()
        .having((e) => e.path, 'path', 'items[1].quantity')),
  );
});

สังเกตว่าเทสต์ที่ทดสอบความล้มเหลว ไม่ได้ใช้ไฟล์ แต่สร้าง JSON ในเทสต์แล้วแก้เฉพาะจุด

Map<String, dynamic> taskJson([Map<String, Object?> overrides = const {}]) => {
  'id': 'T-001',
  'palletCode': 'PAL-0101',
  'quantity': 12,
  'status': 'PENDING',
  'createdAt': '2026-08-15T03:00:00Z',
  ...overrides,
};
ไฟล์ fixtureตัวช่วยสร้างในเทสต์
เหมาะกับกรณีที่ถูกต้อง และหน้าตาของจริงกรณีที่ผิด ทีละจุด
อ่านง่ายไหมเห็นข้อมูลทั้งก้อนเห็นว่าเทสต์นี้สนใจอะไร
ถ้ามี 20 กรณีผิดต้องมี 20 ไฟล์เขียน 20 บรรทัด

การมีทั้งสองแบบทำให้ไฟล์ไม่บานปลาย และเทสต์แต่ละข้ออ่านแล้วรู้ทันทีว่าทดสอบอะไร

7. รายการกรณีที่ควรครอบ

กลุ่มกรณีบท
ค่าที่จำเป็นไม่มีคีย์ / คีย์มีแต่เป็น null / ข้อความว่าง6.2, 6.4
ตัวเลข12 / 12.0 / "12" / 1.9 / ข้อความที่ไม่ใช่เลข6.3
เวลามี Z / มี offset / ไม่มีเขตเวลา / รูปแบบผิด6.6
ชื่อคีย์camelCase / PascalCase6.7
สถานะค่าที่รู้จัก / ชื่อเล่น / ค่าที่ไม่รู้จัก6.9
ซองปกติ / ว่าง / totalCount ขัดแย้ง / items ไม่ใช่ array6.10
ชั้นซ้อนปกติ / แถวในพัง / รายการในว่าง6.5

lab มี 27 เทสต์ที่ครอบตารางนี้ทั้งหมด ซึ่งรันจบใน ไม่ถึงหนึ่งวินาที เพราะไม่ต้องต่อเน็ตและไม่ต้องสร้างหน้าจอ

8. อย่าลืมทิศทางขาขึ้น

คนส่วนใหญ่เทสต์แค่การอ่าน แต่การส่งผิดสัญญาก็เสียหายเท่ากัน

test('ส่งคีย์ครบตามที่ตกลง รวมถึงคีย์ที่ค่าเป็น null', () {
  const request = CompleteTaskRequest(
    taskId: 'T-001',
    quantity: 12,
    commandId: 'cmd-1',
  );

  expect(request.toJson(), {
    'taskId': 'T-001',
    'quantity': 12,
    'commandId': 'cmd-1',
    'note': null,
  });
});

จุดสำคัญคือ เทียบทั้ง map ไม่ใช่เช็กทีละคีย์

วิธีเทียบจับอะไรได้
expect(json['taskId'], 'T-001')จับได้ว่าคีย์นี้ถูก
expect(json, {...})จับได้ทั้งคีย์ที่หาย คีย์ที่เกิน และชื่อคีย์ที่เปลี่ยน

คีย์ที่เกินมาโดยไม่ตั้งใจอันตรายกว่าที่คิด — บาง endpoint จะปฏิเสธคำขอที่มีคีย์แปลกปลอม และบางตัวจะรับไปแล้วทำอะไรที่ไม่ได้ตั้งใจ

9. ตรวจสองฝั่งถึงจะจับ drift ได้

เทสต์ฝั่งแอปพิสูจน์ได้แค่ว่า แอปอ่านไฟล์ตัวอย่างได้ ไม่ได้พิสูจน์ว่าเซิร์ฟเวอร์ยังส่งแบบนั้นอยู่

ไฟล์ตัวอย่างในแอป  ←──?──→  สิ่งที่เซิร์ฟเวอร์ส่งจริงวันนี้
วิธีปิดช่องว่างนี้รายละเอียด
ฝั่ง .NET มีเทสต์ที่ตรวจผลของการแปลงเป็น JSONยืนยันว่าชื่อคีย์และชนิดยังเหมือนเดิม
ใช้ไฟล์ตัวอย่างชุดเดียวกันทั้งสองฝั่งถ้าฝั่งไหนเปลี่ยน อีกฝั่งจะแดง
มีเทสต์ที่ยิงเข้าเซิร์ฟเวอร์จริงเป็นระยะรันแยกจากเทสต์ปกติ เพราะช้าและต้องมีเซิร์ฟเวอร์

รายละเอียดฝั่ง .NET อยู่ใน zero-to-senior-dotnet8-course ส่วนการยิงเข้าเซิร์ฟเวอร์จริงจะทำใน Part 14

แม้ยังไม่ได้ทำสองฝั่ง เทสต์ฝั่งแอปก็ยังคุ้มมาก — เพราะมันจับ การเปลี่ยนแปลงที่มาจากฝั่งเราเอง ได้ทั้งหมด ซึ่งเป็นสาเหตุของปัญหาส่วนใหญ่

10. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
มีคนแก้กฎที่ตั้งใจไว้ แล้วเทสต์ยังเขียวไม่มีเทสต์ตรึงกฎนั้นเขียนเทสต์ที่ชื่อบอกเหตุผล
เทสต์ผ่านหมดแต่ของจริงพังมีแต่ไฟล์ที่ถูกต้องเพิ่มกรณีที่ผิด
ไฟล์ตัวอย่างมี 40 ไฟล์สร้างไฟล์ใหม่ทุกกรณีผิดใช้ตัวช่วยสร้างในเทสต์
token หลุดเข้า gitคัดลอก response จริงมาทั้งก้อนลบข้อมูลอ่อนไหวก่อน
ไม่เคยเจอบั๊กภาษาไทยจนขึ้น productionแทนข้อมูลจริงด้วย "test"รักษาหน้าตาและความยาวเดิม
ส่งคีย์เกินไปแล้วเซิร์ฟเวอร์ปฏิเสธเช็ก toJson ทีละคีย์เทียบทั้ง map
เซิร์ฟเวอร์เปลี่ยนแล้วแอปไม่รู้ตรวจฝั่งเดียวให้ฝั่งหลังบ้านมีเทสต์ด้วย

11. แบบฝึกหัด

ทีมกำลังจะทำ endpoint ใหม่สำหรับใบเบิก และได้สัญญามาแล้ว

{
  "requestNo": "RQ-2026-0451",
  "statusCode": 20,
  "requestedAt": "2026-08-15T03:00:00Z",
  "lines": [
    {"productCode": "SKU-1001", "quantity": 5, "pickedQuantity": null}
  ],
  "totalCount": 1
}
  1. เขียนรายการเทสต์ที่ควรมี อย่างน้อยสิบข้อ พร้อมชื่อที่บอกเหตุผล
  2. ตัดสินว่าข้อไหนควรใช้ไฟล์ และข้อไหนควรสร้าง JSON ในเทสต์
  3. ตอบว่าจะรู้ได้อย่างไรว่าเทสต์ที่เขียนครอบพอแล้ว
ดูแนวคำตอบและเหตุผล

ข้อ 1 และ 2 — รายการเทสต์

#ชื่อเทสต์ใช้
1อ่านใบเบิกที่ถูกต้องได้ครบทุก fieldไฟล์
2รับ PascalCase จาก endpoint เก่าไฟล์
3ไม่มี requestNo ต้องบอกว่าคีย์ไหนหายในเทสต์
4requestNo เป็นข้อความว่าง ไม่นับว่ามีค่าในเทสต์
5รับ quantity ทั้ง 5, 5.0 และ "5"ในเทสต์
6ปฏิเสธ quantity ที่มีเศษ ไม่ปัดให้ในเทสต์
7เวลาที่ไม่บอกเขตเวลาต้องถูกปฏิเสธในเทสต์
8เวลาที่มี offset ต้องถูกแปลงเป็น UTCในเทสต์
9statusCode ที่ไม่รู้จักต้องเป็น unknown ไม่ใช่พังในเทสต์
10statusCode ที่ไม่รู้จักต้องกดทำงานต่อไม่ได้ในเทสต์
11pickedQuantity เป็น null แปลว่ายังไม่เริ่มหยิบในเทสต์
12บรรทัดที่พังต้องบอกว่าเป็นบรรทัดที่เท่าไหร่ในเทสต์
13lines ที่ไม่ใช่ array ต้องพังที่ linesในเทสต์
14totalCount น้อยกว่าจำนวนที่ส่งมาคือข้อมูลขัดแย้งในเทสต์
15คำสั่งยืนยันส่งคีย์ครบ รวมคีย์ที่ค่าเป็น nullในเทสต์

เกณฑ์แยกสองแบบ

ใช้ไฟล์เมื่อสร้างในเทสต์เมื่อ
อยากเห็นหน้าตาของจริงทั้งก้อนสนใจแค่จุดเดียวที่เปลี่ยน
เป็นกรณีที่ถูกต้องเป็นกรณีที่ผิด
ใช้ซ้ำได้หลายเทสต์ใช้ครั้งเดียว

สังเกตว่ามีแค่สองข้อที่ใช้ไฟล์ จาก 15 ข้อ — ไฟล์มีไว้แสดงหน้าตาของจริง ส่วนกรณีขอบใช้การแก้ทีละจุด

ตัวอย่างข้อ 11 ที่คนมักลืม

test('pickedQuantity เป็น null แปลว่ายังไม่เริ่มหยิบ', () {
  final line = PickLineDto.fromJson(lineJson({'pickedQuantity': null}));
  expect(toPickLine(line).pickedQuantity, 0);
});

test('pickedQuantity เป็น 0 ก็แปลว่ายังไม่หยิบเหมือนกัน', () {
  final line = PickLineDto.fromJson(lineJson({'pickedQuantity': 0}));
  expect(toPickLine(line).pickedQuantity, 0);
});

สองข้อนี้ดูซ้ำกัน แต่ตรึงคนละเรื่อง — ข้อแรกตรึงว่า null แปลงเป็น 0 ได้ ข้อที่สองตรึงว่าไม่ได้เผลอปฏิเสธ 0 ที่ส่งมาจริง

ข้อ 3 — จะรู้ได้อย่างไรว่าครอบพอ

ไม่มีตัวเลขความครอบคลุมที่ตอบได้ แต่มีคำถามสามข้อที่ตอบได้

ถ้าลบแล้วเทสต์ยังเขียวหมด แปลว่าบรรทัดนั้นไม่ได้ถูกทดสอบ — วิธีนี้ตรงกว่าการดูเปอร์เซ็นต์ความครอบคลุม เพราะความครอบคลุมบอกแค่ว่าบรรทัดถูกรัน ไม่ได้บอกว่าผลถูกตรวจ

คำถามที่สอง — ทุกการตัดสินใจที่อธิบายยาก มีเทสต์ไหม

การตัดสินใจมีเทสต์ไหม
ทำไมรับ "5" ที่เป็นข้อความข้อ 5
ทำไมไม่ปัด 1.9ข้อ 6
ทำไม unknown ไม่โยนข้อผิดพลาดข้อ 9
ทำไม null กลายเป็น 0 ได้ตรงนี้ข้อ 11

ถ้ามีการตัดสินใจไหนที่ตอบว่า “ไม่มี” นั่นคือช่องที่ต้องเติม

คำถามที่สาม — ถ้าเซิร์ฟเวอร์เปลี่ยนแบบที่เคยเกิดมาก่อน จะจับได้ไหม

การเปลี่ยนที่เคยเกิดจริงเทสต์ข้อไหนจับได้
เปลี่ยนชื่อคีย์ข้อ 3
เปลี่ยน int เป็น decimalข้อ 5
เพิ่มสถานะใหม่ข้อ 9, 10
เลิกส่ง field ที่ไม่ค่อยได้ใช้ข้อ 3
เปลี่ยนจาก DateTime เป็น DateTimeOffsetข้อ 7, 8

บทเรียนที่ควรจำ: เทสต์ที่ดีของ Part นี้ไม่ได้วัดที่จำนวน แต่วัดที่ว่า ทุกบรรทัดที่มีคนอาจอยากลบ มีเทสต์คอยแดงให้ — เพราะโค้ดตรวจสอบทั้งหมดใน Part 6 หน้าตาเหมือนโค้ดที่ระแวงเกินไป จนกว่าจะรู้ว่าทำไมถึงต้องมี

12. สรุปท้ายบท

  • เทสต์คือที่เก็บเหตุผลว่าทำไมโค้ดถึงเขียนแบบนั้น
  • ตั้งชื่อเทสต์เป็นประโยคที่บอกเหตุผล ไม่ใช่ชื่อเมธอด
  • ไฟล์ตัวอย่างใช้แสดงหน้าตาของจริง ส่วนกรณีผิดสร้างในเทสต์
  • ใส่ความหลากหลายในไฟล์เดียว เพื่อให้ครอบหลายเรื่องพร้อมกัน
  • ลบข้อมูลอ่อนไหวออก แต่รักษาหน้าตาและความยาวเดิม
  • ต้องมีทั้งกรณีที่ถูกและที่ผิด
  • อย่าลืมทดสอบทิศทางขาขึ้น และเทียบทั้ง map
  • เทสต์ฝั่งแอปจับการเปลี่ยนจากฝั่งเราได้ แต่ต้องมีสองฝั่งถึงจะจับ drift ได้
  • วัดความพอด้วยการลองลบโค้ดตรวจสอบ แล้วดูว่ามีเทสต์แดงไหม
