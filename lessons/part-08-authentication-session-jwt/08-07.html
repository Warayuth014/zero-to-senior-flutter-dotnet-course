ระดับ: กลาง–สูงใช้เวลา: 30–40 นาทีลงมือทำ: รับมือกับ token ที่ใช้ไม่ได้ระหว่างทำงาน

1. ปัญหาที่บทนี้แก้

8.6 ตรวจอายุตอนเปิดแอปแล้ว แต่ token ยังใช้ไม่ได้กลางทางได้อยู่ดี

สาเหตุตรวจล่วงหน้าได้ไหม
หมดอายุระหว่างที่ใช้งานอยู่ได้ — รู้เวลาหมดอายุอยู่แล้ว
ผู้ดูแลระบบยกเลิก tokenไม่ได้
ผู้ใช้เปลี่ยนรหัสผ่านจากเครื่องอื่นไม่ได้
บัญชีถูกระงับกลางกะไม่ได้

สามแถวล่างรู้ได้ทางเดียวคือ ยิงคำขอไปแล้วได้ 401 กลับมา

และเมื่อเกิดขึ้น มันมักเกิดกับหลายคำขอพร้อมกัน

polling ทุก 10 วินาที + คำขอที่ผู้ใช้กด + คำขอโหลดรูป
  → token ถูกยกเลิก
  → ทั้งสามได้ 401 ภายในเสี้ยววินาทีเดียวกัน

2. คำศัพท์ที่ต้องรู้ก่อน

คำความหมาย
revokeยกเลิก token ก่อนหมดอายุ
reasonเหตุผลที่หลุดออกจากระบบ
idempotentทำซ้ำแล้วผลเท่ากับทำครั้งเดียว

3. Mental model: หลายคนเห็นปัญหาเดียวกัน

คำขอ A ──401──┐
คำขอ B ──401──┼──→ ต้องออกจากระบบ "หนึ่งครั้ง" ไม่ใช่สามครั้ง
คำขอ C ──401──┘
ประโยคเดียวที่ต้องจำ: 401 หลายตัวคือหลักฐานของเหตุการณ์เดียว ไม่ใช่หลายเหตุการณ์ — การจัดการต้องเกิดครั้งเดียว

4. ให้จุดเดียวเป็นคนจัดการ

เปิด lib/auth/session_store.dart

/// เรียกเมื่อ server ตอบ 401 — token ใช้ไม่ได้แล้ว
///
/// ทำครั้งเดียวแม้จะได้ 401 มาหลายครั้งพร้อมกัน เพราะคำขอหลายตัว
/// ที่วิ่งขนานกันจะได้ 401 พร้อมกันหมด (5.4)
Future<void> handleUnauthorized() async {
  if (_status != AuthStatus.signedIn) return;
  await _clearTo(AuthStatus.signedOut, reason: SignOutReason.sessionExpired);
}

บรรทัดแรกคือทั้งหมดของการกันซ้ำ — ถ้าไม่ได้อยู่ในสถานะเข้าสู่ระบบแล้ว ก็ไม่ต้องทำอะไร

test('ได้ 401 หลายครั้งพร้อมกัน ต้องล้าง token ครั้งเดียว', () async {
  await store.signIn(username: 'somchai', password: 'x', serverId: 'wh-01');

  await Future.wait([
    store.handleUnauthorized(),
    store.handleUnauthorized(),
    store.handleUnauthorized(),
  ]);

  // ['token-abc', null] — ไม่ใช่ null ซ้ำสามครั้ง
  expect(tokens, ['token-abc', null]);
});

เทสต์นี้จับอาการที่ทดสอบด้วยมือไม่มีทางเจอ — เพราะต้องมีคำขอสามตัววิ่งพร้อมกันแล้วล้มเหลวพร้อมกันพอดี

5. ทำไมการทำซ้ำถึงเป็นปัญหา

ถ้าทำสามครั้งผลที่ตามมา
เขียนที่เก็บสามครั้งเปลืองและอาจชนกัน
แจ้งผู้ฟังสามครั้งหน้าจอวาดใหม่โดยไม่จำเป็น
พาไปหน้าเข้าสู่ระบบสามครั้งหน้าเข้าสู่ระบบซ้อนกันสามชั้น

แถวสุดท้ายคืออาการที่ 5.4 หัวข้อ 10 พูดถึง — กดปุ่มย้อนกลับแล้วเจอหน้าเข้าสู่ระบบอีกใบ

// ที่ชั้นบนซึ่งฟัง store อยู่
void _onSessionChanged() {
  if (sessionStore.status == AuthStatus.signedOut && _wasSignedIn) {
    _wasSignedIn = false;
    navigator.pushAndRemoveUntil(loginRoute, (route) => false);   // 5.4
  }
}

การเทียบกับสถานะก่อนหน้าตาม 4.6 หัวข้อ 8 ทำให้การเปลี่ยนเส้นทางเกิดครั้งเดียว แม้ store จะแจ้งหลายครั้ง

6. เหตุผลเปลี่ยนข้อความที่ผู้ใช้เห็น

/// เหตุผลที่หลุดออกจากระบบ — ใช้บอกผู้ใช้ว่าเกิดอะไร (8.7)
enum SignOutReason {
  /// ผู้ใช้กดออกเอง
  userRequested,

  /// token หมดอายุ หรือ server ปฏิเสธ
  sessionExpired,

  /// สลับไปเซิร์ฟเวอร์อื่น
  serverChanged,
}
เหตุผลข้อความควรพาไปไหน
userRequestedไม่ต้องมีข้อความหน้าเข้าสู่ระบบ
sessionExpired“เซสชันหมดอายุ กรุณาเข้าสู่ระบบใหม่”หน้าเข้าสู่ระบบ พร้อมชื่อผู้ใช้เดิม
serverChanged“เปลี่ยนเซิร์ฟเวอร์แล้ว กรุณาเข้าสู่ระบบใหม่”หน้าเข้าสู่ระบบ

ถ้าไม่เก็บเหตุผล ทั้งสามกรณีจะได้ข้อความเดียวกัน — และผู้ใช้ที่ถูกเด้งออกกลางงานจะไม่รู้ว่าเกิดอะไร

แถวกลางมีรายละเอียดที่ช่วยได้มาก — เติมชื่อผู้ใช้เดิมให้ เพราะเขาเพิ่งใช้งานอยู่ และชื่อผู้ใช้ไม่ใช่ความลับ

// ชื่อผู้ใช้เก็บแยกได้ เพราะไม่ใช่ข้อมูลลับ (8.10)
_lastUsername = session.username;

7. อย่าทิ้งงานของผู้ใช้เงียบ ๆ

สถานการณ์ที่ต้องคิดให้ครบ

คนหน้างานสแกนของเข้าใบรับเข้าไป 18 บรรทัด
  → token ถูกยกเลิก
  → แอปเด้งไปหน้าเข้าสู่ระบบ
  → 18 บรรทัดหายไปพร้อมกับ store ที่ถูกทำลาย
ทางเลือกผลที่ตามมา
ทิ้งไปเลยผู้ใช้ต้องเดินสแกนใหม่ 18 บรรทัด
เก็บร่างไว้ก่อนออกจากระบบล็อกอินกลับมาแล้วทำต่อได้

ทางที่สองถูก แต่มีเงื่อนไขที่ 5.4 แบบฝึกหัดข้อ 3 บอกไว้ — ร่างต้องผูกกับผู้ใช้ที่สร้างมัน

Future<void> handleUnauthorized() async {
  if (_status != AuthStatus.signedIn) return;

  // เก็บงานที่ทำค้างไว้ก่อน แล้วค่อยล้าง session
  await draftStorage.save(userId: _session!.username, drafts: ...);

  await _clearTo(AuthStatus.signedOut, reason: SignOutReason.sessionExpired);
}

ลำดับสำคัญ — เก็บก่อนล้าง เพราะหลังล้างแล้วจะไม่รู้ว่าร่างเป็นของใคร

8. ต่ออายุอัตโนมัติ ถ้ามี refresh token

ถ้าฝั่งหลังบ้านให้ refresh token มาตาม 8.5 การได้ 401 ไม่จำเป็นต้องพาออกจากระบบทันที

Future<bool> tryRefresh() async {
  final refreshToken = _session?.refreshToken;
  if (refreshToken == null) return false;

  final outcome = await repository.refresh(refreshToken);
  return switch (outcome) {
    RefreshSucceeded(:final session) => _applyAndSave(session),
    RefreshRejected() => false,      // refresh token ก็ใช้ไม่ได้แล้ว
    RefreshFailed() => false,        // เน็ตมีปัญหา
  };
}
ต้องระวังเพราะ
ต่ออายุครั้งเดียวแม้หลายคำขอได้ 401เหตุผลเดียวกับหัวข้อ 4
คำขอที่ได้ 401 ต้องรอผลการต่ออายุไม่งั้นจะล้มเหลวทั้งที่ต่ออายุสำเร็จ
ต่ออายุล้มเหลวแล้วต้องออกจากระบบไม่ใช่ลองไปเรื่อย ๆ
อย่าให้การต่ออายุเองได้ 401 แล้วเรียกต่ออายุอีกจะวนไม่รู้จบ

แถวสุดท้ายเป็นกับดักที่เจอบ่อย — คำขอต่ออายุต้องไม่ผ่านตัวดักที่เรียก handleUnauthorized

การต่ออายุอัตโนมัติเพิ่มความซับซ้อนมาก — ถ้า token อายุยาวพอสำหรับหนึ่งกะ อาจไม่คุ้มที่จะทำ ให้เริ่มจากการพาออกจากระบบอย่างสุภาพก่อน

9. ใครเป็นคนเรียก handleUnauthorized

// ที่ชั้นที่เรียก repository
try {
  await taskRepository.complete(id, commandId: commandId);
} on ApiException catch (error) {
  if (error.kind == ApiFailureKind.unauthorized) {
    await sessionStore.handleUnauthorized();
    return;
  }
  ...
}

เขียนแบบนี้ในทุก store จะซ้ำและลืมง่าย — ทางที่ดีกว่าคือดักที่ชั้นเดียว

// ที่ ApiClient — แจ้งขึ้นไปโดยไม่ตัดสินใจเอง
ApiClient({
  required this.baseUrl,
  http.Client? client,
  this.onUnauthorized,          // ไม่รู้จัก SessionStore
});

final void Function()? onUnauthorized;
// ที่จุดประกอบแอป
final apiClient = ApiClient(
  baseUrl: profile.baseUrl,
  onUnauthorized: () => sessionStore.handleUnauthorized(),
);

รูปแบบเดียวกับ onTokenChanged ใน 8.4 — ApiClient บอกว่าเกิดอะไร แต่ไม่ตัดสินใจว่าต้องทำอะไร ตามหลักใน 7.9

10. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
หน้าเข้าสู่ระบบซ้อนกันหลายชั้นจัดการ 401 หลายครั้งกันด้วยการตรวจสถานะ
ถูกเด้งออกโดยไม่มีข้อความไม่เก็บเหตุผลเก็บ SignOutReason
งานที่ทำค้างหายไปล้าง session ก่อนเก็บร่างเก็บก่อนล้าง
ต้องเขียนดัก 401 ในทุก storeดักที่ชั้นบนเกินไปดักที่ ApiClient แล้วแจ้งขึ้นไป
ต่ออายุแล้ววนไม่รู้จบคำขอต่ออายุเองได้ 401ยกเว้น endpoint นั้น
ถูกเด้งออกเมื่อไม่มีสิทธิ์เหมารวม 403 เป็น 401แยกสองรหัส (7.5)

11. แบบฝึกหัด

แอปมีสามอย่างที่ทำงานพร้อมกัน

อะไรความถี่
polling รายการงานทุก 10 วินาที
คำขอที่ผู้ใช้กดเมื่อกด
อัปโหลดรูปเบื้องหลังเมื่อมีรูปรอส่ง

ผู้ดูแลระบบยกเลิก token ตอน 14:32:10 ขณะที่ผู้ใช้กำลังกรอกใบรับเข้าอยู่ 18 บรรทัด

  1. ไล่ลำดับว่าเกิดอะไรบ้างในสามวินาทีถัดมา
  2. ระบุว่าถ้าไม่กันซ้ำ จะเกิดอะไร
  3. เขียนสิ่งที่ผู้ใช้ควรเห็น
ดูแนวคำตอบและเหตุผล

ข้อ 1 — ลำดับเหตุการณ์

14:32:10  ผู้ดูแลยกเลิก token
14:32:12  อัปโหลดรูปที่กำลังวิ่งอยู่ → 401
14:32:14  polling ยิงตามรอบ → 401
14:32:15  ผู้ใช้กดปุ่มบันทึกบรรทัดที่ 19 → 401

สามคำขอได้ 401 ภายในสามวินาที และแต่ละตัวเรียก handleUnauthorized

ครั้งที่สถานะตอนเรียกผล
1signedInทำงาน — ล้าง token, เก็บร่าง, เปลี่ยนสถานะ
2signedOutออกทันทีที่บรรทัดแรก
3signedOutออกทันทีเหมือนกัน
if (_status != AuthStatus.signedIn) return;

บรรทัดเดียวนี้ทำให้ครั้งที่ 2 และ 3 ไม่ทำอะไรเลย

ข้อ 2 — ถ้าไม่กันซ้ำ

ผลที่เกิดอาการที่ผู้ใช้เห็น
เขียนที่เก็บสามครั้งไม่เห็นอะไร แต่เปลืองและอาจชนกัน
เก็บร่างสามครั้งอาจเขียนทับกันเองถ้าเขียนไม่พร้อมกัน
แจ้งผู้ฟังสามครั้งหน้าจอกระพริบ
เปลี่ยนเส้นทางสามครั้งหน้าเข้าสู่ระบบสามชั้น

แถวสุดท้ายร้ายที่สุด เพราะผู้ใช้ล็อกอินสำเร็จแล้วยังเจอหน้าเข้าสู่ระบบอีกสองใบซ้อนอยู่

ล็อกอินสำเร็จ → เข้าหน้าหลัก
  → กดปุ่มย้อนกลับ → เจอหน้าเข้าสู่ระบบอีกใบ
  → งงว่าออกจากระบบตอนไหน

และแถวที่สองอันตรายเงียบ ๆ — การเก็บร่างสามครั้งพร้อมกันอาจทำให้ได้ร่างที่ไม่ครบ ถ้าการเขียนไม่ใช่การกระทำที่จบในครั้งเดียว

ข้อ 3 — สิ่งที่ผู้ใช้ควรเห็น

ส่วนประกอบทำไมต้องมี
บอกว่าเซสชันหมดอายุผู้ใช้ไม่ได้กดออกเอง ต้องรู้ว่าทำไม
บอกว่างานถูกเก็บไว้ลดความตกใจมากที่สุด — เขากำลังกลัวว่างานหาย
เติมชื่อผู้ใช้ให้เขาเพิ่งใช้งานอยู่ และไม่ใช่ความลับ
ไม่เติมรหัสผ่านไม่เคยเก็บไว้ตั้งแต่แรก (8.5)

สิ่งที่ห้ามแสดง

ห้ามเพราะ
“เกิดข้อผิดพลาด (401)”ผู้ใช้ไม่รู้ว่า 401 คืออะไร
“ไม่มีสิทธิ์เข้าถึง”เป็นข้อความของ 403 ไม่ใช่ 401 (7.5)
ข้อความเดียวกับตอนที่ผู้ใช้กดออกเองสองกรณีนี้ต่างกันสิ้นเชิง

และหลังล็อกอินกลับมา

เข้าหน้าหลัก → เจอข้อความ
  "มีใบรับเข้าที่ทำค้างไว้ 18 บรรทัด เมื่อ 14:32  [ทำต่อ] [ยกเลิก]"

ซึ่งเป็นรูปแบบเดียวกับที่ 5.7 แบบฝึกหัดข้อ 3 ออกแบบไว้ — กู้คืนแล้วต้องบอก ไม่ใช่กู้คืนเงียบ ๆ

บทเรียนที่ควรจำ: เมื่อมีหลายอย่างทำงานพร้อมกัน เหตุการณ์เดียวจะถูกรายงานหลายครั้ง — และการกันซ้ำที่ง่ายที่สุดคือตรวจว่าสถานะเปลี่ยนไปแล้วหรือยัง ซึ่งเป็นรูปแบบเดียวกับธงกันกดซ้ำใน 4.14

12. สรุปท้ายบท

  • token ใช้ไม่ได้กลางทางได้ด้วยสาเหตุที่ตรวจล่วงหน้าไม่ได้
  • 401 หลายตัวคือหลักฐานของเหตุการณ์เดียว
  • กันซ้ำด้วยการตรวจว่าสถานะยังเป็น signedIn อยู่ไหม
  • ไม่กันซ้ำแล้วจะได้หน้าเข้าสู่ระบบซ้อนกันหลายชั้น
  • เก็บเหตุผล เพราะข้อความที่ผู้ใช้เห็นต่างกัน
  • เก็บร่างงานก่อนล้าง session
  • ดัก 401 ที่ ApiClient แล้วแจ้งขึ้นไป ไม่ตัดสินใจเอง
  • ถ้าต่ออายุอัตโนมัติ ต้องยกเว้น endpoint ที่ใช้ต่ออายุเอง

บทถัดไป: 8.8 จะรวบทุกอย่างที่ต้องล้างเมื่อออกจากระบบ ซึ่งมากกว่าแค่ token
