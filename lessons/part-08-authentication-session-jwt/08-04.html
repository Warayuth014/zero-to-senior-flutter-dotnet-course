1. ปัญหาที่บทนี้แก้

8.2 ทำให้ SessionStore รู้ว่า token คืออะไร ส่วน 7.5 ทำให้ ApiClient แนบ token ไปกับคำขอ — บทนี้ต่อสองอย่างเข้าด้วยกัน

ปัญหาคือมันเป็นคนละชั้นกัน และการต่อผิดวิธีทำให้เกิดอาการที่ไล่หาสาเหตุยาก

วิธีต่อปัญหา
ส่ง token เข้า ApiClient ตอนสร้างต่ออายุแล้วยังใช้ค่าเก่า (7.5)
ให้ repository อ่าน token จาก store เองทุก repository ต้องรู้จัก store
ให้ store ถือ ApiClient ไว้ทดสอบ store ต้องมีตัวส่ง HTTP

2. คำศัพท์ที่ต้องรู้ก่อน

คำความหมาย
bearer tokenบัตรผ่านแบบที่ใครถือก็ใช้ได้
composition rootจุดที่ประกอบทุกอย่างเข้าด้วยกันตอนแอปเริ่ม
callbackฟังก์ชันที่ส่งไปให้เรียกกลับ

3. Mental model: บอกทางเดียว ไม่ถามกลับ

SessionStore  ──บอก──→  ApiClient
    │                       │
    รู้ว่า token คืออะไร      แค่ใช้ค่าล่าสุดที่ได้รับ ไม่ต้องรู้ว่ามาจากไหน
ประโยคเดียวที่ต้องจำ: ให้ฝ่ายที่รู้เป็นคนบอก ไม่ใช่ให้ฝ่ายที่ต้องใช้ไปถาม — เพราะฝ่ายที่ต้องใช้จะไม่รู้ว่าควรถามเมื่อไหร่

4. ต่อกันที่จุดประกอบแอป

final apiClient = ApiClient(baseUrl: profile.baseUrl);

final sessionStore = SessionStore(
  repository: RemoteAuthRepository(apiClient),
  storage: sessionStorage,
  onTokenChanged: (token) => apiClient.authToken = token,
);

สังเกตว่า RemoteAuthRepository ใช้ apiClient ตัวเดียวกัน — ซึ่งดูเหมือนวนกลับมาหาตัวเอง แต่ไม่มีปัญหา

ตอนไหนauthToken เป็นอะไร
ยิงคำขอเข้าสู่ระบบnull — endpoint นี้ไม่ต้องใช้ token
หลังเข้าสู่ระบบสำเร็จค่าที่เพิ่งได้มา
หลังออกจากระบบnull อีกครั้ง

และเพราะ _headers() ใส่ Authorization เฉพาะเมื่อมีค่า ตามที่ทำใน 7.5 คำขอเข้าสู่ระบบจึงไม่มี header นั้นติดไปโดยอัตโนมัติ

5. ทำไมใช้ callback แทนการถือไว้ตรง ๆ

/// เรียกทุกครั้งที่ token เปลี่ยน เพื่อให้ ApiClient ใช้ค่าล่าสุด
///
/// ใช้ callback แทนการถือ ApiClient ไว้ตรง ๆ เพื่อให้ทดสอบได้
/// โดยไม่ต้องสร้างตัวส่ง HTTP
final void Function(String? token) onTokenChanged;

ประโยชน์ที่เห็นได้ชัดที่สุดคือในเทสต์

tokens = <String?>[];
store = SessionStore(
  repository: RemoteAuthRepository(api),
  storage: storage,
  onTokenChanged: tokens.add,     // เก็บทุกครั้งที่เปลี่ยน
  clock: () => now,
);

ทำให้ตรวจ ลำดับ ของการเปลี่ยนแปลงได้ ซึ่งเป็นสิ่งที่อ่านจาก ApiClient ไม่ได้

test('ออกจากระบบต้องล้าง token ก่อนล้างที่เก็บ', () async {
  await store.signIn(username: 'somchai', password: 'x', serverId: 'wh-01');

  await store.signOut();

  expect(tokens, ['token-abc', null]);
});

ถ้าตรวจจาก ApiClient.authToken จะเห็นแค่ค่าสุดท้ายคือ null — ไม่รู้ว่าเคยถูกตั้งแล้วล้าง หรือไม่เคยถูกตั้งเลย

6. ล้าง token ก่อนเสมอ

Future<void> _clearTo(AuthStatus status, {SignOutReason? reason}) async {
  // ล้าง token ก่อนเสมอ ไม่ให้มีคำขอไหนยิงด้วย token เก่าได้อีก
  onTokenChanged(null);
  _session = null;
  _status = status;
  ...
}

ลำดับสำคัญ เพราะระหว่างที่ล้างที่เก็บ (ซึ่งเป็นงาน async) อาจมีคำขออื่นยิงออกไป

// ถ้าล้างที่เก็บก่อน
await storage.clear();          // ใช้เวลา 20 ms
  ← ระหว่างนี้ polling ยิงคำขอด้วย token เก่าได้
onTokenChanged(null);

เป็นหลักเดียวกับ 5.4 หัวข้อ 8 — ล้างสิ่งที่ทำให้เข้าถึงได้ก่อน แล้วค่อยจัดการที่เหลือ

และการเรียก onTokenChanged(null) ทำทุกครั้งแม้ตอนที่ยังไม่เคยมี token

test('ไม่มีข้อมูลเก็บไว้ ต้องไปหน้าเข้าสู่ระบบ', () async {
  await store.restore(currentServerId: 'wh-01');

  expect(store.status, AuthStatus.signedOut);
  // ล้าง token เสมอแม้ไม่เคยมี เพื่อรับประกันว่า ApiClient ไม่มีของค้าง
  expect(tokens, [null]);
});

7. ต้องมี ApiClient ตัวเดียว

// อันตราย — สอง repository สร้าง ApiClient คนละตัว
final taskRepository = RemoteTaskRepository(ApiClient(baseUrl: url));
final inboundRepository = RemoteInboundRepository(ApiClient(baseUrl: url));

ผลคือ onTokenChanged ตั้งค่าให้ตัวเดียว ส่วนอีกตัวไม่เคยได้ token เลย

อาการที่เห็นสาเหตุ
บางหน้าใช้ได้ บางหน้าได้ 401คนละตัวส่ง
ออกจากระบบแล้วบางส่วนยังใช้ได้ล้างให้ตัวเดียว
// ถูก — ตัวเดียว ส่งต่อให้ทุก repository
final apiClient = ApiClient(baseUrl: profile.baseUrl);
final taskRepository = RemoteTaskRepository(apiClient);
final inboundRepository = RemoteInboundRepository(apiClient);

เป็นเหตุผลเดียวกับ 7.6 หัวข้อ 8 เรื่องความเป็นเจ้าของ — และเป็นเหตุผลที่ ApiClient ควรถูกสร้างที่จุดประกอบแอปที่เดียว

8. คำขอที่กำลังวิ่งอยู่ตอนที่ token เปลี่ยน

การล้าง token ไม่ได้ยกเลิกคำขอที่ยิงไปแล้ว

0.0s  ยิงคำขอโหลดรายการ พร้อม token เก่า
0.2s  ผู้ใช้กดออกจากระบบ → token ถูกล้าง
0.8s  คำขอกลับมาพร้อมข้อมูล
      → store ของรายการงานเขียนข้อมูลลง state ทั้งที่ออกจากระบบแล้ว
สิ่งที่กันได้สิ่งที่กันไม่ได้
คำขอ ใหม่ จะไม่มี tokenคำขอที่ยิงไปแล้วยังเดินทางต่อ

ทางแก้คือให้ store ที่ถือข้อมูลถูกทำลายพร้อมกับการออกจากระบบ ซึ่งเป็นเรื่องของ 8.8

หรือใช้ตัวนับรุ่นตาม 4.10 ถ้า store นั้นอยู่ต่อ

9. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ได้ 401 ทั้งที่เพิ่งเข้าสู่ระบบส่ง token ตอนสร้าง ApiClientตั้งค่าเมื่อเปลี่ยน
บางหน้าใช้ได้ บางหน้าไม่ได้มี ApiClient หลายตัวสร้างตัวเดียวที่จุดประกอบ
ออกจากระบบแล้วยังมีคำขอด้วย token เก่าล้างที่เก็บก่อนล้าง tokenล้าง token ก่อน
ทดสอบ store ต้องมีตัวส่ง HTTPstore ถือ ApiClientใช้ callback
ข้อมูลของผู้ใช้เก่าโผล่มาหลังออกจากระบบคำขอที่ค้างอยู่กลับมาทำลาย store พร้อมกัน (8.8)

10. แบบฝึกหัด

ทีมพบว่าแอปมีอาการนี้ที่สาขาหนึ่ง

ผู้ใช้เข้าสู่ระบบสำเร็จ → หน้ารายการงานโหลดได้ปกติ
  → กดเข้าหน้ารับเข้า → ได้ 401 ทุกครั้ง
  → กลับมาหน้ารายการงาน → ยังใช้ได้ปกติ
  1. ระบุสาเหตุที่น่าจะเป็นที่สุด
  2. เขียนโค้ดจุดประกอบแอปที่ถูกต้อง
  3. เขียนเทสต์ที่จะจับปัญหานี้ได้
ดูแนวคำตอบและเหตุผล

ข้อ 1 — สาเหตุ

มี ApiClient มากกว่าหนึ่งตัว และ onTokenChanged ตั้งค่าให้ตัวที่หน้ารายการงานใช้เท่านั้น

หลักฐานที่ชี้ชัดสามอย่าง

สิ่งที่สังเกตได้ตัดอะไรออกไป
หน้าหนึ่งใช้ได้ อีกหน้าไม่ได้ไม่ใช่ token หมดอายุ — ถ้าหมดอายุจะพังทั้งคู่
กลับมาหน้าเดิมยังใช้ได้ไม่ใช่ token ถูกล้าง
เกิดทุกครั้ง ไม่ใช่บางครั้งไม่ใช่ปัญหาเครือข่ายหรือจังหวะ

“ทุกครั้ง” กับ “เฉพาะบางหน้า” รวมกันแล้วเหลือความเป็นไปได้เดียว — โค้ดของสองหน้าใช้ตัวส่งคนละตัว

ส่วนที่ว่า “เกิดที่สาขาหนึ่ง” น่าจะเป็นเรื่องบังเอิญ — สาขาอื่นอาจยังไม่มีใครใช้หน้ารับเข้า

ข้อ 2 — จุดประกอบที่ถูกต้อง

Future<AppDependencies> buildDependencies(ConnectionProfile profile) async {
  // ตัวส่ง HTTP ตัวเดียวสำหรับทั้งแอป
  final apiClient = ApiClient(baseUrl: profile.baseUrl);

  final sessionStore = SessionStore(
    repository: RemoteAuthRepository(apiClient),
    storage: SecureSessionStorage(),
    onTokenChanged: (token) => apiClient.authToken = token,
  );

  return AppDependencies(
    apiClient: apiClient,
    sessionStore: sessionStore,
    // ทุก repository ใช้ตัวส่งตัวเดียวกัน
    taskRepository: RemoteTaskRepository(apiClient),
    inboundRepository: RemoteInboundRepository(apiClient),
  );
}

สิ่งที่ทำให้ผิดยากขึ้น — ไม่มีที่ไหนอื่นในโปรเจกต์ที่เรียก ApiClient(...) ได้

// ค้นทั้งโปรเจกต์ ควรเจอที่เดียว
grep -rn "ApiClient(" lib/

ถ้าเจอมากกว่าหนึ่งที่ (ไม่นับเทสต์) แปลว่ามีปัญหานี้อยู่

ข้อ 3 — เทสต์ที่จับได้

ปัญหานี้อยู่ที่ การประกอบ ไม่ใช่ที่คลาสใดคลาสหนึ่ง — เทสต์ของ SessionStore หรือของ repository แต่ละตัวจะผ่านหมด

test('ทุก repository ต้องใช้ตัวส่งตัวเดียวกัน', () async {
  final deps = await buildDependencies(testProfile);

  // ยืนยันว่าเป็นวัตถุตัวเดียวกันจริง ไม่ใช่แค่ค่าเท่ากัน
  expect(
    identical(deps.taskRepository.api, deps.inboundRepository.api),
    isTrue,
  );
});
test('เข้าสู่ระบบแล้ว token ต้องถึงทุก repository', () async {
  final deps = await buildDependencies(testProfile);
  await deps.sessionStore.signIn(
    username: 'somchai',
    password: 'x',
    serverId: testProfile.id,
  );

  expect(deps.apiClient.authToken, isNotNull);
  // ตัวส่งที่ repository ถืออยู่ ต้องเป็นตัวเดียวกับที่ถูกตั้งค่า
  expect(identical(deps.taskRepository.api, deps.apiClient), isTrue);
  expect(identical(deps.inboundRepository.api, deps.apiClient), isTrue);
});

สังเกตว่าใช้ identical ไม่ใช่ equals — เพราะสิ่งที่ต้องพิสูจน์คือ เป็นวัตถุตัวเดียวกันในหน่วยความจำ ไม่ใช่แค่มีค่าเหมือนกัน

ทางที่ทำให้ผิดไม่ได้เลย

เทสต์จับได้หลังเขียนผิดไปแล้ว — แต่ทำให้เขียนผิดไม่ได้ตั้งแต่แรกดีกว่า

// ให้ repository รับสัญญา ไม่ใช่รับคลาสจริง
class RemoteTaskRepository implements TaskRepository {
  const RemoteTaskRepository(this.api);
  final JsonApi api;      // ไม่ใช่ ApiClient
}

เมื่อ repository รับ JsonApi มันจะสร้าง ApiClient เองไม่ได้ เพราะไม่รู้จักคลาสนั้น — คนที่ประกอบต้องส่งเข้ามาเสมอ

บทเรียนที่ควรจำ: อาการ “บางหน้าใช้ได้ บางหน้าไม่ได้” เกือบทุกครั้งแปลว่ามีของที่ควรมีตัวเดียวแต่มีหลายตัว — และบั๊กแบบนี้อยู่ที่จุดประกอบ ซึ่งเป็นที่ที่มักไม่มีเทสต์ครอบ

11. สรุปท้ายบท

  • ให้ฝ่ายที่รู้เป็นคนบอก ไม่ใช่ให้ฝ่ายที่ใช้ไปถาม
  • ต่อกันที่จุดประกอบแอปที่เดียว
  • ใช้ callback แทนการให้ store ถือตัวส่ง HTTP
  • ล้าง token ก่อนล้างที่เก็บเสมอ
  • ล้างทุกครั้งแม้ยังไม่เคยมี เพื่อรับประกันว่าไม่มีของค้าง
  • ต้องมี ApiClient ตัวเดียวในแอป
  • ให้ repository รับสัญญา ไม่ใช่คลาสจริง จะได้สร้างเองไม่ได้
  • การล้าง token ไม่ได้ยกเลิกคำขอที่ยิงไปแล้ว

บทถัดไป: 8.5 จะทำให้ผู้ใช้ไม่ต้องเข้าสู่ระบบใหม่ทุกครั้งที่เปิดแอป และตอบว่าอะไรเก็บได้ อะไรเก็บไม่ได้
