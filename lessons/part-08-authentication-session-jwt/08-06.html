1. ปัญหาที่บทนี้แก้

อาการที่เห็นได้ในแอปจำนวนมาก

เปิดแอป → หน้าเข้าสู่ระบบแวบหนึ่ง → เด้งเข้าหน้าหลักเอง

ผู้ใช้ไม่ได้ทำอะไรผิด และแอปก็ทำงานถูก — แต่ประสบการณ์แย่ และทำให้คนกดปุ่มผิดถ้ากดเร็ว

สาเหตุคือแอปตอบคำถาม “เข้าสู่ระบบอยู่ไหม” เร็วเกินไป ก่อนที่จะอ่านข้อมูลที่เก็บไว้เสร็จ

0 ms   แอปเริ่ม → session เป็น null → แสดงหน้าเข้าสู่ระบบ
40 ms  อ่านข้อมูลจากเครื่องเสร็จ → มี session → เด้งเข้าหน้าหลัก

2. คำศัพท์ที่ต้องรู้ก่อน

คำความหมาย
bootstrapขั้นตอนเตรียมทุกอย่างก่อนแอปเริ่มทำงานจริง
splashหน้าจอที่แสดงระหว่างเตรียม
restoreการกู้สถานะเดิมกลับมา

3. Mental model: ยังไม่รู้ ไม่เท่ากับ ยังไม่ได้เข้าสู่ระบบ

unknown    ยังอ่านข้อมูลไม่เสร็จ  →  แสดงหน้ารอ
signedOut  อ่านเสร็จแล้ว ไม่มี     →  แสดงหน้าเข้าสู่ระบบ
signedIn   อ่านเสร็จแล้ว ใช้ได้     →  แสดงหน้าหลัก
ประโยคเดียวที่ต้องจำ: การไม่มีสถานะ unknown บังคับให้แอปต้องเดา และการเดาที่ผิดคือที่มาของการเด้งหน้าจอ

4. สถานะที่มีสี่ค่า

เปิด lib/auth/session_store.dart

enum AuthStatus {
  /// ยังไม่รู้ — กำลังอ่านข้อมูลที่เก็บไว้ตอนเปิดแอป
  unknown,

  signedOut,

  /// กำลังส่งชื่อผู้ใช้และรหัสผ่าน
  signingIn,

  signedIn,
}
AuthStatus _status = AuthStatus.unknown;   // ค่าเริ่มต้น

ค่าเริ่มต้นเป็น unknown ไม่ใช่ signedOut — ซึ่งเป็นรายละเอียดเล็ก ๆ ที่แก้ปัญหาทั้งหมดในหัวข้อ 1

Widget build(BuildContext context) => ListenableBuilder(
  listenable: sessionStore,
  builder: (context, child) => switch (sessionStore.status) {
    AuthStatus.unknown => const SplashScreen(),
    AuthStatus.signedOut || AuthStatus.signingIn => LoginScreen(...),
    AuthStatus.signedIn => HomeShell(...),
  },
);

สังเกตว่า signingIn ยังแสดงหน้าเข้าสู่ระบบ — เพราะผู้ใช้ต้องเห็นฟอร์มของตัวเองระหว่างที่รอ ตาม 4.7

5. ขั้นตอนการอ่านข้อมูลเดิม

Future<void> restore({required String currentServerId}) async {
  _status = AuthStatus.unknown;
  notifyListeners();

  AuthSession? stored;
  try {
    stored = await storage.read();
  } on SessionStorageException {
    // อ่านไม่ได้ก็แค่ให้ล็อกอินใหม่ ไม่ควรทำให้เปิดแอปไม่ได้
    stored = null;
  }

  if (stored == null) {
    await _clearTo(AuthStatus.signedOut);
    return;
  }
  if (stored.serverId != currentServerId) {
    await _clearTo(AuthStatus.signedOut, reason: SignOutReason.serverChanged);
    return;
  }
  if (stored.isExpiredAt(_clock())) {
    await _clearTo(AuthStatus.signedOut, reason: SignOutReason.sessionExpired);
    return;
  }

  _applySession(stored);
}

สี่ทางออก และแต่ละทางมีเหตุผลที่ต่างกัน

กรณีผลลัพธ์บอกผู้ใช้ว่า
ไม่มีข้อมูลหน้าเข้าสู่ระบบไม่ต้องบอกอะไร — เป็นเรื่องปกติ
คนละเซิร์ฟเวอร์หน้าเข้าสู่ระบบ“เปลี่ยนเซิร์ฟเวอร์แล้ว กรุณาเข้าสู่ระบบใหม่”
หมดอายุหน้าเข้าสู่ระบบ“เซสชันหมดอายุ”
ใช้ได้หน้าหลัก—

สามกรณีแรกพาไปที่เดียวกัน แต่ เหตุผลต่างกัน และผู้ใช้ควรได้ข้อความต่างกัน ซึ่งเป็นเหตุผลที่ต้องเก็บ SignOutReason

6. ลำดับการตรวจสำคัญ

1. มีข้อมูลไหม
2. เซิร์ฟเวอร์ตรงกันไหม
3. หมดอายุหรือยัง
4. ใช้ได้

ทำไมตรวจเซิร์ฟเวอร์ก่อนอายุ

ถ้าตรวจอายุก่อน
  token ของสาขา A ที่ยังไม่หมดอายุ + ตอนนี้ตั้งค่าเป็นสาขา B
  → ผ่านการตรวจอายุ → เข้าใช้งาน → ทุกคำขอได้ 401
  → ผู้ใช้เห็น "เซสชันหมดอายุ" ทั้งที่ปัญหาคือเซิร์ฟเวอร์ผิด

การตรวจเซิร์ฟเวอร์ก่อนทำให้ได้ข้อความที่ถูกต้อง และตัดปัญหาตั้งแต่ต้น

7. ตรวจอายุเองก่อน ไม่ต้องรอ 401

ตรวจเองตอนเปิดแอปรอให้ได้ 401
ผู้ใช้เห็นอะไรหน้าเข้าสู่ระบบทันทีหน้าหลัก แล้วเด้งออกหลังกดอะไรสักอย่าง
เสียคำขอไปกี่ครั้งไม่เสียอย่างน้อยหนึ่ง
รู้ได้ตอนไหนก่อนแสดงหน้าจอหลังผู้ใช้เริ่มทำงานแล้ว

แต่การตรวจเองไม่ได้แทนที่การรับมือกับ 401 — ต้องมีทั้งคู่

ตรวจเองตอนเปิด  →  กันกรณีที่รู้อยู่แล้วว่าหมดอายุ
รับมือกับ 401    →  กันกรณีที่ token ถูกยกเลิกก่อนหมดอายุ (8.7)

เพราะ token อาจใช้ไม่ได้ทั้งที่ยังไม่หมดอายุ เช่นเมื่อผู้ดูแลระบบยกเลิกมัน หรือเมื่อผู้ใช้เปลี่ยนรหัสผ่านจากเครื่องอื่น

8. อย่ายิงคำขอตรวจสอบตอนเปิดแอป

ทางเลือกที่ดูรอบคอบกว่า

// เพิ่มขั้นตอน — ยิงไปถาม server ว่า token ยังใช้ได้ไหม
final me = await api.getJson('/api/Authenticate/me');
ได้อะไรเสียอะไร
รู้แน่ว่า token ยังใช้ได้เปิดแอปช้าลงเท่ากับหนึ่งคำขอ
ได้ข้อมูลผู้ใช้ล่าสุดถ้าเน็ตไม่มี จะเปิดแอปไม่ได้เลย

แถวขวาล่างเป็นปัญหาจริงในคลัง — ถ้าเน็ตมีปัญหาตอนเช้า ทุกคนเปิดแอปไม่ได้พร้อมกัน ทั้งที่ token ยังดีอยู่

lab จึงไม่ยิงคำขอตอนเปิด — เชื่อสิ่งที่เก็บไว้ แล้วให้คำขอจริงเป็นตัวบอกถ้ามีปัญหา ซึ่งเป็นหลักเดียวกับ 7.16 หัวข้อ 7

ถ้าจำเป็นต้องยิงจริง ให้ทำแบบไม่บล็อก — เข้าหน้าหลักไปก่อน แล้วยิงเบื้องหลัง ถ้าได้ 401 ค่อยพาออก

9. หน้ารอต้องไม่ค้าง

} on SessionStorageException {
  // อ่านไม่ได้ก็แค่ให้ล็อกอินใหม่ ไม่ควรทำให้เปิดแอปไม่ได้

ถ้าไม่ดักไว้ แอปจะค้างอยู่ที่ unknown ตลอดกาล เพราะไม่มีอะไรเปลี่ยนสถานะ

test('อ่านที่เก็บไม่ได้ ต้องไม่ทำให้เปิดแอปไม่ได้', () async {
  storage.failOnRead = true;
  final store = storeWithClock(now);

  await store.restore(currentServerId: 'wh-01');

  expect(store.status, AuthStatus.signedOut);
});

เป็นหลักเดียวกับ 4.8 — ทุกเส้นทางต้องจบที่สถานะที่ชัดเจน ไม่มีทางไหนที่ปล่อยให้ค้าง

10. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
เด้งหน้าเข้าสู่ระบบแวบหนึ่งทุกครั้งที่เปิดไม่มีสถานะ unknownเพิ่มเข้าไปเป็นค่าเริ่มต้น
แอปค้างที่หน้ารออ่านข้อมูลล้มเหลวแล้วไม่มีทางออกดักแล้วไปหน้าเข้าสู่ระบบ
เปิดแอปไม่ได้เมื่อเน็ตมีปัญหายิงคำขอตรวจสอบก่อนเข้าเชื่อสิ่งที่เก็บไว้ก่อน
บอกว่าหมดอายุทั้งที่เปลี่ยนเซิร์ฟเวอร์ตรวจอายุก่อนตรวจเซิร์ฟเวอร์สลับลำดับ
เข้าหน้าหลักแล้วเด้งออกทันทีไม่ตรวจอายุตอนเปิดตรวจก่อนแสดงหน้าจอ
ทุกกรณีขึ้นข้อความเดียวกันไม่เก็บเหตุผลเก็บ SignOutReason

11. แบบฝึกหัด

ทีมพบว่าตอนเช้าที่ทุกคนเปิดแอปพร้อมกัน มีรายงานสองแบบ

#อาการ
1บางเครื่องค้างที่หน้าโลโก้ ต้องปิดแล้วเปิดใหม่
2บางเครื่องเข้าได้ แต่พอกดปุ่มแรกก็เด้งออกทันที
  1. ระบุสาเหตุที่น่าจะเป็นของแต่ละข้อ
  2. เขียนวิธีแก้
  3. ตอบว่าทำไมทั้งสองอาการถึงเกิดพร้อมกันตอนเช้า
ดูแนวคำตอบและเหตุผล

ข้อ 1 — สาเหตุ

#สาเหตุหลักฐาน
1ขั้นตอนเปิดแอปมีทางที่ไม่จบค้างที่หน้ารอ = สถานะไม่เคยเปลี่ยนจาก unknown
2ไม่ได้ตรวจอายุตอนเปิดเข้าได้แต่คำขอแรกล้มเหลว

อาการที่ 1 มีสองสาเหตุย่อยที่เป็นไปได้

ก. อ่านข้อมูลจากเครื่องล้มเหลว แล้วไม่มีทางออก
ข. ยิงคำขอตรวจสอบตอนเปิด แล้วรอจนหมดเวลา (15 วินาที)

ข้อ ข น่าจะเป็นสาเหตุจริงมากกว่า เพราะ “ต้องปิดแล้วเปิดใหม่” สื่อว่ารอนานจนผู้ใช้ยอมแพ้ ไม่ใช่ค้างถาวร

ข้อ 2 — วิธีแก้

อาการที่ 1

// เอาการยิงคำขอออกจากขั้นตอนเปิดแอป

และเพราะเผื่อเวลาไว้ 60 วินาทีตาม 8.3 token ที่เหลืออีกไม่ถึงนาทีก็จะถูกถือว่าหมดอายุ — กันกรณีที่หมดอายุระหว่างที่ผู้ใช้กำลังเปิดแอป

ข้อ 3 — ทำไมเกิดพร้อมกันตอนเช้า

นี่คือคำถามที่ทำให้เห็นภาพรวม — ทั้งสองอาการมาจากสาเหตุร่วมกันข้อเดียว

token มีอายุ 8 ชั่วโมง
กะเช้าเริ่ม 08:00 → เข้าสู่ระบบ → token หมดอายุ 16:00
กะบ่ายเริ่ม 16:00 → เปิดแอปที่เครื่องเดิม → token หมดอายุพอดี
สิ่งที่เกิดอาการที่เห็น
token หมดอายุพร้อมกันทุกเครื่องทุกคนต้องเข้าสู่ระบบใหม่พร้อมกัน
ทุกเครื่องยิงคำขอเข้าสู่ระบบพร้อมกันเซิร์ฟเวอร์ช้าลง
คำขอตรวจสอบตอนเปิดแอปช้าตามไปด้วยอาการที่ 1
เครื่องที่ token เพิ่งหมดพอดีอาการที่ 2

สิ่งที่ควรทำเพิ่มนอกจากแก้สองข้อข้างบน

ทำอะไรแก้อะไร
ให้ token หมดอายุกระจายกัน ไม่ตรงกันเป๊ะลดการกระจุกตัว — เป็นหลักเดียวกับ jitter ใน 7.14
ใช้ refresh token ต่ออายุเบื้องหลัง (8.5)ผู้ใช้ไม่ต้องเข้าสู่ระบบใหม่พร้อมกัน
ตั้งอายุ token ให้ยาวกว่ากะไม่หมดอายุกลางกะ

แถวสุดท้ายดูง่ายที่สุดแต่ต้องคิดให้ครบ — token ที่อายุยาวกว่ากะ แปลว่าถ้าเครื่องหาย มันใช้ได้ต่ออีกทั้งกะ

บทเรียนที่ควรจำ: อาการที่เกิด “พร้อมกันตอนเวลาเดิมทุกวัน” เกือบทุกครั้งมาจากสิ่งที่หมดอายุพร้อมกัน — และการกระจายเวลาหมดอายุออกจากกัน เป็นทางแก้เดียวกับที่ใช้กับการลองใหม่ใน 7.14

12. สรุปท้ายบท

  • ค่าเริ่มต้นของสถานะต้องเป็น unknown ไม่ใช่ signedOut
  • “ยังไม่รู้” กับ “ยังไม่ได้เข้าสู่ระบบ” เป็นคนละเรื่อง
  • ตรวจตามลำดับ — มีข้อมูลไหม เซิร์ฟเวอร์ตรงไหม หมดอายุหรือยัง
  • ตรวจเซิร์ฟเวอร์ก่อนอายุ เพื่อให้ได้ข้อความที่ถูกต้อง
  • ตรวจอายุเองตอนเปิด แต่ยังต้องรับมือกับ 401 อยู่ดี
  • อย่ายิงคำขอตรวจสอบก่อนเข้าแอป
  • ทุกเส้นทางต้องจบที่สถานะที่ชัดเจน
  • เก็บเหตุผลของการออกจากระบบ เพราะข้อความที่ผู้ใช้เห็นต่างกัน
