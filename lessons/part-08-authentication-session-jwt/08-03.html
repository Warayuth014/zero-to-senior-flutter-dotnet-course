3. Mental model: ซองที่ปิดผนึกแต่โปร่งใส

header . payload . signature
  │        │          │
  │        │          └─ พิสูจน์ว่าไม่ถูกแก้ — ต้องมีกุญแจถึงจะตรวจได้
  │        └─ ข้อมูล — ใครก็อ่านได้
  └─ บอกว่าเซ็นด้วยวิธีไหน
ประโยคเดียวที่ต้องจำ: JWT ไม่ได้เข้ารหัส — มันแค่เซ็นชื่อกำกับ ใครถือ token ก็อ่านข้างในได้หมด สิ่งที่ทำไม่ได้คือแก้แล้วให้ยังผ่านการตรวจ

ผลตามมาที่สำคัญที่สุด — ห้ามใส่อะไรที่เป็นความลับลงใน token

4. แกะเนื้อหาออกมา

เปิด lib/auth/jwt.dart

/// อ่านเนื้อหาใน JWT
///
/// อ่านได้อย่างเดียว — ตรวจสอบว่า token ถูกต้องจริงไม่ได้ เพราะการตรวจ
/// ต้องใช้กุญแจที่มีแต่ server รู้ ฝั่งแอปจึงใช้ข้อมูลนี้เพื่อปรับหน้าจอเท่านั้น
/// ห้ามใช้ตัดสินสิทธิ์ (8.9)
library;
JwtPayload? tryDecodeJwt(String token) {
  final parts = token.split('.');
  if (parts.length != 3) return null;

  try {
    // base64url ใน JWT ตัด = ท้ายออก จึงต้องเติมกลับก่อนถอด
    final normalized = base64Url.normalize(parts[1]);
    final decoded = utf8.decode(base64Url.decode(normalized));
    final claims = jsonDecode(decoded);
    if (claims is! Map<String, dynamic>) return null;
    return JwtPayload(claims);
  } on FormatException {
    return null;
  }
}

บรรทัด base64Url.normalize จำเป็นเพราะ JWT ตัดเครื่องหมาย = ท้ายทิ้ง เพื่อให้ใส่ใน URL ได้ — ถ้าไม่เติมกลับ จะถอดไม่ได้

และคืน null แทนการโยน ตามเหตุผลเดียวกับ 6.9

test('token ที่รูปแบบผิดต้องคืน null ไม่ใช่โยน', () {
  expect(tryDecodeJwt('ไม่ใช่ token'), isNull);
  expect(tryDecodeJwt('a.b'), isNull);
  expect(tryDecodeJwt('a.ข้อความที่ไม่ใช่ base64.c'), isNull);
});

5. เวลาหมดอายุเป็นตัวเลข ไม่ใช่ข้อความ

/// เวลาหมดอายุ — `exp` เป็นจำนวนวินาทีนับจากปี 1970 ไม่ใช่ ISO-8601
DateTime? get expiresAt {
  final exp = claims['exp'];
  if (exp is! num) return null;
  return DateTime.fromMillisecondsSinceEpoch(exp.toInt() * 1000, isUtc: true);
}
ที่ไหนรูปแบบเวลา
ใน JWT (exp)1791972800 — วินาทีนับจากปี 1970
ในคำตอบของ login"2026-08-15T10:00:00Z" — ISO-8601 (6.6)

สองรูปแบบในคำตอบเดียวกัน ซึ่งเป็นเรื่องปกติของ JWT — และการคูณด้วย 1000 จำเป็นเพราะ Dart นับเป็นมิลลิวินาที ไม่ใช่วินาที

test('อ่าน exp ที่เป็นวินาทีนับจากปี 1970 ได้', () {
  final payload = tryDecodeJwt(
    fakeJwt({'exp': expiry.millisecondsSinceEpoch ~/ 1000}),
  );

  expect(payload!.expiresAt, expiry);
});

ลืมคูณ 1000 แล้วจะได้เวลาที่อยู่ในปี 1970 ซึ่งแปลว่าหมดอายุไปแล้วเสมอ — และแอปจะเด้งออกทันทีที่เข้าสู่ระบบสำเร็จ

6. ชื่อ claim ของ .NET ยาวผิดปกติ

/// ชื่อ claim ที่ .NET ใช้เก็บ role โดยค่าตั้งต้น
///
/// ยาวแบบนี้เพราะเป็นมาตรฐานเก่าของ Microsoft — ถ้าฝั่งหลังบ้านตั้งค่าให้ใช้
/// ชื่อสั้นกว่านี้ได้ ควรขอให้เปลี่ยน แต่ต้องรองรับของเดิมไว้ด้วย
const String dotNetRoleClaim =
    'http://schemas.microsoft.com/ws/2008/06/identity/claims/role';

ครั้งแรกที่เห็นจะคิดว่าเป็นที่อยู่เว็บ แต่จริง ๆ เป็นแค่ชื่อ — และไม่มีอะไรอยู่ที่ที่อยู่นั้น

Set<String> get roles {
  final raw = claims['role'] ?? claims[dotNetRoleClaim] ?? claims['roles'];
  ...
}

ลองสามชื่อตามลำดับ เพราะแต่ละระบบตั้งค่าไม่เหมือนกัน — เป็นรูปแบบเดียวกับการรองรับ camelCase และ PascalCase ใน 6.7

7. role อาจเป็นข้อความเดียวหรือหลายค่า

/// role อาจมาเป็นข้อความเดี่ยวหรือรายการ ขึ้นกับว่าผู้ใช้มีกี่ role
///
/// .NET ส่งข้อความเดี่ยวเมื่อมี role เดียว และส่ง array เมื่อมีหลาย role
/// การรับได้ทั้งสองแบบจึงจำเป็น ไม่ใช่ความใจดี
Set<String> get roles {
  final raw = claims['role'] ?? claims[dotNetRoleClaim] ?? claims['roles'];
  return switch (raw) {
    String value when value.trim().isNotEmpty => {value.trim()},
    List<Object?> values => {for (final value in values) ?_readString(value)},
    _ => const <String>{},
  };
}

นี่คือพฤติกรรมที่ทำให้เกิดบั๊กที่หายากมาก

ผู้ใช้ที่มี role เดียว   → "role": "PICKER"              → ทดสอบผ่าน
ผู้ใช้ที่มีสอง role      → "role": ["PICKER", "ADMIN"]   → พัง

ถ้าทดสอบด้วยบัญชีที่มี role เดียวอย่างเดียว จะไม่มีทางเจอ — จนกว่าหัวหน้าที่มีหลาย role จะเข้าสู่ระบบ

test('รับ role ทั้งแบบข้อความเดี่ยวและแบบรายการ', () {
  expect(tryDecodeJwt(fakeJwt({'role': 'PICKER'}))!.roles, {'PICKER'});
  expect(
    tryDecodeJwt(fakeJwt({'role': ['PICKER', 'SUPERVISOR']}))!.roles,
    {'PICKER', 'SUPERVISOR'},
  );
});

8. เชื่อคำตอบก่อน แล้วค่อยดู token

เปิด lib/auth/auth_repository.dart

// เชื่อ expiration ที่ server ส่งมาก่อน แล้วค่อยดูใน token
// เพราะ server อาจตั้งอายุสั้นกว่าที่เขียนไว้ใน token ได้
final expiresAt = _readTime(json['expiration']) ?? payload?.expiresAt;
// อ่านจากคำตอบก่อน ถ้าไม่มีค่อยแกะจาก token และสุดท้ายใช้ชื่อที่กรอกมา
username: _readText(json['username']) ?? payload?.username ?? fallbackUsername,
roles: _readRoles(json['roles']) ?? payload?.roles ?? const {},
ลำดับเหตุผล
คำตอบของ loginเป็นสัญญาที่ตกลงกันไว้ชัด และไม่ผูกกับชื่อ claim
ข้างใน tokenทางสำรองเมื่อ endpoint ยังไม่ได้ส่งมาให้
ค่าที่ผู้ใช้กรอกทางสำรองสุดท้ายสำหรับชื่อผู้ใช้

ประโยชน์คือ เมื่อฝั่งหลังบ้านเพิ่ม field มาให้ โค้ดนี้จะใช้ค่าใหม่เองโดยไม่ต้องแก้

test('อ่าน role จาก token เมื่อ server ไม่ได้ส่งมาแยก', () async {
  api.loginResponse = loginJson(
    token: fakeJwt({'name': 'somchai', 'role': 'SUPERVISOR'}),
  );

  final outcome = await repository.login(...);

  expect((outcome as LoginSucceeded).session.roles, {'SUPERVISOR'});
});

9. เผื่อเวลาไว้ก่อนหมดอายุ

เปิด lib/auth/auth_session.dart

/// หมดอายุแล้วหรือยัง โดยเผื่อเวลาไว้ล่วงหน้า
///
/// เผื่อ [skew] เพราะนาฬิกาของ PDA กับของ server ไม่ตรงกันเป๊ะ และเพราะ
/// คำขอที่ยิงตอนเหลืออีก 2 วินาที จะหมดอายุระหว่างเดินทาง
bool isExpiredAt(
  DateTime now, {
  Duration skew = const Duration(seconds: 60),
}) => !now.toUtc().isBefore(expiresAt.subtract(skew));
ถ้าไม่เผื่ออาการ
นาฬิกา PDA เร็วกว่า server 30 วินาทีแอปคิดว่าหมดอายุแล้ว ทั้งที่ยังใช้ได้
นาฬิกา PDA ช้ากว่า server 30 วินาทีแอปส่งคำขอด้วย token ที่หมดอายุแล้ว
ยิงคำขอตอนเหลือ 2 วินาทีหมดอายุระหว่างเดินทาง ได้ 401
test('ถือว่าหมดอายุก่อนเวลาจริง เพราะเผื่อเวลาเดินทาง', () {
  // เหลืออีก 30 วินาที แต่เผื่อไว้ 60 จึงถือว่าหมดแล้ว
  final almost = expiry.subtract(const Duration(seconds: 30));

  expect(sessionExpiringAt(expiry).isExpiredAt(almost), isTrue);
});

การเผื่อเวลาไม่ได้แก้ปัญหานาฬิกาไม่ตรงทั้งหมด — มันแค่ลดโอกาส และ เซิร์ฟเวอร์ยังต้องตรวจอายุเองอยู่ดี ตามหลักใน 6.12

10. token ไม่ควรไปโผล่ในบันทึก

/// ไม่พิมพ์ token ออกมา เพราะ toString ไปโผล่ใน log ได้ง่าย (7.17)
@override
String toString() =>
    'AuthSession($username @ $serverId, หมดอายุ $expiresAt)';

เขียน toString เองเพราะค่าตั้งต้นของ Dart จะพิมพ์แค่ชื่อคลาส — แต่พอมีคนเพิ่มการพิมพ์ field ทั้งหมดเพื่อช่วยไล่หาสาเหตุ token จะหลุดไปทันที

test('ไม่พิมพ์ token ออกมาใน toString', () {
  final text = sessionExpiringAt(expiry).toString();

  expect(text, contains('somchai'));
  expect(text, isNot(contains('token')));
});

เทสต์นี้ไม่ได้กันบั๊กที่มีอยู่ — มันกันการเปลี่ยนแปลงในอนาคตที่ดูเหมือนไม่มีพิษภัย ซึ่งเป็นหน้าที่ของเทสต์ตาม 6.13

11. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
เข้าสู่ระบบสำเร็จแล้วเด้งออกทันทีลืมคูณ 1000 ตอนอ่าน expคูณให้ถูกหน่วย
หัวหน้าที่มีหลาย role เข้าใช้ไม่ได้รับ role แบบข้อความอย่างเดียวรับทั้งข้อความและรายการ
อ่าน role ไม่เจอเลยชื่อ claim ของ .NET ยาวไม่ตรงที่คิดลองหลายชื่อ
ถอด token ไม่ได้ไม่ได้เติม = กลับใช้ base64Url.normalize
ได้ 401 เป็นครั้งคราวโดยไม่มีเหตุผลไม่เผื่อเวลาเผื่อไว้ประมาณหนึ่งนาที
token อยู่ในไฟล์บันทึกtoString พิมพ์ทุก fieldเขียนเอง แล้วมีเทสต์กัน
ซ่อนปุ่มตาม role แล้วคิดว่าปลอดภัยใช้ token ตัดสินสิทธิ์เซิร์ฟเวอร์ต้องตรวจ (8.9)

12. แบบฝึกหัด

ทีมได้ token ตัวอย่างจากฝั่งหลังบ้านมาตรวจ

{
  "sub": "EMP001",
  "http://schemas.xmlsoap.org/ws/2005/05/identity/claims/name": "สมชาย ใจดี",
  "http://schemas.microsoft.com/ws/2008/06/identity/claims/role": [
    "PICKER", "SUPERVISOR"
  ],
  "warehouseCode": "WH-01",
  "employeePin": "1234",
  "exp": 1791972800,
  "iss": "wms-api",
  "aud": "wms-mobile"
}
  1. ระบุว่าอ่าน claim ไหนได้บ้างด้วยโค้ดใน lab
  2. มี claim หนึ่งที่ไม่ควรอยู่ในนี้ — บอกว่าอันไหนและทำไม
  3. ตอบว่า warehouseCode ควรใช้ทำอะไรได้และทำอะไรไม่ได้
ดูแนวคำตอบและเหตุผล

ข้อ 1 — อ่านได้อะไรบ้าง

claimอ่านได้ด้วยได้ค่า
subpayload.subjectEMP001
claim ชื่อยาวของ namepayload.usernameสมชาย ใจดี
claim ชื่อยาวของ rolepayload.roles{PICKER, SUPERVISOR}
exppayload.expiresAtเวลาที่แปลงเป็น UTC แล้ว
warehouseCode, iss, audpayload.claims['...']ต้องอ่านเอง

สังเกตว่า username ลองสามทางตามลำดับ และในกรณีนี้เจอที่ทางที่สอง

String? get username =>
    _readString(claims['name']) ??
    _readString(claims[dotNetNameClaim]) ??
    subject;

ถ้าไม่มี claim ชื่อยาว ก็จะได้ EMP001 จาก sub แทน ซึ่งยังใช้งานได้ แค่ไม่สวย

ข้อ 2 — claim ที่ไม่ควรอยู่

"employeePin": "1234"

เหตุผลรายละเอียด
ใครถือ token ก็อ่านได้JWT ไม่ได้เข้ารหัส — แค่ถอด base64 ก็เห็น
เป็นความลับที่ใช้ยืนยันตัวตนได้ถ้า PIN ใช้เข้าระบบอื่นได้ ก็เท่ากับรหัสผ่านรั่ว
ไม่จำเป็นต้องอยู่ในนั้นแอปไม่ต้องรู้ PIN ของผู้ใช้เลย

วิธีพิสูจน์ให้ทีมเห็น — เอา token ไปวางในเว็บที่ถอด JWT ได้ แล้วให้ดูว่าเห็น PIN ทันทีโดยไม่ต้องมีกุญแจอะไรเลย

ถอด base64 ส่วนกลาง → เห็น {"employeePin": "1234", ...}
ไม่ต้องมีรหัสผ่าน ไม่ต้องมีสิทธิ์ ไม่ต้องเจาะอะไร

และเพราะ token ถูกส่งไปกับ ทุกคำขอ มันจึงผ่านตัวกลางในเครือข่ายทุกตัว และอาจไปอยู่ในบันทึกของระบบอื่นที่เราไม่ได้ดูแล

นอกจากนี้ token ที่มี claim เยอะจะใหญ่ขึ้น และถูกส่งไปทุกคำขอ — เป็นค่าใช้จ่ายที่จ่ายซ้ำตลอด

ข้อ 3 — warehouseCode ใช้ทำอะไรได้

บรรทัดล่างเขียนได้ แต่ เซิร์ฟเวอร์ต้องไม่เชื่อค่านี้ — มันต้องดูจาก token ที่แนบมาว่าผู้ใช้คนนี้เข้าถึงคลังไหนได้ แล้วจำกัดเอง

ถ้าเซิร์ฟเวอร์เชื่อค่าที่แอปส่งมา ใครก็ตามที่ยิงคำขอเองจะดูข้อมูลของทุกคลังได้ โดยแค่เปลี่ยนค่าในคำขอ

บทเรียนที่ควรจำ: ข้อมูลใน token ใช้ทำให้หน้าจอสะดวกขึ้นได้ทั้งหมด — แต่ทุกครั้งที่มันถูกใช้ตัดสินว่า เข้าถึงอะไรได้ นั่นคือการย้ายการตัดสินใจเรื่องความปลอดภัยมาไว้ฝั่งที่ควบคุมไม่ได้

13. สรุปท้ายบท

  • JWT ไม่ได้เข้ารหัส — ใครถือก็อ่านได้ ห้ามใส่ความลับ
  • ลายเซ็นตรวจได้เฉพาะฝั่งที่มีกุญแจ ซึ่งไม่ใช่ฝั่งแอป
  • exp เป็นวินาที ต้องคูณ 1000 ก่อนแปลง
  • ชื่อ claim ของ .NET ยาวผิดปกติ ต้องลองหลายชื่อ
  • role อาจเป็นข้อความเดี่ยวหรือรายการ ต้องรับได้ทั้งคู่
  • เชื่อคำตอบของ login ก่อน แล้วค่อยดูใน token
  • เผื่อเวลาก่อนหมดอายุ เพราะนาฬิกาไม่ตรงและคำขอใช้เวลาเดินทาง
  • เขียน toString เองไม่ให้ token หลุด และมีเทสต์กันไว้
  • ข้อมูลใน token ใช้ปรับหน้าจอได้ แต่ตัดสินสิทธิ์ไม่ได้
