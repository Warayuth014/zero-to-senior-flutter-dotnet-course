1. ปัญหาที่บทนี้แก้

5.4 บอกไว้แล้วว่าล้างหน้าจอไม่เท่ากับออกจากระบบ — บทนี้ลงรายละเอียดว่าต้องล้างอะไรบ้าง

และเหตุผลที่เรื่องนี้สำคัญกว่าในแอปทั่วไป

โทรศัพท์ส่วนตัว  →  คนเดียวใช้  →  ล้างไม่ครบก็ไม่มีใครเห็น
PDA ในคลัง       →  เปลี่ยนมือทุกกะ  →  ล้างไม่ครบ = คนถัดไปเห็นของคนก่อน
สิ่งที่ลืมล้างคนถัดไปเห็นอะไร
tokenทำงานด้วยสิทธิ์ของคนก่อน
ข้อมูลใน storeรายการงานของคนก่อน
ตัวจับเวลาคำขอยิงต่อด้วย token เก่า
ร่างงานที่ทำค้างงานของคนก่อน และอาจกดส่งในชื่อตัวเอง

3. Mental model: อะไรที่มีอายุเท่ากับ session

อายุเท่าแอป        ApiClient, ที่อยู่เซิร์ฟเวอร์, การตั้งค่า
อายุเท่า session   token, ข้อมูลงาน, ร่างงาน, ตัวจับเวลา
อายุเท่าหน้าจอ     controller, ตำแหน่งที่เลื่อน
ประโยคเดียวที่ต้องจำ: ทุกอย่างในชั้นกลางต้องหายไปพร้อมกันเมื่อออกจากระบบ — และวิธีที่ทำให้ลืมไม่ได้คือให้มันถูกสร้างพร้อมกันตั้งแต่แรก

4. ลำดับการล้าง

เปิด lib/auth/session_store.dart

Future<void> _clearTo(AuthStatus status, {SignOutReason? reason}) async {
  // ล้าง token ก่อนเสมอ ไม่ให้มีคำขอไหนยิงด้วย token เก่าได้อีก
  onTokenChanged(null);
  _session = null;
  _status = status;
  _lastSignOutReason = reason;
  try {
    await storage.clear();
  } on SessionStorageException {
    // ล้างไม่ได้ก็ยังต้องออกจากระบบในรอบนี้ให้ได้
  }
  notifyListeners();
}
ลำดับทำอะไรทำไมต้องอยู่ตำแหน่งนี้
1ล้าง tokenตัดความสามารถในการยิงคำขอทันที
2ล้างสถานะในหน่วยความจำเร็ว ไม่มีทางล้มเหลว
3ล้างที่เก็บช้าและล้มเหลวได้ จึงอยู่ท้าย
4แจ้งผู้ฟังหลังจากทุกอย่างเรียบร้อยแล้ว

ข้อ 3 ที่ล้มเหลวได้ ต้องไม่ทำให้ข้อ 1 และ 2 เสียไป — ออกจากระบบในรอบนี้ต้องสำเร็จเสมอ แม้ครั้งหน้าจะยังเจอข้อมูลเก่าอยู่

5. store อื่นต้องหายไปด้วย

SessionStore ล้างของตัวเองได้ แต่ไม่รู้จัก store อื่น

// ที่จุดประกอบแอป — สร้าง store ที่มีอายุเท่า session พร้อมกัน
class SignedInScope {
  SignedInScope({required this.apiClient, required this.session}) {
    taskStore = TaskStore(RemoteTaskRepository(apiClient));
    inboundStore = InboundStore(RemoteInboundRepository(apiClient));
  }

  late final TaskStore taskStore;
  late final InboundStore inboundStore;

  /// ทำลายทุกอย่างที่ผูกกับผู้ใช้คนนี้
  void dispose() {
    taskStore.dispose();      // หยุด polling และเพิ่มเลขรุ่น (4.11)
    inboundStore.dispose();
  }
}
// หน้าจอที่ฟัง session
void _onSessionChanged() {
  if (sessionStore.isSignedIn && _scope == null) {
    _scope = SignedInScope(apiClient: apiClient, session: sessionStore.session!);
  } else if (!sessionStore.isSignedIn && _scope != null) {
    _scope!.dispose();
    _scope = null;
  }
}

ประโยชน์คือ เพิ่ม store ใหม่แล้วลืมล้างไม่ได้ เพราะมันถูกสร้างในที่เดียวกับที่ถูกทำลาย

เป็นหลักเดียวกับ 4.12 — ใครสร้าง คนนั้นทำลาย

6. ตัวจับเวลาต้องหยุด

สิ่งที่ลืมบ่อยที่สุดและมองไม่เห็น

@override
void dispose() {
  stopPolling();      // หยุดที่จะเกิดในอนาคต
  _generation++;      // ทิ้งผลของที่เกิดไปแล้ว (4.10, 4.11)
  super.dispose();
}

ถ้าลืม จะเกิดสิ่งนี้

ออกจากระบบ → token ถูกล้าง
  → 10 วินาทีต่อมา polling ยิงคำขอ
  → ไม่มี token → ได้ 401
  → เรียก handleUnauthorized ทั้งที่ออกจากระบบไปแล้ว
  → (ถ้าไม่กันซ้ำตาม 8.7 จะพาไปหน้าเข้าสู่ระบบอีกครั้ง)

อาการที่ผู้ใช้เห็นคือหน้าเข้าสู่ระบบเด้งซ้ำทุก 10 วินาที ทั้งที่ไม่ได้ทำอะไร

และแม้จะกันซ้ำไว้แล้ว มันก็ยังเปลืองคำขอและกินแบตโดยไม่ได้อะไร

7. ร่างงานเก็บได้ แต่ต้องผูกกับเจ้าของ

ล้างทิ้งเก็บไว้แต่ผูกกับผู้ใช้
tokenร่างงานที่ยังไม่ได้ส่ง
ข้อมูลที่โหลดมาจากเซิร์ฟเวอร์รหัสคำสั่งที่ยังค้าง (4.14)
ชื่อและข้อมูลส่วนตัวการตั้งค่าที่ไม่อ่อนไหว

เหตุผลที่คอลัมน์ขวาเก็บได้ — มันเป็นงานของผู้ใช้ ไม่ใช่ความลับ และการทิ้งไปคือการทำลายงานที่เขาทำมา

await draftStorage.save(userId: session.username, draft: draft);

แต่ต้องผูกกับเจ้าของเสมอ ไม่งั้นคนกะถัดไปจะเห็น ซึ่งเป็นข้อสรุปของ 5.4 แบบฝึกหัดข้อ 3

// ตอนเปิดกลับมา — ดึงเฉพาะของคนที่ล็อกอินอยู่
final draft = await draftStorage.load(userId: session.username);

8. เซิร์ฟเวอร์ควรรู้ไหม

8.2 บอกว่า AuthRepository ไม่มีเมธอด logout — เพราะการออกจากระบบเป็นเรื่องของฝั่งแอป

ถ้าไม่แจ้งเซิร์ฟเวอร์ถ้าแจ้ง
token ยังใช้ได้จนหมดอายุtoken ถูกยกเลิกทันที
ออกจากระบบสำเร็จเสมอต้องตัดสินว่าถ้าแจ้งไม่ได้จะทำอย่างไร
ทำงานได้แม้ไม่มีเน็ตต้องมีเน็ต

สำหรับ token ที่อายุสั้น การไม่แจ้งก็เพียงพอ — แต่ถ้าตัดสินใจแจ้ง ต้อง ไม่ให้การแจ้งล้มเหลวขวางการออกจากระบบ

Future<void> signOut({SignOutReason reason = SignOutReason.userRequested}) async {
  // แจ้ง server แบบไม่รอผล และไม่ให้ล้มเหลวขวางการออก (4.8)
  unawaited(_notifyServerBestEffort());

  await _clearTo(AuthStatus.signedOut, reason: reason);
}

เพราะถ้าเน็ตหลุด แล้วออกจากระบบไม่ได้ ผู้ใช้จะติดอยู่ในบัญชีของตัวเองบนเครื่องที่ต้องส่งต่อให้กะถัดไป

9. รายการตรวจ

ล้างอะไรถ้าลืม
token ในตัวส่ง HTTPยิงคำขอด้วยสิทธิ์คนเก่าได้
token ในที่เก็บเปิดแอปใหม่แล้วเข้าได้เลย
store ที่ถือข้อมูลของผู้ใช้เห็นรายการงานของคนเก่า
ตัวจับเวลาและ pollingยิงคำขอต่อและได้ 401 วนไป
คำขอที่ยังค้างอยู่ผลกลับมาเขียนลง store ที่ควรตายแล้ว (4.10)
เส้นทางบนหน้าจอกดย้อนกลับแล้วเข้าหน้าเดิมได้ (5.4)
ไฟล์ชั่วคราว เช่นรูปที่รอส่งคนถัดไปเห็นรูปของคนก่อน

แถวสุดท้ายมักถูกลืมที่สุด — และเป็นข้อมูลที่มองเห็นได้ชัดที่สุดถ้าหลุด

10. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
คนกะถัดไปเห็นงานของคนก่อนไม่ทำลาย storeสร้างและทำลายพร้อมกัน
หน้าเข้าสู่ระบบเด้งซ้ำทุก 10 วินาทีไม่หยุด pollingหยุดใน dispose
เปิดแอปใหม่แล้วเข้าได้เลยไม่ล้างที่เก็บล้าง
ออกจากระบบไม่ได้เมื่อไม่มีเน็ตรอผลการแจ้งเซิร์ฟเวอร์แจ้งแบบไม่รอผล
ร่างงานของทุกคนปนกันไม่ผูกกับผู้ใช้เก็บพร้อมรหัสผู้ใช้
ล้างแล้วการตั้งค่าเซิร์ฟเวอร์หายด้วยล้างที่เก็บทั้งหมดล้างเฉพาะของ session (8.5)
เพิ่ม store ใหม่แล้วลืมล้างสร้างกระจายหลายที่รวมไว้ที่เดียว

11. แบบฝึกหัด

ทีมได้รายงานจากหัวหน้ากะว่า

“คนกะบ่ายบอกว่าเปิดแอปมาแล้วเห็นชื่อคนกะเช้าค้างอยู่บนหัวจอ กดออกจากระบบแล้วเข้าใหม่ก็ยังเห็น ต้องปิดแอปแล้วเปิดใหม่ถึงจะหาย”

  1. ระบุสาเหตุที่น่าจะเป็น
  2. อธิบายว่าทำไมปิดแอปแล้วเปิดใหม่ถึงหาย
  3. เขียนวิธีแก้และเทสต์ที่จะจับได้
ดูแนวคำตอบและเหตุผล

ข้อ 1 — สาเหตุ

มี store หรือ widget ที่ถือชื่อผู้ใช้ไว้ และไม่ถูกทำลายตอนออกจากระบบ

สิ่งที่ตัดออกได้เพราะ
token ไม่ถูกล้างถ้าไม่ล้าง คนกะบ่ายจะทำงานด้วยสิทธิ์คนเก่า ซึ่งไม่มีรายงาน
ที่เก็บไม่ถูกล้างถ้าไม่ล้าง ปิดแอปเปิดใหม่ก็ยังเห็น — แต่รายงานบอกว่าหาย

ข้อมูลที่ชี้ชัดที่สุดคือ “ปิดแอปแล้วเปิดใหม่ถึงจะหาย” — แปลว่าข้อมูลอยู่ในหน่วยความจำอย่างเดียว ไม่ได้อยู่ในที่เก็บ

สิ่งที่น่าจะเป็นคือ store ที่ถูกสร้างครั้งเดียวตอนแอปเริ่ม แล้วอยู่ยาวข้าม session

// ผิด — สร้างครั้งเดียวตอนแอปเริ่ม
final profileStore = ProfileStore();   // อายุเท่าแอป

// แต่ข้อมูลข้างในมีอายุเท่า session
class ProfileStore extends ChangeNotifier {
  String? displayName;    // ค้างข้าม session
}

ข้อ 2 — ทำไมปิดแอปแล้วหาย

ปิดแอป → หน่วยความจำถูกคืนทั้งหมด → store ถูกสร้างใหม่ตอนเปิด
      → displayName เป็น null → ไม่มีอะไรค้าง

ซึ่งอธิบายว่าทำไมการกดออกจากระบบไม่ช่วย — เพราะการออกจากระบบล้างแค่ SessionStore ส่วน ProfileStore ไม่มีใครแตะ

สิ่งที่เกิดข้อมูลไปไหน
กดออกจากระบบSessionStore ถูกล้าง แต่ ProfileStore ไม่
เข้าสู่ระบบใหม่ProfileStore ยังมีค่าเก่า จนกว่าจะโหลดทับ
ปิดแอปแล้วเปิดทุก store ถูกสร้างใหม่

ข้อ 3 — วิธีแก้และเทสต์

วิธีแก้ — ทำให้ store ที่ถือข้อมูลของผู้ใช้มีอายุเท่า session

class SignedInScope {
  SignedInScope({required this.apiClient, required AuthSession session}) {
    profileStore = ProfileStore(session: session);
    taskStore = TaskStore(RemoteTaskRepository(apiClient));
  }

  late final ProfileStore profileStore;
  late final TaskStore taskStore;

  void dispose() {
    profileStore.dispose();
    taskStore.dispose();
  }
}

สิ่งที่ทำให้ผิดยากขึ้น — ProfileStore รับ AuthSession เข้ามาทาง constructor

class ProfileStore extends ChangeNotifier {
  ProfileStore({required this.session});
  final AuthSession session;      // final — ไม่มีทางค้างจาก session เก่า

  String get displayName => session.username;
}

เมื่อชื่อผู้ใช้มาจาก session ที่เป็น final มันจึงไม่มีทางค้าง — เพราะ store ตัวใหม่ต้องได้ session ตัวใหม่เสมอ

เป็นการแก้ปัญหาด้วยชนิดข้อมูล แทนที่จะแก้ด้วยวินัย ซึ่งเป็นหลักเดียวกับ 4.7

เทสต์

test('ล็อกอินคนใหม่ต้องไม่เห็นชื่อคนก่อนหน้า', () async {
  await app.signIn('somchai');
  expect(app.scope!.profileStore.displayName, 'somchai');

  await app.signOut();
  expect(app.scope, isNull);          // ทำลายแล้วจริง

  await app.signIn('somsri');
  expect(app.scope!.profileStore.displayName, 'somsri');
});
test('ออกจากระบบต้องทำลาย store ที่ผูกกับผู้ใช้ทั้งหมด', () async {
  await app.signIn('somchai');
  final taskStore = app.scope!.taskStore;

  await app.signOut();

  // store เก่าต้องใช้ต่อไม่ได้ ไม่ใช่แค่ถูกลืม
  expect(() => taskStore.load(), throwsA(isA<FlutterError>()));
});

เทสต์ตัวที่สองแรงกว่า — มันพิสูจน์ว่า store ถูก ทำลาย ไม่ใช่แค่ไม่มีใครอ้างถึงแล้ว

เพราะ ChangeNotifier ที่ถูก dispose แล้วจะโยนเมื่อมีคนเรียก notifyListeners ซึ่งเป็นการยืนยันว่าถูกทำลายจริง

บทเรียนที่ควรจำ: อาการ “ปิดแอปแล้วหาย” เป็นหลักฐานว่าข้อมูลอยู่ในหน่วยความจำอย่างเดียว — และเมื่อรวมกับ “กดออกจากระบบแล้วไม่หาย” ก็เหลือความเป็นไปได้เดียวคือมีของที่อายุยาวกว่าที่ควร

12. สรุปท้ายบท

  • PDA เปลี่ยนมือทุกกะ การล้างไม่ครบจึงเห็นได้ทันที
  • ล้าง token ก่อน แล้วค่อยล้างที่เก็บที่ล้มเหลวได้
  • ออกจากระบบในรอบนี้ต้องสำเร็จเสมอ
  • สร้าง store ที่มีอายุเท่า session พร้อมกัน เพื่อให้ทำลายพร้อมกัน
  • หยุดตัวจับเวลา ไม่งั้นจะยิงคำขอต่อและได้ 401 วนไป
  • ร่างงานเก็บได้ แต่ต้องผูกกับเจ้าของ
  • ถ้าแจ้งเซิร์ฟเวอร์ อย่าให้การแจ้งล้มเหลวขวางการออก
  • ให้ข้อมูลของผู้ใช้มาจาก session ที่เป็น final จะได้ไม่มีทางค้าง
