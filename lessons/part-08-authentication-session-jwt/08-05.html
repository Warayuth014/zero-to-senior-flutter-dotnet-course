1. ปัญหาที่บทนี้แก้

ทุกอย่างที่ทำมาอยู่ในหน่วยความจำ — ปิดแอปแล้วหายหมด

คนหน้างานเปิดแอป → เข้าสู่ระบบ → ทำงาน
  → วางเครื่องไว้ ระบบปิดแอปเอง (5.7)
  → หยิบขึ้นมาใหม่ → ต้องเข้าสู่ระบบอีกรอบ

บน PDA ที่ใส่ถุงมือและใช้ทั้งกะ การพิมพ์รหัสผ่านใหม่ทุกครั้งเป็นเรื่องใหญ่กว่าที่คิด

แต่การจำไว้ก็เปิดคำถามใหม่ — เก็บอะไรได้บ้าง และเก็บไว้ที่ไหน

2. คำศัพท์ที่ต้องรู้ก่อน

คำความหมาย
persistเก็บให้อยู่รอดข้ามการปิดแอป
secure storageที่เก็บที่ระบบปฏิบัติการช่วยปกป้อง
schema versionเลขรุ่นของโครงข้อมูลที่เก็บ
atomicสำเร็จทั้งหมดหรือไม่สำเร็จเลย

3. Mental model: เก็บบัตรผ่าน ไม่เก็บกุญแจ

รหัสผ่าน  →  ห้ามเก็บ  →  ใช้เข้าระบบอื่นได้ และเปลี่ยนไม่ได้ถ้ารั่ว
token     →  เก็บได้   →  หมดอายุเอง และยกเลิกได้
ประโยคเดียวที่ต้องจำ: เก็บสิ่งที่หมดอายุเองได้ ไม่เก็บสิ่งที่ใช้ได้ตลอดไป — และถ้าจำเป็นต้องเก็บรหัสผ่านเพื่อให้ฟีเจอร์ทำงาน แปลว่าออกแบบผิด

4. เก็บอะไรบ้าง

เปิด lib/auth/session_storage.dart

String encode(AuthSession session) => jsonEncode({
  'schemaVersion': sessionSchemaVersion,
  'serverId': session.serverId,
  'username': session.username,
  'accessToken': session.accessToken,
  'expiresAt': session.expiresAt.toIso8601String(),
  'roles': session.roles.toList(),
});
fieldทำไมต้องเก็บ
accessTokenสิ่งที่ทำให้ไม่ต้องเข้าสู่ระบบใหม่
expiresAtรู้ได้ว่ายังใช้ได้ไหมโดยไม่ต้องยิงคำขอ
serverIdรู้ว่า token นี้เป็นของเซิร์ฟเวอร์ไหน (8.11)
usernameแสดงบนหน้าจอทันทีที่เปิดแอป
rolesปรับหน้าจอได้ก่อนที่จะยิงคำขอแรก
schemaVersionดูหัวข้อ 6

สังเกตว่า ไม่มีรหัสผ่าน — และไม่มีอะไรที่จำเป็นต้องมีรหัสผ่านถึงจะทำงานได้

5. เก็บลงเครื่องก่อน แล้วค่อยประกาศว่าเข้าสู่ระบบแล้ว

เปิด lib/auth/session_store.dart

case LoginSucceeded(:final session):
  // เก็บลงเครื่องก่อน แล้วค่อยประกาศว่าเข้าสู่ระบบแล้ว
  // ถ้าเก็บไม่ได้ ยังใช้งานได้ในรอบนี้ แต่เปิดแอปใหม่ต้องล็อกอินอีก
  try {
    await storage.write(session);
  } on SessionStorageException {
    _errorMessage = 'เข้าสู่ระบบได้ แต่จำไว้ให้ไม่ได้ '
        'ครั้งหน้าต้องเข้าสู่ระบบใหม่';
  }
  _applySession(session);

การตัดสินใจที่ซ่อนอยู่ตรงนี้คือ — เก็บไม่ได้ ไม่ทำให้เข้าสู่ระบบล้มเหลว

ทางเลือกผลที่ตามมา
เก็บไม่ได้ = เข้าสู่ระบบไม่สำเร็จผู้ใช้ทำงานไม่ได้เลย ทั้งที่รหัสผ่านถูก
เก็บไม่ได้ = ใช้งานรอบนี้ได้ แต่บอกให้รู้ทำงานต่อได้ และรู้ว่าครั้งหน้าต้องพิมพ์ใหม่

ทางที่สองถูกสำหรับงานคลัง เพราะ การจำไว้เป็นความสะดวก ไม่ใช่ความจำเป็น

test('เก็บลงเครื่องไม่ได้ ยังใช้งานรอบนี้ได้ แต่ต้องบอกผู้ใช้', () async {
  storage.failOnWrite = true;

  await store.signIn(username: 'somchai', password: 'x', serverId: 'wh-01');

  expect(store.status, AuthStatus.signedIn);
  expect(store.errorMessage, contains('ครั้งหน้าต้องเข้าสู่ระบบใหม่'));
});

แต่ถ้าเป็นแอปธนาคาร คำตอบอาจกลับกัน — ซึ่งเป็นการตัดสินใจของธุรกิจ ไม่ใช่ของเทคนิค

6. เลขรุ่นของโครงข้อมูล

/// รุ่นของโครงข้อมูลที่เก็บ
///
/// ต้องมีตั้งแต่วันแรก เพราะวันที่โครงเปลี่ยน ต้องแยกให้ออกระหว่าง
/// "ข้อมูลเสีย" กับ "ข้อมูลของรุ่นเก่า" ซึ่งต้องจัดการคนละแบบ
const int sessionSchemaVersion = 1;
// รุ่นที่ไม่รู้จักถือว่าอ่านไม่ได้ ไม่ใช่พยายามเดา
if (json['schemaVersion'] != sessionSchemaVersion) return null;

ทำไมต้องมีตั้งแต่วันแรก ทั้งที่ยังมีรุ่นเดียว

วันที่เพิ่ม field ใหม่ที่จำเป็น
  ถ้าไม่มีเลขรุ่น → อ่านข้อมูลเก่าแล้วไม่เจอ field → ไม่รู้ว่าเสียหรือเก่า
  ถ้ามีเลขรุ่น    → รู้ทันทีว่าเป็นของรุ่นก่อน → ให้ล็อกอินใหม่อย่างมีเหตุผล

และการเพิ่มทีหลังทำไม่ได้ — เพราะข้อมูลที่เก็บไว้แล้วไม่มีเลขรุ่นอยู่ในนั้น

test('ข้อมูลของรุ่นเก่าต้องไม่ถูกเดา', () {
  final old = jsonEncode({
    'schemaVersion': 0,
    ...
  });

  expect(codec.decode(old), isNull);
});

7. ข้อมูลที่เสียต้องไม่ทำให้เปิดแอปไม่ได้

/// คืน null เมื่ออ่านไม่ได้ — ข้อมูลที่เสียไม่ควรทำให้เปิดแอปไม่ได้ (8.6)
AuthSession? decode(String? raw) {
  if (raw == null || raw.isEmpty) return null;

  final Object? json;
  try {
    json = jsonDecode(raw);
  } on FormatException {
    return null;
  }
  ...
}
สาเหตุที่ข้อมูลเสียได้
แอปถูกปิดกลางการเขียน
พื้นที่เก็บเต็ม
โครงข้อมูลเปลี่ยนโดยไม่ได้เพิ่มเลขรุ่น
แบตหมดตอนกำลังบันทึก

ทั้งสี่กรณีให้ผลเดียวกันที่ควรเป็น — ไปหน้าเข้าสู่ระบบ ไม่ใช่จอขาว

test('ข้อมูลที่เสียต้องคืน null ไม่ใช่ทำให้เปิดแอปไม่ได้', () {
  expect(codec.decode('ไม่ใช่ JSON'), isNull);
  expect(codec.decode('{}'), isNull);
  expect(codec.decode(null), isNull);
});

8. ล้างเฉพาะของตัวเอง

// อันตราย — ล้างทุกอย่างที่แอปเก็บไว้
await preferences.clear();
สิ่งที่หายไปด้วยโดยไม่ตั้งใจผลที่ตามมา
ที่อยู่เซิร์ฟเวอร์ที่ตั้งไว้ต้องตั้งค่าใหม่ทั้งหมด (7.3)
การตั้งค่าเสียงและขนาดตัวอักษรผู้ใช้ต้องตั้งใหม่
ร่างงานที่ยังไม่ได้ส่งงานของผู้ใช้หาย (5.7)

แถวสุดท้ายร้ายที่สุด — และมันเกิดจากโค้ดบรรทัดเดียวที่ดูเหมือนทำความสะอาด

// ถูก — ล้างเฉพาะสิ่งที่เกี่ยวกับการเข้าสู่ระบบ
@override
Future<void> clear() async {
  raw = null;
}

ที่เก็บของ session ควรเป็นคนละก้อนกับที่เก็บอย่างอื่น เพื่อให้ล้างผิดไม่ได้ตั้งแต่แรก

9. ที่เก็บเป็นสัญญา ไม่ใช่คลาสจริง

/// ที่เก็บ session ข้ามการปิดแอป
///
/// เป็นสัญญา ไม่ใช่คลาสจริง เพราะที่เก็บจริงต่างกันตามแพลตฟอร์ม
/// และเทสต์ต้องใช้ของปลอมที่ควบคุมได้ (8.10)
abstract interface class SessionStorage {
  Future<AuthSession?> read();
  Future<void> write(AuthSession session);
  Future<void> clear();
}

และของปลอมที่ทำให้ทดสอบกรณีล้มเหลวได้

/// จำลองความล้มเหลวได้ เพราะที่เก็บจริงเขียนไม่สำเร็จได้ และนั่นคือกรณี
/// ที่ต้องออกแบบไว้ ไม่ใช่กรณีที่หวังว่าจะไม่เกิด
class InMemorySessionStorage implements SessionStorage {
  String? raw;
  bool failOnWrite = false;
  bool failOnRead = false;
  ...
}

เป็นรูปแบบเดียวกับ MockClient ใน 7.6 — กรณีที่ทำให้เกิดจริงยาก คือกรณีที่ต้องทดสอบมากที่สุด

10. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
รหัสผ่านอยู่ในเครื่องเก็บไว้เพื่อเข้าสู่ระบบอัตโนมัติเก็บ token แทน
เปิดแอปแล้วจอขาวข้อมูลเสียแล้วโยนออกมาคืน null แล้วไปหน้าเข้าสู่ระบบ
อัปเดตแอปแล้วผู้ใช้ตั้งค่าหายหมดล้างที่เก็บทั้งหมดล้างเฉพาะของ session
ร่างงานหายตอนออกจากระบบเหมือนข้างบนเหมือนข้างบน
ไม่รู้ว่าข้อมูลเสียหรือเป็นของรุ่นเก่าไม่มีเลขรุ่นเพิ่มตั้งแต่วันแรก
เข้าสู่ระบบไม่ได้เพราะที่เก็บมีปัญหาถือว่าเก็บไม่ได้คือล้มเหลวให้ใช้งานรอบนี้ได้ แล้วบอกผู้ใช้
ทดสอบกรณีที่เก็บล้มเหลวไม่ได้ผูกกับที่เก็บจริงใช้สัญญา + ของปลอม

11. แบบฝึกหัด

ทีมจะเพิ่มความสามารถ “จำฉันไว้” ที่หน้าเข้าสู่ระบบ โดยมีข้อกำหนดจากธุรกิจ

ข้อกำหนด
ถ้าติ๊ก ให้ไม่ต้องเข้าสู่ระบบใหม่นาน 7 วัน
ถ้าไม่ติ๊ก ให้จำแค่จนกว่าจะปิดแอป
แต่ token จากเซิร์ฟเวอร์มีอายุแค่ 8 ชั่วโมง
  1. ระบุว่าข้อกำหนดนี้ทำได้ด้วยสิ่งที่มีอยู่ไหม
  2. ถ้าไม่ได้ ต้องขออะไรจากฝั่งหลังบ้าน
  3. ออกแบบว่าจะเก็บอะไรต่างกันระหว่างติ๊กกับไม่ติ๊ก
ดูแนวคำตอบและเหตุผล

ข้อ 1 — ทำได้ไหม

ทำไม่ได้ ด้วยสิ่งที่มีอยู่ และเหตุผลตรงไปตรงมา

token อายุ 8 ชั่วโมง
ธุรกิจต้องการ 7 วัน = 168 ชั่วโมง
  → หลังชั่วโมงที่ 8 token ใช้ไม่ได้แล้ว
  → ไม่ว่าเก็บไว้นานแค่ไหนก็ไม่ช่วย

สิ่งที่ทำได้คือ เก็บ token ไว้ 8 ชั่วโมง ซึ่งไม่ตรงกับที่ธุรกิจขอ

ทางที่ผิดที่คนมักคิดถึง

ทางเลือกที่ผิดทำไมผิด
เก็บรหัสผ่านไว้แล้วเข้าสู่ระบบใหม่เงียบ ๆขัดกับหลักในหัวข้อ 3 — รหัสผ่านใช้ได้ตลอดไปและเปลี่ยนยาก
ขอให้เซิร์ฟเวอร์ออก token อายุ 7 วันtoken ที่อายุยาวคือความเสี่ยงที่ยาวตามไปด้วย

ทางที่สองน่าสนใจเพราะดูสมเหตุสมผล — แต่ถ้า PDA หาย token ที่อายุ 7 วันจะใช้ได้ต่ออีกเจ็ดวัน และยกเลิกไม่ได้ถ้าไม่มีระบบเพิ่ม

ข้อ 2 — ต้องขออะไร

ขอ refresh token — บัตรผ่านอีกใบที่อายุยาว แต่ทำได้อย่างเดียวคือขอ token ใหม่

{
  "token": "eyJ...",              // อายุ 8 ชั่วโมง ใช้ยิงคำขอ
  "expiration": "2026-08-15T18:00:00Z",
  "refreshToken": "rt_9f3a...",   // อายุ 7 วัน ใช้ขอ token ใหม่เท่านั้น
  "refreshExpiration": "2026-08-22T10:00:00Z"
}
access tokenrefresh token
ใช้ทำอะไรยิงคำขอทุกคำขอขอ access token ใหม่เท่านั้น
อายุสั้นยาว
ถูกส่งบ่อยแค่ไหนทุกคำขอนาน ๆ ครั้ง
ยกเลิกได้ไหมรอหมดอายุยกเลิกได้ทันที

แถวสุดท้ายคือสิ่งที่ทำให้ทางนี้ปลอดภัยกว่าการออก token อายุยาว — ถ้า PDA หาย ทีมสั่งยกเลิก refresh token ได้ทันที

สิ่งที่ต้องตกลงเพิ่ม

ตกลงเรื่องเพราะ
endpoint สำหรับขอ token ใหม่ต้องรู้ว่ายิงไปที่ไหน
ใช้ refresh token ซ้ำได้ไหมบางระบบออกใบใหม่ทุกครั้ง
ยกเลิกอย่างไรเมื่อเครื่องหายเป็นเหตุผลหลักที่ใช้วิธีนี้

ข้อ 3 — เก็บอะไรต่างกัน

ติ๊ก “จำฉันไว้”ไม่ติ๊ก
access tokenเก็บเก็บ
refresh tokenเก็บไม่เก็บ
ปิดแอปแล้วเปิดใหม่ต่ออายุเองได้ 7 วันใช้ต่อได้จนครบ 8 ชั่วโมง
String encode(AuthSession session, {required bool rememberMe}) => jsonEncode({
  'schemaVersion': sessionSchemaVersion,
  'serverId': session.serverId,
  'username': session.username,
  'accessToken': session.accessToken,
  'expiresAt': session.expiresAt.toIso8601String(),
  'roles': session.roles.toList(),
  // เก็บเฉพาะเมื่อผู้ใช้ยอมให้จำ
  'refreshToken': ?(rememberMe ? session.refreshToken : null),
});

จุดที่ต้องคิดเพิ่ม — “ไม่ติ๊ก” ไม่ได้แปลว่าไม่เก็บอะไรเลย

คนมักคิดว่าไม่ติ๊ก = ไม่เก็บ แต่ถ้าทำแบบนั้น ระบบปิดแอปเองระหว่างพักกลางวันจะทำให้ต้องเข้าสู่ระบบใหม่ ทั้งที่ยังอยู่ในกะเดียวกัน

การเก็บ access token ไว้ทั้งสองกรณีจึงถูกต้อง — เพราะมันหมดอายุใน 8 ชั่วโมงอยู่แล้ว ซึ่งประมาณหนึ่งกะพอดี

และเมื่อออกจากระบบ ต้องล้างทั้งสองอย่าง พร้อมแจ้งเซิร์ฟเวอร์ให้ยกเลิก refresh token ด้วยถ้าทำได้

บทเรียนที่ควรจำ: เมื่อข้อกำหนดของธุรกิจขัดกับข้อจำกัดทางเทคนิค อย่าแก้ด้วยการเก็บสิ่งที่ไม่ควรเก็บ — ให้กลับไปคุยว่าต้องขออะไรเพิ่ม เพราะปัญหาส่วนใหญ่แบบนี้มีทางแก้ที่ถูกต้องอยู่แล้ว แค่ต้องทำสองฝั่ง

12. สรุปท้ายบท

  • เก็บสิ่งที่หมดอายุเองได้ ไม่เก็บสิ่งที่ใช้ได้ตลอดไป
  • ไม่มีเหตุผลใดที่ต้องเก็บรหัสผ่าน
  • เก็บลงเครื่องก่อน แล้วค่อยประกาศว่าเข้าสู่ระบบแล้ว
  • เก็บไม่ได้ไม่ควรทำให้เข้าสู่ระบบล้มเหลว แต่ต้องบอกผู้ใช้
  • เลขรุ่นของโครงข้อมูลต้องมีตั้งแต่วันแรก เพราะเพิ่มทีหลังไม่ได้
  • ข้อมูลที่เสียต้องคืนค่าว่าง ไม่ใช่ทำให้เปิดแอปไม่ได้
  • ล้างเฉพาะของ session ไม่ล้างที่เก็บทั้งหมด
  • ที่เก็บเป็นสัญญา เพื่อให้ทดสอบกรณีล้มเหลวได้

บทถัดไป: 8.6 จะใช้สิ่งที่เก็บไว้ตอนเปิดแอป และแก้อาการเด้งไปหน้าเข้าสู่ระบบแวบหนึ่งที่คนเจอบ่อย
