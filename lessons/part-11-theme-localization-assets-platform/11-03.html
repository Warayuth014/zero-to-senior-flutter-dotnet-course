1. design token คืออะไร

ค่าเล็ก ๆ ที่ใช้ซ้ำทั่วแอป — สี ระยะห่าง ขนาดตัวอักษร ความโค้ง — ที่ถูกตั้งชื่อและเก็บไว้ที่เดียว

คำว่า token มาจากวงการออกแบบ หมายถึงหน่วยที่เล็กที่สุดที่ยังมีความหมาย ไม่ใช่ศัพท์ของ Flutter โดยเฉพาะ

2. กฎข้อเดียวที่สำคัญที่สุด

ตั้งชื่อตามหน้าที่ ไม่ใช่ตามหน้าตา
ชื่อที่ผูกกับหน้าตาชื่อที่ผูกกับหน้าที่
blueprimary
lightGreyborder
reddanger
size14bodyMedium
padding12screenPadding

เหตุผลชัดตอนที่ต้องเปลี่ยน — วันที่แบรนด์เปลี่ยนจากน้ำเงินเป็นเขียว โค้ดที่เขียนว่า blue จะกลายเป็นคำโกหกทั้งโปรเจกต์ ทางเลือกคือไล่เปลี่ยนชื่อทุกที่ หรือปล่อยให้ AppColors.blue เป็นสีเขียวไปตลอดกาล ซึ่งทั้งสองทางแย่

กฎนี้มีข้อยกเว้นเดียว คือชั้นล่างสุดที่เป็น ชุดสีดิบ

// ชั้นล่าง — ชื่อตามสีได้ เพราะมันคือสีจริง ๆ
class Palette {
  static const blue600 = Color(0xFF2E73B9);
  static const blue900 = Color(0xFF174D80);
}

// ชั้นบน — ชื่อตามหน้าที่ อ้างถึงชั้นล่าง
class PdaColors {
  static const primary = Palette.blue600;
  static const primaryDark = Palette.blue900;
}

สองชั้นแบบนี้คุ้มเมื่อมีธีมหลายชุด ถ้ามีธีมเดียวก็ใช้ชั้นเดียวพอ ตามหลักใน 9.12

3. token ที่มีเหตุผลเขียนไว้

/// ขนาดต่ำสุดของสิ่งที่กดได้
///
/// 48 มาจากแนวทางของ Material แต่บน PDA มีเหตุผลเพิ่มคือผู้ใช้ใส่ถุงมือ
/// และกดขณะเดิน ปุ่มหลักในแอปนี้จึงสูง 54 ไม่ใช่ 48
static const double minTapTarget = 48;
static const double primaryButtonHeight = 54;

คอมเมนต์นี้ตอบคำถามที่จะถูกถามแน่นอน คือ “ทำไม 54 ไม่ใช่ 48 หรือ 56” ถ้าไม่เขียนไว้ วันหนึ่งจะมีคนเปลี่ยนเป็น 48 เพราะ “มาตรฐานบอกว่า 48” แล้วผู้ใช้ที่ใส่ถุงมือจะกดพลาดบ่อยขึ้นโดยไม่มีใครโยงสาเหตุได้

เกณฑ์: ตัวเลขที่ไม่ใช่ค่าที่ชัดในตัวเอง ต้องมีคอมเมนต์บอกที่มา — screenPadding = 12 ไม่ต้อง แต่ primaryButtonHeight = 54 ต้อง

4. อย่าตั้ง token ให้ทุกค่า

ควรเป็น tokenไม่ควร
ใช้ซ้ำอย่างน้อยสามที่ใช้ที่เดียว
เปลี่ยนแล้วต้องเปลี่ยนพร้อมกันทุกที่ค่าที่เฉพาะเจาะจงกับจอเดียว
มีคนถามว่า “ค่านี้ควรเป็นเท่าไหร่”ค่าที่ไม่มีใครสนใจ
// อย่าทำ — token ที่ไม่มีใครใช้ซ้ำ
static const double receivingCardTopMargin = 7;
static const double palletBadgeLetterSpacing = 0.4;

ผลของการทำเกินคือรายการ token ยาวจนหาไม่เจอ แล้วคนก็เขียนค่าลงไปตรง ๆ อยู่ดี — token ที่หาไม่เจอ มีค่าเท่ากับไม่มี

5. ระยะห่างควรมาจากมาตราส่วน

class Spacing {
  static const double xs = 4;
  static const double sm = 8;
  static const double md = 12;
  static const double lg = 16;
  static const double xl = 24;
}

ประโยชน์ไม่ใช่การประหยัดการพิมพ์ แต่คือการตัดตัวเลือกที่ไม่ควรมี — เมื่อมีแค่ห้าค่า จะไม่มีจอไหนใช้ 7 หรือ 13 และหน้าจอทั้งแอปจะดูเป็นชุดเดียวกันโดยอัตโนมัติ

ถ้าเจอที่ที่ค่าในมาตราส่วนใช้ไม่ได้เลย ให้ถามว่าเป็นเพราะ layout ผิดหรือเปล่า ก่อนที่จะเพิ่มค่าใหม่เข้าไป

6. token ไม่ควรรู้จัก Flutter มากเกิน

// ดี — เก็บค่า
static const Color primary = Color(0xFF2E73B9);
static const double cardRadius = 16;

// ไม่ดี — เก็บ widget
static Widget get primaryButton => FilledButton(...);
static BoxDecoration get cardDecoration => BoxDecoration(...);

บรรทัดล่างทำให้ไฟล์ token กลายเป็นไฟล์ UI ซึ่งจะโตขึ้นเรื่อย ๆ จนกลายเป็นไฟล์ที่ทุกคนต้องแก้ — ปัญหาเดียวกับ common_widgets.dart ที่ 9.14 พูดถึง

สิ่งที่ประกอบจาก token ควรอยู่ในธีม (ถ้าเป็นค่าตั้งต้นของ widget สำเร็จรูป) หรือเป็น widget ของตัวเอง (ถ้าเป็นความหมายเฉพาะ) ตามที่ 11.1 §9 สรุปไว้

7. ของจริง: สองชั้นที่ทับกัน

// core/app_theme.dart — ของกลางทั้งแอป
class AppColors {
  static const Color primary = Color(0xFF2E73B9);
  static const Color heading = Color(0xFF102A43);
  static const Color success = Color(0xFF1F9D62);
  ...
}

// features/warehouse/ui/pda_colors.dart — เฉพาะฟีเจอร์คลัง
class PdaColors {
  static const shellNavy = Color(0xFF062B50);      // ของฟีเจอร์นี้
  static const agv = Color(0xFF16845C);            // สีของรถ AGV
  static const fork = Color(0xFFB4770C);           // สีของรถโฟล์คลิฟท์
  static const hdr = AppColors.primary;            // ← อ้างของกลาง
  static const green = AppColors.success;          // ← อ้างของกลาง
}

โครงนี้ตรงกับกฎใน 9.11 คือของกลางอยู่ core ของเฉพาะฟีเจอร์อยู่ในฟีเจอร์ และฟีเจอร์อ้างของกลางได้ แต่ไม่กลับกัน

จุดที่ต้องระวังคือบรรทัดอย่าง static const green = AppColors.success; — เป็นการสร้างชื่อที่สองให้ของเดิม ซึ่งทำให้ค้นหาว่าใครใช้สีนี้บ้างยากขึ้น เพราะต้องค้นสองชื่อ

ได้เสีย
โค้ดในฟีเจอร์เขียนสั้น อ่านง่ายค้นหาไม่เจอครบ
เปลี่ยนใจภายหลังได้ว่าฟีเจอร์นี้จะใช้สีเขียวของตัวเองคนใหม่ไม่รู้ว่าสองชื่อนี้คือสีเดียวกัน

คุ้มไม่คุ้มขึ้นกับว่าฟีเจอร์นั้นมีโอกาสแยกตัวจริงไหม ถ้าไม่มี ให้ใช้ชื่อเดียว

8. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
AppColors.blue เป็นสีเขียวตั้งชื่อตามสีตั้งตามหน้าที่
ไม่มีใครใช้ token ที่มีอยู่รายการยาวเกินจนหาไม่เจอตัดตัวที่ใช้ที่เดียวออก
ระยะห่างในแอปไม่เป็นชุดเดียวกันไม่มีมาตราส่วนกำหนด 5–6 ค่าแล้วใช้แค่นั้น
ไฟล์ token กลายเป็นไฟล์ที่ทุกคนต้องแก้เก็บ widget ไว้ในนั้นเก็บแต่ค่า
เปลี่ยนตัวเลขแล้วไม่มีใครรู้ว่าทำไมเคยเป็นค่านั้นไม่มีคอมเมนต์บอกที่มาเขียนเหตุผลไว้กับค่าที่ไม่ชัดในตัวเอง

9. แบบฝึกหัด

ทีมกำลังจะเริ่มโปรเจกต์ใหม่ และเสนอโครง token นี้

class Tokens {
  static const white = Color(0xFFFFFFFF);
  static const black = Color(0xFF000000);
  static const blue = Color(0xFF2E73B9);
  static const lightBlue = Color(0xFFE9F3FF);
  static const grey1 = Color(0xFFF6F9FC);
  static const grey2 = Color(0xFFD9E5EF);
  static const grey3 = Color(0xFF60758A);
  static const grey4 = Color(0xFF102A43);
}

ปรับปรุงอย่างไร

แนวคำตอบ

ปัญหาสามข้อ

ปัญหาผลที่เกิด
ชื่อตามสีเปลี่ยนแบรนด์แล้วชื่อโกหก
grey1..grey4 ไม่บอกอะไรเลยคนเลือกใช้แบบสุ่ม แล้วสีเทาในแอปจะปนกันสี่แบบ
ไม่มีสีคู่ไม่มีใครรู้ว่าตัวอักษรบน blue ควรเป็นสีอะไร (11.2)
class AppColors {
  // พื้นผิว เรียงจากลึกไปตื้น
  static const pageBg = Color(0xFFF6F9FC);
  static const surface = Color(0xFFFFFFFF);

  // เส้น
  static const border = Color(0xFFD9E5EF);

  // ตัวอักษร เรียงตามลำดับความสำคัญ
  static const heading = Color(0xFF102A43);
  static const body = Color(0xFF102A43);
  static const subtitle = Color(0xFF60758A);

  // แบรนด์ พร้อมสีที่วางบนได้
  static const primary = Color(0xFF2E73B9);
  static const onPrimary = Color(0xFFFFFFFF);
  static const primarySoft = Color(0xFFE9F3FF);
  static const onPrimarySoft = Color(0xFF174D80);
}

สังเกตสองอย่าง

  • heading กับ body เป็นสีเดียวกันตอนนี้ แต่แยกชื่อไว้ เพราะวันที่ฝ่ายออกแบบอยากให้หัวข้อเข้มกว่าเนื้อความ จะแก้ได้โดยไม่ต้องไล่ดูว่าที่ไหนเป็นหัวข้อ
  • ไม่มี black และ white ล้วน — ดำสนิทบนขาวสนิทมีความต่างสูงถึง 21 ซึ่งจ้าเกินไปเมื่ออ่านนาน ๆ ตัวอักษรจึงใช้ #102A43 ที่เป็นน้ำเงินเข้มมากแทน

ข้อสุดท้ายเป็นรายละเอียดที่ทำให้แอปดูดีขึ้นโดยที่ผู้ใช้บอกไม่ถูกว่าเพราะอะไร

10. สรุป

  • ตั้งชื่อตามหน้าที่ ยกเว้นชั้นล่างสุดที่เป็นชุดสีดิบ
  • เขียนเหตุผลไว้กับตัวเลขที่ไม่ชัดในตัวเอง
  • ทำ token เฉพาะค่าที่ใช้ซ้ำจริง — รายการที่ยาวเกินคือรายการที่ไม่มีใครใช้
  • ระยะห่างมาจากมาตราส่วนที่มีไม่กี่ค่า
  • เก็บแต่ค่า อย่าเก็บ widget

บทหน้า: เรื่องที่กระทบทุกบรรทัดของข้อความไทย
