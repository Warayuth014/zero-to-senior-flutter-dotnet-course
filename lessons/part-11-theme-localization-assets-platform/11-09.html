1. ทำไมเรื่องนี้อันตราย

ข้อความที่แปลผิด ผู้ใช้เห็นแล้วรู้ทันที แต่ตัวเลขและวันที่ที่จัดรูปแบบผิด ดูปกติทุกประการ

สิ่งที่แสดงอาจหมายถึงผู้ใช้รู้ไหม
08/09/20269 สิงหาคม หรือ 8 กันยายนไม่
1,234หนึ่งพันสองร้อยสามสิบสี่ หรือ 1.234 ในบางประเทศไม่
15 มี.ค. 26ปี ค.ศ. 2026 หรือ พ.ศ. 2526ไม่
14:32เวลาเครื่อง หรือเวลา UTCไม่

แถวสุดท้ายเจ็บที่สุดในงานคลัง — ใบรับของที่บอกว่ารับเข้าตอน 14:32 ทั้งที่จริงคือ 21:32 จะทำให้การสอบทานย้อนหลังผิดทั้งกะ

2. รวมไว้ที่เดียว

/// การจัดรูปแบบวันที่และตัวเลขให้ตรงกับภาษาที่ผู้ใช้เลือก
///
/// รวมไว้ที่เดียวเพราะกฎเหล่านี้ต้องเหมือนกันทุกจอ และเพราะกฎบางข้อ
/// (เช่นปี พ.ศ.) ผิดง่ายมากถ้าปล่อยให้แต่ละจอทำเอง
library;

เหตุผลเดียวกับ json_read.dart ใน Part 6 — กฎที่ต้องเหมือนกันทุกที่ ต้องอยู่ที่เดียว

3. ตัวเลข

/// จำนวนชิ้น
///
/// ใส่ตัวคั่นหลักพันเสมอ เพราะ 12000 กับ 1200 บนจอเล็กที่มองแวบเดียว
/// แยกยาก แต่ 12,000 กับ 1,200 แยกออกทันที
String formatCount(num value, AppLanguage language) =>
    NumberFormat.decimalPattern(language.code).format(value);
/// จำนวนที่มีทศนิยม เช่นน้ำหนักหรือความยาว
///
/// บังคับให้แสดงทศนิยมตามที่กำหนดเสมอ ไม่ตัดศูนย์ท้ายทิ้ง — 12.0 กับ 12
/// สื่อความแม่นยำต่างกันในใบรับของ
String formatQuantity(num value, AppLanguage language, {int decimals = 2}) {
  final pattern = decimals == 0 ? '#,##0' : '#,##0.${'0' * decimals}';
  return NumberFormat(pattern, language.code).format(value);
}
สัญลักษณ์ในรูปแบบความหมาย
#แสดงถ้ามีค่า ไม่แสดงถ้าเป็นศูนย์นำหน้า
0แสดงเสมอ เติมศูนย์ถ้าไม่มี
,ตัวคั่นกลุ่ม

#,##0.00 จึงแปลว่า “คั่นหลักพัน อย่างน้อยหนึ่งหลักหน้าจุด และสองหลักหลังจุดเสมอ”

4. เวลา — กฎที่ห้ามสลับ

เก็บเป็น UTC แสดงเป็นเวลาท้องถิ่น — สลับเมื่อไหร่พังเมื่อนั้น
String formatTime(DateTime utc, AppLanguage language) =>
    DateFormat.Hm(language.code).format(utc.toLocal());
ขั้นรูปแบบใครรับผิดชอบ
เซิร์ฟเวอร์ส่งมาISO พร้อม Zทีมหลังบ้าน
อ่านเข้าโมเดลDateTime ที่เป็น UTCrequireUtcTime (6.9)
เก็บในหน่วยความจำUTCstore
แสดงบนจอเวลาท้องถิ่นformatTime
ส่งกลับเซิร์ฟเวอร์UTCrepository

มีจุดเดียวที่แปลงเป็นเวลาท้องถิ่น คือตอนแสดง เทสต์บันทึกกฎนี้ไว้

test('เวลาแปลงจาก UTC เป็นเวลาเครื่องก่อนแสดง', () {
  final utc = DateTime.utc(2026, 8, 21, 7, 32);
  final expected = utc.toLocal();
  final shown = formatTime(utc, AppLanguage.english);

  expect(shown, '${expected.hour.toString().padLeft(2, '0')}:'
      '${expected.minute.toString().padLeft(2, '0')}');
});

สังเกตว่าเทสต์คำนวณค่าที่คาดหวังจาก toLocal() เอง ไม่ได้เขียน '14:32' ตรง ๆ — ไม่งั้นเทสต์จะผ่านเฉพาะบนเครื่องที่ตั้งเขตเวลาเป็นไทย และล้มบนเครื่องของทีมงานที่อยู่คนละประเทศหรือบน CI

5. ปี พ.ศ. — กรณีที่ intl ไม่ช่วย

String formatDate(DateTime utc, AppLanguage language) {
  final local = utc.toLocal();
  if (language == AppLanguage.thai) {
    // ไทยใช้ พ.ศ. ซึ่ง intl ไม่ได้แปลงให้ ต้องบวก 543 เอง แล้วใส่ลงไป
    // เป็นข้อความ ไม่ใช่ปล่อยให้ DateFormat อ่านปีจาก DateTime
    final buddhistYear = local.year + 543;
    final day = local.day.toString().padLeft(2, '0');
    final month = DateFormat.MMM('th').format(local);
    return '$day $month ${buddhistYear % 100}';
  }
  return DateFormat('dd MMM yy', language.code).format(local);
}

นี่คือกับดักที่คนไทยเจอกันทุกโปรเจกต์ — DateFormat('dd MMM yyyy', 'th') ให้ชื่อเดือนเป็นภาษาไทยถูกต้อง แต่ปียังเป็น ค.ศ. ผลคือได้ “15 มี.ค. 2026” ซึ่งดูเป็นภาษาไทยทุกอย่าง ยกเว้นปีที่ผิดไป 543 ปี

ทางที่ปลอดภัยคือคำนวณปีเองแล้วประกอบข้อความ ไม่ใช่หวังว่า DateFormat จะรู้

test('วันที่ภาษาไทยใช้ปี พ.ศ.', () {
  final utc = DateTime.utc(2026, 3, 15, 4, 0);
  final local = utc.toLocal();
  final thai = formatDate(utc, AppLanguage.thai);

  // 2026 + 543 = 2569 → แสดงสองหลักท้ายคือ 69
  expect(thai, contains(((local.year + 543) % 100).toString()));
  // และต้องไม่มี ค.ศ. หลุดออกมา
  expect(thai, isNot(contains('26')));
});
คำถามที่ต้องถามทีมก่อนเขียน: ใบรับของที่พิมพ์ออกมาใช้ พ.ศ. หรือ ค.ศ. — เพราะถ้าหน้าจอกับใบพิมพ์ไม่ตรงกัน จะเกิดปัญหาตอนสอบทาน และเป็นคำถามที่โปรแกรมเมอร์ตอบเองไม่ได้

6. เวลาที่ผ่านมาแล้ว

String formatElapsed(DateTime utc, AppLanguage language, {DateTime? now}) {
  final elapsed = (now ?? DateTime.now()).toUtc().difference(utc.toUtc());
  final thai = language == AppLanguage.thai;

  if (elapsed.isNegative || elapsed.inSeconds < 60) {
    return thai ? 'เมื่อครู่' : 'just now';
  }
  ...
  // เกินหนึ่งวันแล้ว เวลาที่ผ่านมาไม่มีประโยชน์เท่าวันที่จริง
  return formatDate(utc, language);
}
รายละเอียดเหตุผล
now รับเข้ามาได้เทสต์คุมเวลาได้ ไม่ต้องพึ่งนาฬิกาจริง (9.8)
elapsed.isNegativeนาฬิกาเครื่องกับเซิร์ฟเวอร์ไม่ตรงกันเป็นเรื่องปกติ (8.3) — ถ้าไม่ดัก จะได้ “-3 นาทีที่แล้ว”
เกินหนึ่งวันแล้วเปลี่ยนเป็นวันที่“73 ชั่วโมงที่แล้ว” ต้องเอาไปคำนวณในหัว ส่วน “18 ส.ค.” อ่านแล้วรู้เลย
test('เวลาในอนาคตต้องไม่กลายเป็นค่าติดลบประหลาด', () {
  final now = DateTime.utc(2026, 8, 21, 12, 0);
  final future = now.add(const Duration(minutes: 2));

  expect(formatElapsed(future, AppLanguage.thai, now: now), 'เมื่อครู่');
});

7. intl ต้องโหลดข้อมูลก่อนใช้

// ในเทสต์
setUpAll(() => initializeDateFormatting());

ถ้าไม่เรียก DateFormat.MMM('th') จะคืนชื่อเดือนภาษาอังกฤษเงียบ ๆ ไม่โยนข้อผิดพลาด — เป็นความล้มเหลวที่เงียบที่สุดในบทนี้

ในแอปจริง flutter_localizations เรียกให้อัตโนมัติเมื่อ MaterialApp ถูกสร้าง แต่ในเทสต์ที่ทดสอบฟังก์ชันล้วน ๆ ไม่มี MaterialApp จึงต้องเรียกเอง

8. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ปีไทยผิดไป 543หวังว่า DateFormat จะแปลงให้บวกเอง
เวลาเพี้ยนไป 7 ชั่วโมงลืม toLocal() หรือเรียกสองครั้งแปลงที่จุดแสดงจุดเดียว
เทสต์ผ่านบนเครื่องเรา ล้มบน CIเขียนค่าเวลาที่คาดหวังตรง ๆคำนวณจาก toLocal() ในเทสต์
ชื่อเดือนเป็นอังกฤษทั้งที่ตั้ง locale แล้วลืม initializeDateFormatting()เรียกใน setUpAll
“-5 นาทีที่แล้ว”นาฬิกาไม่ตรงดัก isNegative
จำนวนแสดง 12 แทน 12.00ใช้ toString()ใช้รูปแบบที่บังคับทศนิยม

9. แบบฝึกหัด

หน้าจอประวัติการรับเข้าแสดงเวลาแบบ formatElapsed ผู้ใช้รายงานสองเรื่อง

  1. “รายการที่รับเข้าเมื่อวานขึ้นว่า 23 ชั่วโมงที่แล้ว ทั้งที่เป็นคนละวัน”
  2. “หัวหน้าให้ส่งรายงานว่ารับของเข้าเวลาไหนบ้าง แต่หน้าจอไม่บอกเวลาจริง”
แนวคำตอบ

1. ถูกตามที่เขียนไว้ แต่ผิดตามที่ผู้ใช้คิด — เขาไม่ได้นับเป็นชั่วโมง เขานับเป็น “วันนี้” กับ “เมื่อวาน”

String formatElapsed(DateTime utc, AppLanguage language, {DateTime? now}) {
  final current = (now ?? DateTime.now()).toLocal();
  final local = utc.toLocal();
  ...
  // เทียบวันตามปฏิทิน ไม่ใช่ตามจำนวนชั่วโมง
  final today = DateTime(current.year, current.month, current.day);
  final day = DateTime(local.year, local.month, local.day);
  final dayDiff = today.difference(day).inDays;

  if (dayDiff == 1) {
    return thai ? 'เมื่อวาน ${formatTime(utc, language)}' : 'Yesterday ...';
  }
  ...
}

บทเรียนที่ใหญ่กว่าโค้ด: “หนึ่งวัน” ในความหมายของผู้ใช้คือขอบเขตของปฏิทิน ไม่ใช่ 24 ชั่วโมง — เวลา 23:50 กับ 00:10 ห่างกันยี่สิบนาทีแต่เป็นคนละวัน

2. นี่ไม่ใช่บั๊ก แต่เป็นข้อจำกัดของการเลือกใช้ “เวลาที่ผ่านมา” ซึ่งดีสำหรับการดูว่าเพิ่งเกิดหรือเปล่า แต่ใช้อ้างอิงไม่ได้

ทางแก้ผล
แสดงทั้งสองอย่าง“3 นาทีที่แล้ว · 14:32” — กินที่ แต่ตอบได้ทั้งสองคำถาม
เวลาที่ผ่านมาในลิสต์ เวลาจริงในหน้ารายละเอียดลิสต์ยังสะอาด และคนที่ต้องการเวลาจริงกดเข้าไปดูได้
กดค้างเพื่อดูเวลาจริงไม่กินที่เลย แต่ไม่มีใครรู้ว่าทำได้

ทางที่สองดีที่สุดสำหรับ PDA เพราะจอแคบ และสิ่งที่หัวหน้าต้องการคือรายงาน ซึ่งควรมาจากระบบเว็บที่ออกรายงานได้ ไม่ใช่จากการที่พนักงานอ่านทีละรายการจากหน้าจอ PDA แล้วจดใส่กระดาษ

ข้อสังเกตสุดท้ายสำคัญกว่าโค้ด — เมื่อผู้ใช้ขอฟีเจอร์ ให้ถามว่าเขาจะเอาไปทำอะไร บ่อยครั้งคำตอบที่ดีที่สุดไม่ได้อยู่ในแอปนี้

10. สรุป

  • ตัวเลขและวันที่ที่ผิด ดูเหมือนถูกทุกประการ
  • เก็บเป็น UTC แสดงเป็นเวลาท้องถิ่น แปลงที่จุดเดียว
  • ปี พ.ศ. ต้องบวกเอง intl ไม่แปลงให้
  • เทสต์ต้องคำนวณเวลาที่คาดหวังเอง ไม่ใช่เขียนค่าตรง ๆ
  • ดักเวลาในอนาคต เพราะนาฬิกาไม่ตรงกันเป็นเรื่องปกติ
  • “หนึ่งวัน” ของผู้ใช้คือขอบเขตปฏิทิน ไม่ใช่ 24 ชั่วโมง
