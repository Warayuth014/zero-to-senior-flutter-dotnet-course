class ThemeProvider extends ValueNotifier<ThemeMode> {

  void toggle() { value = isDark ? ThemeMode.light : ThemeMode.dark; }

3. ถ้าจะทำ ทำอย่างไรให้ไม่พัง

MaterialApp(
  theme: buildPdaTheme(Brightness.light),
  darkTheme: buildPdaTheme(Brightness.dark),
  themeMode: controller.mode,     // light / dark / system
)
themeModeหมายถึง
ThemeMode.systemตามการตั้งค่าของเครื่อง
ThemeMode.light / darkบังคับ ไม่สนเครื่อง

สิ่งที่ต้องระวังที่สุดคือ อย่าเขียนฟังก์ชันสร้างธีมสองตัวแยกกัน ให้เขียนตัวเดียวที่รับ Brightness เข้าไป ไม่งั้นจะมีค่าที่แก้ในตัวหนึ่งแต่ลืมแก้ในอีกตัวเสมอ

ThemeData buildPdaTheme(Brightness brightness) {
  final dark = brightness == Brightness.dark;
  final scheme = ColorScheme(
    brightness: brightness,
    primary: dark ? PdaColors.primaryLight : PdaColors.primary,
    onPrimary: Colors.white,
    surface: dark ? PdaColors.surfaceDark : PdaColors.surface,
    onSurface: dark ? PdaColors.headingDark : PdaColors.heading,
    ...
  );
  // ส่วนที่ไม่เกี่ยวกับสี เช่นความสูงปุ่มและมุมโค้ง เขียนครั้งเดียว
  return ThemeData(colorScheme: scheme, ...);
}

4. คำถามที่แท้จริงไม่ใช่สว่างหรือมืด แต่คืออ่านออกไหม

Contrast ratio คือตัวเลขบอกความต่างระหว่างสีตัวอักษรกับสีพื้น มีค่าตั้งแต่ 1 (สีเดียวกัน) ถึง 21 (ดำสนิทบนขาวสนิท)
เกณฑ์ WCAGใช้กับ
4.5 ขึ้นไปตัวอักษรขนาดปกติ
3.0 ขึ้นไปตัวใหญ่ 18pt ขึ้นไป หรือ 14pt ตัวหนา
3.0 ขึ้นไปเส้นขอบของช่องกรอกและไอคอนที่สื่อความหมาย

ในคลังที่แสงจ้า ตัวเลขที่ผ่านเกณฑ์บนจอคอมอาจยังอ่านไม่ออกที่หน้างาน ตัวเลขนี้จึงเป็นขั้นต่ำ ไม่ใช่เป้าหมาย

5. วัดด้วยโค้ด

/// ความสว่างสัมพัทธ์ตามสูตรของ WCAG
///
/// ไม่ใช่ค่าเฉลี่ยของ R G B ธรรมดา — ตาคนไวต่อสีเขียวมากที่สุดและไวต่อ
/// สีน้ำเงินน้อยที่สุด สูตรนี้จึงถ่วงน้ำหนักต่างกัน
double relativeLuminance(Color color) {
  double channel(double value) => value <= 0.03928
      ? value / 12.92
      : math.pow((value + 0.055) / 1.055, 2.4).toDouble();

  return 0.2126 * channel(color.r) +
      0.7152 * channel(color.g) +
      0.0722 * channel(color.b);
}

double contrastRatio(Color foreground, Color background) {
  final a = relativeLuminance(foreground);
  final b = relativeLuminance(background);
  final lighter = math.max(a, b);
  final darker = math.min(a, b);
  return (lighter + 0.05) / (darker + 0.05);
}

สังเกตน้ำหนัก 0.7152 ของสีเขียว เทียบกับ 0.0722 ของสีน้ำเงิน — นี่คือเหตุผลที่ตัวอักษรสีน้ำเงินเข้มบนพื้นดำอ่านไม่ออก ทั้งที่ดูเหมือนต่างกันมากตอนเลือกสี

6. เทสต์ที่จับปัญหาจริงได้

test('ตัวอักษรบนพื้นต้องผ่านเกณฑ์ความต่างของ WCAG', () {
  final pairs = <String, (Color, Color)>{
    'หัวข้อบนพื้นขาว': (PdaColors.heading, PdaColors.surface),
    'หัวข้อบนพื้นหน้า': (PdaColors.heading, PdaColors.pageBg),
    'ตัวอักษรขาวบนปุ่มหลัก': (Colors.white, PdaColors.primary),
  };

  for (final entry in pairs.entries) {
    final (foreground, background) = entry.value;
    expect(contrastRatio(foreground, background), greaterThanOrEqualTo(4.5),
        reason: '${entry.key} ความต่างไม่พอ');
  }
});

เทสต์นี้จับปัญหาจริงตอนเขียน lab นี้ — สีบอกสถานะสามสีที่คัดลอกมาจากโปรเจกต์จริง ไม่ผ่านเกณฑ์เมื่อใช้ตัวอักษรสีขาว

สีพื้นตัวอักษรขาวเกณฑ์ 4.5
#D64545 แดง4.38ไม่ผ่าน (เฉียด)
#1F9D62 เขียว3.47ไม่ผ่าน
#D78A13 เหลืองส้ม2.79ไม่ผ่านชัดเจน

ตัวเลขเหล่านี้ไม่ได้แปลว่าโปรเจกต์จริงเขียนผิด — สีชุดนี้เป็นสีมาตรฐานที่แอปจำนวนมากใช้กัน และป้ายสถานะส่วนใหญ่ในแอปนั้นใช้สีเข้มบนพื้นจาง (greenSoft, redSoft, amberSoft) ซึ่งเป็นการแก้ปัญหาแบบหนึ่ง แต่มันแปลว่าถ้าวันไหนมีคนเผลอใช้ตัวขาวบนพื้นสีเต็ม จะไม่มีอะไรเตือน

7. ทางแก้สามทาง และทางที่ lab เลือก

ทางผล
ทำสีให้เข้มขึ้นได้ผลกับแดงและเขียว แต่เหลืองส้มที่เข้มพอจะรับตัวขาวได้ จะกลายเป็นสีน้ำตาล
ใช้ตัวอักษรใหญ่ขึ้นหรือหนาขึ้นเกณฑ์ลดเหลือ 3.0 แต่ป้ายสถานะบนจอเล็กใหญ่ขึ้นไม่ได้มาก
เปลี่ยนสีตัวอักษรตามพื้นได้ผลทุกกรณี แต่ต้องประกาศเป็นคู่
// ค่าที่ใช้ถูกทำให้เข้มกว่าสีแบรนด์ดั้งเดิม เพราะเทสต์จับได้ว่า
// ตัวอักษรขาวบนสีเดิมอ่านไม่ออกตามเกณฑ์
static const Color danger = Color(0xFFB3261E);
static const Color success = Color(0xFF157C4C);
static const Color warning = Color(0xFFD78A13);

/// สีตัวอักษรที่คู่กับพื้นสีสถานะ
///
/// ต้องประกาศคู่กันเสมอ ไม่ใช่สมมติว่าใช้สีขาวได้ทุกอัน
static const Color onDanger = Colors.white;
static const Color onSuccess = Colors.white;
static const Color onWarning = heading;      // ← ตัวเดียวที่ต่าง

และมีเทสต์ที่บันทึกข้อจำกัดไว้ ไม่ใช่บันทึกบั๊ก

test('ตัวอักษรขาวใช้กับสีเหลืองส้มไม่ได้ ไม่ว่าจะอยากใช้แค่ไหน', () {
  // เทสต์นี้บันทึกข้อจำกัดทางฟิสิกส์ไว้ เพื่อให้คนที่มาเปลี่ยน onWarning
  // เป็นสีขาวในอนาคต เห็นทันทีว่าทำไมถึงทำไม่ได้
  expect(contrastRatio(Colors.white, PdaColors.warning), lessThan(4.5));
});

8. สีอย่างเดียวไม่พอ

ผู้ชายประมาณ 8 คนใน 100 แยกแดงกับเขียวได้ไม่ดี ป้ายสถานะที่ต่างกันแค่สีจึงไม่ควรใช้

ไม่พอพอ
จุดเขียว / จุดแดงจุดเขียว + ✓ / จุดแดง + ✕
แถวสีแดงในตารางแถวสีแดง + ไอคอน + ข้อความ

ในคลังยังมีเหตุผลเพิ่มอีกข้อ คือผู้ใช้บางคนใส่แว่นตานิรภัยสีเหลือง ซึ่งเปลี่ยนการรับรู้สีทั้งหมด

9. แบบฝึกหัด

ฝ่ายออกแบบส่งสีใหม่มาสำหรับข้อความรอง คือ #9AA8B5 บนพื้นขาว บอกว่า “ดูสบายตากว่าเดิม”

  1. ควรรับไหม
  2. ถ้าไม่รับ จะอธิบายอย่างไร
แนวคำตอบ

1. ต้องวัดก่อนตอบ ไม่ใช่ตอบจากความรู้สึก

test('สีข้อความรองที่เสนอมา', () {
  expect(contrastRatio(const Color(0xFF9AA8B5), Colors.white),
      greaterThanOrEqualTo(4.5));
});

สีนี้ได้ประมาณ 2.5 ซึ่งไม่ผ่านแม้แต่เกณฑ์ของตัวใหญ่ ส่วนสีเดิม #60758A ได้ประมาณ 4.9 ผ่าน

2. สามอย่างที่ควรพูด และลำดับสำคัญ

  • เห็นด้วยกับเจตนาก่อน — ฝ่ายออกแบบพูดถูกว่าข้อความรองไม่ควรเด่นเท่าหัวข้อ นั่นคือเป้าหมายที่ถูกต้อง
  • แสดงตัวเลข ไม่ใช่ความเห็น — 2.5 กับ 4.5 เป็นข้อเท็จจริงที่เถียงกันไม่ได้ ต่างจาก “ผมว่าจางไป”
  • เสนอทางที่ได้ทั้งสองอย่าง — ลดขนาดหรือน้ำหนักตัวอักษรแทนการลดความเข้มของสี ได้ลำดับความสำคัญเหมือนกันโดยยังอ่านออก

ที่สำคัญกว่านั้น ให้เสนอว่าเพิ่มเทสต์นี้เข้าไปในโปรเจกต์ — ครั้งหน้าที่มีคนเสนอสีใหม่ จะรู้คำตอบภายในสิบวินาทีโดยไม่ต้องถกเถียงกัน

และควรบอกด้วยว่าสีที่ดูจางบนจอคอมในออฟฟิศ จะจางกว่านั้นมากบนจอ PDA กลางแดด ซึ่งเป็นข้อมูลที่ฝ่ายออกแบบไม่มี เพราะเขาไม่ได้ยืนอยู่ตรงนั้น

10. สรุป

  • โหมดมืดมีราคา ตัดสินจากที่ที่ผู้ใช้ยืนอยู่จริง ไม่ใช่จากว่ามันดูดี
  • ถ้าทำ ให้เขียนฟังก์ชันเดียวที่รับ Brightness ไม่ใช่สองฟังก์ชัน
  • ความต่างของสีวัดเป็นตัวเลขได้ และวัดในเทสต์ได้
  • สีพื้นกับสีตัวอักษรต้องประกาศเป็นคู่เสมอ
  • เหลืองส้มรับตัวอักษรขาวไม่ได้ ไม่ว่าจะปรับอย่างไร
  • อย่าให้สีเป็นสัญญาณเดียว
