1. ปัญหาที่บทนี้แก้

แอปคลังไม่ได้มีแค่งานเดียว — มีงานวันนี้ ประวัติ และข้อมูลของผู้ใช้ ซึ่งผู้ใช้ต้องสลับไปมาได้ตลอด

วิธีที่คนเขียนครั้งแรกมักคิดถึงคือ push หน้าใหม่เมื่อแตะแถบล่าง

แตะ "ประวัติ"  → push หน้าประวัติ
แตะ "งานวันนี้" → push หน้างานวันนี้อีกใบ
แตะ "ประวัติ"  → push อีกใบ

4. โครงพื้นฐาน

เปิด lib/nav_basics/nav_basics_app.dart

class _HomeShellState extends State<HomeShell> {
  static const List<String> _titles = ['งานวันนี้', 'ประวัติ', 'ของฉัน'];

  int _tabIndex = 0;                       // แหล่งความจริงเดียว

  @override
  Widget build(BuildContext context) => Scaffold(
    appBar: AppBar(title: Text(_titles[_tabIndex])),
    body: IndexedStack(

      children: [

    ),
    bottomNavigationBar: NavigationBar(

      onDestinationSelected: (index) => setState(() => _tabIndex = index),
      destinations: const [...],
    ),
  );
}

จุดสำคัญคือ _tabIndex ตัวเดียวคุมสามอย่าง — ชื่อบนแถบบน เนื้อหาตรงกลาง และปุ่มที่ถูกเน้นในแถบล่าง

// อย่าทำแบบนี้
int _selectedTab = 0;
Widget _currentBody = const TaskListTab();   // ค่าที่สองที่ต้องดูแลให้ตรงกัน

สองค่าที่ต้องตรงกันเสมอ คือสองค่าที่วันหนึ่งจะไม่ตรงกัน — เป็นหลักเดียวกับ 4.2

5. IndexedStack: เก็บทุกแท็บไว้

IndexedStack สร้างลูกทุกตัวและเก็บไว้ในต้นไม้ แต่วาดเฉพาะตัวที่เลือก

index: 1

TaskListTab   ← อยู่ในต้นไม้ State ยังอยู่ แต่ไม่ถูกวาด (offstage)
HistoryTab    ← ถูกวาด
ProfileTab    ← อยู่ในต้นไม้ แต่ไม่ถูกวาด
ได้อะไรแลกอะไร
ตำแหน่งที่เลื่อนไว้ยังอยู่ทุกแท็บกินหน่วยความจำพร้อมกัน
ข้อความที่พิมพ์ค้างไว้ยังอยู่สร้างทุกแท็บตั้งแต่เปิดหน้าจอ
ไม่ต้องโหลดข้อมูลใหม่ทุกครั้งที่สลับตัวจับเวลาในแท็บที่ไม่ได้ดูก็ยังเดิน

แถวสุดท้ายคือกับดักที่ต้องระวัง — ถ้าแท็บประวัติมี polling ทุก 10 วินาที (4.11) มันจะยิงต่อไปแม้ผู้ใช้ดูแท็บอื่นอยู่ ต้องหยุดเองเมื่อไม่ได้ถูกแสดง

รายการงานที่เลื่อนมาลึกเก็บไว้เลื่อนใหม่ทุกครั้งน่ารำคาญมาก

และมีข้อสำคัญที่ต้องเข้าใจ — ข้อมูลทางธุรกิจไม่ควรอยู่ใน widget ตั้งแต่แรก ถ้ามันอยู่ใน store ตาม Part 4 การสร้าง widget ใหม่ก็ไม่ทำให้ข้อมูลหาย เหลือแค่เรื่องตำแหน่งที่เลื่อนและข้อความที่พิมพ์ค้าง ซึ่งเป็นของ widget จริง ๆ

7. offstage: เรื่องที่ต้องรู้ตอนเขียนเทสต์

ตอนเขียนเทสต์ของ lab นี้ ผมเจอสิ่งนี้เข้าเต็ม ๆ

await tester.tap(find.text('ประวัติ'));
await tester.pumpAndSettle();

expect(find.byKey(const Key('task-list')), findsOneWidget);   // ล้มเหลว

ทั้งที่ IndexedStack เก็บแท็บแรกไว้ในต้นไม้จริง ๆ

สาเหตุคือ ตัวค้นหาของ Flutter ข้าม widget ที่ไม่ถูกวาดโดยค่าเริ่มต้น ซึ่งสมเหตุสมผล เพราะเทสต์ส่วนใหญ่ต้องการรู้ว่าผู้ใช้เห็นอะไร

expect(find.byKey(const Key('task-list')), findsNothing);
// IndexedStack เก็บแท็บที่ไม่ได้เลือกไว้แบบ offstage — ยังอยู่ในต้นไม้
// แต่ไม่ถูกวาด ตัวค้นหาจึงต้องบอกให้รวม offstage ด้วย
expect(
  find.byKey(const Key('task-list'), skipOffstage: false),
  findsOneWidget,
);

สองบรรทัดนี้อธิบายความต่างได้ชัดกว่าคำพูด — ไม่เห็น ไม่ได้แปลว่าไม่มี

8. แท็บที่มีหน้าซ้อนของตัวเอง

คำถามที่จะเจอทันทีเมื่อแอปโตขึ้น — ถ้าเปิดรายละเอียดงานจากแท็บแรก แล้วสลับไปแท็บสอง แล้วกลับมา ควรเห็นอะไร

แบบกลับมาแล้วเห็นความซับซ้อน
หน้าซ้อนอยู่บน shell (ที่ lab ใช้)หน้ารายละเอียดทับแถบล่างอยู่ตลอดง่าย
แต่ละแท็บมี Navigator ของตัวเองกลับมาเจอหน้ารายละเอียดของแท็บนั้นสูงขึ้นมาก

ใน lab หน้ารายละเอียดถูก push ขึ้นไปทับทั้ง shell รวมถึงแถบล่างด้วย

┌──────────────────┐
│ รายละเอียดงาน     │  ← ทับทั้งหมด ไม่เห็นแถบล่าง
├──────────────────┤
│ HomeShell        │
└──────────────────┘

ซึ่งเหมาะกับ PDA เพราะงานหนึ่งชิ้นควรได้พื้นที่เต็มจอ และไม่ควรให้ผู้ใช้สลับแท็บหนีระหว่างทำงานที่ยังไม่จบอยู่แล้ว

การให้แต่ละแท็บมี Navigator ของตัวเอง ทำให้ต้องตัดสินอีกหลายเรื่อง เช่นกล่องโต้ตอบควรอยู่บน navigator ตัวไหน และปุ่มย้อนกลับของเครื่องควรถอยใน navigator ตัวไหน — อย่าทำจนกว่าจะมีความต้องการจริง เรื่องนี้อยู่ใน Part 11

9. อย่าให้ shell ถูกสร้างใหม่

// ผิด — แตะ "งานวันนี้" แล้ว push shell ใหม่ทับ shell เดิม
onDestinationSelected: (index) => Navigator.of(context).push(
  MaterialPageRoute(builder: (_) => HomeShell(session: session)),
),

ผลที่ตามมาไม่ใช่แค่กองสูงขึ้น แต่ทุกอย่างที่ shell ถืออยู่จะถูกสร้างใหม่หมด

ถ้า shell ถูกสร้างใหม่ผลที่ตามมา
store ที่ shell สร้างไว้ ถูกสร้างใหม่โหลดข้อมูลใหม่ทั้งหมด
ตัวจับเวลาถูกตั้งใหม่มีสองตัววิ่งพร้อมกัน ถ้าตัวเก่าไม่ถูกยกเลิก (4.11)
ข้อมูลที่ยังไม่ได้บันทึกหายผู้ใช้เสียงาน

shell ควรถูกสร้างครั้งเดียวหลังเข้าสู่ระบบ และถูกทำลายตอนออกจากระบบเท่านั้น — ซึ่งเป็นเหตุผลที่ pushAndRemoveUntil ใน 5.4 สำคัญ เพราะมันคือจุดที่ shell ถูกทำลายจริงพร้อมกับทุกอย่างที่มันถืออยู่

11. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
กดย้อนกลับแล้วเจอแท็บเก่าไปเรื่อย ๆpush หน้าใหม่เมื่อแตะแถบล่างเปลี่ยนเฉพาะ index
ปุ่มที่เน้นกับเนื้อหาไม่ตรงกันเก็บสองค่าที่ต้องตรงกันใช้ค่าเดียว
สลับแท็บแล้วตำแหน่งที่เลื่อนหายสร้างใหม่ทุกครั้งIndexedStack
เปิดแอปช้าลงIndexedStack สร้างทุกแท็บพร้อมกันทำให้แท็บที่หนักโหลดข้อมูลเมื่อถูกเลือกเท่านั้น
ยังยิงคำขอของแท็บที่ไม่ได้ดูตัวจับเวลาไม่รู้ว่าถูกซ่อนหยุดเมื่อไม่ได้ถูกแสดง (4.11)
เทสต์หา widget ของแท็บอื่นไม่เจอตัวค้นหาข้าม offstageskipOffstage: false
ข้อมูลโหลดใหม่ทุกครั้งที่แตะแถบล่างpush shell ใหม่อย่า push shell

12. แบบฝึกหัด

แอปคลังหนึ่งมีสี่แท็บ

แท็บเนื้อหา
งานวันนี้รายการ 200 บรรทัด รีเฟรชทุก 10 วินาที
สแกนด่วนฟอร์มที่กรอกค้างไว้ได้
ประวัติรายการที่โหลดครั้งเดียวตอนเปิด ไม่เปลี่ยนอีก
ของฉันสามบรรทัด ไม่มีอะไรเลย
  1. ตัดสินว่าแต่ละแท็บควรถูกเก็บไว้หรือสร้างใหม่
  2. อธิบายว่าจะจัดการ polling ของแท็บแรกอย่างไร
  3. ตอบว่าถ้าเปิดแอปแล้วช้าเพราะสร้างทุกแท็บพร้อมกัน ควรแก้ที่ไหน
ดูแนวคำตอบและเหตุผล

ข้อ 1 — เก็บหรือสร้างใหม่

แท็บตัดสินเหตุผล
งานวันนี้เก็บเลื่อนมาลึกแล้ว การเด้งกลับบนสุดทำให้หาที่เดิมไม่เจอ
สแกนด่วนเก็บฟอร์มที่กรอกค้างไว้จะหาย
ประวัติเก็บโหลดครั้งเดียวแพง ไม่ควรโหลดซ้ำทุกครั้งที่สลับ
ของฉันสร้างใหม่ก็ได้ไม่มีอะไรให้เสียดาย

ในทางปฏิบัติ เมื่อสามในสี่ต้องเก็บ ก็ใช้ IndexedStack ทั้งหมดไปเลย — การผสมสองวิธีในหน้าเดียวเพิ่มความซับซ้อนโดยได้กลับมาน้อย

แต่ข้อสังเกตที่สำคัญกว่าคือ สองในสามเหตุผลนั้นไม่ควรเป็นเหตุผลตั้งแต่แรก

เหตุผลถ้าออกแบบดีแล้ว
รายการ 200 บรรทัดจะโหลดใหม่ไม่ควรเป็นปัญหา ถ้าข้อมูลอยู่ใน store ไม่ใช่ใน widget
ประวัติโหลดใหม่แพงเหมือนกัน — เก็บไว้ที่ store
ตำแหน่งที่เลื่อนไว้หายนี่คือเหตุผลที่แท้จริง เพราะเป็นของ widget จริง ๆ

ถ้าย้ายข้อมูลไปไว้ที่ store ตาม Part 4 แล้ว เหตุผลที่เหลือจะเหลือแค่ตำแหน่งที่เลื่อนและข้อความที่พิมพ์ค้าง ซึ่งแก้ด้วย PageStorageKey ได้โดยไม่ต้องเก็บทั้งแท็บไว้ในหน่วยความจำ

ข้อ 2 — polling ของแท็บแรก

ปัญหาคือ IndexedStack ไม่บอกลูกว่าตอนนี้ถูกแสดงอยู่หรือเปล่า มันแค่ไม่วาด

// shell เป็นคนรู้ จึงเป็นคนบอก
body: IndexedStack(
  index: _tabIndex,
  children: [
    TaskListTab(isActive: _tabIndex == 0),
    ScanTab(isActive: _tabIndex == 1),
    ...
  ],
),
// ในแท็บ — เริ่มหรือหยุดตามว่าถูกแสดงอยู่ไหม
@override
void didUpdateWidget(covariant TaskListTab oldWidget) {
  super.didUpdateWidget(oldWidget);
  if (oldWidget.isActive == widget.isActive) return;   // ต้องเทียบก่อน (4.12)
  if (widget.isActive) {
    _store.load();                    // กลับมาดู — รีเฟรชทันที
    _store.startPolling(const Duration(seconds: 10));
  } else {
    _store.stopPolling();             // ไม่ได้ดูแล้ว — หยุด
  }
}

สองอย่างที่ต้องมีคู่กันเสมอ ตาม 4.11 — หยุดเมื่อไม่มีใครดู และรีเฟรชทันทีเมื่อกลับมาดู ถ้าทำแค่อย่างแรก ผู้ใช้จะกลับมาเจอข้อมูลค้างจากเมื่อสิบนาทีก่อน

และยังต้องหยุดเมื่อแอปถูกย่อลงด้วย ซึ่งเป็นคนละเรื่องกับการสลับแท็บ

ข้อ 3 — เปิดแอปช้า

อาการนี้บอกว่าแท็บถูกสร้างพร้อมกันทั้งหมด แต่ ต้องแยกให้ออกก่อนว่าอะไรช้า

สิ่งที่ช้าแก้ที่
สร้าง widget สี่แท็บแทบไม่มีทางเป็นสาเหตุ — การสร้าง widget ถูกมาก (1.4)
initState ของทุกแท็บยิงคำขอพร้อมกันนี่คือสาเหตุจริง
สร้าง store สี่ตัวพร้อมกันถ้าแต่ละตัวโหลดข้อมูลทันที ก็เป็นเรื่องเดียวกัน

ดังนั้นไม่ควรแก้ด้วยการเลิกใช้ IndexedStack แต่ควรแก้ที่ แยกการสร้าง widget ออกจากการโหลดข้อมูล

// แท็บถูกสร้างตั้งแต่ต้น แต่ยังไม่โหลดอะไรจนกว่าจะถูกเปิดดูครั้งแรก
bool _loadedOnce = false;

void _loadIfNeeded() {
  if (_loadedOnce || !widget.isActive) return;
  _loadedOnce = true;
  _store.load();
}

ผลที่ได้คือเปิดแอปแล้วยิงคำขอเดียว ไม่ใช่สี่ และแท็บที่ผู้ใช้ไม่เคยแตะเลยทั้งกะ ก็ไม่เคยยิงคำขอเลย

บทเรียนที่ควรจำ: เวลาเจออาการช้า ให้หาว่าอะไรช้าจริง ก่อนเปลี่ยนโครงสร้าง — ตามลำดับใน 2.10 คือวัดก่อนเสมอ การเลิกใช้ IndexedStack จะแก้อาการได้จริง แต่แลกด้วยการเสียประโยชน์ทั้งหมดของมัน ทั้งที่สาเหตุจริงอยู่คนละที่

13. สรุปท้ายบท

  • แตะแถบล่างคือเปลี่ยนไส้ ไม่ใช่ push หน้าใหม่
  • ใช้ค่าเดียวคุมทั้งชื่อ เนื้อหา และปุ่มที่ถูกเน้น
  • IndexedStack เก็บทุกแท็บไว้ในต้นไม้ แต่วาดทีละตัว
  • สิ่งที่เก็บได้คือของที่เป็นของ widget จริง ๆ ส่วนข้อมูลควรอยู่ที่ store
  • แท็บที่ไม่ได้ถูกแสดงยังทำงานอยู่ — ต้องบอกให้มันหยุดเอง
  • เทสต์ต้องใช้ skipOffstage: false เมื่อจะหาของในแท็บที่ไม่ได้แสดง
  • อย่า push shell ใหม่ — shell ควรมีตัวเดียวตลอด session
  • ปลายทางบน PDA ควรมี 3–4 อัน ตามข้อจำกัดพื้นที่แตะ

บทถัดไป: 5.7 จะตอบคำถามที่ค้างจากบทนี้ — ข้อมูลที่ผู้ใช้ทำค้างไว้ ควรอยู่รอดแค่ไหน เมื่อสลับแท็บ เมื่อเปิดกล่องโต้ตอบ และเมื่อออกจากงานกลางคัน
