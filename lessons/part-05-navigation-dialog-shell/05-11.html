3. Mental model: แผ่นยืมค่าไปแก้ แล้วค่อยคืน

AppSession (ค่าจริง)          SettingsSheet (ร่าง)
  scanSound = true    ──คัดลอก──→  _scanSound = true
                                        │
                                   ผู้ใช้แตะสวิตช์
                                        │
                                   _scanSound = false   ← ค่าจริงยังเป็น true
                                        │
                                   กดบันทึก
  scanSound = false   ←──ทับ────────────┘
ประโยคเดียวที่ต้องจำ: ระหว่างที่แผ่นเปิดอยู่ ค่าจริงต้องไม่เปลี่ยนเลย — ทางเดียวที่มันจะเปลี่ยนคือผู้ใช้กดบันทึก

ถ้าไม่ทำแบบนี้ ปุ่มยกเลิกจะไม่มีความหมาย และผู้ใช้ที่แค่เปิดดูการตั้งค่าอาจเปลี่ยนค่าไปโดยไม่ตั้งใจ

0.0s  กดทดสอบที่อยู่ A → กำลังรอ
1.0s  ผู้ใช้แก้เป็นที่อยู่ B

10. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
เปิดดูการตั้งค่าเฉย ๆ แล้วค่าเปลี่ยนแก้ค่าจริงทันทีที่แตะเก็บร่างแล้วทับตอนบันทึก
ปุ่มยกเลิกไม่มีผลเหมือนข้างบนเหมือนข้างบน
ปุ่มบันทึกกดได้ทั้งที่ไม่ได้แก้อะไรไม่ได้เทียบกับค่าจริงคำนวณ _isDirty
ถามยืนยันทั้งที่แก้แล้วแก้กลับเก็บ dirty เป็น fieldเทียบค่าจริงทุกครั้ง
ลากแผ่นลงแล้วของหายไม่มี PopScopeครอบ
ผลทดสอบไม่ตรงกับค่าที่กรอกอยู่ไม่ได้จับคู่ผลกับค่าที่ทดสอบตัวนับรุ่น (4.10)

11. แบบฝึกหัด

ต้องทำแผ่นตั้งค่าการเชื่อมต่อ ที่มี

ส่วนรายละเอียด
รายการโปรไฟล์เลือกได้ว่าจะใช้อันไหน
ที่อยู่และพอร์ตแก้ได้ของโปรไฟล์ที่เลือก
ปุ่มทดสอบยิงไปเช็กว่าเชื่อมต่อได้ไหม
ปุ่มบันทึกบันทึกและเริ่มใช้โปรไฟล์นั้น
  1. ระบุว่าอะไรเป็นร่าง อะไรเป็นค่าจริง
  2. ตอบว่าปุ่มทดสอบควรทดสอบค่าไหน และผลควรผูกกับอะไร
  3. ผู้ใช้แก้ที่อยู่แล้วกดทดสอบ ระหว่างรอเขาสลับไปเลือกโปรไฟล์อื่น — ควรเกิดอะไร
ดูแนวคำตอบและเหตุผล

ข้อ 1 — ร่างกับค่าจริง

สิ่งของเป็นอะไรอยู่ที่ไหน
รายการโปรไฟล์ทั้งหมดค่าจริงstore
โปรไฟล์ที่ใช้งานอยู่จริงค่าจริงstore
โปรไฟล์ที่เลือกอยู่ในแผ่นร่างแผ่น
ที่อยู่และพอร์ตที่กำลังพิมพ์ร่างแผ่น
ผลการทดสอบล่าสุดร่างแผ่น

จุดที่คนมักจัดผิดคือแถวที่สาม — การเลือกโปรไฟล์ในแผ่น ยังไม่ใช่การสลับโปรไฟล์ ถ้าสลับทันทีที่แตะ คำขอที่กำลังวิ่งอยู่เบื้องหลังจะเปลี่ยนเซิร์ฟเวอร์กลางคัน

ข้อ 2 — ทดสอบค่าไหน และผลผูกกับอะไร

ทดสอบ ค่าในร่าง ไม่ใช่ค่าจริง — เพราะจุดประสงค์คือดูว่าค่าที่กำลังจะบันทึกใช้ได้ไหม

Future<void> _testConnection() async {
  if (_testing) return;                          // 4.14

  // จับค่าที่กำลังทดสอบไว้ ณ ตอนเริ่ม (4.8 หัวข้อ 7)
  final testedUrl = _buildUrl(_host, _port);
  final generation = ++_testGeneration;          // 4.10

  setState(() {
    _testing = true;
    _testResult = null;
  });

  try {
    final ok = await repository.ping(testedUrl);
    if (generation != _testGeneration || !mounted) return;
    setState(() => _testResult = TestResult(url: testedUrl, success: ok));
  } on ConnectionException catch (error) {
    if (generation != _testGeneration || !mounted) return;
    setState(() => _testResult = TestResult(
      url: testedUrl, success: false, message: error.message,
    ));
  } finally {                                     // 4.8
    if (generation == _testGeneration && mounted) {
      setState(() => _testing = false);
    }
  }
}

  // ค่าเปลี่ยนไปแล้ว — บอกว่ายังไม่ได้ทดสอบค่านี้

ข้อ 3 — สลับโปรไฟล์ระหว่างรอผล

สิ่งที่ต้องเกิดตามลำดับ

1. ผลการทดสอบเดิมต้องถูกทิ้ง
     _testGeneration++ → ผลของ A ที่กลับมาทีหลังจะถูกทิ้ง (4.10)

2. ปุ่มทดสอบต้องกลับมากดได้
     _testing = false — ไม่ใช่ค้างอยู่เพราะคำขอเก่ายังวิ่ง

3. ร่างต้องเปลี่ยนเป็นค่าของโปรไฟล์ใหม่
     _host, _port อ่านจากโปรไฟล์ที่เพิ่งเลือก

4. ผลการทดสอบที่แสดงอยู่ต้องหายไป
     เพราะเป็นผลของอีกโปรไฟล์หนึ่ง

แต่มีคำถามที่ต้องตอบก่อนทั้งหมดนี้ — ถ้าผู้ใช้แก้ที่อยู่ของโปรไฟล์ A ค้างไว้ แล้วสลับไปโปรไฟล์ B การแก้ของ A ควรหายไหม

ทางเลือกผลที่ตามมา
ทิ้งเงียบ ๆผู้ใช้กลับมาที่ A แล้วพบว่าที่แก้ไว้หาย โดยไม่มีใครบอก
ถามก่อนสลับชัดเจน แต่ถามบ่อยถ้าผู้ใช้สลับไปมาเพื่อเทียบค่า
เก็บร่างของทุกโปรไฟล์ไว้สลับไปมาได้อิสระ แต่ต้องบอกให้ชัดว่าอันไหนแก้ค้างไว้

คำตอบที่ควรให้คือทางที่สอง สำหรับแผ่นตั้งค่าทั่วไป เพราะทางที่สามซับซ้อนเกินความจำเป็น — คนไม่ได้แก้โปรไฟล์หลายอันพร้อมกันบ่อย ๆ

แต่ถ้าเลือกทางที่สาม ต้องแสดงเครื่องหมายบอกว่าโปรไฟล์ไหนมีการแก้ค้างอยู่ ไม่งั้นผู้ใช้จะกดบันทึกโดยไม่รู้ว่ากำลังบันทึกอะไรบ้าง

ข้อสังเกตสุดท้าย: แผ่นตั้งค่านี้มีทั้งร่าง การกันปิด คำสั่ง async ตัวนับรุ่น ธงกันซ้ำ และการแยกกติกาออกไปที่ store — มันคือแอปเล็ก ๆ หนึ่งตัวที่บังเอิญแสดงเป็นแผ่น การคิดว่ามันเป็น "แค่หน้าตั้งค่า" คือที่มาของบั๊กเกือบทั้งหมดในหัวข้อ 10

12. สรุปท้ายบท

  • แผ่นคัดลอกค่าจริงมาเป็นร่างครั้งเดียวใน initState
  • ค่าจริงต้องไม่เปลี่ยนจนกว่าผู้ใช้จะกดบันทึก
  • _isDirty คำนวณจากการเทียบกับค่าจริง ไม่ใช่เก็บเป็น field
  • บันทึกคือจุดเดียวที่แตะค่าจริง และเปลี่ยนทุกค่าพร้อมกัน
  • ครอบด้วย PopScope เพราะแผ่นปิดง่ายกว่าหน้าจอ
  • ปุ่มออกจากระบบต้องปิดแผ่นก่อน แล้วให้เจ้าของหน้าจอจัดการเส้นทาง
  • ปุ่มทดสอบต้องทดสอบค่าในร่าง และผูกผลกับค่าที่ทดสอบ
  • กติกาทางธุรกิจอยู่ที่ store ไม่ใช่ในแผ่น
