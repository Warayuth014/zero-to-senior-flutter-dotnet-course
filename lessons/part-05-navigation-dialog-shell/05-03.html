1. ปัญหาที่บทนี้แก้

หน้ารายละเอียดถูกปิดแล้ว — หน้ารายการต้องทำอะไรต่อ

ผู้ใช้ทำอะไรหน้ารายการต้องทำ
กดบันทึกอัปเดตแถวนั้น + แจ้งว่าบันทึกแล้ว
กดพักไว้ก่อนเก็บร่างไว้ แต่ทำเครื่องหมายว่ายังไม่เสร็จ
ยืนยันว่าจะทิ้งที่แก้ไว้ไม่เปลี่ยนอะไร + แจ้งว่ายกเลิกแล้ว
กดย้อนกลับทั้งที่ไม่ได้แก้อะไรไม่ทำอะไรเลย และไม่ต้องแจ้ง

สี่กรณี สี่การกระทำที่ต่างกัน แต่โค้ดที่คนเขียนครั้งแรกมักได้แค่สอง

final saved = await Navigator.push<bool>(...);
if (saved == true) refresh();     // แยกได้แค่ "บันทึก" กับ "ไม่บันทึก"

คำความหมาย

3. Mental model: หน้าที่ปิดไปกำลังรายงานผล

หน้ารายการ: "ไปทำงาน T-001 มา แล้วบอกฉันว่าเกิดอะไรขึ้น"
                    ↓ push
หน้ารายละเอียด: ... ทำงาน ...
                    ↓ pop(รายงาน)
หน้ารายการ: อ่านรายงาน แล้วตัดสินใจว่าจะทำอะไรต่อ
ประโยคเดียวที่ต้องจำ: ผลที่ส่งกลับต้องบอกว่า เกิดอะไรขึ้น ไม่ใช่แค่ สำเร็จหรือไม่ — เพราะหน้าเดิมต้องทำต่างกันตามเหตุการณ์ ไม่ใช่ตามความสำเร็จ

4. ทำไม bool ไม่พอ

final saved = await Navigator.push<bool>(...);

bool? เก็บได้สามค่า คือ true, false, null ซึ่งดูเหมือนพอสำหรับสามกรณี แต่พอเขียนจริงจะพบว่า

ค่าควรแปลว่าอะไรปัญหา
trueบันทึกแล้วแต่บันทึกอะไร? ต้องไปโหลดใหม่ทั้งรายการ
falseยกเลิก?ต้องจำเองว่า false แปลว่าอะไร ไม่มีอะไรบอก
nullกดย้อนกลับเผลอเขียน if (saved!) แล้วพังทันที

และเมื่อมีกรณีที่สี่คือ “พักไว้ก่อน” ก็หมดที่ให้เก็บ ต้องเปลี่ยนไปใช้ String หรือ Map ซึ่งแย่กว่าเดิม

// เคยเห็นในโปรเจกต์จริง
final result = await Navigator.push<Map<String, dynamic>>(...);
if (result?['status'] == 'saved') { ... }   // พิมพ์ผิดแล้วรู้ตอนรัน

5. sealed class คือคำตอบ

เปิด lib/nav_basics/nav_models.dart

sealed class TaskOutcome {
  const TaskOutcome();
}

/// บันทึกแล้ว ต้องอัปเดตรายการและแจ้งผู้ใช้
class TaskSaved extends TaskOutcome {
  const TaskSaved(this.task);
  final NavTask task;          // พาข้อมูลที่บันทึกกลับมาด้วย
}

/// พักไว้ก่อน เก็บร่างไว้แต่ยังไม่ถือว่าเสร็จ
class TaskPaused extends TaskOutcome {
  const TaskPaused(this.draft);
  final NavTask draft;
}

/// ทิ้งการแก้ไข กลับไปใช้ค่าเดิม
class TaskCancelled extends TaskOutcome {
  const TaskCancelled();       // ไม่มีข้อมูลอะไรต้องพากลับ
}

สังเกตว่าแต่ละกรณีพกสิ่งที่ต่างกัน ตามที่เรียนใน 4.15 — TaskSaved มีข้อมูลแน่นอน ส่วน TaskCancelled ไม่มีอะไรเลย

ประโยชน์ที่ได้ทันทีคือ หน้ารายการไม่ต้องโหลดข้อมูลใหม่ เพราะข้อมูลที่บันทึกแล้วเดินทางกลับมาพร้อมผลลัพธ์

6. รับผลด้วย switch ที่ครบทุกทาง

เปิด lib/nav_basics/nav_basics_app.dart

final outcome = await Navigator.of(context).push<TaskOutcome>(...);
if (!mounted) return;

switch (outcome) {
  case TaskSaved(task: final saved):
    setState(() {
      _tasks = [
        for (final item in _tasks) item.id == saved.id ? saved : item,
      ];
      _pausedIds.remove(saved.id);
    });
    showResultBanner(messenger, 'บันทึก ${saved.id} แล้ว');

  case TaskPaused(draft: final draft):
    setState(() {
      _tasks = [
        for (final item in _tasks) item.id == draft.id ? draft : item,
      ];
      _pausedIds.add(draft.id);
    });
    showResultBanner(messenger, 'พัก ${draft.id} ไว้ก่อน');

  case TaskCancelled():
    showResultBanner(messenger, 'ยกเลิกการแก้ไข ${task.id}');

  // ผู้ใช้กดปุ่มย้อนกลับโดยไม่ได้แก้อะไร — ไม่ต้องทำอะไรและไม่ต้องแจ้ง
  case null:
    break;
}

สามอย่างที่ควรสังเกต

สิ่งที่เห็นทำไมสำคัญ
ไม่มี default:เพิ่มผลลัพธ์ชนิดใหม่แล้ว compiler จะไล่ชี้ทุกจุดที่ต้องกลับมาตัดสินใจ
มี case null:ชนิดจริงคือ TaskOutcome? — ถ้าลืม compiler ก็เตือน
ดึงข้อมูลออกมาใช้เลยTaskSaved(task: final saved) ไม่ต้อง cast และไม่ต้องเช็ก null

7. null คือกรณีที่ต้องคิด ไม่ใช่กรณีที่เหลือ

นี่คือหัวใจของบทนี้

ผู้ใช้กดปุ่มย้อนกลับ → pop() ไม่มีค่า → outcome เป็น null

คนส่วนใหญ่จับ null รวมกับ “ยกเลิก” ซึ่งผิดในทางที่มองไม่เห็น

กดย้อนกลับ (null)ยืนยันว่าจะทิ้ง (TaskCancelled)
ผู้ใช้ตั้งใจอะไรแค่ออกจากหน้า ไม่ได้แก้อะไรอยู่แล้วแก้ไว้แล้ว แต่ตัดสินใจทิ้ง
ควรแจ้งอะไรไหมไม่ — ไม่มีอะไรเกิดขึ้นควร — เพื่อยืนยันว่างานที่แก้ไว้หายไปแล้วจริง

ถ้ารวมสองอย่างนี้เข้าด้วยกัน ผู้ใช้ที่แค่เปิดดูงานแล้วกดย้อนกลับ จะโดนข้อความ “ยกเลิกการแก้ไข” เด้งขึ้นมาทุกครั้ง ทั้งที่ไม่เคยแก้อะไร — ซึ่งกวนใจและทำให้เขาเริ่มไม่อ่านข้อความอื่นด้วย

ใน lab มีเทสต์ตรึงพฤติกรรมนี้ไว้

testWidgets('กดย้อนกลับโดยไม่ได้แก้อะไร ต้องไม่มีข้อความใด ๆ', (tester) async {
  await pumpLoggedIn(tester);
  await openFirstTask(tester);

  await tester.pageBack();          // จำลองการกดปุ่มย้อนกลับ
  await tester.pumpAndSettle();

  expect(find.byKey(const Key('result-banner')), findsNothing);
  expect(find.text('12 ชิ้น'), findsOneWidget);
});

8. อย่าใช้ผลลัพธ์เป็นทางส่งข้อมูลทางเดียว

ข้อจำกัดที่ต้องรู้ตั้งแต่ต้น

หน้าใหม่ ── pop(ผล) ──→ หน้าเดิม        ✓ ได้
หน้าใหม่ ── ระหว่างที่ยังเปิดอยู่ ──→ หน้าเดิม   ✗ ไม่ได้

ผลลัพธ์เดินทางกลับได้ครั้งเดียว คือตอนปิด ถ้าต้องการให้หน้าเดิมรู้เรื่องระหว่างที่หน้าใหม่ยังเปิดอยู่ ต้องใช้ store ร่วมกันตาม Part 4

สถานการณ์ใช้อะไร
แก้เสร็จแล้วกลับมาroute result
หน้าใหม่ปิดงานไปเรื่อย ๆ และหน้าเดิมต้องเห็นทันทีstore ที่ทั้งสองหน้าฟังอยู่
เปิดหน้าใหม่ซ้อนอีกสามชั้นแล้วต้องบอกหน้าล่างสุดstore — อย่าไล่ pop ส่งค่าต่อกันเป็นทอด ๆ
การส่งผลต่อกันเป็นทอด ๆ ผ่านหลาย pop ทำงานได้ในตอนแรก แต่พอมีเส้นทางเข้าออกมากกว่าหนึ่งทาง จะเริ่มมีกรณีที่ค่าหายกลางทางโดยไม่มีใครสังเกต

9. เมื่อหน้าเดิมถูกปิดไปแล้ว

กรณีที่คนลืมบ่อย — ผลลัพธ์กลับมาแต่ไม่มีใครรออยู่แล้ว

final outcome = await Navigator.of(context).push<TaskOutcome>(...);
if (!mounted) return;      // หน้ารายการถูกล้างไปแล้ว (เช่นออกจากระบบ)

ตรงนี้ return ได้อย่างปลอดภัย เพราะข้อมูลจริงไม่ได้อยู่ที่นี่ — แต่ถ้าหน้ารายการเป็นที่เดียวที่เก็บผล การ return เฉย ๆ คือการทิ้งงานของผู้ใช้ ซึ่งเป็นปัญหาเดียวกับ 4.9 แบบฝึกหัดข้อ 3

หลักคือ ถ้าผลลัพธ์นั้นสำคัญกว่าหน้าจอ ต้องให้มันไปจบที่ store ไม่ใช่ที่หน้าจอ

10. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
Null check operator used on a null valueเขียน result! โดยลืมว่ากดย้อนกลับได้ nullจัดการ null ให้ครบ
ขึ้นข้อความยกเลิกทั้งที่แค่เปิดดูรวม null เข้ากับ “ยกเลิก”แยกสองกรณีออกจากกัน
บันทึกแล้วรายการไม่อัปเดตไม่ได้ await หรือไม่ได้ใช้ผลawait แล้ว switch
ต้องโหลดรายการใหม่ทุกครั้งที่กลับมาผลลัพธ์เป็น bool ไม่พาข้อมูลกลับให้ผลลัพธ์พาข้อมูลที่บันทึกกลับมาด้วย
เพิ่มผลลัพธ์ชนิดใหม่แล้วบางจุดไม่รองรับมี default: ใน switchตัดออก
result?['status'] อ่านค่าไม่เจอใช้ Map เป็นผลลัพธ์ใช้ sealed class
ค่าหายเมื่อปิดหลายหน้าพร้อมกันส่งผลต่อกันเป็นทอด ๆใช้ store ร่วมกัน

11. แบบฝึกหัด

หน้าเลือกช่องเก็บสินค้าถูกเปิดจากหน้างาน putaway และผู้ใช้ทำได้สี่อย่าง

ผู้ใช้ทำหน้างานต้องทำต่อ
เลือกช่องเก็บหนึ่งช่องใช้ช่องนั้นและไปขั้นถัดไป
สแกนช่องที่เต็มแล้ว ระบบแจ้ง แล้วเขากดออกอยู่ขั้นเดิม + บอกว่าให้ลองช่องอื่น
กดปุ่ม “ยังไม่มีที่ว่าง แจ้งหัวหน้า”ส่งเรื่องให้หัวหน้า + ออกจากงานนี้
กดปุ่มย้อนกลับเฉย ๆไม่ทำอะไร
  1. ออกแบบชนิดของผลลัพธ์
  2. เขียน switch ฝั่งที่รับผล
  3. ตอบว่าทำไมกรณีที่สองไม่ควรใช้ null ทั้งที่ผู้ใช้ก็แค่ออกจากหน้าเหมือนกัน
ดูแนวคำตอบและเหตุผล

ข้อ 1 — ชนิดของผลลัพธ์

sealed class LocationPickOutcome {
  const LocationPickOutcome();
}

/// เลือกได้แล้ว
class LocationPicked extends LocationPickOutcome {
  const LocationPicked(this.locationCode);
  final String locationCode;
}

/// ลองแล้วแต่ช่องที่หมายตาไว้ใช้ไม่ได้ ยังอยากทำงานนี้ต่อ
class LocationUnavailable extends LocationPickOutcome {
  const LocationUnavailable(this.triedCode, this.reason);
  final String triedCode;
  final String reason;
}

/// ไม่มีที่ว่างจริง ๆ ส่งเรื่องต่อแล้ว
class LocationEscalated extends LocationPickOutcome {
  const LocationEscalated(this.ticketId);
  final String ticketId;
}

สังเกตว่า LocationUnavailable พา triedCode กลับมาด้วย เพื่อให้หน้างานบอกได้ว่า “ช่อง A-12 เต็ม ลองช่องอื่น” แทนที่จะบอกลอย ๆ ว่า “ไม่สำเร็จ”

ข้อ 2 — ฝั่งที่รับผล

final outcome = await Navigator.of(context).push<LocationPickOutcome>(
  MaterialPageRoute<LocationPickOutcome>(
    builder: (_) => LocationPickerScreen(palletCode: _palletCode),
  ),
);
if (!mounted) return;

switch (outcome) {
  case LocationPicked(:final locationCode):
    setState(() => _locationCode = locationCode);
    _goToConfirmStep();

  case LocationUnavailable(:final triedCode, :final reason):
    showResultBanner(messenger, 'ช่อง $triedCode ใช้ไม่ได้: $reason');
    // อยู่ขั้นเดิม ให้ลองใหม่ได้เลย

  case LocationEscalated(:final ticketId):
    showResultBanner(messenger, 'แจ้งหัวหน้าแล้ว เลขที่ $ticketId');
    _exitTask();

  case null:
    break;    // ไม่ทำอะไร
}

ข้อ 3 — ทำไมกรณีที่สองไม่ใช่ null

เพราะ null กับ LocationUnavailable ตอบคำถามคนละข้อ

nullLocationUnavailable
บอกว่าอะไร“ไม่มีอะไรเกิดขึ้น”“มีบางอย่างเกิดขึ้น และมันล้มเหลว”
ผู้ใช้ได้ลองอะไรไปหรือยังยังลองแล้ว
หน้างานมีอะไรจะบอกไหมไม่มีมี — และต้องบอก

ถ้าใช้ null ในกรณีที่สอง จะเกิดสิ่งนี้ — ผู้ใช้เดินไปที่ช่อง A-12 สแกนแล้วพบว่าเต็ม กดออกมา แล้ว หน้าจอไม่บอกอะไรเลย เขาจะไม่แน่ใจว่าระบบรับรู้หรือเปล่า และมีโอกาสสูงที่จะเดินกลับไปสแกนช่องเดิมอีกรอบ

วิธีตัดสินที่ใช้ได้ทุกครั้ง — ถามว่า "หน้าที่เรียกจะทำอะไรต่างไปหรือไม่"

คำตอบแปลว่า
ทำเหมือนกันทุกอย่างยุบรวมเป็นกรณีเดียวได้
ทำต่างกันแม้แต่นิดเดียวต้องเป็นคนละกรณี

เกณฑ์นี้ทำให้ไม่ออกแบบผลลัพธ์เยอะเกินจำเป็น และไม่ยุบสิ่งที่ต้องแยก

ข้อสังเกตสุดท้าย: null เป็นกรณีเดียวที่เรากำหนดไม่ได้ เพราะระบบเป็นคนส่งมาให้เมื่อผู้ใช้กดย้อนกลับ — จึงควรใช้มันแทน “ไม่มีอะไรเกิดขึ้น” เท่านั้น และอย่าเอาความหมายอื่นไปฝากไว้กับมัน

12. สรุปท้ายบท

  • ผลลัพธ์ต้องบอกว่าเกิดอะไรขึ้น ไม่ใช่แค่สำเร็จหรือไม่
  • bool และ Map ไม่พอ และพังตอนรัน
  • sealed class ให้แต่ละกรณีพกข้อมูลของตัวเองมาด้วย
  • ให้ผลลัพธ์พาข้อมูลที่บันทึกกลับมา จะได้ไม่ต้องโหลดใหม่
  • switch ที่ไม่มี default: ทำให้ compiler ไล่จุดที่ลืมให้
  • null มาจากการกดย้อนกลับ และควรแปลว่า “ไม่มีอะไรเกิดขึ้น” เท่านั้น
  • ตัดสินว่าต้องแยกกรณีไหม จากคำถามว่าหน้าที่เรียกจะทำต่างกันหรือเปล่า
  • ผลลัพธ์เดินทางกลับได้ครั้งเดียวตอนปิด — เรื่องที่ต้องรู้ระหว่างทางให้ใช้ store
