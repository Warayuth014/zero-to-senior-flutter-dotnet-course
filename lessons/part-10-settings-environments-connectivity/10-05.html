1. เก็บกี่กุญแจ

มีสองอย่างที่ต้องเก็บ คือรายการ profile และ id ของตัวที่ใช้งานอยู่ ทางเลือกแรกที่นึกออกคือแยกกัน

// อย่าทำแบบนี้
await storage.write('profiles', jsonEncode(profiles));
await storage.write('active_id', activeId);        // ← เขียนไม่สำเร็จตรงนี้ล่ะ?

ถ้าเขียนสำเร็จแค่บรรทัดแรก จะได้สถานะที่ไม่มีทางเกิดจากการใช้งานปกติ คือมีรายการใหม่แต่ตัวที่ใช้งานอยู่ยังชี้ของเก่าที่อาจถูกลบไปแล้ว

ทางที่ถูกคือเก็บรวมใต้กุญแจเดียว

String encodeProfiles(List<ServerProfile> profiles, String? activeId) =>
    jsonEncode({
      'version': profileSchemaVersion,
      'profiles': [
        for (final profile in profiles)
          profile.toJson(isActive: profile.id == activeId),
      ],
    });

สังเกตว่า activeId ไม่ได้เก็บเป็นฟิลด์แยก แต่กลายเป็นธง isActive บนแถว — ผลคือตัวที่ใช้งานอยู่ต้องเป็นแถวที่มีอยู่จริงเสมอ เพราะธงอยู่บนแถวนั้นเอง

2. อ่านกลับอย่างไรเมื่อข้อมูลเสีย

นี่คือจุดที่ Part นี้ตัดสินใจต่างจาก Part 6 โดยตั้งใจ และต้องเข้าใจว่าทำไม

ข้อมูลจากเซิร์ฟเวอร์ (Part 6)การตั้งค่าบนเครื่องนี้ (Part 10)
ใครเขียนทีมหลังบ้านแอปนี้เอง
ผิดรูปแบบแปลว่าสัญญาถูกละเมิด มีคนต้องแก้ไฟล์เสีย ผู้ใช้แก้ไม่ได้และไม่ต้องรู้
ทำอะไรเมื่อผิดโยน ContractExceptionทิ้งแถวนั้น ใช้ที่เหลือต่อ
ราคาของการโยนทีมเห็นปัญหาแล้วแก้ผู้ใช้เปิดแอปไม่ได้
/// คืน null สำหรับแถวที่เชื่อไม่ได้ แทนที่จะโยน
///
/// ต่างจากกฎใน Part 6 โดยตั้งใจ — ข้อมูลนี้เราเขียนเองลงเครื่องนี้
/// ไม่ใช่สัญญากับเซิร์ฟเวอร์ และแถวเสียหนึ่งแถวต้องไม่ทำให้ผู้ใช้
/// เสีย profile อื่นที่เหลือทั้งหมด
static ServerProfile? fromJson(Object? raw) { ... }
กฎที่ใช้ตัดสิน: โยนเมื่อมีคนแก้ได้ กลืนเมื่อไม่มีใครแก้ได้และการหยุดทำงานแพงกว่า — และเมื่อกลืน ต้องกลืนให้แคบที่สุด คือทิ้งเฉพาะแถวที่เสีย

3. โค้ดที่อ่านกลับ

ProfileSnapshot decodeProfiles(String? raw) {
  const empty = (profiles: <ServerProfile>[], activeId: null);
  if (raw == null || raw.trim().isEmpty) return empty;

  Object? decoded;
  try {
    decoded = jsonDecode(raw);
  } on FormatException {
    return empty;              // ไฟล์เสียทั้งไฟล์ ถือว่าไม่เคยมี
  }
  if (decoded is! Map) return empty;

  final rows = decoded['profiles'];
  if (rows is! List) return empty;

  final profiles = <ServerProfile>[];
  String? activeId;
  for (final row in rows) {
    final profile = ServerProfile.fromJson(row);
    if (profile == null) continue;                  // ทิ้งเฉพาะแถวที่เสีย
    profiles.add(profile);
    if (activeId == null && row is Map && row['isActive'] == true) {
      activeId = profile.id;
    }
  }

  if (profiles.isEmpty) return empty;
  return (profiles: profiles, activeId: activeId ?? profiles.first.id);
}

บรรทัดสุดท้ายคือหัวใจ — activeId ?? profiles.first.id ทำให้กฎ “มีตัวที่ใช้งานอยู่หนึ่งตัวเสมอ” เป็นจริงแม้ไฟล์จะถูกแก้ด้วยมือมาก่อน หรือแถวที่เคยเป็นตัวใช้งานถูกทิ้งไปเพราะเสีย

4. typedef ที่คืนสองค่า

typedef ProfileSnapshot = ({List<ServerProfile> profiles, String? activeId});

นี่คือ record ของ Dart 3 ที่ตั้งชื่อฟิลด์ได้ ใช้เมื่อฟังก์ชันต้องคืนหลายค่าที่ไปด้วยกัน

ทางเลือกปัญหา
คืน List แล้วให้คนเรียกหา active เองทุกคนต้องเขียนกฎหาเอง และเขียนไม่เหมือนกัน
สร้างคลาสใหม่ได้ไฟล์เพิ่มเพื่อคืนค่าจากฟังก์ชันเดียว
คืน recordเข้าถึงด้วย snapshot.activeId ชื่อชัด ไม่ต้องมีคลาส

5. เลขรุ่นตั้งแต่วันแรก

const int profileSchemaVersion = 1;

เหตุผลเดียวกับ 8.5 — วันที่ต้องเปลี่ยนโครงข้อมูล ถ้าไม่เคยเก็บเลขรุ่นไว้ จะแยกไม่ออกว่าไฟล์ที่อ่านมาเป็นรุ่นเก่าหรือรุ่นใหม่ที่เสีย ทางเดียวที่เหลือคือทิ้งของผู้ใช้ทั้งหมด

ตอนนี้ยังไม่มีโค้ดที่อ่านเลขนี้ และนั่นถูกแล้ว — เขียนลงไปตอนนี้ อ่านตอนที่ต้องใช้ ต้นทุนของการเขียนคือหนึ่งบรรทัด ต้นทุนของการไม่เขียนคือทางเลือกที่หายไปตลอดกาล

6. เทสต์ที่ต้องมี

test('แถวเสียหนึ่งแถว ต้องไม่ทำให้เสีย profile อื่นทั้งหมด', () {
  final raw = jsonEncode({
    'version': 1,
    'profiles': [
      profile('p1').toJson(isActive: true),
      {'id': 'broken'},                        // ไม่มี host และ port
      profile('p3').toJson(),
    ],
  });

  final snapshot = decodeProfiles(raw);
  expect(snapshot.profiles.map((p) => p.id), ['p1', 'p3']);
  expect(snapshot.activeId, 'p1');
});
test('ข้อมูลเสียทั้งไฟล์ ต้องได้ค่าว่าง ไม่ใช่โยน', () {
  for (final raw in [null, '', 'ไม่ใช่ JSON', '[]', '{}']) {
    final snapshot = decodeProfiles(raw);
    expect(snapshot.profiles, isEmpty, reason: 'ผิดที่ "$raw"');
  }
});

ห้าค่าในลูปนี้คือทุกรูปแบบของ “ไฟล์ที่ไม่ใช่สิ่งที่เราคาด” และแต่ละอันเข้าเส้นทางป้องกันคนละเส้นในฟังก์ชัน

7. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
เปิดแอปแล้วค้างที่หน้าขาว หลังจากอัปเดตแอปโครงข้อมูลเปลี่ยนแล้วอ่านของเก่าไม่ได้ แล้วโยนอ่านไม่ได้ให้ถือว่าไม่เคยมี
profile หายทั้งหมดเพราะแถวเดียวเสียใช้ map().toList() ที่โยนกลางทางวนทีละแถว ทิ้งเฉพาะที่เสีย
ไม่มีตัวที่ใช้งานอยู่เลยเก็บ activeId แยก แล้วชี้ไปที่แถวที่ถูกลบเก็บเป็นธงบนแถว พร้อมค่าสำรอง
มีตัวที่ใช้งานอยู่สองตัวเขียนธงมากกว่าหนึ่งแถวactiveId == null && ตอนอ่าน — อันแรกชนะ

8. แบบฝึกหัด

รุ่นถัดไปต้องเพิ่มฟิลด์ note ใน profile และเปลี่ยน port จาก int เป็น String เพื่อรองรับพอร์ตที่เป็นชื่อบริการ

  1. ต้องเพิ่มเลขรุ่นเป็น 2 ไหม
  2. ผู้ใช้ที่มีข้อมูลรุ่น 1 อยู่แล้ว จะเกิดอะไรขึ้น
แนวคำตอบ

1. แยกเป็นสองเรื่อง

  • เพิ่ม note ไม่ต้องขึ้นรุ่น เพราะเป็นการเพิ่มฟิลด์ที่มีค่าปริยาย ข้อมูลรุ่น 1 อ่านได้ปกติ จะได้ note เป็นค่าว่าง ซึ่งถูกต้อง
  • เปลี่ยนชนิดของ port ต้องขึ้นเป็นรุ่น 2 เพราะความหมายของฟิลด์เดิมเปลี่ยน

2. ตอนนี้ fromJson อ่านด้วย int.tryParse(raw['port']?.toString() ?? '') ซึ่งบังเอิญรับได้ทั้งเลขและข้อความที่เป็นเลข แต่จะทิ้งแถวที่พอร์ตเป็น "https" ไป — ผู้ใช้จะเสีย profile นั้นเงียบ ๆ

ทางที่ถูกคือเขียนตัวแปลงตามรุ่น

ProfileSnapshot decodeProfiles(String? raw) {
  ...
  final version = decoded['version'];
  final rows = decoded['profiles'];
  return switch (version) {
    1 => _decodeV1(rows),      // แปลง int เป็น String ให้
    2 => _decodeV2(rows),
    // รุ่นที่ใหม่กว่าที่แอปนี้รู้จัก แปลว่าผู้ใช้เพิ่ง downgrade
    // อ่านไม่ได้แน่นอน อย่าเดา
    _ => (profiles: [], activeId: null),
  };
}

ประเด็นที่ควรจำ: การมีเลขรุ่นไม่ได้แก้ปัญหาให้เอง มันแค่เปิดทางให้แก้ได้ — ถ้าไม่เคยเขียนเลขไว้ตั้งแต่วันแรก บรรทัด switch นี้เขียนไม่ได้เลย

9. สรุป

  • เก็บของที่ต้องตรงกันไว้ใต้กุญแจเดียว เขียนและอ่านพร้อมกัน
  • ตัวที่ใช้งานอยู่เก็บเป็นธงบนแถว ไม่ใช่ id แยก
  • ข้อมูลของเราเองบนเครื่องนี้ ให้ทิ้งแถวที่เสีย ไม่ใช่โยนทั้งหมด — ต่างจากสัญญากับเซิร์ฟเวอร์
  • เขียนเลขรุ่นตั้งแต่วันแรก อ่านตอนที่ต้องใช้
