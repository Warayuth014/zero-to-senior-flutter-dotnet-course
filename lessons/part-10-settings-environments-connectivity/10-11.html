1. ทำไมต้องมีคลาสแยก

ทางที่ง่ายที่สุดคือให้ฟอร์มแก้ ServerProfile ตรง ๆ แต่ทำไม่ได้ด้วยเหตุผลสองข้อ

ปัญหารายละเอียด
ค่าระหว่างพิมพ์ยังแปลงไม่ได้ผู้ใช้ลบพอร์ตจนหมดเพื่อพิมพ์ใหม่ ระหว่างนั้น port คือ '' ซึ่งเป็น int ไม่ได้
ยังไม่รู้ว่าจะบันทึกไหมถ้าแก้ของจริงตรง ๆ แล้วผู้ใช้กดยกเลิก ต้องหาทางย้อนกลับ

ทางแก้ที่คนมักเลือกคือทำให้ port เป็น int? ซึ่งย้ายปัญหาไปให้ทุกที่ที่ใช้ profile ต้องเช็ค null ทั้งที่ profile ที่บันทึกแล้วมีพอร์ตเสมอ

/// แยกจาก [ServerProfile] เพราะระหว่างพิมพ์ ค่ายังเป็นข้อความที่แปลงไม่ได้
/// ...ถ้าใช้โมเดลจริงเป็นตัวเก็บ จะต้องยอมให้ `port` เป็น `int?` ซึ่งทำให้
/// ทุกที่ที่ใช้ profile ต้องเช็ค null ทั้งที่ profile ที่บันทึกแล้วมีพอร์ตเสมอ
class ProfileDraft { ... }
หลักที่ใช้ได้ทั่วไป: เมื่อของชิ้นหนึ่งมีสองสถานะที่กฎไม่เหมือนกัน — “กำลังทำ” กับ “ทำเสร็จแล้ว” — ให้แยกเป็นสองชนิด อย่าทำชนิดเดียวให้หลวมพอที่จะเป็นได้ทั้งสองอย่าง

2. หน้าตาของร่าง

class ProfileDraft {
  final String? id;         // null = สร้างใหม่ ไม่ใช่แก้ของเดิม
  String name;              // ไม่ใช่ final เพราะฟอร์มแก้ได้
  String host;
  String port;              // เป็น String ไม่ใช่ int
  String basePath;

  bool get isNew => id == null;

  factory ProfileDraft.from(ServerProfile profile) => ProfileDraft(...);
}
ต่างจาก ServerProfileเพราะ
ฟิลด์แก้ได้ ไม่ใช่ finalฟอร์มแก้ทีละช่อง การใช้ copyWith ทุกตัวอักษรจะเปลืองโดยไม่ได้อะไร
port เป็นข้อความรองรับค่าระหว่างพิมพ์
id เป็น nullableแยกสร้างใหม่กับแก้ของเดิม
ไม่มี baseUrlร่างที่ยังไม่ผ่านการตรวจ ประกอบ URL ไม่ได้

3. แปลงกลับตอนบันทึก

ServerProfile toProfile({required String id}) {
  final parsedPort = int.tryParse(port.trim());
  if (parsedPort == null) {
    throw StateError('toProfile ถูกเรียกทั้งที่ยังไม่ผ่านการตรวจ: port="$port"');
  }
  return ServerProfile(...);
}

ทำไมโยน ทั้งที่ Part นี้บอกตลอดว่าอย่าให้แอปพัง คำตอบอยู่ที่ใครผิด

สถานการณ์ใครผิดควรทำ
ผู้ใช้พิมพ์พอร์ตผิดไม่มีใครผิดแสดงข้อความใต้ช่อง
ไฟล์ที่เก็บไว้เสียไม่มีใครแก้ได้ทิ้งแถวนั้น
โค้ดเรียก toProfile โดยไม่ตรวจก่อนเราโยนให้ดัง

ถ้าคืนค่าเงียบ ๆ แทน เช่นใช้พอร์ต 80 เป็นค่าปริยาย ผู้ใช้จะได้ profile ที่มีพอร์ตที่ตัวเองไม่เคยพิมพ์ แล้วสงสัยว่าทำไมต่อไม่ติด — บั๊กของเรากลายเป็นปัญหาของผู้ใช้ ซึ่งเป็นสิ่งที่ต้องหลีกเลี่ยงที่สุด

4. รู้ได้อย่างไรว่าแก้ไปแล้ว

bool isDirtyFrom(ServerProfile? saved) {
  if (saved == null) {
    // ร่างใหม่ที่ยังไม่พิมพ์อะไรเลย ไม่ถือว่าแก้ — ปิดได้โดยไม่ต้องเตือน
    return name.trim().isNotEmpty ||
        host.trim().isNotEmpty ||
        port.trim().isNotEmpty ||
        basePath.trim().isNotEmpty;
  }
  return name.trim() != saved.name ||
      environment != saved.environment ||
      protocol != saved.protocol ||
      host.trim() != saved.host ||
      port.trim() != saved.port.toString() ||
      basePath.trim() != saved.basePath;
}

รายละเอียดสามข้อที่มีผลจริง

รายละเอียดถ้าไม่ทำจะเกิด
ร่างเปล่าถือว่ายังไม่แก้ผู้ใช้กดเปิดหน้าสร้างใหม่ แล้วกดปิดทันที ก็โดนเตือน
เทียบด้วย trim()ช่องว่างท้ายบรรทัดที่ผู้ใช้ไม่เห็นทำให้เตือนทั้งที่ไม่ได้แก้
เทียบทีละช่อง ไม่ใช่เทียบ baseUrlเปลี่ยนชื่ออย่างเดียวจะไม่นับว่าแก้ แล้วผู้ใช้เสียงานเงียบ ๆ

เทสต์บังคับทั้งสามข้อ

test('เปลี่ยนแค่ชื่อก็นับว่าแก้ ทั้งที่ที่อยู่เท่าเดิม', () {
  final saved = profile('p1');
  final edited = ProfileDraft.from(saved)..name = 'Office ชั้น 2';
  expect(edited.isDirtyFrom(saved), isTrue);
});

test('เว้นวรรคหัวท้ายไม่นับว่าแก้', () {
  final saved = profile('p1');
  final edited = ProfileDraft.from(saved)..host = ' 192.168.1.194 ';
  expect(edited.isDirtyFrom(saved), isFalse);
});

5. ต่อเข้ากับปุ่มย้อนกลับ

PopScope(
  canPop: !draft.isDirtyFrom(saved),
  onPopInvokedWithResult: (didPop, _) async {
    if (didPop) return;                     // ออกไปแล้ว ไม่ต้องทำอะไร
    final leave = await showDialog<bool>(
      context: context,
      builder: (context) => AlertDialog(
        title: const Text('ยังไม่ได้บันทึก'),
        content: const Text('ออกตอนนี้ค่าที่แก้จะหาย'),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(context, false),
            child: const Text('กลับไปแก้ต่อ'),
          ),
          FilledButton(
            onPressed: () => Navigator.pop(context, true),
            child: const Text('ออกโดยไม่บันทึก'),
          ),
        ],
      ),
    );
    if (leave == true && context.mounted) Navigator.pop(context);
  },
  child: ...,
)

สามจุดที่พลาดกันบ่อย ตามที่ 5.9 อธิบายไว้

จุดถ้าลืม
if (didPop) return;ถ้า canPop เป็นจริง หน้าปิดไปแล้ว แล้วยังเปิด dialog ซ้อนบนหน้าที่ไม่มีอยู่
context.mountedระหว่างรอ dialog ผู้ใช้อาจออกไปทางอื่นแล้ว
canPop คำนวณสดถ้าคำนวณครั้งเดียวตอนสร้าง จะไม่อัปเดตตามที่พิมพ์

6. ข้อความในกล่องเตือนสำคัญกว่าที่คิด

ข้อความที่ไม่ดีปัญหา
“คุณแน่ใจหรือไม่?” + ปุ่ม ตกลง/ยกเลิก“ตกลง” หมายถึงตกลงออก หรือตกลงว่าจะอยู่ต่อ
“มีการเปลี่ยนแปลงที่ยังไม่ได้บันทึก”บอกสถานะ แต่ไม่บอกว่าถ้ากดปุ่มไหนจะเกิดอะไร

ปุ่มควรเขียนเป็นการกระทำ ไม่ใช่ ใช่/ไม่ใช่ — “ออกโดยไม่บันทึก” กับ “กลับไปแก้ต่อ” อ่านแล้วรู้ทันทีโดยไม่ต้องอ่านข้อความข้างบน

และถ้าทำได้ ให้เพิ่มปุ่มที่สามคือ “บันทึกแล้วออก” เพราะนั่นคือสิ่งที่ผู้ใช้ส่วนใหญ่ตั้งใจตั้งแต่แรก

7. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
เตือนทั้งที่เพิ่งเปิดหน้ามาร่างเปล่านับว่าแก้แล้วเทียบกับ null แยกกรณี
เตือนหลังบันทึกสำเร็จแล้วไม่ได้อัปเดตตัวที่ใช้เทียบหลังบันทึกตั้ง saved เป็นตัวใหม่ที่เพิ่งบันทึก
ไม่เตือนเลยcanPop คำนวณครั้งเดียวคำนวณใน build
dialog ซ้อนกันสองอันลืม if (didPop) return;เช็คก่อนเสมอ
แก้แล้วกดบันทึก แต่ค่าเก่ายังอยู่ฟอร์มเขียนลง controller แต่ไม่ได้เขียนกลับลงร่างอ่านจาก controller ตอนบันทึก หรืออัปเดตร่างทุกครั้งที่เปลี่ยน

8. แบบฝึกหัด

ผู้ใช้แก้ profile ที่กำลังใช้งานอยู่ เปลี่ยน IP แล้วกดบันทึก

  1. ควรตรวจการเชื่อมต่อให้อัตโนมัติไหม
  2. ถ้าที่อยู่ใหม่ต่อไม่ติด ควรทำอย่างไร
แนวคำตอบ

1. ควร เพราะผู้ใช้เพิ่งเปลี่ยนที่อยู่ของเครื่องที่กำลังใช้อยู่ คำตอบทั้งใหม่และกำลังจะถูกมอง ตรงกับเกณฑ์ใน 10.8 พอดี

2. ตัวเลือกที่ต้องชั่งน้ำหนัก

ทางเลือกปัญหา
ย้อนกลับให้อัตโนมัติผู้ใช้อาจรู้อยู่แล้วว่าเซิร์ฟเวอร์ยังไม่เปิด และตั้งใจตั้งไว้ล่วงหน้า
ห้ามบันทึกจนกว่าจะต่อติดตั้งค่าล่วงหน้าไม่ได้เลย และถ้าเน็ตหลุดตอนนั้นก็ทำอะไรไม่ได้
บันทึกแล้วแสดงว่าต่อไม่ติดผู้ใช้อาจไม่ทันสังเกต

ทางที่ควรเลือกคือข้อสาม แต่ทำให้สังเกตยาก — บันทึกไว้ตามที่ผู้ใช้สั่ง แล้วแสดงผลการตรวจแบบที่ต้องรับรู้ พร้อมปุ่ม “ย้อนกลับไปที่อยู่เดิม” ที่กดได้ในคลิกเดียว

เหตุผลคือ ผู้ใช้สั่งอะไร ต้องเกิดสิ่งนั้น — แอปที่ตัดสินใจแทนโดยการย้อนค่าให้ จะทำให้ผู้ใช้ที่รู้ว่าตัวเองทำอะไรอยู่ทำงานไม่ได้ และเขาจะไม่มีทางรู้ด้วยว่าทำไม

อีกอย่างที่ควรทำคืออย่าลบที่อยู่เดิมทิ้ง ถ้าผู้ใช้แก้ profile ที่ใช้งานอยู่ ให้เสนอทางเลือก “บันทึกเป็นตัวใหม่” ด้วย ผู้ใช้จะได้เก็บที่อยู่เดิมไว้กลับไปใช้ได้ — ซึ่งเป็นเหตุผลที่ saveAsNew กับ saveEdits แยกกันตั้งแต่ 10.6

9. สรุป

  • แยก “กำลังพิมพ์” กับ “บันทึกแล้ว” เป็นสองชนิด อย่าทำชนิดเดียวให้หลวม
  • ร่างเก็บค่าเป็นข้อความ เพราะระหว่างพิมพ์ยังแปลงไม่ได้
  • เรียกแปลงโดยไม่ตรวจก่อนคือบั๊กของเรา ต้องโยนให้ดัง
  • เทียบทีละช่องด้วย trim() ไม่ใช่เทียบผลลัพธ์รวม
  • ปุ่มในกล่องเตือนต้องเขียนเป็นการกระทำ ไม่ใช่ ใช่/ไม่ใช่
