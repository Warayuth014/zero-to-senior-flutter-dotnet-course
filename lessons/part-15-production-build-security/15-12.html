3. checklist ที่รันได้

ReleaseReport checkRelease({
  required String androidManifest,
  required String androidGradle,
  required String pubspec,
}) => ReleaseReport([
  ...checkAndroidManifest(androidManifest),
  ...checkAndroidGradle(androidGradle),
  ...checkPubspec(pubspec),
]);
test('โปรเจกต์ที่เพิ่งสร้างจากแม่แบบ ปล่อยไม่ได้', () {
  final report = checkRelease(...);

  expect(report.canRelease, isFalse);
  // สามข้อนี้คือสิ่งที่ทำให้ APK ที่ build ออกมาใช้งานจริงไม่ได้
  expect(report.has('missing-internet-permission'), isTrue);
  expect(report.has('cleartext-blocked'), isTrue);
  expect(report.has('default-application-id'), isTrue);
  expect(report.has('debug-signing'), isTrue);
});

สี่ข้อนี้คือสถานะของโปรเจกต์ Flutter ที่ยังไม่ได้ตั้งค่าอะไรเลย และเป็นสิ่งที่ 15.3 พบใน wmsapp จริง

6. คำใบ้สำหรับข้อที่ยาก

เกณฑ์ 1 — อ่านไฟล์จริง

test('โปรเจกต์นี้พร้อมปล่อย', () {
  final report = checkRelease(
    androidManifest: File('android/app/src/main/AndroidManifest.xml')
        .readAsStringSync(),
    androidGradle: File('android/app/build.gradle.kts').readAsStringSync(),
    pubspec: File('pubspec.yaml').readAsStringSync(),
  );

  expect(report.blockers, isEmpty, reason: report.toString());
});

reason: report.toString() ทำให้ข้อความตอนล้มมีวิธีแก้ครบทุกข้อ

เกณฑ์ 5 — เทียบกับรุ่นก่อน

test('เลข build มากกว่ารุ่นที่ปล่อยล่าสุด', () {
  final current = AppVersion.parse(readVersionLine())!;
  final lastTag = Process.runSync('git', ['tag', '--list', 'v*', '--sort=-v:refname'])
      .stdout.toString().split('\n').first.trim();
  if (lastTag.isEmpty) return;   // ยังไม่เคยปล่อย

  final released = AppVersion.parse(lastTag.substring(1))!;
  expect(current.canUpgradeFrom(released), isTrue,
      reason: 'รุ่นล่าสุดคือ $released');
});

เกณฑ์ 8 และ 9 — ตรวจ APK หลัง build

# tool/release.sh
APK="build/app/outputs/flutter-apk/app-release.apk"

if apksigner verify --print-certs "$APK" | grep -q "CN=Android Debug"; then
  echo "APK ถูกเซ็นด้วยกุญแจ debug"
  exit 1
fi

if ! aapt dump permissions "$APK" | grep -q "android.permission.INTERNET"; then
  echo "APK ไม่มีสิทธิ์ INTERNET"
  exit 1
fi

9. สรุปท้าย Part 15

สิ่งที่ทำได้แล้ว

  • เลือกโหมด build ให้ตรงกับสิ่งที่ทำ และรู้ว่าอะไรหายไปในโหมด release
  • ตั้งเลขรุ่นให้ติดตั้งทับได้และตามหาได้ว่าเครื่องไหนใช้รุ่นอะไร
  • ประกาศสิทธิ์ให้ครบสำหรับ build ที่ปล่อยจริง
  • อนุญาต http เฉพาะเซิร์ฟเวอร์ที่จำเป็น
  • เซ็นแอปด้วยกุญแจของตัวเองและเก็บให้ปลอดภัย
  • รู้ว่าอะไรเก็บในแอปได้และอะไรต้องอยู่ที่เซิร์ฟเวอร์
  • เลือกรูปแบบไฟล์และลดขนาดแอป
  • ตั้งชื่อแพ็กเกจ ชื่อแอป และไอคอนก่อนปล่อยรุ่นแรก
  • รู้ว่าต้องเตรียมอะไรถ้าต้องรองรับ iOS
  • เก็บรายงานข้อผิดพลาดจากหน้างานโดยไม่ทำให้ความลับหลุด
  • ตรวจว่าแอปรับมือกับเครือข่ายคลังได้ครบทุกด้าน
  • เปลี่ยน checklist ให้เป็นเทสต์ที่รันได้
