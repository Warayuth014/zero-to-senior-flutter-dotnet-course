3. Mental model: ปิดปุ่มไม่ใช่การป้องกัน

สิ่งที่คนใหม่มักคิดคือปิดปุ่มแล้วจบ ซึ่งไม่จริงด้วยเหตุผลสองข้อ

  1. ปุ่มไม่ใช่ทางเดียว — ยังมีการกด Enter เครื่องสแกน และปุ่มลัด
  2. ปุ่มปิดช้ากว่าที่คิด — การปิดปุ่มเกิดหลังวาดหน้าจอใหม่ ซึ่งช้ากว่าการแตะสองครั้งติดกัน
ผู้ใช้แตะครั้งที่ 1  →  _submit() เริ่มทำงาน
ผู้ใช้แตะครั้งที่ 2  →  _submit() เริ่มทำงานอีกครั้ง   ← ปุ่มยังไม่ทันปิด
                    →  setState ทำให้ปุ่มปิด          ← สายไปแล้ว
ประโยคเดียวที่ต้องจำ: การป้องกันจริงต้องอยู่ที่ บรรทัดแรกของฟังก์ชัน และต้องตั้งธง ก่อน await ตัวแรก — ส่วนการปิดปุ่มเป็นการสื่อสารกับผู้ใช้ ไม่ใช่การป้องกัน

4. รูปแบบที่ถูกต้อง

Future<void> _submit() async {
  if (_submitting || _session.isEmpty) return;      // 1. ตรวจก่อน
  if (!(_formKey.currentState?.validate() ?? false)) return;

  final confirmed = await _confirmSubmit();
  if (!mounted || confirmed != true) return;

  setState(() => _submitting = true);               // 2. ตั้งธงก่อน await
  _session.beginSubmit();
  try {
    final send = widget.onSubmit;
    final ok = send == null ? true : await send(_session.accepted);
    if (!mounted) return;
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(content: Text(ok ? 'บันทึกสำเร็จ' : 'บันทึกไม่สำเร็จ')),
    );
  } finally {
    _session.endSubmit();                           // 3. ปลดใน finally เสมอ
    if (mounted) setState(() => _submitting = false);
  }
}

สามขั้นที่ต้องครบ ตรงกับ D1.15 หัวข้อ 7 ทุกประการ

ขั้นถ้าขาดจะเกิดอะไร
ตรวจก่อนส่งซ้อนได้
ตั้งธงก่อน awaitการเรียกครั้งที่สองผ่านการตรวจไปได้
ปลดใน finallyปุ่มค้างตลอดไปเมื่อเกิดข้อผิดพลาด

5. ทำไมต้องตั้งธงก่อน await

นี่คือจุดที่ละเอียดที่สุดของบทนี้

// ผิด — ตั้งธงหลัง await
Future<void> _submit() async {
  if (_submitting) return;

  final confirmed = await _confirmSubmit();   // ← ระหว่างนี้ _submitting ยังเป็น false
  setState(() => _submitting = true);
  ...
}

ทบทวนจาก D1.15: โค้ดก่อน await ทำงานทันที ส่วนโค้ดหลัง await ทำงานทีหลัง — ระหว่างที่รอกล่องยืนยัน ถ้ามีการเรียก _submit() อีกครั้ง มันจะผ่าน if (_submitting) return; ไปได้เพราะธงยังไม่ถูกตั้ง

ใน lab เราแก้ด้วยการมีธงสองระดับ

// ระดับหน้าจอ — คุมปุ่มและช่องกรอก
bool _submitting = false;

// ระดับตรรกะ — คุมการรับบาร์โค้ด
_session.beginSubmit();

โดย _session.beginSubmit() ทำให้ accept() คืน ScanOutcome.busy ทันที ซึ่งเป็นการป้องกันที่ไม่ขึ้นกับหน้าจอเลย

6. เทสต์ที่พิสูจน์ได้จริง

การกันส่งซ้อนทดสอบยาก เพราะต้องจำลอง “ระหว่างที่ยังไม่เสร็จ” ให้ได้ — lab ใช้ Completer ซึ่งเป็นสวิตช์ที่เราควบคุมเองว่าจะให้งานเสร็จเมื่อไหร่

testWidgets('กดบันทึกรัวไม่ทำให้ส่งซ้ำ', (tester) async {
  var sent = 0;
  final gate = Completer<bool>();

  await tester.pumpWidget(
    MaterialApp(
      home: ReceiveScreen(
        onSubmit: (codes) {
          sent++;
          return gate.future;          // ค้างไว้จนกว่าเราจะปล่อย
        },
      ),
    ),
  );

  await scan(tester, 'PAL-1001');
  await tester.tap(find.byKey(const Key('submit-button')));
  await tester.pumpAndSettle();
  await tester.tap(find.byKey(const Key('confirm-ok')));
  await tester.pump();

  // ระหว่างที่ยังบันทึกไม่เสร็จ กดซ้ำอีกสองครั้ง
  await tester.tap(find.byKey(const Key('submit-button')));
  await tester.pump();
  await tester.tap(find.byKey(const Key('submit-button')));
  await tester.pump();

  gate.complete(true);
  await tester.pumpAndSettle();

  expect(sent, 1);
});

Completer คือ Future ที่เราสั่งให้เสร็จเองได้ ทำให้ควบคุมเวลาในเทสต์ได้เต็มที่ — ต่างจากการใช้ Future.delayed ซึ่งขึ้นกับจังหวะและทำให้เทสต์ไม่แน่นอน

อีกเทสต์หนึ่งพิสูจน์ว่าการยิงบาร์โค้ดถูกกันด้วย

testWidgets('ระหว่างบันทึก ยิงบาร์โค้ดเพิ่มไม่ได้', (tester) async {
  // ... กดยืนยันแล้วค้างไว้ ...
  expect(countText(tester), 'รับแล้ว 1 รายการ');
});

7. ปิดทุกทางที่เข้าถึงได้

ทางปิดอย่างไรระดับ
ปุ่มบันทึกonPressed: _submitting ? null : _submitสื่อสารกับผู้ใช้
ช่องกรอกenabled: !_submittingสื่อสารกับผู้ใช้
ปุ่มลบรายการonPressed: _submitting ? null : ...สื่อสารกับผู้ใช้
ตัวฟังก์ชันเองif (_submitting) return;ป้องกันจริง
ตรรกะการรับสแกนif (_busy) return ScanOutcome.busy;ป้องกันจริง

สามแถวแรกทำให้ผู้ใช้เห็นว่าระบบกำลังทำงานอยู่ ซึ่งสำคัญต่อประสบการณ์ใช้งาน — แต่ถ้ามีแค่สามแถวแรกโดยไม่มีสองแถวหลัง ระบบจะยังส่งซ้อนได้

8. ยังไม่ครบถ้าไม่มีที่ server

ทบทวนจาก 3.7 หัวข้อ 7 — ธงในแอปกันได้เฉพาะในเครื่องเดียวและในการทำงานครั้งเดียว

สถานการณ์ธงในแอปกันได้ไหม
กดปุ่มรัวได้
เน็ตหลุดตอนส่ง แล้วผู้ใช้กดใหม่ไม่ได้ — ครั้งแรกอาจถึง server แล้ว
แอปค้างแล้วเปิดใหม่ ส่งซ้ำไม่ได้
สองคนส่งใบเดียวกันจากคนละเครื่องไม่ได้

แถวที่สองสำคัญเป็นพิเศษและเชื่อมกับ D1.15 หัวข้อ 6 — การหมดเวลาแปลว่าเราเลิกรอ ไม่ได้แปลว่า server ไม่ได้ทำงาน คำสั่งนั้นอาจสำเร็จไปแล้ว การให้ผู้ใช้กดใหม่จึงเสี่ยงสร้างใบงานซ้ำ

ทางแก้ที่ถูกคือให้แต่ละคำสั่งมีรหัสประจำตัว ส่งไปกับข้อมูล ถ้า server เจอรหัสเดิมก็คืนผลเดิมโดยไม่บันทึกใหม่ — จะทำจริงใน Part 7

9. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
กดรัวแล้วเกิดใบงานซ้ำปิดแค่ปุ่ม ไม่มี guard ในฟังก์ชันเพิ่ม if (_submitting) return; บรรทัดแรก
ยังส่งซ้อนได้ทั้งที่มี guardตั้งธงหลัง awaitตั้งก่อน await ตัวแรก
ปุ่มค้างกดไม่ได้หลังเกิดข้อผิดพลาดปลดธงไว้ท้าย tryย้ายไป finally
setState() called after dispose()เรียกหลังผู้ใช้ออกจากหน้าจอตรวจ mounted ก่อนเรียก แต่ปลดธงในตรรกะโดยไม่ตรวจ
ยิงบาร์โค้ดเข้ามาระหว่างบันทึกกันแค่ที่ปุ่มกันที่ตรรกะการรับด้วย
เน็ตหลุดแล้วกดใหม่ ได้ใบงานสองใบไม่มีกฎที่ serverใช้รหัสประจำคำสั่ง
เทสต์การกันซ้อนไม่แน่นอน ผ่านบ้างไม่ผ่านบ้างใช้ Future.delayed จำลองใช้ Completer ควบคุมเวลาเอง

10. แบบฝึกหัด

โค้ดนี้มีปัญหาสามข้อ จงหาให้ครบและแก้

bool _saving = false;

Future<void> _save() async {
  final confirmed = await _showConfirmDialog();
  if (confirmed != true) return;

  if (_saving) return;
  setState(() => _saving = true);

  final result = await api.save(_items);
  setState(() => _message = result.message);
  setState(() => _saving = false);
}
ดูแนวคำตอบและเหตุผล

ปัญหาที่ 1 — ตรวจธงหลัง await

บรรทัด if (_saving) return; อยู่หลังการรอกล่องยืนยัน ระหว่างที่กล่องเปิดอยู่ ธงยังเป็น false ดังนั้นถ้ามีการเรียก _save() อีกครั้ง กล่องยืนยันจะเปิดซ้อนขึ้นมาอีกใบ และเมื่อผู้ใช้กดยืนยันทั้งสองใบ ก็จะส่งสองครั้ง

ปัญหาที่ 2 — ไม่มี finally

ถ้า api.save() โยนข้อผิดพลาด บรรทัดปลดธงจะไม่ถูกเรียก ธงค้างเป็น true ตลอดไป ปุ่มจะกดไม่ได้อีกเลยจนกว่าจะปิดแอป — และผู้ใช้จะไม่เห็นข้อความอะไรด้วย เพราะข้อผิดพลาดหลุดออกไปโดยไม่มีใครดัก

ปัญหาที่ 3 — ไม่ตรวจ mounted

ถ้าผู้ใช้กดย้อนกลับระหว่างที่กำลังบันทึก การเรียก setState หลังจากนั้นจะเกิด setState() called after dispose() ตามที่เตือนไว้ใน 1.6

Future<void> _save() async {
  // 1. ตรวจธงเป็นสิ่งแรกสุด ก่อน await ใด ๆ
  if (_saving) return;

  final confirmed = await _showConfirmDialog();
  if (!mounted || confirmed != true) return;

  // 2. ตั้งธงก่อน await ที่ทำงานจริง
  setState(() => _saving = true);
  try {
    final result = await api.save(_items);
    if (!mounted) return;
    setState(() => _message = result.message);
  } on Exception catch (error) {
    if (!mounted) return;
    setState(() => _message = 'บันทึกไม่สำเร็จ กรุณาลองใหม่');
    debugPrint('save failed: $error');       // รายละเอียดเข้า log (D1.14)
  } finally {
    // 3. ปลดธงเสมอ ไม่ว่าจะสำเร็จ ล้มเหลว หรือออกจากหน้าจอ
    if (mounted) setState(() => _saving = false);
  }
}

จุดที่ต้องคิดเพิ่ม — ธงที่ปลดไม่ได้เมื่อออกจากหน้าจอ

สังเกตว่า finally ยังมี if (mounted) ครอบอยู่ ซึ่งถูกต้องสำหรับ setState แต่ถ้าธงนั้นเป็นของสิ่งที่อยู่ยาวกว่าหน้าจอ เช่น ScanSession ใน lab จะต้องปลดโดยไม่ตรวจ mounted

} finally {
  _session.endSubmit();                      // ปลดเสมอ ไม่ขึ้นกับหน้าจอ
  if (mounted) setState(() => _submitting = false);   // อัปเดตจอเมื่อยังอยู่
}

ความต่างคือ _session เป็นตรรกะที่อาจถูกใช้ต่อ ส่วน setState เป็นการแตะหน้าจอซึ่งทำได้เฉพาะเมื่อหน้าจอยังอยู่ — เป็นเหตุผลว่าทำไม lab ถึงแยกสองบรรทัดนี้ออกจากกัน

ลองต่อ: เขียนเทสต์ที่กดยืนยันแล้วถอด widget ออกก่อนงานเสร็จ แล้วตรวจว่าไม่มี exception — ดูรูปแบบได้จากเทสต์ “ออกจากหน้าจอระหว่างบันทึกแล้วไม่พัง” ใน lab

11. สรุปท้ายบท

  • กันยิงซ้ำกับกันส่งซ้อนเป็นคนละเรื่อง ต้องทำทั้งคู่
  • ปิดปุ่มคือการสื่อสาร ไม่ใช่การป้องกัน
  • ตรวจธงเป็นบรรทัดแรก และตั้งธงก่อน await ตัวแรก
  • ปลดธงใน finally เสมอ
  • ปิดทุกทางที่เข้าถึงได้ รวมถึงการรับบาร์โค้ด
  • ใช้ Completer ในเทสต์เพื่อควบคุมเวลาเอง
  • ธงในแอปกันไม่ได้ทุกกรณี — ต้องมีกฎที่ server ด้วย
