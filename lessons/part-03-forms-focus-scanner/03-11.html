1. ปัญหาที่บทนี้แก้

ความจริงของงานสแกนในคลัง — พนักงานไม่ได้มองหน้าจอ เขามองของที่อยู่ในมือ มองฉลากบนกล่อง มองทางเดิน หน้าจอเป็นสิ่งที่เขาเงยขึ้นมาดูเป็นครั้งคราว

ถ้าเราบอกผลด้วยข้อความบนจออย่างเดียว พนักงานจะยิงต่อไปเรื่อย ๆ โดยไม่รู้ว่ามีอันที่ไม่เข้า แล้วมารู้ตอนจบว่ายอดไม่ตรง

2. คำศัพท์ที่ต้องรู้ก่อน

3. Mental model: สามช่องทางที่ไม่ต้องมองจอ

ช่องทางรับรู้ได้เมื่อข้อจำกัด
การสั่นถือเครื่องอยู่ในมือบางเครื่องไม่รองรับ
เสียงคลังไม่เสียงดังเกินไปอาจปิดเสียงไว้
สีที่เห็นด้วยหางตาเครื่องอยู่ในระยะสายตาผู้ใช้ตาบอดสีแยกไม่ออก
ประโยคเดียวที่ต้องจำ: ทุกช่องทางมีข้อจำกัด จึงต้องมีหลายช่องทางพร้อมกัน — และต้องมีข้อความบนจอเป็นตัวยืนยันสุดท้ายเสมอ

4. ดูของจริงใน lab

import 'package:flutter/services.dart';

/// สัญญาณตอบรับหลายช่องทาง — พนักงานกำลังมองของในมือ ไม่ได้มองจอ
void _giveFeedback(ScanOutcome outcome) {
  if (outcome.isFailure) {
    HapticFeedback.heavyImpact();      // สั่นแรง = มีปัญหา
  } else {
    HapticFeedback.lightImpact();      // สั่นเบา = ผ่าน
  }
}

ความต่างของแรงสั่นสำคัญกว่าที่คิด — พนักงานเรียนรู้ได้ภายในไม่กี่นาทีว่าสั่นเบาคือผ่าน สั่นแรงคือต้องเงยหน้ามาดู หลังจากนั้นเขาจะยิงได้เร็วโดยแทบไม่ต้องมองจอเลย

ส่วนบนจอ lab แสดงแถบที่มีครบสามอย่างตาม 2.12

class ScanFeedbackBanner extends StatelessWidget {
  @override
  Widget build(BuildContext context) {
    final scheme = Theme.of(context).colorScheme;
    final failed = outcome.isFailure;

    return Container(
      decoration: BoxDecoration(
        color: failed ? scheme.errorContainer : scheme.tertiaryContainer,
      ),
      child: Row(
        children: [
          Icon(failed ? Icons.error_outline : Icons.check_circle_outline),
          const SizedBox(width: 8),
          Expanded(child: Text(outcome.message)),
        ],
      ),
    );
  }
}

5. ข้อความมาจาก enum ที่เดียว

enum ScanOutcome {
  accepted('รับเข้าแล้ว'),
  duplicate('สแกนซ้ำ'),
  empty('ไม่มีข้อมูล'),
  badFormat('รูปแบบบาร์โค้ดไม่ถูกต้อง'),
  busy('กำลังบันทึกรายการก่อนหน้า');

  const ScanOutcome(this.message);

  final String message;

  bool get isFailure => this != ScanOutcome.accepted;
}

ประโยชน์ตรงกับที่คุยใน 2.12 — เมื่อเพิ่มผลลัพธ์แบบใหม่ในอนาคต เราต้องกำหนดข้อความให้มันตั้งแต่บรรทัดที่ประกาศ ไม่มีทางลืม

และมี shouldClearInput ผูกไว้กับผลลัพธ์ด้วย

/// สำเร็จเท่านั้นที่ควรล้างช่องกรอก — กรณีอื่นเก็บค่าไว้ให้ผู้ใช้ตรวจ
bool get shouldClearInput => this == ScanOutcome.accepted;

นี่คือส่วนหนึ่งของการตอบรับที่คนมักมองข้าม — การเก็บค่าที่ยิงผิดไว้ในช่องคือการบอกผู้ใช้ว่าเกิดอะไรขึ้น เขาจะเห็นเองว่ารหัสมีอักขระแปลกปน หรือยิงผิดใบ

6. ตารางการตอบรับที่ควรมี

ผลลัพธ์สั่นสีและไอคอนช่องกรอกโฟกัส
รับเข้าแล้วเบา หนึ่งครั้งเขียว เครื่องหมายถูกล้างกลับมาที่เดิม
สแกนซ้ำแรงส้ม ไอคอนซ้ำไม่ล้างกลับมาที่เดิม
รูปแบบผิดแรงแดง เครื่องหมายตกใจไม่ล้างกลับมาที่เดิม
กำลังบันทึกอยู่แรงเทา ไอคอนนาฬิกาไม่ล้างรอจนเสร็จ

สังเกตว่าทุกกรณีที่ไม่สำเร็จมีสองอย่างเหมือนกัน — สั่นแรง และไม่ล้างช่อง ซึ่งเป็นสัญญาณที่สอดคล้องกันว่า “หยุดแล้วดู”

7. เสียงของเครื่องสแกนไม่ใช่เสียงของเรา

นี่คือความเข้าใจผิดที่ทำให้ยอดไม่ตรงบ่อยที่สุด

ยิงบาร์โค้ด
   ↓
"ปี๊บ"  ← เสียงจากเครื่องสแกน แปลว่า "อ่านฉลากได้"
   ↓
แอปรับข้อมูล ตรวจ แล้วอาจปฏิเสธ
   ↓
ต้องมีสัญญาณของแอปแยกต่างหาก
พนักงานส่วนใหญ่เข้าใจว่าเสียงปี๊บแปลว่าสำเร็จ ซึ่งไม่จริง — มันแปลว่าเครื่องอ่านฉลากออก แต่แอปอาจปฏิเสธด้วยเหตุผลอะไรก็ได้ ถ้าแอปไม่มีสัญญาณของตัวเอง พนักงานจะยิงต่อไปโดยเข้าใจว่าทุกอันเข้าหมด

ทางแก้คือทำให้สัญญาณของแอป ต่างจากเสียงปี๊บอย่างชัดเจน เช่นใช้การสั่นเป็นหลัก เพราะเครื่องสแกนไม่สั่น จึงไม่มีทางสับสน

8. ห้ามบอกว่าสำเร็จก่อนสำเร็จจริง

// ผิด — สั่นแบบสำเร็จตั้งแต่ยังไม่ได้ส่ง
void _submit() {
  HapticFeedback.lightImpact();
  api.save(items);
}

// ถูก — รอผลจริงก่อน
Future<void> _submit() async {
  final ok = await api.save(items);
  if (!mounted) return;
  ok ? HapticFeedback.lightImpact() : HapticFeedback.heavyImpact();
}

เป็นเรื่องเดียวกับที่ 3.10 หัวข้อ 8 เตือนไว้ — และเมื่อผลไม่แน่ชัดเพราะเน็ตหลุด อย่าใช้สัญญาณของกรณีสำเร็จ ให้ใช้สัญญาณอีกแบบพร้อมข้อความว่าต้องตรวจสอบ

9. อย่าให้ข้อความกองทับกัน

วิธีปัญหา
แสดงข้อความลอยขึ้นมาทุกครั้งที่ยิงยิง 100 ครั้งได้ข้อความ 100 อัน กองซ้อนกันจนอันล่าสุดถูกบัง
ใช้แถบสถานะที่แทนที่ตัวเดิมเห็นผลล่าสุดเสมอ — วิธีที่ควรใช้

ใน lab เราใช้แถบสถานะที่แสดงผลล่าสุดเพียงอันเดียว

if (outcome != null)
  ScanFeedbackBanner(
    key: const Key('feedback-banner'),
    outcome: outcome,
  ),

ส่วนข้อความลอยแบบชั่วคราวเก็บไว้ใช้กับเหตุการณ์ที่เกิดไม่บ่อย เช่นผลการบันทึกทั้งชุด

10. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
พนักงานไม่รู้ว่ามีอันที่ไม่เข้าบอกผลด้วยข้อความบนจออย่างเดียวเพิ่มการสั่นที่ต่างกันตามผล
เข้าใจว่าเสียงปี๊บคือสำเร็จแอปไม่มีสัญญาณของตัวเองใช้การสั่นซึ่งเครื่องสแกนไม่มี
ยิงผิดแล้วไม่รู้ว่ายิงอะไรไปล้างช่องทุกกรณีล้างเฉพาะเมื่อสำเร็จ
ข้อความกองซ้อนกันจนอ่านไม่ทันใช้ข้อความลอยทุกครั้งที่ยิงใช้แถบสถานะที่แทนที่ตัวเดิม
ผู้ใช้ตาบอดสีแยกไม่ออกใช้สีอย่างเดียวเพิ่มไอคอนและข้อความ (2.12)
บอกว่าสำเร็จทั้งที่ยังไม่ส่งให้สัญญาณตั้งแต่กดปุ่มรอผลจาก server
ทุกความผิดพลาดได้สัญญาณเหมือนกันหมดไม่แยกชนิดของผลลัพธ์ใช้ enum ที่มีข้อความและพฤติกรรมของตัวเอง

11. แบบฝึกหัด

พนักงานรายงานปัญหาว่า “ยิงไปแล้ว 50 ชิ้น แต่ระบบขึ้นแค่ 43” และบอกว่าไม่เห็นมีอะไรผิดปกติระหว่างทาง เมื่อไปดูโค้ดพบว่า

void _handleScan() {
  final outcome = _session.accept(_controller.text);
  _controller.clear();                              // ล้างทุกกรณี
  setState(() => _lastMessage = outcome.message);   // ข้อความเล็ก ๆ ด้านล่าง
  _focus.requestFocus();
}
  1. อธิบายว่าทำไมพนักงานถึงไม่เห็นความผิดปกติ
  2. แก้โค้ดให้ปัญหานี้ไม่เกิดอีก
  3. อธิบายว่าการแก้แต่ละจุดช่วยอะไร
ดูแนวคำตอบและเหตุผล

ข้อ 1 — ทำไมไม่เห็น

สามเหตุผลที่รวมกันจนกลายเป็นปัญหาใหญ่

  1. ไม่มีสัญญาณที่รับรู้ได้โดยไม่มองจอ — พนักงานมองของในมือตลอด ข้อความเล็ก ๆ ด้านล่างจอจึงไม่มีใครเห็น
  2. ล้างช่องทุกกรณี — แม้จะเงยหน้ามาดู ก็ไม่เห็นว่ายิงอะไรผิดไป เพราะช่องว่างเปล่าเหมือนกันหมด
  3. เครื่องสแกนปี๊บทุกครั้ง — พนักงานได้ยินเสียงเดิมทั้ง 50 ครั้ง จึงเข้าใจว่าเข้าครบ

ที่ทำให้ปัญหานี้ร้ายแรงคือ ไม่มีใครรู้ว่าอันไหนหายไป — พอยอดไม่ตรงตอนจบ ต้องยิงใหม่ทั้ง 50 ชิ้นเพราะไล่ไม่ถูกว่าขาดใบไหน

void _handleScan() {
  final outcome = _session.accept(_controller.text);

  // 1. ล้างเฉพาะเมื่อสำเร็จ — ยิงผิดจะเห็นค่าค้างอยู่
  if (outcome.shouldClearInput) {
    _controller.clear();
  }

  // 2. สั่นต่างกันตามผล — รับรู้ได้โดยไม่ต้องมองจอ
  if (outcome.isFailure) {
    HapticFeedback.heavyImpact();
  } else {
    HapticFeedback.lightImpact();
  }

  setState(() => _lastOutcome = outcome);
  _focus.requestFocus();
}
// 3. แถบสถานะที่เห็นชัดด้วยหางตา แทนข้อความเล็ก ๆ
if (outcome != null)
  ScanFeedbackBanner(outcome: outcome),

ข้อ 3 — การแก้แต่ละจุดช่วยอะไร

แก้อะไรช่วยตอนไหน
สั่นแรงเมื่อผิดรู้ทันทีขณะยิง โดยไม่ต้องเงยหน้า
ไม่ล้างช่องเมื่อผิดเมื่อเงยหน้ามาดู เห็นว่ายิงอะไรผิดไป
แถบสีชัดแทนข้อความเล็กเห็นด้วยหางตาโดยไม่ต้องอ่าน

สิ่งที่ควรเพิ่มอีกอย่าง — แสดงจำนวนที่รับแล้วให้เห็นตลอดเวลา

Text('รับแล้ว ${_session.count} รายการ', key: const Key('accepted-count'))

เพราะพนักงานที่นับของในมือไปด้วยจะเทียบได้เองว่าตัวเลขตรงไหม โดยไม่ต้องรอจนจบงาน — เป็นการให้เขาตรวจสอบระบบได้ระหว่างทาง ซึ่งสำคัญกว่าการบอกผลทีละครั้งด้วยซ้ำ

บทเรียนที่ใช้ได้ต่อไป: เมื่อได้ยินคำว่า “ระบบทำงานปกติแต่ยอดไม่ตรง” ให้สงสัยเรื่องการตอบรับก่อนเสมอ — บ่อยครั้งระบบทำถูกทุกอย่างและปฏิเสธอย่างถูกต้อง แต่ไม่มีใครรู้ว่ามันปฏิเสธ

ลองต่อ: ในเทสต์ของ lab ตัวที่ตรวจ “ยิงซ้ำไม่เพิ่มรายการ และไม่ล้างช่องกรอก” ให้ลบเงื่อนไข shouldClearInput ออกจากโค้ด แล้วรัน — เทสต์จะล้มเหลวที่บรรทัดตรวจค่าในช่อง ซึ่งยืนยันว่าเทสต์ปกป้องพฤติกรรมนี้ไว้

12. สรุปท้ายบท

  • พนักงานไม่ได้มองจอ ต้องมีสัญญาณที่รับรู้ได้โดยไม่มอง
  • ใช้หลายช่องทางพร้อมกัน เพราะแต่ละช่องทางมีข้อจำกัด
  • สั่นเบาเมื่อผ่าน สั่นแรงเมื่อมีปัญหา
  • เสียงปี๊บของเครื่องสแกนไม่ได้แปลว่าสำเร็จ
  • ล้างช่องเฉพาะเมื่อสำเร็จ เพื่อให้เห็นว่ายิงอะไรผิด
  • ใช้แถบสถานะที่แทนที่ตัวเดิม ไม่ใช่ข้อความที่กองทับกัน
  • อย่าให้สัญญาณสำเร็จก่อนสำเร็จจริง
  • แสดงจำนวนสะสมให้ผู้ใช้ตรวจสอบเองได้ระหว่างทาง

บทถัดไป: 3.12 จะรวบเรื่องการคืนทรัพยากรที่กระจายอยู่ตลอด Part นี้ให้เป็นกฎเดียว เพราะหน้าจอสแกนหนึ่งหน้ามีของที่ต้องคืนหลายชิ้น
