1. ปัญหาที่บทนี้แก้

บางการกระทำย้อนกลับยาก — บันทึกรายการรับเข้า ยกเลิกใบงาน ลบข้อมูลที่สแกนไว้ทั้งหมด การถามยืนยันก่อนช่วยลดความเสียหาย

แต่ปัญหาที่ตามมาคือ ถ้าถามบ่อยเกินไป ผู้ใช้จะกดตกลงโดยไม่อ่าน ซึ่งทำให้กล่องยืนยันไร้ประโยชน์และยังทำให้ทำงานช้าลง — พนักงานที่ต้องยืนยัน 100 ครั้งต่อกะจะกดปุ่มเป็นอัตโนมัติภายในวันแรก

2. คำศัพท์ที่ต้องรู้ก่อน

คำความหมาย
showDialogเปิดกล่องโต้ตอบและรอผล
AlertDialogกล่องมาตรฐานที่มีหัวข้อ เนื้อหา และปุ่ม
Navigator.popปิดกล่องพร้อมส่งค่ากลับ
barrierDismissibleแตะนอกกล่องแล้วปิดได้ไหม
destructiveการกระทำที่ทำให้ข้อมูลหายหรือย้อนยาก

3. Mental model: กล่องคืนค่ากลับมาเป็น Future

final confirmed = await showDialog<bool>(...);
// ตรงนี้กล่องปิดไปแล้ว และเรามีคำตอบ

if (confirmed == true)  →  ผู้ใช้กดยืนยัน
if (confirmed == false) →  ผู้ใช้กดยกเลิก
if (confirmed == null)  →  ผู้ใช้แตะนอกกล่องหรือกดย้อนกลับ
ประโยคเดียวที่ต้องจำ: ผลลัพธ์เป็น bool? ไม่ใช่ bool — ต้องเขียน confirmed != true ไม่ใช่ !confirmed เพราะกรณี null ต้องถือว่าไม่ยืนยัน

4. ดูของจริงใน lab

Future<bool?> _confirmSubmit() {
  return showDialog<bool>(
    context: context,
    builder: (dialogContext) => AlertDialog(
      title: const Text('ยืนยันการบันทึก'),
      content: Text('บันทึก ${_session.count} รายการใช่หรือไม่'),
      actions: [
        TextButton(
          key: const Key('confirm-cancel'),
          // ใช้ context ของกล่อง ไม่ใช่ของหน้าจอ (1.7)
          onPressed: () => Navigator.pop(dialogContext, false),
          child: const Text('ยกเลิก'),
        ),
        FilledButton(
          key: const Key('confirm-ok'),
          onPressed: () => Navigator.pop(dialogContext, true),
          child: const Text('ยืนยัน'),
        ),
      ],
    ),
  );
}

และฝั่งที่เรียกใช้

final confirmed = await _confirmSubmit();
if (!mounted || confirmed != true) return;   // ตรวจทั้งสองอย่าง

ทบทวนจาก 1.7 — ต้องใช้ dialogContext ที่ builder ให้มา ไม่ใช่ context ของหน้าจอ ไม่งั้นอาจปิดทั้งหน้าจอแทนที่จะปิดแค่กล่อง

และตรวจ mounted ด้วยเสมอ เพราะระหว่างที่กล่องเปิดอยู่ ผู้ใช้อาจกดย้อนกลับออกจากหน้าจอไปแล้ว

5. เนื้อหาต้องบอกสิ่งที่จะเกิดขึ้นจริง

แย่ดี
“คุณแน่ใจหรือไม่”“บันทึก 12 รายการใช่หรือไม่”
“ยืนยันการทำรายการ”“ย้าย PAL-1001 ไปที่ A-01-02 จำนวน 24 ชิ้น”
ปุ่ม “ตกลง / ไม่”ปุ่ม “บันทึก / ยกเลิก”

เหตุผลสองข้อ

  1. ผู้ใช้ต้องตรวจได้ว่าถูกต้อง — กล่องที่ไม่บอกตัวเลขไม่ได้ช่วยอะไรเลย เป็นแค่ขั้นตอนที่เพิ่มมา
  2. ปุ่มที่บอกกริยาชัดจะกดผิดยากกว่า — เมื่อพนักงานกวาดตาเร็ว ๆ คำว่า “บันทึก” กับ “ยกเลิก” แยกกันชัดกว่า “ตกลง” กับ “ไม่”

6. เมื่อไหร่ควรถาม เมื่อไหร่ไม่ควร

สถานการณ์ถามไหมเหตุผล
บันทึกรายการทั้งชุดขึ้น serverถามย้อนยาก กระทบสต็อกจริง
ยกเลิกใบงานที่ทำไปครึ่งทางถามงานที่ทำไปแล้วหายหมด
ลบรายการที่สแกนไว้ทั้งหมดถามต้องยิงใหม่ทั้งหมด
ยิงบาร์โค้ดทีละใบไม่ถามเกิดบ่อยมาก และลบทีละใบได้ง่าย
เอารายการออกทีละใบไม่ถามยิงใหม่ได้ทันที
กฎที่ใช้ตัดสินได้: ถามเมื่อการยกเลิกทำได้ยากกว่าการทำใหม่ — ถ้าผู้ใช้แก้ได้เองในสองวินาที อย่าถาม ให้ทำเลยแล้วมีปุ่มเลิกทำแทน

ใน lab เราถามครั้งเดียวตอนบันทึกทั้งชุด ไม่ถามตอนยิงแต่ละใบ — ซึ่งเป็นการวางจุดถามที่ถูก เพราะการยิงผิดใบเดียวแก้ได้ด้วยการกดถังขยะข้างรายการนั้น

7. กันไม่ให้กล่องถูกกดโดยไม่ตั้งใจ

ปัญหาเฉพาะของงานสแกน — เครื่องสแกนกด Enter ให้ และ Enter อาจไปกดปุ่มที่โฟกัสอยู่ในกล่อง

Future<void> _submit() async {
  _session.beginSubmit();               // ปิดรับสแกนก่อนเปิดกล่อง
  try {
    final confirmed = await _confirmSubmit();
    if (!mounted || confirmed != true) return;
    // ... บันทึก ...
  } finally {
    _session.endSubmit();
    if (mounted) _barcodeFocus.requestFocus();   // คืนโฟกัสหลังปิด (3.2)
  }
}

สำหรับการกระทำที่ย้อนยากมาก ควรทำเพิ่มอีกสองอย่าง

ทำอะไรเพื่ออะไร
barrierDismissible: falseต้องเลือกอย่างใดอย่างหนึ่ง แตะนอกกล่องปิดไม่ได้
ให้โฟกัสเริ่มที่ปุ่มยกเลิกถ้ามีการกด Enter โดยไม่ตั้งใจ จะได้ยกเลิกแทนที่จะยืนยัน

8. ยืนยันแล้วไม่ได้แปลว่าสำเร็จ

ลำดับที่ถูกต้องคือ ยืนยัน แล้วส่ง แล้วรอผล แล้วค่อยบอกว่าสำเร็จ

// ผิด — บอกว่าสำเร็จตั้งแต่ยังไม่ส่ง
if (confirmed == true) {
  showSnackBar('บันทึกสำเร็จ');
  api.save(items);
}

// ถูก — รอผลจริงก่อน
if (confirmed != true || !mounted) return;
final ok = await api.save(items);
if (!mounted) return;
showSnackBar(ok ? 'บันทึกสำเร็จ' : 'บันทึกไม่สำเร็จ');

และเมื่อเน็ตหลุดจนไม่รู้ผล อย่าเปิดกล่องยืนยันใหม่แล้วให้ผู้ใช้กดส่งซ้ำทันที เพราะครั้งแรกอาจถึง server ไปแล้วตามที่คุยใน 3.8 หัวข้อ 8 — ควรบอกว่าสถานะไม่แน่ชัดและให้โหลดข้อมูลใหม่เพื่อตรวจสอบก่อน

9. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
กดยกเลิกแล้วยังบันทึกเขียน if (!confirmed) กับค่า nullableใช้ if (confirmed != true) return;
ปิดกล่องแล้วออกจากหน้าจอไปเลยใช้ context ของหน้าจอไปปิดกล่องใช้ dialogContext
setState() called after dispose()ไม่ตรวจ mounted หลังกล่องปิดตรวจทุกครั้งหลัง await
ยิงบาร์โค้ดตอนกล่องเปิด แล้วกดยืนยันเองไม่ได้ปิดรับสแกนปิดก่อนเปิดกล่อง
ผู้ใช้กดตกลงโดยไม่อ่านถามบ่อยเกินไปถามเฉพาะที่ย้อนยาก
กล่องบอกแค่ “แน่ใจหรือไม่”ไม่ได้สรุปสิ่งที่จะเกิดใส่ตัวเลขและรายละเอียดจริง
บอกว่าสำเร็จทั้งที่ยังไม่ได้ส่งแสดงผลตั้งแต่กดยืนยันรอผลจาก server ก่อน

10. แบบฝึกหัด

หน้าจอตรวจซีเรียลก่อนส่งออก ผู้ใช้ยิงซีเรียล 50 ชิ้นต่อหนึ่งใบงาน

  1. ควรถามยืนยันตรงจุดไหนบ้าง และไม่ถามตรงไหน
  2. เขียนกล่องยืนยันสำหรับจุดที่เลือก โดยเนื้อหาต้องช่วยให้ผู้ใช้ตรวจได้จริง
  3. อธิบายว่าจะกันการกด Enter โดยไม่ตั้งใจอย่างไร
ดูแนวคำตอบและเหตุผล

ข้อ 1 — จุดที่ควรถามและไม่ควรถาม

การกระทำถามไหมเหตุผล
ยิงซีเรียลแต่ละชิ้นไม่เกิด 50 ครั้ง ถ้าถามทุกครั้งจะใช้เวลาเพิ่มเท่าตัวและผู้ใช้จะกดโดยไม่อ่าน
เอาซีเรียลออกทีละชิ้นไม่ยิงกลับเข้ามาใหม่ได้ในสองวินาที
ล้างทั้งใบงานถามต้องยิงใหม่ทั้ง 50 ชิ้น
ปิดใบงานและส่งขึ้น serverถามย้อนยาก และกระทบข้อมูลการส่งออกจริง

ข้อ 2 — กล่องยืนยันตอนปิดใบงาน

Future<bool?> _confirmClose({
  required int scanned,
  required int expected,
  required List<String> missing,
}) {
  final hasProblem = scanned != expected;

  return showDialog<bool>(
    context: context,
    barrierDismissible: false,        // ต้องเลือกอย่างใดอย่างหนึ่ง
    builder: (dialogContext) => AlertDialog(
      title: const Text('ปิดใบงานและส่งข้อมูล'),
      content: Column(
        mainAxisSize: MainAxisSize.min,
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Text('ตรวจแล้ว $scanned จาก $expected ชิ้น'),
          if (hasProblem) ...[
            const SizedBox(height: 8),
            Text(
              'ยังขาดอีก ${expected - scanned} ชิ้น',
              style: TextStyle(
                color: Theme.of(dialogContext).colorScheme.error,
              ),
            ),
            if (missing.isNotEmpty) ...[
              const SizedBox(height: 4),
              // แสดงตัวอย่างให้ตรวจได้ แต่ไม่ยาวจนล้นจอ (2.11)
              Text(
                'เช่น ${missing.take(3).join(', ')}'
                '${missing.length > 3 ? ' และอีก ${missing.length - 3}' : ''}',
                style: Theme.of(dialogContext).textTheme.bodySmall,
              ),
            ],
          ],
        ],
      ),
      actions: [
        TextButton(
          onPressed: () => Navigator.pop(dialogContext, false),
          child: const Text('กลับไปตรวจต่อ'),   // กริยาชัด ไม่ใช่ "ไม่"
        ),
        FilledButton(
          onPressed: () => Navigator.pop(dialogContext, true),
          child: Text(hasProblem ? 'ปิดทั้งที่ยังขาด' : 'ปิดใบงาน'),
        ),
      ],
    ),
  );
}

สิ่งที่ทำให้กล่องนี้มีประโยชน์จริง

  • บอกตัวเลขเทียบกัน — “ตรวจแล้ว 47 จาก 50” ทำให้ผู้ใช้เห็นปัญหาทันทีโดยไม่ต้องคิดเอง
  • ยกตัวอย่างที่ขาด — ช่วยให้เขานึกออกว่าลืมตรงไหน แต่จำกัดไว้สามตัวเพื่อไม่ให้กล่องยาวเกิน
  • ข้อความปุ่มเปลี่ยนตามสถานการณ์ — “ปิดทั้งที่ยังขาด” ทำให้ผู้ใช้รู้ตัวว่ากำลังทำอะไร ต่างจาก “ยืนยัน” ที่กดผ่านได้โดยไม่คิด
  • ปุ่มยกเลิกบอกว่าจะไปไหนต่อ — “กลับไปตรวจต่อ” ชัดกว่า “ยกเลิก”

ข้อ 3 — กันการกด Enter โดยไม่ตั้งใจ

Future<void> _closeTask() async {
  if (_closing) return;                    // กันส่งซ้อน (3.8)

  _session.beginSubmit();                  // ปิดรับสแกนก่อนเปิดกล่อง
  try {
    final confirmed = await _confirmClose(...);
    if (!mounted || confirmed != true) return;

    setState(() => _closing = true);
    final ok = await api.closeTask(...);
    if (!mounted) return;
    _showResult(ok);
  } finally {
    _session.endSubmit();                  // ปลดเสมอ
    if (mounted) {
      setState(() => _closing = false);
      _serialFocus.requestFocus();         // คืนโฟกัสหลังกล่องปิด (3.2)
    }
  }
}

สามชั้นที่ทำงานร่วมกัน

  1. _session.beginSubmit() ทำให้ accept() ปฏิเสธทันทีถ้ามีรหัสหลุดเข้ามา
  2. barrierDismissible: false ทำให้ปิดกล่องได้เฉพาะผ่านปุ่ม
  3. ปุ่มยกเลิกอยู่ตำแหน่งแรก จึงเป็นตัวที่ได้โฟกัสก่อน — ถ้ามี Enter หลุดเข้ามา ผลคือกลับไปตรวจต่อ ซึ่งปลอดภัยกว่าปิดใบงาน

ลองต่อ: เขียนเทสต์ที่เปิดกล่องแล้วยิงบาร์โค้ดเข้าไป จากนั้นตรวจว่าจำนวนที่ตรวจแล้วไม่เพิ่ม — ดูรูปแบบได้จากเทสต์ “ระหว่างบันทึก ยิงบาร์โค้ดเพิ่มไม่ได้” ใน lab

11. สรุปท้ายบท

  • ผลของกล่องเป็น bool? — ต้องเขียน confirmed != true
  • ใช้ dialogContext ในการปิดกล่อง ไม่ใช่ context ของหน้าจอ
  • ตรวจ mounted หลังกล่องปิดเสมอ
  • เนื้อหาต้องบอกสิ่งที่จะเกิดขึ้นจริงพร้อมตัวเลข
  • ปุ่มต้องใช้คำกริยาที่ชัด ไม่ใช่ “ตกลง/ไม่”
  • ถามเฉพาะเมื่อการยกเลิกยากกว่าการทำใหม่
  • ปิดรับสแกนก่อนเปิดกล่อง และคืนโฟกัสหลังปิด
  • ยืนยันแล้วยังไม่ใช่สำเร็จ ต้องรอผลจาก server
