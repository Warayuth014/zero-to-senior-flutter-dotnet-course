2. รูปแบบมาตรฐานของคลาส

class PickOrderListItem {
  final String pickOrderId;
  final String status;          // WAITING | PICKING
  final String owner;
  final String? customerOrderId;
  final int partCount;
  final int totalRequiredQty;
  final int palletCount;
  final DateTime createdAt;

  PickOrderListItem({
    required this.pickOrderId,
    required this.status,
    required this.owner,
    this.customerOrderId,
    ...
  });

  factory PickOrderListItem.fromJson(Map<String, dynamic> json) =>
      PickOrderListItem(
        pickOrderId: json['pickOrderId'],
        status: json['status'] ?? 'WAITING',
        owner: json['owner'] ?? '',
        customerOrderId: json['customerOrderId'],
        partCount: json['partCount'] ?? 0,
        ...
      );

  bool get isWaiting => status == 'WAITING';
  bool get isPicking => status == 'PICKING';
}

มี getter ตั้งชื่อให้เงื่อนไขisWaiting อ่านง่ายกว่า status == 'WAITING'6.12

3. สามชั้นของการป้องกัน

ไล่ดูวิธีอ่านค่าที่ใช้จริงในโปรเจกต์ เรียงจากปลอดภัยน้อยไปมาก

// ชั้น 1 — ไม่ป้องกันเลย
pickOrderId: json['pickOrderId'],

// ชั้น 2 — มีค่าเริ่มต้น
status: json['status'] ?? 'WAITING',
partCount: json['partCount'] ?? 0,

// ชั้น 3 — แปลงแล้วมีค่าเริ่มต้น
createdAt: DateTime.tryParse(json['createdAt'] ?? '') ?? DateTime.now(),
ชั้นถ้าฝั่ง .NET ไม่ส่งค่านี้มา
1TypeError ตอนแปลง — จอเห็นเป็นข้อความผิดพลาดที่อ่านไม่รู้เรื่อง
2ได้ค่าเริ่มต้น ทำงานต่อได้
3ได้เวลาปัจจุบัน ทำงานต่อได้

สังเกตว่าฟิลด์ชั้น 1 มักเป็นรหัสหลัก — pickOrderId, palletId, partId ซึ่งเป็นการตัดสินใจที่สมเหตุสมผล เพราะถ้าไม่มีรหัส ข้อมูลชิ้นนั้นก็ไม่มีความหมายอยู่ดี

แต่มีปัญหาตรงที่ข้อความผิดพลาดที่ได้ไม่ช่วยอะไร

// ผู้ใช้เห็นแบบนี้
เกิดข้อผิดพลาด: type 'Null' is not a subtype of type 'String'

เทียบกับสิ่งที่ 6.9 สอน — อ่านผ่านตัวช่วยที่บอกได้ว่าฟิลด์ไหนหาย

pickOrderId: requireString(json, 'pickOrderId', path: 'PickOrderListItem'),
// ล้มเหลว → 'PickOrderListItem.pickOrderId: ต้องมีค่าแต่ไม่พบ'

ต่างกันตรงที่ข้อความหลังบอกได้ทันทีว่าต้องไปคุยกับใครเรื่องอะไร

4. ชั้น 3 มีกับดักที่ต้องรู้

createdAt: DateTime.tryParse(json['createdAt'] ?? '') ?? DateTime.now(),

ทนทานมาก ไม่มีทางพัง แต่ทนทานเกินไป

ถ้า .NET ส่งได้ถูกไหม
"2026-08-16T09:30:00Z"เวลานั้นถูก
ไม่ส่งมาเลยเวลาปัจจุบันผิด แต่ไม่มีใครรู้
"16/08/2026"เวลาปัจจุบันผิด แต่ไม่มีใครรู้
nullเวลาปัจจุบันผิด แต่ไม่มีใครรู้

อาการที่เกิดในคลัง: รายการงานที่เรียงตามเวลาสร้างจะเรียงผิด งานเก่าโผล่ขึ้นบนสุดเพราะได้เวลาปัจจุบัน และยิ่งค้างนานยิ่งดูใหม่

สาเหตุนี้หายากมาก เพราะไม่มี error ไม่มี log และหน้าจอดูปกติทุกอย่าง

หลักที่ใช้ตัดสิน: ค่าเริ่มต้นที่ดีต้องบอกได้ว่าเป็นค่าเริ่มต้น — 0 กับ '' บอกได้ แต่ DateTime.now() บอกไม่ได้ เพราะดูเหมือนข้อมูลจริงทุกประการ
สำหรับวันเวลา ทางที่ตรงกว่าคือให้เป็น DateTime? แล้วให้จอแสดง '—' เมื่อไม่มี

รูปแบบเดียวกันนี้อยู่ใน receiving_models.dart ด้วย

createdAt: json['createdAt'] != null
    ? DateTime.tryParse(json['createdAt']) ?? DateTime.now()
    : DateTime.now(),

เขียนยาวกว่าแต่ได้ผลเหมือนกันทุกประการ

5. รายการซ้อนรายการ

factory PickOrderDetailFull.fromJson(Map<String, dynamic> json) =>
    PickOrderDetailFull(
      ...
      pallets: ((json['pallets'] ?? []) as List)
          .map((e) => PickOrderPalletInfo.fromJson(e))
          .toList(),
      parts: ((json['parts'] ?? []) as List)
          .map((e) => PickOrderPartInfo.fromJson(e))
          .toList(),
    );

อ่านจากในออก

ขั้นทำอะไร
json['pallets'] ?? []ไม่มีก็ใช้รายการว่าง
as Listบอก Dart ว่าเป็นรายการ
.map(...fromJson(e))แปลงทีละตัว
.toList()ปิด lazy ให้เป็นรายการจริง

.toList() จำเป็น — map ใน Dart เป็นแบบขี้เกียจ ถ้าไม่ปิด การแปลงจะเกิดใหม่ทุกครั้งที่วนอ่าน ซึ่งใน ListView ที่เลื่อนขึ้นลงจะแปลงซ้ำนับร้อยครั้ง ตรงกับที่ 6.10 อธิบาย

แต่มีจุดที่ยังไม่ปลอดภัย

.map((e) => PickOrderPalletInfo.fromJson(e))
//                                       ^ e เป็น dynamic

ถ้าสมาชิกในรายการไม่ใช่ Map — เช่นเซิร์ฟเวอร์ส่งรายการของสตริงมา — จะพังตอนอ่านฟิลด์แรก ไม่ใช่ตอนนี้ ทำให้ข้อความยิ่งไกลจากสาเหตุ

ในไฟล์เดียวกันมีอีกที่ที่เขียนไว้ปลอดภัยกว่า

.map((e) => PickOrderListItem.fromJson(e as Map<String, dynamic>))

เขียนแบบหลังดีกว่า เพราะพังตรงจุดที่ผิดจริงและข้อความบอกชนิดที่ได้มา

6. สิ่งที่ไม่มีในโมเดลทั้งหมด

ค้นทั้งโฟลเดอร์ models/ เจอ toJson แค่ที่เดียว

ไม่มีผลที่ตามมา
toJson()body ของ POST ถูกประกอบเป็น Map ด้วยมือใน services/api/
copyWith()อัปเดตทีละฟิลด์ไม่ได้ ต้องสร้างใหม่ทั้งตัวหรือเก็บใน state แยก
== / hashCodeเทียบสองตัวว่าเหมือนกันไม่ได้ ต้องเทียบทีละฟิลด์
การสร้างโค้ดอัตโนมัติเขียนมือทั้งหมด — ควบคุมได้เต็มที่ แต่พิมพ์ผิดชื่อคีย์แล้วไม่มีใครจับ

สามข้อแรกไม่มีเพราะไม่ได้ใช้ ซึ่งถูกต้อง — ข้อมูลไหลทางเดียวจากเซิร์ฟเวอร์มาที่จอ ไม่มีการแก้แล้วส่งกลับทั้งก้อน

// body ถูกประกอบเองใน picking_api.dart
final response = await _post('/picking/confirm-pick', {
  'pickOrderId': pickOrderId,
  'sourcePalletId': sourcePalletId,
  'destPalletId': destPalletId,
  'items': items,          // ← List<Map<String, dynamic>> ที่จอประกอบมาเอง
  'operatorId': operatorId,
});

items ตรงนี้เป็นจุดที่เสี่ยงที่สุด — เป็น List<Map<String, dynamic>> ที่จอสร้างขึ้นเอง ไม่มีชนิดกำกับ พิมพ์ชื่อคีย์ผิดคอมไพเลอร์ไม่จับ และจะรู้ตอนเซิร์ฟเวอร์ตอบกลับมาว่าข้อมูลไม่ครบ

7. ชื่อคีย์กับสิ่งที่ .NET ส่งมา

ทุกคีย์ในโปรเจกต์นี้เป็น camelCase

json['pickOrderId'], json['partCount'], json['createdAt']

ตรงกับค่าเริ่มต้นของ ASP.NET Core ที่แปลง PascalCase เป็น camelCase ให้อัตโนมัติ — PickOrderId ฝั่ง C# กลายเป็น pickOrderId ใน JSON

สิ่งที่ต้องระวัง: ถ้ามีใครเปลี่ยนการตั้งค่า serialization ฝั่ง .NET หรือมี endpoint ที่คืน object ที่สร้างเองแบบไม่ผ่านตัวแปลงมาตรฐาน คีย์จะกลายเป็น PascalCase แล้วทุกฟิลด์จะเป็น null พร้อมกัน
อาการคือหน้าจอว่างเปล่าหรือขึ้นศูนย์หมด ทั้งที่เซิร์ฟเวอร์ตอบ 200 และข้อมูลมีจริง — 14.9 เจอเคสนี้ตอนเขียนเทสต์กับ fixture ที่เป็น PascalCase

วิธีป้องกันคือให้ตัวอ่านรับได้ทั้งสองแบบ

/// อ่านคีย์แบบ camelCase ก่อน ถ้าไม่มีลอง PascalCase
Object? pick(Map<String, dynamic> json, String key) =>
    json[key] ?? json[key[0].toUpperCase() + key.substring(1)];

แต่ในโปรเจกต์นี้ยังไม่จำเป็น เพราะทุก endpoint มาจาก controller เดียวกันและตั้งค่าเหมือนกัน — จดไว้เป็นสิ่งที่ต้องตรวจถ้าวันหนึ่งหน้าจอว่างเปล่าโดยไม่มี error

8. ข้อผิดพลาดที่พบบ่อย

เข้าใจผิดที่จริง
ค่าเริ่มต้นทำให้ปลอดภัยขึ้นเสมอDateTime.now() ซ่อนปัญหาแทนที่จะบอก
ไม่มี error แปลว่าข้อมูลถูกวันเวลาผิดไม่ทำให้เกิด error
as List รับประกันชนิดของสมาชิกรับประกันแค่ว่าเป็นรายการ
ลืม .toList() แล้วแค่ช้าลงนิดหน่อยแปลงใหม่ทุกครั้งที่วนอ่าน
โมเดลมี toJson อยู่แล้วมีที่เดียวในทั้งโฟลเดอร์ — body ประกอบเองด้วยมือ

9. แบบฝึกหัด

ผู้ใช้แจ้งว่า “รายการ Pick Order เรียงมั่ว งานที่สั่งเมื่อวานอยู่บนสุดเหมือนเพิ่งสั่ง”

  1. สาเหตุที่น่าจะเป็นคืออะไร และยืนยันอย่างไรโดยไม่ต้องแก้โค้ด
  2. แก้อย่างไรให้ปัญหานี้ไม่กลับมาเงียบ ๆ อีก
  3. ถ้าฝั่ง .NET แก้ไม่ได้ในเร็ว ๆ นี้ ฝั่งแอปควรทำอย่างไร
แนวคำตอบ

ข้อ 1

createdAt ของรายการเหล่านั้นอ่านไม่ได้ จึงกลายเป็น DateTime.now() ทุกครั้งที่โหลด — ยิ่งโหลดใหม่ยิ่งดูใหม่

ยืนยันโดยไม่แก้โค้ด

วิธีดูอะไร
ยิง endpoint ตรงด้วยเบราว์เซอร์หรือ curlcreatedAt มีในคำตอบไหม รูปแบบเป็นอย่างไร
รีเฟรชหน้าจอสองครั้งห่างกันหนึ่งนาทีถ้าเวลาที่แสดงขยับตาม แปลว่าเป็นเวลาปัจจุบัน ไม่ใช่เวลาจริง

วิธีที่สองยืนยันได้เร็วที่สุดและทำได้จากหน้างานเลย

ข้อ 2 — แก้ไม่ให้กลับมาเงียบ ๆ

class PickOrderListItem {
  ...
  final DateTime? createdAt;      // เปลี่ยนเป็นยอมให้ไม่มี

  factory PickOrderListItem.fromJson(Map<String, dynamic> json) =>
      PickOrderListItem(
        ...
        // ไม่มีหรืออ่านไม่ได้ = ไม่มี ไม่ใช่ "ตอนนี้"
        createdAt: DateTime.tryParse(json['createdAt']?.toString() ?? ''),
      );
}

คอมไพเลอร์จะบังคับให้ทุกที่ที่ใช้ createdAt จัดการกรณีไม่มีค่า ซึ่งเป็นสิ่งที่ต้องการ

// ตอนแสดง
Text(item.createdAt != null ? formatThaiDateTime(item.createdAt!) : '—')

// ตอนเรียง — ตัวที่ไม่มีเวลาไปอยู่ท้าย ไม่ใช่บนสุด
items.sort((a, b) {
  if (a.createdAt == null) return 1;
  if (b.createdAt == null) return -1;
  return b.createdAt!.compareTo(a.createdAt!);
});

ตอนนี้ปัญหาเดิมจะแสดงตัวเป็น — บนหน้าจอ ซึ่งมองเห็นได้ แทนที่จะซ่อนอยู่

ข้อ 3 — ถ้า .NET แก้ไม่ได้เร็ว

ทำไม่ทำ
แสดง — ในช่องเวลาใส่เวลาปัจจุบันแทน
เรียงด้วยฟิลด์อื่นที่เชื่อได้ เช่นรหัสงานที่เรียงตามลำดับปล่อยให้เรียงมั่วต่อไป
บันทึกไว้ว่ามีกี่รายการที่อ่านเวลาไม่ได้เงียบ

ข้อสุดท้ายสำคัญ เพราะจะได้รู้ว่าปัญหาใหญ่แค่ไหน — สองรายการกับสองพันรายการต้องได้ความสำคัญต่างกัน

final missing = items.where((e) => e.createdAt == null).length;
if (missing > 0) {
  debugPrint('createdAt อ่านไม่ได้ $missing/${items.length} รายการ');
}

ข้อสรุปที่สำคัญกว่าตัวบั๊ก: ค่าเริ่มต้นทุกตัวคือการตัดสินใจว่า “ถ้าไม่มีข้อมูล จะเดาว่าอะไร” — เดาแล้วผู้ใช้เห็นว่าเป็นการเดา (0, '—') ยอมรับได้ แต่เดาแล้วดูเหมือนข้อมูลจริง (DateTime.now()) จะกลายเป็นข้อมูลผิดที่ไม่มีใครจับได้
