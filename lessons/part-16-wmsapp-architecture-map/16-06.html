2. ApiResult — ผลลัพธ์สองทาง

class ApiResult<T> {
  final bool success;
  final T? data;
  final String? error;
  final int? statusCode;

  ApiResult.success(this.data) : success = true, error = null, statusCode = 200;

  ApiResult.error(this.error, {this.statusCode}) : success = false, data = null;

  bool get isNotFound => statusCode == 404;
}

4. หาว่าจะยิงไปที่ไหน

static String? _cachedBase;

static Future<String> _resolveBase() async {
  if (_cachedBase != null) return _cachedBase!;

  final custom = await _configService.getCustomBase();
  // kIsWeb ต้องเช็คก่อนเสมอ — Platform.* throw บน web (dart:io ไม่รองรับ)
  final isAndroid = !kIsWeb && Platform.isAndroid;

  final candidates = <String>[
    if (custom != null) custom,
    if (isAndroid) 'http://10.0.2.2:$_defaultPort',
    if (!isAndroid) 'http://localhost:$_defaultPort',
  ];

  for (final base in candidates) {
    final uri = Uri.parse(base);
    try {
      final socket = await Socket.connect(uri.host, uri.port,
          timeout: const Duration(seconds: 1));
      socket.destroy();
      _cachedBase = '$base/api';
      return _cachedBase!;
    } catch (_) {}
  }

  _cachedBase = '${candidates.first}/api';
  return _cachedBase!;
}

ลำดับความสำคัญคือค่าที่ผู้ใช้ตั้ง มาก่อนค่าเริ่มต้นของแพลตฟอร์ม ตรงกับที่ 10.3 สอน และคอมเมนต์บนบรรทัด kIsWeb อธิบายกับดักจริงไว้ให้ด้วย

สถานการณ์ยิงไปที่
ผู้ใช้ตั้ง IP แล้วIP นั้น
ยังไม่ตั้ง + รันบน emulator10.0.2.2:5000
ยังไม่ตั้ง + รันบนเดสก์ท็อป/เว็บlocalhost:5000
ยังไม่ตั้ง + รันบน PDA จริง10.0.2.2:5000 ซึ่งไม่มีอยู่จริง

แถวสุดท้ายคือเหตุผลที่หน้า Settings มีอยู่ — บนเครื่องจริง 10.0.2.2 เป็นที่อยู่ที่ไม่มีความหมาย ต้องตั้งเอง

5. cache ที่ต้องรู้จัก

_cachedBase เป็น static จำไว้จนกว่าจะถูกล้างด้วยมือ

static void resetBaseUrl() => _cachedBase = null;

เหตุผลที่ต้อง cache: _resolveBase() ทำ Socket.connect ที่รอได้ถึง 1 วินาทีต่อ candidate ถ้าไม่จำไว้ ทุกคำขอจะเสียเวลาตรงนี้ก่อน

ผลข้างเคียงที่ต้องรู้

เมื่อเกิดอะไร
ผู้ใช้บันทึก IP ใหม่ต้องเรียก resetBaseUrl() ไม่งั้นยังยิงที่เก่า
ย้ายวง Wi-Fi ระหว่างใช้งานcache ยังเป็นที่อยู่เดิม ซึ่งอาจถูกอยู่
เซิร์ฟเวอร์ที่ผู้ใช้ตั้งล่มตอนเปิดแอปfallback ไป localhost แล้วจำไว้ทั้ง session

แถวสุดท้ายเป็นกรณีที่หายากแต่หาสาเหตุยากที่สุด ไล่ตามลำดับ

ผู้ใช้ตั้ง 192.168.1.50 ไว้แล้ว
  → เปิดแอปตอนเซิร์ฟเวอร์กำลังรีสตาร์ท
    → Socket.connect ไป 192.168.1.50 ล้มเหลว
      → ไม่มี candidate อื่น (ไม่ใช่ Android emulator)
        → _cachedBase = '192.168.1.50/api' ← ยังถูกอยู่ เพราะใช้ candidates.first
          → เซิร์ฟเวอร์กลับมา → ใช้งานได้ปกติ

ในเคสนี้บรรทัด fallback candidates.first ช่วยไว้ เพราะ candidate แรกคือค่าที่ผู้ใช้ตั้ง — คนเขียนคิดเรื่องนี้แล้วและเขียนคอมเมนต์อธิบายไว้

// ไม่มี candidate ไหน connect ได้ตอนนี้ — ใช้ตัวแรก (custom ถ้ามี) เป็น fallback
// request จริงจะ throw SocketException ให้เห็น error ชัดเจนแทน
แต่ยังมีช่องที่ไม่ได้ปิด: ถ้าผู้ใช้ไม่เคยตั้งค่าและรันบน Android จริง candidate แรกคือ 10.0.2.2 ซึ่งจะถูกจำไว้ทั้ง session — ผู้ใช้ต้องปิดเปิดแอปหลังตั้งค่า ยกเว้นจะตั้งผ่านหน้า Settings ที่เรียก resetBaseUrl() ให้

6. สามเมธอดที่ยิงจริง

Future<ApiResult<Map<String, dynamic>>> _get(String path) async {
  try {
    final base = await ApiService._resolveBase();
    final response = await http
        .get(Uri.parse('$base$path'), headers: _headers)
        .timeout(const Duration(seconds: 10));
    return _handle(response);
  } on SocketException {
    return ApiResult.error('ไม่สามารถเชื่อมต่อ server ได้');
  } on TimeoutException {
    return ApiResult.error('การเชื่อมต่อหมดเวลา กรุณาลองใหม่');
  } catch (error) {
    return ApiResult.error('เกิดข้อผิดพลาด: $error');
  }
}

7. _handle — แปลงคำตอบ

ApiResult<Map<String, dynamic>> _handle(http.Response response) {
  dynamic decoded;

  try {
    decoded = jsonDecode(response.body);
  } catch (_) {
    return ApiResult.error('Server ส่งข้อมูลไม่ถูกต้อง (${response.statusCode})');
  }

  final body = decoded is List
      ? <String, dynamic>{'items': decoded}
      : decoded as Map<String, dynamic>;

  if (response.statusCode >= 200 && response.statusCode < 300) {
    return ApiResult.success(body);
  }

  final error = body['error']?.toString();
  final detail = body['detail']?.toString();
  final message = (error != null && detail != null)
      ? '$error\n$detail'
      : error ?? detail ?? 'เกิดข้อผิดพลาด (${response.statusCode})';

  return ApiResult.error(message, statusCode: response.statusCode);
}

สามเรื่องที่ทำได้ดี

เรื่องรายละเอียด
JSON พังไม่ทำให้แอปพังครอบ jsonDecode ด้วย try แล้วคืนข้อความที่มีรหัสสถานะให้ตามต่อได้
รับได้ทั้ง array และ objectห่อ array เป็น {'items': [...]} ให้ชั้นบนอ่านแบบเดียวกัน
อ่าน detail จาก ProblemDetailsตรงกับที่ .NET ส่งมา (7.7)

จุดที่เสี่ยง

: decoded as Map<String, dynamic>;

ถ้าเซิร์ฟเวอร์ตอบ JSON ที่ถูกต้องแต่เป็นค่าเดี่ยว — เช่น true, 42, หรือ "ok" — jsonDecode จะผ่าน แล้ว as Map จะโยน TypeError ซึ่งถูกจับโดย catch (error) ใน _get แล้วกลายเป็นข้อความ 'เกิดข้อผิดพลาด: type ...' ที่ผู้ใช้อ่านไม่รู้เรื่อง

เขียนให้ตรงกว่าได้ด้วยการตรวจชนิดแทน cast

final Map<String, dynamic> body;
if (decoded is List) {
  body = {'items': decoded};
} else if (decoded is Map<String, dynamic>) {
  body = decoded;
} else {
  return ApiResult.error(
    'Server ส่งข้อมูลผิดรูปแบบ (${response.statusCode})',
    statusCode: response.statusCode,
  );
}

รูปแบบนี้ตรงกับที่ 6.9 สอนว่าให้ตรวจชนิดที่ขอบแล้วคืนข้อความที่ผู้ใช้เข้าใจ ดีกว่าปล่อยให้ TypeError หลุดไปโผล่เป็นข้อความภาษาอังกฤษ

8. ข้อผิดพลาดที่พบบ่อย

เข้าใจผิดที่จริง
singleton ทำให้เทสต์ง่ายขึ้นตรงกันข้าม — ปิดช่องฉีดของทดสอบ
ApiService() สร้างตัวใหม่ทุกครั้งได้ตัวเดิมเสมอ
เปลี่ยน IP แล้วมีผลทันทีเสมอต้องผ่าน resetBaseUrl()
_resolveBase() ตรวจว่าเป็นเซิร์ฟเวอร์เราจริงตรวจแค่ว่าพอร์ตเปิด — งานตรวจจริงอยู่ที่ probeConnection
_handle รับ JSON ทุกรูปแบบค่าเดี่ยวจะกลายเป็นข้อความผิดพลาดที่อ่านไม่รู้เรื่อง
ทุกความล้มเหลวเหมือนกันต่างกันมาก แต่โค้ดนี้ยุบเป็นข้อความอย่างเดียว

9. แบบฝึกหัด

ทีมอยากเพิ่มการลองใหม่อัตโนมัติเมื่อสัญญาณกระพริบ

  1. ควรใส่ที่ไหนในไฟล์นี้ และห้ามใส่ที่ไหน
  2. ต้องเปลี่ยนอะไรใน ApiResult ก่อน
  3. เขียนโค้ดที่แก้จริง
แนวคำตอบ

ข้อ 1 — ใส่ที่ไหน

เมธอดลองใหม่ได้ไหมทำไม
_getได้อ่านอย่างเดียว ยิงซ้ำไม่เปลี่ยนอะไร
_postไม่ได้อาจสร้างรายการซ้ำ
_uploadFileไม่ได้เหมือนกัน และไฟล์ใหญ่ทำให้เสียเวลามาก

นี่คือกฎเดียวกับ 7.12 และเป็นเหตุผลที่ต้องแยก _get ออกจาก _post ตั้งแต่แรก

ข้อ 2 — ต้องเปลี่ยน ApiResult ก่อน

ตอนนี้ ApiResult.error เก็บแค่ข้อความ ตัวลองใหม่จะไม่รู้ว่าความล้มเหลวนี้ลองแล้วมีโอกาสต่างไหม การเดาจากข้อความเป็นวิธีที่พังง่ายที่สุด

enum ApiFailureKind { network, timeout, server, client, decode }

class ApiResult<T> {
  ...
  final ApiFailureKind? kind;

  ApiResult.error(this.error, {this.statusCode, this.kind})
      : success = false, data = null;

  bool get isRetryable =>
      kind == ApiFailureKind.network || kind == ApiFailureKind.timeout;
}

ต้องแก้จุดที่สร้าง ApiResult.error ทั้งหมดให้ระบุ kind ด้วย ซึ่งมีในไฟล์นี้ 12 จุด

ข้อ 3 — โค้ดที่แก้จริง

Future<ApiResult<Map<String, dynamic>>> _get(String path) async {
  const maxAttempts = 3;

  for (var attempt = 1; ; attempt++) {
    final result = await _getOnce(path);
    if (result.success || !result.isRetryable || attempt >= maxAttempts) {
      return result;
    }
    // ถอยห่างขึ้นทุกครั้ง + สุ่มเล็กน้อย ไม่ให้ทุกเครื่องยิงพร้อมกันหลังเน็ตกลับมา
    final backoff = Duration(milliseconds: 300 * attempt + _random.nextInt(200));
    await Future<void>.delayed(backoff);
  }
}

ไม่มีอะไรบอกว่ากำลังลองใหม่ผู้ใช้เห็นแค่วงกลมหมุนนานผิดปกติ

ทางแก้ที่สมดุลกว่าคือลดเวลา timeout ต่อครั้งลงเมื่อจะลองใหม่

// ครั้งแรกใจร้อน ครั้งหลังใจเย็น รวมแล้วไม่เกิน ~18 วินาที
final timeout = Duration(seconds: attempt == 1 ? 5 : 10);

ข้อสรุปที่สำคัญกว่าโค้ด: การลองใหม่ทำให้แอปทนสัญญาณกระพริบได้จริง แต่แลกมาด้วยเวลารอที่ผู้ใช้เห็น ถ้าไม่บอกผู้ใช้ว่ากำลังทำอะไร จะกลายเป็น “แอปช้าลง” ในสายตาเขา ไม่ใช่ “แอปทนขึ้น”

10. สรุป

  • ทุกคำขอในแอปผ่าน _get, _post, _uploadFile ในไฟล์เดียว 154 บรรทัด
  • ApiResult แยกสำเร็จ/ล้มเหลวได้ แต่ไม่แยกชนิดของความล้มเหลว
  • singleton ทำให้เรียกง่ายจาก 26 จุด แต่ปิดช่องฉีดของทดสอบ — เป็นเหตุผลที่แอปนี้ไม่มีเทสต์
  • _resolveBase() เรียงค่าที่ผู้ใช้ตั้งไว้ก่อน แล้วค่อยตกไปที่ค่าของแพลตฟอร์ม
  • _cachedBase จำไว้ทั้ง session ต้องล้างด้วย resetBaseUrl() เมื่อเปลี่ยนค่า
  • มี timeout ครบทุกเมธอด แต่ไม่มีการลองใหม่ ไม่มีรหัสคำสั่ง
  • _handle รับ array และ object ได้ แต่ค่าเดี่ยวจะกลายเป็นข้อความที่ผู้ใช้อ่านไม่รู้เรื่อง
