lib/main.dart ทั้งไฟล์มี 39 บรรทัด

void main() async {
  WidgetsFlutterBinding.ensureInitialized();

  runApp(const WmsApp());
}

class WmsApp extends StatelessWidget {
  const WmsApp({super.key});

  @override
  Widget build(BuildContext context) {
    return ValueListenableBuilder<ThemeMode>(
      valueListenable: themeProvider,
      builder: (_, themeMode, __) => MaterialApp(
        title: 'WMS',
        theme: AppTheme.light,
        darkTheme: AppTheme.dark,
        themeMode: themeMode,
        debugShowCheckedModeBanner: false,
        localizationsDelegates: const [
          GlobalMaterialLocalizations.delegate,
          GlobalWidgetsLocalizations.delegate,
          GlobalCupertinoLocalizations.delegate,
        ],
        supportedLocales: const [
          Locale('th', 'TH'),
          Locale('en', 'US'),
        ],
        locale: const Locale('th', 'TH'),
        home: const HomeScreen(),
      ),
    );
  }
}

2. อ่านทีละบรรทัด

บรรทัดบอกอะไร
main() asyncประกาศเป็น async แต่ไม่มี await เลย — ไม่ผิด แต่เป็นร่องรอยว่าเคยมีอะไรอยู่
ensureInitialized()จำเป็นเมื่อจะเรียก plugin ก่อน runApp — ตอนนี้ไม่ได้เรียก แต่ปล่อยไว้ไม่เสียหาย
ValueListenableBuilderทั้งแอปสร้างใหม่เมื่อธีมเปลี่ยน
themeProviderตัวแปรระดับไฟล์จาก theme.dart — ไม่ได้ส่งเข้ามา
debugShowCheckedModeBanner: falseซ่อนแถบ DEBUG มุมขวาบน
supportedLocales มีสองภาษาแต่ locale ตรึงไว้ที่ th_TH
home: ไม่ใช่ routes:ไม่มีตารางเส้นทาง ทุกการนำทางเป็น Navigator.push ตรง ๆ
เรื่อง locale ที่ตรึงไว้ — ประกาศรองรับสองภาษาแต่บังคับใช้ไทยเสมอ แปลว่าผู้ใช้เปลี่ยนภาษาไม่ได้แม้จะตั้งเครื่องเป็นอังกฤษ
สิ่งที่ localizationsDelegates ทำจริงคือแปลข้อความของ Flutter เอง (ปุ่ม “ยกเลิก” ในตัวเลือกวันที่ ข้อความในเมนูคัดลอก/วาง) ไม่ได้แปลข้อความของแอปเรา — ตรงกับที่ 11.7 อธิบายไว้ ส่วนข้อความในแอปทั้งหมดเป็นภาษาไทยฝังในโค้ด

3. ธีมถูกสลับอย่างไร

ใน theme/theme.dart

class ThemeProvider extends ValueNotifier<ThemeMode> {
  ThemeProvider() : super(ThemeMode.light);

  bool get isDark => value == ThemeMode.dark;

  void toggle() {
    value = isDark ? ThemeMode.light : ThemeMode.dark;
  }

  void setMode(ThemeMode mode) => value = mode;
}

// Global instance
final themeProvider = ThemeProvider();

ตัวแปรระดับไฟล์หนึ่งตัว ใครก็เรียกได้ถ้า import ไฟล์นี้ ใน home_screen.dart จึงเขียนได้เลยว่า

onPressed: () => themeProvider.toggle(),
วิธีนี้วิธีใน Part 11
เข้าถึงจากimport ไฟล์อ่านจาก context
โค้ดที่ต้องเขียนบรรทัดเดียวต้องมี Scope ครอบ
สลับตัวปลอมในเทสต์ทำไม่ได้ทำได้
รู้ว่าใครใช้บ้างต้องค้นทั้งโปรเจกต์ดูจากที่ครอบ

สำหรับค่าที่มีตัวเดียวทั้งแอปและไม่ต้องทดสอบ อย่างธีม วิธีนี้ตรงไปตรงมาและอ่านง่าย ปัญหาจะเกิดเมื่อเอาวิธีเดียวกันไปใช้กับสิ่งที่ต้องทดสอบ — ซึ่งจะเจอที่ 16.6

ธีมที่เลือกไว้ไม่ถูกจำ — ThemeProvider() เริ่มที่ ThemeMode.light เสมอ ปิดแอปแล้วเปิดใหม่กลับเป็นสว่างทุกครั้ง ทั้งที่โปรเจกต์นี้มี shared_preferences ใช้อยู่แล้วในที่อื่น

4. AppTheme — สองแบบในไฟล์เดียว

AppTheme ยาว 324 บรรทัด และมีสามส่วน

ส่วนเนื้อหา
ค่าคงที่สีแบรนด์ 7 สี, จานสีสว่าง 7 ค่า, จานสีมืด 7 ค่า, รัศมี 5 ระดับ, ระยะห่าง 5 ระดับ
ตัวช่วยที่รู้ธีมtextPrimary(context), textGrey(context), border(context)…
ตัวสร้าง ThemeDatalight, dark เรียก _build() ตัวเดียวกัน
static Color textPrimary(BuildContext context) =>
    _isDark(context) ? _darkTextPrimary : _lightTextPrimary;

static bool _isDark(BuildContext context) =>
    Theme.of(context).brightness == Brightness.dark;

รูปแบบนี้แก้ปัญหาจริง — ค่าที่ ThemeData ไม่มีช่องให้เก็บ (เช่นสีเทาของข้อความรอง) ต้องเก็บไว้เองแล้วเลือกตามธีม

สังเกตว่า _build() ตัวเดียวรับ 9 พารามิเตอร์แล้วสร้างทั้งสองธีม แปลว่าเพิ่มการตกแต่งใหม่ครั้งเดียวได้ทั้งสองโหมด — เป็นการตัดสินใจที่ดีและควรทำตาม

ปุ่มหลักทุกปุ่มในแอปจึงสูง 52 และกว้างเต็มพื้นที่ — ตัดสินใจครั้งเดียวมีผลทั้งแอป ตรงกับที่ 11.3 บอกว่าค่าปุ่มต้องมาจากที่เดียว

Sarabun เป็นฟอนต์ไทยที่ฝังมากับแอป (3 น้ำหนัก อยู่ใน assets/fonts/) ไม่ได้พึ่งฟอนต์ของเครื่อง — สำคัญบน PDA ที่ฟอนต์ไทยของระบบมักไม่ครบ

6. จุดที่ต้องระวังเมื่อจะแก้

จุดทำไมต้องระวัง
ค่าคงที่ใน AppThemeถูกอ้างจากทุกจอ เปลี่ยนแล้วกระทบทั้งแอป
textPrimaryColor / borderColorเวอร์ชันไม่รู้ธีม ใช้สีสว่างเสมอ — ในโหมดมืดจะเพี้ยน
_build()เปลี่ยนพารามิเตอร์เดียวกระทบทั้งสองธีม
themeProviderไม่มีที่ครอบ ต้องค้นทั้งโปรเจกต์ว่าใครแตะบ้าง

แถวที่สองมาจากส่วนนี้ในไฟล์จริง

// ── Backwards-compat static colors ────────
// (used by screens that don't have context yet)
static const textPrimaryColor = _lightTextPrimary;
static const textGreyColor    = _lightTextGrey;
static const borderColor      = _lightBorder;

คอมเมนต์บอกตรง ๆ ว่าเป็นของเก่าที่เก็บไว้ให้โค้ดเดิมใช้ได้ ถ้าเจอจอที่สีเพี้ยนในโหมดมืด ให้สงสัยสามตัวนี้ก่อน — ค้นชื่อมันทั้งโปรเจกต์แล้วเปลี่ยนเป็นเวอร์ชันที่รับ context

7. ข้อผิดพลาดที่พบบ่อย

เข้าใจผิดที่จริง
main() เป็น async แปลว่ารออะไรอยู่ไม่มี await เลย เป็นร่องรอยเก่า

8. แบบฝึกหัด

  1. ทำให้ธีมที่ผู้ใช้เลือกถูกจำไว้หลังปิดแอป — เขียนโค้ดที่ต้องแก้ทั้งหมด
  2. ถ้าต้องให้ผู้ใช้สลับภาษาไทย/อังกฤษได้จริง ต้องเปลี่ยนอะไรบ้าง เรียงตามลำดับที่ทำจริงได้
แนวคำตอบ

ข้อ 1 — จำธีมไว้

ใน theme.dart

class ThemeProvider extends ValueNotifier<ThemeMode> {
  ThemeProvider() : super(ThemeMode.light);

  static const _key = 'themeMode';

  Future<void> restore() async {
    final prefs = await SharedPreferences.getInstance();
    value = prefs.getString(_key) == 'dark'
        ? ThemeMode.dark
        : ThemeMode.light;
  }

  Future<void> _persist() async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.setString(_key, isDark ? 'dark' : 'light');
  }

  void toggle() {
    value = isDark ? ThemeMode.light : ThemeMode.dark;
    unawaited(_persist());
  }
}

ใน main.dart

void main() async {
  WidgetsFlutterBinding.ensureInitialized();
  await themeProvider.restore();   // ตอนนี้ async มีเหตุผลแล้ว
  runApp(const WmsApp());
}

สังเกตว่า ensureInitialized() ที่ดูเหมือนเกินความจำเป็น กลายเป็นสิ่งจำเป็นทันทีที่เรียก SharedPreferences ก่อน runApp — คนเขียนคนแรกอาจเคยตั้งใจทำแบบนี้

มีกับดักหนึ่งข้อ: AuthSessionService.clearSession() เรียก prefs.clear() ซึ่งลบทุกคีย์ — ธีมที่จำไว้จะหายตอนออกจากระบบด้วย เรื่องนี้อธิบายเต็ม ๆ ที่ 16.4

ข้อ 2 — สลับภาษาได้จริง

ลำดับทำอะไรขนาดงาน
1ทำ localeProvider แบบเดียวกับ themeProvider แล้วผูกกับ locale:เล็ก
2สร้างที่เก็บข้อความแบบ 11.8กลาง
3ย้ายข้อความไทยที่ฝังในโค้ดออกมาทีละจอใหญ่มาก
4ตรวจว่าคำอังกฤษไม่ล้นปุ่มที่ออกแบบด้วยความยาวคำไทยกลาง

ขั้นที่ 3 คืองานจริง — ข้อความไทยกระจายอยู่ใน 80 ไฟล์ และหลายอันอยู่กลาง build() ที่ยาวเป็นพันบรรทัด

ข้อสรุปที่สำคัญกว่าตัวคำตอบ: ขั้นที่ 1 ใช้เวลาไม่ถึงชั่วโมงและทำให้ดูเหมือน “ทำเสร็จแล้ว” แต่ถ้าหยุดแค่นั้น ผู้ใช้กดสลับภาษาแล้วทุกอย่างยังเป็นไทยหมด ซึ่งแย่กว่าไม่มีปุ่มเลย

ถ้าจะทำ ต้องทำทีละจอและปล่อยเมื่อครบทั้งจอ ไม่ใช่ปล่อยปุ่มก่อนแล้วค่อยตามแปล

9. สรุป

  • main.dart 39 บรรทัด: MaterialApp ห่อด้วย ValueListenableBuilder ที่ฟังธีม
  • ไม่มีตารางเส้นทาง ใช้ home: แล้ว Navigator.push ตรง ๆ ทุกที่
  • ประกาศรองรับสองภาษาแต่ตรึง locale ไว้ที่ไทย
  • themeProvider เป็นตัวแปรระดับไฟล์ — เรียกง่าย แต่ไม่มีที่ครอบและสลับตัวปลอมไม่ได้
  • ธีมที่เลือกไม่ถูกจำหลังปิดแอป
  • AppTheme เก็บทั้งค่าคงที่ ตัวช่วยที่รู้ธีม และตัวสร้าง ThemeData ในไฟล์เดียว
  • ระวังสามค่าท้ายไฟล์ที่เป็นของเก่าและไม่รู้ธีม
