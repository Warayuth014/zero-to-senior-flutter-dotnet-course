2. คอนโทรลเลอร์

enum AppLanguage { english, thai }

class AppLanguageController extends ChangeNotifier {
  AppLanguageController._();                                    // ①
  static final AppLanguageController instance = AppLanguageController._();
  static const _preferenceKey = 'app_language';

  AppLanguage _language = AppLanguage.english;                  // ②

  AppLanguage get language => _language;
  Locale get locale => Locale(_language == AppLanguage.thai ? 'th' : 'en');
  bool get isThai => _language == AppLanguage.thai;
  String get code => isThai ? 'TH' : 'EN';

  Future<void> load() async { ... }
  Future<void> setLanguage(AppLanguage value) async { ... }
  Future<void> toggle() =>
      setLanguage(isThai ? AppLanguage.english : AppLanguage.thai);

  String text(String english, String thai) => isThai ? thai : english;
}

3. load() และ setLanguage()

Future<void> load() async {
  try {
    final saved = await SharedPreferencesAsync().getString(_preferenceKey);
    _language = saved == 'th' ? AppLanguage.thai : AppLanguage.english;
  } on StateError {
    // Widget tests and unsupported platforms may not register the plugin.
    _language = AppLanguage.english;
  }
}

สังเกตว่า load() ไม่เรียก notifyListeners()

ถูกต้อง เพราะมันถูกเรียกใน main() ก่อน runApp() — ตอนนั้นยังไม่มีใครฟังอยู่ (20.2)

Future<void> setLanguage(AppLanguage value) async {
  if (_language == value) return;              // ①
  _language = value;
  notifyListeners();                           // ② แจ้งก่อน
  try {
    await SharedPreferencesAsync().setString(
      _preferenceKey,
      value == AppLanguage.thai ? 'th' : 'en',
    );
  } on StateError {
    // Language switching still works for the current session.
  }
}

8. แบบฝึกหัด

  1. เขียนเทสต์ที่ตรวจว่าจอแสดงข้อความไทยเมื่อสลับภาษา — และอธิบายว่าทำไมต้องคืนค่าเมื่อจบ
  2. ถ้าต้องเพิ่มภาษาที่สาม ให้วางแผนการย้ายที่เจ็บน้อยที่สุด
แนวคำตอบ

ข้อ 1

testWidgets('the sign-in button reads Thai once the language is switched',
    (tester) async {
  addTearDown(() => AppLanguageController.instance
      .setLanguage(AppLanguage.english));          // ← คืนค่า

  await AppLanguageController.instance.setLanguage(AppLanguage.thai);

  await tester.pumpWidget(MaterialApp(
    home: LoginScreen(auth: AuthRepository(ApiClient())),
  ));

  expect(find.text('เข้าสู่ระบบ'), findsWidgets);
});

addTearDown จำเป็นเพราะ controller เป็น singleton ที่มีชีวิตข้ามเทสต์

ถ้าไม่คืนค่า เทสต์ที่รันถัดไปจะเจอแอปเป็นภาษาไทย
อาการที่ตามมาคือเทสต์ตัวอื่นที่หา find.text('SIGN IN') จะแดง — และจะแดงเฉพาะเมื่อรันทั้งไฟล์ ไม่แดงเมื่อรันตัวเดียว
เป็นความล้มเหลวที่ขึ้นกับลำดับ ซึ่งเสียเวลาหาสาเหตุมากที่สุดประเภทหนึ่ง
กฎ: ทุกครั้งที่เทสต์แตะ state ระดับโลก ให้เขียน addTearDown ในบรรทัดถัดไปทันที ไม่ใช่ตอนท้าย

ข้อ 2 — เพิ่มภาษาที่สาม

ปัญหาคือ tr(english, thai) รับสองอาร์กิวเมนต์ตายตัว มีการเรียกหลายร้อยจุด

ทางวิธีประเมิน
Aเพิ่มพารามิเตอร์ที่สาม tr(en, th, [String? my])ทุกจุดที่ไม่ใส่จะตกกลับเป็นอังกฤษ — ย้ายทีละส่วนได้ แต่ค้างครึ่ง ๆ กลาง ๆ นาน
Bย้ายไป .arb ทั้งหมดทีเดียวสะอาดที่สุด แต่ต้องแก้ทุกจุดพร้อมกัน และชั้น core/ ต้องหาทางจัดการเรื่อง context
Cเก็บ tr() ไว้เป็นหน้ากาก แล้วเปลี่ยนข้างในให้ค้นจากตารางเจ็บน้อยที่สุด

ทาง C ทำได้เพราะอาร์กิวเมนต์อังกฤษใช้เป็นคีย์ได้ในตัว

String tr(String english, String thai) {
  final c = AppLanguageController.instance;
  if (c.language == AppLanguage.english) return english;
  if (c.language == AppLanguage.thai) return thai;
  return c.lookup(english) ?? english;   // ภาษาที่สามค้นจากตาราง
}
ได้เสีย
ไม่ต้องแก้จุดเรียกแม้แต่จุดเดียวคีย์คือประโยคอังกฤษ — แก้คำเดียวแล้วคำแปลหลุด
ทำงานได้ทันทีแม้ตารางยังไม่ครบข้อความซ้ำที่ความหมายต่างกันจะแปลแยกไม่ได้
ชั้น core/ ไม่ต้องแตะยังส่งให้นักแปลได้ยากอยู่ดี
ข้อเสียข้อแรกจัดการได้ — เขียนสคริปต์ที่กวาด tr('...', '...') ทั้งโปรเจกต์แล้วสร้างตารางคีย์อัตโนมัติ
ถ้าคีย์ในตารางกับในโค้ดไม่ตรง ให้ CI แดง
เมื่อทำแบบนี้ ก็ได้ประโยชน์ของ .arb (ตรวจความครบได้) โดยไม่ต้องแก้จุดเรียกเลย
ทางนี้เป็นที่มาของคำแนะนำทั่วไปที่ว่า ให้หุ้มไลบรารีภายนอกไว้หลังฟังก์ชันของตัวเอง — tr() ทำหน้าที่นั้นอยู่แล้วโดยบังเอิญ ซึ่งเป็นเหตุผลที่การย้ายในอนาคตจะไม่แย่อย่างที่บทที่ 4 เตือนไว้

9. สรุป

  • ระบบสองภาษาทั้งหมดคือ tr(english, thai) — ไม่มีไฟล์แปล ไม่มีการ generate
  • ข้อความในแอปแปล แต่ข้อมูลจากเซิร์ฟเวอร์ไม่แปล เพราะต้องตรงกับป้ายที่ชั้นวาง
  • setLanguage แจ้งก่อนบันทึก — optimistic update ที่ปลอดภัยเพราะความล้มเหลวราคาถูก
  • load() ไม่แจ้ง เพราะรันก่อน runApp()
  • กลืน StateError พร้อมคอมเมนต์ — ชนิดเจาะจง เข้าใจสาเหตุ ทำงานต่อได้ถูก
  • tr() ทำงานได้โดยไม่มี BuildContext จึงใช้ในชั้น core/ ได้
  • ยังต้องมี localizationsDelegates เพราะข้อความของ Flutter เองไม่ผ่าน tr()
  • LanguageButton ไม่ได้ฟัง controller — พึ่ง ListenableBuilder ที่ระดับ MaterialApp
  • singleton ตัวนี้ไม่มี debugReset() — เทสต์ที่แตะภาษาต้อง addTearDown เอง
