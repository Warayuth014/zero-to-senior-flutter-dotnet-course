1. session เดินทางไปไหนบ้าง

LoginScreen._submit()
  └─ auth.login(id, pw)
       ├─ _api.authToken = token          ← ①  token ผูกกับ client
       └─ _session = session              ← ②  เก็บไว้ที่ repository

  └─ pushReplacement(WarehouseShell(auth: widget.auth))

WarehouseShell.initState()
  └─ auth.session!.access                 ← ③  หยิบสิทธิ์
  └─ auth.session!.username               ←     หยิบชื่อ
  └─ WarehouseStore(username: ..., access: ...)

HomeView
  └─ store.access.canView(PdaFeature.picking)   ← ④ ตัดสินว่าวาดการ์ดไหม

UserChipMenu → logout
  └─ auth.logout()                        ← ⑤ ล้าง token + session
  └─ pushReplacement(LoginScreen)

สังเกตข้อ ② และ ③ — จอไม่ได้รับ AuthSession ตรง ๆ แต่รับ AuthRepository แล้วไปหยิบ session จากในนั้น

2. PdaFeature — enum ที่พก key

enum PdaFeature {
  receiving('/mobile/pda/receiving'),

รูปแบบ enhanced enum (D1.17, 6.5) — สมาชิกแต่ละตัวพกข้อมูลติดตัว

ฝั่ง Dartฝั่ง JSON จากเซิร์ฟเวอร์
PdaFeature.picking"/mobile/pda/picking"

ทำไม key เป็นเส้นทาง URL

เพราะฝั่ง WMS เว็บใช้เส้นทางเป็นตัวระบุสิทธิ์ของเมนูอยู่แล้ว — พีดีเอยืมระบบเดิมมาใช้ ไม่ได้สร้างชื่อชุดใหม่

ข้อดี — ผู้ดูแลตั้งสิทธิ์ในหน้าจัดการเดียวกับที่ใช้ตั้งสิทธิ์เว็บ ไม่ต้องเรียนคำศัพท์ชุดที่สอง
ข้อเสีย — คำว่า “เส้นทาง” ทำให้เข้าใจผิดได้ว่าพีดีเอยิงไปที่ URL นี้ ทั้งที่ไม่เคยยิง มันเป็นแค่สตริงที่ใช้เป็นกุญแจ
คอมเมนต์ในไฟล์เรียกมันว่า “stable permission key” ซึ่งเป็นคำที่ตรงกว่า

ทำไมไม่ใช้ name ของ enum

PdaFeature.picking.name   // → "picking"

ใช้ได้ในทางเทคนิค แต่จะผูกชื่อฟิลด์ใน Dart เข้ากับข้อตกลงกับเซิร์ฟเวอร์ — เปลี่ยนชื่อตัวแปรใน Dart แล้วสิทธิ์พังทันที โดยที่คอมไพเลอร์ไม่เตือน

การประกาศ key แยกทำให้เปลี่ยนชื่อใน Dart ได้อย่างปลอดภัย

3. PdaAccessLevel — ระดับที่เรียงลำดับได้

enum PdaAccessLevel {
  none,
  view,
  operate,
  fullControl;

  static PdaAccessLevel parse(Object? value) =>
      switch (value?.toString().trim().toLowerCase()) {
        'view' => PdaAccessLevel.view,
        'operate' => PdaAccessLevel.operate,
        'fullcontrol' || 'full-control' => PdaAccessLevel.fullControl,
        _ => PdaAccessLevel.none,
      };
}
ระดับindexทำอะไรได้
none0ไม่เห็นเมนูเลย
view1เปิดดูได้ แต่กดยืนยันไม่ได้
operate2ทำงานปกติได้
fullControl3ทำสิ่งที่ต้องระดับหัวหน้าได้

parse ที่ไม่มีวันโยน

ค่าที่ไม่รู้จักตกเข้า _ => none — ไม่มีทางล้ม

การเลือกค่าตกถังให้เป็น none ไม่ใช่ view คือการเลือกที่ปลอดภัย
ถ้าเซิร์ฟเวอร์ส่งค่าใหม่ที่แอปรุ่นเก่ายังไม่รู้จัก แอปจะปิดฟีเจอร์ไว้ก่อน ไม่ใช่เปิดให้
ผู้ใช้อาจงงว่าเมนูหาย แต่ดีกว่าให้คนที่ไม่มีสิทธิ์เข้าไปยืนยันงานได้ — เรียกว่า fail closed

รูปแบบ || ใน pattern

'fullcontrol' || 'full-control' => PdaAccessLevel.fullControl,

สองสตริงนำไปสู่ผลเดียวกัน (Dart 3, D1.19) — รับได้ทั้งรูปแบบที่มีขีดและไม่มี

บวกกับ .trim().toLowerCase() ข้างหน้า ทำให้ "FullControl", " full-control ", "FULLCONTROL" ได้ผลเหมือนกันหมด

ความยืดหยุ่นนี้จำเป็นเมื่อสองระบบพัฒนากันคนละที — เหมือนกับที่ _decode ใน 20.4 ลองอ่านชื่อฟิลด์สามแบบ
ทางที่สะอาดกว่าคือตกลงรูปแบบเดียวให้แน่ แล้วลดเงื่อนไขลง — แต่ตราบใดที่ยังไม่ได้ตกลง การรับหลายรูปดีกว่าปิดฟีเจอร์ให้ผู้ใช้เพราะตัวพิมพ์ใหญ่เล็ก

4. PdaAccess

class PdaAccess {
  const PdaAccess(this._levels);

  factory PdaAccess.fromJson(Object? value) {
    final raw = value is Map ? value : const <Object?, Object?>{};
    return PdaAccess({
      for (final feature in PdaFeature.values)
        feature: PdaAccessLevel.parse(raw[feature.key]),
    });
  }

  final Map<PdaFeature, PdaAccessLevel> _levels;

  PdaAccessLevel level(PdaFeature feature) =>
      _levels[feature] ?? PdaAccessLevel.none;

  bool canView(PdaFeature feature) => level(feature) != PdaAccessLevel.none;

  bool canOperate(PdaFeature feature) =>
      level(feature).index >= PdaAccessLevel.operate.index;

  bool hasFullControl(PdaFeature feature) =>
      level(feature) == PdaAccessLevel.fullControl;

  Map<PdaFeature, PdaAccessLevel> get levels => Map.unmodifiable(_levels);
}

fromJson วนจาก enum ไม่ใช่จาก JSON

for (final feature in PdaFeature.values)
  feature: PdaAccessLevel.parse(raw[feature.key]),

ทิศทางของการวนสำคัญมาก

วนจากผลเมื่อ JSON ขาดฟีเจอร์ผลเมื่อ JSON มีคีย์แปลกปลอม
JSONฟีเจอร์นั้นหายจาก map — ต้องจัดการ null ทุกที่เข้ามาปนใน map
enum (ที่ทำจริง)ได้ none — map ครบทุกฟีเจอร์เสมอถูกเมินเงียบ ๆ
ผลลัพธ์คือ _levels มีครบ 8 คีย์เสมอ ไม่ว่าเซิร์ฟเวอร์จะส่งอะไรมา
รวมถึงกรณีส่ง null มาทั้งก้อน (บรรทัด value is Map ? value : {})
เป็นรูปแบบเดียวกับ 4.14 — ทำให้สถานะที่ไม่ครบสร้างไม่ได้ แทนที่จะไปตรวจทีหลัง

สาม getter ที่ต่างกันตรงตัวเปรียบเทียบ

canView        →  != none            (มากกว่า 0)
canOperate     →  index >= operate    (2 หรือ 3)
hasFullControl →  == fullControl      (3 เท่านั้น)

การใช้ index ได้ผลเพราะ enum เรียงจากน้อยไปมากโดยตั้งใจ

ข้อควรระวังของการพึ่ง index
ถ้ามีคนแทรกสมาชิกใหม่ผิดตำแหน่ง เช่น
enum PdaAccessLevel { none, view, audit, operate, fullControl }
//                                    ↑ แทรกตรงนี้ปลอดภัย
แต่ถ้าแทรกเป็น { none, operate, view, fullControl } ตรรกะจะกลับด้านทันทีโดยไม่มีอะไรเตือน
ลำดับของ enum นี้จึงเป็นข้อตกลง ไม่ใช่แค่การจัดเรียง — ควรมีคอมเมนต์กำกับว่าห้ามสลับ และมีเทสต์ตรึงไว้
test('access levels stay ordered from least to most', () {
  expect(PdaAccessLevel.values, [
    PdaAccessLevel.none, PdaAccessLevel.view,
    PdaAccessLevel.operate, PdaAccessLevel.fullControl,
  ]);
});

Map.unmodifiable

Map<PdaFeature, PdaAccessLevel> get levels => Map.unmodifiable(_levels);

ให้ข้างนอกอ่านได้ (เมนูผู้ใช้เอาไปแสดงรายการสิทธิ์) แต่แก้ไม่ได้ — levels[x] = y จะโยนตอนรัน

ค่าใช้จ่ายคือสร้าง wrapper ใหม่ทุกครั้งที่อ่าน getter ซึ่งไม่มีผลกับการเรียกไม่กี่ครั้ง

const PdaAccess.unrestricted()

/// Used by isolated widget/store tests that do not exercise authentication.
const PdaAccess.unrestricted()
  : _levels = const {
      PdaFeature.receiving: PdaAccessLevel.fullControl,
      ... ครบทั้ง 8 ตัว
    };

ตัวสร้างที่มีไว้เพื่อเทสต์โดยเฉพาะ พร้อมคอมเมนต์บอกตรง ๆ

WarehouseStore({
  ...
  this.access = const PdaAccess.unrestricted(),   // ค่าปริยาย
});
ทำไมค่าปริยายเป็น “เปิดหมด” ไม่ใช่ “ปิดหมด”
ดูขัดกับหลัก fail closed ในบทที่ 3 — แต่บริบทต่างกัน
  • parse จัดการข้อมูลจากเซิร์ฟเวอร์ ซึ่งไม่น่าเชื่อถือ → ปิดไว้ก่อน
  • ค่าปริยายของ WarehouseStore จัดการโค้ดที่โปรแกรมเมอร์เขียนเอง → ถ้าปิดหมด เทสต์ทุกตัวที่ไม่สนใจสิทธิ์จะต้องเขียนสิทธิ์เต็มก่อน
ความเสี่ยงคือถ้ามีใครลืมส่ง access ในโค้ดจริง สิทธิ์จะเปิดหมด — ที่นี่มีที่สร้าง store จริงแค่ที่เดียวคือ warehouse_shell.dart จึงตรวจสอบได้ง่าย

'home only builds feature cards with View access'

เทสต์นี้ไม่ได้ทดสอบ PdaAccess ตรง ๆ แต่ทดสอบผลที่ผู้ใช้เห็น — สร้าง store ที่มีสิทธิ์จำกัด แล้วยืนยันว่าการ์ดที่ไม่มีสิทธิ์ไม่ถูกวาด

6. สิ่งที่โมเดลสิทธิ์นี้ยังไม่ได้ทำ

เรื่องสถานะ
ซ่อนเมนูตามสิทธิ์ทำแล้ว มีเทสต์
ปิดปุ่มยืนยันเมื่อมีแค่ viewทำบางจอ
เซิร์ฟเวอร์ตรวจซ้ำอยู่นอกขอบเขตของแอป
ข้อสุดท้ายสำคัญที่สุดและมักถูกลืม
การซ่อนเมนูในแอปคือความสะดวก ไม่ใช่ความปลอดภัย
ใครก็ตามที่มี token สามารถยิงเส้นทางที่ถูกซ่อนได้โดยตรงด้วยเครื่องมือทั่วไป
ความปลอดภัยจริงต้องอยู่ที่ .NET — endpoint ต้องตรวจสิทธิ์เองทุกเส้น ไม่ว่าแอปจะซ่อนอะไรไว้
เมื่ออ่านโค้ดสิทธิ์ฝั่งแอป ให้ถามเสมอว่า “แล้วฝั่งเซิร์ฟเวอร์ตรวจด้วยหรือเปล่า”

7. ข้อผิดพลาดที่พบบ่อย

ผิดอาการถูก
ตรวจชื่อบทบาทในแอปเพิ่มบทบาทแล้วต้อง build ใหม่รับสิทธิ์ที่คำนวณแล้วจากเซิร์ฟเวอร์
ใช้ enum.name เป็นคีย์กับเซิร์ฟเวอร์rename ใน Dart แล้วสิทธิ์พังเงียบประกาศ key แยก
วน fromJson จาก JSONฟีเจอร์ที่ขาดกลายเป็น nullวนจาก values
ค่าไม่รู้จัก → เปิดสิทธิ์คนไม่มีสิทธิ์ทำงานได้ตกถังเป็น none
คืน _levels ตรง ๆข้างนอกแก้สิทธิ์ได้Map.unmodifiable
คิดว่าซ่อนเมนู = ปลอดภัยยิงตรงยังทำได้เซิร์ฟเวอร์ต้องตรวจ

ข้อ 2 น่าสนใจ — unrestricted() เป็น const ที่ระบุครบทั้ง 8 ตัว ถ้าเพิ่ม enum แล้วไม่เพิ่มตรงนี้ จะยังคอมไพล์ผ่าน เพราะ map ไม่ต้องครบ

แต่จะเจอตอนรันเทสต์: level() คืน none ฟีเจอร์ใหม่จึงไม่โผล่ใน widget test ที่ควรเห็น

ทางแก้ที่ดูน่าจะได้คือให้มันสร้างรายการเอง

PdaAccess.unrestricted()
  : _levels = const {
      for (final f in PdaFeature.values) f: PdaAccessLevel.fullControl,
    };   // ← คอมไพล์ไม่ผ่าน
ใช้ไม่ได้ — Dart อนุญาต if ใน const collection แต่ไม่อนุญาต for

เปลี่ยนเป็น factory ธรรมดาแล้วใช้ forครบอัตโนมัติไม่เป็น const อีกต่อไป จึงใช้เป็นค่าปริยายของพารามิเตอร์ไม่ได้
คงรายการ 8 บรรทัดไว้ แล้วเพิ่มเทสต์ตรึงยัง constต้องมีเทสต์คอยจับ

ค่าปริยายของ WarehouseStore เขียนว่า this.access = const PdaAccess.unrestricted() ซึ่งบังคับให้ต้องเป็น const — ทางที่สองจึงเป็นทางเดียวที่ไม่ต้องรื้อ

test('unrestricted really covers every feature', () {
  const access = PdaAccess.unrestricted();
  for (final feature in PdaFeature.values) {
    expect(access.hasFullControl(feature), isTrue, reason: '${feature.name}');
  }
});
บทเรียนยังเหมือนเดิม แม้ทางแก้จะต่างจากที่คิดไว้ตอนแรก
รายการที่ต้อง “อย่าลืมเติมให้ครบ” คือรายการที่วันหนึ่งจะไม่ครบ
ถ้าทำให้มันสร้างตัวเองไม่ได้ ก็ต้องมีอะไรมาจับแทน — เทสต์สี่บรรทัดถูกกว่าการไล่หาว่าทำไมเมนูใหม่ไม่โผล่ในเทสต์

9. สรุป

  • จอรับ AuthRepository ไม่ใช่ AuthSession — จึงล็อกเอาต์และสร้างชั้น API ได้จากตัวเดียวกัน
  • PdaFeature พก key แยกจากชื่อตัวแปร — rename ใน Dart ได้โดยไม่กระทบสิทธิ์
  • PdaAccessLevel เรียงจากน้อยไปมากโดยตั้งใจ canOperate จึงใช้ index เทียบได้
  • ลำดับของ enum เป็นข้อตกลง ควรมีเทสต์ตรึงไว้
  • parse ตกถังเป็น none — fail closed สำหรับข้อมูลจากเซิร์ฟเวอร์
  • fromJson วนจาก PdaFeature.values — map ครบ 8 คีย์เสมอ
  • unrestricted() เปิดหมดเป็นค่าปริยายของ store เพราะเป็นเรื่องของโค้ดที่เราคุมเอง ไม่ใช่ข้อมูลภายนอก
  • เทสต์ทดสอบที่ระดับ “การ์ดไม่โผล่” ไม่ใช่ระดับ “getter คืน false”
  • การซ่อนเมนูคือความสะดวก ความปลอดภัยต้องอยู่ที่ .NET
