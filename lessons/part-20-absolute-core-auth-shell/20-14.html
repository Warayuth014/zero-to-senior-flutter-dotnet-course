2. initState — จุดที่ทุกชั้นมาบรรจบ

@override
void initState() {
  super.initState();
  WidgetsBinding.instance.addObserver(this);
  store = WarehouseStore(
        username: widget.auth.session?.username ?? '0000',
        receivingRepository: ReceivingApi(widget.auth.client),
        tasksRepository: TasksApi(widget.auth.client),
        inventoryRepository: InventoryApi(widget.auth.client),
        cycleCountRepository: CycleCountApi(widget.auth.client),
        access: widget.auth.session?.access ?? const PdaAccess({}),
      )
      ..onToast = _showToast
      ..onFlash = _showFlash;
  unawaited(store.loadAll()); // badges must be right on the first screen
  // One probe on entry so the header dot reflects the server rather than a
  // guess. No repeating timer — an all-day handheld should not spend radio
  // on a status nobody is looking at.
  unawaited(ConnectionController.instance.refreshStatus());
}

สี่ repository ที่ใช้ client ตัวเดียวกัน

ReceivingApi(widget.auth.client)
TasksApi(widget.auth.client)
InventoryApi(widget.auth.client)
CycleCountApi(widget.auth.client)

เป็นผลของ client getter ใน 20.5 — ทั้งสี่ตัวมี token เดียวกัน และการสลับเซิร์ฟเวอร์กระทบทั้งสี่พร้อมกัน

const PdaAccess({}) เป็นทางถอย

access: widget.auth.session?.access ?? const PdaAccess({}),
สังเกตว่าไม่ได้ใช้ PdaAccess.unrestricted()
PdaAccess({}) คือ map ว่าง ซึ่ง level() จะคืน none ทุกฟีเจอร์ (20.6)
ถ้าเข้ามาโดยไม่มี session — ซึ่งไม่ควรเกิด — ผู้ใช้จะไม่เห็นเมนูอะไรเลย แทนที่จะเห็นทุกเมนู
เป็น fail closed ที่ถูกต้อง และตรงข้ามกับค่าปริยายของ store ที่เป็น unrestricted() — เพราะที่นี่คือโค้ดจริง ไม่ใช่เทสต์

onToast / onFlash — store พูดกับจอผ่าน callback

WarehouseStore(...)
  ..onToast = _showToast
  ..onFlash = _showFlash;
ทางเลือกปัญหา
store เรียก ScaffoldMessenger เองต้องมี BuildContext — store จะผูกกับ UI
store เก็บข้อความไว้ให้จอมาอ่านต้องจัดการว่าอ่านแล้วหรือยัง
callback ที่จอเสียบให้store ไม่รู้จัก Flutter เลย

รูปแบบนี้ทำให้ WarehouseStore ทดสอบได้โดยไม่ต้องมี widget — เทสต์เสียบ callback ที่เก็บข้อความลงลิสต์แล้วตรวจได้

final toasts = <String>[];
store.onToast = (msg, {bool err = false}) => toasts.add(msg);

unawaited สองครั้ง

initState เป็น void จึง await ไม่ได้ — unawaited ประกาศว่า “ตั้งใจไม่รอ” ไม่ใช่ลืม

ผลคือจอวาดทันทีด้วยข้อมูลว่าง แล้วเติมเมื่อข้อมูลมา — ถูกต้องสำหรับหน้าแรกที่ต้องขึ้นเร็ว

3. ตรวจสถานะสองจังหวะ ไม่มีตัวจับเวลา

/// The dot goes stale while the app is in the background — the handheld can
/// leave Wi-Fi range, or the server can be restarted, and nothing tells the
/// app. Re-checking on the way back is the one moment a fresh answer is both
/// cheap and about to be looked at.
@override
void didChangeAppLifecycleState(AppLifecycleState state) {
  if (state == AppLifecycleState.resumed) {
    unawaited(ConnectionController.instance.refreshStatus());
  }
}
จังหวะทำไมคำตอบมีค่าตอนนั้น
เข้าแอปผู้ใช้กำลังจะเริ่มทำงาน
กลับมาจากพักอาจออกนอกพื้นที่สัญญาณไปแล้ว
หลังสลับโปรไฟล์ (20.13)เปลี่ยนปลายทาง
เทียบกับตัวจับเวลาที่ store มี
// warehouse_store.dart
_taskRefreshTimer = Timer.periodic(
  const Duration(seconds: 5),
  (_) => _refreshTaskStatus(),
);
สองการตัดสินใจที่ตรงข้ามกันในโปรเจกต์เดียว และทั้งคู่ถูก
สถานะเซิร์ฟเวอร์สถานะงาน
เปลี่ยนเองโดยไม่มีใครทำอะไรไหมแทบไม่เปลี่ยน — คนอื่นหยิบงาน หุ่นยนต์เดินถึง
ข้อมูลเก่าทำให้ทำงานผิดไหมไม่ (แค่จุดสี)ใช่ — เดินไปหยิบของที่คนอื่นหยิบไปแล้ว
เกณฑ์: ยิงเป็นรอบเมื่อข้อมูลเปลี่ยนโดยไม่มีใครบอก และการรู้ช้าทำให้ทำงานผิด

เทียบกับ wmsapp ที่ใช้ SignalR (16.8) — วิธีนั้นดีกว่าในทางทฤษฎี (รู้ทันทีและไม่ยิงเปล่า) แต่แลกมาด้วยไลบรารีเพิ่มและข้อบกพร่องห้าข้อที่ 19.12 นับไว้ รวมถึงการไม่มี onreconnected

4. การนำทางที่ไม่มี push

  if (!store.canViewScreen(store.screen)) {

ทั้งแอปมี route แค่สองอัน: LoginScreen และ WarehouseShell — สิบสองจอที่เหลือคือค่าของตัวแปร store.screen

Navigator.pushสลับด้วยตัวแปร
แถบบน/ล่างต้องวาดใหม่ทุกจอวาดครั้งเดียว อยู่ตลอด
สถานะของงานที่ค้างอยู่ใน State ของแต่ละจอ — หายเมื่อ popอยู่ใน store — ไม่หาย
ปุ่มย้อนกลับของเครื่องทำงานเองต้องจัดการเอง
ตรวจสิทธิ์ต้องเช็คทุกจุดที่ pushเช็คที่เดียวใน _body
แถวที่สองคือเหตุผลหลัก
งานในคลังเป็นขั้นตอนหลายสเต็ปที่ต้องพักได้ — คนงานอาจต้องไปช่วยงานอื่นกลางคัน
ถ้าแต่ละสเต็ปเป็น route ที่มี State ของตัวเอง การพักงานแปลว่าสถานะหาย
เมื่อสถานะอยู่ใน store ทั้งหมด การพักคือแค่เปลี่ยนค่า screen แล้วกลับมาทีหลัง — ซึ่งเป็นสิ่งที่ปุ่ม “พักงานชั่วคราว” ทำ
18.16 ชี้ว่า wmsapp กระจายสถานะไว้ในแต่ละจอ ทำให้ทำแบบนี้ไม่ได้

แถวที่สามคือราคาที่จ่าย — และเป็นข้อจำกัดที่ยังไม่ถูกแก้

ไม่มี PopScope ที่ระดับเปลือก
กดปุ่มย้อนกลับของ Android ระหว่างอยู่ในขั้นตอนกลางคัน จะ pop WarehouseShell ทั้งอัน
ผลคือกลับไปหน้าล็อกอิน — และเนื่องจาก _logout() ไม่ได้ถูกเรียก session จึงยังอยู่ แต่ WarehouseShell ถูก dispose ไปแล้ว พร้อม store ทั้งตัว
งานที่ค้างอยู่ในหน่วยความจำหายทั้งหมด
ทางแก้ตรงไปตรงมา: ครอบ Scaffold ด้วย PopScope(canPop: store.screen == 'home', ...) แล้วให้กลับไปจอก่อนหน้าแทน — ซึ่งเป็นตรรกะที่ store.goBack มีอยู่แล้ว

7. ข้อความชั่วคราวและแฟลชสี

void _showFlash(bool ok) {
  if (ok) {
    unawaited(HapticFeedback.mediumImpact());
  } else {
    unawaited(HapticFeedback.heavyImpact());
  }
  setState(() => _flashColor = ok ? const Color(0x5228A745) : const Color(0x59E04B3F));
  _flashTimer?.cancel();
  _flashTimer = Timer(const Duration(milliseconds: 220), () {
    if (mounted) setState(() => _flashColor = null);
  });
}

ทั้งจอกลายเป็นสีเขียวหรือแดงจาง ๆ 220 มิลลิวินาที พร้อมการสั่นที่ต่างกัน

ออกแบบมาสำหรับสภาพจริงในคลัง
ช่องทางใช้ได้เมื่อ
สีทั้งจอเห็นด้วยหางตาโดยไม่ต้องอ่าน
การสั่นในที่เสียงดัง และเมื่อจออยู่ในมือ
ข้อความเมื่อต้องการรายละเอียด
คนงานที่สแกนติดกันสิบชิ้นไม่ได้อ่านจอทุกครั้ง — เขาสแกนแล้วรอสัญญาณ
ความต่างระหว่าง mediumImpact กับ heavyImpact ทำให้แยกสำเร็จกับล้มเหลวได้โดยไม่ต้องมอง

สังเกต IgnorePointer รอบชั้นสี — ไม่งั้นแฟลชจะกินการกดของผู้ใช้ในช่วง 220 มิลลิวินาทีนั้น ซึ่งเป็นช่วงที่คนสแกนเร็ว ๆ กำลังจะกดพอดี

และ _flashTimer?.cancel() ก่อนตั้งใหม่ทุกครั้ง — สแกนรัวแล้วตัวจับเวลาไม่ทับกัน

ทั้งสองตัวจับเวลาถูกยกเลิกใน dispose

@override
void dispose() {
  WidgetsBinding.instance.removeObserver(this);
  _toastTimer?.cancel();
  _flashTimer?.cancel();
  store.dispose();
  super.dispose();
}

ครบทั้งสี่อย่าง รวม store.dispose() ที่จะไปยกเลิกตัวจับเวลา 5 วินาทีข้างใน

8. ข้อผิดพลาดที่พบบ่อย

ผิดอาการถูก
ให้ store เรียก ScaffoldMessenger เองstore ผูกกับ UI ทดสอบไม่ได้callback ที่จอเสียบให้
ทางถอยของสิทธิ์เป็น “เปิดหมด”คนไม่มีสิทธิ์เห็นทุกเมนูconst PdaAccess({})
ตรวจสถานะเซิร์ฟเวอร์เป็นรอบแบตหมดก่อนจบกะตรวจตอนเข้าและตอนกลับมา
ไม่ยิงสถานะงานเป็นรอบหยิบของที่คนอื่นหยิบไปแล้วตัวจับเวลาใน store
สลับจอด้วยตัวแปรแต่ไม่จัดการปุ่มย้อนกลับกดแล้วหลุดออกทั้งเปลือก งานหายPopScope ที่เปลือก
ไม่ cancel ตัวจับเวลาก่อนตั้งใหม่สแกนรัวแล้วแฟลชค้าง?.cancel() ทุกครั้ง
ชั้นสีทับโดยไม่มี IgnorePointerกดไม่ติดช่วงสั้น ๆครอบไว้
ตรวจสิทธิ์สองที่ด้วยเงื่อนไขคนละชุดเมนูซ่อนแต่เข้าได้ หรือกลับกันเรียกฟังก์ชันเดียวกัน

9. แบบฝึกหัด

  1. เขียน PopScope ที่แก้ปัญหาปุ่มย้อนกลับ พร้อมบอกว่าต้องระวังอะไร
  2. ผู้ใช้พักหน้าจอตอนอยู่กลางขั้นตอนรับสินค้า แล้วกลับมาอีกสิบนาที — ไล่ว่าเกิดอะไรขึ้นบ้าง
แนวคำตอบ

ข้อ 1

@override
Widget build(BuildContext context) {
  return PopScope(
    canPop: store.screen == 'home' && !store.inFlow,
    onPopInvokedWithResult: (didPop, _) {
      if (didPop) return;
      store.goBack();
    },
    child: Scaffold(...),
  );
}
ต้องระวังเหตุผล
canPop ต้องอ่านจาก storeต้องอยู่ใน ListenableBuilder ไม่งั้นค่าไม่อัปเดตเมื่อเปลี่ยนจอ
ที่หน้าหลักควรออกจากแอปได้ไม่งั้นผู้ใช้ติดอยู่ในแอป
ระหว่างอยู่ในขั้นตอนควรถามก่อน เหมือนปุ่ม ✕ ที่เรียก _confirmExit

เวอร์ชันที่คิดครบกว่า

onPopInvokedWithResult: (didPop, _) {
  if (didPop) return;
  if (store.inFlow) {
    unawaited(_confirmExit());   // ใช้ตัวเดิม ไม่เขียนกล่องใหม่
  } else {
    store.goBack();
  }
},
การเรียก _confirmExit() ที่มีอยู่แล้ว สำคัญกว่าที่ดู
ถ้าเขียนกล่องยืนยันใหม่ จะได้ข้อความสองชุดสำหรับการกระทำเดียวกัน แล้ววันหนึ่งจะแก้แค่ชุดเดียว
นี่คือ 19.15 แถวที่ 8 (“แจ้งความสำเร็จ”) ในรูปแบบเดียวกันเป๊ะ — และหลีกเลี่ยงได้ด้วยการมองหาของที่มีอยู่ก่อนเขียนใหม่

ข้อ 2 — พักหน้าจอสิบนาที

พักหน้าจอ
  didChangeAppLifecycleState(paused)   → ไม่ได้จัดการ ไม่มีอะไรเกิด
  Timer.periodic ใน store              → Android อาจชะลอหรือหยุดชั่วคราว
  state ทั้งหมดยังอยู่ใน store         → ไม่หาย

กลับมา
  didChangeAppLifecycleState(resumed)
    → refreshStatus() → จุดสถานะถูกต้อง
  Timer เดินต่อ → _refreshTaskStatus() ภายใน 5 วินาที
  จอวาดด้วยข้อมูลใน store (อาจเก่าสิบนาที) ก่อนข้อมูลใหม่มาถึง
สิ่งที่ถูกสิ่งที่ยังขาด
จุดสถานะสดทันทีข้อมูลงานยังเก่าอยู่ไม่เกิน 5 วินาที
งานที่ค้างไม่หายไม่มีการเรียก store.loadAll() ตอนกลับมา
token ยังอยู่ถ้า token หมดอายุระหว่างพัก จะรู้ตอนยิงครั้งถัดไปเป็น 401 (20.5)
ช่องว่างที่แท้จริงคือแถวที่สอง
ผู้ใช้กลับมาเห็นตัวเลขตราแจ้งของสิบนาทีที่แล้วเป็นเวลาสั้น ๆ
ในคลังที่คนหลายคนหยิบงานจากคิวเดียวกัน สิบนาทีคือระยะที่งานเปลี่ยนมือได้หลายใบ
เติมได้บรรทัดเดียว
if (state == AppLifecycleState.resumed) {
  unawaited(ConnectionController.instance.refreshStatus());
  unawaited(store.loadAll());          // ← เติม
}
แต่ต้องตรวจก่อนว่า loadAll() ปลอดภัยที่จะเรียกซ้ำระหว่างที่ผู้ใช้อยู่กลางขั้นตอน — ถ้ามันรีเซ็ตอะไรที่กำลังกรอกอยู่ จะแก้ปัญหาหนึ่งแล้วสร้างอีกปัญหา
เป็นคำถามที่ต้องไปอ่าน WarehouseStore.loadAll() ก่อนตอบ ซึ่งเป็นเนื้อหาของ Part 21

10. สรุป

  • เปลือก 742 บรรทัดที่ไม่มีตรรกะธุรกิจ — ทั้งหมดอยู่ใน store
  • initState สร้าง repository ทั้งสี่จาก auth.client ตัวเดียว
  • ทางถอยของสิทธิ์เป็น PdaAccess({}) ที่ปิดทุกเมนู ไม่ใช่ unrestricted()
  • store พูดกับจอผ่าน onToast/onFlash จึงไม่รู้จัก Flutter
  • สถานะเซิร์ฟเวอร์ตรวจสองจังหวะ ส่วนสถานะงานยิงทุก 5 วินาที — เกณฑ์คือข้อมูลเปลี่ยนเองไหมและรู้ช้าแล้วเสียหายไหม
  • สิบสองจอเป็นค่าของ store.screen ไม่ใช่ route — งานที่ค้างจึงไม่หายเมื่อสลับจอ
  • ยังไม่มี PopScope ที่เปลือก — ปุ่มย้อนกลับของเครื่องทำให้หลุดออกทั้งอัน
  • สิทธิ์ตรวจสองที่แต่ผ่านฟังก์ชันเดียวกัน จึงไม่มีทางไม่ตรงกัน
  • แฟลชสี + การสั่นสองระดับ สำหรับคนที่ไม่ได้มองจอทุกครั้งที่สแกน
  • dispose ครบทั้ง observer ตัวจับเวลาสองตัว และ store
