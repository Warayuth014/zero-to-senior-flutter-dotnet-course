2. ApiException

/// Raised when the backend answers with a non-success status or is unreachable.
class ApiException implements Exception {
  ApiException(this.message, {this.statusCode});

  final String message;
  final int? statusCode;

  @override
  String toString() => message;
}

เก้าบรรทัด และทุกบรรทัดมีเหตุผล

implements Exception ไม่ใช่ extends

Exception ใน Dart เป็น interface ที่ไม่มีอะไรเลย — การ implements เป็นการประกาศเจตนาว่า “ของชิ้นนี้ตั้งใจให้ throw”

statusCode เป็น int?

สถานการณ์statusCode
เซิร์ฟเวอร์ตอบ 401401
เซิร์ฟเวอร์ตอบ 500500
ต่อไม่ติดnull
หมดเวลารอnull

null จึงมีความหมายชัดเจน: คำขอไปไม่ถึงเซิร์ฟเวอร์ หรือไม่รู้ว่าถึงหรือเปล่า

19.12 เจอปัญหานี้ใน wmsapp — จอไม่รู้ว่าคำสั่งไปถึงหรือไม่ จึงขึ้น “ส่งไม่สำเร็จ” ทั้งที่งานเดินไปแล้ว
การมี statusCode เป็น null แยกจาก 500 คือขั้นแรกของการแก้ปัญหานั้น

toString() คืน message เฉย ๆ

// ถ้าไม่ override
'$error'   // → "Instance of 'ApiException'"

// เมื่อ override
'$error'   // → "เชื่อมต่อเซิร์ฟเวอร์ไม่ได้ ตรวจสอบสัญญาณหรือค่าเซิร์ฟเวอร์"

ทำให้จอที่เขียน catch (e) { showToast('$e'); } ได้ข้อความที่อ่านรู้เรื่อง แม้จะเขียนแบบขี้เกียจ

นี่คือการออกแบบให้ “ทางที่ง่ายที่สุด” เป็นทางที่ถูก
ถ้า toString() ให้ผลแย่ คนก็จะเขียน e is ApiException ? e.message : '...' ทุกที่ แล้วมีที่ที่ลืม
เมื่อ toString() ให้ผลดี การเขียนแบบสั้นที่สุดก็ถูกอยู่แล้ว

3. ตัวสร้างที่ฉีดของทดสอบได้

ApiClient({http.Client? client}) : _client = client ?? http.Client();

final http.Client _client;

อ่านว่า “ถ้ามีคนส่ง client มาก็ใช้ตัวนั้น ถ้าไม่ส่งก็สร้างของจริง”

ใครเรียกส่งอะไร
main()ไม่ส่ง — ได้ http.Client() จริง
เทสต์ส่ง MockClient ที่ตอบตามสคริปต์
// รูปแบบในเทสต์
final client = MockClient((request) async {
  expect(request.url.path, '/api/WMS/mobile_receiving_list');
  return http.Response('{"data": []}', 200);
});
final api = ApiClient(client: client);
ความต่างจาก wmsapp ที่ใหญ่ที่สุดอยู่ตรงนี้
// wmsapp — 16.6
class ApiService {
  static final ApiService _instance = ApiService._internal();
  factory ApiService() => _instance;
  final http.Client _client = http.Client();     // ← ตายตัว
}
ไม่มีทางใส่ของปลอมเข้าไปได้ — ต่อให้อยากเขียนเทสต์ก็เขียนไม่ได้
เทสต์ 101 ข้อของโปรเจกต์นี้ไม่ได้เกิดจากความขยัน แต่เกิดจากตัวสร้างสองบรรทัดนี้

4. post()

/// POST JSON. [path] is appended to the mobile route group prefix, unless
/// [absolute] is set — then it is used as-is after the base URL (for shared
/// WMS endpoints like /api/Authenticate/login that aren't PDA-specific).
Future<Map<String, dynamic>> post(
  String path,
  Map<String, dynamic> body, {
  bool absolute = false,
}) async {
  final prefix = absolute ? '' : AppConfig.apiPrefix;
  final uri = Uri.parse('${AppConfig.apiBaseUrl}$prefix$path');

  late http.Response response;
  try {
    response = await _client
        .post(uri, headers: _headers(), body: jsonEncode(body))
        .timeout(_timeout);
  } on SocketException {
    throw ApiException(tr('Unable to connect to the server. ...', 'เชื่อมต่อเซิร์ฟเวอร์ไม่ได้ ...'));
  } on TimeoutException {
    throw ApiException(tr('The server did not respond. ...', 'เซิร์ฟเวอร์ไม่ตอบสนอง ...'));
  }

  return _decode(response);
}

ทำไมมีแต่ post()

ทุกอย่างในแอปนี้ยิงเป็น POST แม้กระทั่งการดึงรายการ

await _api.post('/mobile_receiving_list', {'q': query, 'page': page});
ข้อดีข้อเสีย
พารามิเตอร์อยู่ในตัว JSON ไม่ต้อง encode ใส่ query stringผิดหลัก REST
ตัวส่งเหลือเมธอดเดียวแคชระดับ HTTP ใช้ไม่ได้
รหัสสินค้าที่มี / หรือ # ไม่พังยิงซ้ำจากเบราว์เซอร์/พร็อกซีไม่ปลอดภัยตามนิยาม
เป็นการเลือกที่ป้องกันปัญหาที่ 18.9 เจอจริง — wmsapp ต่อ URL ด้วยการแปะสตริง แล้วรหัสที่มีอักขระพิเศษทำให้เส้นทางเพี้ยน
ที่นี่ไม่มีอะไรอยู่ใน URL เลย ทุกค่าเดินทางในตัว JSON ปัญหานั้นจึงเป็นไปไม่ได้ตั้งแต่ต้น
ราคาที่จ่าย (แคชกับ REST) ไม่มีความหมายในแอปที่ยิงเซิร์ฟเวอร์ในวงแลนเดียวกันและต้องการข้อมูลสดเสมอ

late http.Response response;

ประกาศไว้นอก try เพราะต้องใช้หลัง try — late บอกคอมไพเลอร์ว่า “มันจะมีค่าแน่นอนก่อนถูกอ่าน”

ถูกต้องเพราะทุกทางออกของ try ที่ไม่ได้ตั้งค่า จะ throw ไปเลย

สอง catch ที่แยกกัน

ข้อยกเว้นแปลว่าบอกผู้ใช้ว่า
SocketExceptionไปไม่ถึงเครื่องปลายทางตรวจสัญญาณหรือค่าเซิร์ฟเวอร์
TimeoutExceptionถึงแล้วแต่ไม่ตอบใน 15 วินาทีลองใหม่

สองอย่างนี้ต้องการการกระทำต่างกัน — อันแรกไปแก้ตั้งค่า อันที่สองกดซ้ำ

แต่มีข้อควรระวัง — TimeoutException ไม่ได้แปลว่างานไม่เกิด
เซิร์ฟเวอร์อาจรับคำสั่งไปทำแล้ว แค่ตอบกลับไม่ทัน
ข้อความ “กรุณาลองใหม่” จึงถูกสำหรับการดึงข้อมูล แต่อันตรายสำหรับการสั่งงาน — กดซ้ำอาจได้งานสองใบ
14.7 เรียกสถานะนี้ว่า outcome unknown และเสนอ Idempotency-Key เป็นทางแก้ — โปรเจกต์นี้ยังไม่มี จะกลับมาที่ Part 22

15 วินาที

static const Duration _timeout = Duration(seconds: 15);

ค่าเดียวใช้กับทุกคำขอ ทั้งการดึงรายการและการยืนยันงาน

บนวงแลนของคลัง 15 วินาทีถือว่าใจกว้างพอสำหรับคำขอที่ช้าที่สุด และไม่นานจนคนคิดว่าแอปค้าง

5. _headers()

Map<String, String> _headers() => {
  'Content-Type': 'application/json',
  if (authToken != null) 'Authorization': 'Bearer $authToken',
};

collection if ใน map literal (D1.14) — ถ้ายังไม่ล็อกอินก็ไม่มี header ตัวนี้เลย ไม่ใช่ส่ง Bearer null

เรียกใหม่ทุกครั้งที่ยิง จึงหยิบ token ล่าสุดเสมอ — เป็นแนวคิดเดียวกับ getter ใน 20.3

6. _decode() — ส่วนที่ละเอียดที่สุด

Map<String, dynamic> _decode(http.Response response) {
  Map<String, dynamic>? json;
  if (response.body.isNotEmpty) {
    try {
      json = jsonDecode(response.body) as Map<String, dynamic>;
    } on FormatException {
      json = null;                                      // ①
    }
  }

  if (response.statusCode >= 200 && response.statusCode < 300) {
    return json ?? <String, dynamic>{};                 // ②
  }

  final message =
      json?['Message'] as String? ??                    // ③
      json?['message'] as String? ??
      json?['msg'] as String? ??
      tr('An error occurred (${response.statusCode})',
         'เกิดข้อผิดพลาด (${response.statusCode})');
  throw ApiException(message, statusCode: response.statusCode);   // ④
}

① แปลง JSON ไม่ได้ก็ไม่ล้ม

เมื่อเซิร์ฟเวอร์ตอบ HTML (หน้าข้อผิดพลาดของ IIS, หน้าล็อกอินของพร็อกซี) jsonDecode จะโยน FormatException

ถ้าไม่จับ ผู้ใช้จะเห็น FormatException: Unexpected character ซึ่งไม่ช่วยอะไร — ที่นี่ตั้ง json = null แล้วไปต่อ สุดท้ายได้ข้อความ “เกิดข้อผิดพลาด (500)” ซึ่งอย่างน้อยบอกรหัสได้

② สำเร็จแต่ตัวว่าง

return json ?? <String, dynamic>{};

204 No Content หรือ 200 ที่ตัวว่าง จะได้ map ว่างแทนที่จะล้ม — คำสั่งที่ไม่คืนข้อมูล (เช่น ยืนยันงาน) จึงเรียกได้โดยไม่ต้องเช็ค null

③ ลองอ่านชื่อฟิลด์สามแบบ

ชื่อมาจากไหน
MessageASP.NET ทั่วไป (ตัวใหญ่ตามธรรมเนียม C#)
messageเมื่อ .NET ตั้งค่าแปลงชื่อเป็น camelCase
msgเส้นทางที่คนเขียนเองแบบสั้น
สามบรรทัดนี้เป็นหลักฐานว่าฝั่ง .NET ไม่ได้ตอบรูปแบบเดียวกันหมด
ถ้าตอบเหมือนกันทั้งหมด บรรทัดเดียวก็พอ
ทางแก้ที่ถูกจริงคือทำให้ฝั่งเซิร์ฟเวอร์ตอบรูปเดียว แล้วลดเหลือบรรทัดเดียว — แต่ตราบใดที่ยังไม่ทำ การเผื่อไว้ตรงนี้ดีกว่าปล่อยให้ผู้ใช้เห็น “เกิดข้อผิดพลาด (400)” ทั้งที่เซิร์ฟเวอร์อุตส่าห์อธิบายมาแล้ว

④ ส่ง statusCode ติดไปด้วย

throw ApiException(message, statusCode: response.statusCode);

เทียบกับ wmsapp ที่ 19.12 ชี้ว่าโยนแค่ข้อความ ทำให้จอปลายทางแยก 401 ออกจาก 500 ไม่ได้

เมื่อมี statusCode จอจึงตัดสินใจต่างกันได้

on ApiException catch (e) {
  if (e.statusCode == 401) {  /* เด้งกลับไปล็อกอิน */ }
  else if (e.statusCode == null) { /* ไม่รู้ผล — อย่ากดซ้ำอัตโนมัติ */ }
  else { /* แสดงข้อความ */ }
}

7. สามอย่างที่ไฟล์นี้ทำถูกและ wmsapp ไม่ได้ทำ

เรื่องที่นี่wmsappผลของการไม่ทำ
ฉีด http.Client ได้ได้ไม่ได้เขียนเทสต์ไม่ได้เลย
statusCode ติดมากับ errorมีไม่มีแยก 401 จาก 500 ไม่ได้
ค่าทั้งหมดอยู่ในตัว bodyใช่ต่อ URL ด้วยสตริงรหัสที่มี / ทำให้เส้นทางเพี้ยน

และหนึ่งอย่างที่ยังขาดเหมือนกันทั้งคู่

ไม่มีการกันการยิงซ้ำ — ไม่มี Idempotency-Key และไม่มีการแยกว่าคำขอนี้ปลอดภัยที่จะลองใหม่หรือไม่
เมื่อหมดเวลารอตอนกด “ยืนยันรับสินค้า” แอปบอกให้ลองใหม่ ซึ่งอาจสร้างรายการซ้ำ
เป็นข้อจำกัดที่ต้องแก้ร่วมกับฝั่ง .NET — แอปฝ่ายเดียวทำไม่ได้

8. ข้อผิดพลาดที่พบบ่อย

ผิดอาการถูก
สร้าง http.Client ในคลาสตายตัวเทสต์ไม่ได้รับผ่านตัวสร้าง
โยน error ที่ไม่มี statusCodeปลายทางแยกกรณีไม่ได้พก int? ไปด้วย
ไม่ override toString()ผู้ใช้เห็น “Instance of…”คืน message
jsonDecode โดยไม่จับ FormatExceptionHTML ทำแอปล้มจับแล้วปล่อยเป็น null
ไม่ตั้ง timeoutหมุนค้างไม่จบ.timeout(...)
บอก “ลองใหม่” เมื่อหมดเวลาของคำสั่งอาจได้งานซ้ำแยกดึงข้อมูลกับสั่งงาน

9. แบบฝึกหัด

  1. เซิร์ฟเวอร์ตอบ 200 พร้อมตัว body เป็น HTML — เดินตามโค้ดแล้วบอกว่าผู้เรียก post() ได้อะไร และปัญหาคืออะไร
  2. เขียนเทสต์ที่พิสูจน์ว่า statusCode ถูกส่งต่อจริง โดยใช้ MockClient
แนวคำตอบ

ข้อ 1 — เดินตามโค้ด

body ไม่ว่าง        → เข้า try
jsonDecode('<html>...')  → FormatException
                     → json = null
statusCode 200      → เข้าเงื่อนไขสำเร็จ
return json ?? {}   → คืน map ว่าง

ผู้เรียกได้ {} โดยไม่มีข้อผิดพลาดใด ๆ

นี่คือกรณีที่แอบอันตราย
ตัวแปลง (parser) ที่รับ map ว่างจะได้รายการว่าง แล้วจอขึ้น “ไม่พบข้อมูล” — ทั้งที่ความจริงคือยิงไปโดนหน้าเว็บ ไม่ใช่ API
อาการที่ผู้ใช้เห็นคือ “ค้นหาแล้วไม่เจออะไรเลย” ซึ่งพาไปหาที่ผิดฝั่ง
เกิดได้จริงเมื่อ: ตั้ง Base Path ผิด, พร็อกซีคืนหน้าล็อกอิน, หรือชน MapFallbackToPage ของ Blazor

ทางแก้ที่พอดี

if (response.statusCode >= 200 && response.statusCode < 300) {
  if (json == null && response.body.isNotEmpty) {
    // ตอบ 200 แต่ไม่ใช่ JSON = ไม่ได้คุยกับ API
    throw ApiException(
      tr('The server replied with an unexpected format. Check the server settings.',
         'เซิร์ฟเวอร์ตอบมาในรูปแบบที่ไม่ใช่ของ API ตรวจสอบค่าเซิร์ฟเวอร์'),
      statusCode: response.statusCode,
    );
  }
  return json ?? <String, dynamic>{};
}

เงื่อนไข response.body.isNotEmpty จำเป็น มิฉะนั้น 204 No Content จะกลายเป็นข้อผิดพลาด

สังเกตว่า connection_probe.dart แก้ปัญหาเดียวกันนี้ไปแล้ว (20.12) — มันตรวจว่า json['service'] == 'WMS' ไม่ใช่แค่ดูรหัส 200
แปลว่าคนเขียนรู้จักปัญหานี้ แต่ป้องกันไว้เฉพาะตอนตรวจสถานะ ยังไม่ได้ป้องกันในเส้นทางคำขอปกติ
เป็นตัวอย่างของบทเรียนที่เรียนแล้วแต่ยังไม่ได้ใช้ทั่วถึง — ตรงข้ามกับ 19.15 ที่เป็นกฎเดียวเขียนสองแบบ อันนี้คือกฎเดียวเขียนที่เดียวทั้งที่ควรเขียนสองที่

ข้อ 2

import 'package:flutter_test/flutter_test.dart';
import 'package:http/http.dart' as http;
import 'package:http/testing.dart';
import 'package:wms_absolute_mobile/core/api_client.dart';

void main() {
  test('a 400 reply keeps both the server message and the status code',
      () async {
    final api = ApiClient(
      client: MockClient((_) async => http.Response(
            '{"Message":"พาเลทถูกใช้แล้ว"}',
            400,
            headers: {'content-type': 'application/json'},
          )),
    );

    await expectLater(
      () => api.post('/mobile_receive', {'palletCode': 'P001'}),
      throwsA(
        isA<ApiException>()
            .having((e) => e.message, 'message', 'พาเลทถูกใช้แล้ว')
            .having((e) => e.statusCode, 'statusCode', 400),
      ),
    );
  });

  test('an unreachable server yields a null status code', () async {
    final api = ApiClient(
      client: MockClient((_) async => throw const SocketException('no route')),
    );

    await expectLater(
      () => api.post('/mobile_receive', {}),
      throwsA(isA<ApiException>().having((e) => e.statusCode, 'statusCode', isNull)),
    );
  });
}

เทสต์ที่สองสำคัญกว่าที่ดู — มันตรึงความหมายของ null ไว้ว่า “ไม่รู้ผล” ถ้ามีใครมาเติม statusCode: 0 ในอนาคต เทสต์จะแดง

เขียนได้เพราะบรรทัดเดียวใน 20.4 ข้อ 3 — ApiClient({http.Client? client})

10. สรุป

  • 96 บรรทัด เมธอดสาธารณะเดียวคือ post() — ทุกอย่างยิง POST ค่าทั้งหมดอยู่ใน body
  • ApiClient({http.Client? client}) คือบรรทัดที่ทำให้เทสต์ 101 ข้อเป็นไปได้
  • ApiException พก int? statusCode — null แปลว่าไม่รู้ว่าคำขอถึงหรือไม่
  • toString() คืนข้อความ ทำให้ทางที่เขียนง่ายที่สุดเป็นทางที่ถูก
  • แยก SocketException กับ TimeoutException เพราะผู้ใช้ต้องทำคนละอย่าง
  • _decode จับ FormatException ไว้ ไม่ให้ HTML ทำแอปล้ม
  • ลองอ่านชื่อฟิลด์ข้อความสามแบบ — หลักฐานว่าฝั่งเซิร์ฟเวอร์ยังตอบไม่เป็นรูปเดียว
  • ยังขาดการกันยิงซ้ำ และยังไม่ตรวจว่า 200 ที่ได้เป็น JSON จริงหรือไม่
