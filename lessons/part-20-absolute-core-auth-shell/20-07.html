เจ็ดฟิลด์ที่ไม่มีตัวไหนซ้ำซ้อนกัน — เทียบกับ 19.6 ที่จอเช็คอินของ wmsapp มีเจ็ดฟิลด์เหมือนกัน แต่สามตัวเป็น nullable ที่รวมกันได้แปดสถานะ ทั้งที่ตั้งใจให้มีแค่สาม

2. _submit()

Future<void> _submit() async {
  if (_submitting) return;                                       // ①
  if (!(_formKey.currentState?.validate() ?? false)) return;      // ②

  FocusScope.of(context).unfocus();                              // ③
  setState(() {
    _submitting = true;
    _error = null;                                               // ④
  });

  try {
    await widget.auth.login(
      _employeeIdController.text.trim(),                          // ⑤
      _passwordController.text,                                   // ⑥
    );
    if (!mounted) return;                                         // ⑦
    Navigator.of(context).pushReplacement(
      MaterialPageRoute(builder: (_) => WarehouseShell(auth: widget.auth)),
    );
  } on ApiException catch (e) {                                   // ⑧
    if (!mounted) return;
    setState(() => _error = e.message);
  } finally {
    if (mounted) setState(() => _submitting = false);             // ⑨
  }
}

① ยามกันกดซ้ำ

บนพีดีเอที่ปุ่มตอบสนองช้า คนกดสองครั้งเป็นเรื่องปกติ — บรรทัดนี้กันการยิงล็อกอินซ้อน

สังเกตว่ามีการกันสองชั้น: บรรทัดนี้ และ onPressed: _submitting ? null : _submit ที่ตัวปุ่ม

สองชั้นไม่ใช่ความซ้ำซ้อนที่ไร้ประโยชน์
ชั้นที่ปุ่มทำให้ปุ่มเป็นสีเทา — สื่อสารกับผู้ใช้
ชั้นในเมธอดกันเส้นทางอื่นที่เรียกเข้ามา — ในไฟล์นี้คือ onFieldSubmitted: (_) => _submit() ที่ช่องรหัสผ่าน ซึ่งกดปุ่ม “เสร็จ” บนแป้นพิมพ์แล้วยิงได้โดยไม่ผ่านปุ่ม

② ตรวจฟอร์ม

if (!(_formKey.currentState?.validate() ?? false)) return;

validate() เรียก validator ของทุกช่องในฟอร์ม แล้วให้ช่องที่ไม่ผ่านแสดงข้อความสีแดงเอง

?? false กันกรณี currentState เป็น null (ฟอร์มยังไม่ถูกวาด) — ถือว่าไม่ผ่าน ปลอดภัยกว่า

③ ปิดแป้นพิมพ์

บนจอพีดีเอที่เล็ก แป้นพิมพ์กินพื้นที่ครึ่งจอ — ปิดก่อนแล้วผู้ใช้จะเห็นตัวหมุนและข้อความผิดพลาด

④ ล้างข้อผิดพลาดเก่า

ถ้าไม่ล้าง ผู้ใช้จะเห็นข้อความจากครั้งก่อนค้างอยู่ระหว่างที่กำลังลองใหม่ ทำให้คิดว่าครั้งนี้ก็ผิดอีก

⑤⑥ trim() รหัสพนักงาน แต่ไม่ trim รหัสผ่าน

ช่องtrimเหตุผล
รหัสพนักงานใช่เครื่องสแกนหรือแป้นพิมพ์อาจแถมช่องว่างมา และช่องว่างไม่มีความหมายในรหัส
รหัสผ่านไม่ช่องว่างเป็นอักขระที่ถูกต้องในรหัสผ่าน — ตัดทิ้งคือแก้รหัสผ่านของผู้ใช้
รายละเอียดเล็กที่หลายโปรเจกต์ทำผิด — .trim() ทั้งสองช่องเพราะเขียนติดมือ แล้วผู้ใช้ที่ตั้งรหัสผ่านลงท้ายด้วยเว้นวรรคจะล็อกอินไม่ได้ตลอดกาลโดยไม่มีใครหาสาเหตุเจอ

⑦ if (!mounted) return; ก่อนใช้ context

ระหว่างรอเซิร์ฟเวอร์ ผู้ใช้อาจปิดแอปหรือจอถูกถอดออก — ใช้ context ตอนนั้นจะพัง (4.12)

⑧ จับเฉพาะ ApiException

นี่คือช่องโหว่ของจอนี้
20.5 ชี้ว่า json['token'] as String จะโยน TypeError ถ้าเซิร์ฟเวอร์ตอบ 200 แต่ไม่มี token
TypeError ไม่ใช่ ApiException จึงหลุดผ่าน catch นี้ไป
สิ่งที่ผู้ใช้เห็น: ตัวหมุนหยุด (เพราะ finally ทำงาน) แต่ไม่มีข้อความอะไรเลย และไม่ได้ไปไหน — เหมือนกดปุ่มแล้วไม่มีอะไรเกิดขึ้น
เป็นอาการที่หาสาเหตุยากที่สุดแบบหนึ่ง เพราะไม่มีร่องรอยให้ผู้ใช้รายงาน

ทางแก้

} on ApiException catch (e) {
  if (!mounted) return;
  setState(() => _error = e.message);
} catch (_) {
  if (!mounted) return;
  setState(() => _error = tr(
    'Sign-in failed. Please try again or check the server settings.',
    'เข้าสู่ระบบไม่สำเร็จ กรุณาลองใหม่หรือตรวจสอบค่าเซิร์ฟเวอร์',
  ));
}

ข้อความกว้าง ๆ แต่มีข้อความ — ผู้ใช้รู้ว่าเกิดอะไรขึ้นและมีอะไรให้ลอง

⑨ finally ที่เช็ค mounted

} finally {
  if (mounted) setState(() => _submitting = false);
}

finally ทำงานทุกทาง รวมทางที่สำเร็จแล้วเปลี่ยนจอไปแล้ว — จึงต้องเช็ค mounted

ในเส้นทางที่สำเร็จ pushReplacement ถอดจอนี้ออก mounted จึงเป็น false และไม่มีการ setState ที่ไร้ประโยชน์

3. pushReplacement ไม่ใช่ push

Navigator.of(context).pushReplacement(
  MaterialPageRoute(builder: (_) => WarehouseShell(auth: widget.auth)),
);
pushpushReplacement
จอล็อกอินยังอยู่ในกองอยู่ถูกถอดออก
กดปุ่มย้อนกลับของเครื่องกลับไปหน้าล็อกอินที่ยังกรอกไว้ออกจากแอป
หน่วยความจำค้างจอที่ไม่ใช้แล้วคืน

บนพีดีเอที่ใช้ร่วมกัน การกดย้อนกลับแล้วเจอหน้าล็อกอินที่ยังมีรหัสพนักงานคนก่อนค้างอยู่คือปัญหา — pushReplacement จึงถูก

ทางกลับก็เหมือนกัน — 20.15 จะเห็นว่าเมนูผู้ใช้ใช้ pushReplacement กลับมาที่ LoginScreen เช่นกัน กองจึงมีจอเดียวเสมอ

5. รายละเอียดของฟอร์มที่ทำถูก

สิ่งที่ทำผลกับผู้ใช้พีดีเอ
autofocus: true ที่ช่องแรกสแกนบัตรได้ทันทีไม่ต้องแตะจอ
textInputAction: next + onFieldSubmitted ย้ายโฟกัสสแกนเสร็จโฟกัสไปช่องรหัสผ่านเอง
textInputAction: done + onFieldSubmitted: _submitจบได้โดยไม่ต้องแตะปุ่ม
autocorrect: false, TextCapitalization.noneรหัสพนักงานไม่ถูกแก้เป็นคำอื่น
enabled: !_submitting ทั้งสองช่องแก้ข้อความระหว่างส่งไม่ได้
ปุ่มตาดู/ซ่อนรหัสผ่านพิมพ์ผิดแล้วตรวจได้ — สำคัญเมื่อสวมถุงมือ
maxWidth: 420ไม่ยืดเต็มจอบนเครื่องจอกว้าง
SingleChildScrollViewแป้นพิมพ์ขึ้นมาแล้วไม่ล้น

ทั้งชุดนี้รวมกันแปลว่าล็อกอินได้โดยไม่ต้องแตะจอเลย — สแกนบัตร ยิงรหัสผ่าน จบ

17.5 อธิบายว่าทำไมการไม่ต้องแตะจอถึงสำคัญ: คนงานสวมถุงมือ มือถือกล่องอยู่ และจอเปียกได้
จอนี้ทำครบตั้งแต่จอแรกที่ผู้ใช้เจอ

6. AnimatedSize รอบกล่องข้อผิดพลาด

AnimatedSize(
  duration: const Duration(milliseconds: 180),
  child: _error == null
      ? const SizedBox(width: double.infinity)
      : Container(... กล่องสีแดง ...),
),

เมื่อข้อความผิดพลาดโผล่ ฟอร์มจะสูงขึ้น — ถ้าเปลี่ยนทันทีปุ่มจะกระโดด และผู้ใช้ที่กำลังจะกดอาจกดโดนที่อื่น

AnimatedSize ทำให้ขยายภายใน 180 มิลลิวินาที เร็วพอที่ไม่รู้สึกว่าช้า แต่ตาตามทัน

สังเกต SizedBox(width: double.infinity) ในกรณีไม่มีข้อผิดพลาด — กว้างเต็มแต่สูงศูนย์ ทำให้ AnimatedSize เปลี่ยนแค่ความสูง ไม่เปลี่ยนความกว้างไปด้วย

7. ข้อผิดพลาดที่พบบ่อย

ผิดอาการถูก
จับเฉพาะข้อยกเว้นชนิดเดียวกดแล้วเงียบ ไม่มีข้อความมี catch (_) ปิดท้าย
trim() รหัสผ่านรหัสผ่านที่มีช่องว่างใช้ไม่ได้trim เฉพาะรหัสพนักงาน
ใช้ push ไปหน้าหลักกดย้อนกลับเจอฟอร์มคนก่อนpushReplacement
ไม่มีทางเข้าตั้งค่าก่อนล็อกอินตั้ง IP ผิดแล้วตันปุ่มตั้งค่าบนจอล็อกอิน
ไม่ล้าง _error ก่อนยิงใหม่ข้อความเก่าค้างล้างใน setState แรก
setState ใน finally โดยไม่เช็ค mountedข้อยกเว้นหลังเปลี่ยนจอif (mounted)

8. แบบฝึกหัด

  1. ผู้ใช้กดปุ่มล็อกอินแล้วปิดจอทันที ระหว่างนั้นเซิร์ฟเวอร์ตอบสำเร็จ — เดินตามโค้ดว่าเกิดอะไรขึ้นตามลำดับ
  2. เขียน widget test ที่พิสูจน์ว่าข้อผิดพลาดจากเซิร์ฟเวอร์ถูกแสดงบนจอ โดยไม่ต้องมีเซิร์ฟเวอร์จริง
แนวคำตอบ

ข้อ 1

1. _submitting = true, จอวาดตัวหมุน
2. await login() เริ่ม
3. ผู้ใช้ปิดจอ → State ถูกถอด → mounted = false
   dispose() ทำงาน ปล่อย controller ทั้งสาม
4. เซิร์ฟเวอร์ตอบสำเร็จ → login() คืนค่า
   _api.authToken ถูกตั้งแล้ว  ← สำเร็จจริง
5. if (!mounted) return;  → ออกจาก try
6. finally: if (mounted) → เป็น false ไม่ทำอะไร

ไม่มีข้อยกเว้น ไม่มีการเข้าถึงของที่ปล่อยไปแล้ว

แต่มีผลข้างเคียงที่น่าสังเกต — ในขั้นที่ 4 การล็อกอินสำเร็จจริง และ token ถูกตั้งบน ApiClient ที่ยังมีชีวิตอยู่ใน main()
AuthRepository._session ก็มีค่าแล้ว
แต่ผู้ใช้กลับมาเจอจอล็อกอินอีกครั้ง เพราะไม่มีใครตรวจว่ามี session ค้างอยู่
ไม่ใช่บั๊กเรื่องความปลอดภัย (ยังต้องกรอกใหม่อยู่ดี) แต่เป็นโอกาสที่พลาดไป — เติมได้ง่ายมาก
// LoginScreen.initState
if (widget.auth.session != null) {
  WidgetsBinding.instance.addPostFrameCallback((_) {
    if (!mounted) return;
    Navigator.of(context).pushReplacement(...);
  });
}
แต่ต้องคิดให้ครบก่อนทำ: เส้นทางที่มาจาก logout ก็ผ่าน LoginScreen เหมือนกัน — ถ้า logout() ล้าง session ไม่หมด โค้ดนี้จะเด้งกลับเข้าแอปทันที กลายเป็นล็อกเอาต์ไม่ได้
ตรวจแล้วว่า logout() ตั้ง _session = null จริง จึงปลอดภัย — แต่นี่คือประเภทของการเชื่อมโยงที่ต้องไล่ให้ครบก่อนเพิ่มโค้ดสั้น ๆ แบบนี้

ข้อ 2

testWidgets('a rejected sign-in shows the server message on screen',
    (tester) async {
  final api = ApiClient(
    client: MockClient((_) async => http.Response(
          '{"Message":"รหัสพนักงานหรือรหัสผ่านไม่ถูกต้อง"}',
          401,
        )),
  );

  await tester.pumpWidget(MaterialApp(
    home: LoginScreen(auth: AuthRepository(api)),
  ));

  await tester.enterText(find.byType(TextFormField).first, 'EMP001');
  await tester.enterText(find.byType(TextFormField).last, 'wrong');
  await tester.tap(find.byType(FilledButton));
  await tester.pumpAndSettle();

  expect(find.text('รหัสพนักงานหรือรหัสผ่านไม่ถูกต้อง'), findsOneWidget);
  expect(find.byType(WarehouseShell), findsNothing);
});

บรรทัดสุดท้ายสำคัญพอ ๆ กับบรรทัดก่อนหน้า — ยืนยันว่าไม่ได้เข้าไป ไม่ใช่แค่ว่ามีข้อความขึ้น

เขียนได้เพราะสองอย่างที่มีอยู่แล้ว: ApiClient รับ client ได้ (20.4) และ LoginScreen รับ AuthRepository ผ่านตัวสร้าง (20.2)

ลองต่อยอด: เปลี่ยน mock ให้ตอบ '{"username":"x"}' พร้อมสถานะ 200 — จะเห็นอาการเงียบที่บทที่ 2 ข้อ ⑧ พูดถึงด้วยตาตัวเอง

9. สรุป
