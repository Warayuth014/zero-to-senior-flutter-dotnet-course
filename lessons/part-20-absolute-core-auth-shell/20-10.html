2. normalizeBasePath — สี่บรรทัดที่ทำงานหนัก

String normalizeBasePath(String raw) {
  final trimmed = raw.trim();
  if (trimmed.isEmpty) return '';
  final collapsed = trimmed
      .split('/')
      .where((segment) => segment.isNotEmpty)
      .join('/');
  return collapsed.isEmpty ? '' : '/$collapsed';
}

สัญญาที่คอมเมนต์ประกาศไว้: ผลลัพธ์เป็น '' หรือ /something ที่ไม่มีสแลชท้ายและไม่มีสแลชซ้อน

ใส่ได้
''''
' '''
'/'''
'wms''/wms'
'/wms/''/wms'
'//wms//api//''/wms/api'

เทคนิคที่ใช้คือ split แล้วทิ้งส่วนว่าง แล้ว join กลับ ซึ่งจัดการสแลชซ้อนได้ในตัว โดยไม่ต้องเขียนเงื่อนไขเป็นชั้น ๆ

ทำไมต้องมีฟังก์ชันนี้เลย
เพราะผลลัพธ์ถูกนำไปต่อสตริงตรง ๆ
String get baseUrl =>
    '${protocol.name}://$host:$port${normalizeBasePath(basePath)}';
ถ้าไม่ normalize จะได้ http://host:6191//wms/ แล้วต่อกับ /api/WMS กลายเป็น //wms//api/WMS
เซิร์ฟเวอร์บางตัวยอม บางตัวไม่ยอม — และเมื่อไม่ยอม ข้อความผิดพลาดจะไม่บอกว่าเป็นเพราะสแลชเกิน

เก็บส่วนประกอบ ไม่เก็บผลลัพธ์

ทางเลือกปัญหา
เก็บ baseUrl เป็นฟิลด์ต้องแยกกลับเป็นชิ้นตอนแก้ไข — และการแยก URL ให้ถูกทุกกรณีเป็นงานยาก
เก็บชิ้น แล้วคำนวณฟอร์มแก้ทีละช่องได้ตรง ๆ

เป็นรูปแบบเดียวกับ apiBaseUrl ใน 20.3 — อย่าเก็บสิ่งที่คำนวณได้

basePath เก็บดิบ

/// Raw value as typed; use [baseUrl] (or [normalizeBasePath]) to consume it.
final String basePath;

เก็บตามที่ผู้ใช้พิมพ์ ไม่ normalize ตอนเก็บ

เหตุผล: ถ้า normalize ตอนเก็บ ผู้ใช้จะเปิดฟอร์มมาเจอข้อความที่ไม่เหมือนที่ตัวเองพิมพ์ ซึ่งชวนสับสนว่าระบบแก้อะไรให้
เก็บดิบแล้ว normalize ตอนใช้ = สิ่งที่เห็นคือสิ่งที่พิมพ์ และสิ่งที่ยิงคือสิ่งที่ถูก
คอมเมนต์บรรทัดเดียวบอกกฎการใช้งานไว้ชัด จึงไม่มีใครเผลอเอา basePath ไปต่อสตริงเอง

4. fromJson ที่คืน null แทนที่จะโยน

/// Returns null for a row that cannot be trusted rather than throwing — one
/// bad entry must not cost the operator every other saved profile.
static ConnectionProfile? fromJson(Object? raw) {
  if (raw is! Map) return null;
  final id = raw['id']?.toString();
  final host = raw['host']?.toString();
  final port = int.tryParse(raw['port']?.toString() ?? '');
  if (id == null || id.isEmpty || host == null || host.isEmpty) return null;
  if (port == null || port < 1 || port > 65535) return null;
  ...
}

คอมเมนต์บอกเหตุผลไว้ตรง ๆ และมันถูก

// ใน decodeConnectionSnapshot
for (final row in rows) {
  final profile = ConnectionProfile.fromJson(row);
  if (profile == null) continue;      // ข้ามแถวเสีย
  profiles.add(profile);
  ...
}
ผลต่างที่ผู้ใช้เห็น
ถ้าโยนที่ทำจริง — คืน null
โปรไฟล์เสียหนึ่งอัน → เสียทั้งหมด → กลับไปเป็นค่าปริยายเสียอันเดียว อีกสี่อันยังอยู่
ผู้ใช้ต้องพิมพ์ที่อยู่ทุกเซิร์ฟเวอร์ใหม่ผู้ใช้อาจไม่ทันสังเกตด้วยซ้ำ
กฎทั่วไป: เมื่ออ่านข้อมูลที่ผู้ใช้สะสมมา ให้รักษาส่วนที่ยังดีไว้เสมอ

ตัวถอยของแต่ละฟิลด์

ฟิลด์ขาดแล้ว

สามฟิลด์แรกคือสิ่งที่ขาดแล้วโปรไฟล์ไม่มีความหมาย ที่เหลือเป็นของประกอบที่เดาแทนได้

เป็นหลักเดียวกับ token ใน 20.5 — แยกให้ออกว่าอะไรคือแก่น อะไรคือของแถม

);
เทสต์ 37 ข้อในไฟล์นี้ไม่ขึ้นกับภาษาเลย เพราะบรรทัดเดียวนี้

กฎที่ตรวจ

ช่องห้ามข้อความบอกให้ไปทำอะไร
ชื่อว่างตั้งชื่อก่อน
hostว่างใส่ IP หรือชื่อเครื่อง
hostมี :// หรือขึ้นต้นด้วย httpเลือกที่ช่อง Protocol

hostมีช่องว่างห้ามมีช่องว่าง

ข้อความผิดพลาดที่ดีบอกการกระทำถัดไป ไม่ใช่บอกว่าผิด

6. ConnectionSnapshot — record ที่รวมสองอย่างไว้ด้วยกัน

/// Everything persisted under one preferences key, so "exactly one active
/// profile" is written and read atomically instead of drifting across keys.
typedef ConnectionSnapshot = ({
  List<ConnectionProfile> profiles,
  String? activeId,
});

record ของ Dart 3 (D1.20) — กลุ่มค่าที่มีชื่อฟิลด์ โดยไม่ต้องประกาศคลาส

คอมเมนต์อธิบายเหตุผลที่ต้องรวม

ถ้าแยกสองคีย์ที่ทำจริง — คีย์เดียว
เขียนคีย์แรกสำเร็จ คีย์สองล้ม → activeId ชี้ไปโปรไฟล์ที่ไม่มีเขียนทีเดียว สำเร็จหรือไม่สำเร็จทั้งก้อน

encodeConnectionSnapshot

String encodeConnectionSnapshot(
  List<ConnectionProfile> profiles,
  String? activeId,
) => jsonEncode({
  'version': 1,
  'profiles': [
    for (final profile in profiles)
      profile.toJson(isActive: profile.id == activeId),
  ],
});

สองอย่างที่น่าสังเกต

รายละเอียดทำไม
'version': 1เผื่อเปลี่ยนรูปแบบในอนาคต — อ่านได้ว่าเป็นรุ่นไหน
isActive เก็บในแต่ละแถวไม่ต้องมีคีย์ activeId แยก
ข้อสังเกตเรื่อง version
เขียนลงไปแต่ decodeConnectionSnapshot ไม่ได้อ่าน
final rows = decoded['profiles'];    // ข้าม version ไปเลย
ตอนนี้ไม่มีผล เพราะมีรุ่นเดียว — แต่ตอนที่มีรุ่นสอง คนเขียนต้องไม่ลืมว่าข้อมูลรุ่นหนึ่งจะถูกอ่านด้วยตัวอ่านรุ่นสองโดยไม่มีการตรวจ
เป็นการเตรียมที่ทำครึ่งทาง: ฝั่งเขียนพร้อม ฝั่งอ่านยังไม่พร้อม
ไม่ใช่บั๊ก และตรงกับหลัก “อย่าออกแบบเผื่ออนาคต” — แต่ควรมีคอมเมนต์บอกว่าตั้งใจไม่อ่าน ไม่ใช่ลืม

decodeConnectionSnapshot — รักษาข้อตกลงไว้ให้ได้

/// Rebuilds the snapshot, dropping unreadable rows. `activeId` comes from the
/// stored `isActive` flag and falls back to the first profile, so the invariant
/// holds even if the stored file was hand-edited.
ConnectionSnapshot decodeConnectionSnapshot(String? raw) {
  if (raw == null || raw.trim().isEmpty) return (profiles: [], activeId: null);

  Object? decoded;
  try {
    decoded = jsonDecode(raw);
  } on FormatException {
    return (profiles: [], activeId: null);
  }
  if (decoded is! Map) return (profiles: [], activeId: null);

  final rows = decoded['profiles'];
  if (rows is! List) return (profiles: [], activeId: null);

  final profiles = <ConnectionProfile>[];
  String? activeId;
  for (final row in rows) {
    final profile = ConnectionProfile.fromJson(row);
    if (profile == null) continue;
    profiles.add(profile);
    if (activeId == null && row is Map && row['isActive'] == true) {
      activeId = profile.id;
    }
  }
  if (profiles.isEmpty) return (profiles: [], activeId: null);
  return (profiles: profiles, activeId: activeId ?? profiles.first.id);
}

ข้อตกลง (invariant) ที่ฟังก์ชันนี้รับประกัน: ถ้ามีโปรไฟล์อย่างน้อยหนึ่ง จะมี activeId ที่ชี้ไปโปรไฟล์ที่มีอยู่จริงเสมอ

ข้อมูลที่เจอผลลัพธ์
ไม่มี isActive เลยใช้ตัวแรก
มี isActive สองแถวใช้แถวแรกที่เจอ (activeId == null &&)
แถวที่ isActive เสีย ถูกทิ้งตกไปใช้ตัวแรกที่เหลือ
JSON พังว่างเปล่า → controller จะสร้างโปรไฟล์เริ่มต้นให้
ทุกเส้นทางออกจากฟังก์ชันนี้ให้ผลที่ใช้งานได้ ไม่มีทางไหนคืนสถานะที่ขัดแย้งกันเอง
เป็นรูปแบบเดียวกับ PdaAccess.fromJson ใน 20.6 ที่ประกันว่า map ครบ 8 คีย์เสมอ
หลักร่วม: ให้ตัวแปลงเป็นด่านที่ข้อมูลผิดรูปเข้ามาไม่ได้ แล้วโค้ดที่เหลือทั้งหมดไม่ต้องระแวง

8. ข้อผิดพลาดที่พบบ่อย

ผิดอาการถูก
fromJson โยนเมื่อแถวเสียโปรไฟล์เสียหนึ่ง เสียทั้งหมดคืน null แล้วข้าม
เก็บ URL เต็มเป็นฟิลด์ต้องแยกกลับตอนแก้ไขเก็บชิ้น คำนวณ

ข้อ 1

ConnectionProfile p(String id, {bool active = false}) => ...;

group('decodeConnectionSnapshot keeps exactly one valid active profile', () {
  test('null and empty input give an empty snapshot', () {
    expect(decodeConnectionSnapshot(null).profiles, isEmpty);
    expect(decodeConnectionSnapshot('   ').profiles, isEmpty);
  });

  test('malformed json does not throw', () {
    expect(decodeConnectionSnapshot('{not json').activeId, isNull);
  });

  test('a row without isActive falls back to the first profile', () {
    final raw = encodeConnectionSnapshot([p('a'), p('b')], null);
    expect(decodeConnectionSnapshot(raw).activeId, 'a');
  });

  test('a broken row is dropped without losing the others', () {
    final raw = jsonEncode({'version': 1, 'profiles': [
      {'id': '', 'host': 'x', 'port': 1},          // id ว่าง
      {'id': 'b', 'host': 'h', 'port': 6191, 'isActive': true},
    ]});
    final snap = decodeConnectionSnapshot(raw);
    expect(snap.profiles.length, 1);
    expect(snap.activeId, 'b');
  });

  test('when the active row is the broken one, the survivor takes over', () {
    final raw = jsonEncode({'version': 1, 'profiles': [
      {'id': 'a', 'host': 'h', 'port': 99999, 'isActive': true},  // พอร์ตเกิน
      {'id': 'b', 'host': 'h', 'port': 6191},
    ]});
    final snap = decodeConnectionSnapshot(raw);
    expect(snap.activeId, 'b');      // ข้อตกลงยังอยู่
  });
});

แต่การไม่จับกรณีเหล่านี้เป็นการตัดสินใจที่ถูก ไม่ใช่ช่องโหว่
กฎที่มีอยู่ทั้งหมดจับความสับสนเรื่องว่าค่าควรอยู่ช่องไหน ซึ่งเป็นความผิดพลาดที่ผู้ใช้แก้เองไม่ได้ถ้าไม่มีใครบอก
ส่วน 192.168.1.999 เป็นที่อยู่ที่พิมพ์ผิด ซึ่งปุ่ม “ทดสอบการเชื่อมต่อ” (20.12) ตอบได้ตรงกว่าและแม่นกว่า — เพราะที่อยู่ที่ถูกรูปแบบทุกอย่างก็ยังต่อไม่ติดได้ถ้าเครื่องปิดอยู่
หลักที่ได้: ให้การตรวจรูปแบบจับสิ่งที่รูปแบบบอกได้ และให้การทดสอบจริงจับสิ่งที่ต้องลองถึงจะรู้ — อย่าให้อันแรกพยายามทำงานของอันหลัง
การเขียน regex ตรวจ IP ที่นี่จะเพิ่มโค้ดที่ต้องดูแล แล้วยังตอบคำถามที่ผู้ใช้อยากรู้จริง ๆ ไม่ได้อยู่ดี

10. สรุป

  • 348 บรรทัดที่ import แค่ dart:convert — จึงมีเทสต์ 37 ข้อ
  • normalizeBasePath ใช้ split/where/join จัดการสแลชซ้อนโดยไม่ต้องไล่เงื่อนไข
  • ConnectionProfile เก็บส่วนประกอบ ไม่เก็บ URL ที่คำนวณได้
  • basePath เก็บดิบตามที่พิมพ์ normalize ตอนใช้
  • fromJson คืน null แทนโยน — แถวเสียหนึ่งไม่ทำให้เสียทั้งหมด
  • แยกให้ออกว่าฟิลด์ไหนคือแก่น (ทิ้งแถว) ฟิลด์ไหนเดาแทนได้
  • validateConnectionInput รับสตริงล้วนและรับ translate เข้ามา จึงบริสุทธิ์และไม่ขึ้นกับภาษา
  • ข้อความผิดพลาดบอกว่าค่านั้นควรไปช่องไหน ไม่ใช่แค่บอกว่าผิด
  • เก็บทุกอย่างในคีย์เดียวเพื่อรักษาข้อตกลง “มีตัวที่เลือกอยู่หนึ่งตัวเสมอ”
  • decode รับประกันว่าทุกทางออกให้สถานะที่ใช้ได้
