1. ปัญหาที่บทนี้แก้

ใน 1.10 เราสรุปว่าตัวชี้ขาดของความลื่นคือ ขอบเขตของ rebuild แต่ยังไม่ได้บอกว่าจะควบคุมมันอย่างไร

ปัญหาของ setState คือมันวาดใหม่ทั้ง build ของ State นั้น ถ้าหน้าจอมีรายการร้อยแถวและเราแค่อยากอัปเดตตัวเลขบรรทัดเดียว รายการทั้งร้อยแถวก็ถูกวาดใหม่ไปด้วย

ทบทวนจาก D1.16: Stream คือสายพานที่ส่งค่าทยอยมา — ValueNotifier ก็คล้ายกัน แต่เรียบง่ายกว่ามาก

setStateValueNotifier + builder

4. ดูของจริงใน lab

เปิด lib/widget_basics/widget_basics_app.dart

// ประกาศกล่องเก็บค่า
final ValueNotifier<int> _tapCount = ValueNotifier<int>(0);

@override
void dispose() {
  _tapCount.dispose();   // ใครสร้าง คนนั้นต้องคืน (1.6)
  super.dispose();
}
// ส่วนนี้เท่านั้นที่ rebuild เมื่อ _tapCount เปลี่ยน
ValueListenableBuilder<int>(
  valueListenable: _tapCount,
  builder: (context, count, child) {
    return Text('กดไปแล้ว $count ครั้ง', key: const Key('tap-count'));
  },
),
// เปลี่ยนค่า — ไม่ต้องเรียก setState เลย
_tapCount.value++;

ผลคือเมื่อกดงานหนึ่งใบ จะมีสองอย่างเกิดขึ้นพร้อมกันแต่คนละขอบเขต

สิ่งที่เปลี่ยนวิธีแจ้งใครวาดใหม่
งานใบไหนเสร็จแล้วsetStateทั้งหน้าจอ (เพราะการ์ดสรุปกับรายการต้องเปลี่ยนพร้อมกัน)
จำนวนครั้งที่กด_tapCount.value++เฉพาะข้อความบรรทัดเดียว

ตัวอย่างนี้จงใจใช้ทั้งสองแบบในหน้าจอเดียว เพื่อให้เห็นว่าเลือกใช้ตามความหมายของข้อมูล ไม่ใช่เลือกอันใดอันหนึ่งไปทั้งไฟล์

5. เมื่อมีหลายค่า ใช้ ListenableBuilder

ValueNotifier เก็บได้ค่าเดียว ถ้าต้องการหลายค่าที่เปลี่ยนพร้อมกัน ให้สร้าง class ที่แจ้งได้เอง

class ScanCounter extends ChangeNotifier {
  int _scanned = 0;
  int _rejected = 0;

  int get scanned => _scanned;
  int get rejected => _rejected;

  void accept() {
    _scanned++;
    notifyListeners();      // แจ้งครั้งเดียวหลังเปลี่ยนครบ
  }

  void reject() {
    _rejected++;
    notifyListeners();
  }
}
ListenableBuilder(
  listenable: counter,
  builder: (context, child) => Text(
    'สแกน ${counter.scanned} ปฏิเสธ ${counter.rejected}',
  ),
)
กฎที่ห้ามลืม: ถ้าเปลี่ยนค่าแล้วไม่เรียก notifyListeners() หน้าจอจะไม่อัปเดต และไม่มี error อะไรเตือนเลย — เป็นสาเหตุอันดับหนึ่งของอาการ “ข้อมูลเปลี่ยนแล้วแต่จอไม่เปลี่ยน” เมื่อเริ่มใช้รูปแบบนี้ ทางกันคือทำ field เป็น private แล้วเปิดเฉพาะ getter กับ method ตามหลัก D1.7

ถ้าเปลี่ยนหลายค่าในการกระทำเดียว ให้เปลี่ยนให้ครบก่อนแล้วค่อยแจ้งครั้งเดียว ไม่ใช่แจ้งทีละค่า เพราะจะทำให้หน้าจอวาดใหม่หลายรอบและอาจเห็นสถานะกลางที่ไม่ถูกต้อง

6. child: ส่วนที่ไม่ต้องวาดใหม่

ValueListenableBuilder<int>(
  valueListenable: _tapCount,
  child: const Icon(Icons.touch_app),      // สร้างครั้งเดียว
  builder: (context, count, child) => Row(
    children: [
      child!,                              // ใช้ตัวเดิมซ้ำ ไม่สร้างใหม่
      Text('กดไปแล้ว $count ครั้ง'),
    ],
  ),
)

สิ่งที่ส่งผ่าน child จะถูกสร้างครั้งเดียวและส่งเข้ามาให้ใช้ซ้ำทุกครั้งที่วาดใหม่ ใช้เมื่อมีส่วนที่ไม่ขึ้นกับค่าที่ฟังอยู่และสร้างแพง

ในทางปฏิบัติ ถ้าส่วนนั้นเป็น const ได้อยู่แล้ว ประโยชน์จะน้อยมาก — ให้ใช้เมื่อวัดแล้วเห็นว่าคุ้มจริงตามลำดับใน 1.10

7. builder ไม่ได้คืนทรัพยากรให้

สิ่งที่ builder จัดการให้สิ่งที่เราต้องจัดการเอง
เริ่มฟังเมื่อเข้าหน้าจอสร้าง ValueNotifier
เลิกฟังเมื่อออกจากหน้าจอdispose ตัว ValueNotifier

เชื่อมกับกฎความเป็นเจ้าของใน 1.6 โดยตรง

  • ถ้า ValueNotifier ถูกสร้างใน State ของเรา → เราต้อง dispose
  • ถ้ามันถูกส่งเข้ามาจากข้างนอก → เราห้าม dispose

8. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
เปลี่ยนค่าแล้วหน้าจอไม่อัปเดตลืม notifyListeners()เปิดเฉพาะ method ที่แจ้งให้เสมอ
เปลี่ยนค่าใน list แล้วไม่อัปเดตแก้ list ก้อนเดิม .value จึงเท่าเดิมกำหนด list ก้อนใหม่ notifier.value = [...old, x] (D1.5)
หน่วยความจำรั่วเมื่อเข้าออกหน้าจอซ้ำไม่ได้ disposeคืนใน dispose ของ State
A ValueNotifier was used after being disposedคืนของที่ยืมมาจากข้างนอกคืนเฉพาะที่สร้างเอง
ใช้ builder แล้วยังวาดใหม่ทั้งหน้าจอวาง builder ไว้สูงเกินไปย้ายลงมาให้ครอบเฉพาะส่วนที่ใช้ค่านั้น
หน้าจอกระพริบเห็นค่ากลาง ๆเรียก notifyListeners หลายครั้งต่อการกระทำเดียวเปลี่ยนค่าให้ครบก่อน แล้วแจ้งครั้งเดียว

แถวที่สองเป็นกับดักที่เจอบ่อยมากและงงที่สุด เพราะโค้ดดูถูกทุกอย่าง — ValueNotifier ตรวจว่าค่าเปลี่ยนโดยเทียบว่าเป็นก้อนเดียวกันหรือเปล่า เมื่อเราแก้ข้างใน list ก้อนเดิม มันจึงเห็นว่า “ไม่เปลี่ยน” ตรงกับเรื่องที่เรียนใน D1.5 หัวข้อ 7 พอดี

9. แบบฝึกหัด

หน้าจอสแกนมีสามส่วน

  • หัวข้อและคำแนะนำ — ไม่เปลี่ยนเลย
  • จุดสถานะการเชื่อมต่อ — เปลี่ยนทุกไม่กี่วินาที
  • รายการที่สแกนแล้ว 200 แถว — เปลี่ยนเมื่อสแกนใหม่เท่านั้น
  1. ถ้าใช้ setState ทั้งหมด จะเกิดอะไรขึ้นเมื่อสถานะการเชื่อมต่อเปลี่ยน
  2. ควรวาง builder ตรงไหน
  3. เขียนโครงโค้ด
ดูแนวคำตอบและเหตุผล

ข้อ 1

ทุกครั้งที่สถานะการเชื่อมต่อเปลี่ยน — ซึ่งเกิดทุกไม่กี่วินาที — รายการทั้ง 200 แถวจะถูกสร้างคำอธิบายใหม่หมด ทั้งที่ข้อมูลของมันไม่ได้เปลี่ยนเลยแม้แต่แถวเดียว บนเดสก์ท็อปอาจไม่รู้สึก แต่บน PDA ที่ CPU ช้ากว่ามาก จะเห็นรายการกระตุกเป็นจังหวะขณะเลื่อน

ข้อ 2

วาง builder ให้ครอบ เฉพาะจุดสถานะ เท่านั้น ส่วนหัวข้อไม่ต้องอยู่ในนั้นเพราะไม่เปลี่ยน และรายการก็ไม่ต้องอยู่ในนั้นเพราะเปลี่ยนคนละจังหวะ

class _ScanScreenState extends State<ScanScreen> {
  final ValueNotifier<bool> _online = ValueNotifier<bool>(true);
  List<String> _scanned = <String>[];

  @override
  void dispose() {
    _online.dispose();
    super.dispose();
  }

  void _addScan(String code) {
    setState(() => _scanned = [..._scanned, code]);   // รายการเปลี่ยนจริง
  }

  @override
  Widget build(BuildContext context) {
    return Column(
      crossAxisAlignment: CrossAxisAlignment.stretch,
      children: [
        // ไม่เปลี่ยนเลย — เป็น const ได้
        const Padding(
          padding: EdgeInsets.all(16),
          child: Text('สแกนบาร์โค้ดเพื่อบันทึก'),
        ),

        // เปลี่ยนบ่อย แต่กระทบแค่จุดเดียว
        ValueListenableBuilder<bool>(
          valueListenable: _online,
          builder: (context, online, child) => Row(
            children: [
              Icon(
                online ? Icons.wifi : Icons.wifi_off,
                color: online ? Colors.green : Colors.red,
              ),
              const SizedBox(width: 8),
              Text(online ? 'เชื่อมต่ออยู่' : 'ออฟไลน์'),
            ],
          ),
        ),

        // เปลี่ยนเฉพาะตอนสแกนใหม่
        Expanded(
          child: ListView.builder(
            itemCount: _scanned.length,
            itemBuilder: (context, index) => ListTile(
              key: ValueKey<String>(_scanned[index]),
              title: Text(_scanned[index]),
            ),
          ),
        ),
      ],
    );
  }
}

ทำไม _addScan ยังใช้ setState: เพราะรายการเป็นข้อมูลที่กระทบส่วนใหญ่ของหน้าจออยู่แล้ว และเกิดไม่บ่อย (เฉพาะตอนสแกน) การใช้ setState จึงตรงไปตรงมาและอ่านง่ายกว่า — ไม่ใช่ทุกอย่างต้องเป็น ValueNotifier

ทำไมเขียน _scanned = [..._scanned, code] แทน _scanned.add(code): ทั้งสองแบบทำงานได้กับ setState แต่การสร้าง list ก้อนใหม่เป็นนิสัยที่ปลอดภัยกว่า และจำเป็นทันทีถ้าวันหนึ่งย้ายไปใช้ ValueNotifier ตามที่เตือนไว้ในตารางหัวข้อ 8

ผลลัพธ์: เมื่อสถานะเชื่อมต่อเปลี่ยน มีเพียง Row เล็ก ๆ ที่ถูกวาดใหม่ ส่วนรายการ 200 แถวและหัวข้อไม่ถูกแตะเลย — นี่คือความหมายของ “ขอบเขต rebuild” ที่ 1.10 บอกว่าสำคัญกว่า const

ลองต่อ: ใส่ print ไว้ใน itemBuilder ของรายการ แล้วสลับ _online.value ไปมา — จะเห็นว่าไม่มีอะไรพิมพ์ออกมาเลย จากนั้นลองย้ายรายการเข้าไปอยู่ใน builder แล้วทำแบบเดิม เพื่อเห็นความต่างด้วยตา

10. สรุปท้ายบท

  • setState วาดใหม่ทั้ง build ส่วน builder วาดใหม่เฉพาะข้างในตัวเอง
  • ValueNotifier เก็บค่าเดียว ส่วน ChangeNotifier เก็บได้หลายค่า
  • ลืม notifyListeners() = หน้าจอไม่อัปเดตโดยไม่มี error
  • แก้ข้างใน list ก้อนเดิมจะไม่ถูกมองว่าเปลี่ยน — ต้องสร้างก้อนใหม่
  • วาง builder ต่ำที่สุดที่ยังครอบทุกคนที่ต้องใช้ค่านั้น
  • builder เลิกฟังให้ แต่ไม่ได้ dispose ให้ — เจ้าของต้องคืนเอง
  • ไม่ใช่ทุกอย่างต้องเป็น ValueNotifier — เลือกตามความหมายของข้อมูล
