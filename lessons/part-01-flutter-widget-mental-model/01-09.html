1. ปัญหาที่บทนี้แก้

บทก่อน ๆ เตือนเรื่องนี้ไว้หลายครั้งแล้ว — 1.4 บอกว่าห้ามทำงานที่มีผลข้างเคียงใน build, 1.6 บอกว่าอย่ายิง API ใน build บทนี้จะรวบให้เป็นกฎเดียวที่ใช้ตัดสินใจได้ทุกกรณี

เหตุที่ต้องมีกฎ เพราะ build ถูกเรียกเมื่อไหร่ก็ได้ กี่ครั้งก็ได้ และเราควบคุมไม่ได้ โค้ดที่ “เห็นว่ารันครั้งเดียว” ในวันนี้ อาจรันร้อยครั้งพรุ่งนี้เมื่อมีคนเพิ่มอนิเมชันเข้ามา

2. คำศัพท์ที่ต้องรู้ก่อน

คำความหมาย

side effect / ผลข้างเคียงการเปลี่ยนแปลงที่ค้างอยู่หลังฟังก์ชันจบ เช่นบันทึกข้อมูล เริ่มจับเวลา

3. Mental model: build คือการถ่ายรูป ไม่ใช่การลงมือทำ

build ควรตอบคำถามเดียวคือ “ด้วยข้อมูลที่มีอยู่ตอนนี้ หน้าจอควรมีหน้าตาอย่างไร”

การถ่ายรูปห้องไม่ได้ทำให้ห้องเปลี่ยน — ถ่ายกี่ครั้งห้องก็เหมือนเดิม ถ้าการถ่ายรูปทำให้ของในห้องขยับ นั่นแปลว่าเราทำอะไรผิด

คำถามทดสอบถ้าตอบว่าใช่
ถ้าเรียกฟังก์ชันนี้ 100 ครั้งติดกัน จะมีอะไรเสียหายไหมห้ามอยู่ใน build
มันเปลี่ยนอะไรที่อยู่นอกฟังก์ชันไหมห้ามอยู่ใน build
มันแค่คำนวณแล้วคืนค่าใช่ไหมอยู่ใน build ได้
ประโยคเดียวที่ต้องจำ: ถ้าเรียกซ้ำแล้วเสียหาย มันไม่ควรอยู่ใน build

4. สิ่งที่ทำได้และทำไม่ได้

ทำได้ใน buildห้ามทำใน build
คำนวณค่าจากข้อมูลที่มีเรียก API
เลือก widget ตามสถานะบันทึกข้อมูลลงเครื่อง
สร้าง callback ที่จะถูกเรียกทีหลังเริ่มจับเวลา
อ่านธีมและขนาดหน้าจอเริ่มฟังเหตุการณ์
ประกอบ widget เข้าด้วยกันเรียก setState
สร้างตัวควบคุมที่ต้องคืนทีหลัง
// ผิดทั้งหมด
@override
Widget build(BuildContext context) {
  repository.loadTasks();          // ยิง API ทุกครั้งที่วาดใหม่
  _timer = Timer.periodic(...);    // สะสมตัวจับเวลาไปเรื่อย ๆ
  store.addListener(_onChange);    // ผู้ฟังทวีคูณ

อาการที่จะเจอจากโค้ดข้างบน: แอปยิง API รัวเป็นร้อยครั้งต่อวินาที, ข้อความที่พิมพ์หายทุกครั้งที่วาดใหม่, และหน่วยความจำเพิ่มไม่หยุด

5. ย้ายไปไว้ที่ไหน

โค้ดแบบย้ายไปที่
โหลดข้อมูลครั้งแรกinitState
สร้างตัวควบคุม / เริ่มฟังinitState (คู่กับ dispose)
โหลดใหม่เมื่อข้อมูลจากแม่เปลี่ยนdidUpdateWidget
งานที่ต้องใช้ธีมหรือขนาดหน้าจอdidChangeDependencies
งานที่เกิดจากการกดของผู้ใช้callback ของปุ่มนั้น
// ถูก — แยกให้ชัดว่าอะไรเกิดครั้งเดียว อะไรเกิดทุกรอบ
@override
void initState() {
  super.initState();
  _controller = TextEditingController();
  _load();                    // เกิดครั้งเดียว
}

Future<void> _load() async {
  setState(() => _loading = true);
  try {
    final result = await repository.loadTasks();
    if (!mounted) return;
    setState(() => _tasks = result);
  } finally {
    if (mounted) setState(() => _loading = false);
  }
}

@override
Widget build(BuildContext context) {
  // เกิดทุกรอบ แต่แค่อ่านสถานะแล้วเลือกว่าจะแสดงอะไร
  if (_loading) return const CircularProgressIndicator();
  if (_tasks.isEmpty) return const Text('ยังไม่มีงานในกะนี้');
  return ListView(children: [for (final t in _tasks) TaskTile(task: t)]);
}

สังเกตว่า build ไม่ได้ทำอะไรเลยนอกจากอ่านสถานะสามอย่างแล้วเลือกว่าจะแสดงอะไร — เรียกกี่ครั้งก็ได้ผลเหมือนกัน และไม่มีอะไรเสียหาย

6. ทำไม build ถึงถูกเรียกบ่อยกว่าที่คิด

สาเหตุคาดเดาได้ไหม
เราเรียก setState เองได้
widget แม่ถูก build ใหม่ยาก — ขึ้นกับคนอื่น
ธีมหรือภาษาเปลี่ยนยาก
แป้นพิมพ์ขึ้นมาบังจอยาก
หมุนอุปกรณ์ยาก
อนิเมชันกำลังเล่นไม่ได้ — เกิดได้ 60 ครั้งต่อวินาที
แถวสุดท้ายคือเหตุผลว่าทำไมกฎนี้ถึงเข้มงวด — ถ้ามีคนเพิ่มอนิเมชันเข้ามาในหน้าจอที่คุณเขียน API ที่คุณเผลอยิงไว้ใน build จะกลายเป็น 60 ครั้งต่อวินาที และคุณจะไม่รู้ตัวจนกว่าจะโดนบิลค่าเซิร์ฟเวอร์

7. งานคำนวณที่หนัก

บางอย่าง “ไม่มีผลข้างเคียง” แต่ก็ไม่ควรอยู่ใน build เพราะช้า

@override
Widget build(BuildContext context) {
  // pure แต่ช้า — ถ้ามีหมื่นรายการ จะทำใหม่ทุกครั้งที่วาด
  final sorted = allItems.where(...).toList()..sort(...);
  return ListView(...);
}

ลำดับการแก้ตามที่เรียนใน D1.17

  1. วัดก่อนว่าช้าจริงไหม — รายการ 20 แถวไม่มีปัญหาแน่นอน
  2. ถ้าช้าจริง ให้คำนวณตอนที่ข้อมูลเปลี่ยนแล้วเก็บผลไว้ ไม่ใช่คำนวณตอนวาด
  3. ใช้รายการแบบสร้างทีละแถว (ListView.builder) แทนการสร้างทุกแถวพร้อมกัน

8. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
API ถูกเรียกซ้ำไม่หยุดเรียกใน buildย้ายไป initState
setState() called during buildเรียก setState ใน buildย้ายไป callback หรือ lifecycle
ข้อความที่พิมพ์หายทุกครั้งที่วาดใหม่สร้างตัวควบคุมใน buildย้ายไปเป็น field ของ State
ผู้ฟังทำงานซ้ำหลายรอบต่อหนึ่งเหตุการณ์เริ่มฟังใน buildย้ายไป initState
ห่อด้วยการเลื่อนไปทำทีหลังแล้วยังซ้ำอยู่ถ้า build ซ้ำ การเลื่อนก็ซ้ำแก้ที่ต้นเหตุ ให้เจ้าของงานเป็น lifecycle
โหลดข้อมูลใหม่ทุกครั้งที่หน้าจอวาด ทั้งที่ใช้ FutureBuilderสร้าง Future ใหม่ใน buildเก็บ Future ไว้ใน State

แถวสุดท้ายเป็นกับดักที่พบบ่อยมาก และจะลงรายละเอียดพร้อมตัวอย่างเต็มใน 4.7

9. แบบฝึกหัด

หน้าจอรายละเอียดงานรับ taskId จากแม่ และต้องโหลดข้อมูลของงานนั้นจาก API

  1. ต้องโหลดตอนเปิดหน้าจอครั้งแรก
  2. ถ้าแม่ส่ง taskId ใหม่มา ต้องโหลดใหม่
  3. ห้ามโหลดซ้ำเมื่อเพียงแค่ธีมเปลี่ยนหรือหน้าจอวาดใหม่
  4. ระหว่างโหลดต้องแสดงสถานะกำลังโหลด และถ้าล้มเหลวต้องมีปุ่มลองใหม่

จงเขียนโครงว่าโค้ดแต่ละส่วนอยู่ตรงไหน

ดูแนวคำตอบและเหตุผล
class _TaskDetailScreenState extends State<TaskDetailScreen> {
  Task? _task;
  String? _error;
  bool _loading = false;

  @override
  void initState() {
    super.initState();
    _load();                          // ข้อ 1
  }

  @override
  void didUpdateWidget(covariant TaskDetailScreen oldWidget) {
    super.didUpdateWidget(oldWidget);
    if (oldWidget.taskId != widget.taskId) {
      _load();                        // ข้อ 2 — โหลดเฉพาะเมื่อ id เปลี่ยนจริง
    }
  }

  Future<void> _load() async {
    setState(() {
      _loading = true;
      _error = null;
    });
    try {
      final result = await repository.loadTask(widget.taskId);
      if (!mounted) return;
      setState(() => _task = result);
    } on Exception {
      if (!mounted) return;
      setState(() => _error = 'โหลดไม่สำเร็จ');
    } finally {
      if (mounted) setState(() => _loading = false);
    }
  }

  @override
  Widget build(BuildContext context) {
    // ข้อ 3 และ 4 — อ่านสถานะอย่างเดียว ไม่มีผลข้างเคียงเลย
    if (_loading) return const Center(child: CircularProgressIndicator());
    if (_error != null) {
      return Center(
        child: Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            Text(_error!),
            FilledButton(onPressed: _load, child: const Text('ลองอีกครั้ง')),
          ],
        ),
      );
    }
    final task = _task;
    if (task == null) return const Center(child: Text('ไม่พบงานนี้'));
    return TaskDetailView(task: task);
  }
}

ทำไมข้อ 2 ต้องเทียบ oldWidget.taskId != widget.taskId: เพราะ didUpdateWidget ถูกเรียกทุกครั้งที่แม่ build ใหม่ ไม่ใช่เฉพาะตอนที่ค่าเปลี่ยน ถ้าไม่เทียบก่อนก็จะโหลดซ้ำทุกครั้งที่แม่วาดใหม่ ซึ่งแทบไม่ต่างจากการเรียกใน build เลย

ทำไมข้อ 3 ไม่ต้องเขียนโค้ดอะไรเพิ่ม: เพราะเราไม่ได้เรียก _load จาก build ตั้งแต่แรก การที่ธีมเปลี่ยนแล้ว build ทำงานใหม่จึงไม่กระทบอะไร — ข้อกำหนดนี้ได้มาฟรีจากการวางโค้ดถูกที่ ซึ่งเป็นประเด็นหลักของบทนี้

ทำไมปุ่มลองใหม่ส่ง _load ไปตรง ๆ ได้: เพราะมันเป็นการส่ง ตัวฟังก์ชัน ไม่ใช่การเรียก ตามที่เรียนใน D1.11 — ถ้าเขียน onPressed: _load() มันจะเรียกทันทีตอน build ซึ่งกลายเป็นความผิดพลาดแบบเดียวกับที่บทนี้ห้ามพอดี

ทำไม final task = _task; ถึงจำเป็น: เพราะ _task เป็น field ที่เปลี่ยนได้ Dart จึงไม่เลื่อนชนิดให้หลังตรวจ null ต้องคัดลอกลงตัวแปรในเครื่องก่อน — ตรงกับที่เรียนใน D1.13 หัวข้อ 5

ลองต่อ: เพิ่ม print ไว้ต้น _load แล้วสลับโหมดมืด — ควรเห็นว่ามันไม่ถูกเรียกเลย จากนั้นลองย้าย _load() ไปไว้ใน build แล้วทำแบบเดิม เพื่อเห็นความต่างด้วยตาตัวเอง

10. สรุปท้ายบท

  • build คือการถ่ายรูป ไม่ใช่การลงมือทำ
  • ถ้าเรียกซ้ำแล้วเสียหาย มันไม่ควรอยู่ใน build
  • งานครั้งเดียวไป initState, งานตามข้อมูลที่เปลี่ยนไป didUpdateWidget, งานจากผู้ใช้ไป callback
  • didUpdateWidget ต้องเทียบค่าเก่ากับใหม่ก่อนลงมือ
  • build ถูกเรียกบ่อยกว่าที่คิดมาก และเราคาดเดาไม่ได้
  • งานคำนวณหนักก็ไม่ควรอยู่ใน build แม้จะไม่มีผลข้างเคียง
