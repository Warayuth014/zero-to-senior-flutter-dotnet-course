1. ปัญหาที่บทนี้แก้

ใน 1.11 เราแตก TaskTile ออกมาจากหน้าจอ แต่มีคำถามค้างอยู่ — เมื่อผู้ใช้กดที่แถวนั้น ใครเป็นคนเปลี่ยนข้อมูล

ถ้าให้ TaskTile เปลี่ยนเอง มันจะต้องรู้จักที่เก็บข้อมูลของทั้งหน้าจอ และจะใช้ในหน้าจออื่นไม่ได้อีกเลย คำตอบคือ ให้มันแค่บอกว่าเกิดอะไรขึ้น แล้วให้แม่ตัดสินใจ

2. คำศัพท์ที่ต้องรู้ก่อน

คำความหมาย
VoidCallbackชื่อเล่นของ void Function()
ValueChanged<T>ชื่อเล่นของ void Function(T)
unidirectionalข้อมูลไหลลง เหตุการณ์ไหลขึ้น ทางเดียวทั้งคู่
intentเจตนาของผู้ใช้ เช่น “อยากทำเครื่องหมายว่าเสร็จ”

3. Mental model: ข้อมูลไหลลง เหตุการณ์ไหลขึ้น

          ┌──────────────┐
          │  หน้าจอ (แม่) │  ← เก็บข้อมูล ตัดสินใจ
          └──────┬───────┘
     ข้อมูล ↓         ↑ เหตุการณ์
          ┌──────┴───────┐
          │  TaskTile     │  ← แค่แสดงและแจ้ง
          └──────────────┘
ทิศทางผ่านอะไรตัวอย่าง
ลงconstructortask: task, done: true
ขึ้นcallbackonToggle: () => ...
ประโยคเดียวที่ต้องจำ: ลูกบอกว่า “เกิดอะไรขึ้น” ไม่ใช่ “ต้องทำอะไรต่อ” — การตัดสินใจว่าจะทำอะไรเป็นหน้าที่ของเจ้าของข้อมูล

4. ดูของจริงใน lab

class TaskTile extends StatelessWidget {
  const TaskTile({
    super.key,
    required this.task,
    required this.done,
    required this.onToggle,
  });

  final DashboardTask task;
  final bool done;
  final VoidCallback onToggle;

  @override
  Widget build(BuildContext context) {
    return ListTile(
      leading: Icon(done ? Icons.check_circle : Icons.radio_button_unchecked),
      title: Text(task.title),
      onTap: onToggle,        // แค่ส่งต่อ ไม่ตัดสินใจอะไร
    );
  }
}

ฝั่งแม่เป็นคนตัดสินใจว่าการกดนั้นแปลว่าอะไร

TaskTile(
  key: ValueKey<String>(task.id),
  task: task,
  done: _doneIds.contains(task.id),
  onToggle: () => _toggleDone(task.id),   // แม่รู้ว่าต้องทำอะไร
)

void _toggleDone(String id) {
  setState(() {
    if (!_doneIds.remove(id)) _doneIds.add(id);
  });
  _tapCount.value++;
}

สังเกตว่า TaskTile ไม่รู้เลยว่าการกดจะทำให้เกิดการสลับสถานะ การนับจำนวนครั้ง หรือแม้แต่จะเกิดอะไรขึ้นหรือเปล่า — จึงเอาไปใช้ในหน้าจออื่นที่การกดมีความหมายต่างไปได้ทันที

5. ระบุชนิดของ callback ให้ชัด

เขียนแบบผลที่ได้
final VoidCallback onToggle;ชัดเจน — ไม่รับอะไร ไม่คืนอะไร
final ValueChanged<int> onChanged;ชัดเจน — รับจำนวนเต็ม
final Future<void> Function(String) onSubmit;ชัดเจน — เป็นงานที่ต้องรอ
final Function onTap;แย่ — compiler ตรวจอะไรไม่ได้เลย

ทบทวนจาก D1.11: การเขียน Function เปล่า ๆ ทำให้ส่ง argument ผิดจำนวนหรือผิดชนิดได้โดยไม่มีใครเตือน แล้วไปพังตอนผู้ใช้กด

ชนิดของ callback ยังบอกข้อมูลสำคัญให้ผู้ใช้งานด้วย — Future<void> Function() บอกว่านี่เป็นงานที่ใช้เวลา ผู้เรียกจึงรู้ว่าควรแสดงสถานะกำลังทำงาน

6. ห่อ callback เพื่อเพิ่มข้อมูล

// TaskTile ประกาศว่ารับ VoidCallback ธรรมดา
final VoidCallback onToggle;

// แม่ห่อไว้เพื่อผูกกับงานใบที่ถูกต้อง
onToggle: () => _toggleDone(task.id),

นี่คือ closure จาก D1.4 ที่ใช้งานจริง — ฟังก์ชันที่สร้างขึ้นจำค่า task.id ของแถวนั้นไว้ ทำให้แต่ละแถวรู้ว่าตัวเองต้องแจ้งงานใบไหน โดยที่ TaskTile ไม่ต้องรับ id เพิ่ม

ทางเลือกอีกแบบคือให้ลูกส่งข้อมูลกลับมาเอง

final ValueChanged<String> onToggle;   // ลูกส่ง id กลับมา

// ในลูก
onTap: () => onToggle(task.id),

// ในแม่
onToggle: _toggleDone,   // ส่งชื่อ method ไปตรง ๆ ได้เลย
แบบเหมาะเมื่อ
แม่ห่อด้วย closureลูกไม่จำเป็นต้องรู้ว่า id คืออะไร
ลูกส่งค่ากลับลูกมีข้อมูลที่แม่ยังไม่รู้ เช่นข้อความที่ผู้ใช้พิมพ์

7. callback ที่เป็นงานต้องรอ

ปุ่มของ Flutter รับ callback ที่ไม่คืนค่า จึง ไม่รอ ให้งานเสร็จ ผลคือถ้าไม่ระวังจะเกิดการกดซ้ำ

class SubmitButton extends StatelessWidget {
  const SubmitButton({super.key, required this.busy, required this.onSubmit});

  final bool busy;
  final Future<void> Function() onSubmit;

  @override
  Widget build(BuildContext context) {
    return FilledButton(
      onPressed: busy ? null : onSubmit,   // ปิดปุ่มระหว่างทำงาน
      child: busy
          ? const SizedBox(
              width: 16, height: 16,
              child: CircularProgressIndicator(strokeWidth: 2),
            )
          : const Text('บันทึก'),
    );
  }
}
สถานะ busy ต้องมีแหล่งเดียว — ให้แม่เป็นเจ้าของแล้วส่งลงมา ห้ามให้ลูกเก็บ busy ของตัวเองอีกชุด เพราะเมื่อสองค่าไม่ตรงกัน ปุ่มจะกดได้ทั้งที่งานยังไม่เสร็จ ซึ่งเป็นบั๊กที่เกิดเฉพาะตอนเน็ตช้าและทดสอบเจอยาก

ส่วนการกันกดซ้ำที่ฝั่งตรรกะยังต้องมีตามที่เรียนใน D1.15 หัวข้อ 7 — การปิดปุ่มเป็นเพียงการสื่อสารกับผู้ใช้ ไม่ใช่การป้องกันจริง

8. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ปุ่มทำงานเองตั้งแต่หน้าจอเปิดใส่วงเล็บตอนส่ง เช่น onTap: _toggle()ส่งชื่อเปล่า หรือห่อด้วย () => _toggle(x)
ส่ง argument ผิดแล้วไม่มีใครเตือนประกาศชนิดเป็น Function เปล่าระบุให้ครบ เช่น ValueChanged<String>
กดปุ่มรัวแล้วเกิดรายการซ้ำไม่ได้กันการกดซ้ำปิดปุ่มด้วย busy และกันที่ตรรกะด้วย
ปุ่มยังกดได้ทั้งที่กำลังทำงานมีสถานะ busy สองชุดให้แม่เป็นเจ้าของแหล่งเดียว
ลูกไปเรียก API เองให้ลูกตัดสินใจแทนที่จะแจ้งลูกแจ้งเจตนา แม่เรียก API
ต้องส่ง callback ผ่านห้าชั้นเจ้าของข้อมูลอยู่ไกลจากผู้ใช้เกินไปย้ายเจ้าของให้ใกล้ขึ้น หรือใช้เครื่องมือใน 1.14 และ Part 27

9. แบบฝึกหัด

ออกแบบ ScanBox ที่ใช้ได้ทั้งหน้ารับเข้าและหน้าหยิบของ โดย

  1. รับบาร์โค้ดจากผู้ใช้แล้วแจ้งกลับเมื่อกดยืนยัน
  2. ต้องไม่รู้จัก API หรือที่เก็บข้อมูลใด ๆ
  3. รองรับสถานะกำลังทำงาน โดยห้ามเก็บสถานะนั้นเอง
  4. ตัดช่องว่างและแปลงเป็นตัวพิมพ์ใหญ่ก่อนแจ้ง

แล้วเขียนตัวอย่างการใช้จากฝั่งแม่

ดูแนวคำตอบและเหตุผล
class ScanBox extends StatefulWidget {
  const ScanBox({
    super.key,
    required this.busy,
    required this.onSubmitted,
    this.hintText = 'สแกนบาร์โค้ด',
  });

  final bool busy;                                       // รับมา ไม่เก็บเอง
  final Future<void> Function(String barcode) onSubmitted;
  final String hintText;

  @override
  State<ScanBox> createState() => _ScanBoxState();
}

class _ScanBoxState extends State<ScanBox> {
  final _controller = TextEditingController();

  @override
  void dispose() {
    _controller.dispose();     // ของที่เราสร้าง เราคืน (1.6)
    super.dispose();
  }

  void _submit() {
    final code = _controller.text.trim().toUpperCase();
    if (code.isEmpty) return;
    _controller.clear();
    widget.onSubmitted(code);   // แจ้งขึ้นไป ไม่ตัดสินใจเอง
  }

  @override
  Widget build(BuildContext context) {
    return Row(
      children: [
        Expanded(
          child: TextField(
            controller: _controller,
            enabled: !widget.busy,
            decoration: InputDecoration(hintText: widget.hintText),
            onSubmitted: (_) => _submit(),
          ),
        ),
        const SizedBox(width: 8),
        FilledButton(
          onPressed: widget.busy ? null : _submit,
          child: const Text('ยืนยัน'),
        ),
      ],
    );
  }
}
// ฝั่งแม่ — หน้ารับเข้า
ScanBox(
  busy: _submitting,
  onSubmitted: (code) => _receive(code),
)

// ฝั่งแม่ — หน้าหยิบของ ใช้ widget ตัวเดียวกัน คนละความหมาย
ScanBox(
  busy: _submitting,
  hintText: 'สแกนพาเลทที่จะหยิบ',
  onSubmitted: (code) => _pick(code),
)

ทำไม ScanBox ต้องเป็น StatefulWidget: เพราะมันเป็นเจ้าของ TextEditingController ซึ่งต้องคืนใน dispose — นี่คือ “มีของที่ต้องดูแลเอง” ตามเกณฑ์ใน 1.11 ส่วนสถานะ busy ไม่ใช่ของมัน จึงรับมาจากแม่

ทำไมการตัดช่องว่างถึงทำในลูกได้: เพราะมันเป็นการจัดรูปแบบข้อมูลนำเข้าขั้นพื้นฐาน ไม่ใช่กฎทางธุรกิจ — ส่วนกฎอย่าง “บาร์โค้ดต้องขึ้นต้นด้วย PAL-” หรือ “ห้ามซ้ำในกะนี้” ต้องอยู่ที่แม่ เพราะแต่ละหน้าจอมีกฎต่างกัน

ทำไมล้างช่องกรอกก่อนแจ้ง: เพราะบน PDA เครื่องสแกนยิงต่อเนื่อง ถ้าไม่ล้าง บาร์โค้ดถัดไปจะต่อท้ายอันเดิม การล้างทันทีทำให้พร้อมรับตัวถัดไปเสมอ — เป็นรายละเอียดที่รู้ได้จากการเข้าใจอุปกรณ์ปลายทาง และจะลงลึกใน Part 3

จุดที่คนมักพลาด: ให้ ScanBox เก็บ bool _busy ของตัวเองแล้วตั้งค่าเมื่อเรียก onSubmitted ซึ่งดูสมเหตุสมผลแต่จะเพี้ยนทันทีเมื่อแม่มีปุ่มอื่นที่ทำให้ระบบไม่ว่างด้วย — ค่าสองชุดจะไม่ตรงกัน และปุ่มจะกดได้ทั้งที่ไม่ควร

ลองต่อ: เขียนเทสต์ที่ประกอบ ScanBox เดี่ยว ๆ พิมพ์ ' pal-1001 ' แล้วตรวจว่า callback ได้รับ 'PAL-1001' — ทำได้โดยไม่ต้องมี API เลย นั่นคือผลตอบแทนของการออกแบบตามบทนี้

10. สรุปท้ายบท

  • ข้อมูลไหลลงผ่าน constructor เหตุการณ์ไหลขึ้นผ่าน callback
  • ลูกบอกว่า “เกิดอะไรขึ้น” ไม่ใช่ “ต้องทำอะไรต่อ”
  • ระบุชนิด callback ให้ชัด อย่าใช้ Function เปล่า
  • ใช้ closure ห่อเพื่อผูก callback กับข้อมูลของแถวนั้น
  • callback ที่เป็นงานต้องรอ ให้ประกาศชนิดเป็น Future เพื่อสื่อสาร
  • สถานะ busy ต้องมีแหล่งเดียว และเป็นของแม่
  • ลูกจัดรูปแบบข้อมูลได้ แต่กฎธุรกิจอยู่ที่แม่

บทถัดไป: 1.13 จะแก้ปัญหาที่เริ่มเห็นแล้วในบทนี้ — เมื่อข้อมูลเปลี่ยน เราจะวาดใหม่เฉพาะส่วนที่เกี่ยวข้องได้อย่างไร โดยไม่ต้อง setState ทั้งหน้าจอ
