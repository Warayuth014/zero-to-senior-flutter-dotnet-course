1. ปัญหาที่บทนี้แก้

ใน 1.5 เราเก็บแค่ตัวเลขไว้ใน State ซึ่งไม่ต้องดูแลอะไร แต่หน้าจอจริงถือของที่ต้องคืน

  • ตัวควบคุมช่องกรอกข้อความ
  • การฟังเหตุการณ์จากเครื่องสแกน
  • ตัวจับเวลาที่ยิงซ้ำทุกวินาที

ถ้าไม่คืนของเหล่านี้ตอนออกจากหน้าจอ จะเกิดอาการที่หาสาเหตุยากมาก — แอปกินหน่วยความจำเพิ่มเรื่อย ๆ หรือโค้ดของหน้าจอเก่ายังทำงานอยู่หลังผู้ใช้ออกไปแล้ว

2. คำศัพท์ที่ต้องรู้ก่อน

คำความหมาย
lifecycleลำดับเหตุการณ์ตั้งแต่ State เกิดจนตาย
initStateเรียกครั้งเดียวตอนเกิด
disposeเรียกครั้งเดียวตอนตาย
didUpdateWidgetเรียกเมื่อแม่ส่งค่าใหม่มา แต่ State ตัวเดิมยังอยู่
mountedบอกว่า State นี้ยังอยู่ในหน้าจอหรือยัง
ownershipใครเป็นเจ้าของของชิ้นนั้น และต้องเป็นคนคืน

3. Mental model: เข้ากะ ทำงาน เลิกกะ

createState()            สร้าง State ครั้งเดียว
      ↓
initState()              เข้ากะ — เตรียมของ เริ่มฟัง
      ↓
build()                  ทำงาน — เกิดซ้ำได้ไม่จำกัดครั้ง
      ↕
didUpdateWidget()        แม่ส่งข้อมูลชุดใหม่มา
      ↓
dispose()                เลิกกะ — คืนของให้ครบ
ประโยคเดียวที่ต้องจำ: ทุกอย่างที่ทำใน initState ต้องมีคู่ตรงข้ามใน dispose — สร้างคู่กับคืน, เริ่มฟังคู่กับเลิกฟัง, เริ่มจับเวลาคู่กับหยุด

4. initState: เตรียมของครั้งเดียว

@override
void initState() {
  super.initState();                      // ต้องเรียกก่อนเสมอ
  _controller = TextEditingController();
  _scanner.addListener(_onScan);
}
เหมาะกับไม่เหมาะกับ
สร้างตัวควบคุมงานที่ต้องใช้ธีมหรือขนาดหน้าจอ (ใช้ didChangeDependencies)
เริ่มฟังเหตุการณ์งานที่ต้องทำซ้ำเมื่อข้อมูลเปลี่ยน
เริ่มโหลดข้อมูลครั้งแรกงานที่ต้องรอผลก่อนวาดหน้าจอ (จะทำให้จอค้าง)

การโหลดข้อมูลครั้งแรกใน initState เป็นรูปแบบที่ถูกต้องและใช้กันทั่วไป โดยไม่ต้องรอผลตรงนั้น

@override
void initState() {
  super.initState();
  _load();            // ปล่อยให้ทำงานไป หน้าจอวาดสถานะกำลังโหลดก่อน
}

Future<void> _load() async {
  final result = await api.fetchTasks();
  if (!mounted) return;             // ผู้ใช้อาจออกไปแล้วระหว่างรอ
  setState(() => _tasks = result);
}

5. dispose: คืนของให้ครบ

@override
void dispose() {
  _scanner.removeListener(_onScan);   // คู่กับ addListener
  _controller.dispose();              // คู่กับ TextEditingController()
  _timer?.cancel();                   // คู่กับ Timer.periodic
  super.dispose();                    // เรียกท้ายสุดเสมอ
}

สังเกตลำดับ: super.initState() เรียก ก่อน ส่วน super.dispose() เรียก ท้ายสุด — เหมือนการเข้าและออกจากห้อง เข้าต้องเปิดประตูก่อน ออกต้องปิดประตูทีหลัง

ดูตัวอย่างจริงใน lab widget_basics_app.dart

final ValueNotifier<int> _tapCount = ValueNotifier<int>(0);

@override
void dispose() {
  _tapCount.dispose();   // ใครสร้าง คนนั้นต้องคืน
  super.dispose();
}

6. กฎความเป็นเจ้าของ

คำถามสำคัญเมื่อจะเขียน dispose: ของชิ้นนี้เป็นของเราหรือแค่ยืมมา

สถานการณ์ใครคืน
เราสร้างเองใน initStateเรา ต้องคืนใน dispose
แม่ส่งเข้ามาผ่าน constructorแม่ — เราห้ามคืน
เราเป็นคนเริ่มฟังเรา ต้องเลิกฟัง
อาการเมื่อคืนของที่ยืมมา: จะได้ error ว่า A TextEditingController was used after being disposed ตอนที่แม่หรือหน้าจออื่นพยายามใช้มันต่อ — เป็นบั๊กที่โผล่ห่างจากจุดที่เขียนผิดมาก จึงหายาก

รูปแบบที่รองรับทั้งสองกรณี — รับตัวควบคุมจากภายนอกได้ ถ้าไม่ส่งมาก็สร้างเอง

TextEditingController? _owned;

TextEditingController get _controller => widget.controller ?? _owned!;

@override
void initState() {
  super.initState();
  if (widget.controller == null) _owned = TextEditingController();
}

@override
void dispose() {
  _owned?.dispose();    // คืนเฉพาะตัวที่เราสร้างเอง
  super.dispose();
}

7. didUpdateWidget: เมื่อแม่ส่งของชิ้นใหม่มา

ทบทวนจาก 1.3: เมื่อแม่ build ใหม่ widget เป็น object ใหม่ แต่ State ตัวเดิมยังอยู่ ปัญหาเกิดเมื่อ State ไปฟังสิ่งที่แม่ส่งมา

@override
void didUpdateWidget(covariant ScanPanel oldWidget) {
  super.didUpdateWidget(oldWidget);
  if (oldWidget.source != widget.source) {
    oldWidget.source.removeListener(_sync);   // เลิกฟังตัวเก่า
    widget.source.addListener(_sync);         // เริ่มฟังตัวใหม่
  }
}

ถ้าไม่เขียนส่วนนี้ เราจะยังฟังตัวเก่าอยู่ทั้งที่แม่เปลี่ยนไปใช้ตัวใหม่แล้ว ผลคือหน้าจอไม่อัปเดต และตัวเก่าถูกยึดไว้ไม่ให้คืนหน่วยความจำ

คำว่า covariant เป็นเพียงการบอก Dart ว่าเรารับ parameter ชนิดที่แคบกว่าที่แม่ประกาศไว้ ไม่ต้องคิดมาก เขียนตามรูปแบบนี้ได้เลย

8. mounted: กันการทำงานหลังออกจากหน้าจอ

นี่คือปัญหาที่ค้างไว้ตั้งแต่ 1.5 และ D1.15 — โลกเปลี่ยนไปแล้วระหว่างที่เรารอ

Future<void> _submit() async {
  final result = await api.submit(code);
  if (!mounted) return;              // ผู้ใช้กดย้อนกลับไปแล้ว หยุดตรงนี้
  setState(() => _message = result.message);
}
สิ่งที่เกิดถ้าไม่ตรวจข้อความ error
เรียก setState หลังหน้าจอถูกทำลายsetState() called after dispose()
ใช้ context ที่หมดอายุerror เกี่ยวกับ deactivated widget's ancestor
อย่ามอง if (!mounted) return เป็นพิธีกรรมที่ใส่เพื่อให้ analyzer เงียบ ให้ใส่ ทุกครั้งที่มี await คั่นก่อนจะแตะหน้าจอ เพราะระหว่างนั้นผู้ใช้ทำอะไรก็ได้ — เรื่องนี้จะลงลึกอีกครั้งใน 3.13 และ 4.9

9. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
แอปกินหน่วยความจำเพิ่มเรื่อย ๆ เมื่อเข้าออกหน้าจอซ้ำลืมคืนของใน disposeไล่ดูว่าทุก addListener/Timer/Controller มีคู่ครบ
setState() called after dispose()งานที่รออยู่ทำงานเสร็จหลังออกจากหน้าจอใส่ if (!mounted) return; หลัง await
A TextEditingController was used after being disposedคืนของที่ยืมมาจากแม่คืนเฉพาะที่สร้างเอง
ยิง API ซ้ำหลายรอบโดยไม่ตั้งใจเริ่มโหลดใน buildย้ายไป initState
แม่เปลี่ยนข้อมูลแล้วหน้าจอไม่ตามไม่ได้เขียน didUpdateWidgetเทียบค่าเก่ากับใหม่แล้วย้ายการฟัง
คัดลอกค่าจากแม่มาเก็บใน initState แล้วค่าไม่อัปเดตinitState ทำงานครั้งเดียวอ่านจาก widget.xxx ตรง ๆ หรือซิงก์ใน didUpdateWidget

10. แบบฝึกหัด

สร้าง StatefulWidget ชื่อ ShiftClock ที่

  1. แสดงจำนวนวินาทีที่ผ่านไปตั้งแต่เปิดหน้าจอ
  2. อัปเดตทุกหนึ่งวินาที
  3. ต้องไม่ทำงานต่อหลังผู้ใช้ออกจากหน้าจอ
  4. รับ onTick แบบไม่บังคับ ที่ถูกเรียกทุกวินาทีพร้อมค่าปัจจุบัน

แล้วตอบว่าจะพิสูจน์ได้อย่างไรว่าข้อ 3 ทำงานจริง

ดูแนวคำตอบและเหตุผล
import 'dart:async';
import 'package:flutter/material.dart';

class ShiftClock extends StatefulWidget {
  const ShiftClock({super.key, this.onTick});

  final void Function(int seconds)? onTick;

  @override
  State<ShiftClock> createState() => _ShiftClockState();
}

class _ShiftClockState extends State<ShiftClock> {
  Timer? _timer;
  int _seconds = 0;

  @override
  void initState() {
    super.initState();
    _timer = Timer.periodic(const Duration(seconds: 1), (_) {
      if (!mounted) return;
      setState(() => _seconds++);
      widget.onTick?.call(_seconds);    // อ่านจาก widget เสมอ ไม่คัดลอกมาเก็บ
    });
  }

  @override
  void dispose() {
    _timer?.cancel();      // คู่ตรงข้ามของ Timer.periodic
    super.dispose();
  }

  @override
  Widget build(BuildContext context) => Text('ผ่านไป $_seconds วินาที');
}

ทำไมต้องเก็บ Timer ไว้ในตัวแปร: เพราะถ้าไม่เก็บ เราจะไม่มีทางยกเลิกมันได้ใน dispose ตัวจับเวลาจะยิงต่อไปเรื่อย ๆ ตลอดอายุแอป และแต่ละครั้งที่ผู้ใช้เข้าหน้าจอนี้ใหม่ก็จะเพิ่มอีกตัวสะสมไปเรื่อย ๆ

ทำไมมี if (!mounted) return; ทั้งที่มี cancel แล้ว: เพราะมีช่วงเวลาสั้น ๆ ที่ตัวจับเวลาอาจยิงพอดีระหว่างที่หน้าจอกำลังถูกทำลาย การป้องกันสองชั้นทำให้ไม่พังในกรณีที่จังหวะไม่เข้าข้างเรา ซึ่งบนอุปกรณ์ช้าอย่าง PDA เกิดได้จริง

ทำไมเรียก widget.onTick ไม่ใช่คัดลอกมาเก็บใน initState: เพราะแม่อาจส่ง callback ตัวใหม่มาระหว่างทาง ถ้าเราคัดลอกไว้ตอนเริ่ม เราจะเรียกตัวเก่าตลอดไป — การอ่านจาก widget. ทุกครั้งได้ค่าล่าสุดเสมอโดยไม่ต้องเขียน didUpdateWidget

ข้อ 5 — พิสูจน์อย่างไร

วิธีที่ตรงที่สุดคือเขียนเทสต์ที่ประกอบ widget ขึ้นมา ปล่อยเวลาผ่านไป ถอด widget ออก แล้วปล่อยเวลาผ่านไปอีก จากนั้นตรวจว่าจำนวนครั้งที่ onTick ถูกเรียกหยุดนิ่ง

testWidgets('หยุดทำงานหลังถูกถอดออกจากหน้าจอ', (tester) async {
  var ticks = 0;
  await tester.pumpWidget(MaterialApp(
    home: ShiftClock(onTick: (_) => ticks++),
  ));

  await tester.pump(const Duration(seconds: 3));
  expect(ticks, 3);

  await tester.pumpWidget(const MaterialApp(home: SizedBox()));  // ถอดออก
  await tester.pump(const Duration(seconds: 3));
  expect(ticks, 3);      // ต้องไม่เพิ่ม
});

ถ้าลืม _timer?.cancel() เทสต์นี้จะฟ้องทันที — และถ้าไม่มีเทสต์ คุณจะรู้ตัวก็ต่อเมื่อแอปทำงานยาว ๆ แล้วเริ่มหน่วง ซึ่งสายเกินไป

ลองต่อ: ลบบรรทัด _timer?.cancel() ออกแล้วรันเทสต์ดู เพื่อยืนยันด้วยตาว่าเทสต์จับได้จริง จากนั้นใส่กลับ

11. สรุปท้ายบท

  • initState เตรียมของครั้งเดียว, dispose คืนของครั้งเดียว
  • ทุกอย่างที่เริ่มไว้ต้องมีคู่ตรงข้ามที่ปิด
  • super.initState() เรียกก่อน, super.dispose() เรียกท้ายสุด
  • คืนเฉพาะของที่เราสร้างเอง อย่าคืนของที่แม่ส่งมา
  • didUpdateWidget ใช้ย้ายการฟังเมื่อแม่ส่งของชิ้นใหม่มา
  • ใส่ if (!mounted) return; ทุกครั้งที่มี await คั่นก่อนแตะหน้าจอ
  • อ่านค่าจากแม่ผ่าน widget.xxx เสมอ อย่าคัดลอกมาเก็บ
