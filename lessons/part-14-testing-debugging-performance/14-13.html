1. ปัญหาที่เทสต์กลุ่มนี้แก้

บน PDA มีข้อกำหนดที่ต่างจากแอปทั่วไป และผิดแล้วผู้ใช้ทำงานไม่ได้

ข้อกำหนดผิดแล้วเกิดอะไร
ช่องสแกนต้องอยู่ที่เดิมเสมอเลื่อนรายการแล้วช่องหาย สแกนไม่เข้า
ทั้งจอต้องพอดีโดยไม่ต้องเลื่อนทั้งหน้าปุ่มยืนยันอยู่ล่างจอ ต้องเลื่อนหาทุกครั้ง
เนื้อหาต้องไม่ล้นได้แถบเหลืองดำและกดปุ่มไม่ได้
ปุ่มต้องกดโดนแม้ใส่ถุงมือกดพลาดบ่อย

ทั้งสี่ข้อมองด้วยตาบนคอมแล้วดูปกติ เพราะจอคอมกว้างและสูงกว่าเครื่องจริงมาก

2. กำหนดขนาดจอให้ตรงกับของจริง

void useViewport(WidgetTester tester, {double height = 560}) {
  tester.view.devicePixelRatio = 1;
  tester.view.physicalSize = Size(390, height);
  addTearDown(tester.view.reset);
}
ค่าปริยายของเทสต์PDA จริง
800×600390×560 หรือเล็กกว่า

ความต่าง 240 พิกเซลในแนวตั้งคือความต่างระหว่าง “พอดี” กับ “ล้น” — และเป็นเหตุผลที่เทสต์ layout ที่ไม่กำหนดขนาดจอไม่มีความหมายเลย

devicePixelRatio = 1 ทำให้ physicalSize เท่ากับหน่วยที่ layout ใช้พอดี ถ้าตั้งเป็น 2 ต้องใส่ Size(780, 1120) เพื่อให้ได้จอ 390×560 ซึ่งชวนสับสน

4. tester.takeException()

expect(tester.takeException(), isNull);

บรรทัดสั้น ๆ นี้ตรวจสิ่งที่สำคัญมาก — ไม่มีเนื้อหาล้น

Flutter โยนข้อผิดพลาดเมื่อ layout ล้นในโหมด debug แต่ในเทสต์มันจะถูกเก็บไว้แทนที่จะทำให้เทสต์ล้มทันที takeException() คือการหยิบมันออกมาดู

for (final scale in [1.0, 1.15, 1.3]) {
  testWidgets('ใช้งานได้ที่ขนาดตัวอักษร $scale เท่า', (tester) async {
    useViewport(tester);
    await tester.pumpWidget(MediaQuery(
      data: MediaQueryData(textScaler: TextScaler.linear(scale)),
      child: wrap(const ReceivingScreen()),
    ));
    await tester.pumpAndSettle();

    expect(tester.takeException(), isNull);
  });
}

7. ทดสอบพื้นที่กด

testWidgets('ปุ่มสลับภาษากดโดนแม้ใส่ถุงมือ', (tester) async {
  await pumpApp(tester);

  final size = tester.getSize(find.byKey(const Key('language-button')));
  expect(size.height, greaterThanOrEqualTo(PdaMetrics.minTapTarget));
  expect(size.width, greaterThanOrEqualTo(PdaMetrics.minTapTarget));
});

ตรวจสิ่งที่ตามองแล้วบอกไม่ได้ — ปุ่มไอคอนที่ดูใหญ่พอ อาจมีพื้นที่กดจริงแค่ 24×24 ถ้าห่อด้วย GestureDetector แทน IconButton (13.10)

เขียนเป็นฟังก์ชันช่วยแล้วใช้กับทุกปุ่มในจอได้

void expectTappable(WidgetTester tester, Finder finder, {String? label}) {
  final size = tester.getSize(finder);
  expect(size.height, greaterThanOrEqualTo(48), reason: label);
  expect(size.width, greaterThanOrEqualTo(48), reason: label);
}

// แล้วตรวจทุกปุ่มในจอ
for (final key in ['confirm', 'cancel', 'scan', 'back']) {
  expectTappable(tester, find.byKey(Key(key)), label: key);
}

8. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
เทสต์ layout ผ่านแต่เครื่องจริงล้นไม่ได้กำหนดขนาดจอtester.view.physicalSize
เทสต์ล้มเมื่อรันทั้งไฟล์ลืม addTearDown(tester.view.reset)ใส่
ไม่จับการล้นลืม takeException()ใส่ทุกเทสต์ layout
เทสต์ผ่านเพราะข้อมูลน้อยเกินใส่แค่สองรายการใส่จนล้น
ล้นเฉพาะภาษาไทยทดสอบด้วยอังกฤษอย่างเดียวทดสอบทั้งสองภาษา (11.4)
ล้นเมื่อผู้ใช้ตั้งตัวอักษรใหญ่ไม่ได้ทดสอบ textScalerวนทดสอบหลายขนาด

9. แบบฝึกหัด

จอ “นับสต๊อก” มีช่องสแกน รายการที่นับแล้ว และปุ่มยืนยัน เขียนรายการเทสต์ layout ที่ควรมี

แนวคำตอบ
เทสต์จับอะไร
ช่องสแกนอยู่ที่เดิมเมื่อเลื่อนรายการห่อทั้งจอด้วย scroll
ปุ่มยืนยันอยู่ที่เดิมเมื่อเลื่อนรายการเหมือนกัน แต่คนละมุมจอ
ไม่มีการเลื่อนทั้งหน้าผู้ใช้ต้องเห็นทุกอย่างพร้อมกัน
ไม่ล้นเมื่อมี 50 รายการรายการยาวทำให้ layout พัง
ไม่ล้นเมื่อรายการว่างบางครั้ง Expanded ที่ไม่มีลูกทำให้พัง
ไม่ล้นที่ขนาดตัวอักษร 1.3 เท่าผู้ใช้ที่มองไม่ชัด
ไม่ล้นเมื่อชื่อสินค้ายาวมากข้อมูลจริงยาวกว่าที่ทดสอบ
ปุ่มทุกปุ่มกดโดนพื้นที่กดเล็กเกิน

สองข้อที่คนมักลืมคือแถวที่ห้าและเจ็ด

testWidgets('รายการว่าง layout ต้องไม่พัง', (tester) async {
  useViewport(tester);
  await tester.pumpWidget(wrap(CycleCountView(store: emptyStore)));

  expect(tester.takeException(), isNull);
  // และต้องมีข้อความบอก ไม่ใช่หน้าจอเปล่า
  expect(find.byKey(const Key('empty-hint')), findsOneWidget);
});

testWidgets('ชื่อสินค้ายาวมาก ต้องไม่ล้น', (tester) async {
  useViewport(tester);
  final store = storeWith(productName: 'สินค้า' * 30);   // ยาวเกินจริง
  await tester.pumpWidget(wrap(CycleCountView(store: store)));

  expect(tester.takeException(), isNull);
});

ข้อมูลที่ยาวเกินจริงมีค่ามาก เพราะข้อมูลในระบบจริงมักยาวกว่าที่ทีมนึกถึงตอนออกแบบ — ชื่อสินค้าที่มีรายละเอียดครบ หรือชื่อที่ถูกคัดลอกมาจากใบสั่งซื้อ

สิ่งที่ควรทำนอกจากเทสต์ — ถ่ายภาพหน้าจอจากเครื่องจริงมาเทียบกับที่เห็นในเทสต์อย่างน้อยหนึ่งครั้ง เพื่อยืนยันว่าขนาดจอที่ใช้ในเทสต์ตรงกับเครื่องจริง

// รันบนเครื่องจริงเพื่อดูขนาดที่แท้จริง
debugPrint(MediaQuery.sizeOf(context).toString());
debugPrint(MediaQuery.devicePixelRatioOf(context).toString());

ถ้าเลขไม่ตรงกับที่ตั้งในเทสต์ เทสต์ทั้งหมดในกลุ่มนี้จะทดสอบจอที่ไม่มีอยู่จริง

10. สรุป

  • เทสต์ layout ที่ไม่กำหนดขนาดจอ ไม่มีความหมาย
  • จดตำแหน่งก่อนเลื่อน แล้วตรวจว่าไม่ขยับ
  • tester.takeException() จับการล้นที่ตามองไม่เห็น
  • ใส่ข้อมูลมากพอที่จะทำให้เกิดปัญหา
  • ทดสอบทั้งรายการว่าง รายการยาว และข้อความยาวเกินจริง
  • ยืนยันขนาดจอที่ใช้ในเทสต์กับเครื่องจริงอย่างน้อยหนึ่งครั้ง
