1. เทสต์ widget ตัวแรก

testWidgets('TaskTile ผูกกับ id ของงานผ่าน ValueKey', (tester) async {
  await tester.pumpWidget(
    MaterialApp(home: Scaffold(body: TaskTile(task: task('T-1')))),
  );

  expect(find.byKey(const ValueKey('T-1')), findsOneWidget);
});
ส่วนหน้าที่
testWidgetsไม่ใช่ test — ให้ tester มาด้วย
tester.pumpWidgetสร้าง widget แล้ววาดหนึ่งเฟรม
find.byKey(...)หา widget ในต้นไม้ที่วาดแล้ว
findsOneWidgetต้องเจอหนึ่งตัวพอดี
เทสต์ widget รันบนเครื่องคอม ไม่ใช่บนมือถือ — Flutter จำลองการวาดทั้งหมดในหน่วยความจำ จึงเร็วกว่าที่คิด และไม่ต้องมีอุปกรณ์ต่ออยู่

2. ต้องห่ออะไรบ้าง

widget ส่วนใหญ่ต้องการของจากข้างบนต้นไม้ ถ้าไม่ห่อจะได้ข้อผิดพลาดที่อ่านไม่รู้เรื่อง

widget ที่ใช้ต้องห่อด้วยถ้าไม่ห่อ
Text, IconMaterialApp หรือ DirectionalityNo Directionality widget found
ปุ่ม, Card, ListTileMaterialAppNo Material widget found
SnackBar, FloatingActionButtonScaffoldNo Scaffold widget found
ตัวเลือกวันที่MaterialApp ที่มี delegateNo MaterialLocalizations found
widget ที่อ่านธีมMaterialApp(theme: ...)ได้ธีมปริยาย ไม่ใช่ของแอป

วิธีที่ทำให้ไม่ต้องจำ คือเขียนฟังก์ชันห่อไว้ที่เดียวในไฟล์เทสต์

Widget wrap(Widget child) => MaterialApp(home: Scaffold(body: child));

ถ้า widget ต้องการมากกว่านั้น ให้ห่อเพิ่มในตัวเดียวกัน

Widget wrap(Widget child) => LanguageScope(
  controller: controller,
  child: MaterialApp(
    theme: buildPdaTheme(),
    locale: controller.locale,
    localizationsDelegates: const [GlobalMaterialLocalizations.delegate, ...],
    home: Scaffold(body: child),
  ),
);

3. ห่อเท่าที่จำเป็น

ห่อน้อยห่อทั้งแอป
เทสต์เร็วช้ากว่า
ล้มแล้วรู้ว่าอะไรพังล้มแล้วต้องไล่หา
ทดสอบ widget เดี่ยว ๆ ได้ต้องเตรียมของครบทุกอย่าง
ไม่ครอบคลุมการต่อกันครอบคลุม

lab ใช้ทั้งสองแบบตามความเหมาะสม

// ห่อน้อย — ทดสอบ widget เดี่ยว
await tester.pumpWidget(wrap(TaskTile(task: task('T-1'))));

// ห่อทั้งแอป — ทดสอบการต่อกันและการนำทาง
await tester.pumpWidget(PdaApp(controller: controller));

เกณฑ์ที่ใช้ตัดสินคือ เทสต์นี้ทดสอบ widget ตัวนี้ หรือทดสอบว่าหลายตัวต่อกันถูก

4. กำหนดขนาดจอ

ค่าปริยายของเทสต์คือ 800×600 ซึ่งไม่ใช่ขนาดของ PDA จริง

void useViewport(WidgetTester tester, {double height = 560}) {
  tester.view.devicePixelRatio = 1;
  tester.view.physicalSize = Size(390, height);
  addTearDown(tester.view.reset);
}
บรรทัดเหตุผล
devicePixelRatio = 1ทำให้ physicalSize เท่ากับหน่วยที่ใช้ใน layout พอดี คิดเลขง่าย
physicalSize = Size(390, 560)ขนาดจริงของ PDA ที่ใช้งาน
addTearDown(tester.view.reset)ขาดไม่ได้ — ไม่งั้นเทสต์ข้อถัดไปจะได้จอขนาดนี้ต่อ

โค้ดนี้มาจาก scanner_layout_test.dart ของ wms_absolute_mobile จริง และเป็นสิ่งที่ทำให้เทสต์การจัดวางมีความหมาย (14.13)

ลืม addTearDown แล้วจะได้เทสต์ที่ล้มเฉพาะเมื่อรันทั้งไฟล์ แต่ผ่านเมื่อรันเดี่ยว ๆ ซึ่งเป็นอาการที่ทำให้เสียเวลาหาสาเหตุนานมาก

5. pumpWidget วาดกี่เฟรม

await tester.pumpWidget(wrap(child));   // วาดหนึ่งเฟรม
await tester.pump();                     // วาดอีกหนึ่งเฟรม
await tester.pump(const Duration(seconds: 1));   // เดินเวลาไปหนึ่งวินาที แล้ววาด
await tester.pumpAndSettle();            // วาดจนกว่าจะไม่มีอะไรขยับ

ความต่างนี้สำคัญมากและ 14.10 จะพูดถึงทั้งบท ตอนนี้แค่รู้ว่า

อยากทดสอบใช้
สิ่งที่เห็นทันทีหลังสร้างpumpWidget อย่างเดียว
สิ่งที่เห็นระหว่างโหลดpump()
สิ่งที่เห็นหลังทุกอย่างเสร็จpumpAndSettle()
testWidgets('เปิดแอปที่มี session เดิม ต้องไม่เห็นหน้าเข้าสู่ระบบเลย', (tester) async {
  await tester.pumpWidget(AuthLabApp(dependencies: deps));
  // ยังไม่ settle — ดูเฟรมแรกก่อน
  await tester.pump();

  expect(find.byKey(const Key('splash')), findsOneWidget);
  expect(find.byKey(const Key('login-screen')), findsNothing);

  await tester.pumpAndSettle();
  expect(find.byKey(const Key('home-shell')), findsOneWidget);
});

เทสต์นี้จับอาการ “เห็นหน้าเข้าสู่ระบบแวบหนึ่ง” จาก 8.6 ซึ่ง pumpAndSettle อย่างเดียวมองไม่เห็น

6. ใส่ Key ให้สิ่งที่จะหา

FilledButton(
  key: const Key('scan-button'),
  onPressed: () {},
  child: Text(context.t(AppText.scanPallet)),
)
หาด้วยปัญหา
find.text('สแกนพาเลท')พังเมื่อเปลี่ยนคำ หรือเมื่อสลับภาษา
find.byType(FilledButton)เจอหลายตัวถ้ามีปุ่มหลายอัน
find.byKey(const Key('scan-button'))ชัดเจนและไม่พังตามคำ

แต่อย่าใส่ Key ให้ทุกอย่าง — ใส่ให้สิ่งที่เทสต์ต้องหาจริง ซึ่งมักเป็นปุ่ม ช่องกรอก และส่วนที่แสดงสถานะ

ในทางกลับกัน find.text มีที่ใช้เมื่อข้อความคือสิ่งที่ทดสอบ

testWidgets('กดปุ่มเดียวแล้วทั้งจอเปลี่ยนภาษา', (tester) async {
  await tester.tap(find.byKey(const Key('language-button')));
  await tester.pumpAndSettle();

  // ตรงนี้ข้อความคือสิ่งที่ทดสอบ จึงใช้ find.text ถูกแล้ว
  expect(find.text('พาเลท'), findsOneWidget);
  expect(find.text('Pallets'), findsNothing);
});

7. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
No Directionality widget foundไม่ได้ห่อ MaterialAppห่อ
No Material widget foundไม่ได้ห่อ Scaffold หรือ Materialห่อ
เทสต์ล้มเมื่อรันทั้งไฟล์ แต่ผ่านเมื่อรันเดี่ยวลืม addTearDown(tester.view.reset)ใส่
หา widget ไม่เจอทั้งที่เห็นในแอปยังไม่ได้ pump พอดู 14.10
เทสต์พังทุกครั้งที่เปลี่ยนคำหาด้วย find.textใส่ Key
ล้นจอในเทสต์แต่ไม่ล้นในแอปขนาดจอปริยายไม่ตรงกับเครื่องจริงกำหนด physicalSize

8. แบบฝึกหัด

เขียนเทสต์ให้ widget นี้

class QuantityStepper extends StatelessWidget {
  const QuantityStepper({
    super.key,
    required this.value,
    required this.onChanged,
    this.max = 99,
  });

  final int value;
  final ValueChanged<int> onChanged;
  final int max;

  @override
  Widget build(BuildContext context) => Row(
    children: [
      IconButton(
        key: const Key('minus'),
        onPressed: value > 0 ? () => onChanged(value - 1) : null,
        icon: const Icon(Icons.remove),
      ),
      Text('$value', key: const Key('value')),
      IconButton(
        key: const Key('plus'),
        onPressed: value < max ? () => onChanged(value + 1) : null,
        icon: const Icon(Icons.add),
      ),
    ],
  );
}
แนวคำตอบ
Widget wrap(Widget child) => MaterialApp(home: Scaffold(body: child));

testWidgets('แสดงค่าที่ส่งเข้ามา', (tester) async {
  await tester.pumpWidget(wrap(QuantityStepper(value: 5, onChanged: (_) {})));
  expect(find.text('5'), findsOneWidget);
});

testWidgets('กดบวกแล้วแจ้งค่าที่เพิ่มขึ้น', (tester) async {
  int? received;
  await tester.pumpWidget(
    wrap(QuantityStepper(value: 5, onChanged: (v) => received = v)),
  );

  await tester.tap(find.byKey(const Key('plus')));
  expect(received, 6);
});

testWidgets('ที่ศูนย์แล้วกดลบไม่ได้', (tester) async {
  var called = false;
  await tester.pumpWidget(
    wrap(QuantityStepper(value: 0, onChanged: (_) => called = true)),
  );

  await tester.tap(find.byKey(const Key('minus')));
  expect(called, isFalse);
});

testWidgets('ที่ค่าสูงสุดแล้วกดบวกไม่ได้', (tester) async {
  var called = false;
  await tester.pumpWidget(
    wrap(QuantityStepper(value: 99, onChanged: (_) => called = true, max: 99)),
  );

  await tester.tap(find.byKey(const Key('plus')));
  expect(called, isFalse);
});

สามอย่างที่ควรสังเกต

ข้อสังเกตเหตุผล
ไม่ต้อง pump() หลัง tapเพราะตรวจ callback ไม่ได้ตรวจหน้าจอ — widget นี้ไม่ถือ state เอง
ไม่มีเทสต์ว่า “กดบวกแล้วเลขเปลี่ยนเป็น 6”widget นี้ไม่ได้เปลี่ยนเลขเอง คนเรียกต่างหากที่เปลี่ยน — ทดสอบสิ่งที่มันสัญญาไว้ (14.1)
ทดสอบว่าปุ่มกดไม่ได้ ด้วยการกดแล้วดูว่าไม่มีอะไรเกิดตรงกว่าการตรวจว่า onPressed เป็น null ซึ่งเป็นรายละเอียดภายใน

แถวที่สองคือจุดที่คนใหม่มักทำผิด — เขียนเทสต์ที่ตรวจว่าเลขเปลี่ยน แล้วงงว่าทำไมไม่เปลี่ยน ทั้งที่ widget ทำงานถูกต้อง

สิ่งที่ควรทำถ้าอยากทดสอบเรื่องนั้นคือห่อด้วย StatefulBuilder ซึ่งเป็นการจำลองคนเรียก

testWidgets('เมื่อคนเรียกอัปเดตค่า หน้าจอเปลี่ยนตาม', (tester) async {
  var value = 5;
  await tester.pumpWidget(wrap(
    StatefulBuilder(
      builder: (context, setState) => QuantityStepper(
        value: value,
        onChanged: (v) => setState(() => value = v),
      ),
    ),
  ));

  await tester.tap(find.byKey(const Key('plus')));
  await tester.pump();
  expect(find.text('6'), findsOneWidget);
});

9. สรุป

  • testWidgets ให้ tester มาสร้างและกด widget
  • ห่อด้วย MaterialApp และ Scaffold เท่าที่ widget ต้องการ
  • กำหนดขนาดจอให้ตรงกับเครื่องจริง และอย่าลืม addTearDown
  • ใส่ Key ให้สิ่งที่เทสต์ต้องหา
  • find.text ใช้เมื่อข้อความคือสิ่งที่ทดสอบ
  • ทดสอบสิ่งที่ widget สัญญาไว้ ไม่ใช่สิ่งที่คนเรียกทำ
