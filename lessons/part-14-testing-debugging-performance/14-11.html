1. ปัญหาของเวลาจริง

// อย่าเขียนแบบนี้
test('ถามซ้ำทุก 15 วินาที', () async {
  store.startPolling();
  await Future<void>.delayed(const Duration(seconds: 45));   // รอจริง 45 วินาที
  expect(repository.fetchCalls, 3);
});
ปัญหาผล
ช้าเทสต์เดียวใช้ 45 วินาที ชุดเทสต์กลายเป็นชั่วโมง
ไม่แน่นอนเครื่องที่ช้ากว่าอาจได้ 2 หรือ 4 ครั้ง
ไม่มีใครรันเทสต์ที่ช้าจะถูกข้ามในที่สุด

ทางแก้คือปลอมนาฬิกา แทนที่จะรอจริง

2. fakeAsync

dev_dependencies:
  fake_async: ^1.3.1
test('หลุดแล้วเริ่มถามซ้ำ ต่อกลับได้แล้วหยุด', () {
  fakeAsync((async) {
    repository.result = [task('T-1')];
    unawaited(store.start());
    async.flushMicrotasks();
    expect(store.isPolling, isFalse, reason: 'ต่อได้อยู่ ไม่ต้องถามซ้ำ');

    inner.simulateDrop();
    async.flushMicrotasks();
    expect(store.isPolling, isTrue);

    final callsBefore = repository.fetchCalls;
    async.elapse(const Duration(seconds: 45));
    expect(repository.fetchCalls, callsBefore + 3);
    ...
  });
});
fakeAsync แทนที่นาฬิกาและตัวจับเวลาทั้งหมดในบล็อกนั้น — Timer, Future.delayed และ Stopwatch จะใช้เวลาปลอม ทำให้ 45 วินาทีผ่านไปทันที
เมธอดทำอะไรใช้เมื่อ
elapse(duration)เดินเวลาไป ตัวจับเวลาที่ถึงกำหนดทำงานทดสอบสิ่งที่เกิดตามเวลา
flushMicrotasks()ให้ Future ที่พร้อมแล้วทำงานหลังเรียก async ที่ไม่ได้ await
flushTimers()รันตัวจับเวลาที่เหลือทั้งหมดตอนจบ เพื่อไม่ให้ค้าง
elapseBlocking(duration)เดินเวลาโดยไม่ให้อะไรทำงานจำลองเครื่องที่ค้าง

3. unawaited ใน fakeAsync

unawaited(store.start());
async.flushMicrotasks();

ใน fakeAsync จะ await ไม่ได้ เพราะเวลาไม่ได้เดินจริง ต้องเรียกแล้วสั่งให้ microtask ทำงานแทน

ในเทสต์ปกติใน fakeAsync
await store.start();unawaited(store.start()); async.flushMicrotasks();
await Future.delayed(Duration.zero);async.flushMicrotasks();

ถ้าเผลอใช้ async กับฟังก์ชันที่ส่งให้ fakeAsync เทสต์จะค้างหรือผ่านโดยไม่ได้ตรวจอะไร

// ผิด — fakeAsync ไม่รองรับ async function
fakeAsync((async) async { ... });

// ถูก
fakeAsync((async) { ... });

4. จับตัวจับเวลาที่รั่ว

นี่คือสิ่งที่ fakeAsync ทำได้ดีที่สุด และเป็นบั๊กที่มองไม่เห็นด้วยตา (12.8)

test('ทิ้ง store แล้วตัวจับเวลาต้องหยุด', () {
  fakeAsync((async) {
    unawaited(store.start());
    async.flushMicrotasks();
    inner.simulateDrop();
    async.flushMicrotasks();
    expect(store.isPolling, isTrue);

    store.dispose();
    final callsAtDispose = repository.fetchCalls;

    async.elapse(const Duration(minutes: 10));
    expect(repository.fetchCalls, callsAtDispose);
  });
});

สิบนาทีผ่านไปในเสี้ยววินาที และถ้าตัวจับเวลายังเดินอยู่ จำนวนคำขอจะเพิ่มขึ้น 40 ครั้ง — เทสต์จับได้ทันที

เทสต์แบบนี้ควรมีในทุก store ที่มีตัวจับเวลา เพราะการลืม cancel() ไม่มีอาการอะไรให้เห็นเลยตอนพัฒนา

5. ตัวจับเวลาที่ค้างทำให้เทสต์ล้ม

A Timer is still pending even after the widget tree was disposed.

ข้อความนี้มาจาก Flutter ที่ตรวจให้เองในเทสต์ widget — เป็นของขวัญ เพราะมันเปลี่ยนการรั่วที่เงียบเป็นเทสต์ที่ล้ม

ที่มาแก้
ลืม timer.cancel() ใน disposeใส่
สร้างตัวจับเวลาซ้อนกันตรวจก่อนสร้าง (12.11)
ตัวจับเวลาถูกสร้างใน buildย้ายไป initState

ใน fakeAsync ต้องจัดการเอง ด้วยการเรียก flushTimers() ตอนจบ

store.dispose();
async.flushTimers();      // เคลียร์ที่เหลือ ไม่งั้น fakeAsync จะบ่น

6. เวลาที่ต้องส่งเข้ามา

ฟังก์ชันที่อ่านนาฬิกาเองทดสอบไม่ได้

// ทดสอบยาก — ผลเปลี่ยนทุกวินาที
String formatElapsed(DateTime utc) {
  final elapsed = DateTime.now().toUtc().difference(utc);
  ...
}

// ทดสอบง่าย — เทสต์กำหนดเวลาได้
String formatElapsed(DateTime utc, AppLanguage language, {DateTime? now}) {
  final elapsed = (now ?? DateTime.now()).toUtc().difference(utc.toUtc());
  ...
}
test('เวลาที่ผ่านมา เลือกหน่วยตามช่วง', () {
  final now = DateTime.utc(2026, 8, 21, 12, 0);

  expect(
    formatElapsed(now.subtract(const Duration(minutes: 3)),
        AppLanguage.thai, now: now),
    '3 นาทีที่แล้ว',
  );
});

รูปแบบเดียวกับ createCommandId ใน 9.8 — อะไรที่ทำให้เทสต์เดาไม่ได้ ให้เปิดช่องรับเข้ามาพร้อมค่าเริ่มต้น

สิ่งที่ควรรับเข้ามาได้ค่าปริยาย
เวลาปัจจุบันDateTime.now()
ตัวสร้างรหัสอิงเวลา
ช่วงเวลาของตัวจับเวลาค่าที่ใช้จริง
ตัวสุ่มRandom()
ระยะเวลารอก่อนลองใหม่ค่าที่ใช้จริง

7. ทดสอบการลองใหม่แบบถอยหลัง

7.12 ออกแบบการลองใหม่ที่ห่างขึ้นเรื่อย ๆ ซึ่งทดสอบได้ตรง ๆ ด้วย fakeAsync

test('ลองใหม่โดยห่างขึ้นเรื่อย ๆ แต่ไม่เกินเพดาน', () {
  fakeAsync((async) {
    final delays = <Duration>[];
    final client = RetryingClient(
      sleep: (d) {
        delays.add(d);
        return Future<void>.delayed(d);
      },
      random: Random(42),     // สุ่มแบบเดาได้ เพื่อให้ jitter คงที่
    );

    unawaited(client.get('/tasks'));   // ล้มเหลวทุกครั้ง
    async.elapse(const Duration(minutes: 5));

    expect(delays, hasLength(4));
    // ห่างขึ้นเรื่อย ๆ
    for (var i = 1; i < delays.length; i++) {
      expect(delays[i], greaterThan(delays[i - 1]));
    }
    // แต่ไม่เกินเพดาน
    expect(delays.last, lessThanOrEqualTo(const Duration(seconds: 30)));
  });
});

Random(42) คือตัวสุ่มที่มีเมล็ดคงที่ ทำให้ได้ลำดับเดิมทุกครั้ง — จำเป็นเมื่อโค้ดมีการสุ่มแบบ jitter

8. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
เทสต์ใช้เวลาเป็นนาทีรอเวลาจริงfakeAsync
เทสต์ผ่านบ้างไม่ผ่านบ้างพึ่งความเร็วของเครื่องปลอมนาฬิกา
fakeAsync ไม่ทำงานใส่ async ในฟังก์ชันที่ส่งให้เอาออก ใช้ flushMicrotasks
Timer is still pendingลืม cancelยกเลิกใน dispose
fakeAsync บ่นตอนจบมี timer เหลือasync.flushTimers()
ทดสอบเวลาที่ผ่านมาไม่ได้ฟังก์ชันอ่าน DateTime.now() เองรับ now เข้ามา

9. แบบฝึกหัด

ฟีเจอร์ใหม่ต้องล็อกหน้าจอเมื่อไม่มีการใช้งานห้านาที เพื่อไม่ให้คนอื่นหยิบเครื่องไปใช้ต่อ เขียนเทสต์

แนวคำตอบ

ก่อนเขียนเทสต์ ต้องทำให้ทดสอบได้ก่อน

class IdleLockController extends ChangeNotifier {
  IdleLockController({
    this.timeout = const Duration(minutes: 5),
    DateTime Function()? now,
  }) : _now = now ?? DateTime.now;

  final Duration timeout;
  final DateTime Function() _now;
  Timer? _timer;
  bool _locked = false;

  bool get locked => _locked;

  /// เรียกทุกครั้งที่ผู้ใช้แตะหน้าจอหรือสแกน
  void recordActivity() {
    if (_locked) return;
    _timer?.cancel();
    _timer = Timer(timeout, _lock);
  }

  void _lock() {
    _locked = true;
    notifyListeners();
  }

  @override
  void dispose() {
    _timer?.cancel();
    super.dispose();
  }
}
test('ไม่มีการใช้งานครบเวลา ต้องล็อก', () {
  fakeAsync((async) {
    final controller = IdleLockController();
    controller.recordActivity();

    async.elapse(const Duration(minutes: 5));
    expect(controller.locked, isTrue);

    controller.dispose();
    async.flushTimers();
  });
});

test('มีการใช้งานก่อนครบเวลา ต้องนับใหม่', () {
  fakeAsync((async) {
    final controller = IdleLockController();
    controller.recordActivity();

    async.elapse(const Duration(minutes: 4));
    controller.recordActivity();          // แตะตอนนาทีที่ 4
    async.elapse(const Duration(minutes: 4));

    // ผ่านไป 8 นาทีแล้ว แต่แตะล่าสุดเมื่อ 4 นาทีที่แล้ว
    expect(controller.locked, isFalse);

    controller.dispose();
    async.flushTimers();
  });
});

test('ล็อกแล้วการแตะต้องไม่ปลดล็อก', () {
  fakeAsync((async) {
    final controller = IdleLockController();
    controller.recordActivity();
    async.elapse(const Duration(minutes: 5));

    controller.recordActivity();
    async.elapse(const Duration(minutes: 1));

    expect(controller.locked, isTrue, reason: 'ต้องใส่รหัสเท่านั้น');

    controller.dispose();
    async.flushTimers();
  });
});

test('ทิ้ง controller แล้วต้องไม่ล็อกอีก', () {
  fakeAsync((async) {
    final controller = IdleLockController();
    controller.recordActivity();
    controller.dispose();

    // ถ้าลืม cancel บรรทัดนี้จะเรียก notifyListeners บนตัวที่ถูกทิ้งแล้ว
    async.elapse(const Duration(minutes: 10));
  });
});

เทสต์ข้อสุดท้ายคือข้อที่จับการรั่ว และจะล้มด้วย A ChangeNotifier was used after being disposed ถ้าลืม cancel

คำถามที่การเขียนเทสต์ทำให้ต้องถาม

คำถามใครตอบ
ถ้ากำลังสแกนอยู่แล้วครบเวลาพอดี ควรล็อกไหมหัวหน้าคลัง
ถ้ามีงานค้างที่ยังไม่ได้ส่ง ล็อกแล้วงานหายไหมทีมพัฒนา — และคำตอบต้องเป็น “ไม่หาย”
ห้านาทีเหมาะกับคลังที่ต้องเดินไปไกลไหมต้องวัดจากหน้างาน

แถวสุดท้ายเป็นเหตุผลที่ timeout ควรรับเข้ามาได้ — เพื่อให้ปรับได้โดยไม่ต้อง build ใหม่ ตามหลักใน 10.1

10. สรุป

  • อย่ารอเวลาจริงในเทสต์ ให้ปลอมนาฬิกาด้วย fakeAsync
  • ใน fakeAsync ใช้ unawaited กับ flushMicrotasks แทน await
  • เดินเวลาสิบนาทีเพื่อจับตัวจับเวลาที่ลืมยกเลิก
  • เรียก flushTimers() ตอนจบ
  • รับเวลา ตัวสุ่ม และช่วงเวลาเข้ามาพร้อมค่าปริยาย
  • Random(42) ทำให้การสุ่มเดาได้ในเทสต์
