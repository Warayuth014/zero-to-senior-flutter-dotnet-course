2. ตัวนับ

/// นับจำนวนครั้งที่ widget ถูกวาดใหม่ แยกตามป้ายชื่อ
///
/// ประสิทธิภาพเป็นเรื่องที่เถียงกันได้ไม่จบถ้าไม่มีตัวเลข — ตัวนับนี้ทำให้
/// คำว่า "เร็วขึ้น" กลายเป็นสิ่งที่เทสต์ยืนยันได้ แทนที่จะเป็นความรู้สึก
class RebuildCounter {
  final Map<String, int> _counts = {};

  int countOf(String label) => _counts[label] ?? 0;
  void record(String label) => _counts[label] = countOf(label) + 1;
}
class CountedBuilder extends StatelessWidget {
  const CountedBuilder({
    super.key,
    required this.counter,
    required this.label,
    required this.builder,
  });

  @override
  Widget build(BuildContext context) {
    counter.record(label);
    return builder(context);
  }
}
นับใน build เพราะนั่นคือสิ่งที่อยากวัด — จำนวนครั้งที่ Flutter ตัดสินใจวาดส่วนนี้ใหม่ ไม่ใช่จำนวนครั้งที่ setState ถูกเรียก ซึ่งเป็นคนละเรื่อง

3. สองรุ่นที่ต่างกันบรรทัดเดียว

/// รุ่นที่วาดใหม่ทั้งจอทุกครั้งที่มีอะไรเปลี่ยน
class WideRebuildBoard extends StatelessWidget {
  @override
  Widget build(BuildContext context) => ListenableBuilder(
    listenable: state,                    // ← ครอบทั้งจอ
    builder: (context, _) => Column(
      children: [
        CountedBuilder(label: 'header', ...),
        CountedBuilder(label: 'counter', ...),
        CountedBuilder(label: 'expensive', ...),
      ],
    ),
  );
}
/// รุ่นที่วาดใหม่เฉพาะส่วนที่ค่าเปลี่ยนจริง
class NarrowRebuildBoard extends StatelessWidget {
  @override
  Widget build(BuildContext context) => Column(
    children: [
      ListenableBuilder(                  // ← ครอบเฉพาะส่วนนี้
        listenable: state,
        builder: (context, _) => CountedBuilder(label: 'header', ...),
      ),
      ListenableBuilder(
        listenable: state,
        builder: (context, _) => CountedBuilder(label: 'counter', ...),
      ),
      // ไม่ได้อยู่ใต้ ListenableBuilder เลย จึงไม่ถูกวาดใหม่ไม่ว่าอะไรจะเปลี่ยน
      CountedBuilder(label: 'expensive', ...),
    ],
  );
}

ต่างกันที่ตำแหน่งของ ListenableBuilder เท่านั้น และผลต่างวัดได้

4. ตัวเลข

testWidgets('รุ่นที่ครอบทั้งจอ วาดใหม่ทุกส่วนแม้ค่าที่ใช้ไม่เปลี่ยน', (tester) async {
  await tester.pumpWidget(wrap(WideRebuildBoard(state: state, counter: counter)));

  state.scan();
  await tester.pump();

  // ชื่อผู้ปฏิบัติงานไม่ได้เปลี่ยน แต่ยังถูกวาดใหม่
  expect(counter.countOf('header'), 2);
  expect(counter.countOf('counter'), 2);
  // และส่วนที่แพงก็ถูกวาดใหม่ด้วย ทั้งที่ไม่ได้ใช้ค่าอะไรเลย
  expect(counter.countOf('expensive'), 2);
});

testWidgets('รุ่นที่ครอบแคบ ส่วนที่แพงไม่ถูกวาดใหม่', (tester) async {
  await tester.pumpWidget(wrap(NarrowRebuildBoard(state: state, counter: counter)));

  state.scan();
  await tester.pump();

  expect(counter.countOf('counter'), 2);
  expect(counter.countOf('expensive'), 1, reason: 'ไม่ควรถูกวาดใหม่เลย');
});
สแกน 10 ครั้งครอบทั้งจอครอบแคบ
ส่วนที่แพงถูกวาด11 ครั้ง1 ครั้ง
testWidgets('สแกนสิบครั้ง ส่วนที่แพงยังวาดครั้งเดียว', (tester) async {
  for (var i = 0; i < 10; i++) {
    state.scan();
    await tester.pump();
  }

  expect(counter.countOf('counter'), 11);
  expect(counter.countOf('expensive'), 1);
});

ตัวเลขนี้คือสิ่งที่เอาไปคุยกับทีมได้ และเป็นเทสต์ที่จะล้มถ้ามีคนย้าย ListenableBuilder กลับขึ้นไป

5. อีกครึ่งที่สำคัญ — อย่าประกาศทั้งที่ไม่มีอะไรเปลี่ยน

testWidgets('ตั้งค่าเดิมซ้ำ ต้องไม่วาดใหม่เลย', (tester) async {
  await tester.pumpWidget(wrap(NarrowRebuildBoard(state: state, counter: counter)));
  final before = counter.snapshot;

  // ChangeNotifier ที่ไม่ประกาศเมื่อค่าไม่เปลี่ยน (11.7)
  state.changeOperator('somchai');
  await tester.pump();

  expect(counter.snapshot, before);
});
void changeOperator(String name) {
  if (_operator == name) return;    // ← บรรทัดที่ทำให้เทสต์ผ่าน
  _operator = name;
  notifyListeners();
}

บรรทัดเดียวนี้กันการวาดใหม่ทั้งจอ และเป็นสิ่งที่ควรมีในทุก setter ของ store

6. เครื่องมือลดการวาดใหม่

วิธีได้อะไรระวัง
ย้าย ListenableBuilder ลงไปครอบให้แคบได้มากที่สุดต้องรู้ว่าส่วนไหนใช้ค่าอะไร
ใส่ const ให้ widget ที่ไม่เปลี่ยนFlutter ข้ามการสร้างใหม่const Text ไม่เปลี่ยนภาษา (11.6)
ตรวจก่อน notifyListenersกันการวาดที่ไม่จำเป็นตั้งแต่ต้นทางต้องเทียบค่าให้ถูก
แยก widget ที่แพงออกเป็นคลาสของตัวเองทำให้ const ได้—
ใช้ RepaintBoundaryแยกชั้นการวาดกินหน่วยความจำเพิ่ม ใช้เมื่อจำเป็นจริง

สองอันแรกครอบคลุมกรณีส่วนใหญ่ และอันดับความสำคัญคือครอบให้แคบก่อน แล้วค่อยคิดถึงอย่างอื่น

// ทำให้ส่วนที่แพงเป็นคลาสของตัวเอง แล้วมันจะ const ได้
class _ExpensiveSection extends StatelessWidget {
  const _ExpensiveSection();
  ...
}

// แล้วเรียกแบบนี้ Flutter จะข้ามการสร้างใหม่ทั้งก้อน
const _ExpensiveSection()

7. เมื่อไหร่ควรทำ

อย่าปรับก่อนวัด — การวาดใหม่ที่ไม่จำเป็นส่วนใหญ่ไม่มีผลที่ผู้ใช้รู้สึก และการแยก ListenableBuilder เป็นสิบตัวทำให้โค้ดอ่านยากขึ้นโดยไม่ได้อะไร
ควรทำเมื่อไม่ต้องทำเมื่อ
DevTools บอกว่าเฟรมเกิน 16ms (14.14)ยังไม่เคยวัด
ส่วนที่ถูกวาดใหม่มีของแพง เช่นรูปหรือลิสต์เป็นข้อความสองบรรทัด
ผู้ใช้รายงานว่ากระตุกเพราะอ่านมาว่าควรทำ
วาดใหม่เกิดถี่มาก เช่นทุกตัวอักษรที่พิมพ์เกิดปีละครั้ง

ในโค้ดจริงของ wmsapp มีตัวอย่างที่เข้าเกณฑ์ชัดเจน — _filterParts เรียก setState ทุกตัวอักษรที่พิมพ์ แล้วการ์ดทุกใบสร้าง Future ใหม่ (13.6) ซึ่งเป็นทั้ง “ถี่มาก” และ “มีของแพง”

8. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ปรับแล้วไม่รู้ว่าดีขึ้นไหมไม่ได้วัดก่อนนับก่อนและหลัง
ใส่ const แล้วข้อความไม่เปลี่ยนภาษาconst ทำให้ข้ามการวาดใหม่เอาออกจาก widget ที่มีข้อความ (11.6)
วาดใหม่ทั้งจอทุก 15 วินาทีประกาศแม้ข้อมูลเหมือนเดิมตรวจก่อนประกาศ
แยก ListenableBuilder จนอ่านไม่รู้เรื่องปรับก่อนวัดปรับเฉพาะที่วัดแล้วว่ามีปัญหา
ใส่ RepaintBoundary ทุกที่คิดว่ายิ่งเยอะยิ่งดีใช้เฉพาะส่วนที่วาดบ่อยและแพง
เทสต์การวาดใหม่ล้มหลัง refactorโครง widget เปลี่ยนปรับตัวเลขที่คาด ถ้าการเปลี่ยนถูกต้อง

9. แบบฝึกหัด

จอรายการงานมีสามส่วน — หัวจอที่แสดงชื่อผู้ใช้และจุดสถานะ ลิสต์งาน 50 รายการ และแถบสรุปด้านล่าง ทั้งหมดอยู่ใต้ ListenableBuilder ตัวเดียวที่ฟัง TaskStore

ข้อมูลถูกโหลดใหม่ทุก 15 วินาที เขียนเทสต์แล้วปรับปรุง

แนวคำตอบ

ขั้นที่ 1 — วัดก่อน

เทสต์ข้อสุดท้ายคือข้อที่ให้ผลมากที่สุด เพราะการถามซ้ำทุก 15 วินาทีในคลังที่ไม่มีอะไรเปลี่ยน คือกรณีที่เกิดบ่อยที่สุด

แต่มันต้องการให้ store เทียบข้อมูลก่อนประกาศ ซึ่งต้องมีวิธีเทียบที่ถูก

Future<void> load() async {
  final result = await repository.fetchOpen();
  // เทียบด้วยเลขรุ่นรวม ไม่ใช่เทียบลิสต์ทีละตัวซึ่งแพงกว่าการวาดใหม่
  if (_lastVersion == result.version) return;
  _lastVersion = result.version;
  _tasks = result.tasks;
  notifyListeners();
}

ข้อควรระวัง — อย่าเทียบด้วยการวนทั้งลิสต์ถ้าลิสต์ยาว เพราะการเทียบอาจแพงกว่าการวาดใหม่ที่ตั้งใจจะประหยัด ซึ่งเป็นกับดักคลาสสิกของการปรับประสิทธิภาพโดยไม่วัด

10. สรุป

  • นับการวาดใหม่ ทำให้ “เร็วขึ้น” เป็นตัวเลขที่เทสต์ยืนยันได้
  • ตำแหน่งของ ListenableBuilder เปลี่ยนผลได้สิบเท่า
  • ตรวจก่อน notifyListeners กันการวาดที่ไม่จำเป็นตั้งแต่ต้นทาง
  • อย่าปรับก่อนวัด
  • เทสต์การวาดใหม่จะล้มเมื่อมีคนย้ายโค้ดกลับ ซึ่งเป็นการเตือนที่ถูกต้อง
  • ระวังการเทียบข้อมูลที่แพงกว่าการวาดใหม่
