2. เทสต์ที่จับบั๊กจาก 5.4

testWidgets('ออกจากระบบตอนอยู่ลึกในแอป', (tester) async {
  await tester.pumpWidget(NavBasicsApp(session: session));
  await tester.pumpAndSettle();

  // เข้าไปลึก ๆ ก่อน
  await tester.tap(find.byKey(const Key('task-T-1')));
  await tester.pumpAndSettle();
  expect(find.byKey(const Key('task-detail')), findsOneWidget);

  // แล้วออกจากระบบจากตรงนั้น
  await tester.tap(find.byKey(const Key('sign-out')));
  await tester.pumpAndSettle();

  expect(find.byKey(const Key('login-screen')), findsOneWidget);
  // จอเก่าต้องไม่เหลืออยู่ใต้จอเข้าสู่ระบบ
  expect(
    find.byKey(const Key('task-detail'), skipOffstage: false),
    findsNothing,
  );
});

บรรทัดสุดท้ายคือหัวใจ — ถ้าใช้ pushReplacement อย่างเดียว จอรายละเอียดจะยังอยู่ในกอง และผู้ใช้คนถัดไปกดย้อนกลับก็เห็นข้อมูลของคนก่อน

/// pushReplacement อย่างเดียวไม่พอ เพราะมันแทนแค่ route บนสุด
void _signOut() {
  widget.session.signOut();
  Navigator.of(context).pushAndRemoveUntil(
    MaterialPageRoute<void>(builder: (_) => LoginScreen(session: widget.session)),
    (route) => false, // ไม่เก็บ route ไหนไว้เลย
  );
}
skipOffstage: false จำเป็นตรงนี้ (14.8) — จอที่ถูกทับยังอยู่ในต้นไม้แต่ถูกซ่อน ถ้าไม่บอกให้หารวมตัวที่ซ่อน เทสต์จะผ่านทั้งที่บั๊กยังอยู่

3. ผลที่ส่งกลับจากจอลูก

testWidgets('สแกนแล้วค่ากลับมาที่จอหลัก', (tester) async {
  await tester.pumpWidget(wrap(ReceivingScreen()));
  await tester.pumpAndSettle();

  await tester.tap(find.byKey(const Key('open-scanner')));
  await tester.pumpAndSettle();

  await tester.enterText(find.byKey(const Key('manual-code')), 'PLT-001');
  await tester.tap(find.byKey(const Key('confirm-scan')));
  await tester.pumpAndSettle();

  // กลับมาจอเดิมแล้ว และค่าที่สแกนอยู่ในช่อง
  expect(find.byKey(const Key('scanner-screen')), findsNothing);
  expect(find.text('PLT-001'), findsOneWidget);
});

เทสต์นี้ครอบคลุมทั้งเส้นทาง — เปิดจอลูก ทำงาน ปิด และค่ากลับมา ซึ่งเป็นสิ่งที่ทดสอบแยกเป็นชิ้นแล้วไม่ครอบคลุม

final code = await Navigator.push<String>(
  context,
  MaterialPageRoute(builder: (_) => const ScannerScreen()),
);
// ผู้ใช้กดย้อนกลับได้ ซึ่งคืน null — ต้องเผื่อเสมอ
if (code == null || !mounted) return;
_controller.text = code;
testWidgets('กดย้อนกลับจากจอสแกน ค่าเดิมต้องไม่หาย', (tester) async {
  // พิมพ์ค่าไว้ก่อน
  await tester.enterText(find.byKey(const Key('pallet-field')), 'PLT-999');
  await tester.tap(find.byKey(const Key('open-scanner')));
  await tester.pumpAndSettle();

  await tester.pageBack();
  await tester.pumpAndSettle();

  expect(find.text('PLT-999'), findsOneWidget);
});

4. ทดสอบว่าไปจอไหน

สองวิธี และเลือกตามสิ่งที่ทดสอบ

วิธีเขียนเหมาะเมื่อ
ดูสิ่งที่แสดงexpect(find.byKey(const Key('task-detail')), findsOneWidget)ปกติ — ทดสอบสิ่งที่ผู้ใช้เห็น
ดักที่ NavigatorNavigatorObserver ปลอมทดสอบว่าเรียก push อะไร โดยไม่สนใจว่าจอนั้นวาดอะไร
class RecordingObserver extends NavigatorObserver {
  final List<String?> pushed = [];

  @override
  void didPush(Route<dynamic> route, Route<dynamic>? previous) {
    pushed.add(route.settings.name);
  }
}

// ในเทสต์
final observer = RecordingObserver();
await tester.pumpWidget(MaterialApp(
  navigatorObservers: [observer],
  home: const HomeScreen(),
));

await tester.tap(find.byKey(const Key('open-tasks')));
await tester.pumpAndSettle();

expect(observer.pushed, contains('/tasks'));
วิธีแรกดีกว่าในเกือบทุกกรณี เพราะทดสอบสิ่งที่ผู้ใช้เห็น ส่วนวิธีที่สองผูกกับรายละเอียดภายในซึ่งขัดกับหลักใน 14.1 — ใช้เมื่อจอปลายทางแพงเกินจะสร้างในเทสต์

5. การนำทางที่ไม่ใช้ Navigator

wms_absolute_mobile ใช้สถานะแทนกองของ route

// warehouse_store.dart
String screen = 'home';

void nav(String s, {bool preserveFlow = false}) { ... }
void goBack() { ... }

วิธีนี้ทดสอบง่ายกว่ามาก เพราะเป็นเรื่องของ store ล้วน ๆ

// จาก flow_navigation_test.dart ของจริง
test('ออกจาก flow แล้วกลับหน้าแรก', () {
  store.nav('recvrun');
  store.exitFlow();
  expect(store.screen, 'home');
});
Navigatorสถานะใน store
ทดสอบต้องสร้าง widgetเทสต์ unit ธรรมดา
ปุ่มย้อนกลับของเครื่องทำงานเองต้องเขียนจัดการเอง
animation ระหว่างจอมีให้ต้องทำเอง
สถานะของ flow ที่ทำค้างอยู่ในแต่ละจออยู่ที่เดียว ตรวจได้หมด

แถวสุดท้ายคือเหตุผลที่โปรเจกต์นั้นเลือกทางนี้ — งานในคลังเป็นขั้นตอนที่ต่อกัน และการมีสถานะของทั้ง flow อยู่ที่เดียวทำให้ตรวจได้ว่าอยู่ขั้นไหน

6. ทดสอบว่าย้อนกลับไม่ได้

testWidgets('กลางขั้นตอนยืนยัน ปุ่มย้อนกลับต้องเตือนก่อน', (tester) async {
  await startReceivingFlow(tester);
  await tester.enterText(find.byKey(const Key('quantity')), '5');
  await tester.pump();

  await tester.pageBack();
  await tester.pumpAndSettle();

  // ยังอยู่จอเดิม พร้อมกล่องเตือน
  expect(find.byKey(const Key('receiving-run')), findsOneWidget);
  expect(find.text('ยังไม่ได้บันทึก'), findsOneWidget);
});

ทดสอบ PopScope ที่ 5.9 สอน และรายละเอียดที่ต้องระวังคือ tester.pageBack() จำลองปุ่มย้อนกลับของระบบ ซึ่งเป็นทางที่ผู้ใช้ PDA ใช้จริง

7. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
เทสต์ผ่านแต่จอเก่ายังค้างไม่ได้ใส่ skipOffstage: falseใส่
หาจอปลายทางไม่เจอไม่ได้ pumpAndSettle หลัง pushใส่ — การเปลี่ยนจอมี animation
pageBack ไม่ทำงานไม่มี Scaffold หรือไม่มีอะไรให้ popตรวจว่ามีจอซ้อนอยู่จริง
ค่าที่ส่งกลับเป็น nullจอลูก pop โดยไม่ส่งค่าตรวจว่า Navigator.pop(context, value)
เทสต์พังเมื่อเปลี่ยนชื่อ routeใช้ NavigatorObserverทดสอบสิ่งที่แสดงแทน
Looking up a deactivated widget's ancestorใช้ context หลัง popเช็ค mounted

8. แบบฝึกหัด

ผู้ใช้รายงานว่า “กดออกจากระบบแล้วกดย้อนกลับ กลับเข้าไปในแอปได้เลย ไม่ต้องล็อกอิน” เขียนเทสต์ที่จับได้

แนวคำตอบ

เทสต์ต้องจำลองสามขั้น — เข้าไปลึก ออกจากระบบ แล้วกดย้อนกลับ

testWidgets('ออกจากระบบแล้วกดย้อนกลับ ต้องไม่กลับเข้าแอป', (tester) async {
  await tester.pumpWidget(AuthLabApp(dependencies: deps));
  await signIn(tester);
  await tester.pumpAndSettle();

  // เข้าไปลึกสองชั้น
  await tester.tap(find.byKey(const Key('task-T-1')));
  await tester.pumpAndSettle();
  await tester.tap(find.byKey(const Key('task-edit')));
  await tester.pumpAndSettle();

  await tester.tap(find.byKey(const Key('sign-out')));
  await tester.pumpAndSettle();
  expect(find.byKey(const Key('login-screen')), findsOneWidget);

  // กดย้อนกลับ — ต้องไม่มีอะไรให้กลับไป
  await tester.pageBack();
  await tester.pumpAndSettle();

  expect(find.byKey(const Key('login-screen')), findsOneWidget);
  expect(find.byKey(const Key('task-detail'), skipOffstage: false), findsNothing);
});

แต่เทสต์นี้อาจล้มด้วยเหตุผลที่ไม่เกี่ยวกับบั๊ก — pageBack() จะโยนถ้าไม่มีอะไรให้ pop ซึ่งเป็นสิ่งที่เราต้องการพอดี

// ถ้าทำถูก จะไม่มีอะไรให้ pop จึงต้องเผื่อ
expect(
  () async => tester.pageBack(),
  throwsA(anything),
  reason: 'ไม่ควรมี route ให้กลับไป',
);

เขียนแบบนี้อ่านยาก ทางที่ชัดกว่าคือตรวจกองของ route ตรง ๆ

final navigator = tester.state<NavigatorState>(find.byType(Navigator));
expect(navigator.canPop(), isFalse, reason: 'ต้องไม่เหลือ route ไหนเลย');

สิ่งที่ต้องตรวจเพิ่มและสำคัญกว่า — การที่จอหายไม่ได้แปลว่าข้อมูลหาย

testWidgets('ออกจากระบบแล้วข้อมูลของคนก่อนต้องไม่เหลือ', (tester) async {
  await signIn(tester, username: 'somchai');
  await tester.pumpAndSettle();
  expect(find.text('somchai'), findsOneWidget);

  await signOut(tester);
  await signIn(tester, username: 'somsri');
  await tester.pumpAndSettle();

  expect(find.text('somchai'), findsNothing, reason: 'store ต้องถูกล้าง');
  expect(store.tasks, isEmpty, reason: 'ข้อมูลของคนก่อนต้องไม่ค้าง');
});

เทสต์คู่นี้ครอบคลุมทั้งสองด้านของกฎเดียวกันจาก 8.8 — ล้างทั้งกองของ route และล้างทั้ง store ซึ่งเป็นคนละเรื่องและลืมได้ทีละอย่าง

วิธีที่ทำให้ลืมไม่ได้คือผูกอายุของ store ไว้กับ shell ที่มีเฉพาะตอนเข้าสู่ระบบแล้ว (12.9 §8) — แล้วการล้างจะเกิดขึ้นเองโดยไม่ต้องเขียนโค้ด

9. สรุป

  • ทดสอบว่าจอเก่าไม่เหลืออยู่ ด้วย skipOffstage: false
  • pumpAndSettle หลัง push เสมอ เพราะมี animation
  • ทดสอบทั้งค่าที่ส่งกลับ และกรณีที่ผู้ใช้กดย้อนกลับ
  • ทดสอบสิ่งที่แสดง ดีกว่าดักที่ Navigator
  • การนำทางด้วยสถานะใน store ทดสอบง่ายกว่า แต่ต้องจัดการปุ่มย้อนกลับเอง
  • ล้างกองของ route กับล้าง store เป็นคนละเรื่อง ต้องทดสอบทั้งคู่
