1. ปัญหาที่ 14.4 ทิ้งไว้

MockClient ตอบตามที่เราเขียนไว้ ซึ่งแปลว่าเทสต์ทั้งหมดตรวจความเข้าใจของเราเอง ถ้าเข้าใจผิดตั้งแต่ต้น เทสต์จะเขียวและแอปจะพัง

// เราคิดว่าเซิร์ฟเวอร์ส่งแบบนี้
MockClient((_) async => http.Response('{"items": [...]}', 200))

// แต่จริง ๆ มันส่ง {"data": {"tasks": [...]}}
// เทสต์เขียว แอปพัง

ทางแก้คือใช้ข้อมูลจริงแทนข้อมูลที่เราแต่ง

2. ไฟล์ตัวอย่างที่คัดลอกมาจากของจริง

test/fixtures/
  tasks_page.json           ← คัดลอกจากคำตอบจริงของเซิร์ฟเวอร์
  tasks_page_pascal.json    ← จากระบบเก่าที่ส่ง PascalCase
  inbound_order.json
/// อ่านไฟล์ตัวอย่างที่เก็บไว้ใน test/fixtures
///
/// เก็บเป็นไฟล์แยก ไม่ใช่สตริงในเทสต์ เพราะเป็น**สำเนาของสิ่งที่เซิร์ฟเวอร์
/// ส่งมาจริง** — คัดลอกจาก log ได้ตรง ๆ และอัปเดตได้โดยไม่ต้องแก้โค้ด
Map<String, dynamic> loadFixture(String name) {
  final file = File('test/fixtures/$name');
  return jsonDecode(file.readAsStringSync()) as Map<String, dynamic>;
}
เก็บเป็นไฟล์เขียนเป็นสตริงในเทสต์
คัดลอกจาก log มาวางได้ตรง ๆต้อง escape เครื่องหมายคำพูด
อัปเดตได้โดยไม่แตะโค้ดต้องแก้ไฟล์เทสต์
ส่งให้ทีมหลังบ้านดูได้ว่า “นี่คือสิ่งที่เราคาด”ต้องคัดลอกออกมาก่อน
diff อ่านรู้เรื่องเมื่อสัญญาเปลี่ยนdiff รกเพราะปนกับโค้ด

แถวสุดท้ายมีค่ามาก — วันที่ทีมหลังบ้านเปลี่ยนสัญญา การอัปเดตไฟล์ตัวอย่างจะทำให้เห็นชัดใน pull request ว่าอะไรเปลี่ยนไปบ้าง

3. เทสต์สัญญาที่ง่ายที่สุด

test('ไฟล์ตัวอย่างที่เก็บไว้ ต้องอ่านได้ด้วยโมเดลปัจจุบัน', () {
  final json = loadFixture('tasks_page.json');
  final items = requireList(json, 'items');

  final tasks = [
    for (var i = 0; i < items.length; i++)
      TaskDto.fromJson(requireMap(items[i], 'items[$i]')),
  ];

  expect(tasks, isNotEmpty);
  for (final task in tasks) {
    expect(task.id.trim(), isNotEmpty);
    expect(task.quantity, greaterThanOrEqualTo(0));
  }
});

เทสต์นี้จับได้สองอย่างที่ MockClient จับไม่ได้

จับได้อย่างไร
เราแก้โมเดลแล้วอ่านข้อมูลจริงไม่ได้ไฟล์ตัวอย่างไม่เปลี่ยน แต่โค้ดเปลี่ยน
เซิร์ฟเวอร์เปลี่ยนสัญญาเมื่ออัปเดตไฟล์ตัวอย่างแล้วเทสต์ล้ม
แถวที่สองต้องมีคนอัปเดตไฟล์ ซึ่งไม่เกิดเอง — จึงควรมีขั้นตอนที่ทำให้เกิด เช่นให้ CI ดึงคำตอบจริงจากเซิร์ฟเวอร์ทดสอบมาเทียบกับไฟล์ที่เก็บไว้

4. รองรับสองรูปแบบด้วยตัวอ่านชุดเดียว

test('รูปแบบ PascalCase ที่ระบบเก่าส่งมา ต้องอ่านได้ด้วยตัวอ่านชุดเดียวกัน', () {
  // ไฟล์นี้มาจากระบบเก่าที่ส่งชื่อฟิลด์เป็น PascalCase ทั้งหมด
  // โค้ดที่อ่านเป็นชุดเดียวกับที่อ่าน camelCase ไม่มีสาขาแยก (6.6)
  final json = loadFixture('tasks_page_pascal.json');
  final items = requireList(json, 'items');

  final tasks = [
    for (var i = 0; i < items.length; i++)
      TaskDto.fromJson(requireMap(items[i], 'items[$i]')),
  ];

  expect(tasks.single.id, 'T-101');
  expect(tasks.single.quantity, 5);
  expect(tasks.single.palletCode, 'PAL-0201');
});

สังเกตว่าโค้ดที่อ่านทั้งสองไฟล์เหมือนกันทุกบรรทัด — ไม่มี if ที่ตรวจว่าเป็นรูปแบบไหน เพราะ json_read.dart จัดการให้ที่เดียว

Object? _raw(Map<String, dynamic> json, String key) {
  if (json.containsKey(key)) return json[key];
  final pascal = key[0].toUpperCase() + key.substring(1);
  if (json.containsKey(pascal)) return json[pascal];
  return _absent;
}

ไฟล์ตัวอย่างสองไฟล์คือสิ่งที่พิสูจน์ว่ากลไกนี้ทำงานจริง

5. เทสต์ที่ตรวจว่าสัญญาที่หายไปจะดัง

test('ฟิลด์ที่หายไปจากไฟล์ตัวอย่าง ต้องทำให้เทสต์ล้ม ไม่ใช่ผ่านเงียบ ๆ', () {
  // จำลองว่าเซิร์ฟเวอร์รุ่นใหม่เลิกส่ง quantity
  final json = loadFixture('tasks_page.json');
  final items = requireList(json, 'items');
  final broken = Map<String, dynamic>.from(requireMap(items.first, 'x'))
    ..remove('quantity')
    ..remove('Quantity');

  expect(
    () => TaskDto.fromJson(broken),
    throwsA(isA<ContractException>()),
    reason: 'สัญญาที่หายไปต้องดัง ไม่ใช่กลายเป็นศูนย์เงียบ ๆ',
  );
});

เทสต์นี้ทดสอบพฤติกรรมเมื่อสัญญาถูกละเมิด ซึ่งเป็นเรื่องที่ Part 6 ใช้เวลาทั้ง Part ตัดสินใจ

ถ้าวันหนึ่งมีคนเปลี่ยน requireInt เป็นอ่านแล้วคืน 0 เมื่อไม่มีค่า เทสต์นี้จะล้ม — ซึ่งเป็นการเตือนที่ถูกต้อง เพราะการเปลี่ยนแบบนั้นทำให้ของหายในคลังโดยไม่มีใครรู้

6. ระดับที่สูงขึ้น — ยิงเซิร์ฟเวอร์จริง

เทสต์ที่ยิงเซิร์ฟเวอร์ทดสอบจริงคือสิ่งเดียวที่จับ “ฝั่งเซิร์ฟเวอร์เปลี่ยน” ได้

@Tags(['live'])
library;

void main() {
  // รันเฉพาะเมื่อสั่ง: flutter test --tags live
  final baseUrl = Platform.environment['WMS_TEST_BASE_URL'];

  test('สัญญาของ mobile_tasks ยังตรงกับโมเดล', () async {
    final api = ApiClient(baseUrl: baseUrl!);
    final json = await api.getJson('/api/WMS/mobile_tasks');

    // อ่านด้วยโมเดลจริง ถ้าสัญญาเปลี่ยนจะโยน ContractException
    final items = requireList(json, 'items');
    for (var i = 0; i < items.length; i++) {
      TaskDto.fromJson(requireMap(items[i], 'items[$i]'));
    }
  }, skip: baseUrl == null ? 'ไม่ได้ตั้ง WMS_TEST_BASE_URL' : null);
}
รายละเอียดเหตุผล
@Tags(['live'])ไม่รันปนกับเทสต์ปกติ ซึ่งต้องรันได้ทุกที่ทุกเวลา
อ่าน URL จาก environmentแต่ละคนชี้เซิร์ฟเวอร์คนละตัวได้
skip เมื่อไม่มีค่าคนที่ไม่มีเซิร์ฟเวอร์ยังรันเทสต์ที่เหลือได้
ไม่ตรวจค่าเฉพาะเจาะจงข้อมูลบนเซิร์ฟเวอร์ทดสอบเปลี่ยนได้ ตรวจแค่ว่าอ่านได้
แถวสุดท้ายสำคัญ — เทสต์ที่ตรวจว่า “ต้องมีงานสามรายการ” จะล้มทุกครั้งที่มีคนไปกดในเซิร์ฟเวอร์ทดสอบ แล้วจะไม่มีใครเชื่อถือมันอีก

7. ทำให้ไฟล์ตัวอย่างไม่ล้าสมัย

ปัญหาที่แท้จริงของวิธีนี้คือไม่มีอะไรบังคับให้อัปเดตไฟล์ ทางแก้ที่ได้ผลจริง

วิธีราคา
เขียนสคริปต์ที่ดึงคำตอบจริงมาทับไฟล์ แล้วดู diffต่ำ ทำครั้งเดียวใช้ตลอด
ให้ CI รันเทสต์แบบ live สัปดาห์ละครั้งกลาง
ให้ทีมหลังบ้านส่งไฟล์ตัวอย่างมาพร้อมทุกการเปลี่ยนสัญญาต้องตกลงกันเป็นกระบวนการ
# tool/refresh_fixtures.sh
curl -s "$WMS_TEST_BASE_URL/api/WMS/mobile_tasks" \
  | python -m json.tool > test/fixtures/tasks_page.json
git diff --stat test/fixtures/

รันก่อนปล่อยรุ่น แล้วถ้า diff ว่าง แปลว่าสัญญายังเหมือนเดิม ถ้าไม่ว่าง แปลว่ามีอะไรเปลี่ยนที่ต้องดู

8. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
เทสต์เขียวแต่แอปอ่านข้อมูลจริงไม่ได้ข้อมูลในเทสต์เป็นของที่เราแต่งเองใช้ไฟล์ที่คัดลอกจากของจริง
ไฟล์ตัวอย่างล้าสมัยมาหนึ่งปีไม่มีกระบวนการอัปเดตทำสคริปต์ดึงมาทับ
เทสต์ live ล้มบ่อยจนไม่มีใครสนใจตรวจค่าเฉพาะเจาะจงบนข้อมูลที่เปลี่ยนได้ตรวจแค่ว่าอ่านได้
เทสต์ปกติรันไม่ได้เพราะไม่มีเซิร์ฟเวอร์ไม่ได้แยก tagใช้ @Tags และ skip
ไฟล์ตัวอย่างอ่านไม่ออกเป็นบรรทัดเดียวยาวจัดรูปแบบก่อนเก็บ

9. แบบฝึกหัด

ทีมหลังบ้านบอกว่าจะเพิ่มฟิลด์ priority ใน mobile_tasks สัปดาห์หน้า

  1. ต้องทำอะไรกับเทสต์ตอนนี้
  2. ถ้าเขาเพิ่มแล้วแอปรุ่นเก่ายังใช้อยู่ จะเป็นอย่างไร
แนวคำตอบ

1. ยังไม่ต้องทำอะไร — และนี่คือคำตอบที่ถูกแม้จะดูเหมือนขี้เกียจ

การเพิ่มฟิลด์ใหม่ที่แอปไม่ได้อ่าน ไม่กระทบอะไรเลย เพราะ json_read.dart อ่านเฉพาะคีย์ที่ขอ ส่วนคีย์อื่นถูกละไว้

สิ่งที่ควรทำคือถามว่าฟิลด์นี้บังคับหรือไม่ เพราะคำตอบเปลี่ยนสิ่งที่ต้องทำ

ถ้าต้องทำ
เพิ่มฟิลด์ใหม่เฉย ๆไม่ต้องทำอะไร อ่านเมื่อจะใช้
ฟิลด์เดิมจะหายไปด้วยต้องแก้โมเดลก่อนที่เซิร์ฟเวอร์จะเปลี่ยน
ความหมายของฟิลด์เดิมเปลี่ยนอันตรายที่สุด เพราะไม่มีอะไรจับได้เลย

แถวสุดท้ายคือกรณีที่เทสต์สัญญาก็ช่วยไม่ได้ — เช่นถ้า quantity เปลี่ยนจาก “จำนวนที่ต้องหยิบ” เป็น “จำนวนที่หยิบไปแล้ว” ทุกอย่างจะยังอ่านได้และเทสต์จะเขียว แต่ความหมายผิดหมด

ป้องกันได้ทางเดียวคือคุยกัน ซึ่งเป็นเหตุผลที่ 12.2 ยืนยันว่าต้องตกลงสัญญาให้ชัดก่อนเขียน

2. แอปรุ่นเก่าจะทำงานได้ปกติ เพราะไม่ได้อ่านฟิลด์นั้น — นี่คือเหตุผลที่การเพิ่มฟิลด์ปลอดภัยกว่าการลบหรือเปลี่ยนความหมายมาก

การเปลี่ยนแอปรุ่นเก่า
เพิ่มฟิลด์ใหม่ไม่กระทบ
ทำฟิลด์ที่บังคับให้เป็นทางเลือกได้พังถ้าแอปเรียก require
ลบฟิลด์พังทันที
เปลี่ยนชนิดพังทันที
เปลี่ยนความหมายไม่พัง แต่ผิด

ควรเสนอให้ทีมใช้กฎนี้เป็นข้อตกลง — เพิ่มได้เสมอ ลบต้องรอให้แอปทุกเครื่องอัปเดตก่อน ซึ่งเป็นกฎเดียวกับที่ระบบที่มีผู้ใช้หลายรุ่นใช้กัน

10. สรุป

  • ข้อมูลที่เราแต่งเอง ทดสอบความเข้าใจของเราเอง
  • เก็บไฟล์ตัวอย่างที่คัดลอกจากคำตอบจริง
  • ไฟล์แยกดีกว่าสตริงในเทสต์ เพราะ diff อ่านรู้เรื่อง
  • เทสต์ที่ยิงเซิร์ฟเวอร์จริงต้องแยก tag และ skip ได้
  • เทสต์ live ตรวจแค่ว่าอ่านได้ อย่าตรวจค่าเฉพาะเจาะจง
  • ต้องมีกระบวนการที่ทำให้ไฟล์ตัวอย่างไม่ล้าสมัย
