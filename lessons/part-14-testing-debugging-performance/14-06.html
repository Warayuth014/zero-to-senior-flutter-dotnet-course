1. ทำไม store คือที่ที่คุ้มที่สุด

store ถือซึ่งคือที่ที่บั๊กอยู่
ลำดับของสถานะวงหมุนค้าง ปุ่มกดไม่ได้
การกันกดซ้ำคำสั่งซ้ำ ของหายของเกิน
การจัดการคำตอบที่มาไม่เรียงข้อมูลเก่าทับข้อมูลใหม่
การแปลความล้มเหลวเป็นสิ่งที่ผู้ใช้ทำต่อได้ข้อความที่ไม่ช่วยอะไร

และทั้งหมดนี้ทดสอบได้โดยไม่ต้องมี widget เพราะ store ไม่ import material.dart (9.7)

test('เหตุการณ์เก่าที่มาถึงช้า ต้องไม่ทับของใหม่', () async {
  repository.result = [task('T-1', version: 5)];
  await store.start();

  inner.emit(const TaskCompleted(taskId: 'T-1', version: 3));
  await Future<void>.delayed(Duration.zero);

  expect(store.tasks.single.status, LiveTaskStatus.waiting);
});

เจ็ดบรรทัด รันในไมโครวินาที และทดสอบสิ่งที่ทดสอบด้วยมือไม่มีทางทำได้

2. สี่กลุ่มที่ควรมีในทุก store

กลุ่มตัวอย่างจาก lab
เส้นทางที่สำเร็จโหลดแล้วได้ข้อมูล
เส้นทางที่ล้มเหลวโหลดล้มเหลวแล้วเก็บข้อความไว้แสดง
กฎที่ต้องจริงเสมอมีตัวที่ใช้งานอยู่หนึ่งตัวเสมอ
จังหวะเวลากดซ้ำ คำตอบมาไม่เรียง ยกเลิกกลางทาง

กลุ่มที่สามกับสี่คือที่ที่ค่าอยู่ เพราะกลุ่มแรกสองกลุ่มมักถูกทดสอบด้วยมืออยู่แล้วตอนพัฒนา

3. ทดสอบว่ามีการประกาศ ไม่ใช่แค่ค่าเปลี่ยน

test('สลับแล้วประกาศให้หน้าจอวาดใหม่', () async {
  var notified = 0;
  controller.addListener(() => notified++);

  await controller.toggle();
  expect(notified, 1);
  expect(controller.isThai, isTrue);
});

สองบรรทัดสุดท้ายทดสอบคนละเรื่อง — isThai ตรวจว่าค่าเปลี่ยน ส่วน notified ตรวจว่าหน้าจอจะรู้

ค่าที่เปลี่ยนโดยไม่ประกาศ คือบั๊กที่ผู้ใช้เห็นเป็น “กดแล้วไม่มีอะไรเกิดขึ้น” และเป็นบั๊กที่หาสาเหตุยากเพราะข้อมูลใน store ถูกต้องทุกอย่าง

test('เลือกภาษาเดิมซ้ำ ต้องไม่ประกาศ', () async {
  var notified = 0;
  controller.addListener(() => notified++);

  await controller.setLanguage(AppLanguage.english);
  expect(notified, 0);
});

ด้านกลับก็สำคัญ — ประกาศทั้งที่ไม่มีอะไรเปลี่ยน ทำให้ทั้งจอวาดใหม่ฟรี ๆ (14.15)

4. ทดสอบลำดับของสถานะ

test('ตรวจแล้วประกาศสองครั้ง คือกำลังตรวจ และผลลัพธ์', () async {
  final states = <ServerStatus>[];
  store.addListener(() => states.add(store.status));

  await store.refreshStatus(client: ...);

  // ต้องเห็น checking ด้วย ไม่งั้นผู้ใช้กดปุ่มแล้วเหมือนไม่มีอะไรเกิดขึ้น
  expect(states, [ServerStatus.checking, ServerStatus.connected]);
});

เก็บสถานะทุกครั้งที่ประกาศลงลิสต์ แล้วตรวจทั้งลิสต์ — วิธีนี้จับสถานะระหว่างทางได้ ซึ่งการตรวจค่าสุดท้ายอย่างเดียวมองไม่เห็น

ตรวจแค่ค่าสุดท้ายตรวจทั้งลำดับ
รู้ว่าจบถูกรู้ว่าระหว่างทางถูกด้วย
ไม่รู้ว่าผู้ใช้เห็นอะไรระหว่างรอรู้
ไม่จับการวาดใหม่ที่ไม่จำเป็นจับได้

5. ทดสอบกฎที่ต้องจริงเสมอ

test('สลับไป id ที่ไม่มีอยู่ ต้องไม่ทำให้ไม่มีตัวที่ใช้งานอยู่', () async {
  await store.load();

  await store.activate('ไม่มีจริง');

  expect(store.active, isNotNull);
});

test('ลบตัวสุดท้ายไม่ได้', () async {
  await store.load();
  expect(await store.remove(store.active!.id), isFalse);
  expect(store.profiles, hasLength(1));
});

test('ลบตัวที่ใช้งานอยู่ ต้องมีตัวใหม่มาแทนเสมอ', () async { ... });

สามข้อนี้ทดสอบกฎเดียวกันจากสามทิศทาง — กฎ “มีตัวที่ใช้งานอยู่หนึ่งตัวเสมอ” ที่ 10.6 วางไว้

วิธีหากฎที่ควรทดสอบ: อ่านคอมเมนต์ในโค้ดที่ขึ้นต้นด้วย “ต้อง” หรือ “เสมอ” — ทุกประโยคแบบนั้นคือเทสต์ที่ยังไม่ได้เขียน

6. ทดสอบสิ่งที่ต้องไม่เกิด

test('รูปที่ไม่ผ่านการตรวจ ต้องไม่ถูกส่ง', () async {
  picker.result = image(bytes: jpegBytes(size: 9 * 1024 * 1024));

  await store.pick(PickSource.camera);
  await store.upload(path: '/upload');

  expect(store.stage, UploadStage.idle);
  expect(uploadCalls, 0, reason: 'ไม่ควรยิงเลย');
});

เทสต์ที่ตรวจว่า “ต้องไม่เกิด” มักถูกลืม แต่จับบั๊กประเภทที่แพงที่สุด — คำสั่งที่ยิงไปทั้งที่ไม่ควร

ตัวอย่างจาก labกันอะไร
expect(uploadCalls, 0)ส่งไฟล์ที่จะถูกปฏิเสธ
expect(postCount, 1) หลังกดสองครั้งคำสั่งซ้ำ
expect(repository.fetchCalls, callsBefore)โหลดใหม่ทั้งที่ไม่ต้อง
expect(inner.connectCalls, 1) หลังขอสามครั้งเปิดการเชื่อมต่อซ้ำ

รูปแบบที่ใช้ได้คือให้ของปลอมนับ แล้วตรวจตัวเลข ซึ่งเป็นเหตุผลที่ของปลอมใน 14.3 มีตัวนับทุกตัว

7. ทดสอบการเก็บของตอนทิ้ง

test('ทิ้ง store แล้วปล่อยการเชื่อมต่อที่ใช้ร่วมกัน', () async {
  await store.start();
  expect(shared.userCount, 1);

  store.dispose();
  await Future<void>.delayed(Duration.zero);

  expect(shared.userCount, 0);
  expect(inner.disconnectCalls, 1);
});

เทสต์กลุ่มนี้จับการรั่วซึ่งมองไม่เห็นด้วยตา (12.8) และควรมีในทุก store ที่ถือทรัพยากร

store ถืออะไรต้องมีเทสต์ว่า
ตัวจับเวลาหยุดหลัง dispose
การฟัง streamยกเลิกแล้ว
การเชื่อมต่อที่ใช้ร่วมกันปล่อยแล้ว
งาน async ที่ค้างอยู่ผลที่กลับมาไม่ทำให้พัง

8. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ทดสอบ store ไม่ได้เลยstore สร้าง repository เองรับเข้ามาทาง constructor (9.8)
ต้องสร้าง widget เพื่อทดสอบ storestore import material.dartใช้ foundation.dart (9.7)
ทดสอบค่าถูกแต่หน้าจอไม่อัปเดตไม่ได้ตรวจการประกาศนับ notified
ทดสอบจังหวะเวลาไม่ได้ของปลอมตอบทันทีใช้ Completer (14.3)
เทสต์ล้มแบบสุ่มใช้ store ตัวเดิมข้ามข้อสร้างใหม่ใน setUp
ลืมทดสอบว่าอะไรต้องไม่เกิดคิดแต่เส้นทางที่สำเร็จให้ของปลอมนับแล้วตรวจ

9. แบบฝึกหัด

มี store ที่จัดการการนับสต๊อก มีเมธอด scanPart, setQuantity, confirm เขียนรายการเทสต์ที่ควรมี

แนวคำตอบ

เรียงตามสี่กลุ่มใน §2

กลุ่มเทสต์
สำเร็จสแกนแล้วรายการโผล่ในลิสต์
ยืนยันแล้วส่งจำนวนทั้งหมดไปครั้งเดียว (9.5)
ล้มเหลวยืนยันแล้วเน็ตหลุด ต้องเก็บที่นับไว้ ไม่ล้างทิ้ง
เซิร์ฟเวอร์ปฏิเสธ ต้องบอกว่าทำไม
กฎที่ต้องจริงเสมอสแกนรหัสเดิมซ้ำ ต้องไม่เพิ่มแถวใหม่
จำนวนติดลบตั้งไม่ได้
ยืนยันตอนยังไม่ได้สแกนอะไร ต้องไม่ยิง
จังหวะเวลากดยืนยันซ้ำระหว่างรอ ต้องยิงครั้งเดียว
สแกนระหว่างที่กำลังยืนยัน ต้องไม่เข้าไปในชุดที่ส่งไปแล้ว
ทิ้ง store ระหว่างยืนยัน ต้องไม่พัง

แถวที่ควรสังเกตที่สุดคือแถวสุดท้ายของกลุ่มล้มเหลว — “เก็บที่นับไว้ ไม่ล้างทิ้ง”

test('ยืนยันไม่สำเร็จ ต้องเก็บที่นับไว้', () async {
  await store.scanPart('P-001');
  await store.scanPart('P-002');
  repository.throwOnConfirm = Exception('เชื่อมต่อไม่ได้');

  await store.confirm();

  // ผู้ใช้เดินนับมาสี่สิบนาที ห้ามให้เริ่มใหม่เพราะเน็ตหลุด
  expect(store.counted, hasLength(2));
  expect(store.canConfirm, isTrue, reason: 'ต้องกดยืนยันซ้ำได้');
});

เทสต์นี้บันทึกการตัดสินใจที่สำคัญที่สุดของฟีเจอร์นั้นไว้ — และเป็นเทสต์ที่จะช่วยชีวิตในวันที่มีคนมา refactor แล้วเผลอย้าย _counted.clear() ไปไว้ผิดที่

สังเกตว่าเทสต์ส่วนใหญ่ในรายการมาจากการถามว่าอะไรจะทำให้ผู้ใช้เสียงาน ไม่ใช่จากการไล่ดูว่าโค้ดมีกี่บรรทัด

10. สรุป

  • store คือที่ที่ตรรกะอยู่ และทดสอบได้โดยไม่ต้องมี widget
  • ทดสอบสี่กลุ่ม — สำเร็จ ล้มเหลว กฎที่ต้องจริงเสมอ และจังหวะเวลา
  • ตรวจการประกาศด้วย ไม่ใช่แค่ค่าที่เปลี่ยน
  • เก็บสถานะทุกครั้งที่ประกาศลงลิสต์ เพื่อตรวจลำดับ
  • ทดสอบสิ่งที่ต้องไม่เกิด โดยให้ของปลอมนับ
  • คอมเมนต์ที่มีคำว่า “ต้อง” หรือ “เสมอ” คือเทสต์ที่ยังไม่ได้เขียน
