4. ปิดค่าใน URL

เปิด lib/core/api_log.dart

/// แทนค่าที่อ่อนไหวใน URL ด้วยเครื่องหมาย แต่เก็บโครงไว้ให้ครบ
///
/// เก็บชื่อคีย์ไว้เสมอ เพราะการรู้ว่า "มี token ส่งไปด้วย" มีประโยชน์
/// ตอนไล่หาสาเหตุ ส่วนตัวค่าไม่มีประโยชน์เลย
String redactUri(Uri uri) {
  if (uri.queryParameters.isEmpty) return uri.toString();
  final safe = <String, String>{
    for (final entry in uri.queryParameters.entries)
      entry.key: _sensitiveQueryKeys.contains(entry.key.toLowerCase())
          ? _masked
          : entry.value,
  };
  return uri.replace(queryParameters: safe).toString();
}
test('ปิดค่า token ใน URL แต่เก็บชื่อคีย์ไว้', () {
  final uri = Uri.parse(
    'http://wms.local/api/tasks?search=PAL-01&token=secret123',
  );

  final redacted = redactUri(uri);

  expect(redacted, contains('search=PAL-01'));
  // ใช้ตัวอักษรล้วนเป็นเครื่องหมายปิด เพราะ Uri encode สัญลักษณ์
  expect(redacted, contains('token=REDACTED'));
  expect(redacted, isNot(contains('secret123')));
});

บรรทัดคอมเมนต์ในเทสต์บันทึกเรื่องที่เจอตอนเขียนจริง

/// ใช้ตัวอักษรล้วน ไม่ใช้เครื่องหมาย เพราะ Uri.replace จะ encode เครื่องหมาย
/// ให้เป็น %2A ทำให้ log อ่านยากและค้นหาไม่เจอ
const String _masked = 'REDACTED';

ถ้าใช้ *** ตามที่คนมักใช้ ไฟล์บันทึกจะเต็มไปด้วย %2A%2A%2A ซึ่งค้นหาไม่เจอเมื่อต้องการหาว่ามีค่าไหนถูกปิดบ้าง

สังเกตว่า search=PAL-01 ยังอยู่ครบ — คำค้นไม่ใช่ข้อมูลอ่อนไหว และมันจำเป็นตอนไล่หาสาเหตุ

5. ปิดค่าใน header

const Set<String> _sensitiveHeaders = {
  'authorization',
  'cookie',
  'set-cookie',
  'x-api-key',
};

Map<String, String> redactHeaders(Map<String, String> headers) => {
  for (final entry in headers.entries)
    entry.key: _sensitiveHeaders.contains(entry.key.toLowerCase())
        ? _masked
        : entry.value,
};

สองอย่างที่ทำให้ถูกต้อง

สิ่งที่ทำเพราะ
เทียบด้วยตัวเล็กทั้งหมดชื่อ header ไม่สนตัวพิมพ์ — Authorization กับ authorization คืออันเดียวกัน
เก็บชื่อ header ไว้รู้ว่ามี token ส่งไปหรือไม่ ซึ่งเป็นข้อมูลที่ใช้ได้

ข้อแรกเป็นกับดักที่พลาดง่าย — ถ้าเทียบตรง ๆ แล้วโค้ดส่ง authorization ตัวเล็ก ค่าจะหลุดออกไปทั้งที่มีตัวปิดอยู่แล้ว

บรรทัดเดียวนี้ตอบคำถามได้เกือบหมด

คำถามตอบได้จาก

สิ่งเดียวที่ตอบไม่ได้คือ “ส่งจำนวนเท่าไหร่ไป” — ซึ่งถ้าจำเป็นจริง ให้ดูจาก log ฝั่งเซิร์ฟเวอร์ผ่าน traceId แทน

เหตุผลที่ให้ฝั่งเซิร์ฟเวอร์เก็บดีกว่า

ฝั่งแอปฝั่งเซิร์ฟเวอร์
ไฟล์อยู่บนเครื่องที่เปลี่ยนมือทุกกะอยู่ในระบบที่ควบคุมสิทธิ์ได้
ถูกส่งต่อทางแชตได้ง่ายเข้าถึงได้เฉพาะคนที่มีสิทธิ์
ไม่มีนโยบายลบเมื่อครบกำหนดมี

7. บันทึกไม่ได้แปลว่าไม่มีค่าใช้จ่าย

ระดับบันทึกอะไรเปิดเมื่อ
ปกติเฉพาะคำขอที่ล้มเหลวตลอดเวลา
ละเอียดทุกคำขอเมื่อกำลังไล่หาสาเหตุ

เหตุผลที่ไม่บันทึกทุกคำขอตลอดเวลา

คนหน้างานสแกน 400 ครั้งต่อกะ × 3 คำขอต่อครั้ง = 1,200 บรรทัด
× 40 เครื่อง = 48,000 บรรทัดต่อวัน

ซึ่งกินพื้นที่บนเครื่องและทำให้หาบรรทัดที่สำคัญยากขึ้น

แต่ต้องเปิดได้ง่ายเมื่อต้องการ — ไม่ใช่ต้อง build ใหม่

8. บันทึกที่ไม่มีเวลาก็ไร้ประโยชน์

final stopwatch = Stopwatch()..start();
try {
  final response = await _client.get(uri, headers: headers).timeout(timeout);
  stopwatch.stop();
  _log(describeExchange(
    method: 'GET',
    uri: uri,
    statusCode: response.statusCode,
    elapsed: stopwatch.elapsed,
  ));
  return _decode(response);
} on TimeoutException {
  stopwatch.stop();
  _log(describeExchange(
    method: 'GET',
    uri: uri,
    statusCode: null,            // ไม่ได้คำตอบ
    elapsed: stopwatch.elapsed,
  ));
  rethrow;
}
test('ไม่ได้คำตอบก็ยังบันทึกได้', () {
  final line = describeExchange(
    method: 'GET',
    uri: Uri.parse('http://wms.local/api/tasks'),
    statusCode: null,
    elapsed: const Duration(seconds: 15),
  );

  expect(line, contains('ไม่ได้คำตอบ'));
});

กรณีที่ไม่ได้คำตอบเป็นกรณีที่ สำคัญที่สุด ที่ต้องบันทึก — เพราะเป็นกรณีที่ฝั่งเซิร์ฟเวอร์อาจไม่มีอะไรเลย

สิ่งที่เวลาบอกตีความ
ล้มเหลวใน 50 msไปไม่ถึงเซิร์ฟเวอร์ — ที่อยู่ผิดหรือไม่มีเส้นทาง
ล้มเหลวใน 15,000 ms พอดีถึงเซิร์ฟเวอร์แล้วแต่ไม่ตอบ — หมดเวลารอ
สำเร็จใน 4,000 msทำงานได้แต่ช้า — เป็นสัญญาณเตือน

สองแถวแรกให้ผลลัพธ์เหมือนกันในสายตาผู้ใช้ แต่ต้องแก้คนละที่ — และแยกได้จากตัวเลขเวลาอย่างเดียว

9. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
token อยู่ในไฟล์บันทึกบันทึก header ทั้งชุดปิดค่าที่ตัวกลางที่เดียว
ปิดแล้วยังหลุดเทียบชื่อ header แบบสนตัวพิมพ์เทียบด้วยตัวเล็ก
ข้อมูลลูกค้าอยู่ในไฟล์บันทึก bodyไม่บันทึก body เลย
ค้นหาค่าที่ถูกปิดไม่เจอใช้สัญลักษณ์ที่ถูก encodeใช้ตัวอักษรล้วน
ไฟล์ใหญ่จนหาอะไรไม่เจอบันทึกทุกคำขอตลอดเวลาบันทึกเฉพาะที่ล้มเหลว
บันทึกแล้วแต่ยังหาสาเหตุไม่ได้ไม่มีเวลาที่ใช้บันทึกเวลาเสมอ
กรณีที่ไม่ได้คำตอบไม่มีในไฟล์บันทึกเฉพาะตอนได้คำตอบบันทึกในทุกทางออก

ข้อ 1 — เกิดอะไรขึ้น

ภาพรวมคือเซิร์ฟเวอร์หรือเครือข่ายมีปัญหาประมาณสามนาที แล้วค่อย ๆ ฟื้น

สังเกตว่าเวลาที่ใช้เล่าเรื่องได้ทั้งหมด โดยที่ไม่มีเนื้อหาของ body เลยสักบรรทัด

15,000 ms48 ms

สาเหตุที่น่าจะเป็นเซิร์ฟเวอร์ทำงานหนักจนไม่ทันเซิร์ฟเวอร์ปิดไปเลย หรือเครือข่ายขาด

การเปลี่ยนจากหมดเวลารอเป็นต่อไม่ถึง เป็นข้อมูลสำคัญ

เซิร์ฟเวอร์ช้าลงเรื่อย ๆ (หมดเวลารอสองครั้ง)
  → แล้วหยุดตอบเลย (ต่อไม่ถึง)
  → แล้วกลับมาแต่ช้า (4.8 วินาที)
  → แล้วปกติ

รูปแบบนี้ตรงกับ "เซิร์ฟเวอร์ถูกรีสตาร์ต"

ถ้าไม่มีตัวเลขเวลา ทั้งสามบรรทัดจะเป็น “ไม่ได้คำตอบ” เหมือนกัน และเล่าเรื่องนี้ไม่ได้เลย

ข้อ 3 — ข้อมูลที่ควรบันทึกเพิ่ม

ข้อมูลใช้ทำอะไร
ชนิดความล้มเหลวไม่ต้องเดาจากตัวเลขเวลา
รุ่นของแอปรู้ว่าเครื่องนี้ใช้รุ่นไหน
รหัสเครื่องเทียบกับเครื่องอื่นในช่วงเวลาเดียวกัน
ที่อยู่เซิร์ฟเวอร์ที่ใช้อยู่รู้ว่าชี้ไปที่ไหน
ครั้งที่เท่าไหร่ของการลองใหม่แยก "ผู้ใช้กดเอง" ออกจาก "แอปลองเอง"

แถวสุดท้ายสำคัญกับบรรทัด 09:02:11 และ 09:02:27 — สองบรรทัดนี้ห่างกัน 16 วินาที ซึ่งใกล้เคียงกับเวลารอพอดี จึงเดาไม่ได้ว่าผู้ใช้กดใหม่เอง หรือแอปลองเองตาม 7.14

// เพิ่มข้อมูลโดยไม่เพิ่มความเสี่ยง
GET .../api/WMS/mobile_tasks → ไม่ได้คำตอบ (15000 ms)

ทุกอย่างที่เพิ่มเป็นข้อมูลของ ระบบ ไม่ใช่ของ ธุรกิจ จึงบันทึกได้อย่างปลอดภัย

บทเรียนที่ควรจำ: บันทึกที่ดีเล่าเรื่องได้โดยไม่ต้องมีเนื้อหา — และตัวเลขเวลาเป็นข้อมูลที่ให้ประโยชน์สูงสุดต่อความเสี่ยงที่ต่ำที่สุด เพราะมันแยกสาเหตุที่ต่างกันสิ้นเชิงออกจากกันได้ โดยไม่บอกอะไรเกี่ยวกับข้อมูลเลย

11. สรุปท้ายบท

  • บันทึกโครง ไม่บันทึกเนื้อ
  • ปิดค่าที่อ่อนไหวที่ตัวกลางที่เดียว
  • เทียบชื่อ header ด้วยตัวเล็กทั้งหมด
  • ใช้ตัวอักษรล้วนเป็นเครื่องหมายปิด เพื่อให้ค้นหาได้
  • เก็บชื่อคีย์ไว้ เพราะการรู้ว่ามีอะไรส่งไปก็มีประโยชน์
  • ไม่บันทึก body — ให้ฝั่งเซิร์ฟเวอร์เก็บผ่าน traceId
  • บันทึกเฉพาะที่ล้มเหลว และเปิดโหมดละเอียดได้เมื่อต้องการ
  • บันทึกเวลาที่ใช้เสมอ รวมถึงตอนที่ไม่ได้คำตอบ
  • ตัวเลขเวลาแยกสาเหตุที่ต่างกันได้ โดยไม่บอกอะไรเกี่ยวกับข้อมูล
