1. ปัญหาที่บทนี้แก้

โค้ดที่คุยกับเซิร์ฟเวอร์เป็นส่วนที่มีกรณีขอบเยอะที่สุด แต่ทดสอบยากที่สุด

กรณีที่ต้องทดสอบทำให้เกิดจริงได้ไหม
เซิร์ฟเวอร์ตอบ 409ยาก — ต้องหาสถานการณ์ที่ชนกันพอดี
เน็ตหลุดกลางคันต้องถอดสาย หรือปิด Wi-Fi ตอนที่กำลังยิงพอดี
เซิร์ฟเวอร์ตอบ HTML แทน JSONเกือบเป็นไปไม่ได้ — ต้องรอให้ระบบเครือข่ายมีปัญหา
ตอบช้าจนหมดเวลารอต้องทำให้เซิร์ฟเวอร์ช้าจริง

ผลคือคนส่วนใหญ่ทดสอบแค่กรณีที่สำเร็จ แล้วกรณีที่เหลือไปเจอครั้งแรกที่หน้างาน

3. Mental model: เปิดช่องไว้ที่จุดที่ติดต่อโลกภายนอก

โค้ดของเรา ──→ [ช่อง] ──→ เครือข่ายจริง
                  │
                  └──────→ ของปลอมที่เราคุมได้ (ตอนเทสต์)
ประโยคเดียวที่ต้องจำ: เปิดช่องไว้ที่จุดที่ต่ำที่สุดที่ยังเป็นโค้ดของเรา — ยิ่งต่ำ ยิ่งทดสอบของจริงได้มาก

ใน lab ช่องนั้นคือพารามิเตอร์เดียวใน constructor

ApiClient({required this.baseUrl, http.Client? client})
  : _client = client ?? http.Client();

ตอนใช้งานจริงไม่ต้องส่งอะไร มันสร้างของจริงให้เอง ส่วนตอนเทสต์ส่งของปลอมเข้าไป

4. เขียนเทสต์แรก

เปิด test/api_client_test.dart

/// สร้าง ApiClient ที่ตอบตามที่เรากำหนด โดยไม่ต้องมี server จริง
///
/// [onRequest] เก็บคำขอที่ถูกส่งไว้ให้ตรวจได้ — เพราะบั๊กจำนวนมาก
/// อยู่ที่ "ส่งอะไรไป" ไม่ใช่ "แปลผลอย่างไร"
ApiClient clientThatReturns(
  http.Response Function(http.Request request) respond, {
  void Function(http.Request request)? onRequest,
  String baseUrl = 'http://wms.local:5000',
}) => ApiClient(
  baseUrl: baseUrl,
  client: MockClient((request) async {
    onRequest?.call(request);
    return respond(request);
  }),
);

ตัวช่วยตัวเดียวนี้ทำให้เทสต์ที่เหลือสั้นและอ่านง่าย

test('อ่าน JSON ที่เป็นภาษาไทยได้ถูกต้อง', () async {
  final api = clientThatReturns(
    (_) => jsonResponse({'supplierName': 'บริษัท ซัพพลาย จำกัด'}, 200),
  );

  final json = await api.getJson('/api/WMS/inbound');

  expect(json['supplierName'], 'บริษัท ซัพพลาย จำกัด');
});

ไม่มีเซิร์ฟเวอร์ ไม่มีเครือข่าย และรันจบในไม่กี่มิลลิวินาที

5. ตรวจสิ่งที่ส่งไป ไม่ใช่แค่สิ่งที่ได้กลับ

นี่คือส่วนที่คนมักข้าม และเป็นที่ที่บั๊กซ่อนอยู่เยอะที่สุด

test('POST ส่ง body เป็น JSON และแนบ header เพิ่มที่ส่งเข้ามา', () async {
  http.Request? sent;
  final api = clientThatReturns(
    (_) => jsonResponse(const <String, Object>{}, 200),
    onRequest: (request) => sent = request,
  );

  await api.postJson(
    '/api/WMS/mobile_tasks/T-001/complete',
    const {'quantity': 12},
    headers: {'Idempotency-Key': 'cmd-8821'},
  );

  expect(sent!.method, 'POST');
  expect(jsonDecode(sent!.body), {'quantity': 12});
  expect(sent!.headers['Idempotency-Key'], 'cmd-8821');
});
สิ่งที่เทสต์นี้จับได้
เผลอใช้ GET แทน POST
ลืม jsonEncode แล้วส่ง Map ตรง ๆ (7.4)
รหัสคำสั่งไม่ถูกส่งไป — ซึ่งทำให้ระบบกันซ้ำใช้ไม่ได้เลย (7.15)

สามข้อนี้ถ้าไม่ตรวจ จะไม่มีทางรู้จนกว่าจะเจอที่หน้างาน — เพราะเทสต์ที่ตรวจแค่ผลลัพธ์จะผ่านหมด ในเมื่อของปลอมตอบสำเร็จอยู่แล้วไม่ว่าจะส่งอะไรไป

และเทสต์ตัวหนึ่งใน lab ตรึงบั๊กที่ 7.5 พูดถึงไว้โดยเฉพาะ

test('เข้าสู่ระบบแล้วต้องแนบ token และอ่านค่าล่าสุดทุกครั้ง', () async {
  ...
  await api.getJson('/api/WMS/mobile_tasks');

  // จำลองการออกจากระบบ — คำขอถัดไปต้องไม่มี token เก่าติดไป
  api.authToken = null;
  await api.getJson('/api/WMS/mobile_tasks');

  expect(tokens, ['Bearer token-คนแรก', null]);
});

6. ทดสอบกรณีที่ทำให้เกิดจริงไม่ได้

test('ได้ HTML แทน JSON ต้องไม่พังแบบอธิบายไม่ได้', () async {
  // เกิดจริงเมื่อตัวกลางในเครือข่ายคืนหน้าเข้าสู่ระบบขององค์กร
  final api = clientThatReturns(
    (_) => http.Response('<html><body>Login</body></html>', 500),
  );

  await expectLater(
    api.getJson('/api/WMS/mobile_tasks'),
    throwsA(isA<ApiException>()
        .having((e) => e.kind, 'kind', ApiFailureKind.server)),
  );
});
test('POST ที่เน็ตหลุด ไม่รู้ว่า server ทำไปแล้วหรือยัง', () async {
  final api = ApiClient(
    baseUrl: 'http://wms.local',
    client: MockClient((_) async => throw const SocketException('ไม่มีเส้นทาง')),
  );

  await expectLater(
    api.postJson('/api/WMS/mobile_tasks/T-001/complete', const {}),
    throwsA(isA<ApiException>()
        .having((e) => e.outcomeUnknown, 'outcomeUnknown', isTrue)),
  );
});

สังเกตว่าของปลอม โยน exception ได้ ไม่ใช่แค่ตอบกลับ ซึ่งทำให้จำลองเน็ตหลุดได้โดยไม่ต้องถอดสาย

และเทสต์คู่นี้ตรึงข้อสรุปสำคัญของ 7.4 ไว้ — GET กับ POST ตั้ง outcomeUnknown ต่างกัน ซึ่งเป็นสิ่งที่ 4.13 เอาไปใช้ตัดสินว่าจะย้อนกลับหรือไปตรวจสอบ

7. รายการกรณีที่ควรครอบ

กลุ่มกรณี
คำขอmethod, path, header, body ที่ส่งไป
สำเร็จ200 พร้อมข้อมูล, 200 ภาษาไทย, 204 ไม่มีเนื้อหา
ล้มเหลวที่เซิร์ฟเวอร์บอก400 พร้อมข้อผิดพลาดรายช่อง, 401, 403, 409, 500
ล้มเหลวที่รูปแบบHTML แทน JSON, JSON ที่โครงผิด
ล้มเหลวที่การเชื่อมต่อเน็ตหลุดตอน GET, เน็ตหลุดตอน POST, หมดเวลารอ

สังเกตว่าสามกลุ่มล่างเป็นกลุ่มที่ทำให้เกิดจริงยาก — และเป็นกลุ่มที่ทำให้แอปพังที่หน้างานมากที่สุด

8. ใครเป็นคนปิด client

คำถามเรื่องความเป็นเจ้าของกลับมาอีกครั้ง เหมือนใน 4.12

ApiClient({required this.baseUrl, http.Client? client})
  : _client = client ?? http.Client();
...
void close() => _client.close();
สถานการณ์ใครควรปิด
ApiClient สร้าง client เองApiClient ปิดเอง — ถูกต้อง
รับ client จากข้างนอกมาเจ้าของเดิม ไม่ใช่ ApiClient

โค้ดข้างบนปิดทุกกรณี ซึ่งเป็นปัญหาถ้ามีการใช้ client ตัวเดียวร่วมกันหลายที่ — ปิดจากที่หนึ่งแล้วอีกที่จะใช้ไม่ได้

// เข้มขึ้น — จำไว้ว่าใครสร้าง
ApiClient({required this.baseUrl, http.Client? client})
  : _client = client ?? http.Client(),
    _ownsClient = client == null;

void close() {
  if (_ownsClient) _client.close();   // ปิดเฉพาะตัวที่สร้างเอง
}

ในเทสต์เรื่องนี้ไม่กระทบ เพราะ MockClient ไม่ได้ถือทรัพยากรจริง — แต่ในแอปที่ใช้ client ร่วมกันเพื่อใช้การเชื่อมต่อซ้ำ การปิดผิดตัวทำให้คำขอถัดไปล้มเหลวโดยไม่มีเหตุผลที่ชัดเจน

9. อย่าเปิดช่องสูงเกินไป

ทางเลือกที่ดูง่ายกว่าคือปลอมที่ระดับ ApiClient เลย

class FakeApiClient implements JsonApi {
  @override
  Future<Map<String, dynamic>> getJson(String path) async => {'items': []};
}
ปลอมที่ http.Clientปลอมที่ JsonApi
ทดสอบอะไรได้บ้างรหัสตอบกลับ, header, การแปลง JSON, timeoutเฉพาะตรรกะของชั้นบน
เหมาะกับทดสอบApiClient เองrepository และ store

ทั้งสองแบบมีที่ใช้ และควรมีทั้งคู่

ApiClient       ← ทดสอบด้วย MockClient      (api_client_test.dart)
repository      ← ทดสอบด้วย JsonApi ปลอม    (task_repository_test.dart)
store           ← ทดสอบด้วย repository ปลอม (task_store_test.dart)

แต่ละชั้นทดสอบด้วยของปลอมของชั้นที่อยู่ต่ำกว่าตัวเองหนึ่งชั้น ซึ่งตรงกับโครงใน 7.1 พอดี

ถ้าปลอมที่ระดับสูงอย่างเดียว จะไม่มีอะไรทดสอบว่ารหัส 409 ถูกแปลงเป็นชนิดที่ถูกต้อง — เพราะชั้นนั้นถูกข้ามไปทั้งชั้น

10. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ทดสอบได้แต่กรณีที่สำเร็จไม่มีช่องให้แทนของจริงรับ client ผ่าน constructor
เทสต์ช้าและแดงเป็นครั้งคราวยิงเข้าเซิร์ฟเวอร์จริงใช้ของปลอม
ส่งคำขอผิดแต่เทสต์เขียวตรวจแค่ผลลัพธ์ตรวจคำขอที่ส่งไปด้วย
รหัสคำสั่งหายไปโดยไม่มีใครรู้ไม่ได้ตรวจ header ที่ส่งตรวจ
คำขอถัดไปล้มเหลวหลังปิด clientปิด client ที่คนอื่นเป็นเจ้าของปิดเฉพาะตัวที่สร้างเอง
ไม่มีอะไรทดสอบการแปลงรหัสตอบกลับปลอมที่ระดับสูงเกินไปปลอมที่ http.Client ด้วย

11. แบบฝึกหัด

ทีมพบว่าคำขอค้นหาที่มีรหัส A&B 01 ได้ผลไม่ตรง แต่ไม่รู้ว่าปัญหาอยู่ฝั่งแอปหรือฝั่งเซิร์ฟเวอร์

  1. เขียนเทสต์ที่พิสูจน์ว่าฝั่งแอปส่งไปถูก
  2. อธิบายว่าทำไมไม่ควรตรวจด้วยการเทียบ URL ทั้งข้อความ
  3. ถ้าเทสต์ผ่าน แต่ของจริงยังผิด สรุปได้ว่าอย่างไร และควรทำอะไรต่อ
ดูแนวคำตอบและเหตุผล

ข้อ 1 — เทสต์

test('รหัสที่มีอักขระพิเศษต้องถูก encode และถอดกลับได้ค่าเดิม', () async {
  http.Request? sent;
  final api = clientThatReturns(
    (_) => jsonResponse({'items': <Object>[]}, 200),
    onRequest: (request) => sent = request,
  );

  await api.getJson('/api/WMS/mobile_tasks?search=${Uri.encodeQueryComponent('A&B 01')}');

  // ตรวจค่าที่ถอดกลับแล้ว ไม่ใช่ข้อความดิบ
  expect(sent!.url.queryParameters['search'], 'A&B 01');
  expect(sent!.url.queryParameters.keys, ['search']);
});

บรรทัดสุดท้ายสำคัญ — ตรวจว่ามี query แค่ตัวเดียว เพราะถ้า & ไม่ถูก encode มันจะกลายเป็นสองพารามิเตอร์ ซึ่งเป็นอาการที่โจทย์บอกพอดี

?search=A&B 01
  → queryParameters = {'search': 'A', 'B 01': ''}    ← สองตัว
?search=A%26B+01
  → queryParameters = {'search': 'A&B 01'}          ← ตัวเดียว ถูกต้อง

ข้อ 2 — ทำไมไม่เทียบ URL ทั้งข้อความ

// เปราะ
expect(sent!.url.toString(),
    'http://wms.local:5000/api/WMS/mobile_tasks?search=A%26B+01');
ปัญหารายละเอียด
encode ได้หลายแบบที่ถูกต้องเท่ากันช่องว่างเป็น + หรือ %20 ก็ถูกทั้งคู่
ลำดับ query ไม่รับประกันเพิ่มพารามิเตอร์ที่สองแล้วเทสต์แดงทั้งที่ถูก
อ่านไม่ออกว่าตรวจอะไร%26 ต้องแปลในหัวก่อนถึงจะรู้ว่าคือ &

การตรวจที่ queryParameters ตรวจ ความหมาย ส่วนการตรวจข้อความตรวจ วิธีเขียน — และวิธีเขียนที่ต่างกันอาจถูกต้องเท่ากัน

เป็นหลักเดียวกับ 6.3 — ผ่อนได้กับรูปแบบการเขียน แต่ห้ามผ่อนกับความหมาย

ข้อ 3 — เทสต์ผ่านแต่ของจริงยังผิด

สรุปได้ทันทีว่า ฝั่งแอปส่งไปถูกแล้ว ปัญหาอยู่ฝั่งเซิร์ฟเวอร์หรือระหว่างทาง

ซึ่งเป็นข้อมูลที่มีค่ามาก เพราะมันเปลี่ยนคำถามจาก “ใครผิด” เป็น “ผิดตรงไหนหลังจากที่เราส่งไปแล้ว”

สิ่งที่ควรทำต่อ เรียงจากถูกที่สุด

ขั้นทำอะไรบอกอะไร
1ยิง URL เดียวกันด้วยเครื่องมือทดสอบ APIถ้าผิดเหมือนกัน แปลว่าเซิร์ฟเวอร์ ไม่ใช่แอป
2ดู log ฝั่งเซิร์ฟเวอร์ว่ารับค่าอะไรไปแยก "ระหว่างทางเพี้ยน" ออกจาก "เซิร์ฟเวอร์อ่านผิด"
3ถ้าค่าที่เซิร์ฟเวอร์รับถูก แต่ผลผิดปัญหาอยู่ที่การค้นในฐานข้อมูล

ขั้นที่ 1 สำคัญเพราะมันตัดตัวแปรทั้งหมดที่เป็นของแอปออกไป — และถ้าเครื่องมือทดสอบให้ผลถูกต้อง แต่แอปไม่ถูก แปลว่าเทสต์เรายังตรวจไม่ครบ ต้องกลับมาดูว่าของจริงส่งอะไรต่างจากที่เทสต์คิด

บทเรียนที่ควรจำ: เทสต์ที่ตรวจคำขอมีคุณค่ามากกว่าการจับบั๊ก — มันทำให้ ตัดฝั่งหนึ่งออกจากการสงสัยได้ ซึ่งในระบบที่มีสองทีมดูแลคนละฝั่ง เป็นสิ่งที่ประหยัดเวลาได้มากที่สุด

12. สรุปท้ายบท

  • เปิดช่องรับ client ผ่าน constructor เพื่อให้แทนของจริงได้ตอนเทสต์
  • ของปลอมตอบกลับได้และโยน exception ได้ จึงจำลองเน็ตหลุดได้
  • ตรวจคำขอที่ส่งไปด้วย ไม่ใช่แค่ผลลัพธ์ที่ได้
  • กรณีที่ทำให้เกิดจริงยาก คือกรณีที่ทำให้แอปพังที่หน้างานมากที่สุด
  • ตรวจ queryParameters ไม่ใช่เทียบ URL ทั้งข้อความ
  • ปิดเฉพาะ client ที่สร้างเอง
  • แต่ละชั้นทดสอบด้วยของปลอมของชั้นที่ต่ำกว่าหนึ่งชั้น
  • เทสต์ที่ตรวจคำขอ ช่วยตัดฝั่งหนึ่งออกจากการสงสัยได้
