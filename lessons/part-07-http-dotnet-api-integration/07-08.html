3. Mental model: สามด่านที่ต้องผ่าน

ประโยคเดียวที่ต้องจำ: รหัส 200 ผ่านแค่ด่านแรก — และการเขียนโค้ดเหมือนว่ามันผ่านครบทุกด่าน คือที่มาของข้อผิดพลาดที่อ่านไม่รู้เรื่องตอนรัน

4. ใครทำอะไรใน lab

ด่านทำที่โยนอะไรเมื่อไม่ผ่าน
รหัสตอบกลับApiClient._decodeApiException พร้อม kind

เส้นแบ่งคือ ApiClient ตอบว่า “คุยกันสำเร็จไหม” ส่วนชั้นบนตอบว่า “ข้อมูลใช้ได้ไหม”

เหตุผลที่แบ่งแบบนี้ — ApiClient ไม่รู้ว่ากำลังคุยเรื่องอะไร มันเรียกได้ทั้ง endpoint งานและ endpoint พาเลท จึงตัดสินไม่ได้ว่า field ไหนต้องมี

5. body ที่ไม่ใช่ JSON

เปิด lib/core/api_client.dart

Map<String, dynamic>? json;
try {
  if (response.body.isNotEmpty) {
    json = jsonDecode(response.body) as Map<String, dynamic>;
  }
} on FormatException {
  json = null;
}

โค้ดนี้กลืน FormatException ไว้ ซึ่งดูขัดกับ 4.8 ที่บอกว่าห้ามกลืน — แต่ตรงนี้มีเหตุผล

สถานการณ์สิ่งที่เกิด
รหัส 500 พร้อมหน้า HTMLแปลงไม่ได้ แต่รหัสบอกอยู่แล้วว่าล้มเหลว
รหัส 200 พร้อมหน้า HTMLแปลงไม่ได้ และคืน map ว่าง

แถวแรกคือกรณีที่การกลืนถูกต้อง — ถ้าโยน FormatException ออกไป ข้อความจะเป็น “แปลง JSON ไม่ได้” ทั้งที่เรื่องจริงคือเซิร์ฟเวอร์ล่ม ซึ่งชี้ผิดทาง

เทสต์ใน lab ตรึงพฤติกรรมนี้ไว้

test('ได้ HTML แทน JSON ต้องไม่พังแบบอธิบายไม่ได้', () async {
  // เกิดจริงเมื่อตัวกลางในเครือข่ายคืนหน้าเข้าสู่ระบบขององค์กร
  final api = clientThatReturns(
    (_) => http.Response('<html><body>Login</body></html>', 500),
  );

  await expectLater(
    api.getJson('/api/WMS/mobile_tasks'),
    throwsA(isA<ApiException>()
        .having((e) => e.kind, 'kind', ApiFailureKind.server)),
  );
});
แต่แถวที่สองยังเป็นช่องโหว่ — รหัส 200 พร้อม HTML จะได้ map ว่าง แล้วชั้นบนจะบอกว่า “ไม่มีคีย์ items” ซึ่งชี้ไปที่สัญญาทั้งที่จริง ๆ เป็นปัญหาเครือข่าย เรื่องนี้แก้ได้ด้วยการตรวจ Content-Type ตามหัวข้อ 7

6. ภาษาไทยกับการเข้ารหัส

// ปลอดภัยกว่า — อ่านจากไบต์ดิบแล้วแปลงเอง
final text = utf8.decode(response.bodyBytes);
final json = jsonDecode(text);
response.bodyutf8.decode(bodyBytes)
ใช้อะไรตัดสินการเข้ารหัสดูจาก header ที่เซิร์ฟเวอร์ส่งมาบังคับเป็น UTF-8 เสมอ
ถ้า header ไม่ระบุเดาเป็นแบบเก่าที่ไม่รองรับภาษาไทยยังอ่านถูก

อาการที่เกิดคือชื่อผู้จำหน่ายภาษาไทยกลายเป็นตัวอักษรแปลก ๆ ทั้งที่ข้อมูลในฐานข้อมูลถูกต้อง

ในทางปฏิบัติ ASP.NET ส่ง header มาถูกอยู่แล้ว จึงมักไม่มีปัญหา — แต่เมื่อมีตัวกลางในเครือข่ายมาแทรก header อาจหายไป

เทสต์ใน lab ครอบเรื่องนี้ไว้

test('อ่าน JSON ที่เป็นภาษาไทยได้ถูกต้อง', () async {
  final api = clientThatReturns(
    (_) => jsonResponse({'supplierName': 'บริษัท ซัพพลาย จำกัด'}, 200),
  );

  final json = await api.getJson('/api/WMS/inbound');

  expect(json['supplierName'], 'บริษัท ซัพพลาย จำกัด');
});

7. ตรวจว่าเป็น JSON จริงก่อน

วิธีปิดช่องโหว่ในหัวข้อ 5

final contentType = response.headers['content-type'] ?? '';
if (response.statusCode >= 200 &&
    response.statusCode < 300 &&
    response.body.isNotEmpty &&
    !contentType.contains('application/json')) {
  throw ApiException(
    'เซิร์ฟเวอร์ตอบกลับมาในรูปแบบที่ไม่รองรับ ($contentType)',
    statusCode: response.statusCode,
    kind: ApiFailureKind.contract,
  );
}
ได้อะไร
รู้ทันทีว่าไม่ใช่ JSON แทนที่จะไปรู้ตอนหา field ไม่เจอ
ข้อความบอกรูปแบบที่ได้จริง ทำให้เดาได้ว่ามีตัวกลางมาแทรก
ชี้ไปที่ปัญหาเครือข่าย ไม่ใช่ปัญหาสัญญา

แต่ต้องระวังว่าบางเซิร์ฟเวอร์ส่ง content-type ที่มีข้อความต่อท้าย เช่น application/json; charset=utf-8 จึงต้องใช้ contains ไม่ใช่เทียบเท่ากันตรง ๆ

8. คำตอบที่ไม่มีเนื้อหา

if (response.statusCode >= 200 && response.statusCode < 300) {
  // 204 และ 200 ที่ body ว่าง ถือว่าสำเร็จ ไม่ต้องมี JSON
  return json ?? <String, dynamic>{};
}

การคืน map ว่างแทนที่จะโยน ทำให้ผู้เรียกไม่ต้องเขียนเงื่อนไขพิเศษ

test('204 ที่ไม่มีเนื้อหา ถือว่าสำเร็จ ไม่ใช่พัง', () async {
  final api = clientThatReturns((_) => http.Response('', 204));

  final json = await api.postJson('/api/WMS/ping', const {});

  expect(json, isEmpty);
});

แต่มีข้อแลกที่ต้องรู้ตัว — ถ้า endpoint ที่ควรคืนข้อมูลกลับคืน body ว่างมา จะได้ map ว่างแทนที่จะได้ข้อผิดพลาด

ชั้นบนจะจับได้อยู่ดีตอนที่หา field ไม่เจอ ตาม 6.2 — แต่ข้อความจะเป็น “ไม่มีคีย์ items” ซึ่งชี้ไปที่สัญญา ทั้งที่จริง ๆ เซิร์ฟเวอร์ไม่ได้ส่งอะไรมาเลย

ถ้าอยากให้ชี้ตรงกว่านั้น ต้องให้ผู้เรียกบอกว่าคาดหวังเนื้อหาหรือไม่

Future<Map<String, dynamic>> getJson(
  String path, {
  bool expectBody = true,     // GET มักคาดหวังเนื้อหา
}) async { ... }

9. อย่าส่งข้อมูลดิบขึ้นไป

// ที่ lab ทำ — คืน Map แล้วให้ repository แปลงต่อ
Future<Map<String, dynamic>> getJson(String path);

คำถามที่สมเหตุสมผล — ทำไมไม่ให้ ApiClient แปลงเป็น model ไปเลย

ถ้า ApiClient แปลงเองปัญหา
ต้องรู้จัก model ทุกตัวในแอปชั้นล่างสุดผูกกับทุกฟีเจอร์
เพิ่มฟีเจอร์ต้องแก้ ApiClientขัดกับหลักใน 7.1

แต่การคืน Map ก็มีข้อแลก — ค่าใน map เป็น dynamic ซึ่งเป็นสิ่งที่ 6.8 บอกให้ระวัง

สิ่งที่ทำให้ยอมรับได้คือ ขอบเขตที่ dynamic อยู่ได้ต้องแคบมาก

ApiClient        คืน Map<String, dynamic>   ← dynamic เริ่มที่นี่
repository       แปลงเป็น DTO ทันที          ← dynamic จบที่นี่
store, หน้าจอ    ไม่มี dynamic แล้ว

ถ้าเห็น Map<String, dynamic> ไปโผล่ที่ store หรือหน้าจอ แปลว่ามีอะไรรั่วออกมาจากขอบเขตนี้

10. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ภาษาไทยเพี้ยนเซิร์ฟเวอร์ไม่ระบุการเข้ารหัสอ่านจาก bodyBytes แล้วแปลงเป็น UTF-8 เอง
FormatException: Unexpected characterได้ HTML แทน JSONตรวจ content-type ก่อน
“ไม่มีคีย์ items” ทั้งที่ปัญหาคือเครือข่ายไม่ตรวจรูปแบบก่อนเหมือนข้างบน
FormatException ทั้งที่คำขอสำเร็จพยายามแปลง body ว่างตรวจว่ามีเนื้อหาก่อน
ข้อผิดพลาดชี้ไปที่หน้าจอ ทั้งที่ปัญหาอยู่ที่สัญญาdynamic รั่วขึ้นไปแปลงเป็น DTO ที่ repository
เพิ่มฟีเจอร์แล้วต้องแก้ ApiClientให้ชั้นล่างรู้จัก modelให้คืนข้อมูลกลาง แล้วแปลงที่ชั้นบน

11. แบบฝึกหัด

ทีมได้รายงานสามอย่างในสัปดาห์เดียวกัน

#อาการ
1ชื่อผู้จำหน่ายภาษาไทยกลายเป็นตัวอักษรแปลก เฉพาะที่สาขาหนึ่ง
2บางครั้งขึ้นว่า “ไม่มีคีย์ items” ทั้งที่ endpoint นั้นใช้งานได้ปกติมาตลอด
3กดปุ่มหนึ่งแล้วขึ้น FormatException ทั้งที่ข้อมูลถูกบันทึกเรียบร้อย
  1. ระบุด่านที่มีปัญหาของแต่ละข้อ ตามหัวข้อ 3
  2. เสนอวิธีแก้
  3. อาการที่ 2 บอกใบ้อะไรที่สำคัญ
ดูแนวคำตอบและเหตุผล

ข้อ 1 และ 2 — ด่านที่มีปัญหาและวิธีแก้

#ด่านสาเหตุวิธีแก้
1แปลงเป็นข้อความheader การเข้ารหัสหายไปอ่านจาก bodyBytes
2เป็น JSON ไหมได้ HTML พร้อมรหัส 200ตรวจ content-type
3เป็น JSON ไหมได้ 204 แต่พยายามแปลงตรวจว่ามีเนื้อหาก่อน

ข้อ 1 ที่เกิด เฉพาะสาขาหนึ่ง ชี้ชัดว่ามีอะไรต่างที่สาขานั้น — น่าจะมีตัวกลางในเครือข่ายที่สาขานั้นเท่านั้น

ข้อ 3 น่าสนใจเพราะ ข้อมูลถูกบันทึกแล้ว แปลว่าคำขอสำเร็จทุกอย่าง ปัญหาอยู่ที่ฝั่งเราตีความคำตอบผิด

// สาเหตุ
if (response.body.isNotEmpty) {          // 204 มี body ว่าง จึงข้ามไป
  json = jsonDecode(response.body) ...   // ถ้าลืมเช็ก จะพังตรงนี้
}

ใน lab เช็กไว้แล้ว แต่ถ้า endpoint นั้นตอบ 200 พร้อม body ว่าง แทนที่จะเป็น 204 ก็จะเข้าเงื่อนไขเดียวกัน

ข้อ 3 — อาการที่ 2 บอกใบ้อะไร

คำว่า “บางครั้ง” และ “ใช้งานได้ปกติมาตลอด” เป็นข้อมูลสำคัญสองอย่าง

ถ้าเป็นปัญหาสัญญาถ้าเป็นปัญหาเครือข่าย
ต้องพังทุกครั้งพังเป็นครั้งคราว
พังตั้งแต่ที่เซิร์ฟเวอร์เปลี่ยนพังโดยไม่มีใครแก้อะไร
พังทุกเครื่องอาจพังบางเครื่องหรือบางช่วงเวลา

อาการที่รายงานตรงกับคอลัมน์ขวาทั้งหมด ดังนั้น ข้อความ “ไม่มีคีย์ items” กำลังชี้ผิดทาง

สิ่งที่น่าจะเกิดจริง

token หมดอายุ หรือ session ที่ตัวกลางหมดอายุ
  → ตัวกลางคืนหน้าเข้าสู่ระบบ พร้อมรหัส 200
  → ApiClient แปลง HTML ไม่ได้ → คืน map ว่าง
  → repository หา items ไม่เจอ → "ไม่มีคีย์ items"

ข้อความสุดท้ายทำให้ทีมไปไล่ดูสัญญากับฝั่งหลังบ้าน ซึ่งไม่มีอะไรผิด และเสียเวลาไปโดยเปล่าประโยชน์

สิ่งที่การตรวจ content-type ให้

// ข้อความที่ได้หลังแก้
เซิร์ฟเวอร์ตอบกลับมาในรูปแบบที่ไม่รองรับ (text/html)

ข้อความนี้ชี้ไปที่ตัวกลางในเครือข่ายทันที และคำว่า text/html ทำให้เดาได้เลยว่าเป็นหน้าเข้าสู่ระบบขององค์กร

บทเรียนที่ควรจำ: เมื่ออาการเป็น “บางครั้ง” และ “ไม่มีใครแก้อะไร” ให้สงสัยชั้นล่างก่อนเสมอ — และข้อความผิดพลาดที่ชี้ผิดชั้น ทำให้เสียเวลามากกว่าการไม่มีข้อความเลย เพราะมันพาไปหาคำตอบผิดที่

12. สรุปท้ายบท

  • รหัส 200 แปลว่าส่งสำเร็จ ไม่ได้แปลว่าข้อมูลใช้ได้
  • มีสี่ด่านที่ต้องผ่าน และแต่ละด่านอยู่คนละชั้น
  • ApiClient ตอบว่าคุยกันสำเร็จไหม ชั้นบนตอบว่าข้อมูลใช้ได้ไหม
  • อ่านจาก bodyBytes เพื่อให้ภาษาไทยไม่เพี้ยน
  • ตรวจ content-type เพื่อจับกรณีที่ได้ HTML แทน JSON
  • body ว่างถือว่าสำเร็จ แต่ต้องตรวจก่อนแปลง
  • คืนข้อมูลกลางแล้วให้ชั้นบนแปลง เพื่อไม่ให้ชั้นล่างรู้จักทุกฟีเจอร์
  • ข้อความผิดพลาดที่ชี้ผิดชั้น เสียเวลากว่าไม่มีข้อความ
