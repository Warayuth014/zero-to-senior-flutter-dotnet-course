1. ปัญหาที่บทนี้แก้

มีสองวิธีที่ใช้กันทั่วไปในการบอกชั้นบนว่าล้มเหลว

// แบบโยน — ที่ lab ใช้
Future<List<WmsTask>> fetchOpen();      // สำเร็จได้ List, ล้มเหลวโยน

// แบบคืนค่า
Future<ApiResult<List<WmsTask>>> fetchOpen();   // คืนค่าเสมอ ไม่โยน

ทั้งสองแบบใช้ได้ และทีมที่เก่งใช้ทั้งสองแบบ — ปัญหาไม่ได้อยู่ที่เลือกผิด แต่อยู่ที่

ปัญหาจริงอาการ
ผสมสองแบบในโปรเจกต์เดียวผู้เรียกต้องรับมือทั้ง try และ switch
ไม่มีข้อตกลงว่าอะไรโยนอะไรคืนต้องเปิดโค้ดดูทุกครั้งว่าเมธอดนี้แบบไหน
ใช้ null แทนความล้มเหลวแยกไม่ออกว่า “ไม่พบ” “เน็ตหลุด” หรือ “แปลงไม่ได้”

2. คำศัพท์ที่ต้องรู้ก่อน

คำความหมาย
throwโยนความล้มเหลวออกไปให้คนอื่นดัก
Resultชนิดที่พกได้ทั้งความสำเร็จและความล้มเหลว
happy pathเส้นทางที่ทุกอย่างสำเร็จ
exhaustiveครอบทุกกรณี ไม่มีทางตกหล่น
rethrowโยนตัวเดิมต่อ โดยรักษาร่องรอยไว้

3. Mental model: ใครควรถูกบังคับให้จัดการ

โยน       →  ผู้เรียกจะลืมจัดการก็ได้ compiler ไม่บ่น
คืนค่า    →  ผู้เรียกต้องแกะค่าออกมา จึงเห็นว่ามีทางล้มเหลวอยู่
ประโยคเดียวที่ต้องจำ: ถ้าความล้มเหลวเป็นสิ่งที่ คาดไว้แล้วและต้องจัดการทุกครั้ง ให้คืนค่า — ถ้าเป็นสิ่งที่ เกิดได้แต่จัดการรวมที่เดียวก็พอ ให้โยน

4. แบบโยน

lab ใช้แบบนี้ทั้งหมด

Future<void> load() async {
  final generation = ++_requestGeneration;
  _loading = true;
  notifyListeners();
  try {
    final result = await repository.fetchOpen();
    if (generation != _requestGeneration) return;
    _tasks = result;
  } catch (exception) {
    if (generation != _requestGeneration) return;
    _error = exception.toString();
  } finally {
    ...
  }
}
ข้อดีข้อเสีย
เส้นทางที่สำเร็จอ่านง่าย ไม่มีการแกะค่าลืมดักแล้ว compiler ไม่เตือน
ผ่านหลายชั้นได้โดยไม่ต้องแกะทุกชั้นต้องอ่านเอกสารว่าโยนอะไรได้บ้าง
ดักรวมที่เดียวได้ดักกว้างเกินไปแล้วกลืนบั๊ก (4.8)

ข้อเสียข้อแรกเป็นเรื่องจริงและแก้ไม่ได้ด้วยตัวภาษา — Dart ไม่มีวิธีบังคับให้ผู้เรียกดัก

สิ่งที่ทดแทนได้คือ ดักที่ชั้นเดียวซึ่งทุกอย่างต้องผ่าน แล้วทำให้ที่นั่นครอบคลุมจริง ๆ

5. แบบคืนค่า

sealed class ApiResult<T> {
  const ApiResult();
}

class ApiSuccess<T> extends ApiResult<T> {
  const ApiSuccess(this.value);
  final T value;
}

class ApiFailure<T> extends ApiResult<T> {
  const ApiFailure(this.kind, this.message, {this.problem});
  final ApiFailureKind kind;
  final String message;
  final ProblemDetails? problem;
}
final result = await repository.fetchOpen();

switch (result) {
  case ApiSuccess(:final value):
    _tasks = value;
    _state = LoadState.ready;
  case ApiFailure(:final kind, :final message):
    _state = LoadState.failed;
    _errorMessage = message;
    _canRetry = kind.isRetryable;      // ใช้ชนิดตัดสินได้เลย
}
ข้อดีข้อเสีย
ลืมจัดการไม่ได้ — switch ต้องครบต้องแกะค่าทุกชั้นที่ผ่าน
เห็นจากลายเซ็นว่าล้มเหลวได้เส้นทางที่สำเร็จอ่านยากขึ้น
เพิ่มชนิดความล้มเหลวแล้ว compiler ไล่ให้ต้องเขียนโครงเพิ่ม

ข้อดีข้อแรกคือเหตุผลหลักที่คนเลือกแบบนี้ — และมันสำคัญมากในงานที่ความล้มเหลวเป็นเรื่องปกติ

6. เกณฑ์เลือก

สถานการณ์เลือกเพราะ
โหลดข้อมูลมาแสดงโยนล้มเหลวแล้วแสดงหน้าผิดพลาด — จัดการรวมที่เดียวได้
ยืนยันคำสั่งที่มีหลายผลลัพธ์คืนค่าสำเร็จ ขัดแย้ง ข้อมูลผิด ต้องทำคนละอย่าง
ตรวจว่าเซิร์ฟเวอร์ยังอยู่ไหมคืนค่าล้มเหลวคือคำตอบที่ต้องการ ไม่ใช่ข้อยกเว้น
เรียกต่อกันหลายชั้นโยนไม่ต้องแกะทุกชั้น

แถวที่สามน่าสนใจ — เมื่อความล้มเหลวคือผลลัพธ์ที่ต้องการ การโยนจะผิดความหมาย

// แปลก — ใช้ exception บอกผลปกติ
try {
  await api.checkHealth();
  return true;
} catch (_) {
  return false;      // ล้มเหลวคือคำตอบ ไม่ใช่ข้อยกเว้น
}

เรื่องนี้จะกลับมาใน 7.16

Future<ApiResult<List<WmsTask>>> fetchOpen() async {
  final json = await api.getJson('/api/WMS/mobile_tasks');   // ยังโยนอยู่
  ...
  return ApiSuccess(tasks);

try {

ถ้าเลือกแบบคืนค่า ต้องจับให้ครบ

Future<ApiResult<List<WmsTask>>> fetchOpen() async {
  try {
    final json = await api.getJson('/api/WMS/mobile_tasks');
    return ApiSuccess(_parseTasks(json));
  } on ApiException catch (error) {
    return ApiFailure(error.kind, error.userMessage, problem: error.problem);
  } on ContractException catch (error) {
    return ApiFailure(ApiFailureKind.contract, 'ข้อมูลไม่ตรงสัญญา: ${error.path}');
  }
  // ชนิดอื่นยังโยนต่อ — เพราะเป็นบั๊ก ไม่ใช่ผลลัพธ์
}

บรรทัดสุดท้ายสำคัญ — แม้เลือกแบบคืนค่า ก็ยังไม่จับทุกอย่าง บั๊กของเราเองต้องโยนต่อให้เห็น ตาม 4.8

เส้นแบ่งคือ — สิ่งที่คาดไว้แล้วคืนเป็นค่า สิ่งที่ไม่ควรเกิดปล่อยให้โยน

8. อย่าใช้ null แทนความล้มเหลว

// แย่กว่าทั้งสองแบบ
Future<List<WmsTask>?> fetchOpen();
null แปลว่าอะไรผู้เรียกทำอะไรได้
เน็ตหลุดควรลองใหม่
ไม่มีสิทธิ์ควรบอกให้ติดต่อหัวหน้า
ข้อมูลผิดสัญญาควรแจ้งทีม
ไม่มีงานจริง ๆควรแสดงว่าไม่มีงาน

สี่ความหมายยัดอยู่ในค่าเดียว และแยกไม่ออกเลย — ซึ่งเป็นปัญหาเดียวกับที่ 4.7 และ 6.4 เจอ

แถวสุดท้ายอันตรายที่สุด เพราะมันเป็นความสำเร็จ แต่ถูกยัดรวมกับความล้มเหลว

9. รักษาร่องรอยไว้

// ผิด — ร่องรอยว่าเกิดที่ไหนหายไป
try {
  ...
} on ApiException catch (error) {
  throw error;        // ร่องรอยเริ่มใหม่ที่บรรทัดนี้
}
// ถูก
} on ApiException {
  rethrow;            // ร่องรอยเดิมยังอยู่
}

และถ้าเลือกแบบคืนค่า ควรเก็บสาเหตุไว้ด้วย

class ApiFailure<T> extends ApiResult<T> {
  const ApiFailure(this.kind, this.message, {this.problem, this.cause});

  /// เก็บไว้สำหรับ log ไม่ใช่สำหรับแสดงให้ผู้ใช้
  final Object? cause;
}

เพราะเมื่อแปลงความล้มเหลวเป็นค่า ร่องรอยเดิมจะหายไปทันที — และถ้าไม่เก็บไว้ จะไล่หาสาเหตุไม่ได้เลย

10. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ผู้เรียกต้องเขียนทั้ง try และ switchผสมสองแบบเลือกแบบเดียวต่อเมธอด
ต้องเปิดโค้ดดูว่าเมธอดนี้โยนหรือคืนไม่มีข้อตกลงในทีมตกลงแล้วเขียนไว้
ลืมดักแล้วแอปพังเลือกแบบโยนแต่ไม่มีจุดดักรวมดักที่ชั้นเดียวที่ทุกอย่างผ่าน
“ไม่มีงาน” กับ “โหลดไม่ได้” แยกไม่ออกใช้ null แทนความล้มเหลวใช้ชนิดที่แยกได้
รู้ว่าพังแต่ไม่รู้ว่าที่ไหนใช้ throw errorใช้ rethrow
บั๊กของเราถูกแปลงเป็นผลลัพธ์ปกติจับทุกอย่างแล้วคืนค่าจับเฉพาะที่คาดไว้

11. แบบฝึกหัด

แอปมี repository สี่เมธอด

#เมธอดผลลัพธ์ที่เป็นไปได้
1fetchOpen()ได้รายการ หรือล้มเหลว
2complete(id, commandId)สำเร็จ / ปิดไปแล้ว / ถูกล็อก / ของไม่พอ / ล้มเหลว
3checkHealth()ต่อได้ หรือต่อไม่ได้
4findByBarcode(code)เจอ / ไม่เจอ / ล้มเหลว
  1. เลือกแบบโยนหรือคืนค่าของแต่ละเมธอด พร้อมเหตุผล
  2. เขียนลายเซ็นของเมธอดที่ 2 และ 4
  3. ตอบว่าการมีทั้งสองแบบในไฟล์เดียวกัน ยอมรับได้หรือไม่
ดูแนวคำตอบและเหตุผล

ข้อ 1 — เลือกแบบไหน

#เลือกเหตุผล
1โยนล้มเหลวมีทางเดียวคือแสดงหน้าผิดพลาด จัดการรวมได้
2คืนค่าห้าผลลัพธ์ที่ต้องทำคนละอย่าง และลืมจัดการไม่ได้
3คืนค่า“ต่อไม่ได้” คือคำตอบ ไม่ใช่ข้อยกเว้น
4คืนค่า“ไม่เจอ” เป็นผลปกติ ไม่ใช่ความล้มเหลว

ข้อ 4 เป็นข้อที่คนตอบผิดบ่อยที่สุด — การสแกนแล้วไม่เจอรหัสนั้นเป็นเรื่องปกติที่เกิดทุกวัน ไม่ใช่ข้อยกเว้น

// ผิด — ทำให้เรื่องปกติกลายเป็นความผิดพลาด
try {
  final product = await repository.findByBarcode(code);
  ...
} on NotFoundException {
  showError('ไม่พบสินค้า');      // ไม่ใช่ error
}

ข้อ 2 — ลายเซ็น

sealed class CompleteOutcome {
  const CompleteOutcome();
}

class CompleteSucceeded extends CompleteOutcome {
  const CompleteSucceeded(this.correlationId);
  final String correlationId;
}

/// ปิดไปแล้ว — ไม่ใช่ความผิดพลาด ผลลัพธ์ที่ต้องการเกิดขึ้นแล้ว
class AlreadyCompleted extends CompleteOutcome {
  const AlreadyCompleted({this.completedBy, this.completedAt});
  final String? completedBy;
  final DateTime? completedAt;
}

/// ถูกล็อก — รอแล้วลองใหม่ได้
class LockedByOther extends CompleteOutcome {
  const LockedByOther(this.lockedBy);
  final String lockedBy;
}

/// ของไม่พอ — ต้องแก้จำนวนก่อน
class InsufficientStock extends CompleteOutcome {
  const InsufficientStock(this.available);
  final int available;
}

/// ล้มเหลวจริง ๆ
class CompleteFailed extends CompleteOutcome {
  const CompleteFailed(this.kind, this.message);
  final ApiFailureKind kind;
  final String message;
}

Future<CompleteOutcome> complete(String id, {required String commandId});

สังเกตว่าแต่ละกรณีพกข้อมูลที่ต่างกัน ตามหลักใน 4.15 — InsufficientStock พาจำนวนที่เหลือจริงกลับมา ทำให้หน้าจออัปเดตได้ทันทีตาม 6.12

sealed class ProductLookup {
  const ProductLookup();
}

class ProductFound extends ProductLookup {
  const ProductFound(this.product);
  final Product product;
}

/// ไม่เจอ — เป็นผลปกติของการสแกน
class ProductNotFound extends ProductLookup {
  const ProductNotFound(this.scannedCode);
  final String scannedCode;
}

class LookupFailed extends ProductLookup {
  const LookupFailed(this.kind, this.message);
  final ApiFailureKind kind;
  final String message;
}

Future<ProductLookup> findByBarcode(String code);

ProductNotFound พารหัสที่สแกนกลับมาด้วย เพื่อให้แสดงได้ว่า “ไม่พบสินค้ารหัส SKU-9931” แทนที่จะบอกลอย ๆ ตาม 5.10

ข้อ 3 — มีทั้งสองแบบในไฟล์เดียวยอมรับได้ไหม

ยอมรับได้ ถ้าเส้นแบ่งอธิบายได้ — และในกรณีนี้อธิบายได้ชัด

เมธอดความล้มเหลวเป็นอะไรแบบ
fetchOpenข้อยกเว้น — ปกติต้องได้ข้อมูลโยน
completeผลลัพธ์ที่คาดไว้ — มีหลายแบบคืนค่า
checkHealthคำตอบ — ไม่ใช่ข้อยกเว้นคืนค่า
findByBarcodeผลลัพธ์ที่เกิดทุกวันคืนค่า

เกณฑ์ที่ทำให้อธิบายได้

ถามว่า "ผลลัพธ์นี้เกิดขึ้นในการใช้งานปกติหรือไม่"
  เกิด    → คืนค่า
  ไม่เกิด  → โยน

แต่มีเงื่อนไขสองข้อที่ต้องมี ไม่งั้นจะกลายเป็นความสับสน

ต้องมีเพราะ
เขียนเกณฑ์ไว้ในเอกสารของทีมคนใหม่จะได้ตัดสินแบบเดียวกัน
ดูจากลายเซ็นแล้วรู้ทันทีFuture<CompleteOutcome> บอกอยู่แล้วว่าคืนค่า

ข้อสองสำคัญ — ลายเซ็นที่คืน sealed class บอกตัวเองอยู่แล้วว่าเป็นแบบคืนค่า ผู้เรียกไม่ต้องเดา ต่างจากการที่ทั้งสองแบบมีลายเซ็นหน้าตาเหมือนกัน

สิ่งที่ยอมรับไม่ได้คือเมธอดที่ คืนค่าแต่ก็ยังโยนด้วย ตามหัวข้อ 7 — เพราะลายเซ็นบอกว่าปลอดภัยแล้ว แต่จริง ๆ ไม่

บทเรียนที่ควรจำ: คำถามที่ตัดสินไม่ใช่ “แบบไหนดีกว่า” แต่คือ “ผลลัพธ์นี้เป็นเรื่องปกติของงานหรือเปล่า” — และการที่สแกนแล้วไม่เจอเป็นเรื่องปกติที่สุดในคลัง จึงไม่ควรเป็น exception

12. สรุปท้ายบท

  • โยนกับคืนค่าใช้ได้ทั้งคู่ ปัญหาอยู่ที่การผสมโดยไม่มีเกณฑ์
  • โยนเมื่อล้มเหลวแล้วจัดการรวมที่เดียวได้
  • คืนค่าเมื่อมีหลายผลลัพธ์ที่ต้องทำคนละอย่าง
  • เกณฑ์คือถามว่าผลลัพธ์นี้เกิดในการใช้งานปกติหรือไม่
  • ห้ามให้เมธอดเดียวทั้งคืนค่าและโยนสิ่งที่คาดไว้แล้ว
  • บั๊กของเราต้องโยนต่อเสมอ แม้จะเลือกแบบคืนค่า
  • ห้ามใช้ null แทนความล้มเหลว
  • ใช้ rethrow ไม่ใช่ throw error
  • ลายเซ็นที่คืน sealed class บอกตัวเองว่าเป็นแบบไหน
