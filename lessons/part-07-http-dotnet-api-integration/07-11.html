1. ปัญหาที่บทนี้แก้

7.9 บอกว่า “ให้อ่านเนื้อหาที่มากับความล้มเหลว” แต่ยังไม่ได้บอกว่าเนื้อหานั้นหน้าตาอย่างไร

ปัญหาคือถ้าแต่ละ endpoint กำหนดรูปแบบเอง แอปจะต้องรู้จักทุกแบบ

{"message": "..."}                    endpoint A
{"error": {"text": "..."}}            endpoint B
{"errors": ["...", "..."]}            endpoint C
{"Message": "...", "Code": 42}        endpoint D ที่เขียนไว้นานแล้ว

ASP.NET แก้เรื่องนี้ด้วยรูปแบบกลางที่มีมาให้อยู่แล้ว

{
  "type": "https://httpstatuses.io/409",
  "title": "Conflict",
  "status": 409,
  "detail": "งานนี้ถูกปิดโดยผู้ใช้อื่นแล้ว เมื่อ 14:32",
  "traceId": "00-4bf92f-a3ce929d-01"
}

ประโยชน์ที่ได้ทันทีคือ เขียนตัวอ่านครั้งเดียว ใช้ได้กับทุก endpoint ที่ทำตามรูปแบบนี้

2. คำศัพท์ที่ต้องรู้ก่อน

fieldบอกอะไร
typeที่อยู่ที่อธิบายข้อผิดพลาดประเภทนี้
titleชื่อประเภทข้อผิดพลาด — กว้าง
statusรหัสตอบกลับ ซ้ำกับที่อยู่ใน HTTP
detailรายละเอียดของคำขอครั้งนี้โดยเฉพาะ
errorsข้อผิดพลาดรายช่อง — มีเฉพาะกรณีตรวจข้อมูลไม่ผ่าน
traceIdรหัสตามหาคำขอนี้ใน log ฝั่งเซิร์ฟเวอร์

3. Mental model: title กว้าง detail เจาะจง

title:  "Conflict"                        ← ทุกคำขอที่ 409 ได้ข้อความนี้เหมือนกัน
detail: "งานนี้ถูกปิดโดยผู้ใช้อื่นแล้ว"      ← เฉพาะคำขอนี้
ประโยคเดียวที่ต้องจำ: detail คือสิ่งเดียวที่บอกเรื่องของคำขอครั้งนี้ — ส่วน title เป็นแค่ชื่อกลุ่ม ซึ่งรหัสตอบกลับบอกอยู่แล้ว

5. เลือกข้อความให้ผู้ใช้

/// ข้อความที่เอาไปแสดงให้ผู้ใช้ได้เลย
///
/// เรียงจากเจาะจงที่สุดไปกว้างที่สุด — detail บอกเรื่องของคำขอนี้
/// ส่วน title มักเป็นชื่อประเภทข้อผิดพลาดกว้าง ๆ
String? get userMessage {
  if (hasFieldErrors) {
    return errors.values.expand((messages) => messages).join('\n');
  }
  return detail ?? title;
}

ลำดับความสำคัญ

ลำดับใช้อะไรเพราะ
1ข้อผิดพลาดรายช่องบอกได้ว่าช่องไหนผิดและผิดอย่างไร
2detailบอกเรื่องของคำขอนี้
3titleดีกว่าไม่มีอะไรเลย
test('ข้อผิดพลาดรายช่องต้องถูกรวมมาแสดงได้', () {
  final problem = ProblemDetails.tryParse(const {
    'title': 'One or more validation errors occurred.',
    'status': 400,
    'errors': {
      'quantity': ['จำนวนต้องมากกว่าศูนย์'],
      'palletCode': ['รูปแบบรหัสพาเลทไม่ถูกต้อง'],
    },
  });

  // ข้อผิดพลาดรายช่องสำคัญกว่า title กว้าง ๆ

});

6. ข้อผิดพลาดรายช่องควรไปอยู่ที่ช่อง

การรวมข้อความทุกช่องมาแสดงรวมกันเป็นทางที่ง่ายที่สุด แต่ยังไม่ดีที่สุด

จำนวนต้องมากกว่าศูนย์
รูปแบบรหัสพาเลทไม่ถูกต้อง

ผู้ใช้ต้องเดาเองว่าข้อความไหนของช่องไหน — ทั้งที่เซิร์ฟเวอร์บอกมาแล้ว

final problem = error.problem;
if (problem != null && problem.hasFieldErrors) {
  setState(() {

TextFormField(
  decoration: InputDecoration(

  ),

แต่มีเรื่องที่ต้องระวัง — ชื่อช่องที่เซิร์ฟเวอร์ใช้อาจไม่ตรงกับชื่อในแอป

เซิร์ฟเวอร์ส่งมาเกิดจาก
quantityตั้งค่าให้เป็น camelCase
Quantityค่าตั้งต้นของ .NET รุ่นเก่า
$.quantityข้อผิดพลาดตอนแปลง JSON
request.Detail[0].Quantityช่องที่อยู่ลึกในโครง

จึงต้องแปลงชื่อก่อนจับคู่ ไม่ใช่เทียบตรง ๆ

/// จับคู่ชื่อช่องจาก server กับชื่อในแอป
///
/// server อาจส่ง Quantity, quantity หรือ request.Detail[0].Quantity
/// จึงเทียบเฉพาะส่วนท้ายแบบไม่สนตัวพิมพ์
String? errorFor(ProblemDetails problem, String field) {
  final target = field.toLowerCase();
  for (final entry in problem.errors.entries) {
    final last = entry.key.split('.').last.toLowerCase();
    if (last == target) return entry.value.firstOrNull;
  }
  return null;
}

8. อย่าแสดง type ให้ผู้ใช้

"type": "https://httpstatuses.io/409"

field นี้เป็นที่อยู่ของเอกสารที่อธิบายข้อผิดพลาด ซึ่งมีไว้ให้คนพัฒนาอ่าน ไม่ใช่คนหน้างาน

lab จึงไม่ได้อ่าน type เลย

fieldlab อ่านไหมเพราะ

statusอ่านใช้ตรวจว่าตรงกับรหัส HTTP ไหม
errorsอ่านชี้ช่องที่ผิดได้

10. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
แสดง “Conflict” ให้ผู้ใช้เห็นใช้ title ทั้งที่มี detailเรียงลำดับให้ detail มาก่อน
แสดง “One or more validation errors occurred.”ไม่ได้อ่าน errorsให้รายช่องมาก่อน title
ข้อผิดพลาดขึ้นรวมกันหมด ไม่รู้ว่าช่องไหนไม่ได้จับคู่กับช่องจับคู่ด้วยชื่อ field

เนื้อหาผิดรูปแบบแล้วพังซ้อนบังคับให้ต้องเป็น ProblemDetailsคืน null แล้วใช้ทางสำรอง
ตามหาใน log ไม่เจอไม่ได้แสดง traceIdแสดงเมื่อเป็นเรื่องที่ทีมต้องแก้
แสดงที่อยู่เอกสารให้ผู้ใช้เอา type ไปแสดงไม่ต้องอ่าน field นี้

ข้อ 2 — โค้ดจับคู่

/// ผลของการแยกคีย์ที่ server ส่งมา
typedef FieldPath = ({String field, int? index});

/// แยก "request.Details[1].LotNumber" เป็น (field: lotnumber, index: 1)
FieldPath parseErrorKey(String key) {
  if (key.isEmpty) return (field: '', index: null);

  final parts = key.split('.');
  final field = parts.last.toLowerCase();

  // หาหมายเลขแถวจากส่วนที่มีวงเล็บเหลี่ยม
  for (final part in parts) {
    final match = RegExp(r'\[(\d+)\]').firstMatch(part);
    if (match != null) {
      return (field: field, index: int.parse(match.group(1)!));
    }
  }
  return (field: field, index: null);
}
void applyServerErrors(ProblemDetails problem) {
  final formErrors = <String>[];
  final fieldErrors = <String, String>{};
  final lineErrors = <int, Map<String, String>>{};

  for (final entry in problem.errors.entries) {
    final path = parseErrorKey(entry.key);
    final message = entry.value.first;

    if (path.field.isEmpty) {
      formErrors.add(message);                       // ของทั้งฟอร์ม
    } else if (path.index case final index?) {
      (lineErrors[index] ??= {})[path.field] = message;   // ของแถวนั้น
    } else {
      fieldErrors[path.field] = message;             // ของช่องเดี่ยว
    }
  }

  setState(() {
    _formErrors = formErrors;
    _quantityError = fieldErrors['quantity'];
    _lineErrors = lineErrors;
  });
}
// ในแถวที่ index
TextFormField(
  decoration: InputDecoration(
    labelText: 'เลขล็อต',
    errorText: _lineErrors[index]?['lotnumber'],
  ),
)

แต่ในกรณีนี้มีเรื่องที่ต้องคิดเพิ่ม

ข้อความ “ใบรับเข้านี้ถูกปิดไปแล้ว” ไม่ใช่ข้อผิดพลาดของการกรอกข้อมูล — มันคือ ข้อมูลว่าสถานะเปลี่ยนไปแล้ว ซึ่งตามหลักใน 7.9 ควรมาเป็น 409 ไม่ใช่ 400

ผู้ใช้แก้แล้วส่งใหม่ได้ไหมได้ไม่ได้
ควรให้แก้ฟอร์มต่อไหมควรไม่ควร — ควรพากลับไปดูรายการ

การที่เซิร์ฟเวอร์ยัดเรื่องนี้มาใน 400 ทำให้แอปแสดงผลเหมือนเป็นเรื่องที่แก้ได้ ทั้งที่แก้ไม่ได้ — ผู้ใช้จะแก้จำนวนแล้วกดส่งใหม่ แล้วก็ได้ข้อความเดิม

สิ่งที่ควรทำคือแจ้งฝั่งหลังบ้านให้แยกออกมา และระหว่างนั้นฝั่งแอปรับมือด้วยการแสดงข้อความระดับฟอร์มให้เด่นพอที่ผู้ใช้จะเห็นก่อนแก้ช่องอื่น

บทเรียนที่ควรจำ: โครงของ errors บอกได้มากกว่าที่เห็น — คีย์ว่างคือระดับคำขอ คีย์ที่มีวงเล็บเหลี่ยมคือระดับแถว และเมื่อพบว่าข้อผิดพลาดระดับคำขอไม่ใช่เรื่องการกรอกข้อมูล นั่นเป็นสัญญาณว่ารหัสตอบกลับเลือกผิด

12. สรุปท้ายบท

  • ProblemDetails เป็นรูปแบบกลาง เขียนตัวอ่านครั้งเดียวใช้ได้ทุก endpoint
  • detail เจาะจง ส่วน title กว้าง — เรียงลำดับให้ถูก
  • ข้อผิดพลาดรายช่องสำคัญกว่าทั้งคู่
  • คืน null เมื่อรูปแบบไม่ตรง ไม่ใช่โยน
  • จับคู่ชื่อช่องด้วยส่วนท้ายแบบไม่สนตัวพิมพ์
  • คีย์ที่มีวงเล็บเหลี่ยมบอกหมายเลขแถว
  • คีย์ว่างคือข้อผิดพลาดระดับคำขอ
  • แสดง traceId เฉพาะเมื่อเป็นเรื่องที่ทีมต้องแก้
  • ตกลงกับหลังบ้านว่า detail ต้องเป็นภาษาที่ผู้ใช้อ่านได้
