Future<void> _clear() async {
  final stationLabel = _formatSortingStationId(widget.stationId);

7. ข้อผิดพลาดที่พบบ่อย

เข้าใจผิดที่จริง
ทุก widget ที่ฟังเหตุการณ์ต้องเรียก connect()ไม่ ถ้าพ่อต่อไว้แล้ว — แต่เรียกก็ปลอดภัยเพราะ ref-count
ผู้ฟังสองตัวรับเหตุการณ์เดียวกันแล้วชนกันทำงานอิสระ ต่างคนต่างกรอง
โหลดใหม่ทุกเหตุการณ์เป็นทางที่ปลอดภัยเสมอยิง 20 คำขอเมื่อกล่องมา 20 ใบ
ตัวหมุนที่หมุนอยู่แปลว่ากำลังโหลดอาจแปลว่าโหลดล้มเหลวแล้ว _detail ยังเป็น null
บริการตัวที่สองแก้ปัญหาของตัวแรกแล้วคัดลอกปัญหามาครบห้าข้อ

8. แบบฝึกหัด

ทีมอยากรวมสองบริการ SignalR เป็นตัวเดียว

  1. อะไรที่เหมือนกันและอะไรที่ต่าง
  2. ออกแบบบริการที่รองรับหลาย hub
  3. คุ้มไหม และควรทำเมื่อไร
แนวคำตอบ

ข้อ 1 — เหมือนและต่าง

ส่วนเหมือนไหม

ข้อ 2 — ออกแบบ

typedef HubCallback = void Function(Map<String, dynamic> data);

/// การเชื่อมต่อ SignalR หนึ่ง hub — จัดการ ref-count และการกระจายเหตุการณ์ให้
class HubClient {
  HubClient({required this.path, required this.eventNames});

  final String path;                    // '/hubs/putaway'
  final List<String> eventNames;        // ['StationDispatched', ...]

  HubConnection? _connection;
  int _refCount = 0;
  final _listeners = <String, List<HubCallback>>{};

  final status = ValueNotifier<RealtimeStatus>(RealtimeStatus.unknown);

  Future<void> connect() async {
    _refCount++;
    if (_connection != null &&
        _connection!.state != HubConnectionState.Disconnected) return;

    status.value = RealtimeStatus.connecting;
    try {
      final serverUrl = await ApiService.resolveServerUrl();
      _connection = HubConnectionBuilder()
          .withUrl('$serverUrl$path')
          .withAutomaticReconnect()
          .build();

      for (final name in eventNames) {
        _connection!.on(name, (args) => _dispatch(name, args));
      }

      // แอปไม่รู้ว่าพลาดอะไรไประหว่างหลุด — ให้ทุกจอโหลดใหม่ (18.8)
      _connection!.onreconnected(({connectionId}) {
        status.value = RealtimeStatus.connected;
        _dispatch(_reconnectedEvent, null);
      });
      _connection!.onreconnecting(({error}) => status.value = RealtimeStatus.connecting);
      _connection!.onclose(({error}) => status.value = RealtimeStatus.disconnected);

      await _connection!.start();
      status.value = RealtimeStatus.connected;
    } catch (e) {
      status.value = RealtimeStatus.disconnected;
      dev.log('[Hub $path] connect error: $e');
    }
  }

  Future<void> disconnect() async {
    _refCount--;
    if (_refCount > 0) return;
    _refCount = 0;

    final connection = _connection;
    _connection = null;
    status.value = RealtimeStatus.disconnected;
    if (connection == null) return;
    try {
      await connection.stop();
    } catch (e) {
      dev.log('[Hub $path] stop error: $e');   // ตัดไม่ได้ก็ไม่ควรทำให้แอปพัง
    }
  }

  void on(String event, HubCallback cb) =>
      _listeners.putIfAbsent(event, () => []).add(cb);

  void off(String event, HubCallback cb) => _listeners[event]?.remove(cb);

  static const _reconnectedEvent = '__reconnected';
  void onReconnected(VoidCallback cb) => on(_reconnectedEvent, (_) => cb());

  void _dispatch(String event, List<Object?>? args) {
    final data = _parseArgs(args);
    for (final cb in List.of(_listeners[event] ?? const [])) {
      cb(data);
    }
  }

}

9. สรุป

  • บริการตัวที่สองมีเหตุการณ์มากกว่าแต่โค้ดสั้นกว่า เพราะมีตัวช่วย _on
  • ห้าปัญหาที่ 18.8 ชี้ ถูกคัดลอกมาครบ
  • ผลกระทบหนักกว่า เพราะจอ sorting ไม่มีตัวจับเวลาสำรอง
  • แผ่นรายละเอียดฟังเหตุการณ์แต่ไม่เรียก connect() — ถูกตราบใดที่เปิดจากจอหลักเท่านั้น
  • เพิ่ม connect()/disconnect() ในแผ่นจะปลอดภัยกว่าและ ref-count จัดการให้
  • ผู้ฟังเจ็ดตัวลงทะเบียนพร้อมกันเมื่อแผ่นเปิด และต่างคนต่างกรอง
  • แผ่นยิงคำขอทุกกล่องที่มาถึง — ควรหน่วงถ้ากล่องมาถี่
  • โหลดล้มเหลวแล้วตัวหมุนหมุนต่อไปเรื่อย ๆ โดยไม่บอกอะไร
  • onChanged แจ้งจอหลักให้โหลดใหม่เฉพาะเมื่อมีการเปลี่ยนจริง
  • การรวมสองบริการคุ้ม และควรทำก่อนแก้ห้าปัญหา ไม่ใช่หลัง
