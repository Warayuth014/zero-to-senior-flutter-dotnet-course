เมธอดนี้มีอยู่เพราะฟิลด์ซ้ำ — ถ้าแยกออกเป็นคลาสเดียวจะไม่ต้องคัดลอก

class OrderPipeline {
  const OrderPipeline({
    this.total = 0,
    this.pickDone = 0,
    this.packDone = 0,
    this.sortingDone = 0,
    this.checkInDone = 0,
  });

  final int total;
  final int pickDone;
  final int packDone;
  final int sortingDone;
  final int checkInDone;

  factory OrderPipeline.fromJson(Map<String, dynamic> json) => OrderPipeline(
    total: json['pipelineTotal'] ?? 0,
    pickDone: json['pickDone'] ?? 0,
    packDone: json['packDone'] ?? 0,
    sortingDone: json['sortingDone'] ?? 0,
    checkInDone: json['checkInDone'] ?? 0,
  );

  /// ขั้นที่ยังทำไม่ครบขั้นแรก — ใช้บอกผู้ใช้ว่างานค้างตรงไหน
  String? get blockedAt {
    if (total == 0) return null;
    if (pickDone < total) return 'Picking';
    if (packDone < total) return 'Packing';
    if (sortingDone < total) return 'Sorting';
    if (checkInDone < total) return 'Check-in';
    return null;
  }
}

ได้ getter ที่ตอบคำถามที่ผู้ใช้อยากรู้จริง — “ทำไมใบสั่งนี้ยังไม่ครบ”

ตอนนี้จอต้องคำนวณเองในส่วนที่วาด

5. ฝั่ง API ที่สม่ำเสมอ

Future<ApiResult<void>> completeCheckIn({
  required String slotId,
  required String operatorId,
}) async {
  final response = await _post('/checkin/complete', {
    'slotId': slotId,
    'operatorId': operatorId,
  });
  if (!response.success) {
    return ApiResult.error(response.error, statusCode: response.statusCode);
  }
  return ApiResult.success(null);
}
ทำถูกทำไม
ทั้งห้าเมธอดส่ง statusCode ต่อจอแยก 404 จาก 500 ได้
คืน ApiResult<void> เมื่อไม่มีข้อมูลกลับตรงกับความจริง ไม่บังคับให้คืน Map เปล่า
มีคอมเมนต์อธิบายทุกเมธอดบอกสิ่งที่ลายเซ็นไม่บอก
/// Preview Pack ก่อน Check-IN (ไม่เขียน DB)
/// สแกน Carton (PackingId) → assign Slot ตาม Owner
/// ดึงรายละเอียด Slot (cartons ทั้งหมด + progress)
/// Complete Slot → status=READY
/// Dispatch Slot → ย้ายขึ้นรถ SHIPPED

คอมเมนต์บรรทัดที่สองบอกกฎที่สำคัญ — ช่องถูกกำหนดตามเจ้าของสินค้า

ผู้ใช้จึงไม่ได้เลือกช่องเอง และนี่คือเหตุผลที่ต้องมี popup แจ้งปลายทาง (19.7)

ส่วน path ทำถูกกว่าโดเมนอื่น

// สี่ในห้าเมธอดใช้ POST — ไม่มีค่าใน path
await _post('/checkin/preview', {'packingId': packingId});
await _post('/checkin/scan', {...});
await _post('/checkin/complete', {...});
await _post('/checkin/dispatch', {...});

// เมธอดเดียวที่มีค่าใน path
await _get('/checkin/slot/$slotId');
การส่งรหัสใน body แทน path เลี่ยงปัญหาการเข้ารหัสไปเลย — 17.9 และ 18.7 พบว่าการต่อสตริงใน path พังถ้ารหัสมีอักขระพิเศษ
ที่นี่มีจุดเดียวที่ต้องระวัง และ slotId มาจากเซิร์ฟเวอร์ ไม่ได้มาจากบาร์โค้ด จึงเสี่ยงน้อย

เข้าใจผิดที่จริง

ข้อ 2 — เทสต์

void main() {
  group('OrderPipeline.blockedAt', () {
    test('ยังไม่มีข้อมูล → null', () {
      expect(const OrderPipeline().blockedAt, isNull);
    });

    test('ค้างที่ picking', () {
      const p = OrderPipeline(total: 5, pickDone: 3);
      expect(p.blockedAt, 'Picking');
    });

    test('หยิบครบแล้ว ค้างที่ packing', () {
      const p = OrderPipeline(total: 5, pickDone: 5, packDone: 2);
      expect(p.blockedAt, 'Packing');
    });

    test('ครบทุกขั้น → null', () {
      const p = OrderPipeline(
        total: 5, pickDone: 5, packDone: 5, sortingDone: 5, checkInDone: 5,
      );
      expect(p.blockedAt, isNull);
    });

    test('ตัวเลขเกิน total ก็ยังถือว่าครบ', () {
      // ข้อมูลไม่สอดคล้อง แต่ต้องไม่ทำให้แสดงผลผิด
      const p = OrderPipeline(
        total: 5, pickDone: 6, packDone: 5, sortingDone: 5, checkInDone: 5,
      );
      expect(p.blockedAt, isNull);
    });

    test('อ่านจาก JSON ที่ไม่มีคีย์เลย → ทุกค่าเป็น 0', () {
      final p = OrderPipeline.fromJson({});
      expect(p.total, 0);
      expect(p.blockedAt, isNull);
    });
  });
}

เทสต์ที่สองจากท้ายสำคัญ — ยืนยันว่าใช้ < ไม่ใช่ !=

if (pickDone < total) return 'Picking';      // ทน
if (pickDone != total) return 'Picking';     // พังถ้าเกิน

เป็นบทเรียนเดียวกับ isDone => qtyRemaining <= 0 ใน 17.13 และ clamp ใน 19.6

ข้อ 3 — ได้อะไรเพิ่ม

ได้รายละเอียด
ฟังก์ชันบริสุทธิ์ที่เทสต์ได้หกเทสต์รันในไม่กี่มิลลิวินาที
blockedAt ที่ตอบคำถามจริงของผู้ใช้“ใบสั่งนี้ค้างตรงไหน”
คัดลอกทีละฟิลด์หายไปเพิ่มขั้นใหม่ไม่ต้องแก้สองที่
เห็นว่าห้าตัวเลขเป็นของชิ้นเดียวกันตอนนี้ดูเหมือนฟิลด์อิสระสิบตัว

แถวสุดท้ายสำคัญที่สุด — และนำไปสู่ความสามารถใหม่

// ตอนนี้จอวาด pipeline เองด้วยการอ่านห้าฟิลด์
Widget _buildPipelineCard(CheckInSlotDetail slot) { ... }

// หลังแยก — widget รับ OrderPipeline ตัวเดียว
PipelineCard(pipeline: slot.pipeline)

แล้ว widget นี้ใช้ได้กับทั้งสองแหล่งโดยไม่ต้องรู้ว่ามาจากไหน

และเปิดทางให้จออื่นใช้ด้วย — 19.9 ชี้ว่า check-in เป็นจุดเดียวที่เห็นภาพรวมของงาน
ถ้าฝั่ง .NET ส่ง pipeline มากับ endpoint ของ packing หรือ picking ด้วย จอเหล่านั้นจะแสดงได้ทันทีโดยใช้ widget เดียวกัน
เป็นตัวอย่างว่าการจัดกลุ่มข้อมูลให้ถูกทำให้ความสามารถใหม่ถูกกว่า ไม่ใช่แค่โค้ดสวยขึ้น

9. สรุป

  • ห้าคลาส และ ScanCheckInResponse ที่มีสองฟิลด์เป็นตัวตัดสินว่าของขึ้นรถไหม
  • ห้าตัวเลข pipeline ปรากฏซ้ำในสองคลาส และต้องคัดลอกทีละตัว
  • pipeline แสดงความคืบหน้าข้ามสี่โดเมน — จุดเดียวในแอปที่เห็นภาพรวม
  • ระดับการป้องกันสูงเท่า packing ยกเว้น cartons ที่ไม่กัน null
  • สองคลาสในไฟล์เดียวกันเขียน as List คนละแบบ
  • packStatus โหลดมาแล้วไม่มีใครอ่าน
  • ทั้งห้าเมธอดส่ง statusCode ต่อ และมีคอมเมนต์ครบ
  • สี่ในห้าเมธอดใช้ POST ส่งรหัสใน body — เลี่ยงปัญหาการเข้ารหัส path
  • สามโดเมนของ Part นี้เขียนตามข้อตกลงเดียวกันเกือบทั้งหมด
