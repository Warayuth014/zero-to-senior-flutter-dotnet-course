2. copyWith ที่รับเฉพาะที่เปลี่ยนได้

SortingStationView copyWith({
  String? status,
  String? palletId,
  int? cartonsCount,
  int? maxCapacity,
  bool? isFull,
  bool? enabled,
}) =>
    SortingStationView(
      stationId: stationId,           // ← ไม่ให้เปลี่ยน
      enabled: enabled ?? this.enabled,
      status: status ?? this.status,
      palletId: palletId ?? this.palletId,
      cartonsCount: cartonsCount ?? this.cartonsCount,
      maxCapacity: maxCapacity ?? this.maxCapacity,
      isFull: isFull ?? this.isFull,
      disableReason: disableReason,   // ← ไม่ให้เปลี่ยน
    );
ฟิลด์เปลี่ยนได้ไหมเหตุผล
stationIdไม่เป็นตัวระบุ เปลี่ยนแล้วกลายเป็นสถานีอื่น
disableReasonไม่เปลี่ยนพร้อม enabled ผ่านการโหลดใหม่
อีกหกตัวได้เปลี่ยนจากเหตุการณ์เรียลไทม์

ใช้ที่เดียวในทั้งโปรเจกต์

void _onCounterUpdated(Map<String, dynamic> data) {
  final stationId = data['stationId'] as int?;
  if (stationId == null || !mounted) return;

  setState(() {
    final idx = _stations.indexWhere((s) => s.stationId == stationId);
    if (idx < 0) return;
    _stations[idx] = _stations[idx].copyWith(
      cartonsCount: data['current'] as int?,
      maxCapacity: data['total'] as int?,
      isFull: data['isFull'] as bool?,
      palletId: data['palletId'] as String?,
      status: 'BUSY',
    );
  });
}
นี่คือสิ่งที่ 18.7 เสนอให้ putaway ทำ — ที่นั่นประกอบ Map ใหม่ด้วยมือและต้องจำเก็บ items ไว้เอง
ที่นี่ copyWith เก็บทุกฟิลด์ที่ไม่ได้ส่งมาให้อัตโนมัติ — ลืมไม่ได้

3. กับดักของ copyWith กับฟิลด์ nullable

palletId: palletId ?? this.palletId,

รูปแบบ ?? this.x แปลว่า “ถ้าไม่ส่งมา ใช้ค่าเดิม”

แต่ palletId เป็น String? — จึงแยกไม่ออกระหว่างสองความหมาย

ผู้เรียกส่งตั้งใจว่าได้ผล
ไม่ส่งไม่เปลี่ยนค่าเดิม ✓
palletId: 'PAL-7'เปลี่ยนเป็นค่านี้ค่าใหม่ ✓
palletId: nullล้างค่าค่าเดิม ✗

แถวสุดท้ายคือกับดัก — ล้างค่าให้เป็น null ไม่ได้

// เหตุการณ์ StationCleared ควรล้าง palletId
_stations[idx] = _stations[idx].copyWith(
  status: 'AVAILABLE',
  palletId: null,        // ← ไม่มีผล พาเลทเดิมยังค้าง
);
โค้ดปัจจุบันเลี่ยงปัญหานี้โดยไม่รู้ตัว — เหตุการณ์ที่ทำให้สถานีว่าง (StationCleared) ไม่ได้ใช้ copyWith แต่เรียก _loadStations() โหลดใหม่ทั้งชุด
void _onAnyStationEvent(Map<String, dynamic> _) {
  // เปลี่ยนสถานะ + อาจมี station ใหม่ assign → reload ทั้งกริด
  _loadStations();
}
คอมเมนต์อธิบายเหตุผลไว้ และเป็นการตัดสินใจที่ถูก — แต่ถ้าวันหนึ่งมีคนอยากประหยัดคำขอแล้วเปลี่ยนมาใช้ copyWith จะเจอบั๊กนี้ทันที

วิธีแก้ที่ใช้กันคือใช้ค่ากลาง

/// ตัวคั่นสำหรับ copyWith — บอกความต่างระหว่าง "ไม่ส่ง" กับ "ส่ง null"
const _unset = Object();

SortingStationView copyWith({
  Object? palletId = _unset,
  ...
}) =>
    SortingStationView(
      stationId: stationId,
      palletId: identical(palletId, _unset) ? this.palletId : palletId as String?,
      ...
    );

อ่านยากขึ้นมาก — และในกรณีนี้ไม่คุ้ม เพราะการโหลดใหม่ทั้งชุดทำงานได้ดีอยู่แล้ว

สิ่งที่ควรทำแทนคือเขียนคอมเมนต์บอกข้อจำกัด

/// สร้างสำเนาที่แก้เฉพาะฟิลด์ที่ส่งมา
///
/// ข้อจำกัด: ส่ง null เพื่อ "ล้างค่า" ไม่ได้ — จะถือว่าไม่เปลี่ยน
/// ถ้าต้องล้าง (เช่นสถานีว่าง) ให้โหลดใหม่ด้วย _loadStations()
SortingStationView copyWith({...})

4. เหตุการณ์ที่อัปเดตทันทีกับที่โหลดใหม่

@override
void initState() {
  super.initState();
  _hub.onCounterUpdated(_onCounterUpdated);      // ← อัปเดตทันที
  _hub.onStationFull(_onAnyStationEvent);        // ← โหลดใหม่
  _hub.onStationCleared(_onAnyStationEvent);     // ← โหลดใหม่
  _hub.onStationToggled(_onAnyStationEvent);     // ← โหลดใหม่
  _hub.onBatchAssigned(_onAnyStationEvent);      // ← โหลดใหม่
  _initSignalR();
  _loadStations();
}
เหตุการณ์ทำอะไรทำไม
StationCounterUpdatedcopyWith ทันทีเกิดบ่อย และเปลี่ยนแค่ตัวเลข
อีกสี่ตัวโหลดใหม่ทั้งชุดเปลี่ยนโครงสร้าง อาจมีสถานีอื่นเปลี่ยนตาม

การแบ่งนี้สมเหตุสมผล

// ตัวนับเพิ่มขึ้นทีละกล่อง — อาจเกิดหลายครั้งต่อวินาที
// ถ้าโหลดใหม่ทุกครั้งจะยิงคำขอถี่มาก
_stations[idx] = _stations[idx].copyWith(cartonsCount: ..., isFull: ...);
// batch ถูก assign — สถานีที่ว่างอาจกลายเป็น BUSY
// และคิวที่รออาจลดลง → ตัวเลขสรุปเปลี่ยนด้วย
_loadStations();
เทียบกับ putaway (18.4) ที่ทำทั้งสองอย่างพร้อมกัน — อัปเดตจากเหตุการณ์แล้วโหลดใหม่ตามมาทันที
// putaway
setState(() { _stationStatus = {...}; });    // อัปเดตทันที
_loadStationStatus();                         // แล้วโหลดใหม่
ที่นั่นได้ทั้งการตอบสนองทันทีและข้อมูลครบ แต่ยิงคำขอทุกเหตุการณ์
ที่นี่เลือกอย่างใดอย่างหนึ่งตามชนิดเหตุการณ์ — ประหยัดกว่าแต่ตัวนับที่อัปเดตทันทีอาจไม่ตรงกับเซิร์ฟเวอร์ถ้าพลาดเหตุการณ์ไป

และมีการป้องกันเมื่อไม่พบสถานี

final idx = _stations.indexWhere((s) => s.stationId == stationId);
if (idx < 0) return;      // ← ยังไม่โหลด หรือสถานีถูกลบ

เกิดได้ถ้าเหตุการณ์มาถึงก่อน _loadStations() แรกจะเสร็จ

ผลเหมาะไหม
ทิ้งเหตุการณ์นั้นไปใช่ — การโหลดที่กำลังทำอยู่จะได้ข้อมูลล่าสุดอยู่แล้ว

5. สองคลาสที่ต่างกันแค่ nullable

class SortingStationView {
  final int? cartonsCount;      // ← nullable
  final int? maxCapacity;
  final bool? isFull;
}

class SortingStationDetail {
  final int cartonsCount;       // ← ไม่ nullable
  final int maxCapacity;
  final bool isFull;
}
คลาสทำไมต่างกัน
Viewสถานีที่ว่างไม่มีตัวเลขเหล่านี้
Detailเปิดดูรายละเอียดได้เฉพาะสถานีที่ BUSY

สอดคล้องกับ _openStation ที่เปิดแผ่นรายละเอียดเฉพาะเมื่อ BUSY (19.10)

// View — ต้องตรวจก่อนใช้
final isFull = s.isFull == true;

// Detail — ใช้ได้เลย
if (_detail!.isFull) { ... }
การให้ชนิดสะท้อนความจริงแบบนี้ถูกต้อง — ไม่ใช่การเขียนไม่สม่ำเสมอ
ต่างจากกรณีที่ 17.13 พบว่า UnloadItem กับ UnloadedItem ต่างกันแค่ชื่อจนสับสน
ที่นี่ชื่อ View กับ Detail บอกชัดว่าใช้คนละที่

ค่าเริ่มต้นก็ต่างกันตาม

// View — ไม่มีค่าเริ่มต้น ปล่อยเป็น null
cartonsCount: json['cartonsCount'],

// Detail — มีค่าเริ่มต้น เพราะต้องมีค่าเสมอ
cartonsCount: json['cartonsCount'] ?? 0,

7. ข้อผิดพลาดที่พบบ่อย

เข้าใจผิดที่จริง
copyWith ล้างค่าเป็น null ได้ไม่ได้ ถ้าเขียนแบบ ?? this.x
เหตุการณ์เรียลไทม์ทุกตัวควรจัดการเหมือนกันแบ่งตามว่าเปลี่ยนตัวเลขหรือเปลี่ยนโครงสร้าง
สองคลาสที่ฟิลด์ต่างกันเป็นความไม่สม่ำเสมอที่นี่สะท้อนความจริงว่าใช้คนละบริบท
ฟิลด์ nullable สองตัวเป็นอิสระoutcome กับ stationId ผูกกัน
เครื่องมือทดสอบต้องปิดที่แอปปิดที่ endpoint ได้ผลทุกเครื่อง

8. แบบฝึกหัด

ทีมอยากลดคำขอโดยให้ StationCleared ใช้ copyWith แทนการโหลดใหม่

  1. จะเกิดอะไรขึ้นถ้าทำตรง ๆ
  2. แก้อย่างไร
  3. คุ้มไหมที่จะทำ
แนวคำตอบ

ข้อ 1 — ทำตรง ๆ แล้วเกิดอะไร

void _onStationCleared(Map<String, dynamic> data) {
  final stationId = data['stationId'] as int?;
  if (stationId == null || !mounted) return;

  setState(() {
    final idx = _stations.indexWhere((s) => s.stationId == stationId);
    if (idx < 0) return;
    _stations[idx] = _stations[idx].copyWith(
      status: 'AVAILABLE',
      palletId: null,          // ← ตั้งใจล้าง
      cartonsCount: null,
      isFull: false,
    );
  });
}
ฟิลด์ตั้งใจได้จริง
statusAVAILABLEAVAILABLE ✓
palletIdnullค่าเดิม ✗
cartonsCountnullค่าเดิม ✗
isFullfalsefalse ✓

ผู้ใช้จะเห็นการ์ดที่บอกว่า “ว่าง” แต่ยังแสดงรหัสพาเลทและจำนวนกล่องของงานที่เพิ่งเคลียร์ไป

และการ์ดตัดสินใจจาก status

// _StationCard
final isBusy = s.status == 'BUSY';
...
isBusy ? _buildBusyRow(...) : _buildAvailableRow(stationLabel)

โชคดีที่แถวว่างไม่แสดง palletId — ค่าที่ค้างจึงไม่ปรากฏบนจอ

แต่จะปรากฏทันทีที่สถานีกลับเป็น BUSY จากเหตุการณ์ถัดไป ก่อนที่ตัวนับจริงจะมาถึง

t=0  StationCleared  → status=AVAILABLE, palletId ค้าง 'PAL-3'
t=5  BatchAssigned   → _loadStations() → ได้ข้อมูลจริง ✓

ถ้า BatchAssigned ยังใช้ _loadStations() อยู่ ปัญหาจะถูกกลบ — แต่ถ้าเปลี่ยนตัวนั้นด้วยจะเห็นชัด

ข้อ 2 — แก้อย่างไร

ทางที่ตรงที่สุดคือเพิ่มเมธอดเฉพาะกิจ แทนที่จะทำ copyWith ให้ซับซ้อน

/// สถานีถูกเคลียร์ — ล้างข้อมูลงานทั้งหมด เหลือแต่ตัวระบุและการตั้งค่า
SortingStationView cleared() => SortingStationView(
      stationId: stationId,
      enabled: enabled,
      status: 'AVAILABLE',
      palletId: null,
      cartonsCount: null,
      maxCapacity: maxCapacity,      // ความจุของสถานีไม่เปลี่ยน
      isFull: false,
      disableReason: disableReason,
    );
_stations[idx] = _stations[idx].cleared();
ข้อดีเทียบกับ copyWith ที่ใช้ _unset
อ่านแล้วรู้ว่าทำอะไรcopyWith(palletId: null, ...) อ่านไม่ออกว่าล้างหรือไม่เปลี่ยน
บังคับให้คิดครบทุกฟิลด์ลืมฟิลด์หนึ่งได้
ไม่ต้องมี Object? กับ identicalโค้ดอ่านง่ายกว่ามาก
หลักที่ใช้ได้ทั่วไป: copyWith เหมาะกับ “แก้ทีละฟิลด์โดยไม่รู้ว่าจะแก้ตัวไหน”
ส่วนการเปลี่ยนสถานะที่มีความหมายชัด (เคลียร์, ปิดใช้งาน, รับงาน) ควรเป็นเมธอดที่ตั้งชื่อตามความหมาย
เมธอดแบบหลังยังเทสต์ได้ตรงกว่าด้วย
test('cleared() ล้างข้อมูลงานแต่เก็บความจุไว้', () {
  final busy = SortingStationView(
    stationId: 3, enabled: true, status: 'BUSY',
    palletId: 'PAL-3', cartonsCount: 12, maxCapacity: 20, isFull: false,
  );
  final after = busy.cleared();

  expect(after.status, 'AVAILABLE');
  expect(after.palletId, isNull);
  expect(after.cartonsCount, isNull);
  expect(after.maxCapacity, 20);      // ← ความจุไม่หาย
  expect(after.stationId, 3);
});

ข้อ 3 — คุ้มไหม

คำถามคำตอบ
StationCleared เกิดบ่อยแค่ไหนเมื่อผู้ใช้เคลียร์สถานี — ไม่กี่ครั้งต่อชั่วโมง
ประหยัดคำขอได้เท่าไรไม่กี่ครั้งต่อชั่วโมงต่อเครื่อง

9. สรุป

  • หกคลาส และสองคลาสท้ายใช้เฉพาะจอทดสอบ
  • _parseDate เป็นตัวช่วยระดับไฟล์ที่ทนกว่าวิธีใน picking และ receiving
  • copyWith ไม่ให้เปลี่ยน stationId และ disableReason
  • รูปแบบ ?? this.x ทำให้ล้างค่าเป็น null ไม่ได้
  • โค้ดเลี่ยงกับดักนั้นโดยให้เหตุการณ์ที่ต้องล้างค่าโหลดใหม่ทั้งชุด
  • แบ่งเหตุการณ์สองแบบ: อัปเดตทันทีเมื่อเปลี่ยนตัวเลข โหลดใหม่เมื่อเปลี่ยนโครงสร้าง
  • View กับ Detail ต่างกันที่ nullable เพราะใช้คนละบริบท — ถูกต้อง
  • outcome กับ stationId ผูกกันแต่ชนิดไม่บังคับ
  • endpoint ทดสอบมีคำว่า test ใน path — ปิดที่เซิร์ฟเวอร์ได้
  • การเปลี่ยนสถานะที่มีความหมายชัด ควรเป็นเมธอดที่ตั้งชื่อ ไม่ใช่ copyWith
