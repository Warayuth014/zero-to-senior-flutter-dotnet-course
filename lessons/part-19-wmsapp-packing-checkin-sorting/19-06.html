3. getter ที่คำนวณจากสองแหล่ง

class PackingSummary {
  final String status;
  final int orderCount;
  final int orderDoneCount;
  ...
  bool get isDone =>
      status == 'DONE' || (orderCount > 0 && orderDoneCount >= orderCount);
}

ตรวจสองอย่างด้วย || — สถานะจากเซิร์ฟเวอร์ หรือคำนวณจากตัวนับ

กรณีstatusorderDoneCount / orderCountisDone
ปกติDONE3/3true
ตัวนับครบแต่สถานะยังไม่อัปเดตPACKING3/3true
สถานะ DONE แต่ตัวนับไม่ครบDONE2/3true
orderCount เป็น 0PACKING0/0false

แถวสุดท้ายคือเหตุผลที่ต้องมี orderCount > 0 — ถ้าไม่มี กล่องที่ยังไม่มีใบสั่งเลยจะถูกนับว่าเสร็จ (เพราะ 0 >= 0)

การใช้ || ตรงนี้เป็นการเผื่อว่าเซิร์ฟเวอร์อาจส่งสถานะมาไม่ทัน
แต่แลกมาด้วยการที่สองแหล่งขัดกันแล้วไม่มีใครรู้ — แถวที่สองและสามจะได้ true ทั้งคู่
ถ้าอยากรู้ว่าข้อมูลไม่สอดคล้อง ต้องแยกออกมา
bool get isDone => status == 'DONE' || _allOrdersDone;
bool get _allOrdersDone => orderCount > 0 && orderDoneCount >= orderCount;

/// สถานะกับตัวนับไม่ตรงกัน = ข้อมูลไม่สอดคล้อง ควรแจ้งทีมหลังบ้าน
bool get hasInconsistentState => (status == 'DONE') != _allOrdersDone;

สังเกตความไม่สม่ำเสมอในไฟล์เดียวกัน

class PackingSummary       { bool get isDone => status == 'DONE' || (...); }  // สองแหล่ง
class PackingDetailResponse{ bool get isDone => status == 'DONE'; }            // แหล่งเดียว
class PackingOrderSummary  { bool get isDone => status == 'DONE'; }            // แหล่งเดียว
class PackingOrderResponse { bool get isDone => status == 'DONE'; }            // แหล่งเดียว
class PackingPartItem      { bool get isDone => scannedQty >= requiredQty; }   // คำนวณล้วน
คลาสisDone มาจาก

PackingSummaryสถานะ หรือ ตัวนับ
PackingPartItemตัวนับล้วน

ชื่อเดียวกันแต่คำนวณคนละแบบสามแบบ — คนอ่านต้องเปิดดูทุกครั้ง

clamp ทำให้สามที่นี้ทำงานถูกโดยไม่ต้องตรวจเอง — เป็นการป้องกันที่จุดเดียวแทนที่จะกระจาย

factory PackingPartItem.fromJson(Map<String, dynamic> json) => PackingPartItem(

ทั้งสองแบบมีข้อดี — toString() ทนกว่า แต่ซ่อนปัญหา; as String พังทันทีแต่บอกว่าข้อมูลผิด
ในระบบที่ S/N เป็นสตริงเสมอตามสัญญา แบบหลังจับปัญหาได้เร็วกว่า
แต่ในระบบที่ผู้ใช้ทำงานอยู่หน้างาน การพังทั้งจอเพราะ S/N ตัวเดียวผิดชนิดไม่คุ้ม

6. ฝั่ง API ที่ทำครบ

Future<ApiResult<PackingPalletResponse>> scanPalletForPacking(String palletId) async {
  final response = await _get('/packing/pallet/$palletId');
  if (!response.success) {
    return ApiResult.error(response.error, statusCode: response.statusCode);
  }
  return ApiResult.success(PackingPalletResponse.fromJson(response.data!));
}

ทั้งสี่เมธอดส่ง statusCode ต่อ — ต่างจาก receiving, unload, upload, putaway ที่ทิ้งทั้งหมด

และมีคอมเมนต์อธิบายทุกเมธอด

/// Scan Pallet → ดึง list ของ Pack ที่ต้อง pack บน Pallet นี้

7. ข้อผิดพลาดที่พบบ่อย

เข้าใจผิดที่จริง
คลาสที่มี fromJson ต้องถูกสร้างจาก JSONConfirmPackResponse ถูกสร้างด้วยมือ
isDone ในไฟล์เดียวกันคำนวณเหมือนกันสามแบบในห้าคลาส
clamp เป็นการซ่อนปัญหาแก้ค่าให้ถูก แต่ควรมีทางรู้ว่าข้อมูลผิด
toString() กับ as String เหมือนกันตัวแรกทน ตัวหลังจับปัญหาได้
ส่ง [] กับไม่ส่งฟิลด์เหมือนกันขึ้นกับฝั่งเซิร์ฟเวอร์

8. แบบฝึกหัด

เขียนเทสต์ให้โมเดลของโดเมนนี้ โดยเน้นสิ่งที่บทนี้ชี้ไว้

  1. เทสต์อะไรได้บ้าง และแต่ละข้อจับปัญหาอะไร
  2. เขียนเทสต์ที่จับความไม่สม่ำเสมอของ isDone
  3. ถ้าเทสต์ผ่านหมด แปลว่าโมเดลนี้ถูกไหม
แนวคำตอบ

ข้อ 1

เทสต์จับอะไร
remaining ไม่ติดลบเมื่อสแกนเกินยืนยัน clamp
isDone ของ PackingSummary ตรวจสองแหล่งบันทึกพฤติกรรมที่ต่างจากคลาสอื่น
orderCount == 0 ไม่ถือว่าเสร็จยืนยันเงื่อนไข > 0
ฟิลด์คำอธิบายหายแล้วไม่พังยืนยันการป้องกัน
availableSerials ที่มีตัวเลขปนยืนยัน toString()
packFinalized ที่ไม่ส่งมา ได้ falseสัญญาที่ 19.5 พึ่งพา

ข้อ 2 — เทสต์ที่จับความไม่สม่ำเสมอ

void main() {
  group('isDone คำนวณคนละแบบในแต่ละคลาส', () {
    test('PackingPartItem — จากตัวนับล้วน ไม่สนสถานะ', () {
      final part = PackingPartItem.fromJson({
        'partId': 'PT-1',
        'requiredQty': 10,
        'scannedQty': 10,
      });
      expect(part.isDone, isTrue);
      expect(part.remaining, 0);
    });

    test('PackingPartItem — สแกนเกิน remaining ต้องเป็น 0 ไม่ใช่ติดลบ', () {
      final part = PackingPartItem.fromJson({
        'partId': 'PT-1',
        'requiredQty': 10,
        'scannedQty': 12,
      });
      expect(part.remaining, 0);
      expect(part.isDone, isTrue);
    });

    test('PackingSummary — ตัวนับครบก็ถือว่าเสร็จ แม้สถานะยังไม่ DONE', () {
      // พฤติกรรมนี้ต่างจากคลาสอื่นในไฟล์เดียวกันโดยตั้งใจ
      final summary = PackingSummary.fromJson({
        'packingId': 'PACK-1',
        'status': 'PACKING',
        'orderCount': 3,
        'orderDoneCount': 3,
      });
      expect(summary.isDone, isTrue);
    });

    test('PackingSummary — ยังไม่มี Order ต้องไม่ถือว่าเสร็จ', () {
      final summary = PackingSummary.fromJson({
        'packingId': 'PACK-1',
        'status': 'PACKING',
        'orderCount': 0,
        'orderDoneCount': 0,
      });
      expect(summary.isDone, isFalse);      // ← ถ้าลืม orderCount > 0 จะล้ม
    });

    test('PackingDetailResponse — จากสถานะล้วน ไม่สนตัวนับ', () {
      final detail = PackingDetailResponse.fromJson({
        'packingId': 'PACK-1',
        'palletId': 'PAL-1',
        'status': 'PACKING',
        'orders': [
          {'pickOrderId': 'PO-1', 'status': 'DONE', 'partCount': 2, 'partDoneCount': 2},
        ],
      });
      // ทุก Order เสร็จแล้ว แต่ isDone ยังเป็น false เพราะดูแค่ status
      expect(detail.orders.every((o) => o.isDone), isTrue);
      expect(detail.isDone, isFalse);
    });
  });

ข้อ 3 — เทสต์ผ่านแปลว่าถูกไหม

ไม่ — เทสต์ยืนยันว่าโค้ดทำตามที่เขียนไว้ ไม่ได้ยืนยันว่าที่เขียนไว้ถูก

9. สรุป

  • หกคลาสตามสี่ชั้น ใช้รูปแบบสรุป-แล้วดึงเต็ม
  • ConfirmPackResponse.fromJson ไม่มีใครเรียก — ร่องรอยของ endpoint ที่หายไป
  • isDone คำนวณสามแบบในห้าคลาสของไฟล์เดียวกัน
  • PackingSummary.isDone ตรวจสองแหล่งด้วย || — เผื่อสถานะมาไม่ทัน แต่ซ่อนความไม่สอดคล้อง
  • clamp(0, requiredQty) แก้ค่าติดลบที่จุดเดียว ทำให้สามที่ที่ใช้ทำงานถูก
  • โดเมนนี้ป้องกันฟิลด์คำอธิบายครบ — สูงที่สุดในโปรเจกต์
  • e.toString() ทนกว่า e as String แต่ซ่อนปัญหา
  • ทั้งสี่เมธอดส่ง statusCode ต่อ และมีคอมเมนต์อธิบายครบ
  • ส่ง serialNumbers เฉพาะเมื่อไม่ว่าง ต่างจากสามไฟล์อื่นที่ส่ง [] เสมอ
