4. _handleBack ที่ตัดสินตามชั้น

void _handleBack() {
  setState(() {
    switch (_state) {
      case _PackState.packList:
        _resetAll();                      // ← เรียก setState ซ้อน
        break;
      case _PackState.orderList:
        _state = _PackState.packList;
        _packDetail = null;
        _currentPackingId = '';
        break;
      case _PackState.orderParts:
        // ถ้า Pack มีหลาย Order กลับไป orderList แทน packList
        if (_packDetail != null && _packDetail!.orders.length > 1) {
          _state = _PackState.orderList;
        } else {
          _state = _PackState.packList;
          _packDetail = null;
          _currentPackingId = '';
        }
        _orderResp = null;
        _currentPickOrderId = '';
        _selectedPartId = null;
        _collectedSerials.clear();
        _collectedQty.clear();
        _expandedSerialPartIds.clear();
        _partScanCtrl.clear();
        _serialScanCtrl.clear();
        break;
      case _PackState.success:
        _resetAll();                      // ← เรียก setState ซ้อน
        break;
      default:
        break;
    }
  });
}
ทำถูกทำไม
ถอยไป orderList เฉพาะเมื่อมีหลายใบสั่งตรงกับที่ _openPack ข้าม orderList เมื่อมีใบเดียว
ล้างข้อมูลของชั้นที่ออกมาไม่ให้ค้างไปชั้นบน
ล้างช่องกรอกด้วยชั้นถัดไปเริ่มใหม่สะอาด

แต่มีปัญหาโครงสร้าง — _resetAll() เรียก setState ของตัวเอง

void _resetAll() {
  setState(() {                    // ← setState ซ้อนใน setState
    _palletResp = null;
    ...
  });
  WidgetsBinding.instance.addPostFrameCallback((_) {
    _scanFocus.requestFocus();
  });
}
Flutter ยอมให้ทำได้ — setState ซ้อนกันแค่ทำเครื่องหมายว่าต้องวาดใหม่สองครั้ง ไม่โยน exception
แต่อ่านยากและทำให้คนที่มาแก้ทีหลังไม่แน่ใจว่าอะไรอยู่ใน scope ไหน
ทางที่ตรงกว่าคือแยกส่วนที่เปลี่ยนค่าออกจากส่วนที่แจ้ง
void _resetFields() {          // ไม่มี setState
  _palletResp = null;
  ...
}

void _resetAll() {
  setState(_resetFields);
  WidgetsBinding.instance.addPostFrameCallback((_) => _scanFocus.requestFocus());
}

@override
void initState() {
  super.initState();
  WidgetsBinding.instance.addPostFrameCallback((_) {
    if (widget.initialPalletId != null && widget.initialPalletId!.isNotEmpty) {
      _scanCtrl.text = widget.initialPalletId!;
      _scanPallet();                   // ← สแกนให้เลย

      _scanFocus.requestFocus();
    }
  });
}

6. ข้อผิดพลาดที่พบบ่อย

เข้าใจผิดที่จริง
ไฟล์ 2,176 บรรทัดต้องอ่านเรียงบรรทัดอ่านตาม enum ห้าสถานะ
enum บังคับว่าข้อมูลต้องตรงกับสถานะบังคับแค่ว่าสถานะเป็นหนึ่งเดียว
setState ซ้อนกันจะโยน exceptionทำงานได้ แต่อ่านยาก
default: ใน switch ปลอดภัยเสียการตรวจว่าครบ
พารามิเตอร์ที่มีต้องถูกใช้initialPalletId ไม่มีใครส่ง

_collectedCountFor เป็นฟังก์ชันที่ขึ้นกับสอง Map เท่านั้น ดึงออกมาเป็นฟังก์ชันบริสุทธิ์แล้วเทสต์ได้ทันที
int collectedCountFor({
  required bool serialRequire,
  required List<String>? serials,
  required int? qty,
}) => serialRequire ? (serials?.length ?? 0) : (qty ?? 0);
เทสต์สี่ข้อครอบคลุมทุกกิ่ง และเป็นตาข่ายชั้นแรกที่ได้มาฟรี ๆ ก่อนงานใหญ่
