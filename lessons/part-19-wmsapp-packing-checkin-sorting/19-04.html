/// สิ่งที่ทั้ง picking และ packing ต้องรู้เพื่อสแกน S/N
abstract interface class ScannableItem {
  String get partId;
  bool get serialRequire;
  int get targetQty;                    // qtyToPickSuggested หรือ remaining
  List<String> get availableSerials;
}
/// ผลของการตรวจ S/N หนึ่งครั้ง — ไม่แตะ UI ไม่แตะ state
sealed class SerialScanResult {}
class SerialAccepted extends SerialScanResult {}
class SerialDuplicate extends SerialScanResult {}
class SerialNotOnPallet extends SerialScanResult {}
class SerialQuotaFull extends SerialScanResult {
  SerialQuotaFull(this.limit);
  final int limit;
}

SerialScanResult checkSerial({
  required ScannableItem item,
  required List<String> alreadyScanned,
  required String serialNo,
}) {
  if (alreadyScanned.contains(serialNo)) return SerialDuplicate();
  if (!item.availableSerials.contains(serialNo)) return SerialNotOnPallet();
  if (alreadyScanned.length >= item.targetQty) return SerialQuotaFull(item.targetQty);
  return SerialAccepted();
}
ฟังก์ชันนี้เทสต์ได้ทันที — รับข้อมูล คืนผลลัพธ์ ไม่แตะ context ไม่แตะ setState
test('สแกนซ้ำต้องถูกปฏิเสธ', () {
  final result = checkSerial(
    item: _fakeItem(targetQty: 5, serials: ['SN-1', 'SN-2']),
    alreadyScanned: ['SN-1'],
    serialNo: 'SN-1',
  );
  expect(result, isA<SerialDuplicate>());
});
สี่เทสต์ครอบคลุมทั้งสองจอพร้อมกัน — และเป็นตาข่ายที่ต้องมีก่อนจะเริ่มรวมโค้ดจริง

เปลี่ยนจากแถบเขียว (สำเร็จ) เป็นแถบเหลือง (เตือน) — ผู้ใช้และคนที่ยืนข้าง ๆ เห็นว่ากำลังใช้ทางลัด

ข้อสรุปที่สำคัญกว่า: คำถามที่ต้องถามทีมไม่ใช่ “จะซ่อนอย่างไร” แต่คือ “ทำไมยังต้องใช้ตอนติดตั้ง”
ถ้าคำตอบคือ “ข้อมูลทดสอบยังไม่มี S/N จริง” นั่นคือปัญหาของข้อมูลทดสอบ ไม่ใช่ของแอป — และควรแก้ที่นั่น

9. สรุป

  • _scanSerial ของสองไฟล์เหมือนกันสิบสี่บรรทัดแรกทุกตัวอักษร
  • packing เพิ่มด่านตรวจ isDone เพราะโหลดรายการทั้งหมดรวมที่เสร็จแล้ว
  • บั๊ก _addQty กับ _activatePart ที่ 18.12 พบ ถูกคัดลอกมาเป๊ะ
  • หาคู่แฝดได้เร็วที่สุดด้วยการค้นข้อความที่ผู้ใช้เห็น
  • แผ่น “S/N สำหรับทดสอบ” มีสองที่ ไม่ใช่ที่เดียว
  • สีเดียวกันประกาศสองที่ และทั้งคู่ไม่ใช่ AppTheme
  • ราว 900 บรรทัดซ้ำกันระหว่างสองไฟล์
  • รวมยากเพราะโมเดลคนละคลาสและชื่อฟิลด์จำนวนต่างกัน
  • _confirmCollectedPack ส่งทีละรายการและตรวจสถานะทุกรอบ
