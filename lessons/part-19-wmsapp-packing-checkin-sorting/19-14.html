// หลัง — ฟังก์ชันระดับไฟล์ รับทุกอย่างเข้ามา

  required List<String>? serials,
  required int? qty,
}) => serialRequire ? (serials?.length ?? 0) : (qty ?? 0);
// ในจอ — เรียกใช้ ไม่เปลี่ยนพฤติกรรม
int _collectedCountFor(PackingPartItem part) => collectedCountFor(

      serials: _collectedSerials[part.partId],
      qty: _collectedQty[part.partId],
    );

test('ยังไม่มีอะไร → 0', () {

7. สิ่งที่ห้ามทำ

ห้ามทำไม
แก้หลายเรื่องในคอมมิตเดียวพังแล้วไม่รู้ว่าเพราะข้อไหน
เปลี่ยนชื่อพร้อมเปลี่ยนพฤติกรรมdiff อ่านไม่ออกว่าอะไรเปลี่ยนจริง
แตกไฟล์พร้อมแก้บั๊กเหมือนกัน
“ปรับปรุงเล็กน้อย” ระหว่างย้ายโค้ดตั้งใจย้าย 100 บรรทัด แต่เปลี่ยนไป 3 บรรทัดที่ไม่มีใครสังเกต
แก้จอที่ยังไม่ได้อ่านครบสถานะที่ซ่อนอยู่มีจริง (17.5, 19.7)

8. รายการเดินด้วยมือของแต่ละโดเมน

เขียนไว้ก่อนเริ่มแก้ ไม่ใช่นึกทีหลัง

Receiving
  1. สแกน PO → เห็นรายการสินค้า
  2. สแกน Part ที่มี S/N → สแกนครบ → ยืนยัน → ผูกพาเลท
  3. สแกน Part ที่ไม่มี S/N → กรอกจำนวน → ยืนยัน
  4. Part ที่มีหลาย condition → เห็น popup เลือก
  5. ผูกพาเลทไม่สำเร็จ → รายการเข้าลิสต์ค้าง
  6. กดย้อนกลับตอนโหมดผูกพาเลท → กลับโหมดสแกน
Packing
  1. พาเลทที่มี Pack เดียว Order เดียว → ข้ามหน้ารายการ Order
  2. พาเลทที่มี Pack เดียว หลาย Order → ผ่านหน้ารายการ
  3. สินค้าที่ต้องมี S/N → ช่อง S/N + ตาราง
  4. สินค้าที่ไม่ต้องมี S/N → ช่องจำนวน
  5. สแกนชิ้นสุดท้าย → ไปหน้าสรุป + เห็นเลขติดตาม
  6. กดย้อนกลับจากทุกสถานะ → ถอยถูกชั้น
Check-in
  1. สแกนกล่องของช่องใหม่ → เห็น pipeline ทันที
  2. สแกนกล่องของช่องเดิม → เห็นกล่องอื่นในช่อง
  3. กดยืนยัน → เห็นปุ่มปริ้นใบส่งสินค้า
  4. สแกนกล่องสุดท้าย → ช่องปิดและส่งขึ้นรถอัตโนมัติ
  5. สแกนกล่องที่เช็คอินไปแล้ว → ถูกปฏิเสธ
Sorting
  1. เปิดจอ → เห็นสิบสถานีพร้อมสถานะ
  2. แตะสถานี AVAILABLE → เมนูปิดใช้งาน
  3. แตะสถานี BUSY → แผ่นรายละเอียด + รายการกล่อง
  4. แตะสถานี DISABLED → เมนูเปิดใช้งาน + เหตุผล
  5. Clear สถานีที่เต็ม → กลับเป็นว่าง และกริดอัปเดต
  6. ลากลงรีเฟรช
สิ่งที่ต้องทดสอบเพิ่มถ้าแตะเรื่องเครือข่าย
ปิด Wi-Fi กลางคำขอ แล้วดูข้อความ
ปิดเซิร์ฟเวอร์ทดสอบ แล้วเปิดจอ
เปิดจอค้างไว้ห้านาที แล้วกลับมาใช้

สามข้อนี้ตรงกับที่ 15.11 ระบุไว้ และเป็นสิ่งที่มักไม่มีใครทดสอบ

9. ข้อผิดพลาดที่พบบ่อย

เข้าใจผิดที่จริง
ไม่มีเทสต์แปลว่าแก้ไม่ได้สามในสี่ระดับตรวจได้โดยไม่ต้องมีเทสต์เดิม
ต้องเขียนเทสต์ให้ครบก่อนแก้เขียนเฉพาะส่วนที่จะดึงออกมาก็พอ
จัดระเบียบโค้ดก่อน แล้วค่อยแก้บั๊กบั๊กที่ผู้ใช้เจออยู่ควรมาก่อน
ปรับปรุงเล็กน้อยระหว่างย้ายโค้ดไม่เป็นไรทำให้ไล่หาสาเหตุไม่ได้
คอมไพเลอร์จับได้ทุกอย่างถ้าเปลี่ยนชนิดลำดับและพฤติกรรมจับไม่ได้

// ขั้นที่ 1 — ไม่แตะโค้ดเดิมเลย
class ApiResult<T> {
  ...
  /// แปลงความล้มเหลวเป็นชนิดอื่นโดยพก statusCode ไปด้วยเสมอ
  ///
  /// ใช้แทน `ApiResult.error(response.error)` ในชั้น extension
  /// เพื่อไม่ให้ statusCode หายระหว่างทาง (isNotFound จะใช้ได้)
  ApiResult<R> asFailure<R>() =>
      ApiResult<R>.error(error, statusCode: statusCode);
}
// ขั้นที่ 2
test('asFailure พก statusCode ไปด้วย', () {
  final source = ApiResult<Map<String, dynamic>>.error('ไม่พบ', statusCode: 404);
  final converted = source.asFailure<POResponse>();

  expect(converted.success, isFalse);
  expect(converted.error, 'ไม่พบ');
  expect(converted.statusCode, 404);
  expect(converted.isNotFound, isTrue);      // ← สิ่งที่ 16.7 บอกว่าใช้ไม่ได้
});

test('asFailure ไม่มี statusCode ก็ยังทำงาน', () {
  final source = ApiResult<Map<String, dynamic>>.error('เน็ตหลุด');
  final converted = source.asFailure<POResponse>();

  expect(converted.statusCode, isNull);
  expect(converted.isNotFound, isFalse);
});
