1. สี่ชั้นของข้อมูล

Pallet (พาเลท)
  └─ Pack (กล่องที่จะส่ง)
       └─ Order (ใบสั่งของลูกค้า)
            └─ Part (สินค้าแต่ละชนิด)
ชั้นendpointเก็บใน
PalletGET /packing/pallet/{id}_palletResp
PackGET /packing/pack/{id}_packDetail
OrderGET /packing/pack/{id}/order/{orderId}_orderResp
Partอยู่ใน _orderResp.parts—

สามคำขอเพื่อไปถึงหน้าที่ทำงานจริง

ทำไมไม่ส่งมาทีเดียว — พาเลทหนึ่งใบอาจมีหลายกล่อง แต่ละกล่องมีหลายใบสั่ง แต่ละใบสั่งมีหลายสินค้า
ถ้าส่งครบทุกชั้นในคำขอเดียว ข้อมูลจะใหญ่มากและส่วนใหญ่ไม่ถูกใช้ — ผู้ใช้ทำทีละกล่อง
การแบ่งเป็นสามคำขอจึงถูกต้อง แต่แลกมาด้วยการรอสามครั้ง

2. ชั้นที่ 1 — สแกนพาเลท

Future<void> _scanPallet() async {
  final palletId = _scanCtrl.text.trim().toUpperCase();
  if (palletId.isEmpty) return;

  setState(() => _loading = true);
  final result = await _api.scanPalletForPacking(palletId);
  if (!mounted) return;
  setState(() => _loading = false);

  if (!result.success) {
    showErrorDialog(context, message: result.error ?? 'ไม่พบ Pallet');
    _scanCtrl.clear();
    _scanFocus.requestFocus();
    return;
  }

  setState(() {
    _palletResp = result.data;
    _state = _PackState.packList;
  });
}

3. ชั้นที่ 2 — เปิดกล่อง และการข้ามชั้น

Future<void> _openPack(String packingId) async {
  setState(() => _loading = true);
  final result = await _api.getPack(packingId);
  if (!mounted) return;
  setState(() => _loading = false);

  if (!result.success) {
    showErrorDialog(context, message: result.error ?? 'โหลด Pack ไม่สำเร็จ');
    return;
  }

  setState(() {
    _currentPackingId = packingId;
    _packDetail = result.data;
    // ถ้ามี Order เดียว เข้า Order เลย
    if (result.data!.orders.length == 1) {
      _openOrder(result.data!.orders.first.pickOrderId);   // ← async ใน setState
    } else {
      _state = _PackState.orderList;
    }
  });
}

บรรทัดที่ทำเครื่องหมายไว้เรียกเมธอด async จากใน setState

เกิดอะไรขึ้นตามลำดับ
1. _openOrder(...) ถูกเรียก แต่ไม่มีใคร await
2. บรรทัดแรกของมันคือ setState(() => _loading = true) — setState ซ้อน
3. setState ชั้นนอกจบ Flutter ทำเครื่องหมายว่าต้องวาดใหม่
4. _openOrder ยังทำงานต่อในเบื้องหลัง
5. คำขอกลับมา → setState อีกครั้ง → _state = orderParts
ผลลัพธ์ถูกต้อง แต่มีช่วงที่สถานะไม่ตรงกับข้อมูล
ระหว่างขั้นที่ 3 ถึง 5:
  _state       = scanPallet (ค่าเดิม — ไม่มีใครเปลี่ยน)
  _packDetail  = มีค่าแล้ว
  _loading     = true
จอจะวาด _buildScanPallet() โดยมีม่านโหลดทับ — ผู้ใช้เห็นหน้าสแกนพาเลทกระพริบก่อนไปหน้าสินค้า

เขียนให้ตรงกว่าได้โดยเอาการเรียก async ออกจาก setState

4. ชั้นที่ 3 — เปิดใบสั่ง

Future<void> _openOrder(String pickOrderId) async {
  setState(() => _loading = true);
  final result = await _api.getPackOrder(
    packingId: _currentPackingId,
    pickOrderId: pickOrderId,
  );
  if (!mounted) return;
  setState(() => _loading = false);

  if (!result.success) {
    showErrorDialog(context, message: result.error ?? 'โหลด Order ไม่สำเร็จ');
    return;                        // ← ค้างอยู่สถานะเดิม
  }

  setState(() {
    _orderResp = result.data;
    _currentPickOrderId = pickOrderId;
    _selectedPartId = null;
    _collectedSerials.clear();
    _collectedQty.clear();
    _expandedSerialPartIds.clear();
    _partScanCtrl.clear();
    _serialScanCtrl.clear();
    _state = _PackState.orderParts;
  });
  WidgetsBinding.instance.addPostFrameCallback((_) {
    _partScanFocus.requestFocus();
  });
}

setState ก้อนนี้ล้างสถานะของชั้นล่างทั้งหมดก่อนเข้า — ทำให้ไม่มีข้อมูลของใบสั่งก่อนหน้าค้าง

5. การกลับไปชั้นบนหลังแพ็กเสร็จ

Future<void> _refreshPackList() async {
  // reload pallet data แล้วกลับหน้า packList
  setState(() => _loading = true);
  final palletId = _palletResp!.palletId;
  final result = await _api.scanPalletForPacking(palletId);
  if (!mounted) return;
  setState(() => _loading = false);

  if (!result.success) {
    // pallet อาจ shipped ไปแล้ว
    showErrorDialog(context, message: result.error ?? 'โหลดข้อมูลไม่สำเร็จ');
    _resetAll();
    return;
  }

  setState(() {
    _palletResp = result.data;
    _packDetail = null;
    _currentPackingId = '';
    _currentPickOrderId = '';
    _orderResp = null;
    _selectedPartId = null;
    _collectedSerials.clear();
    _collectedQty.clear();
    _expandedSerialPartIds.clear();
    _state = _PackState.packList;
  });
}
ทำถูกทำไม
โหลดพาเลทใหม่ ไม่ใช้ข้อมูลเก่ากล่องที่เพิ่งแพ็กเสร็จต้องเปลี่ยนสถานะ
คอมเมนต์บอกว่าพาเลทอาจถูกส่งไปแล้วอธิบายว่าทำไมต้อง _resetAll()
ล้างข้อมูลชั้นล่างทั้งหมดกลับไปชั้นที่ 1

แถวกลางเป็นการจัดการที่ถูก — เมื่อกล่องสุดท้ายถูกแพ็ก พาเลทจะถูกปล่อยและสแกนไม่ได้อีก

message: resp.palletReleased
    ? 'Pack สำเร็จ — Pallet เปล่าพร้อมใช้ใหม่'
    : 'Pack สำเร็จ (ยังเหลือ Pack อื่นใน Pallet)',

ข้อความบอกผู้ใช้ล่วงหน้าว่าพาเลทว่างแล้ว — ผู้ใช้จึงไม่แปลกใจเมื่อสแกนซ้ำแล้วไม่เจอ

แต่ _palletResp! ใช้เครื่องหมายอัศเจรีย์
final palletId = _palletResp!.palletId;
เมธอดนี้เรียกจากหน้า success ซึ่ง _palletResp ต้องมีค่าเสมอตามเส้นทางปัจจุบัน — แต่ไม่มีอะไรบังคับไว้
เป็นรูปแบบเดียวกับ _fullName! ใน 16.3 และ line.palletId! ใน 17.9

7. ข้อผิดพลาดที่พบบ่อย

เข้าใจผิดที่จริง
เรียก async ใน setState ไม่ได้ได้ แต่ทำให้มีช่วงที่สถานะไม่ตรงกับข้อมูล
ล้มเหลวแล้วผู้ใช้อยู่ที่เดิมเสมอขึ้นกับว่าสถานะถูกเปลี่ยนก่อนหรือหลัง

enum ทำให้หาจุดที่เปลี่ยนสถานะยากตรงกันข้าม — ค้นคำเดียวเจอครบ
โหลดใหม่หลังแพ็กเสร็จเป็นการเสียเวลาจำเป็น เพราะสถานะกล่องเปลี่ยน

8. แบบฝึกหัด

ผู้ใช้แจ้งว่า “กดเลือกกล่องแล้วเน็ตสะดุด แอปเด้งกลับไปหน้าสแกนพาเลท ต้องเริ่มใหม่”

  1. อธิบายว่าเกิดได้อย่างไร และเกิดเฉพาะกล่องแบบไหน
  2. แก้อย่างไร
  3. ยังมีที่อื่นในไฟล์นี้ที่มีปัญหาแบบเดียวกันไหม
แนวคำตอบ

ข้อ 1 — เกิดได้อย่างไร

ผู้ใช้อยู่ packList → แตะกล่อง PACK-007
  ▼
_openPack('PACK-007')
  → getPack สำเร็จ
  → setState { _packDetail = ..., }
      → กล่องมี Order เดียว → _openOrder(...) ถูกเรียก
                              แต่ _state ยังเป็น packList
  ▼
_openOrder → getPackOrder → เน็ตสะดุด → ล้มเหลว
  → showErrorDialog → return
  ▼
_state ยังเป็น packList → ผู้ใช้เห็นหน้ารายการกล่อง

เดี๋ยว — ที่นี่ยังอยู่ packList ซึ่งถูกต้อง ไม่ได้เด้งกลับไป scanPallet

ต้องดูเส้นทางที่ทำให้ถึง scanPallet

// _handleBack เมื่ออยู่ packList
case _PackState.packList:
  _resetAll();          // ← กลับไป scanPallet
ผู้ใช้ทำผล
เห็นกล่อง error แล้วกดปิดอยู่ packList
กดปุ่มย้อนกลับของเครื่องเพื่อปิดกล่องกล่องปิด
กดย้อนกลับอีกครั้ง_handleBack() → _resetAll() → กลับหน้าสแกนพาเลท

ผู้ใช้ที่เคยชินกับการกดย้อนกลับสองครั้งเพื่อปิดกล่องแล้วถอย จะเจอเรื่องนี้

แต่มีอีกเส้นทางที่ตรงกว่า — เมื่อ _openOrder ถูกเรียกจาก _openPack แล้ว ผู้ใช้กดย้อนกลับระหว่างที่คำขอยังค้าง

_openPack → setState → _openOrder เริ่ม (คำขอค้าง)
  → _state ยังเป็น packList
    → ผู้ใช้กดย้อนกลับ → canPop = false → _handleBack()
      → case packList: _resetAll() → _state = scanPallet
        → คำขอกลับมา → if (!mounted) ผ่าน (จอยังอยู่)
          → setState { _state = orderParts }   ← เด้งไปหน้าสินค้าเอง!

กรณีนี้ร้ายกว่า — ผู้ใช้กดย้อนกลับแล้วถูกพาไปหน้าสินค้าโดยไม่ได้ตั้งใจ

เกิดเฉพาะกล่องที่มี Order เดียว เพราะเป็นเส้นทางเดียวที่เรียก _openOrder โดยที่สถานะยังไม่เปลี่ยน

_openOrder (เรียกจาก UI)ไม่ผู้ใช้อยู่ orderList ซึ่งมีข้อมูลรองรับ

ที่นี่ไม่เด้งหน้า เพราะไม่ได้แตะ _state — แต่ข้อมูลจะไม่ตรงกับสถานะ

วิธีป้องกันเดียวกัน

if (_state != _PackState.orderParts) return;
หลักที่ได้: ทุกเมธอด async ที่จบด้วยการ setState ต้องตอบคำถามสองข้อ
  1. จอยังอยู่ไหม → if (!mounted)
  2. ผู้ใช้ยังอยู่ที่เดิมไหม → เทียบสถานะหรือตัวระบุงาน
ข้อสองถูกลืมบ่อยกว่ามาก และเป็นสิ่งที่ enum ช่วยให้เขียนได้ง่าย — เทียบค่าเดียวก็รู้

9. สรุป

  • สี่ชั้น: พาเลท → กล่อง → ใบสั่ง → สินค้า และสามคำขอเพื่อไปถึงหน้าทำงาน
  • _scanPallet ตั้งข้อมูลกับสถานะใน setState เดียวกัน — ไม่มีช่วงที่ไม่ตรง
  • _openPack เรียก _openOrder จากใน setState ทำให้เกิด setState ซ้อน
  • และทำให้มีช่วงที่ _state ยังเป็นค่าเดิมขณะที่ข้อมูลเปลี่ยนไปแล้ว
  • ถ้าโหลดใบสั่งล้มเหลวจากเส้นทางนั้น ผู้ใช้ถูกพากลับไปสองชั้น
  • _openOrder ล้างสถานะของชั้นล่างครบทุกตัวก่อนเข้า
  • _refreshPackList โหลดใหม่และรับมือกรณีพาเลทถูกปล่อยไปแล้ว
  • เจ็ดเมธอดแตะ _state และค้นได้ด้วยคำเดียว
  • เมธอด async ที่จบด้วย setState ต้องตรวจทั้ง mounted และว่าผู้ใช้ยังอยู่ที่เดิม
