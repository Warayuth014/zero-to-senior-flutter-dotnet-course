2. โหลดทั้งหมดแล้วกรองในเครื่อง

bool _loading = false;
List<Map<String, dynamic>> _allParts = [];
List<Map<String, dynamic>> _filteredParts = [];

@override
void initState() {
  super.initState();
  _loadParts();
  _searchCtrl.addListener(_filterParts);     // ← กรองทุกครั้งที่พิมพ์
}
void _filterParts() {
  final q = _searchCtrl.text.trim().toLowerCase();
  setState(() {
    if (q.isEmpty) {
      _filteredParts = _allParts;
    } else {
      _filteredParts = _allParts.where((p) {
        final partId = (p['partId'] as String).toLowerCase();
        final desc = (p['itemDesc'] as String).toLowerCase();
        final brand = (p['brand'] as String).toLowerCase();
        return partId.contains(q) || desc.contains(q) || brand.contains(q);
      }).toList();
    }
  });
}

3. เลือกรูปแล้วอัปโหลด

Future<void> _pickAndUpload(Map<String, dynamic> part) async {
  final source = await showModalBottomSheet<ImageSource>(
    context: context,
    builder: (_) => SafeArea(
      child: Column(
        mainAxisSize: MainAxisSize.min,
        children: [
          ListTile(
            leading: Icon(MdiIcons.camera),
            title: const Text('ถ่ายรูป'),
            onTap: () => Navigator.pop(context, ImageSource.camera),
          ),
          ListTile(
            leading: Icon(MdiIcons.imageMultipleOutline),
            title: const Text('เลือกจากแกลเลอรี'),
            onTap: () => Navigator.pop(context, ImageSource.gallery),
          ),
        ],
      ),
    ),
  );

  if (source == null) return;

  final picked = await _picker.pickImage(
    source: source,
    maxWidth: 1024,
    maxHeight: 1024,
    imageQuality: 85,
  );
  if (picked == null) return;

  setState(() => _loading = true);
  final result = await _api.uploadPartImage(
    partId: part['partId'],
    imageFile: File(picked.path),
  );
  if (!mounted) return;
  setState(() => _loading = false);

  if (!result.success) {
    showErrorDialog(context, message: result.error ?? 'อัปโหลดไม่สำเร็จ');
    return;
  }

  showSuccessSnackbar(context, 'อัปโหลดรูป ${part['partId']} สำเร็จ');
  await _loadParts();
}

6. ช่องค้นหาที่ไม่หน่วง

_searchCtrl.addListener(_filterParts);

กรองทุกครั้งที่ตัวอักษรเปลี่ยน

จำนวนสินค้างานต่อการพิมพ์หนึ่งตัว
200วน 200 × เทียบสามฟิลด์ = 600 การเทียบ
2,0006,000 การเทียบ

การเทียบสตริงเร็วมาก — ที่หลักพันยังไม่รู้สึก

สิ่งที่รู้สึกคือการวาดใหม่

setState(() { _filteredParts = ...; });     // ← วาดรายการทั้งหมดใหม่

รวมกับปัญหา FutureBuilder ในหัวข้อ 4 — รูปทุกใบที่เห็นบนจอสร้าง Future ใหม่ทุกตัวอักษร

Timer? _searchDebounce;

@override
void initState() {
  super.initState();
  _loadParts();
  _searchCtrl.addListener(_onSearchChanged);
}

void _onSearchChanged() {
  // ผู้ใช้พิมพ์ต่อเนื่อง — รอให้หยุดก่อนค่อยกรอง
  _searchDebounce?.cancel();
  _searchDebounce = Timer(const Duration(milliseconds: 250), _filterParts);
}

@override
void dispose() {
  _searchDebounce?.cancel();          // ← ห้ามลืม
  _searchCtrl.removeListener(_onSearchChanged);
  _searchCtrl.dispose();
  super.dispose();
}
แต่ควรแก้ FutureBuilder ก่อน — การหน่วงลดจำนวนครั้ง ส่วนการแก้ FutureBuilder ลดต้นทุนต่อครั้ง
ถ้าแก้อย่างหลังแล้ว การวาดใหม่จะถูกมากจนไม่ต้องหน่วง
เป็นหลักเดียวกับที่ 17.4 สรุปว่าให้แก้ที่ต้นทุนต่อครั้งก่อน

สังเกตว่า dispose ปัจจุบันไม่ได้ถอด listener

@override
void dispose() {
  _searchCtrl.dispose();
  super.dispose();
}
ไม่ถอด listener แล้วรั่วไหม
ไม่ — dispose() ของ TextEditingController ล้างผู้ฟังทั้งหมดให้

ไม่โหลดทั้งหมดเปิดจอเร็ว
ค้นที่เซิร์ฟเวอร์ใช้ดัชนีของฐานข้อมูล
ต้องหน่วงไม่ยิงทุกตัวอักษร
เพิ่มตัวกรอง “ยังไม่มีรูป”ตอบโจทย์งานจริงของหัวหน้า
ตัวกรอง “ยังไม่มีรูป” สำคัญกว่าการค้นหาด้วยซ้ำ — งานของจอนี้คือไล่เติมรูปให้ครบ
ผู้ใช้ไม่ได้ค้นหาสินค้าตัวใดตัวหนึ่ง แต่ต้องการรายการที่ยังขาด
ตอนนี้ต้องเลื่อนหาเองจากขอบสีเทา — ซึ่งใช้ไม่ได้ที่หลักพันขึ้นไป

ข้อสรุปที่ใหญ่กว่า: ก่อนแก้เรื่องประสิทธิภาพ ให้ถามว่าผู้ใช้ใช้จอนี้ทำอะไร

คำตอบ “ไล่เติมรูปให้ครบ” เปลี่ยนลำดับความสำคัญ — ตัวกรองสำคัญกว่าการหน่วงช่องค้นหา และการค้นที่เซิร์ฟเวอร์สำคัญกว่าการปรับแต่งการกรองในเครื่อง

9. สรุป

  • จอนี้ทำงานได้ครบแต่ไม่มีจอไหนเปิดเลย — ขาดแค่ปุ่มเข้า
  • โฟลเดอร์ supervisor/ บอกเจตนา แต่ไม่มีการตรวจ role
  • โหลดสินค้าทั้งหมดแล้วกรองในเครื่อง — ดีจนถึงหลักพัน
  • as String ที่ itemDesc ทำให้พังตอนพิมพ์ค้นหา ไม่ใช่ตอนเปิดจอ
  • เลือกรูปได้ทั้งกล้องและแกลเลอรี และย่อตั้งแต่ตอนเลือก
  • เขียน thumbnail เอง 30 บรรทัดที่มีปัญหาเดียวกับ PartThumbnail ทั้งสองข้อ
  • ปัญหา FutureBuilder หนักกว่าที่อื่น เพราะช่องค้นหากรองทุกตัวอักษร
  • สามสัญญาณบอกว่ามีรูปหรือยัง แต่ตัวนับใช้เงื่อนไขต่างจากการ์ด
  • ค้นหาไม่หน่วง — แต่ควรแก้ FutureBuilder ก่อน
  • ก่อนแก้ประสิทธิภาพ ให้ถามว่าผู้ใช้ใช้จอนี้ทำอะไร

บทหน้า: วิธีแก้จอใหญ่โดยไม่พัง ในโปรเจกต์ที่ไม่มีเทสต์
