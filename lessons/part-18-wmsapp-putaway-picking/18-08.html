2. ref-count คืออะไรและแก้ปัญหาอะไร

Future<void> connect() async {
  _refCount++;
  if (_connection != null &&
      _connection!.state != HubConnectionState.Disconnected) {
    return;                      // ต่ออยู่แล้ว ไม่ต่อซ้ำ
  }
  ... สร้างการเชื่อมต่อใหม่ ...
}

Future<void> disconnect() async {
  _refCount--;
  if (_refCount <= 0) {
    _refCount = 0;
    if (_connection != null) {
      await _connection!.stop();
      _connection = null;
    }
  }
}

ปัญหาที่แก้: สองจอที่ใช้ hub เดียวกันอาจเปิดพร้อมกัน

PutawayScreen เปิด        → connect() → _refCount = 1, ต่อจริง
PutawayPreworkScreen เปิด → connect() → _refCount = 2, ไม่ต่อซ้ำ
PutawayScreen ปิด         → disconnect() → _refCount = 1, ไม่ตัด
PutawayPreworkScreen ปิด  → disconnect() → _refCount = 0, ตัดจริง

ถ้าไม่มี ref-count จอแรกที่ปิดจะตัดการเชื่อมต่อของจอที่ยังเปิดอยู่ — ตรงกับที่ 12.7 สอน

ทำถูกทำไม
นับก่อนตรวจสถานะ_refCount++ อยู่บรรทัดแรก ไม่ว่าจะต่อใหม่หรือไม่
ป้องกันติดลบif (_refCount <= 0) _refCount = 0;
ล้าง _connection หลัง stopครั้งต่อไปจะสร้างใหม่ ไม่ใช้ตัวที่ตายแล้ว

3. ที่ ref-count นับผิดได้

Future<void> connect() async {
  _refCount++;                        // ← เพิ่มเสมอ
  ...
  try {
    ...
    await _connection!.start();
  } catch (e) {
    dev.log('[SignalR] Connect error: $e');   // ← ล้มเหลวก็ไม่ลด
  }
}

ถ้าต่อไม่สำเร็จ _refCount ยังเพิ่มขึ้น — และ _connection อาจเป็นค่าที่สร้างแล้วแต่ start ไม่ผ่าน

เน็ตมีปัญหา
  PutawayScreen เปิด → connect() → _refCount = 1, start() โยน, _connection ค้าง
    → ผู้ใช้ปิดจอ → disconnect() → _refCount = 0 → stop() ตัวที่ไม่เคยต่อ

ในกรณีนี้ยังกลับมาเป็นศูนย์ถูก แต่มีเส้นทางที่นับค้าง

PutawayScreen เปิด → connect() → _refCount = 1
  → ผู้ใช้กดเข้า PutawayPreworkScreen ทันที (จอแรกยังไม่ถูก dispose)
    → connect() → _refCount = 2
      → ผู้ใช้กดย้อนกลับสองครั้งเร็ว ๆ
        → dispose() ทั้งสอง → _refCount = 0 ✓

เส้นทางปกติถูกต้อง ปัญหาอยู่ที่เส้นทางที่ dispose() ไม่ถูกเรียก

@override
void dispose() {
  _signalR.removeStationDispatchedListener(_handleStationDispatched);
  ...
  _signalR.disconnect();          // ← ไม่ await
  _stationController.dispose();
  super.dispose();
}
ข้อสังเกตผล
disconnect() เป็น async แต่ไม่ awaitถูกต้อง — dispose() เป็น sync จะ await ไม่ได้
ถ้า stop() โยน exceptionไม่มีใครจับ — กลายเป็น unhandled
Future<void> disconnect() async {
  _refCount--;
  if (_refCount <= 0) {
    _refCount = 0;
    final connection = _connection;
    _connection = null;              // ล้างก่อน กันการเรียกซ้อน
    if (connection != null) {
      try {
        await connection.stop();
      } catch (e) {
        dev.log('[SignalR] Stop error: $e');   // ตัดไม่ได้ก็ไม่ควรทำให้แอปพัง
      }
    }
  }
}

ผู้ใช้เห็นความจริง
จอสถานีปกติเชื่อมต่อสำเร็จ ข้อมูลสดตลอด
จอสถานีปกติเชื่อมต่อไม่ได้ ข้อมูลค้างตั้งแต่โหลดครั้งแรก

สองกรณีนี้หน้าตาเหมือนกันทุกพิกเซล และ dev.log เห็นได้เฉพาะตอนต่อ debugger

ตรงกับที่ 12.11 เรียกว่าความล้มเหลวเงียบสี่แบบ — และนี่คือแบบแรก: ต่อไม่ติดตั้งแต่แรก
ผลในคลัง: ผู้ใช้ยืนดูสถานีที่ยังแสดงว่าว่าง ทั้งที่มีพาเลทอยู่จริง แล้วเดินไปสแกนเพื่อเจอ error

สิ่งที่ 10.12 และ 12.3 สอนคือผู้ใช้ต้องเห็นสถานะของการเชื่อมต่อ

enum RealtimeStatus { unknown, connecting, connected, disconnected }

class SignalRService {
  final ValueNotifier<RealtimeStatus> status =
      ValueNotifier(RealtimeStatus.unknown);

  Future<void> connect() async {

        _connection!.state != HubConnectionState.Disconnected) return;

    status.value = RealtimeStatus.connecting;

    }
  }
}

5. เหตุการณ์ที่หายไประหว่างหลุด

_connection = HubConnectionBuilder()
    .withUrl(hubUrl)
    .withAutomaticReconnect()      // ← ต่อกลับเอง
    .build();

withAutomaticReconnect() ทำให้ไลบรารีต่อกลับเองเมื่อสายหลุด — ทำถูก

แต่ไม่มี onreconnected

เชื่อมต่ออยู่ → สถานีทุกใบว่าง
  ↓ สัญญาณหลุด 40 วินาที
  ระหว่างนั้น: StationDispatched × 3, PalletArrived × 2
  ↓ ต่อกลับได้เอง
  → แอปไม่รู้ว่าพลาดอะไรไป → หน้าจอยังแสดงว่าทุกสถานีว่าง
เหตุการณ์ที่พลาดผลบนหน้าจอ
StationDispatchedสถานีที่กำลังทำงานยังแสดงว่าว่าง
PalletArrivedพาเลทมาถึงแล้วแต่จอไม่รู้
PalletReturnedสถานีว่างแล้วแต่จอแสดงว่ายังทำงาน

ผู้ใช้ที่แตะสถานีที่ “ว่าง” จะเจอ error ตอนสแกน — หรือแย่กว่าคือไม่แตะเลยเพราะเห็นว่า “ทำงานอยู่”

นี่คือข้อ 14 ในรายการตรวจของ 15.11 — ต่อกลับแล้วต้องโหลดใหม่ทั้งชุด ซึ่งเป็นข้อที่ 12.4 ระบุว่าโปรเจกต์นี้ยังไม่ทำ
เหตุผลที่ต้องโหลดใหม่ทั้งชุด ไม่ใช่แค่ที่พลาด: แอปไม่รู้ว่าพลาดอะไรไป จึงเดาไม่ได้ว่าต้องขออะไรเพิ่ม

แก้ได้โดยเปิดช่องให้จอรู้ว่าต่อกลับแล้ว

final List<VoidCallback> _reconnectedListeners = [];

void addReconnectedListener(VoidCallback cb) => _reconnectedListeners.add(cb);
void removeReconnectedListener(VoidCallback cb) => _reconnectedListeners.remove(cb);

Future<void> connect() async {
  ...
  _connection!.onreconnected(({connectionId}) {
    status.value = RealtimeStatus.connected;
    // แอปไม่รู้ว่าพลาดอะไรไประหว่างหลุด — ให้ทุกจอโหลดใหม่ทั้งชุด
    for (final cb in List.of(_reconnectedListeners)) cb();
  });

  _connection!.onreconnecting(({error}) {
    status.value = RealtimeStatus.connecting;
  });

  _connection!.onclose(({error}) {
    status.value = RealtimeStatus.disconnected;
  });
  ...
}
// ในจอ
@override
void initState() {
  super.initState();
  _signalR.addReconnectedListener(_loadStationStatus);   // ← โหลดใหม่ทั้งชุด
  ...
}

@override
void dispose() {
  _signalR.removeReconnectedListener(_loadStationStatus);
  ...
}

เมธอด _loadStationStatus มีอยู่แล้วและโหลดทั้งชุดอยู่แล้ว — แค่ต้องมีใครเรียกตอนต่อกลับ

6. การกระจายเหตุการณ์ที่ทำถูก

void _onStationDispatched(List<Object?>? args) {
  final data = _parseArgs(args);
  for (final cb in List.of(_stationDispatchedListeners)) {
    cb(data);
  }
}

List.of(...) คัดลอกรายการก่อนวน

ถ้าไม่คัดลอก
ผู้ฟังตัวหนึ่งเรียก removeXxxListener ระหว่างที่กำลังวน → ConcurrentModificationError

เกิดได้จริงในจอนี้ — ผู้ฟังเรียก setState ซึ่งอาจทำให้จอถูกถอดและ dispose() ทำงาน

รูปแบบเดียวกับที่ 12.5 สอนและที่ 17.6 ใช้กับ List<String>.of(_serialNumbers)

Map<String, dynamic> _parseArgs(List<Object?>? args) {
  if (args == null || args.isEmpty) return {};
  final first = args.first;
  if (first is Map<String, dynamic>) return first;
  if (first is Map) return Map<String, dynamic>.from(first);
  return {};
}
กรณีคืนอะไร
ไม่มี args{} — ไม่โยน
เป็น Map<String, dynamic> อยู่แล้วตัวเดิม ไม่คัดลอก
เป็น Map ชนิดอื่นแปลงชนิด
ไม่ใช่ Map เลย{} — ไม่โยน
การถอดรหัสที่ไม่โยน exception เป็นสิ่งจำเป็นในตัวจัดการเหตุการณ์ — ถ้าโยน จะเกิดในบริบทที่ไม่มีใครรอ (ไม่มี await ครอบ) และกลายเป็น unhandled error
ตรงกับที่ 12.6 สอน และเป็นสิ่งที่ทำถูกทั้งสองบริการ

แต่มีราคาที่ต้องรู้ — ข้อมูลที่ผิดรูปจะเงียบ

// ถ้าเซิร์ฟเวอร์เปลี่ยนรูปแบบ payload
return {};      // → ผู้ฟังได้ Map ว่าง → data['stationId'] เป็น null → return เงียบ
// เห็นได้ตอนพัฒนา แต่ไม่พังใน release
Map<String, dynamic> _parseArgs(List<Object?>? args) {
  if (args == null || args.isEmpty) return {};
  final first = args.first;
  if (first is Map<String, dynamic>) return first;
  if (first is Map) return Map<String, dynamic>.from(first);

  assert(false, 'SignalR payload ไม่ใช่ Map: ${first.runtimeType}');
  return {};
}

7. ข้อผิดพลาดที่พบบ่อย

เข้าใจผิดที่จริง
withAutomaticReconnect() พอแล้วต่อกลับได้ แต่ไม่รู้ว่าพลาดอะไรไป
connect() ที่ไม่โยน แปลว่าต่อสำเร็จจับ exception ไว้แล้วเขียน log
ref-count ไม่จำเป็นถ้ามีจอเดียวที่นี่มีสองจอที่ใช้ hub เดียวกัน
วน list ตรง ๆ ได้ถ้าไม่มีใครแก้ผู้ฟังเรียก setState → จอถูกถอด → remove ระหว่างวน
ตัวถอดรหัสควรโยนเมื่อข้อมูลผิดในตัวจัดการเหตุการณ์ไม่ควร ไม่มีใครจับ

8. แบบฝึกหัด

ผู้ใช้แจ้งว่า “ตอนเช้าใช้ดี พอบ่ายสถานีบนจอไม่ตรงกับของจริงเลย ต้องปิดแอปเปิดใหม่”

  1. อธิบายว่าเกิดได้อย่างไร
  2. ทำไม “ปิดแอปเปิดใหม่” ถึงแก้ได้
  3. แก้ให้ผู้ใช้ไม่ต้องปิดแอป
แนวคำตอบ

ข้อ 1 — เกิดได้อย่างไร

เช้า: เปิดแอป → connect() สำเร็จ → เหตุการณ์เข้ามาปกติ
      _loadStationStatus() โหลดครั้งแรก → หน้าจอตรง

บ่าย: ผู้ใช้เดินเข้าโซนที่สัญญาณอ่อน
      → สายหลุด
      → withAutomaticReconnect() พยายามต่อกลับ
        ├─ ถ้าต่อกลับได้: ไม่มี onreconnected → ไม่โหลดใหม่
        │                  → เหตุการณ์ที่พลาดหายไปถาวร
        └─ ถ้าต่อกลับไม่ได้: ไม่มีอะไรบอกผู้ใช้
      → หน้าจอค้างที่สถานะตอนก่อนหลุด

สังเกตว่าทั้งสองสาขาให้ผลเดียวกันในสายตาผู้ใช้ — หน้าจอไม่ตรง

และไม่มีอะไรบนจอที่ต่างกันระหว่าง “ยังเชื่อมต่ออยู่” กับ “หลุดไปแล้ว”

ข้อ 2 — ทำไมปิดแอปแล้วหาย

ปิดแอปทำให้ผล
dispose() ทำงาน → _refCount = 0 → ตัดจริงสถานะเก่าถูกล้าง
เปิดใหม่ → initState → connect()ต่อใหม่หมด
_loadStationStatus() ถูกเรียกอีกครั้งโหลดสถานะจริงทั้งชุด

แถวสุดท้ายคือสิ่งที่แก้ปัญหาจริง — ไม่ใช่การต่อใหม่ แต่คือการโหลดข้อมูลใหม่ทั้งชุด

ข้อสังเกตนี้สำคัญ เพราะบอกว่าทางแก้ไม่ต้องแตะเรื่องการเชื่อมต่อเลย — แค่ต้องมีใครเรียก _loadStationStatus() ในจังหวะที่ถูก

ข้อ 3 — แก้

ทำสามชั้น เรียงตามความคุ้มค่า

ชั้นทำอะไรแก้กรณีไหน
1โหลดใหม่เมื่อกลับมาจากพื้นหลังผู้ใช้เก็บเครื่องแล้วหยิบมาใช้
2โหลดใหม่เมื่อ SignalR ต่อกลับหลุดสั้น ๆ แล้วกลับมาเอง
3โหลดใหม่เป็นระยะ + แสดงสถานะต่อกลับไม่ได้เลย
class _PutawayScreenState extends State<PutawayScreen>
    with WidgetsBindingObserver {

  Timer? _safetyNet;

  @override

    super.initState();
    WidgetsBinding.instance.addObserver(this);
    _signalR.addStationDispatchedListener(_handleStationDispatched);
    ...
    // ชั้น 2 — ต่อกลับแล้วโหลดใหม่ทั้งชุด
    _signalR.addReconnectedListener(_loadStationStatus);
    _initSignalR();
    _loadStationStatus();

    // ชั้น 3 — ตาข่ายกันพลาด ห่างพอที่จะไม่กระทบต้นทุน
    _safetyNet = Timer.periodic(
      const Duration(minutes: 3),
      (_) => _loadStationStatus(),
    );
  }

  @override
  void dispose() {
    _safetyNet?.cancel();
    WidgetsBinding.instance.removeObserver(this);
    _signalR.removeReconnectedListener(_loadStationStatus);
    ...
    super.dispose();
  }

  // ชั้น 1 — กลับมาจากพื้นหลัง
  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    if (state == AppLifecycleState.resumed) _loadStationStatus();
  }
}

สามบรรทัดที่ต้องไม่ลืมใน dispose() — ตัวจับเวลา, observer, และผู้ฟังตัวใหม่

สามนาทีเป็นการแลกที่สมเหตุสมผล: ยิงเพิ่ม 20 ครั้งต่อชั่วโมงต่อเครื่อง แลกกับการที่ข้อมูลไม่มีทางค้างเกินสามนาที

สิ่งที่ควรทำพร้อมกัน — แสดงสถานะการเชื่อมต่อตามหัวข้อ 4 เพราะถ้าไม่แสดง ผู้ใช้จะไม่รู้ว่าทำไมบางทีข้อมูลช้าถึงสามนาที

ข้อสรุป: ปัญหานี้แก้ได้โดยไม่ต้องแตะเรื่องการเชื่อมต่อเลย — เพราะสิ่งที่ผู้ใช้ต้องการคือข้อมูลที่ตรง ไม่ใช่การเชื่อมต่อที่ไม่หลุด และ “โหลดใหม่ทั้งชุด” ให้สิ่งนั้นได้ในทุกกรณี

9. สรุป

  • สองบริการโครงเหมือนกัน ต่างที่ URL และรายชื่อเหตุการณ์ — ตัว sorting เขียนกระชับกว่าด้วยตัวช่วย _on
  • ref-count แก้ปัญหาสองจอที่ใช้ hub เดียวกันปิดทับกัน
  • _refCount++ เกิดก่อนการต่อจริง และไม่ลดเมื่อต่อไม่สำเร็จ
  • disconnect() ไม่จับ exception จาก stop()
  • ต่อไม่สำเร็จ → เขียน log → ไม่มีใครรู้ — ความล้มเหลวเงียบแบบที่ 12.11 อธิบาย
  • มี withAutomaticReconnect() แต่ไม่มี onreconnected — เหตุการณ์ที่พลาดหายถาวร
  • List.of() ก่อนวนกระจาย ป้องกันการแก้รายการระหว่างวน
  • _parseArgs ไม่โยน exception ซึ่งจำเป็นในตัวจัดการเหตุการณ์
  • ทางแก้ที่แท้จริงคือโหลดใหม่ทั้งชุดในสามจังหวะ ไม่ใช่ทำให้การเชื่อมต่อไม่หลุด
