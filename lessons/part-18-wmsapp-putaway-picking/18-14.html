ฟิลด์บอกอะไร

ทำถูกทำไม

4. ส่งไปแพ็กแล้วทำอะไรต่อ

Future<void> _sendToPack(String palletId) async {
  setState(() => _loading = true);
  final result = await _api.sendToPack(palletId: palletId);
  if (!mounted) return;
  setState(() => _loading = false);

  if (!result.success) {
    showErrorDialog(context, message: result.error ?? 'ส่งไป PACK ไม่สำเร็จ');
    return;
  }

  ScaffoldMessenger.of(context).showSnackBar(
    SnackBar(
      content: Text(result.data?['message'] ?? 'ส่งไป PACK สำเร็จ'),
      backgroundColor: AppTheme.success,
    ),
  );

  // กลับไปสแกน source ต่อ (ยังอยู่หน้า Pick)
  setState(() => _destPalletId = null);
  _goToScanSource();
}
ขั้นทำอะไร
1ส่งพาเลทไปแพ็ก
2ล้าง _destPalletId — พาเลทใบนั้นไม่อยู่แล้ว
3กลับไปสแกนต้นทาง

ขั้นที่ 2 สำคัญ — ถ้าไม่ล้าง รอบถัดไปจะข้ามการสแกนปลายทาง (18.11) แล้วส่งของไปพาเลทที่ถูกส่งออกไปแล้ว

Future<void> _returnPallet(String palletId, String destination) async {

7. ข้อผิดพลาดที่พบบ่อย

เข้าใจผิดที่จริง
ปุ่มที่ปิดตัวเองแปลว่ามีกรณีที่ปิดจริงที่นี่เงื่อนไขเป็น true เสมอ
ยืนยันไม่สำเร็จ = ไม่มีอะไรเกิดขึ้นอาจบันทึกไปแล้ว
สร้าง SnackBar เองก็เหมือนกันเสียการสั่นและความสม่ำเสมอ
ปุ่มสองปุ่มที่ห่าง 10 พิกเซลพอผู้ใช้ใส่ถุงมือกดพลาดได้
เมธอดในไฟล์เดียวกันทำเหมือนกันสามทางส่งพาเลทป้องกันสามระดับ

8. แบบฝึกหัด

ผู้ใช้แจ้งว่า “กดส่ง Pallet ผิดปุ่ม ของไป ZONE PACK แทนที่จะกลับ ASRS แก้ไม่ได้เลย”

  1. ทำไมถึงกดผิดได้ง่าย
  2. เสนอการแก้สามระดับ พร้อมผลกระทบต่อความเร็วในการทำงาน
  3. ควรมีทางยกเลิกไหม
แนวคำตอบ

ข้อ 1 — ทำไมกดผิดง่าย

ปัจจัยรายละเอียด
ปุ่มขนาดเท่ากันทั้งคู่ width: double.infinity, vertical: 14
ห่างกัน 10 พิกเซลน้อยกว่าครึ่งของนิ้วมือที่ใส่ถุงมือ
ไม่มีการยืนยันกดปุ๊บส่งปั๊บ
สีต่างกันแต่ไม่ได้สื่อความหมายprimary น้ำเงิน กับ secondary ฟ้า — ใกล้กัน

ปัจจัยที่สี่สำคัญ — สองสีนี้อยู่ในตระกูลเดียวกัน

static const primary   = Color(0xFF1565C0);   // น้ำเงิน
static const secondary = Color(0xFF0288D1);   // ฟ้า

ผู้ใช้ที่กดโดยดูสีจากมุมตาแยกไม่ออก

ข้อ 2 — สามระดับ

ระดับทำอะไรเสียเวลาเพิ่มกันได้แค่ไหน
1เพิ่มระยะห่างเป็น 24 + ใช้สีที่ต่างกันจริงไม่มีกดพลาดน้อยลง
2เพิ่มกล่องยืนยัน1 แตะ ต่อครั้งกันได้เกือบหมด
3ใช้ preview แบบ 18.9 ให้เซิร์ฟเวอร์แนะนำ1 คำขอ + 1 แตะกันได้และแนะนำถูกด้วย
// ระดับ 1 — แก้ที่ widget ได้เลย ไม่กระทบ flow
const SizedBox(height: 24),        // จาก 10

// และแยกสีให้ต่างกันจริง
style: ElevatedButton.styleFrom(
  backgroundColor: AppTheme.primary,     // ASRS = น้ำเงิน (กลับที่เดิม)
  ...
),
style: ElevatedButton.styleFrom(
  backgroundColor: AppTheme.teal,        // ZONE PACK = เขียวน้ำเงิน (ไปต่อ)
  ...
),

  }
}

confirmLabel เขียนปลายทางไว้ในปุ่มด้วย — ผู้ใช้ที่กดเร็วโดยไม่อ่านเนื้อหาก็ยังเห็น ตรงกับที่ 18.9 ทำ

ข้อ 3 — ควรมีทางยกเลิกไหม

คำถามคำตอบ
ยกเลิกได้ทางเทคนิคไหมต้องถามฝั่งเซิร์ฟเวอร์ — มี endpoint ไหม
ยกเลิกได้ทางกายภาพไหมขึ้นกับว่า AMR มาถึงหรือยัง
ถ้าของไปถึง ZONE PACK แล้วต้องให้คนไปเอากลับ ซึ่งเป็นงานนอกระบบ

คำตอบที่ตรงที่สุดคือ “ป้องกันดีกว่ายกเลิก” — เพราะคำสั่งนี้ทำให้ของเคลื่อนที่จริง และมีช่วงเวลาสั้นมากก่อนที่หุ่นยนต์จะมารับ

แต่ถ้าจะมี ควรเป็นแบบที่มีเวลาจำกัด

// แถบที่ให้ยกเลิกได้ในช่วงสั้น ๆ ก่อน AMR มาถึง
ScaffoldMessenger.of(context).showSnackBar(
  SnackBar(
    content: Text('ส่ง $palletId ไป $label แล้ว'),
    backgroundColor: AppTheme.success,
    duration: const Duration(seconds: 8),
    action: SnackBarAction(
      label: 'ยกเลิก',
      textColor: Colors.white,
      onPressed: () => _cancelPalletMove(palletId),
    ),
  ),
);
ก่อนทำต้องยืนยันสามข้อกับทีมและฝ่ายคลัง
  1. มี endpoint ยกเลิกไหม
  2. ยกเลิกได้ภายในกี่วินาทีก่อน AMR ออกเดินทาง
  3. ถ้ายกเลิกไม่ทัน ผู้ใช้ควรทำอย่างไร
ถ้าตอบข้อ 1 ว่าไม่มี การใส่ปุ่มยกเลิกจะกลายเป็นการหลอกผู้ใช้ว่าแก้ได้ ซึ่งแย่กว่าไม่มีปุ่ม

ข้อสรุป: สำหรับคำสั่งที่มีผลทางกายภาพและกลับไม่ได้ ลำดับความสำคัญคือ ทำให้กดผิดยาก → ยืนยันก่อน → แล้วค่อยคิดเรื่องยกเลิก

ระดับ 1 กับ 2 รวมกันใช้เวลาแก้ไม่ถึงชั่วโมงและแก้ปัญหาได้เกือบทั้งหมด — ส่วนการยกเลิกต้องแก้ทั้งสองฝั่งและอาจทำไม่ได้จริง

9. สรุป

  • _confirmPick ตรวจ mounted ก่อน setState และไม่ล้าง S/N เมื่อล้มเหลว
  • แต่ไม่บอกว่าคำสั่งอาจบันทึกไปแล้ว และไม่มี Idempotency-Key
  • เซิร์ฟเวอร์คำนวณ isPickOrderComplete และ remainingItems ให้
  • ปุ่ม “ส่งไป PACK” ปิดตัวเองด้วยเงื่อนไขที่เป็น true เสมอ
  • _sendToPack ล้าง _destPalletId เพื่อไม่ให้รอบถัดไปข้ามการสแกนปลายทาง
  • แต่สร้าง SnackBar เอง — เสียการสั่นไป ทั้งที่เมธอดข้าง ๆ ใช้ตัวกลาง
  • การคืนพาเลทต้นทางให้ผู้ใช้เลือกปลายทางเอง ไม่มี preview ไม่มีการยืนยัน
  • สามทางที่ส่งพาเลทในโดเมนเดียวกัน ป้องกันสามระดับ
  • สำหรับคำสั่งที่กลับไม่ได้ — ทำให้กดผิดยาก แล้วยืนยัน ดีกว่าคิดเรื่องยกเลิก

บทหน้า: สถานะ returnSource เต็ม ๆ และการเลือกปลายทางที่ผู้ใช้ตัดสินเอง
