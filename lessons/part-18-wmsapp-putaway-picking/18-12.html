1. สแกนพาเลทต้นทาง

Future<void> _scanSourcePallet() async {
  final palletId = _sourceScanCtrl.text.trim().toUpperCase();
  if (palletId.isEmpty) return;

  setState(() => _loading = true);
  final result = await _api.assignPickStation(palletId: palletId, pickOrderId: _pickOrderId);
  if (!mounted) return;
  setState(() => _loading = false);

  if (!result.success) {
    setState(() {
      _returnPalletId = palletId;
      _state = _PickState.returnSource;
    });
    return;
  }

  _assignment = result.data!;      // ← ไม่ได้อยู่ใน setState
  _resetPickedSerials();

  setState(() {
    _state = _PickState.pickView;
  });
}

สังเกตสองบรรทัดกลาง — _assignment และการล้างข้อมูลอยู่นอก setState

ทำงานได้ไหมเพราะ
ได้setState ที่ตามมาทำให้วาดใหม่อยู่แล้ว
แต่เป็นรูปแบบที่ควรเลี่ยง — setState ไม่ได้ทำหน้าที่เปลี่ยนค่า มันทำหน้าที่บอก Flutter ว่ามีอะไรเปลี่ยน
การแยกการเปลี่ยนค่าออกจาก setState ทำให้คนอ่านต้องไล่ว่าค่าไหนเปลี่ยนที่ไหนบ้าง และถ้าวันหนึ่งมีคนลบ setState บรรทัดล่างออก จะเกิดบั๊กที่หายาก
setState(() {
  _assignment = result.data!;

2. เลือกสินค้าแล้วเตรียมช่องให้

void _activatePart(String partId) {
  final item = _assignment.palletItems.where((i) => i.partId == partId).firstOrNull;
  if (item == null) return;

  setState(() {
    _selectedPartIds..clear()..add(partId);
    _partScanCtrl.text = partId;                    // ← เติมช่อง Part ให้
    if (!item.serialRequire) {
      final remaining = item.qtyToPickSuggested - (_pickedQty[partId] ?? 0);
      _qtyEntryCtrl.text = remaining > 0 ? '$remaining' : '0';   // ← เติมจำนวนที่เหลือ
    }
  });

  if (item.serialRequire) {
    _serialScanFocus.requestFocus();      // ← ไปช่อง S/N
  } else {
    _qtyEntryFocus.requestFocus();        // ← ไปช่องจำนวน
  }
}
สินค้าเติมอะไรให้โฟกัสไปที่
ต้องมี S/Nรหัสสินค้าช่อง S/N
ไม่ต้องมีรหัสสินค้า + จำนวนที่เหลือช่องจำนวน

แถวที่สองประหยัดเวลาผู้ใช้มาก — ส่วนใหญ่หยิบครบตามที่ระบบขอ จึงกดยืนยันได้เลยโดยไม่ต้องพิมพ์

final remaining = item.qtyToPickSuggested - (_pickedQty[partId] ?? 0);
_qtyEntryCtrl.text = remaining > 0 ? '$remaining' : '0';

คำนวณจากที่ต้องหยิบ ลบที่หยิบไปแล้ว — ผู้ใช้ที่กลับมาเลือกสินค้าเดิมซ้ำจะเห็นจำนวนที่เหลือจริง ไม่ใช่จำนวนเต็ม

remaining > 0 ? ... : '0' กันกรณีที่หยิบเกิน ซึ่งทำให้ได้ค่าติดลบ

การเติมค่าล่วงหน้าเป็นดาบสองคม — ประหยัดเวลาเมื่อค่าที่เดาถูก แต่ทำให้ผู้ใช้กดยืนยันโดยไม่อ่านเมื่อเดาผิด
ที่นี่คุ้ม เพราะค่าที่เดาคือ “จำนวนที่ระบบขอให้หยิบ” ซึ่งเป็นค่าที่ถูกในกรณีส่วนใหญ่ และผู้ใช้ที่หยิบได้ไม่ครบต้องแก้อยู่แล้ว

สังเกต putIfAbsent

final serials = _pickedSerials.putIfAbsent(partId, () => <String>[]);

คืนรายการที่มีอยู่ หรือสร้างใหม่ถ้ายังไม่มี — แล้ว serials.add() ทีหลังแก้รายการใน Map โดยตรง

ทำงานได้เพราะ List เป็นวัตถุอ้างอิง แต่ทำให้ setState ครอบแค่บางส่วน

setState(() {
  _selectedPartIds.add(partId);
  serials.add(serialNo);         // ← แก้รายการใน Map ผ่านตัวอ้างอิง
});

ที่นี่อยู่ใน setState จึงถูกต้อง — แต่ด่านที่ 4 เรียก putIfAbsent ก่อนหน้านั้น ซึ่งอาจเพิ่มคีย์ใหม่เข้า Map โดยไม่มี setState

ผลของการเพิ่มคีย์ว่างนอก setState
ไม่มีผลกับหน้าจอ เพราะรายการว่างไม่แสดงอะไร
แต่ถ้า S/N ถูกปฏิเสธที่ด่าน 5 หรือ 6 คีย์ว่างจะค้างอยู่

ไม่เสียหายในทางปฏิบัติ เพราะ _pickedCountFor คืน 0 สำหรับรายการว่าง

5. กรอกจำนวนสำหรับสินค้าที่ไม่มี S/N

void _addQty() {
  final item = _currentItem;
  if (item == null) {
    showWarningSnackbar(context, 'กรุณาสแกน Part ก่อน');
    _partScanFocus.requestFocus();
    return;
  }

  final qty = int.tryParse(_qtyEntryCtrl.text.trim());
  if (qty == null || qty <= 0) {
    showWarningSnackbar(context, 'กรุณาระบุจำนวน');
    _qtyEntryFocus.requestFocus();
    return;
  }
  if (qty > item.qtyToPickSuggested) {
    showErrorDialog(context,
        message: 'จำนวนเกินที่ต้องการ Pick (สูงสุด ${item.qtyToPickSuggested})');
    return;
  }

  setState(() => _pickedQty[item.partId] = qty);
  showSuccessSnackbar(context, 'Part ${item.partId} ระบุจำนวน $qty ชิ้น');
}
ทำถูกทำไม
int.tryParse ไม่ใช่ int.parseผู้ใช้พิมพ์ตัวอักษรก็ไม่พัง
ตรวจ <= 0กันทั้งศูนย์และค่าติดลบ
ตรวจเกินจำนวนที่ขอบอกค่าสูงสุดในข้อความด้วย
คืนโฟกัสตามช่องที่ผิดผิดที่ Part → กลับช่อง Part, ผิดที่จำนวน → กลับช่องจำนวน

แต่ทางที่สามไม่คืนโฟกัส

if (qty > item.qtyToPickSuggested) {
  showErrorDialog(context, message: '...');
  return;                       // ← ไม่คืนโฟกัส
}

ผู้ใช้ที่กรอกเกินต้องแตะช่องเองก่อนแก้ — จุดเดียวกับที่ 18.10 พบ

และสังเกตว่าใช้ = ไม่ใช่ +=

_pickedQty[item.partId] = qty;      // ← แทนที่ ไม่ใช่บวกเพิ่ม

ผู้ใช้ที่กด “เพิ่มจำนวน” สองครั้งจะได้ค่าครั้งหลัง ไม่ใช่ผลรวม — ซึ่งตรงกับที่ _activatePart เติมจำนวนที่เหลือให้

final remaining = item.qtyToPickSuggested - (_pickedQty[partId] ?? 0);
สองบรรทัดนี้ขัดกัน — _activatePart คำนวณจำนวนที่เหลือโดยหักที่หยิบไปแล้ว แต่ _addQty แทนที่ค่าเดิมทั้งหมด
ต้องหยิบ 10 ชิ้น
  → เลือกสินค้า → ช่องเติม "10" → กดเพิ่ม → _pickedQty = 10
    → เลือกซ้ำ → remaining = 10 - 10 = 0 → ช่องเติม "0"
      → กดเพิ่ม → "กรุณาระบุจำนวน" (เพราะ 0 ไม่ผ่านด่าน)
ผู้ใช้ที่อยากแก้จำนวนต้องพิมพ์เองทั้งหมด — และค่าที่เติมให้ (0) ไม่ช่วยอะไร

ถ้าตั้งใจให้แทนที่ ควรเติมค่าปัจจุบันแทนที่เหลือ

// เติมค่าที่ระบุไว้แล้ว หรือจำนวนที่ขอถ้ายังไม่เคยระบุ
final current = _pickedQty[partId];
_qtyEntryCtrl.text = '${current ?? item.qtyToPickSuggested}';

ประโยชน์ความเสี่ยง
ทดสอบ flow ได้โดยไม่ต้องมีของจริงผู้ใช้ยืนยันการหยิบได้โดยไม่ได้หยิบจริง

สแกน S/N ต้องยิงเน็ตเสมอสองในสามจอในโปรเจกต์นี้ไม่ต้อง
putIfAbsent ต้องอยู่ใน setStateที่นี่ไม่จำเป็น เพราะรายการว่างไม่แสดงอะไร
เติมค่าล่วงหน้าดีเสมอทำให้ผู้ใช้กดยืนยันโดยไม่อ่านเมื่อเดาผิด
เครื่องมือทดสอบไม่อยู่ใน releaseอยู่ ถ้าไม่ครอบด้วย kDebugMode

8. แบบฝึกหัด

ผู้ใช้แจ้งว่า “กรอกจำนวนไป 8 แล้วอยากแก้เป็น 5 แต่พอแตะการ์ดซ้ำ ช่องขึ้น 2 ไม่ใช่ 8”

  1. อธิบายว่าเลข 2 มาจากไหน
  2. อะไรคือพฤติกรรมที่ผู้ใช้คาดหวัง
  3. แก้อย่างไร และมีทางเลือกอื่นไหม
แนวคำตอบ

ข้อ 1 — เลข 2 มาจากไหน

ต้องหยิบ 10 ชิ้น (qtyToPickSuggested = 10)
  → เลือกสินค้า → remaining = 10 - 0 = 10 → ช่องขึ้น "10"
    → ผู้ใช้แก้เป็น 8 แล้วกดเพิ่ม → _pickedQty[partId] = 8
      → แตะการ์ดซ้ำเพื่อแก้
        → _activatePart → remaining = 10 - 8 = 2 → ช่องขึ้น "2"

เลข 2 คือจำนวนที่ยังขาด ไม่ใช่จำนวนที่กรอกไว้

สับสนเพราะสองเมธอดตีความ _pickedQty คนละแบบ

เมธอดตีความ _pickedQty ว่า
_activatePartยอดสะสม — จึงหักออกเพื่อหาที่เหลือ
_addQtyค่าปัจจุบัน — จึงแทนที่ทั้งหมด

ถ้า _addQty เป็น += การหักออกจะถูก — แต่ตอนนี้เป็น =

ข้อ 2 — ผู้ใช้คาดหวังอะไร

ผู้ใช้พูดว่าแปลว่าเขาคิดว่า
“กรอกจำนวนไป 8”ตัวเลข 8 คือค่าที่บันทึกไว้
“อยากแก้เป็น 5”ต้องการแทนที่ 8 ด้วย 5
“ช่องควรขึ้น 8”เพื่อจะได้เห็นว่าตอนนี้เป็นเท่าไรก่อนแก้

ผู้ใช้คิดแบบ “ค่าปัจจุบัน” ซึ่งตรงกับที่ _addQty ทำ — _activatePart คือตัวที่ไม่ตรง

ข้อ 3 — แก้

void _activatePart(String partId) {
  final item = _assignment.palletItems.where((i) => i.partId == partId).firstOrNull;
  if (item == null) return;

  setState(() {
    _selectedPartIds..clear()..add(partId);
    _partScanCtrl.text = partId;
    if (!item.serialRequire) {
      // ระบุไว้แล้ว → แสดงค่าเดิมให้แก้ / ยังไม่เคยระบุ → เติมจำนวนที่ระบบขอ
      final current = _pickedQty[partId];
      _qtyEntryCtrl.text = '${current ?? item.qtyToPickSuggested}';
    }
  });
  ...
}

แถวสุดท้ายแก้ปัญหาที่หัวข้อ 5 ชี้ไว้ด้วย — เดิมผู้ใช้ที่กรอกครบแล้วจะแก้ไม่ได้เลย

ทางเลือกอื่น

ทางข้อดีข้อเสีย
เปลี่ยน _addQty เป็น +=_activatePart เดิมถูกทันทีผู้ใช้แก้ค่าลงไม่ได้ — บวกได้อย่างเดียว
เปลี่ยน _activatePart (ข้างบน)ตรงกับที่ผู้ใช้คิด แก้ได้ทั้งขึ้นและลงต้องรู้ว่ามีค่าเดิมอยู่
ปุ่มบวกลบแทนช่องกรอกผิดยาก ไม่ต้องพิมพ์กด 8 ครั้งเพื่อได้ 8

ทางที่สองดีที่สุดเพราะแก้ตรงจุดและไม่เปลี่ยนวิธีใช้งาน

ทางที่สามน่าสนใจถ้าจำนวนมักน้อย — ScanPartScreen มี _buildManualQtyEditor ที่ทำแบบนี้ (17.5) พร้อมทั้งปุ่มบวกลบและช่องกรอก

สิ่งที่ควรทำพร้อมกัน: ปุ่มที่ใช้เรียก _addQty ควรเปลี่ยนคำ
ถ้าพฤติกรรมคือ “แทนที่” ปุ่มไม่ควรเขียนว่า “เพิ่ม” — ควรเป็น “บันทึกจำนวน” หรือ “ระบุจำนวน”
ข้อความสำเร็จก็เขียนถูกอยู่แล้ว: 'Part ${item.partId} ระบุจำนวน $qty ชิ้น' ใช้คำว่า “ระบุ” ไม่ใช่ “เพิ่ม”

ข้อสรุป: บั๊กนี้เกิดจากสองเมธอดที่ตีความข้อมูลเดียวกันคนละแบบ — และวิธีจับคือถามว่า “ค่านี้หมายถึงอะไร” แล้วตรวจว่าทุกที่ที่แตะมันตอบเหมือนกันไหม

9. สรุป

  • _assignment ถูกแก้นอก setState — ทำงานได้แต่ทำให้ตามยาก
  • _activatePart เติมรหัสสินค้าและจำนวนให้ แล้วย้ายโฟกัสตามชนิดสินค้า
  • มีสองทางยกเลิกการเลือก และทางหนึ่งมองเห็นได้ — สิ่งที่ ScanPartScreen ขาด
  • _scanSerial() ไม่ใช่ async เลย — หกด่านตรวจในเครื่องทั้งหมด
  • รายชื่อ S/N มากับ availableSerials ตั้งแต่ตอนจัดสถานี
  • สองในสามจอที่สแกน S/N ในโปรเจกต์นี้ไม่ต้องยิงเน็ต
  • _addQty แทนที่ค่าเดิม แต่ _activatePart คำนวณจำนวนที่เหลือ — สองเมธอดตีความข้อมูลเดียวกันคนละแบบ
  • แผ่น “S/N สำหรับทดสอบ” แสดงรายชื่อ S/N ทั้งหมดและติดมากับ release
  • เครื่องมือที่ทำให้ข้ามการสแกนจริงได้ ควรครอบด้วย kDebugMode หรือผูกกับบทบาท
