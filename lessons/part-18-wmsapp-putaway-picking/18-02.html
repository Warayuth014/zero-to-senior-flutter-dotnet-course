Future<void> _loadStationStatus() async {
  final result = await _api.getStationStatus();
  if (!mounted || !result.success) return;

  final map = <String, Map<String, dynamic>>{};
  for (final station in result.data!) {
    map[station['stationId'] as String] = station;
  }
  setState(() => _stationStatus = map);
}

Future<ApiResult<List<Map<String, dynamic>>>> getStationStatus() async {
  final response = await _get('/putaway/station-status');
  if (!response.success) return ApiResult.error(response.error);
  final items = response.data!['items'] as List;
  return ApiResult.success(items.cast<Map<String, dynamic>>());

4. แอนิเมชันที่ต้องรู้ว่าค่าเปลี่ยนจากภายนอก

class _StationCardState extends State<StationCard>
    with SingleTickerProviderStateMixin {
  late final AnimationController _ctrl;

  @override
  void initState() {
    super.initState();
    _ctrl = AnimationController(vsync: this, duration: const Duration(milliseconds: 800));
    if (widget.isDispatching) _ctrl.repeat();      // ← เริ่มถ้าตอนสร้างกำลังทำงานอยู่
  }

  @override
  void didUpdateWidget(StationCard old) {
    super.didUpdateWidget(old);
    if (widget.isDispatching && !old.isDispatching) {
      _ctrl.repeat();                              // ← เพิ่งเริ่มทำงาน
    } else if (!widget.isDispatching && old.isDispatching) {
      _ctrl.stop();
      _ctrl.reset();                               // ← เพิ่งหยุด
    }
  }

  @override
  void dispose() {
    _ctrl.dispose();
    super.dispose();
  }
}

didUpdateWidget ถูกเรียกเมื่อพ่อสร้าง widget ตัวใหม่ที่ตำแหน่งเดิม ซึ่งเกิดทุกครั้งที่จอ setState

เมธอดถูกเรียกเมื่อในจอนี้คือ
initStateสร้าง State ครั้งแรกเปิดจอ
didUpdateWidgetได้ค่าใหม่จากพ่อSignalR แจ้งว่าสถานีเปลี่ยนสถานะ
disposeState ถูกถอดออกจากจอ

ทำไมต้องเทียบกับค่าเก่า

// ถ้าเขียนแบบนี้ จะเรียก repeat() ซ้ำทุกครั้งที่จอ setState
if (widget.isDispatching) _ctrl.repeat();

repeat() ซ้ำ ๆ ทำให้แอนิเมชันกระตุกเพราะถูกรีสตาร์ทกลางทาง — การเทียบ old.isDispatching ทำให้ทำงานเฉพาะตอนที่เปลี่ยนจริง

นี่คือรูปแบบมาตรฐานที่ 4.15 สอน และทำได้ครบทั้งสี่จุด — สร้าง, เริ่มถ้าจำเป็น, ตอบสนองการเปลี่ยน, และคืนทรัพยากร
SingleTickerProviderStateMixin จำเป็นเพราะ AnimationController ต้องมีตัวจับจังหวะเฟรม — ถ้ามีหลาย controller ต้องใช้ TickerProviderStateMixin แทน

สิ่งที่ต้องระวัง: _ctrl.repeat() วนตลอดเวลาที่สถานีทำงานอยู่ ซึ่งอาจนานหลายนาที

วาดใหม่ 60 ครั้งต่อวินาทีเฉพาะส่วนที่ผูกกับ _ctrl
กินแบตต่อเนื่องสามสถานีทำงานพร้อมกัน = สามตัว

class _StationCardState extends State<StationCard>
    with SingleTickerProviderStateMixin, WidgetsBindingObserver {

  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addObserver(this);
    _ctrl = AnimationController(vsync: this, duration: const Duration(milliseconds: 800));
    if (widget.isDispatching) _ctrl.repeat();
  }

  @override
  void dispose() {
    WidgetsBinding.instance.removeObserver(this);
    _ctrl.dispose();
    super.dispose();
  }

  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    // อยู่ในกระเป๋า = ไม่มีใครเห็น ไม่ต้องวาด
    if (state == AppLifecycleState.resumed) {
      if (widget.isDispatching) _ctrl.repeat();
    } else {
      _ctrl.stop();
    }
  }
}
แต่มีทางที่ง่ายกว่านั้นมาก — Flutter หยุดวาดเฟรมเองอยู่แล้วเมื่อแอปไปอยู่พื้นหลัง ดังนั้นแอนิเมชันที่วนอยู่จะไม่กินเฟรมจริง
สิ่งที่ยังกินคือ Ticker ที่ยังลงทะเบียนอยู่ ซึ่งเบามาก
ก่อนแก้ ควรวัดก่อนตามที่ 14.13 สอน — ถ้าวัดแล้วไม่ต่าง การเพิ่ม observer 5 ตัวคือความซับซ้อนที่ไม่ได้อะไร

ข้อสรุป: ข้อ 1 กับ 2 ต้องแก้แน่นอนเพราะพังจริงและเห็นได้ ส่วนข้อ 3 เป็นสิ่งที่ต้องวัดก่อนตัดสิน — ไม่ใช่ทุกข้อสังเกตจะเป็นปัญหาที่ต้องแก้

9. สรุป

  • จอ 168 บรรทัด state ตัวเดียว เก็บสถานะทุกสถานีเป็น Map ที่มีรหัสเป็นคีย์
  • สถานะเป็น Map<String, dynamic> ดิบ ไม่มีโมเดล — พิมพ์คีย์ผิดรู้ตอนรัน
  • PutawayStationGrid เป็น Row ที่รองรับแค่แถวเดียว
  • ไฟล์ widget ทั้งสองไม่รู้จัก ApiService — แตกถูกทิศ
  • การ์ดเปลี่ยนสีเป็นแดงเมื่อสถานีทำงาน และไล่ระดับด้วย AnimatedContainer
  • didUpdateWidget เทียบกับค่าเก่าก่อนสั่ง repeat() — ไม่งั้นแอนิเมชันกระตุก
  • สถานีที่ทำงานอยู่กดไม่ได้ แต่ไม่มีอะไรบอกผู้ใช้
  • ข้อความ “Supported stations” พิมพ์รายชื่อซ้ำในสามจอ
