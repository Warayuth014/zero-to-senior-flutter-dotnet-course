ระดับ: กลาง–สูงใช้เวลา: 35–45 นาทีลงมือทำ: กันไม่ให้ผลลัพธ์เก่าเขียนทับข้อมูลล่าสุด
ต้องผ่านอะไรมาก่อน: 4.9 (สิ่งที่ mounted ตอบไม่ได้) และ 4.7 (สถานะการโหลด)
จบบทนี้แล้วจะทำได้: ใช้ตัวนับรุ่นกันผลลัพธ์ล้าสมัย และเขียนเทสต์ที่บังคับลำดับการตอบกลับได้

1. ปัญหาที่บทนี้แก้

ผู้ใช้สแกนพาเลท PAL-0001 แล้วรู้ว่าสแกนผิด จึงสแกน PAL-0002 ทันที

0.0s  สแกน PAL-0001 → ยิงคำขอ A
0.3s  สแกน PAL-0002 → ยิงคำขอ B
0.9s  B ตอบกลับ     → จอแสดงข้อมูล PAL-0002   ✓ ถูกต้อง
1.6s  A ตอบกลับ     → จอแสดงข้อมูล PAL-0001   ✗ ผิด

จอค้างอยู่ที่ข้อมูลของพาเลทที่สแกนผิด และ ไม่มีอะไรบอกว่าผิด — ผู้ใช้เห็นข้อมูลบนจอ ก็เชื่อว่านั่นคือของที่ถืออยู่ในมือ แล้วทำงานต่อ

บนคลังจริงนี่คือของขึ้นแร็คผิดช่อง

สิ่งที่ทำให้บั๊กนี้ร้ายคือ มันไม่พัง ไม่มี exception ไม่มี log และเกิดเฉพาะเมื่อเน็ตช้าไม่เท่ากันในสองคำขอ ซึ่งบนเครื่องพัฒนาแทบไม่เกิดเลย

3. Mental model: คำขอไม่ได้ต่อคิว

เรามักเผลอคิดว่าคำขอที่ยิงก่อนจะกลับก่อน ซึ่งไม่จริงเลย

สิ่งที่คิด          สิ่งที่เกิดจริง
A ─────→            A ─────────────→
      B ───→              B ──→
A กลับ B กลับ         B กลับ  A กลับ

เวลาตอบกลับขึ้นกับขนาดข้อมูล เส้นทางเครือข่าย ภาระของ server และสัญญาณ Wi-Fi ในคลัง — ไม่มีอะไรรับประกันลำดับ

ประโยคเดียวที่ต้องจำ: ก่อนจะเขียนผลลัพธ์ลง state ต้องถามก่อนว่า “ผลนี้ยังเป็นของคำขอล่าสุดอยู่ไหม” — ถ้าไม่ ให้ทิ้งไปเงียบ ๆ

4. ตัวนับรุ่น

เปิด lib/state_basics/pick_store.dart

int _generation = 0;

Future<void> load() async {
  final generation = ++_generation;      // 1. เพิ่มเลข และจำเลขของตัวเอง
  _state = LoadState.loading;
  _errorMessage = null;
  notifyListeners();
  try {
    final result = await repository.fetchLines();
    if (generation != _generation) return;   // 2. ล้าสมัยแล้ว ทิ้งเลย
    _lines = result;
    _state = result.isEmpty ? LoadState.empty : LoadState.ready;
  } on PickException catch (error) {
    if (generation != _generation) return;   // 3. error ก็ต้องตรวจเหมือนกัน
    _state = LoadState.failed;
    _errorMessage = error.message;
  } finally {
    if (generation == _generation) notifyListeners();   // 4. แจ้งเฉพาะรุ่นล่าสุด
  }
}

ไล่ตามสถานการณ์เดิม

load() ครั้งแรก : ++_generation → 1, generation ในตัวมันเอง = 1
load() ครั้งที่สอง: ++_generation → 2, generation ในตัวมันเอง = 2

ผลของครั้งที่สองกลับมา : 2 == 2  → เขียนลง state ✓
ผลของครั้งแรกกลับมา    : 1 != 2  → return ทิ้งไป ✓

generation เป็นตัวแปรท้องถิ่นในแต่ละครั้งที่เรียก จึงไม่ถูกใครแก้ ส่วน _generation เป็นของ store จึงสะท้อนว่าตอนนี้รุ่นล่าสุดคือเลขอะไร

5. สี่จุดที่ต้องตรวจ ไม่ใช่จุดเดียว

คนส่วนใหญ่ใส่การตรวจแค่จุดเดียวคือหลัง await แล้วก็ยังเจอบั๊กอยู่ดี

จุดถ้าไม่ตรวจจะเกิดอะไร
หลัง await ในทางสำเร็จข้อมูลเก่าทับข้อมูลใหม่ — ตัวปัญหาหลัก
ใน catchคำขอเก่าล้มเหลว → จอขึ้นหน้าผิดพลาด ทั้งที่คำขอใหม่สำเร็จไปแล้ว
ใน finally ก่อน notifyListeners()แจ้งโดยไม่มีอะไรเปลี่ยน — วาดใหม่เปล่า ๆ
ใน dispose()ผลที่ค้างอยู่กลับมาแตะ store ที่ตายแล้ว

จุดที่สองสำคัญและมองข้ามบ่อยที่สุด — ลองนึกภาพ

0.0s  โหลดครั้งแรก (สัญญาณแย่)
0.5s  ผู้ใช้กดรีเฟรช → โหลดครั้งที่สอง
1.0s  ครั้งที่สองสำเร็จ  → จอแสดงรายการครบ ✓
3.0s  ครั้งแรกหมดเวลารอ → จอเปลี่ยนเป็น "โหลดไม่สำเร็จ" ✗

ผู้ใช้เห็นรายการอยู่ดี ๆ แล้วหน้าจอกลายเป็นหน้าผิดพลาดเอง โดยไม่ได้ทำอะไรเลย

ส่วนจุดที่สี่คือบรรทัดที่เจอใน 4.9

@override
void dispose() {
  stopPolling();
  _generation++;      // ทุกคำขอที่ยังค้างอยู่ กลายเป็นล้าสมัยทันที
  super.dispose();
}

กรณีปัญหา

โหลดใหม่โดยไม่มีคำค้น เช่นกดรีเฟรชไม่มีค่าอะไรให้เทียบเลย

8. เมื่อหลายคำขอต้องอยู่ร่วมกันได้

ตัวนับรุ่นตัวเดียวหมายความว่า คำขอใหม่ยกเลิกความถูกต้องของทุกคำขอเก่า ซึ่งถูกต้องเมื่อทุกคำขอเขียนลงที่เดียวกัน

แต่ถ้าโหลดรายละเอียดของหลายพาเลทพร้อมกัน แต่ละอันเขียนคนละที่ ตัวนับตัวเดียวจะทำให้อันที่ยิงก่อนถูกทิ้งทั้งที่ยังต้องการ

final Map<String, int> _generations = <String, int>{};

Future<void> loadDetail(String palletCode) async {
  final generation = (_generations[palletCode] ?? 0) + 1;
  _generations[palletCode] = generation;

  final result = await repository.fetchDetail(palletCode);
  if (_generations[palletCode] != generation) return;   // เทียบเฉพาะ key ตัวเอง
  _details[palletCode] = result;
  notifyListeners();
}
รูปแบบใช้เมื่อ
ตัวนับเดียวทุกคำขอเขียนลง state ก้อนเดียวกัน (กรณีส่วนใหญ่)
ตัวนับต่อ keyแต่ละคำขอเขียนคนละที่ และต้องอยู่ร่วมกันได้

อย่าเพิ่งใช้แบบต่อ key ถ้ายังไม่มีความต้องการจริง — มันซับซ้อนกว่าและต้องดูแล map เพิ่ม

9. ทดสอบด้วยการบังคับลำดับ

บั๊กชนิดนี้ทดสอบไม่ได้ด้วยการรอเฉย ๆ เพราะลำดับควบคุมไม่ได้ ต้องใช้ Completer ซึ่งให้เราสั่งเองว่า Future จะเสร็จเมื่อไหร่

class FakePickRepository implements PickRepository {
  Completer<List<PickLine>>? fetchGate;   // ประตูที่เราเปิดเอง

  @override
  Future<List<PickLine>> fetchLines() {
    final gate = fetchGate;
    if (gate != null) return gate.future;   // รอจนกว่าเทสต์จะสั่งให้ผ่าน
    return Future.value(lines);
  }
}
test('ผลของคำขอเก่าที่มาช้าต้องไม่ทับคำขอใหม่', () async {
  final firstGate = Completer<List<PickLine>>();
  repository.fetchGate = firstGate;
  final firstLoad = store.load();               // คำขอ A ค้างอยู่ที่ประตู

  final secondGate = Completer<List<PickLine>>();
  repository.fetchGate = secondGate;
  final secondLoad = store.load();              // คำขอ B ค้างอยู่ที่ประตู

  secondGate.complete([lineB]);                 // ปล่อย B ก่อน
  await secondLoad;
  expect(store.lines.single.id, lineB.id);

  firstGate.complete([lineA]);                  // แล้วค่อยปล่อย A
  await firstLoad;
  expect(store.lines.single.id, lineB.id);      // ต้องยังเป็น B
});

บรรทัดสุดท้ายคือหัวใจ — ถ้าลบการตรวจ generation ออกจาก load() เทสต์นี้จะแดงทันที

ลองเอง: เปิด pick_store.dart ลบบรรทัด if (generation != _generation) return; ในทางสำเร็จออก แล้วรัน flutter test --no-pub — จะเห็นว่าเทสต์นี้จับได้ทันที นี่คือความต่างระหว่าง “เขียนเทสต์ไว้” กับ “มีเทสต์ที่กันบั๊กได้จริง”

10. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
จอแสดงข้อมูลของสิ่งที่เลือกก่อนหน้าผลเก่ามาทีหลังแล้วทับตัวนับรุ่น
จอกลายเป็นหน้าผิดพลาดเองทั้งที่โหลดสำเร็จแล้วไม่ตรวจรุ่นใน catchตรวจในทาง error ด้วย
ตัวหมุนหายทั้งที่ยังโหลดอยู่คำขอเก่าตั้งสถานะให้คำขอใหม่ตรวจรุ่นก่อนแตะสถานะ
เทียบ query == controller.text แล้วยังพลาดค่าซ้ำหรือถูกแปลงรูปใช้ตัวนับรุ่นแทน
ปิดหน้าจอแล้วยังมีอะไรมาแตะ storeไม่เพิ่มรุ่นใน dispose()ใส่ _generation++
โหลดหลายรายการพร้อมกันแล้วได้แค่อันสุดท้ายใช้ตัวนับเดียวกับงานที่แยกกันตัวนับต่อ key
บั๊กเกิดเฉพาะที่หน้างาน ทำซ้ำไม่ได้ไม่มีเทสต์ที่บังคับลำดับได้Completer

11. แบบฝึกหัด

ช่องค้นหาที่ยิงคำขอทุกครั้งที่พิมพ์

Future<void> search(String query) async {
  _searching = true;
  notifyListeners();
  try {
    _results = await repository.search(query);
  } on SearchException catch (error) {
    _errorMessage = error.message;
  } finally {
    _searching = false;
    notifyListeners();
  }
}

ผู้ใช้พิมพ์ p → pa → pal → pall เร็ว ๆ

  1. อธิบายว่าอาจเกิดอะไรได้บ้าง
  2. แก้ด้วยตัวนับรุ่น
  3. อธิบายว่ายังควรเพิ่มอะไรอีก และตัวนับรุ่นเพียงพอหรือไม่
ดูแนวคำตอบและเหตุผล

ข้อ 1 — อาจเกิดอะไรได้บ้าง

ยิงไปสี่คำขอ และไม่มีอะไรรับประกันว่าจะกลับตามลำดับ

#อาการสาเหตุ
1พิมพ์ pall อยู่ แต่เห็นผลของ paผลของคำขอเก่ามาทีหลังแล้วทับ
2ผลกระพริบเปลี่ยนไปมาหลายรอบสี่คำขอทยอยกลับ แต่ละอันเขียนทับ
3ตัวหมุนหายทั้งที่ยังค้นอยู่_searching = false ของคำขอแรกไปปิดของคำขอที่สี่
4ขึ้นข้อความผิดพลาดทั้งที่ผลล่าสุดสำเร็จคำขอเก่าล้มเหลวแล้วเขียน _errorMessage
5ยิงคำขอสี่ครั้งทั้งที่ต้องการผลอันเดียวไม่มี debounce

ข้อ 2 — แก้ด้วยตัวนับรุ่น

int _searchGeneration = 0;

Future<void> search(String query) async {
  final generation = ++_searchGeneration;
  _searching = true;
  _errorMessage = null;
  notifyListeners();
  try {
    final result = await repository.search(query);
    if (generation != _searchGeneration) return;      // แก้ข้อ 1, 2
    _results = result;
  } on SearchException catch (error) {
    if (generation != _searchGeneration) return;      // แก้ข้อ 4
    _errorMessage = error.message;
  } finally {
    if (generation == _searchGeneration) {            // แก้ข้อ 3
      _searching = false;
      notifyListeners();
    }
  }
}

ข้อ 3 คือจุดที่ละเอียดที่สุด — ถ้าปล่อยให้ _searching = false อยู่นอกการตรวจ คำขอเก่าที่จบก่อนจะไปปิดตัวหมุนของคำขอใหม่ที่ยังวิ่งอยู่ ผู้ใช้จะเห็นผลเก่านิ่งอยู่โดยไม่มีอะไรบอกว่ายังค้นอยู่

ข้อ 3 — ยังควรเพิ่มอะไรอีก

ตัวนับรุ่นทำให้ ถูกต้อง แล้ว แต่ยัง ไม่ดี เพราะข้อ 5 ยังอยู่ — ยังยิงสี่คำขอเพื่อเอาผลอันเดียว

เพิ่ม debounce — รอให้หยุดพิมพ์สักครู่ก่อนค่อยยิง

Timer? _debounce;

void onQueryChanged(String query) {
  _debounce?.cancel();
  _debounce = Timer(const Duration(milliseconds: 300), () => search(query));
}

@override
void dispose() {
  _debounce?.cancel();      // อย่าลืม (4.11)
  _searchGeneration++;
  super.dispose();
}

เพิ่มความยาวขั้นต่ำ — ค้นด้วยตัวอักษรเดียวได้ผลเป็นพันแถวและไม่มีประโยชน์

if (query.trim().length < 2) {
  _results = const [];
  notifyListeners();
  return;
}

debounce แทนตัวนับรุ่นได้ไหม — ไม่ได้

นี่คือคำถามสำคัญของข้อนี้

debounceตัวนับรุ่น
แก้อะไรลดจำนวนคำขอกันผลเก่าทับผลใหม่

debounce เป็นเรื่อง ประสิทธิภาพ ตัวนับรุ่นเป็นเรื่อง ความถูกต้อง — ต้องมีทั้งคู่ และถ้าเลือกได้แค่อย่างเดียว ต้องเลือกตัวนับรุ่น

12. สรุปท้ายบท

  • คำขอไม่ได้กลับตามลำดับที่ยิงไป
  • ผลเก่าที่มาทีหลังจะเขียนทับผลใหม่โดยไม่มีอะไรพัง ซึ่งอันตรายกว่าการพัง
  • ตัวนับรุ่นใช้สามบรรทัด — เพิ่มเลข จำเลข เทียบเลขก่อนเขียน
  • ต้องตรวจสี่จุด: ทางสำเร็จ ทาง error ก่อน notifyListeners() และใน dispose()
  • อย่าเทียบค่าที่ผู้ใช้พิมพ์แทนเลขรุ่น
  • ตัวนับรุ่นคุมความถูกต้อง การยกเลิกคุมความสิ้นเปลือง — ทำอย่างแรกก่อน
  • ใช้ตัวนับต่อ key เมื่อหลายคำขอต้องอยู่ร่วมกันได้
  • Completer ทำให้ทดสอบลำดับการตอบกลับได้จริง
