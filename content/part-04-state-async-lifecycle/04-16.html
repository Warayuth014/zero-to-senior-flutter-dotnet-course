3. ทุกบรรทัดใน markDone มาจากบทไหน

ก่อนเขียนของตัวเอง ให้ไล่ดูว่า markDone คือ Part 4 ทั้งหมดย่อลงมาอยู่ในฟังก์ชันเดียว

Future<void> markDone(String lineId) async {
  if (_busyLineIds.contains(lineId)) return;              // 4.14 กันกดซ้ำ
  final index = _lines.indexWhere((line) => line.id == lineId);
  if (index == -1) return;                                // 4.8  ตรวจก่อนล็อก
  final before = _lines[index];                           // 4.13 เก็บสำเนา
  if (before.done) return;                                // 4.2  อย่าทำซ้ำ

  final commandId =
      _pendingCommandIds.putIfAbsent(lineId, _createCommandId);   // 4.14
  _busyLineIds.add(lineId);                               // 4.13 ล็อกรายบรรทัด
  _errorMessage = null;                                   // 4.7  ล้างของเก่า
  _setLine(index, before.copyWith(done: true));           // 4.13 optimistic
  notifyListeners();                                      // 4.5

  try {
    await repository.markDone(lineId, commandId: commandId);
    _pendingCommandIds.remove(lineId);                    // 4.14 สำเร็จ → ล้าง
  } on PickException catch (error) {                      // 4.8  จับเฉพาะชนิดนี้
    if (error.outcomeUnknown) {                           // 4.13 ไม่รู้ผล
      _errorMessage = '${error.message} กำลังตรวจสอบสถานะล่าสุด';
      notifyListeners();
      await load();                                       // 4.13 ถามความจริง
      return;                                             // 4.14 คงรหัสเดิมไว้
    }
    _rollback(lineId, before);                            // 4.13 ย้อนด้วย id
    _pendingCommandIds.remove(lineId);                    // 4.14 → รหัสใหม่
    _errorMessage = error.message;
  } finally {
    _busyLineIds.remove(lineId);                          // 4.8  ปลดเสมอ
    notifyListeners();
  }
}

6. โครงที่แนะนำ

// lib/state_lab/cycle_count_repository.dart
class CountLine {
  const CountLine({
    required this.id,
    required this.productCode,
    required this.expected,
    this.counted,
  });

  final String id;
  final String productCode;
  final int expected;
  final int? counted;          // null = ยังไม่ได้นับ ต่างจาก 0 = นับได้ศูนย์

  bool get isCounted => counted != null;
  int get variance => (counted ?? 0) - expected;

  CountLine copyWith({int? counted, int? expected}) => CountLine(
    id: id, productCode: productCode,
    expected: expected ?? this.expected,
    counted: counted ?? this.counted,
  );
}

class CountException implements Exception {
  const CountException(this.message, {this.outcomeUnknown = false});
  final String message;
  final bool outcomeUnknown;
}

abstract interface class CycleCountRepository {
  Future<List<CountLine>> fetchLines(String rackCode);
  Future<void> submit(String rackCode, List<CountLine> lines,
      {required String commandId});
}
copyWith ข้างบนมีข้อจำกัดที่ต้องรู้ตัว — มันตั้ง counted กลับเป็น null ไม่ได้ เพราะ counted ?? this.counted แปลว่า "ถ้าไม่ส่งมาก็ใช้ของเดิม" ถ้าต้องรองรับการล้างยอด ต้องใช้ค่ายามพิเศษหรือแยกเมธอด clearCounted() ออกมา ซึ่งเป็นกับดักคลาสสิกของ copyWith กับ field ที่เป็น null ได้

ส่วนสถานะของการยืนยัน แนะนำให้ใช้ sealed class ตาม 4.15

sealed class SubmitState { const SubmitState(); }
class NotSubmitted extends SubmitState { const NotSubmitted(); }
class Submitting extends SubmitState {
  const Submitting(this.commandId);
  final String commandId;
}
class SubmitFailed extends SubmitState {
  const SubmitFailed(this.message, {this.canRetry = true});
  final String message;
  final bool canRetry;
}
class Submitted extends SubmitState { const Submitted(); }

7. เกณฑ์ข้อ 7: รวมข้อมูลใหม่โดยไม่ทับของผู้ใช้

ปัญหาคือ load() คืนรายการจากเซิร์ฟเวอร์ ซึ่งมี counted เป็น null ทั้งหมด — ถ้าเอามาแทนตรง ๆ ยอดที่ผู้ใช้นับมาสิบบรรทัดจะหายทันที

List<CountLine> _merge(List<CountLine> fromServer) {
  final myCounts = {
    for (final line in _lines)
      if (line.isCounted) line.id: line.counted!,
  };

  return [
    for (final line in fromServer)
      myCounts.containsKey(line.id)
          ? line.copyWith(counted: myCounts[line.id])   // เก็บของผู้ใช้ไว้
          : line,                                        // ของใหม่จาก server
  ];
}
กรณีผลลัพธ์
บรรทัดเดิม ผู้ใช้นับแล้วใช้ expected ใหม่ + counted ของผู้ใช้
บรรทัดเดิม ยังไม่ได้นับใช้ของเซิร์ฟเวอร์ทั้งหมด
บรรทัดใหม่ที่เพิ่งเพิ่มเข้ามาตามปกติ
บรรทัดที่ถูกลบไปแล้วหายไปพร้อมยอดที่นับ — ต้องแจ้งผู้ใช้

แถวสุดท้ายเป็นกรณีที่ต้องตัดสินใจ ไม่ใช่ปล่อยให้เกิดเอง — ถ้าผู้ใช้นับบรรทัดหนึ่งไปแล้วแต่คนอื่นลบทิ้ง การให้มันหายไปเงียบ ๆ คือการทิ้งงานของเขาโดยไม่บอก

11. แบบทดสอบตัวเองก่อนจบ Part 4

ตอบให้ได้โดยไม่เปิดเฉลย ถ้าข้อไหนตอบไม่ได้ ให้กลับไปอ่านบทที่วงเล็บไว้

  1. state ควรอยู่ที่ widget, ValueNotifier หรือ ChangeNotifier ตัดสินจากอะไร (4.1, 4.3, 4.4, 4.5)
  2. ทำไมการเก็บ remainingCount ไว้เป็น field ถึงเป็นบั๊กรอเกิด (4.2)
  3. boolean สามตัวสร้างสถานะได้กี่แบบ และมีความหมายจริงกี่แบบ (4.7)
  4. ทำไมงานเก็บกวาดต้องอยู่ใน finally ไม่ใช่ท้าย try (4.8)
  5. mounted ตอบคำถามอะไรได้ และตอบอะไรไม่ได้ (4.9)
  6. ตัวนับรุ่นต้องตรวจกี่จุด และแต่ละจุดกันอะไร (4.10)
  7. ทำไมยกเลิกตัวจับเวลาแล้วยังต้องมีตัวนับรุ่น (4.11)
  8. ทำไม removeListener ด้วย closure ถึงไม่ทำงาน (4.12)
  9. เมื่อไม่รู้ว่าเซิร์ฟเวอร์ทำสำเร็จหรือยัง ทำไมห้ามย้อนกลับ (4.13)
  10. ลองใหม่เมื่อไหร่ใช้รหัสคำสั่งเดิม เมื่อไหร่ใช้รหัสใหม่ (4.14)
  11. ล้มเหลวกลางทางแล้วควรย้อนไปสถานะไหน ใช้หลักอะไรตัดสิน (4.15)

12. สรุปท้าย Part 4

สิ่งที่ทำได้แล้ว

  • ตัดสินได้ว่า state ควรอยู่ที่ไหน และย้ายออกจาก widget เมื่อถึงเวลา
  • ออกแบบสถานะให้ค่าที่เป็นไปไม่ได้สร้างไม่ได้
  • เขียนงาน async ที่ปลดล็อกตัวเองเสมอ ไม่ว่าจะจบทางไหน
  • กันผลลัพธ์เก่าไม่ให้ทับข้อมูลล่าสุด
  • รีเฟรชอัตโนมัติโดยไม่ซ้อน ไม่กวนผู้ใช้ และหยุดเป็น
  • ทำให้จอตอบสนองทันทีและย้อนกลับได้อย่างปลอดภัย
  • ทำให้การส่งซ้ำไม่กลายเป็นข้อมูลซ้ำ
  • ใช้ sealed class จัดงานหลายขั้นให้ compiler ช่วยตรวจ
  • เขียนเทสต์ที่บังคับลำดับเวลาได้ด้วย Completer และ fakeAsync
