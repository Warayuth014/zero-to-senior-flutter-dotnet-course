ระดับ: กลางใช้เวลา: 30–40 นาทีลงมือทำ: จำกัดว่าใครวาดใหม่บ้างเมื่อ store แจ้ง
ต้องผ่านอะไรมาก่อน: 1.13 (ListenableBuilder และขอบเขต rebuild) และ 4.5 (ChangeNotifier)
จบบทนี้แล้วจะทำได้: วาง builder ให้ถูกตำแหน่ง และรู้ว่าเมื่อไหร่ที่ควรแยก store แทนที่จะแยก builder

1. ปัญหาที่บทนี้แก้

4.5 จบด้วย store ที่ใช้งานได้ แต่มีผลข้างเคียงหนึ่งข้อที่ยังไม่ได้แก้ — notifyListeners() แจ้งทุกคนที่ฟังอยู่ ไม่ว่าจะสนใจส่วนไหน

store.markDone('L-1')  →  notifyListeners()
                            ├→ ยอดคงเหลือ วาดใหม่   ✓ เกี่ยว
                            ├→ รายการทั้งหมด วาดใหม่ ✓ เกี่ยว
                            └→ ปุ่มตั้งค่า วาดใหม่    ✗ ไม่เกี่ยวเลย

บนรายการไม่กี่สิบแถวจะไม่รู้สึกอะไร แต่เมื่อโตขึ้นและ store แจ้งบ่อย จะเริ่มหน่วงบน PDA

3. Mental model: builder คือรั้วกั้น

ทุกอย่างที่อยู่ ข้างใน builder จะถูกวาดใหม่เมื่อ store แจ้ง ส่วนที่อยู่ข้างนอกไม่ถูกแตะ

Scaffold(
  appBar: AppBar(title: const Text('หยิบสินค้า')),   ← ข้างนอก ไม่วาดใหม่
  body: ListenableBuilder(
    listenable: store,
    builder: (context, child) => ...,               ← ข้างใน วาดใหม่
  ),
)
ประโยคเดียวที่ต้องจำ: วาง builder ให้ ครอบเฉพาะส่วนที่ต้องใช้ข้อมูลนั้น — ไม่ครอบกว้างเกินจนของนิ่งถูกวาดใหม่ และไม่ซอยย่อยเกินจนโค้ดอ่านยากโดยไม่ได้อะไร

5. ความจริงที่ต้องเข้าใจ: แยก builder ไม่ได้ลดการแจ้ง

นี่คือจุดที่คนเข้าใจผิดบ่อยที่สุดในบทนี้

store.notifyListeners()
   ├→ builder ตัวที่ 1 ทำงาน
   └→ builder ตัวที่ 2 ทำงาน      ← ทั้งคู่ทำงานเสมอ

ChangeNotifier ไม่รู้ว่าอะไรเปลี่ยน มันแจ้งอย่างเดียว — ดังนั้นการแยก builder ไม่ได้ลดจำนวนครั้งที่ builder ถูกเรียก มันแค่ลด ปริมาณ widget ที่ถูกสร้างใหม่ ในแต่ละครั้ง

สิ่งที่การแยก builder ทำได้สิ่งที่ทำไม่ได้
ลดจำนวน widget ที่สร้างใหม่ลดจำนวนครั้งที่ถูกแจ้ง
ทำให้ของนิ่งไม่ถูกแตะทำให้ builder ไม่ทำงานเมื่อค่าที่มันสนใจไม่เปลี่ยน

ถ้าต้องการอย่างหลังจริง ๆ ต้องแยกที่ระดับ store ไม่ใช่ที่ระดับ builder

6. เมื่อไหร่ควรแยก store

// แทนที่จะมี store เดียวที่แจ้งทุกเรื่อง
class WarehouseStore extends ChangeNotifier {
  // รายการงาน + สถานะการเชื่อมต่อ + การตั้งค่า + ผู้ใช้
}

// แยกตามความถี่และความเกี่ยวข้อง
class PickStore extends ChangeNotifier { ... }        // เปลี่ยนเมื่อทำงาน
class ConnectionStatus extends ChangeNotifier { ... } // เปลี่ยนทุกไม่กี่วินาที

ตัวอย่างที่ชัดที่สุดคือสถานะการเชื่อมต่อ — ถ้ามันอยู่ใน store เดียวกับรายการงาน ทุกครั้งที่ตรวจสัญญาณ (ทุกไม่กี่วินาที) รายการทั้งหมดจะถูกสร้างใหม่ ทั้งที่ไม่มีอะไรเปลี่ยน

เกณฑ์แยก storeตัวอย่าง
เปลี่ยนคนละความถี่สถานะเชื่อมต่อ เทียบกับ รายการงาน
คนละอายุผู้ใช้ที่ล็อกอิน เทียบกับ ข้อมูลของหน้าจอ
คนละส่วนงานรับเข้า เทียบกับ หยิบออก

เหมือนที่เรียนใน 1.13 หัวข้อ 6 — ใช้เมื่อมีส่วนที่ไม่ขึ้นกับ store และสร้างแพง

ระวังกับดัก — ถ้าสิ่งที่ส่งผ่าน child ไปอ่านค่าจาก store เองข้างใน มันจะไม่ได้ค่าใหม่ เพราะถูกสร้างครั้งเดียว ให้ส่งเฉพาะสิ่งที่ไม่เกี่ยวกับ store จริง ๆ

8. ห้ามทำสิ่งที่มีผลข้างเคียงใน builder

// ผิด — builder ถูกเรียกทุกครั้งที่แจ้ง
builder: (context, child) {
  if (store.allDone) {
    showDialog(...);                     // เด้งซ้ำไม่หยุด
    HapticFeedback.lightImpact();        // สั่นรัว
  }
  return ...;
}

เป็นกฎเดียวกับ 1.9 — builder คือ build รูปแบบหนึ่ง จึงห้ามมีผลข้างเคียง

สิ่งที่ต้องเกิดครั้งเดียวเมื่อสถานะเปลี่ยน ให้ใช้การฟังแยกต่างหาก ตามรูปแบบที่ใช้ใน 3.14

@override
void initState() {
  super.initState();
  widget.store.addListener(_onStoreChanged);    // ฟังเพื่อทำผลข้างเคียง
}

@override
void dispose() {
  widget.store.removeListener(_onStoreChanged); // คู่ตรงข้าม (3.12)
  super.dispose();
}

bool _wasAllDone = false;

void _onStoreChanged() {
  final isAllDone = widget.store.allDone;
  if (isAllDone && !_wasAllDone) {
    HapticFeedback.lightImpact();               // เกิดครั้งเดียวตอนเปลี่ยน
  }
  _wasAllDone = isAllDone;
}

รายละเอียดเรื่องการฟังและการเลิกฟังจะลงลึกใน 4.12

9. อย่าซอยย่อยเกินจำเป็น

สัญญาณว่าซอยเกินปัญหาที่ตามมา
builder ห้าชั้นซ้อนกันในหน้าจอเดียวอ่านยากมาก และหาจุดที่วาดใหม่ไม่เจอ
แต่ละส่วนอ่านข้อมูลคนละจังหวะเห็นข้อมูลที่ไม่สอดคล้องกันในจอเดียว
เพิ่ม builder โดยยังไม่เคยวัดเพิ่มความซับซ้อนโดยไม่ได้อะไร

ข้อสองสำคัญและมองข้ามง่าย — ถ้าแยก builder จนแต่ละส่วนอ่าน store คนละครั้ง อาจเกิดจังหวะที่ยอดคงเหลือแสดง 3 แต่รายการแสดง 4 แถวที่ยังไม่เสร็จ

ทางป้องกันคือให้ส่วนที่ต้อง สอดคล้องกัน อยู่ใน builder เดียวกัน

และตามลำดับใน 2.10 — วัดก่อนเสมอ อย่าปรับโดยเดา

10. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ทั้งหน้าจอวาดใหม่เมื่อค่าเล็ก ๆ เปลี่ยนbuilder ครอบกว้างเกินย้ายลงมาให้ครอบเฉพาะส่วนที่ใช้
แยก builder แล้วยังวาดใหม่บ่อยเหมือนเดิมเข้าใจว่าการแยกลดการแจ้งแยกที่ระดับ store แทน
รายการทั้งหมดวาดใหม่ทุกครั้งที่ตรวจสัญญาณรวมของที่เปลี่ยนคนละความถี่ไว้ store เดียวแยก store
กล่องโต้ตอบเด้งซ้ำไม่หยุดเปิดใน builderย้ายไปการฟังแยก
ตัวเลขสองที่บนจอไม่ตรงกันชั่วขณะแยก builder จนอ่านคนละจังหวะรวมส่วนที่ต้องสอดคล้องไว้ builder เดียว
ส่วนที่ส่งผ่าน child ไม่อัปเดตมันอ่าน store เองข้างในส่งเฉพาะสิ่งที่ไม่เกี่ยวกับ store

11. แบบฝึกหัด

หน้าจอหนึ่งฟัง store ตัวเดียวทั้งหน้า และมีสี่ส่วน

ส่วนใช้ข้อมูลความถี่ที่เปลี่ยน
ชื่อหน้าจอไม่ใช้ไม่เคย
จุดสถานะการเชื่อมต่อสถานะเชื่อมต่อทุก 3 วินาที
ยอดคงเหลือรายการงานเมื่อปิดงาน
รายการงาน 200 แถวรายการงานเมื่อปิดงานหรือโหลดใหม่
  1. อธิบายว่าโครงปัจจุบันมีปัญหาอะไร
  2. แก้ไข พร้อมบอกว่าแก้ที่ระดับ builder หรือระดับ store
  3. อธิบายว่าทำไมยอดคงเหลือกับรายการควรอยู่ใน builder เดียวกัน
ดูแนวคำตอบและเหตุผล

ข้อ 1 — ปัญหาปัจจุบัน

ทุก 3 วินาทีที่สถานะเชื่อมต่อเปลี่ยน notifyListeners() ถูกเรียก แล้วทั้งหน้าจอถูกสร้างใหม่ รวมถึงรายการ 200 แถว — ทั้งที่ข้อมูลรายการไม่ได้เปลี่ยนเลยแม้แต่แถวเดียว

บน PDA จะเห็นเป็นการกระตุกเป็นจังหวะทุก 3 วินาทีขณะเลื่อนรายการ

ข้อ 2 — แก้ที่สองระดับ

ต้องแก้ทั้งสองระดับ เพราะแก้ระดับเดียวไม่พอ

// ระดับ store — แยกของที่เปลี่ยนคนละความถี่ออกจากกัน
class PickStore extends ChangeNotifier { ... }         // งาน
class ConnectionStatus extends ChangeNotifier { ... }  // สัญญาณ
// ระดับ builder — แต่ละส่วนฟังเฉพาะตัวที่เกี่ยวข้อง
Scaffold(
  appBar: AppBar(
    title: const Text('หยิบสินค้า'),          // ไม่ฟังอะไรเลย
    actions: [
      ListenableBuilder(
        listenable: connectionStatus,          // ฟังเฉพาะสัญญาณ
        builder: (context, child) => ConnectionDot(
          online: connectionStatus.isOnline,
        ),
      ),
    ],
  ),
  body: ListenableBuilder(
    listenable: pickStore,                     // ฟังเฉพาะงาน
    builder: (context, child) => Column(
      children: [
        Text('เหลือ ${pickStore.remainingCount}'),
        Expanded(child: LineList(lines: pickStore.lines)),
      ],
    ),
  ),
)

ทำไมแก้ระดับเดียวไม่พอ

ถ้าแก้แค่ผลที่ได้
ระดับ builder อย่างเดียวยังฟัง store เดียวกัน จึงยังถูกแจ้งทุก 3 วินาที รายการยังสร้างใหม่
ระดับ store อย่างเดียวถ้ายังมี builder ตัวเดียวครอบทั้งหน้าและฟังทั้งสอง ก็เหมือนเดิม

นี่คือประเด็นหลักของหัวข้อ 5 — การแยก builder ลดปริมาณ widget ที่สร้างใหม่ แต่ลดจำนวนครั้งที่ถูกแจ้งไม่ได้ ต้องแยกแหล่งที่แจ้งด้วย

ข้อ 3 — ทำไมยอดคงเหลือกับรายการต้องอยู่ด้วยกัน

เพราะทั้งสองอ่านจากข้อมูลชุดเดียวกันและต้องสอดคล้องกันเสมอ ถ้าแยกเป็นสอง builder จะมีจังหวะที่ตัวหนึ่งวาดเสร็จแล้วอีกตัวยังไม่วาด ผู้ใช้อาจเห็น

เหลือ 3 รายการ
─────────────
□ L-1     ← ยังไม่เสร็จ
□ L-2
□ L-3
□ L-4     ← นับได้ 4 แถว แต่หัวบอก 3

ในทางปฏิบัติจังหวะนี้สั้นมากจนอาจไม่ทันเห็น แต่บน PDA ที่ช้ากว่าจะเห็นได้จริง และที่สำคัญกว่าคือ มันเป็นการเปิดช่องให้เกิดสิ่งที่ไม่ควรเกิด — ซึ่งเป็นสิ่งที่ 4.2 พยายามกำจัดตั้งแต่ระดับข้อมูล

การรวมไว้ใน builder เดียวทำให้ทั้งสองอ่านข้อมูลชุดเดียวกันในการวาดครั้งเดียว จึงไม่มีทางไม่ตรงกัน

ลองต่อ: ใน lab ให้ใส่ debugPrint ไว้ต้น builder ทั้งสองตัว แล้วกดปิดงานหนึ่งบรรทัด — จะเห็นว่าทั้งคู่ทำงาน ซึ่งยืนยันหัวข้อ 5 ว่าการแยก builder ไม่ได้ลดการแจ้ง

12. สรุปท้ายบท

  • ทุกอย่างที่อยู่ใน builder ถูกวาดใหม่เมื่อ store แจ้ง
  • วาง builder ให้ครอบเฉพาะส่วนที่ต้องใช้ข้อมูลนั้น
  • การแยก builder ลดปริมาณ widget ที่สร้างใหม่ แต่ไม่ลดจำนวนครั้งที่ถูกแจ้ง
  • ถ้าต้องการลดการแจ้ง ต้องแยกที่ระดับ store
  • แยก store ตามความถี่ อายุ และส่วนงาน
  • ห้ามทำสิ่งที่มีผลข้างเคียงใน builder
  • ส่วนที่ต้องสอดคล้องกันควรอยู่ใน builder เดียวกัน
  • วัดก่อนปรับ อย่าซอยย่อยโดยเดา

บทถัดไป: 4.7 จะกลับมาที่เรื่องที่ค้างจาก 4.5 — สถานะของงานที่ต้องรอมีมากกว่า “กำลังโหลด” กับ “เสร็จแล้ว” และการแยกให้ครบคือสิ่งที่ทำให้ผู้ใช้รู้ว่าเกิดอะไรขึ้นจริง
