ระดับ: สูงใช้เวลา: 45–55 นาทีลงมือทำ: เปลี่ยนงานหลายขั้นตอนให้เหลือค่าเดียวที่ตอบได้ทุกอย่าง
ต้องผ่านอะไรมาก่อน: 4.7 (สถานะการโหลด และ sealed class), 4.14 (การกันคำสั่งซ้ำ) และ 3.14 (sealed class กับ pattern matching)
จบบทนี้แล้วจะทำได้: ออกแบบงานหลายขั้นตอนให้ compiler ช่วยจับกรณีที่ลืม และให้สถานะที่เป็นไปไม่ได้สร้างไม่ได้เลย

1. ปัญหาที่บทนี้แก้

4.7 แก้เรื่องสถานะของงานที่มีขั้นเดียวคือ “โหลด” แต่งานจริงในคลังมีหลายขั้น

รับเข้า: สแกนสินค้า → ใส่จำนวน → สแกนพาเลท → ตรวจทาน → ยืนยัน → เสร็จ

โค้ดที่โตขึ้นทีละนิดจะกลายเป็นแบบนี้

bool _loading = false;
bool _hasItem = false;
bool _hasQuantity = false;
bool _hasPallet = false;
bool _submitting = false;
bool _confirmed = false;
String? _error;

Item? _item;
int? _quantity;
String? _palletCode;
Receipt? _receipt;

แล้วทุกที่ในหน้าจอจะเต็มไปด้วยการรวมเงื่อนไข

if (!_loading && _hasItem && _hasQuantity && _hasPallet &&
    !_submitting && !_confirmed && _error == null) {
  // ถึงตรงนี้แปลว่า... อะไรนะ
}

boolean หกตัวสร้างได้ 64 แบบ แต่แบบที่มีความหมายจริงมีหกแบบ — อีก 58 แบบคือสถานะที่ไม่ควรมีอยู่ แต่โค้ดอนุญาตให้เกิดขึ้น

2. คำศัพท์ที่ต้องรู้ก่อน

คำความหมาย
finite state machineเครื่องที่อยู่ได้ทีละหนึ่งสถานะ จากชุดสถานะที่กำหนดไว้แน่นอน
stateสถานะปัจจุบัน ซึ่งบอกได้ว่าตอนนี้ทำอะไรได้บ้าง
eventสิ่งที่เกิดขึ้นแล้วอาจทำให้เปลี่ยนสถานะ เช่นสแกน กดปุ่ม
transitionการย้ายจากสถานะหนึ่งไปอีกสถานะ
guardเงื่อนไขที่ต้องผ่านก่อนจึงจะย้ายได้
entry effectสิ่งที่ต้องทำเมื่อเข้าสู่สถานะหนึ่ง เช่นย้ายโฟกัส
exhaustiveครอบทุกสถานะ ไม่มีทางตกหล่น

3. Mental model: ถามว่าตอนนี้อยู่ตรงไหน ไม่ใช่ผ่านอะไรมาแล้วบ้าง

แบบ boolean:  ผ่านสแกนแล้วไหม? ใส่จำนวนแล้วไหม? กำลังส่งไหม? ...
              (ต้องประกอบคำตอบเองทุกครั้ง และประกอบผิดได้)

แบบ FSM:      ตอนนี้อยู่สถานะไหน?
              (ค่าเดียว ตอบได้ทันที และผิดไม่ได้)
ประโยคเดียวที่ต้องจำ: สถานะไม่ได้บอกแค่ว่าเกิดอะไรไปแล้ว แต่บอกว่า ตอนนี้ทำอะไรได้บ้าง — ถ้าตอบคำถามหลังไม่ได้จากค่าเดียว แสดงว่ายังออกแบบไม่เสร็จ

4. เขียนสถานะให้พกข้อมูลของตัวเอง

4.7 ใช้ enum ซึ่งพอสำหรับกรณีนั้น แต่ที่นี่แต่ละขั้นถือข้อมูลไม่เหมือนกัน จึงต้องใช้ sealed class

sealed class ReceiveState {
  const ReceiveState();
}

class ScanItem extends ReceiveState {
  const ScanItem();                      // ยังไม่มีข้อมูลอะไร
}

class EnterQuantity extends ReceiveState {
  const EnterQuantity(this.item);
  final Item item;                       // มีสินค้าแน่นอน
}

class ScanPallet extends ReceiveState {
  const ScanPallet(this.item, this.quantity);
  final Item item;
  final int quantity;                    // มีทั้งสองอย่างแน่นอน
}

class ReadyToConfirm extends ReceiveState {
  const ReadyToConfirm(this.draft);
  final ReceiveDraft draft;
}

class Submitting extends ReceiveState {
  const Submitting(this.draft, this.commandId);
  final ReceiveDraft draft;
  final String commandId;                // รหัสคำสั่งอยู่ในสถานะเลย (4.14)
}

class Completed extends ReceiveState {
  const Completed(this.receipt);
  final Receipt receipt;
}

เทียบกับแบบเดิม

boolean + field ที่เป็น null ได้sealed class
อยู่ขั้นใส่จำนวน แต่ไม่มีสินค้าสร้างได้ ต้องเช็กเองสร้างไม่ได้ — constructor บังคับ
เสร็จแล้วแต่ไม่มีใบรับสร้างได้สร้างไม่ได้
กำลังส่งและเสร็จแล้วพร้อมกันสร้างได้สร้างไม่ได้ — เป็นได้ทีละคลาส
ต้องเขียน item!ทุกที่ไม่มีเลย

สังเกตว่า commandId ถูกยัดเข้าไปใน Submitting ซึ่งแก้ปัญหาจาก 4.14 ได้สวยกว่า map แยก — เพราะรหัสจะมีอยู่ก็ต่อเมื่ออยู่ในสถานะที่ส่งอยู่จริงเท่านั้น และหายไปเองเมื่อออกจากสถานะนั้น

5. ตารางการย้ายสถานะ

ก่อนเขียนโค้ด ให้เขียนตารางนี้ก่อนเสมอ — มันคือข้อตกลงทั้งหมดของฟีเจอร์ในหน้าเดียว

สถานะปัจจุบันเหตุการณ์เงื่อนไขสถานะถัดไป
ScanItemสแกนพบสินค้าEnterQuantity
ScanItemสแกนไม่พบอยู่ที่เดิม + แจ้งเตือน
EnterQuantityใส่จำนวน1–999ScanPallet
EnterQuantityยกเลิก—ScanItem
ScanPalletสแกนรูปแบบถูกต้องReadyToConfirm
ReadyToConfirmยืนยัน—Submitting
Submittingสำเร็จ—Completed
Submittingถูกปฏิเสธ—ReadyToConfirm + ข้อความ
Submittingไม่รู้ผล—อยู่ที่เดิม แล้วไปตรวจสอบ (4.13)
Completedเริ่มใหม่—ScanItem

ทุกช่องที่ไม่ได้อยู่ในตารางนี้ แปลว่า เหตุการณ์นั้นถูกปฏิเสธ — ซึ่งเป็นพฤติกรรมที่ต้องตั้งใจ ไม่ใช่ผลข้างเคียง

ประโยชน์ที่ได้ทันทีคือคำถามอย่าง “ถ้าสแกนพาเลทตอนที่ยังไม่ได้ใส่จำนวนล่ะ” ตอบได้จากตารางโดยไม่ต้องเปิดโค้ด

6. เหตุการณ์ต้องตรวจสถานะก่อนเสมอ

void onBarcodeScanned(String raw) {
  final code = normalizeBarcode(raw);

  switch (_state) {
    case ScanItem():
      final item = catalog.findByBarcode(code);
      if (item == null) {
        _flash('ไม่พบสินค้ารหัสนี้');     // ปฏิเสธโดยไม่เปลี่ยนสถานะ
        return;
      }
      _state = EnterQuantity(item);

    case ScanPallet(:final item, :final quantity):
      if (!isValidPalletCode(code)) {
        _flash('รูปแบบรหัสพาเลทไม่ถูกต้อง');
        return;
      }
      _state = ReadyToConfirm(ReceiveDraft(item, quantity, code));

    case EnterQuantity() || ReadyToConfirm() || Submitting() || Completed():
      _flash('ตอนนี้ยังไม่ใช่ขั้นตอนสแกน');   // ปฏิเสธอย่างชัดเจน
      return;
  }
  notifyListeners();
}

บาร์โค้ดเดียวกันให้ผลต่างกันตามสถานะ ซึ่งจำเป็นบน PDA เพราะ เครื่องสแกนยิงข้อมูลเข้ามาได้ตลอดเวลา ไม่สนใจว่าจอกำลังแสดงอะไร

สิ่งที่ FSM ให้เพิ่มคือกรณีสุดท้าย — แทนที่จะเงียบหรือทำอะไรมั่ว ๆ มันบอกผู้ใช้ตรง ๆ ว่าตอนนี้ไม่ใช่จังหวะสแกน

การใช้ switch บน sealed class โดยไม่มี default: ทำให้ compiler บังคับให้จัดการทุกสถานะ — พอเพิ่มสถานะใหม่ในภายหลัง flutter analyze จะไล่ชี้ทุกจุดที่ต้องกลับมาตัดสินใจ ซึ่งเป็นสิ่งที่ boolean ทำให้ไม่ได้เลย

7. คำสั่งที่ต้องรอผล

ตรงนี้คือจุดที่ FSM แก้ปัญหาจากบทก่อน ๆ ได้ในคราวเดียว

Future<void> confirm() async {
  final current = _state;
  if (current is! ReadyToConfirm) return;     // guard — กันกดซ้ำ (4.14)

  final commandId = _createCommandId();
  _state = Submitting(current.draft, commandId);
  notifyListeners();

  try {
    final receipt = await repository.confirm(current.draft, commandId: commandId);
    if (_state is! Submitting) return;         // ออกจาก flow ไปแล้ว
    _state = Completed(receipt);
  } on ReceiveException catch (error) {
    if (_state is! Submitting) return;
    if (error.outcomeUnknown) {
      _errorMessage = '${error.message} กำลังตรวจสอบสถานะล่าสุด';
      notifyListeners();
      await _reconcile(commandId);             // ยังอยู่ Submitting (4.13)
      return;
    }
    _errorMessage = error.message;
    _state = ReadyToConfirm(current.draft);    // ย้อนกลับให้ลองใหม่ได้
  } finally {
    notifyListeners();                          // (4.8)
  }
}
บรรทัดแทนที่อะไรจากบทก่อน
if (current is! ReadyToConfirm) return;ธง _submitting — และแข็งแรงกว่า เพราะบอกว่า อนุญาตจากสถานะไหน ไม่ใช่แค่ ตอนนี้ไม่ว่าง
_state = Submitting(...)การตั้ง _loading = true — และพก draft กับรหัสคำสั่งไปด้วย
if (_state is! Submitting) return;ตัวนับรุ่น (4.10) สำหรับกรณีที่ผู้ใช้ออกจาก flow ไปแล้ว
_state = ReadyToConfirm(current.draft)การย้อนกลับ (4.13) — และ draft ยังอยู่ครบ ไม่ต้องกรอกใหม่

ข้อควรระวังของ if (_state is! Submitting) — ถ้าผู้ใช้ออกจาก flow แล้วเข้ามาใหม่จนกลับมาอยู่ Submitting อีกครั้งพอดี การตรวจแค่ชนิดจะไม่พอ ต้องเทียบ commandId ด้วย

final active = _state;
if (active is! Submitting || active.commandId != commandId) return;

8. ผลข้างเคียงผูกกับการย้าย ไม่ใช่กับการวาด

บางอย่างต้องเกิดเมื่อ เข้าสู่ สถานะหนึ่ง

เข้าสู่สถานะต้องทำ
EnterQuantityย้ายโฟกัสไปช่องจำนวน และเลือกข้อความทั้งหมด
ScanPalletย้ายโฟกัสกลับช่องสแกน
Completedสั่นเครื่อง และเริ่มนับถอยหลังกลับหน้าแรก
ออกจาก flowยกเลิกคำขอที่ค้าง เพิ่มเลขรุ่น

สิ่งเหล่านี้ห้ามอยู่ใน build เพราะ build ถูกเรียกหลายครั้ง (1.9, 4.6) ให้ทำในจุดที่รู้ว่า เพิ่งเปลี่ยน

ReceiveState _state = const ScanItem();

void _transitionTo(ReceiveState next) {
  final previous = _state;
  if (previous.runtimeType == next.runtimeType) {
    _state = next;
    notifyListeners();
    return;
  }
  _state = next;
  _onEnter(next);                 // เกิดครั้งเดียวตอนเปลี่ยนจริง
  notifyListeners();
}

void _onEnter(ReceiveState state) {
  switch (state) {
    case EnterQuantity():
      onFocusQuantity?.call();
    case ScanPallet():
      onFocusScanner?.call();
    case Completed():
      onCompleted?.call();
    case ScanItem() || ReadyToConfirm() || Submitting():
      break;                       // ไม่ต้องทำอะไร แต่ต้องเขียนให้ครบ
  }
}

สังเกตว่า store ไม่ได้เรียก FocusNode เอง แต่เรียก callback ที่หน้าจอส่งเข้ามา — เพราะ store ต้องไม่รู้จักเรื่อง widget (4.5) และเพื่อให้ทดสอบ store ได้โดยไม่ต้องสร้างหน้าจอ

9. แยกส่วนที่คิด ออกจากส่วนที่ทำ

รูปแบบที่ทดสอบง่ายที่สุดคือแยกฟังก์ชันที่ตัดสินใจออกมาเป็นฟังก์ชันบริสุทธิ์

// รับสถานะเก่า + เหตุการณ์ → คืนสถานะใหม่ ไม่แตะอะไรข้างนอกเลย
ReceiveState reduce(ReceiveState state, ReceiveEvent event) => switch ((state, event)) {
  (ScanItem(), ItemScanned(:final item)) => EnterQuantity(item),
  (EnterQuantity(:final item), QuantityEntered(:final quantity)) =>
      ScanPallet(item, quantity),
  (ScanPallet(:final item, :final quantity), PalletScanned(:final code)) =>
      ReadyToConfirm(ReceiveDraft(item, quantity, code)),
  _ => state,        // เหตุการณ์ที่ไม่ตรงเงื่อนไข ไม่เปลี่ยนอะไร
};
test('สแกนพาเลทตอนอยู่ขั้นใส่จำนวน ต้องไม่เปลี่ยนสถานะ', () {
  const state = EnterQuantity(item);
  final next = reduce(state, const PalletScanned('PAL-0001'));
  expect(next, same(state));
});

ไม่ต้องมี widget ไม่ต้องมี repository ไม่ต้อง await — เป็นการทดสอบตรรกะล้วน ๆ แบบเดียวกับที่ทำกับ ScanSession ใน Part 3

10. เมื่อไหร่ที่ยังไม่ควรใช้ FSM

FSM ไม่ฟรี — ต้องเขียนคลาสหลายตัว และทุกจุดที่อ่านสถานะต้อง switch

ใช้ FSM เมื่อยังไม่ต้องใช้เมื่อ
มีสามขั้นตอนขึ้นไปมีแค่ “โหลด/สำเร็จ/ล้มเหลว”
ลำดับสำคัญ ข้ามขั้นไม่ได้ทำอะไรก่อนหลังก็ได้
แต่ละขั้นถือข้อมูลไม่เหมือนกันข้อมูลชุดเดียวตลอด
มีคำถามว่า “ตอนนี้กดอันนี้ได้ไหม” หลายจุดไม่มีคำถามแบบนั้น

PickStore ใน lab เป็นตัวอย่างของฝั่งขวา — มันมี LoadState ที่เป็น enum ธรรมดา เพราะการหยิบสินค้าไม่มีลำดับขั้น แต่ละบรรทัดปิดงานได้อิสระ

สัญญาณที่บอกว่าถึงเวลาต้องเปลี่ยนแล้ว

สัญญาณความหมาย
เขียน if รวมสาม boolean ขึ้นไปกำลังประกอบสถานะเองด้วยมือ
มี field! กระจายอยู่หลายที่ข้อมูลไม่ได้ผูกกับสถานะ
เจอบั๊กแบบ “กดปุ่มนี้ตอนนี้ไม่ควรได้”ไม่มี guard ที่ระดับสถานะ
อธิบาย flow ให้คนอื่นฟังไม่ได้ในหนึ่งนาทียังไม่มีตารางการย้าย

11. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ยังต้องเขียน item! เต็มไปหมดใช้ enum แต่เก็บข้อมูลเป็น field ที่เป็น null ได้ย้ายข้อมูลเข้าไปในคลาสของสถานะ
เพิ่มสถานะใหม่แล้วบางจุดไม่รองรับมี default: ใน switchตัดออก ให้ compiler ไล่ให้
สแกนแล้วเกิดสิ่งที่ไม่ควรเกิดในขั้นนั้นเหตุการณ์ไม่ได้ตรวจสถานะก่อนให้ทุกเหตุการณ์ switch บนสถานะ
โฟกัสเด้งซ้ำ หรือสั่นรัวทำผลข้างเคียงใน buildผูกกับการย้ายสถานะ
ผลของคำสั่งเก่าย้ายสถานะที่ผู้ใช้ออกไปแล้วไม่ตรวจสถานะหลัง awaitตรวจทั้งชนิดและ commandId
ล้มเหลวแล้วต้องกรอกใหม่ทั้งหมดย้อนกลับไปสถานะแรกสุดย้อนไปสถานะก่อนหน้าพร้อม draft เดิม
โค้ดยาวขึ้นแต่ไม่ได้อะไรใช้ FSM กับงานที่มีขั้นเดียวใช้ enum ก็พอ

12. แบบฝึกหัด

งาน putaway — สแกนพาเลท แล้วสแกนช่องเก็บ แล้วยืนยัน

String? _palletCode;
String? _locationCode;
bool _submitting = false;
bool _done = false;
String? _error;

void onScan(String code) {
  if (_palletCode == null) {
    _palletCode = code;
  } else {
    _locationCode = code;
  }
  notifyListeners();
}

Future<void> confirm() async {
  _submitting = true;
  notifyListeners();
  try {
    await repository.putaway(_palletCode!, _locationCode!);
    _done = true;
  } catch (e) {
    _error = 'ไม่สำเร็จ';
  }
  _submitting = false;
  notifyListeners();
}
  1. ระบุสถานะที่ไม่ควรมีอยู่ แต่โค้ดนี้สร้างได้
  2. เขียนตารางการย้ายสถานะ แล้วเขียนใหม่ด้วย sealed class
  3. ตอบว่าถ้าช่องเก็บเต็ม เซิร์ฟเวอร์ปฏิเสธ ควรไปสถานะไหน และทำไมไม่ใช่สถานะแรกสุด

ข้อ 1 — สถานะที่ไม่ควรมีอยู่

#สถานะที่สร้างได้ทำไมไม่ควรมี
1_submitting และ _done เป็น true พร้อมกันกำลังส่งและเสร็จแล้วพร้อมกันไม่มีความหมาย
2_done เป็น true แต่ _locationCode เป็น nullเก็บของสำเร็จโดยไม่มีช่องเก็บ
3สแกนช่องเก็บได้ทั้งที่ยังไม่มีพาเลทที่จริงกันไว้ด้วย if แต่กันด้วยการเดา ไม่ใช่ด้วยกติกา
4_error มีค่า พร้อมกับ _done เป็น trueสำเร็จและล้มเหลวพร้อมกัน
5กด confirm() ตอนที่ยังไม่มีช่องเก็บ_locationCode! จะพังทันที
6กด confirm() ซ้ำระหว่างส่งไม่มี guard เลย — ส่งซ้อนได้ (4.14)

และมีข้อบกพร่องเชิงพฤติกรรมเพิ่ม — onScan ตัดสินจาก “ค่าไหนยังว่าง” ทำให้สแกนพาเลทผิดแล้วแก้ไม่ได้ ต้องปิดหน้าจอเริ่มใหม่

ข้อ 2 — ตารางและโค้ด

สถานะเหตุการณ์เงื่อนไขถัดไป
ScanPalletสแกนรูปแบบถูกScanLocation
ScanPalletสแกนรูปแบบผิดอยู่ที่เดิม + เตือน
ScanLocationสแกนรูปแบบถูกReadyToConfirm

ReadyToConfirmยืนยัน—Submitting

sealed class PutawayState { const PutawayState(); }

class ScanPallet extends PutawayState { const ScanPallet(); }

class ScanLocation extends PutawayState {
  const ScanLocation(this.palletCode);
  final String palletCode;
}

class ReadyToConfirm extends PutawayState {
  const ReadyToConfirm(this.palletCode, this.locationCode);
  final String palletCode;
  final String locationCode;
}

class Submitting extends PutawayState {
  const Submitting(this.palletCode, this.locationCode, this.commandId);
  final String palletCode;
  final String locationCode;
  final String commandId;
}

class Completed extends PutawayState {
  const Completed(this.palletCode, this.locationCode);
  final String palletCode;
  final String locationCode;
}
void onScan(String raw) {
  final code = normalizeBarcode(raw);
  switch (_state) {
    case ScanPallet():
      if (!isValidPalletCode(code)) {
        _flash('รูปแบบรหัสพาเลทไม่ถูกต้อง');
        return;
      }
      _transitionTo(ScanLocation(code));

    case ScanLocation(:final palletCode):
      if (!isValidLocationCode(code)) {
        _flash('รูปแบบรหัสช่องเก็บไม่ถูกต้อง');
        return;
      }
      _transitionTo(ReadyToConfirm(palletCode, code));

    case ReadyToConfirm(:final palletCode):
      // สแกนตอนตรวจทาน = ตั้งใจเปลี่ยนช่องเก็บ
      if (!isValidLocationCode(code)) {
        _flash('รูปแบบรหัสช่องเก็บไม่ถูกต้อง');
        return;
      }
      _transitionTo(ReadyToConfirm(palletCode, code));

    case Submitting() || Completed():
      _flash('กำลังบันทึกอยู่ กรุณารอสักครู่');
  }
}
Future<void> confirm() async {
  final current = _state;
  if (current is! ReadyToConfirm) return;              // แก้ข้อ 5, 6

  final commandId = _createCommandId();
  _transitionTo(Submitting(
      current.palletCode, current.locationCode, commandId));

  try {
    await repository.putaway(
      current.palletCode, current.locationCode, commandId: commandId);
    if (!_isStillSubmitting(commandId)) return;
    _transitionTo(Completed(current.palletCode, current.locationCode));
  } on LocationFullException {
    if (!_isStillSubmitting(commandId)) return;
    _errorMessage = 'ช่องเก็บนี้เต็มแล้ว กรุณาสแกนช่องอื่น';
    _transitionTo(ScanLocation(current.palletCode));   // ข้อ 3
  } on PutawayException catch (error) {
    if (!_isStillSubmitting(commandId)) return;
    _errorMessage = error.message;
    _transitionTo(ReadyToConfirm(current.palletCode, current.locationCode));
  } finally {
    notifyListeners();
  }
}

bool _isStillSubmitting(String commandId) {
  final active = _state;
  return active is Submitting && active.commandId == commandId;
}

สังเกตว่าข้อ 1, 2, 4 หายไปเองโดยไม่ต้องเขียนโค้ดอะไรกัน เพราะสถานะเป็นได้ทีละคลาสเดียว และข้อมูลผูกอยู่กับคลาสนั้น

ข้อ 3 — ช่องเก็บเต็ม ควรไปสถานะไหน

ควรไป ScanLocation โดยยังถือ palletCode เดิมไว้

เหตุผลมาจากการถามว่า ข้อมูลชิ้นไหนที่ยังใช้ได้อยู่

ข้อมูลยังถูกต้องไหมหลังถูกปฏิเสธ
รหัสพาเลทถูกต้อง — ของยังอยู่บนรถยกคันเดิม
รหัสช่องเก็บไม่ถูกต้องแล้ว — ช่องนั้นเต็ม

ทำไมไม่กลับไป ScanPallet — เพราะจะบังคับให้ผู้ใช้สแกนพาเลทใหม่ทั้งที่ข้อมูลนั้นไม่ได้ผิดอะไรเลย คนที่ยืนถือ PDA อยู่หน้าแร็คต้องก้มไปสแกนพาเลทบนรถยกอีกรอบโดยไม่มีเหตุผล ทั้งที่สิ่งเดียวที่ต้องเปลี่ยนคือช่องเก็บ

ทำไมไม่อยู่ ReadyToConfirm — เพราะสถานะนั้นแปลว่า “พร้อมยืนยัน” ซึ่งไม่จริงแล้ว ถ้าคาไว้ตรงนั้น ผู้ใช้กดยืนยันซ้ำได้และจะถูกปฏิเสธซ้ำ

ทำไมไม่มีสถานะ Failed แยก — เพราะความล้มเหลวส่วนใหญ่ไม่ใช่ปลายทาง แต่คือ การกลับไปยังจุดที่แก้ได้ การมีสถานะ Failed ที่มีทางออกทางเดียวคือเริ่มใหม่ทั้งหมด คือการทิ้งงานที่ผู้ใช้ทำมาแล้วโดยไม่จำเป็น

สิ่งที่ควรทำเพิ่มคือรีเฟรชรายการช่องเก็บว่างพร้อมกับการย้าย เพื่อให้ผู้ใช้เลือกช่องถัดไปได้เลย — ซึ่งเป็น entry effect ตามหัวข้อ 8

หลักที่ใช้ตอบคำถามทำนองนี้ได้ทุกครั้ง: ย้อนกลับไปให้ใกล้จุดที่ผิดที่สุด และเก็บทุกอย่างที่ยังถูกต้องไว้

13. สรุปท้ายบท

  • boolean หลายตัวสร้างสถานะที่ไม่ควรมีอยู่เป็นสิบเป็นร้อยแบบ
  • สถานะเดียวตอบได้ว่า “ตอนนี้ทำอะไรได้บ้าง” ไม่ใช่แค่ “ผ่านอะไรมาแล้ว”
  • sealed class ทำให้ข้อมูลผูกกับสถานะ จนไม่ต้องเขียน ! อีกเลย
  • เขียนตารางการย้ายก่อนเขียนโค้ดเสมอ
  • ทุกเหตุการณ์ต้อง switch บนสถานะก่อน และปฏิเสธอย่างชัดเจนเมื่อไม่ตรง
  • switch ที่ไม่มี default: ทำให้ compiler ไล่จุดที่ลืมให้เมื่อเพิ่มสถานะใหม่
  • guard ที่ระดับสถานะแทนธง _submitting ได้ และแข็งแรงกว่า
  • หลัง await ต้องตรวจทั้งชนิดของสถานะและรหัสคำสั่ง
  • ผลข้างเคียงผูกกับการย้ายสถานะ ไม่ใช่กับ build
  • ล้มเหลวแล้วย้อนไปให้ใกล้จุดที่ผิดที่สุด เก็บสิ่งที่ยังถูกต้องไว้
  • งานขั้นเดียวใช้ enum ก็พอ อย่าใช้ FSM เพราะรู้จักมัน
