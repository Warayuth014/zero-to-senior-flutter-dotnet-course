1. ปัญหาที่บทนี้แก้

สามอาการที่มาจากสาเหตุเดียวกัน

อาการเกิดจาก
กดปุ่มครั้งเดียวแต่มีเสียงเตือนสามครั้งฟังซ้ำโดยไม่ได้เลิกฟังของเก่า

แอปช้าลงเรื่อย ๆ ยิ่งใช้นานยิ่งช้าผู้ฟังสะสมไม่มีวันหาย

ทั้งสามมาจากคำถามเดียวที่ไม่ได้ตอบ — ใครเป็นคนถอดสิ่งที่ต่อไว้ออก

2. คำศัพท์ที่ต้องรู้ก่อน

คำความหมาย
listenerฟังก์ชันที่ถูกเรียกเมื่อมีการแจ้ง
subscriptionการสมัครรับข้อมูลจาก stream ซึ่งต้องยกเลิกเป็น
ownershipใครเป็นเจ้าของ ก็คือคนที่ต้องทำลาย
one-shot eventเหตุการณ์ที่ควรเกิดครั้งเดียว เช่นเสียงเตือนหรือการเปลี่ยนหน้า
closureฟังก์ชันไม่มีชื่อที่เขียนตรงจุดใช้งาน

3. Mental model: ทุกการต่อ ต้องมีการถอด

addListener      ↔  removeListener
listen           ↔  subscription.cancel()
addObserver      ↔  removeObserver
Timer.periodic   ↔  timer.cancel()
TextEditingController()  ↔  controller.dispose()
ประโยคเดียวที่ต้องจำ: ทันทีที่เขียนบรรทัดฝั่งซ้าย ให้เลื่อนลงไปเขียนบรรทัดฝั่งขวาใน dispose() ก่อนทำอย่างอื่น — ไม่ใช่ "เดี๋ยวค่อยกลับมาใส่"

นี่คือกฎเดียวกับที่ใช้กับ TextEditingController และ FocusNode ใน 3.12 — ต่างกันแค่ชื่อของสิ่งที่ต้องเก็บกวาด

4. เมื่อไหร่ที่ไม่ต้องเขียนเอง

เริ่มจากข่าวดีก่อน — ถ้าฟังเพื่อวาดหน้าจอใหม่ ไม่ต้องเขียนอะไรเลย

ListenableBuilder(
  listenable: widget.store,
  builder: (context, child) => Text('เหลือ ${widget.store.remainingCount}'),
)

ListenableBuilder จัดการให้ครบ

ทำอะไรให้เมื่อไหร่
addListenerตอนเข้าต้นไม้
removeListener ตัวเก่า + addListener ตัวใหม่เมื่อ listenable เปลี่ยนตัว
removeListenerตอนออกจากต้นไม้

เปิด lib/state_basics/state_basics_app.dart — มี ListenableBuilder สองที่ และ ไม่มี addListener เลยแม้แต่ที่เดียว นั่นเป็นเพราะทุกการฟังในหน้าจอนี้มีจุดประสงค์เดียวคือวาดใหม่

ฟังเพื่ออะไรใช้อะไร
วาดหน้าจอใหม่ListenableBuilder — ไม่ต้องเขียนเอง
ทำผลข้างเคียง เช่นสั่นเครื่อง เปลี่ยนหน้าaddListener เอง และต้องถอดเอง

ก่อนจะเขียน addListener ทุกครั้ง ให้ถามก่อนว่าอยู่กรณีบนหรือกรณีล่าง

5. เมื่อต้องฟังเอง

class _PickScreenState extends State<PickScreen> {
  bool _wasAllDone = false;

  @override
  void initState() {
    super.initState();
    widget.store.addListener(_onStoreChanged);    // ต่อ
    widget.store.load();
  }

  @override
  void dispose() {
    widget.store.removeListener(_onStoreChanged); // ถอด
    super.dispose();
  }

  void _onStoreChanged() {
    final isAllDone = widget.store.allDone;
    if (isAllDone && !_wasAllDone) {
      HapticFeedback.mediumImpact();              // เกิดครั้งเดียวตอนเปลี่ยน
    }
    _wasAllDone = isAllDone;
  }
}

สองจุดที่ต้องสังเกต

จุดเหตุผล
ใช้เมธอดที่มีชื่อ ไม่ใช่ closureดูหัวข้อ 6
เทียบกับค่าก่อนหน้าเสมอnotifyListeners() ถูกเรียกหลายครั้ง แต่ผลข้างเคียงต้องเกิดครั้งเดียว

6. กับดักที่ทำให้ถอดไม่ออก

// ผิด — ถอดไม่ออกเลย
store.addListener(() => _showToast());
...
store.removeListener(() => _showToast());

สอง closure นี้หน้าตาเหมือนกันแต่เป็นคนละวัตถุ removeListener เทียบด้วยความเป็นตัวเดียวกัน จึงหาไม่เจอและ ไม่แจ้งอะไรเลยว่าถอดไม่สำเร็จ

// ถูก — เมธอดที่มีชื่อ อ้างถึงตัวเดียวกันเสมอ
store.addListener(_showToast);
store.removeListener(_showToast);

ถ้าจำเป็นต้องใช้ closure จริง ๆ ให้เก็บไว้ในตัวแปร

late final VoidCallback _listener;

@override
void initState() {
  super.initState();
  _listener = () => _showToast(widget.store.errorMessage);
  widget.store.addListener(_listener);
}

@override
void dispose() {
  widget.store.removeListener(_listener);
  super.dispose();
}

7. ห้ามฟังใน build

@override
Widget build(BuildContext context) {
  widget.store.addListener(_onStoreChanged);   // หายนะ
  return ...;
}

ไล่ดูว่าเกิดอะไร

build ครั้งที่ 1 → ผู้ฟัง 1 ตัว
store แจ้ง → build ครั้งที่ 2 → ผู้ฟัง 2 ตัว
store แจ้ง → callback ทำงาน 2 ครั้ง → build 2 ครั้ง → ผู้ฟัง 4 ตัว
store แจ้ง → 4 ครั้ง → ผู้ฟัง 8 ตัว ...

เพิ่มเป็นทวีคูณจนแอปค้าง เป็นตัวอย่างที่ชัดที่สุดของกฎใน 1.9 — build ต้องไม่มีผลข้างเคียง

สิ่งที่ต้องทำทำที่ไหน
เริ่มฟังinitState
เปลี่ยนตัวที่ฟังdidUpdateWidget
เลิกฟังdispose
วาดbuild เท่านั้น

8. เมื่อสิ่งที่ฟังถูกเปลี่ยนตัว

initState ทำงานครั้งเดียว แต่ widget แม่อาจส่ง store ตัวใหม่ลงมาทีหลัง

@override
void didUpdateWidget(covariant PickScreen oldWidget) {
  super.didUpdateWidget(oldWidget);
  if (oldWidget.store == widget.store) return;      // ตัวเดิม ไม่ต้องทำอะไร
  oldWidget.store.removeListener(_onStoreChanged);  // ถอดจากตัวเก่า
  widget.store.addListener(_onStoreChanged);        // ต่อกับตัวใหม่
}

ถ้าลืมเมธอดนี้จะได้อาการแรกในหัวข้อ 1 — ฟังทั้งตัวเก่าและตัวใหม่พร้อมกัน callback ทำงานสองรอบ

บรรทัด if (oldWidget.store == widget.store) return; จำเป็น เพราะ didUpdateWidget ถูกเรียกทุกครั้งที่แม่วาดใหม่ ไม่ใช่เฉพาะตอนที่ store เปลี่ยน — เป็นรูปแบบเดียวกับที่เจอใน 4.3

9. ใครสร้าง คนนั้นทำลาย

คำถามที่ต้องตอบให้ได้ทุกครั้ง — store ตัวนี้ใครทำลาย

store มาจากไหนใครเรียก store.dispose()
หน้าจอสร้างเองใน initStateหน้าจอนั้น ใน dispose
รับมาทาง constructorห้ามทำลาย — คนที่สร้างเป็นคนทำลาย
อยู่ระดับแอปทั้งตัวตัวแอป หรือไม่ทำลายเลยก็ได้
// หน้าจอสร้างเอง จึงต้องทำลายเอง
class _ScreenState extends State<Screen> {
  late final PickStore _store;

  @override
  void initState() {
    super.initState();
    _store = PickStore(repository: widget.repository);
    _store.load();
  }

  @override
  void dispose() {
    _store.dispose();
    super.dispose();
  }
}

ใน lab นั้น PickScreen รับ store มาทาง constructor จึงไม่ทำลาย — ถ้าทำลาย เทสต์ที่สร้าง store ไว้ข้างนอกแล้วตรวจผลหลังจากนั้นจะพังทันที

ทำลาย store ที่คนอื่นเป็นเจ้าของ คือบั๊กที่หายากที่สุดชนิดหนึ่ง — หน้าจออื่นที่ยังใช้ store ตัวเดียวกันจะพังในภายหลัง โดยไม่มีอะไรชี้กลับมาที่ต้นเหตุ

10. เหตุการณ์ที่ควรเกิดครั้งเดียว

สถานะกับเหตุการณ์เป็นคนละอย่าง และการเก็บเหตุการณ์ไว้ในสถานะทำให้มันเกิดซ้ำ

สถานะเหตุการณ์
ตัวอย่าง“กำลังโหลด” “มีข้อผิดพลาด”“แสดงข้อความ” “ปิดหน้าจอ” “สั่นเครื่อง”
อ่านซ้ำได้ไหมได้ อ่านกี่ครั้งก็ได้ค่าเดิมไม่ได้ ต้องเกิดครั้งเดียว
// ผิด — เก็บเหตุการณ์ไว้เป็นสถานะ
String? errorMessage;   // ถ้ามีค่า ให้แสดง SnackBar

// หน้าจอกลับเข้ามาใหม่ → อ่านเจอ errorMessage เดิม → เด้ง SnackBar ซ้ำ

ทางแก้ที่ตรงไปตรงมาที่สุดคือ ลบทิ้งเมื่อใช้แล้ว

void _onStoreChanged() {
  final message = widget.store.takeErrorMessage();   // อ่านแล้วเคลียร์
  if (message == null) return;
  ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(message)));
}

สังเกตว่าใน lab นั้น _errorMessage ถูกใช้เป็น สถานะ — แสดงเป็นแถบอยู่กับที่ในหน้าจอ ไม่ใช่เด้งขึ้นมาแล้วหาย จึงไม่มีปัญหานี้

การเลือกระหว่างสองแบบขึ้นกับว่าอยากให้ข้อความนั้นอยู่ต่อหรือหายไป ซึ่งเป็นการตัดสินใจเรื่อง UX ไม่ใช่เรื่องเทคนิค

11. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
callback ทำงานหลายรอบฟังซ้ำโดยไม่ถอดของเก่าdidUpdateWidget หรือย้ายออกจาก build
removeListener แล้วยังทำงานอยู่ถอดคนละ closure กับที่ต่อไว้ใช้เมธอดที่มีชื่อ
แอปช้าลงเรื่อย ๆฟังใน buildย้ายไป initState
used after being disposedยังฟังของที่ถูกทำลายแล้วถอดใน dispose
หน้าจออื่นพังโดยไม่มีสาเหตุทำลาย store ที่รับมาใครสร้าง คนนั้นทำลาย
SnackBar เด้งซ้ำเมื่อกลับเข้าหน้าจอเก็บเหตุการณ์ไว้เป็นสถานะอ่านแล้วเคลียร์
เขียน addListener เต็มไปหมดใช้ผิดเครื่องมือถ้าฟังเพื่อวาด ใช้ ListenableBuilder

12. แบบฝึกหัด

class _DashboardState extends State<Dashboard> {
  @override
  Widget build(BuildContext context) {
    widget.store.addListener(() {
      if (widget.store.hasError) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text(widget.store.errorMessage!)),
        );
      }
    });
    return Text('เหลือ ${widget.store.remainingCount}');
  }

  @override
  void dispose() {
    widget.store.dispose();
    super.dispose();
  }
}

store ถูกสร้างที่หน้าจอแม่แล้วส่งลงมาทาง constructor

  1. หาข้อบกพร่องให้ได้อย่างน้อยห้าข้อ
  2. เขียนใหม่
  3. อธิบายว่าข้อไหนอาการจะโผล่ทันที และข้อไหนจะโผล่ตอนขึ้น production แล้ว
ดูแนวคำตอบและเหตุผล

ข้อ 1 — ข้อบกพร่อง

#ข้อบกพร่องผลที่เกิด
1addListener ใน buildผู้ฟังเพิ่มเป็นทวีคูณจนแอปค้าง
2ใช้ closure ไม่มีชื่อต่อให้ย้ายไป initState ก็ยังถอดไม่ออก
3ไม่มี removeListener เลยฟัง store ที่ตายแล้ว
4ทำลาย store ที่รับมาหน้าจอแม่และหน้าจออื่นพัง
5ใช้ addListener เพื่อวาด remainingCountผิดเครื่องมือ — ควรเป็น ListenableBuilder
6ใช้ context ใน callback โดยไม่ตรวจ mountedพังเมื่อ callback ทำงานหลังหน้าจอปิด (4.9)
7ข้อผิดพลาดเป็นสถานะค้างเด้ง SnackBar ซ้ำทุกครั้งที่แจ้ง
8ไม่มี didUpdateWidgetแม่เปลี่ยน store แล้วยังฟังตัวเก่า
9errorMessage!พังถ้า hasError กับ errorMessage ไม่ตรงกัน (4.7)

ข้อ 2 — เขียนใหม่

class _DashboardState extends State<Dashboard> {
  @override
  void initState() {
    super.initState();
    widget.store.addListener(_onStoreChanged);      // แก้ข้อ 1, 2
  }

  @override
  void didUpdateWidget(covariant Dashboard oldWidget) {   // แก้ข้อ 8
    super.didUpdateWidget(oldWidget);
    if (oldWidget.store == widget.store) return;
    oldWidget.store.removeListener(_onStoreChanged);
    widget.store.addListener(_onStoreChanged);
  }

  @override
  void dispose() {
    widget.store.removeListener(_onStoreChanged);   // แก้ข้อ 3
    super.dispose();                                // แก้ข้อ 4 — ไม่ทำลาย store
  }

  void _onStoreChanged() {
    if (!mounted) return;                           // แก้ข้อ 6
    final message = widget.store.takeErrorMessage();// แก้ข้อ 7, 9
    if (message == null) return;
    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(content: Text(message)),
    );
  }

  @override
  Widget build(BuildContext context) => ListenableBuilder(   // แก้ข้อ 5
    listenable: widget.store,
    builder: (context, child) => Text('เหลือ ${widget.store.remainingCount}'),
  );
}

มีจุดที่ดูขัดกันแต่ถูกต้อง — หน้าจอนี้ทั้ง addListener เองและใช้ ListenableBuilder ด้วย เพราะฟังสองจุดประสงค์ที่ต่างกัน

ทางทำอะไร
addListener เองผลข้างเคียง — เด้ง SnackBar
ListenableBuilderวาดตัวเลขใหม่

และ takeErrorMessage() ต้องเพิ่มใน store

String? takeErrorMessage() {
  final message = _errorMessage;
  _errorMessage = null;      // อ่านแล้วหาย เกิดครั้งเดียว
  return message;
}

ข้อ 3 — อันไหนโผล่เมื่อไหร่

โผล่ทันทีโผล่ตอน production
ข้อ 1 — แอปช้าลงจนสังเกตได้ในไม่กี่วินาทีข้อ 3 — ต้องปิดหน้าจอตอนที่มีคำขอค้างพอดี
ข้อ 4 — เปิดหน้าจอนี้แล้วกลับ หน้าแม่พังทันทีข้อ 6 — ต้องกดย้อนกลับตอนที่คำขอกำลังจะเสร็จ
ข้อ 7 — เห็นซ้ำตั้งแต่ทดสอบครั้งแรกข้อ 8 — เกิดเฉพาะเมื่อแม่เปลี่ยน store ซึ่งอาจยังไม่มีในเวอร์ชันแรก
ข้อ 9 — เกิดเมื่อ store เข้าสถานะที่ไม่สอดคล้อง

ทำไมการแยกแบบนี้สำคัญ

ข้อที่โผล่ทันทีจะถูกแก้อยู่แล้วเพราะเจอตอนพัฒนา — ข้อที่อันตรายจริงคือกลุ่มขวา ซึ่งต้องอาศัยจังหวะบังเอิญจึงจะเกิด แปลว่ามันจะผ่านการทดสอบไปได้ทั้งหมด แล้วไปโผล่ที่หน้างานในวันที่เน็ตช้ากว่าปกติ

ทางกันเดียวที่ได้ผลคือ ทำให้เป็นนิสัย ไม่ใช่รอให้เจอ — เขียนฝั่งถอดทันทีที่เขียนฝั่งต่อ และตอบคำถามเรื่องเจ้าของก่อนเขียน dispose() ทุกครั้ง

ลองต่อ: ใน lab ลองเพิ่ม addListener ใน build ของ PickScreen พร้อม debugPrint แล้วกดปิดงานสองสามครั้ง — จะเห็นจำนวนบรรทัดที่พิมพ์ออกมาเพิ่มเป็น 1, 2, 4, 8 ซึ่งเป็นภาพของหัวข้อ 7 ที่ชัดที่สุด

13. สรุปท้ายบท

  • ทุกการต่อต้องมีการถอด และเขียนคู่กันทันที
  • ถ้าฟังเพื่อวาดหน้าจอ ใช้ ListenableBuilder แล้วไม่ต้องเขียนอะไรเลย
  • ถ้าฟังเพื่อผลข้างเคียง ต้อง addListener เองและถอดเอง
  • ใช้เมธอดที่มีชื่อเสมอ ไม่ใช่ closure ที่เขียนตรงจุด
  • ห้าม addListener ใน build — ผู้ฟังจะเพิ่มเป็นทวีคูณ
  • didUpdateWidget ต้องเทียบก่อนว่าเปลี่ยนตัวจริงไหม
  • ใครสร้าง คนนั้นทำลาย — สิ่งที่รับมาทาง constructor ห้ามทำลาย
  • เหตุการณ์ที่ควรเกิดครั้งเดียว ต้องเคลียร์เมื่ออ่านแล้ว
