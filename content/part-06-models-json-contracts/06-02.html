ระดับ: กลางใช้เวลา: 35–45 นาทีลงมือทำ: เขียนตัวแปลง JSON ที่ไม่พังแบบอธิบายไม่ได้
ต้องผ่านอะไรมาก่อน: 6.1 (การแยกชั้น) และ D1.19 (JSON และ dynamic)
จบบทนี้แล้วจะทำได้: เขียน fromJson/toJson ที่บอกได้ว่าพังตรงไหน และรู้ว่าเมื่อไหร่ควรพัง เมื่อไหร่ไม่ควร

1. ปัญหาที่บทนี้แก้

โค้ดที่ทุกคนเขียนครั้งแรก

factory TaskDto.fromJson(Map<String, dynamic> json) => TaskDto(
  id: json['id'] as String,
  quantity: json['quantity'] as int,
  status: json['status'] as String,
);

ใช้ได้ตอนทดสอบ แล้ววันหนึ่งได้ข้อความนี้จากหน้างาน

type 'Null' is not a subtype of type 'String' in type cast
คำถามที่ข้อความนี้ตอบไม่ได้
field ไหนที่เป็น null
แถวที่เท่าไหร่ในรายการ 40 แถว
เป็นเพราะเซิร์ฟเวอร์ผิด หรือแอปอ่านผิด

ผลคือต้องไปเปิด log ดู payload ดิบทั้งก้อนแล้วไล่หาเอง ซึ่งบางทีก็ไม่มี log เก็บไว้

2. คำศัพท์ที่ต้องรู้ก่อน

คำความหมาย
jsonDecodeแปลงข้อความ JSON เป็นวัตถุ Dart
boundary / ขอบจุดที่ข้อมูลจากภายนอกเข้ามาในแอป
untrustedข้อมูลที่ยังเชื่อไม่ได้ ต้องตรวจก่อน
castการบอก compiler ว่าค่านี้เป็นชนิดนี้
forward compatibleทนได้เมื่ออีกฝั่งเพิ่มของใหม่

3. Mental model: ทุกอย่างที่มาจากสายคือของแปลกหน้า

jsonDecode(body)  →  Map<String, dynamic>
                          │
                          └─ dynamic แปลว่า "อะไรก็ได้"
                             compiler ช่วยอะไรไม่ได้เลยตั้งแต่จุดนี้
ประโยคเดียวที่ต้องจำ: fromJson คือด่านที่ข้อมูลจากโลกภายนอกเข้ามา — สิ่งที่ผ่านด่านนี้ไปได้ ต้องเชื่อถือได้แล้ว ส่วนที่ผ่านไม่ได้ ต้องบอกให้ชัดว่าทำไม

ที่สำคัญคือ ข้อมูลที่ผิดไม่ได้แปลว่ามีคนโจมตีเสมอไป ส่วนใหญ่มาจากเซิร์ฟเวอร์คนละรุ่น ข้อมูลเก่าในฐานข้อมูล หรือบั๊กฝั่งหลังบ้าน — ซึ่งทั้งหมดเกิดขึ้นจริงและบ่อย

4. อ่านผ่านตัวช่วยกลาง ไม่อ่านเอง

เปิด lib/json_basics/json_read.dart

/// ตัวอ่านค่าจาก JSON ที่ใช้ร่วมกันทั้งแอป
///
/// ทุก model ต้องอ่านผ่านไฟล์นี้ ไม่เขียน `json['x'] as String` เอง
/// เพราะกฎการแปลงค่าต้องเหมือนกันทุกที่ และข้อความผิดพลาดต้องบอกตำแหน่งได้
class ContractException implements Exception {
  const ContractException(this.path, this.message);

  /// ตำแหน่งในก้อน JSON เช่น `items[3].quantity`
  final String path;
  final String message;

  @override
  String toString() => 'ContractException ที่ $path: $message';
}

ทำไมต้องมีชนิดข้อผิดพลาดของตัวเอง แทนที่จะใช้ FormatException

เหตุผลรายละเอียด
พก path มาด้วยรู้ว่าพังตรงไหนในก้อนที่ซ้อนหลายชั้น
แยกจากความผิดพลาดชนิดอื่นได้“เน็ตหลุด” กับ “ข้อมูลผิดสัญญา” ต้องรับมือคนละแบบ (4.8)
ดักได้ตรงจุดon ContractException catch ไม่กลืนบั๊กอื่น

5. fromJson ที่อ่านง่ายเพราะไม่มีเงื่อนไข

เปิด lib/json_basics/task_dto.dart

factory TaskDto.fromJson(Map<String, dynamic> json, {String path = ''}) =>
    TaskDto(
      id: requireString(json, 'id', path: path),
      palletCode: requireString(json, 'palletCode', path: path),
      quantity: requireInt(json, 'quantity', path: path),
      status: requireString(json, 'status', path: path),
      createdAt: requireUtcTime(json, 'createdAt', path: path),
      note: readString(json, 'note', path: path),
    );

สังเกตว่า ไม่มี if เลยสักตัว — เงื่อนไขทั้งหมดอยู่ในตัวช่วย ทำให้ตัวแปลงอ่านแล้วเห็นทันทีว่าเซิร์ฟเวอร์ต้องส่งอะไรมาบ้าง

ตัวช่วยความหมาย
requireXxxต้องมี ถ้าไม่มีคือผิดสัญญา
readXxxมีก็ได้ ไม่มีก็ได้

ชื่อสองแบบนี้ทำให้อ่านโค้ดแล้วรู้ทันทีว่า field ไหนจำเป็น โดยไม่ต้องเปิดเอกสาร

6. แยกให้ออก: ไม่มีคีย์ กับ มีคีย์แต่เป็น null

สองอย่างนี้ต่างกัน และ json['x'] == null แยกไม่ออก

{"note": null}   →  json['note'] เป็น null
{}               →  json['note'] เป็น null เหมือนกัน

ใน lab จึงใช้ค่าพิเศษแทน “ไม่มีคีย์”

/// ค่าที่ใช้แทน "ไม่มีคีย์นี้เลย" ซึ่งต่างจาก "มีคีย์แต่ค่าเป็น null"
const Object _absent = Object();
Object? _require(Map<String, dynamic> json, String key, String path) {
  final value = _raw(json, key);
  if (identical(value, _absent)) {
    throw ContractException(_at(path, key), 'ไม่มีคีย์นี้ในข้อมูลที่ได้รับ');
  }
  if (value == null) {
    throw ContractException(_at(path, key), 'ค่าต้องไม่เป็น null');
  }
  return value;
}

ทำไมต้องแยก — เพราะสองกรณีนี้บอกคนละเรื่องตอนไล่หาสาเหตุ

ข้อความบอกอะไร
“ไม่มีคีย์นี้”เซิร์ฟเวอร์คนละรุ่น หรือเราสะกดชื่อคีย์ผิด
“ค่าต้องไม่เป็น null”เซิร์ฟเวอร์รู้จักคีย์นี้ แต่ข้อมูลในฐานข้อมูลไม่ครบ

เทสต์ใน lab ตรึงความต่างนี้ไว้สองข้อ

test('ไม่มีคีย์ ต้องบอกว่าคีย์ไหนหาย', ...);
test('คีย์มีอยู่แต่ค่าเป็น null ต้องบอกคนละอย่างกับคีย์หาย', ...);

7. ทิศทางขึ้นกับทิศทางลง เป็นคนละก้อน

/// คำสั่งที่ส่งขึ้นไป — คนละก้อนกับที่รับลงมา
class CompleteTaskRequest {
  const CompleteTaskRequest({
    required this.taskId,
    required this.quantity,
    required this.commandId,
    this.note,
  });

  Map<String, Object?> toJson() => {
    'taskId': taskId,
    'quantity': quantity,
    'commandId': commandId,
    'note': note,
  };
}

เทียบกับ TaskDto ที่รับลงมา

fieldตอนรับตอนส่ง
id / taskIdมี ชื่อ idมี ชื่อ taskId
createdAtมีไม่มี — เซิร์ฟเวอร์เป็นคนกำหนด
commandIdไม่มีมี — เพื่อกันการบันทึกซ้ำ (4.14)
statusมีไม่มี — การส่งคำสั่งคือการขอเปลี่ยนสถานะอยู่แล้ว

การใช้คลาสเดียวทั้งขึ้นและลง จะบังคับให้ทุก field เป็น nullable เพื่อให้ใช้ได้ทั้งสองทาง ซึ่งทำให้ไม่มีอะไรบังคับได้อีกเลยว่าตอนส่งต้องมีอะไรบ้าง

8. ส่งคีย์ที่ค่าเป็น null หรือไม่ส่งเลย

// ส่งคีย์ไปเสมอแม้ค่าเป็น null เพื่อให้เซิร์ฟเวอร์แยกได้ว่า
// "ตั้งใจล้างหมายเหตุ" ต่างจาก "ไม่ได้แตะหมายเหตุ"
'note': note,
สิ่งที่ส่งเซิร์ฟเวอร์ควรเข้าใจว่า
{"note": "ของแตก"}ตั้งหมายเหตุเป็นข้อความนี้
{"note": null}ล้างหมายเหตุทิ้ง
ไม่มีคีย์ note เลยไม่ได้แตะหมายเหตุ ใช้ค่าเดิม

สามความหมายนี้ต่างกัน และต้องตกลงกับฝั่งหลังบ้านให้ชัดว่าใช้แบบไหน — ถ้าไม่ตกลงกัน จะเจออาการ “แก้หมายเหตุแล้วหายไปเอง” ซึ่งไล่หาสาเหตุยากมาก

เทสต์ใน lab จึงตรึงคีย์ที่ส่งไว้ทั้งหมด รวมคีย์ที่ค่าเป็น null ด้วย

expect(request.toJson(), {
  'taskId': 'T-001',
  'quantity': 12,
  'commandId': 'cmd-1',
  'note': null,
});

สังเกตว่าเทียบทั้ง map ไม่ใช่เช็กทีละคีย์ — ทำให้จับได้ทั้งคีย์ที่หายและคีย์ที่เกินมาโดยไม่ตั้งใจ

9. คีย์ที่ไม่รู้จัก ต้องไม่ทำให้พัง

วันหนึ่งฝั่งหลังบ้านเพิ่ม field ใหม่

{
  "id": "T-001",
  "quantity": 12,
  "priority": "HIGH"      ← ของใหม่ที่แอปรุ่นนี้ไม่รู้จัก
}

ตัวแปลงที่เขียนแบบนี้จะไม่สนใจคีย์ที่ไม่ได้อ่าน จึงทำงานต่อได้ตามปกติ — และนั่นถูกต้อง

ถ้าพังเมื่อเจอคีย์ใหม่ผลที่ตามมา
ทุกเครื่องที่ยังไม่อัปเดตใช้งานไม่ได้ทันทีต้องหยุดทั้งคลังเพื่ออัปเดตแอป
ฝั่งหลังบ้านเพิ่มอะไรไม่ได้เลยทุกการเปลี่ยนแปลงต้องปล่อยพร้อมกันสองฝั่ง

กฎที่ได้คือ เข้มกับสิ่งที่เราต้องใช้ ผ่อนกับสิ่งที่เราไม่ได้ใช้

สถานการณ์ทำอย่างไร
คีย์ที่ต้องใช้ หายไปพัง — เพราะทำงานต่อไม่ได้
คีย์ที่ต้องใช้ ชนิดผิดพัง
คีย์ใหม่ที่ไม่รู้จักไม่สนใจ
คีย์ที่ไม่จำเป็น หายไปใช้ค่าตั้งต้นตามที่ตกลง

10. เขียนเองหรือให้เครื่องสร้างให้

Dart มีเครื่องมือสร้าง fromJson/toJson ให้อัตโนมัติ

เขียนเองให้เครื่องสร้าง
โค้ดที่ต้องเขียนมากกว่าน้อยกว่ามาก
ควบคุมการแปลงค่าได้แค่ไหนทั้งหมดต้องเขียนตัวแปลงเสริมอยู่ดี
ข้อความผิดพลาดกำหนดเองได้ตามที่เครื่องมือให้
เข้าใจว่าเกิดอะไรขึ้นเห็นหมดต้องไปอ่านไฟล์ที่ถูกสร้าง

บทนี้ให้เขียนเองก่อน เพราะ เครื่องมือช่วยลดการพิมพ์ แต่ไม่ได้ตัดสินใจเรื่องยากแทนเรา — เรื่องที่ยากคือรับ "12" ไหม เวลาที่ไม่มีเขตเวลาทำอย่างไร สถานะที่ไม่รู้จักแปลว่าอะไร ซึ่งทั้งหมดยังต้องเราตัดสินเองอยู่ดี

การสร้างอัตโนมัติจะเรียนใน Part 30

11. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
type 'Null' is not a subtype of type 'String'ใช้ as ตรง ๆอ่านผ่านตัวช่วยที่ตรวจก่อน
รู้ว่าพังแต่ไม่รู้ว่าตรงไหนไม่มี path ในข้อความส่ง path ลงไปทุกชั้น
แก้หมายเหตุแล้วหายไปเองไม่ได้ตกลงว่า null แปลว่าอะไรตกลงกับหลังบ้านให้ชัด
ฝั่งหลังบ้านเพิ่ม field แล้วแอปพังทั้งคลังเข้มกับคีย์ที่ไม่ได้ใช้ไม่สนใจคีย์ที่ไม่รู้จัก
ตอนส่งลืมใส่ field ที่จำเป็นใช้คลาสเดียวทั้งขึ้นและลงแยกคลาสตามทิศทาง
ทุก model เขียนกฎการแปลงเอง คนละแบบไม่มีตัวช่วยกลางรวมไว้ไฟล์เดียว
บั๊กในโค้ดเราถูกกลืนเป็น “ข้อมูลผิด”ดักกว้างเกินไปใช้ชนิดข้อผิดพลาดของตัวเอง

12. แบบฝึกหัด

เซิร์ฟเวอร์ส่งข้อมูลพาเลทมาแบบนี้

{
  "palletCode": "PAL-0101",
  "locationCode": "A-01-02",
  "totalWeight": 245.5,
  "sealNo": "",
  "lastMovedAt": "2026-08-15T03:00:00Z"
}

โดยที่ locationCode เป็น null ได้เมื่อพาเลทยังไม่ถูกเก็บเข้าช่อง และ sealNo เป็นข้อความว่างเมื่อไม่มีซีล

  1. เขียน fromJson
  2. เขียน toJson สำหรับคำสั่งย้ายพาเลท ซึ่งส่ง palletCode, toLocationCode และ commandId
  3. ตอบว่าถ้าเซิร์ฟเวอร์ส่ง "totalWeight": "245.5" เป็นข้อความ ควรรับหรือปฏิเสธ
ดูแนวคำตอบและเหตุผล

ข้อ 1 — fromJson

class PalletDto {
  const PalletDto({
    required this.palletCode,
    required this.totalWeight,
    required this.lastMovedAt,
    this.locationCode,
    this.sealNo,
  });

  factory PalletDto.fromJson(Map<String, dynamic> json, {String path = ''}) =>
      PalletDto(
        palletCode: requireString(json, 'palletCode', path: path),
        totalWeight: requireDouble(json, 'totalWeight', path: path),
        lastMovedAt: requireUtcTime(json, 'lastMovedAt', path: path),
        locationCode: readString(json, 'locationCode', path: path),
        sealNo: readString(json, 'sealNo', path: path),
      );

  final String palletCode;
  final double totalWeight;
  final DateTime lastMovedAt;
  final String? locationCode;
  final String? sealNo;
}

จุดที่ต้องสังเกตคือ sealNo — โจทย์บอกว่าเซิร์ฟเวอร์ส่งข้อความว่างมาเมื่อไม่มีซีล และ readString แปลงข้อความว่างเป็น null ให้แล้ว

final trimmed = value.trim();
return trimmed.isEmpty ? null : trimmed;

ผลคือทั้งฝั่งที่ส่ง null และฝั่งที่ส่ง "" ให้ผลเหมือนกันในแอป ซึ่งถูกต้องเพราะทั้งคู่แปลว่า “ไม่มีซีล”

ข้อ 2 — toJson

class MovePalletRequest {
  const MovePalletRequest({
    required this.palletCode,
    required this.toLocationCode,
    required this.commandId,
  });

  final String palletCode;
  final String toLocationCode;
  final String commandId;

  Map<String, Object?> toJson() => {
    'palletCode': palletCode,
    'toLocationCode': toLocationCode,
    'commandId': commandId,
  };
}

สังเกตว่า ไม่มี field ไหนเป็น nullable เลย — เพราะการย้ายพาเลทโดยไม่ระบุปลายทางไม่มีความหมาย การประกาศเป็น required ทำให้ compiler กันไม่ให้ลืมตั้งแต่ตอนเขียน

และไม่ส่ง fromLocationCode ไปด้วย เพราะเซิร์ฟเวอร์รู้อยู่แล้วว่าพาเลทนี้อยู่ไหน — การส่งไปมีแต่จะทำให้เกิดคำถามว่าถ้าไม่ตรงกับที่เซิร์ฟเวอร์มีจะเชื่อฝั่งไหน

ข้อ 3 — "245.5" เป็นข้อความ

คำตอบคือ รับได้ แต่ต้องเป็นการตัดสินใจที่บันทึกไว้ ไม่ใช่เพราะบังเอิญ

ใน lab requireDouble รับข้อความที่แปลงเป็นตัวเลขได้

double? _toDouble(Object? value) => switch (value) {
  num n => n.toDouble(),
  String s => double.tryParse(s.trim()),
  _ => null,
};
เหตุผลที่รับเหตุผลที่ไม่ควรรับ
ระบบเก่าบางตัวส่งตัวเลขเป็นข้อความจริง ๆปิดบังปัญหาฝั่งหลังบ้าน
ปฏิเสธแล้วคนหน้างานทำงานไม่ได้ ทั้งที่ข้อมูลอ่านออกวันหนึ่งจะเจอ "245,5" ที่ใช้จุลภาค แล้วพัง

    expect(TaskDto.fromJson(taskJson({'quantity': raw})).quantity, 12);
  }
});

requireInt ยังปฏิเสธ 1.9

if (parsed != parsed.roundToDouble()) {
  throw ContractException(_at(path, key), 'ต้องเป็นจำนวนเต็ม แต่ได้ $value');
}

ความต่างคือ — "12" กับ 12 เป็นค่าเดียวกันที่เขียนคนละแบบ ส่วน 1.9 กับ 2 เป็นคนละค่า การปัดให้จึงเป็นการตัดสินใจแทนธุรกิจ ซึ่งในงานคลังแปลว่าของขาดหรือของเกิน

หลักที่ได้: ผ่อนได้กับ รูปแบบการเขียน แต่ห้ามผ่อนกับ ความหมายของค่า

13. สรุปท้ายบท

  • fromJson คือด่านที่ข้อมูลจากภายนอกเข้ามา ต้องตรวจก่อนเชื่อ
  • อย่าใช้ as ตรง ๆ เพราะข้อความผิดพลาดบอกอะไรไม่ได้
  • รวมกฎการอ่านไว้ที่เดียว และให้ข้อผิดพลาดพก path มาด้วย
  • ใช้ชนิดข้อผิดพลาดของตัวเอง เพื่อแยกจากบั๊กและจากปัญหาเครือข่าย
  • แยก requireXxx กับ readXxx ให้อ่านโค้ดแล้วรู้ว่า field ไหนจำเป็น
  • “ไม่มีคีย์” กับ “มีคีย์แต่เป็น null” ต้องแยกกัน
  • คลาสที่ส่งขึ้นกับที่รับลงเป็นคนละก้อน
  • ตกลงกับหลังบ้านว่า null ในคำสั่งแปลว่าล้างค่า หรือไม่แตะ
  • เข้มกับสิ่งที่ต้องใช้ ผ่อนกับสิ่งที่ไม่ได้ใช้
  • ผ่อนได้กับรูปแบบการเขียน แต่ห้ามผ่อนกับความหมายของค่า
