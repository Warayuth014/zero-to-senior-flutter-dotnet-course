1. ปัญหาที่บทนี้แก้

ใบรับเข้าใบเดียวมีบรรทัดสินค้าหลายบรรทัด

3. Mental model: ที่อยู่ของค่าที่พัง

{                              ← ''
  "details": [                 ← 'details'
    { ... },                   ← 'details[0]'
    { ... },                   ← 'details[1]'
    { "productCode": "..." }   ← 'details[2]'
                                  └─ ขาด → 'details[2].expectedQuantity'
  ]
}
ประโยคเดียวที่ต้องจำ: ทุกครั้งที่เดินลึกเข้าไปอีกชั้น ต้องต่อ path ให้ยาวขึ้นด้วย — ไม่งั้นข้อความผิดพลาดจะบอกได้แค่ชื่อ field ที่อาจซ้ำกันเป็นสิบที่

4. ตรวจตัวครอบก่อนเดินเข้าไป

เปิด lib/json_basics/json_read.dart

/// ตรวจว่าเป็น object จริงก่อนอ่านข้างใน
Map<String, dynamic> requireMap(Object? value, String path) {
  if (value is Map<String, dynamic>) return value;
  throw ContractException(path, 'ต้องเป็น object แต่ได้ ${value.runtimeType}');
}

/// ตรวจว่าเป็น array จริงก่อนวนอ่าน
List<Object?> requireList(Map<String, dynamic> json, String key, {
  String path = '',
}) {
  final value = _require(json, key, path);
  if (value is List) return value;
  throw ContractException(
    _at(path, key),
    'ต้องเป็น array แต่ได้ ${value.runtimeType}',
  );
}

เทียบกับการ cast ตรง ๆ

as ListrequireList
ถ้าเซิร์ฟเวอร์ส่ง object มาแทน arraytype '_Map' is not a subtype of type 'List'“ที่ items: ต้องเป็น array แต่ได้ _Map”
รู้ว่าพังที่คีย์ไหนไม่รู้รู้

เทสต์ตรึงไว้ว่าต้องพังที่ตัวครอบ ไม่ใช่ที่อื่น

test('items ที่ไม่ใช่ array ต้องพังที่ items ไม่ใช่ที่อื่น', () {
  expect(
    () => TaskPageDto.fromJson({'items': {'0': {}}, 'totalCount': 1}),
    throwsA(isA<ContractException>().having((e) => e.path, 'path', 'items')),
  );
});

5. ส่ง path ลงไปทุกชั้น

เปิด lib/json_basics/task_dto.dart

factory TaskPageDto.fromJson(Map<String, dynamic> json) {
  final rawItems = requireList(json, 'items');
  final items = <TaskDto>[];
  for (var index = 0; index < rawItems.length; index++) {
    // ส่ง path ลงไปด้วย เพื่อให้ error บอกได้ว่าพังที่แถวไหน
    final path = 'items[$index]';
    items.add(TaskDto.fromJson(requireMap(rawItems[index], path), path: path));
  }
  ...
}

สังเกตสามอย่าง

สิ่งที่เห็นทำไม
ใช้ for ธรรมดา ไม่ใช้ mapต้องใช้ index เพื่อสร้าง path
requireMap(rawItems[index], path)ตรวจว่าสมาชิกเป็น object ก่อน
ส่ง path: ต่อเข้าไปใน fromJsonให้ชั้นในต่อ path ได้อีก

ผลลัพธ์คือข้อความที่ชี้ตำแหน่งได้จริง

test('แถวที่พังต้องบอกว่าเป็นแถวที่เท่าไหร่', () {
  ...
  expect(
    () => TaskPageDto.fromJson(json),
    throwsA(isA<ContractException>()
        .having((e) => e.path, 'path', 'items[1].quantity')),
  );
});

8. สมาชิกที่เสียในรายการ: ข้ามหรือปฏิเสธ

คำถามที่ต้องตัดสินตอนออกแบบ ไม่ใช่ตอนเจอปัญหา

ทางเลือกผลที่ได้เหมาะกับ
ปฏิเสธทั้งก้อนผู้ใช้ไม่เห็นอะไรเลย แต่ไม่มีข้อมูลผิดงานที่กระทบสต๊อก
ข้ามแถวที่เสียเห็นแถวที่เหลือ แต่ไม่รู้ว่ามีแถวหายรายการที่แค่ดูเฉย ๆ
ข้ามแล้วบอกจำนวนที่ข้ามเห็นแถวที่เหลือ และรู้ว่าไม่ครบทางสายกลาง

lab เลือกปฏิเสธทั้งก้อน เพราะเป็นรายการงานที่ต้องทำจริง

if (details.isEmpty) {

การข้ามแถวเงียบ ๆ ในงานคลังอันตรายมาก — ถ้าใบรับเข้ามี 20 บรรทัดแต่แสดงแค่ 19 คนหน้างานจะรับของไม่ครบแล้วปิดใบว่าเสร็จ โดยที่ไม่มีอะไรบอกว่ามีของหายไปหนึ่งบรรทัด

ถ้าจำเป็นต้องข้ามจริง ๆ ต้องบันทึกไว้เสมอว่าข้ามอะไรไป เพื่อให้ตามไปแก้ที่ต้นทางได้

12. แบบฝึกหัด

เซิร์ฟเวอร์ส่งข้อมูลรอบส่งของมาแบบนี้

{
  "shipmentNo": "SH-2026-0091",
  "vehicle": {
    "plateNo": "70-1234",
    "driver": {"name": "สมชาย ใจดี", "phone": null}
  },
  "stops": [
    {
      "sequence": 1,
      "customerName": "ร้านค้า ก",
      "orders": [
        {"orderNo": "SO-001", "boxCount": 12},
        {"orderNo": "SO-002", "boxCount": 8}
      ]
    },
    {
      "sequence": 2,
      "customerName": "ร้านค้า ข",
      "orders": []
    }
  ]
}
  1. ระบุว่าควรมีกี่คลาส และแต่ละคลาสรับผิดชอบอะไร
  2. เขียน path ที่ควรได้ ถ้า boxCount ของ SO-002 หายไป
  3. ตัดสินว่า orders ที่ว่างของจุดส่งที่สอง ควรยอมรับหรือปฏิเสธ
ดูแนวคำตอบและเหตุผล

ข้อ 1 — คลาสที่ควรมี

คลาสตรงกับ object ไหน
ShipmentDtoก้อนนอกสุด
VehicleDtovehicle
DriverDtovehicle.driver
StopDtoสมาชิกใน stops
StopOrderDtoสมาชิกใน stops[].orders

ห้าคลาส ตรงกับจำนวน object ที่มีใน JSON พอดี ตามกฎในหัวข้อ 10

factory ShipmentDto.fromJson(Map<String, dynamic> json, {String path = ''}) {
  final rawStops = requireList(json, 'stops', path: path);
  final stops = <StopDto>[];
  for (var index = 0; index < rawStops.length; index++) {
    final stopPath = path.isEmpty ? 'stops[$index]' : '$path.stops[$index]';
    stops.add(
      StopDto.fromJson(requireMap(rawStops[index], stopPath), path: stopPath),
    );
  }

  final vehiclePath = path.isEmpty ? 'vehicle' : '$path.vehicle';
  return ShipmentDto(
    shipmentNo: requireString(json, 'shipmentNo', path: path),
    vehicle: VehicleDto.fromJson(
      requireMap(json['vehicle'], vehiclePath),
      path: vehiclePath,
    ),
    stops: stops,
  );
}

ส่วน phone: null ของคนขับใช้ readString ได้เลย เพราะคนขับที่ไม่ได้ให้เบอร์เป็นเรื่องปกติ

ข้อ 2 — path ที่ควรได้

stops[0].orders[1].boxCount

อ่านแล้วเดินตามได้ทันทีโดยไม่ต้องเดา — จุดส่งแรก ใบสั่งที่สอง field boxCount

เทียบกับสิ่งที่ได้ถ้าไม่ส่ง path

ข้อความต้องทำอะไรต่อ
boxCount: ไม่มีคีย์นี้เปิด payload ดิบมาไล่หาเองทุกจุดส่ง
stops[0].orders[1].boxCount: ไม่มีคีย์นี้เปิดไปดูตรงจุดนั้นได้เลย

ความต่างนี้ยิ่งสำคัญเมื่อรอบส่งมี 30 จุด จุดละ 10 ใบ

ข้อ 3 — orders ที่ว่าง

ต้องถามก่อนว่า จุดส่งที่ไม่มีใบสั่งเลย เกิดขึ้นได้ตามปกติไหม

ถ้าเป็นไปได้ตัวอย่างสถานการณ์
จุดแวะรับของกลับไปรับของคืนอย่างเดียว ไม่มีของไปส่ง
จุดที่ใบถูกยกเลิกไปแล้วลูกค้ายกเลิกหลังจัดรอบแล้ว
จุดพักรถไม่ใช่จุดส่งของ

คำตอบคือยอมรับ แต่ต้องยืนยันกับฝั่งหลังบ้านก่อน — และถ้ายอมรับ ต้องทำสองอย่างเพิ่ม

ต้องทำเพราะ
หน้าจอต้องบอกให้ชัดว่าจุดนี้ไม่มีของส่งไม่งั้นคนขับจะคิดว่าแอปโหลดไม่ครบ
ต้องรู้ว่าทำไมถึงว่าง“ยกเลิกแล้ว” กับ “จุดพักรถ” ต้องแสดงต่างกัน

ข้อสองมักเผยว่าสัญญายังขาดข้อมูล — ถ้าตอบไม่ได้ว่าจุดนั้นว่างเพราะอะไร แปลว่าควรขอ field เพิ่ม เช่น stopType ซึ่งเป็นข้อสรุปเดียวกับ 6.3 แบบฝึกหัดข้อ 3

เปรียบเทียบกับใบรับเข้าใน lab

ใบรับเข้าไม่มีบรรทัดจุดส่งไม่มีใบสั่ง
เป็นสถานะปกติไหมไม่ — ใบที่ไม่มีของไม่มีความหมายได้ — มีเหตุผลที่อธิบายได้
ทำอย่างไรปฏิเสธยอมรับ

บทเรียนที่ควรจำ: คำถาม “รายการว่างยอมรับได้ไหม” ไม่มีคำตอบสำเร็จรูป ต้องกลับไปถามธุรกิจทุกครั้ง — และการตอบว่า “ไม่แน่ใจ” คือสัญญาณว่ายังไม่ควรเขียนโค้ดส่วนนั้น

13. สรุปท้ายบท

  • ตรวจว่าตัวครอบเป็น object หรือ array จริง ก่อนเดินเข้าไป
  • ต่อ path ทุกครั้งที่ลงลึกอีกชั้น
  • ใช้ for พร้อม index แทน map เพื่อสร้าง path ของแต่ละสมาชิก
  • ทุก object ใน JSON ควรมีคลาสของตัวเอง
  • สมาชิกที่เสีย — ข้ามเงียบ ๆ อันตรายที่สุดในงานคลัง
  • รายการว่างเป็นความสำเร็จหรือความล้มเหลว ขึ้นกับว่าเป็นสถานะปกติของธุรกิจไหม
  • object ชั้นในที่อาจไม่มี ต้องตรวจก่อนแล้วคืน null
  • ข้อความผิดพลาดที่ดีต้องบอกได้ว่า ชั้นไหน แถวไหน field ไหน
