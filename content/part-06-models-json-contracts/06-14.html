3. Dart DTO

class TaskDto {
  TaskDto({required this.id, required this.qty, required this.status});
  factory TaskDto.fromJson(Map<String,dynamic> j) => TaskDto(
    id: requireString(j,'id'), qty: requireDouble(j,'qty'),
    status: requireString(j,'status'));
  final String id; final double qty; final String status;
}
