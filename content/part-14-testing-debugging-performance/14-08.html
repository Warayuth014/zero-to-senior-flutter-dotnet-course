ระดับ: กลางใช้เวลา: 50–65 นาทีลงมือทำ: หา widget ที่หายากในเทสต์ของตัวเอง
ต้องผ่านอะไรมาก่อน: 14.7
จบบทนี้แล้วจะทำได้: หา widget ในต้นไม้ได้แม่นยำ และสั่งให้มันทำงานเหมือนผู้ใช้

1. ตัวหาที่ใช้บ่อย

เขียนหาใช้เมื่อ
find.byKey(const Key('scan-button'))ตาม keyปกติที่สุด แนะนำ
find.text('พาเลท')ตามข้อความเมื่อข้อความคือสิ่งที่ทดสอบ
find.byType(FilledButton)ตามชนิดเมื่อมีตัวเดียวในจอ
find.byIcon(Icons.refresh)ตามไอคอนปุ่มไอคอนที่ไม่มีข้อความ
find.byTooltip('โหลดใหม่')ตาม tooltipปุ่มไอคอนที่ตั้ง tooltip ไว้
find.byWidgetPredicate(...)ตามเงื่อนไขเมื่อต้องตรวจค่าข้างใน
// ตรวจค่าข้างใน widget ที่หาไม่ได้ด้วยวิธีอื่น
expect(
  find.byWidgetPredicate(
    (w) => w is Image && w.width == 56 && w.height == 56,
  ),
  findsWidgets,
);

2. อ่านค่าจาก widget ที่เจอ

final app = tester.widget<MaterialApp>(find.byType(MaterialApp));
// ถ้าไม่ส่ง locale ลงไป widget สำเร็จรูปของ Material จะยังเป็นอังกฤษ
expect(app.locale, const Locale('th'));

tester.widget<T>() คืนตัว widget จริง ทำให้ตรวจค่าที่ไม่ได้แสดงบนหน้าจอได้ — เช่น locale, onPressed ที่เป็น null, หรือค่าที่ส่งเข้า constructor

เมธอดคืนอะไร
tester.widget<T>(finder)widget ตัวเดียว — โยนถ้าเจอมากกว่าหนึ่ง
tester.widgetList<T>(finder)ทุกตัวที่เจอ
tester.getSize(finder)ขนาดจริงหลังจัดวางแล้ว
tester.getTopLeft(finder)ตำแหน่งบนจอ
testWidgets('ปุ่มสลับภาษากดโดนแม้ใส่ถุงมือ', (tester) async {
  final size = tester.getSize(find.byKey(const Key('language-button')));
  expect(size.height, greaterThanOrEqualTo(PdaMetrics.minTapTarget));
  expect(size.width, greaterThanOrEqualTo(PdaMetrics.minTapTarget));
});

เทสต์นี้ตรวจสิ่งที่ตาคนมองแล้วบอกไม่ได้ — ปุ่มที่ดูใหญ่พออาจมีพื้นที่กดจริงเล็กกว่าที่เห็น

3. ผสมตัวหาเข้าด้วยกัน

// หาข้อความที่อยู่ในการ์ดใบนั้นเท่านั้น
find.descendant(
  of: find.byKey(const Key('pallet-PLT-001')),
  matching: find.text('ล็อกแล้ว'),
)

// หา widget ที่มีลูกเป็นสิ่งนี้
find.ancestor(
  of: find.text('PLT-001'),
  matching: find.byType(Card),
)

find.descendant จำเป็นเมื่อมีข้อความซ้ำกันหลายที่

// ลิสต์มีคำว่า "ล็อกแล้ว" หลายแถว — ตรวจแบบนี้จะกำกวม
expect(find.text('ล็อกแล้ว'), findsOneWidget);   // ล้มเมื่อมีสองแถว

// ตรวจว่าแถวนี้ล็อกแล้ว
expect(
  find.descendant(
    of: find.byKey(const Key('pallet-PLT-001')),
    matching: find.text('ล็อกแล้ว'),
  ),
  findsOneWidget,
);

4. จำนวนที่คาด

เขียนหมายถึง
findsOneWidgetหนึ่งตัวพอดี
findsNothingไม่มีเลย
findsWidgetsอย่างน้อยหนึ่งตัว
findsNWidgets(3)สามตัวพอดี
findsAtLeastNWidgets(2)อย่างน้อยสองตัว

ของจริงใน wms_absolute_mobile ใช้ findsNWidgets พร้อมคอมเมนต์อธิบาย

expect(find.text('SIGN IN'), findsNWidgets(2)); // heading + button

คอมเมนต์นั้นสำคัญ — ถ้าไม่มี คนที่มาอ่านทีหลังจะไม่รู้ว่า 2 มาจากไหน และไม่กล้าแก้เมื่อเลขเปลี่ยน

5. สั่งให้ทำงาน

เขียนทำอะไร
tester.tap(finder)แตะ
tester.enterText(finder, 'PLT-001')พิมพ์ลงช่อง
tester.drag(finder, Offset(0, -300))ลาก — ใช้เลื่อนลิสต์
tester.longPress(finder)กดค้าง
tester.pageBack()กดย้อนกลับ
tester.testTextInput.receiveAction(TextInputAction.done)กด Enter บนแป้นพิมพ์

ทุกตัวต้องตามด้วย pump ถ้าจะตรวจผลบนหน้าจอ

await tester.tap(find.byKey(const Key('language-button')));
await tester.pumpAndSettle();

expect(find.text('พาเลท'), findsOneWidget);

แถวสุดท้ายในตารางสำคัญสำหรับ PDA — เครื่องสแกนส่งข้อมูลเหมือนแป้นพิมพ์แล้วจบด้วย Enter ซึ่งจำลองได้แบบนี้

await tester.enterText(find.byKey(const Key('scan-field')), 'PLT-001');
await tester.testTextInput.receiveAction(TextInputAction.done);
await tester.pump();

expect(store.lastScanned, 'PLT-001');

6. widget ที่อยู่นอกจอ

นี่คือกับดักที่เสียเวลาหาสาเหตุนานที่สุดในบทนี้

// find.byKey ข้าม widget ที่ถูกซ่อนอยู่ (offstage) โดยปริยาย
expect(find.byKey(const Key('task-list')), findsNothing);        // ล้ม

// บอกให้หารวมตัวที่ซ่อนอยู่ด้วย
expect(
  find.byKey(const Key('task-list'), skipOffstage: false),
  findsOneWidget,
);
เกิดเมื่อตัวอย่าง
IndexedStackแท็บที่ไม่ได้เลือกอยู่ ยังอยู่ในต้นไม้แต่ถูกซ่อน (5.6)
Offstagewidget ที่ถูกซ่อนโดยตั้งใจ
เส้นทางที่ถูก push ทับจอเก่ายังอยู่ใต้จอใหม่

ส่วน widget ที่อยู่ในลิสต์แต่ยังไม่ถูกเลื่อนมาถึง เป็นคนละเรื่อง — มันไม่มีอยู่ในต้นไม้เลย ต้องเลื่อนไปหาก่อน

await tester.scrollUntilVisible(
  find.byKey(const Key('pallet-PLT-450')),
  200,                                    // เลื่อนทีละ 200
  scrollable: find.byType(Scrollable),
);
expect(find.byKey(const Key('pallet-PLT-450')), findsOneWidget);

7. การกดที่ไม่โดน

// ปุ่มถูก disable แล้ว แต่ยิง tap ซ้ำที่ตำแหน่งเดิมเพื่อจำลองการกดรัว
await tester.tap(find.byKey(const Key('hold-PLT-001')), warnIfMissed: false);

warnIfMissed: false ใช้เมื่อตั้งใจกดสิ่งที่รับไม่ได้ ซึ่งเป็นการจำลองผู้ใช้ที่กดรัว ๆ บน PDA

ถ้าไม่ใส่ จะได้คำเตือนยาว ๆ ที่บอกว่า “widget ที่กดไม่ได้รับเหตุการณ์” ซึ่งในกรณีนี้เป็นสิ่งที่ต้องการ

แต่ระวัง — ถ้าใส่ warnIfMissed: false โดยไม่ตั้งใจ จะกลบข้อผิดพลาดจริงที่ว่า “กดไม่โดนเพราะมีอะไรบัง” ซึ่งเป็นบั๊กที่ควรรู้

8. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
หาไม่เจอทั้งที่มั่นใจว่ามีอยู่ใน IndexedStack ที่ไม่ได้เลือกskipOffstage: false
หาไม่เจอเพราะอยู่ล่างสุดของลิสต์ยังไม่ถูกสร้างscrollUntilVisible
เจอมากกว่าหนึ่งตัวข้อความซ้ำในหลายแถวfind.descendant
กดแล้วไม่มีอะไรเกิดลืม pumpใส่หลังทุกการกระทำที่จะตรวจผล
คำเตือน “tap missed”กดสิ่งที่ถูก disable หรือมีอะไรบังตรวจว่าตั้งใจ ถ้าใช่ใส่ warnIfMissed: false
เทสต์พังเมื่อสลับภาษาหาด้วย find.textใส่ Key

9. แบบฝึกหัด

เทสต์นี้ล้มด้วยข้อความว่า “Found 0 widgets” ทั้งที่เปิดแอปดูแล้วเห็นปุ่มอยู่

testWidgets('กดยืนยันแล้วส่งคำสั่ง', (tester) async {
  await tester.pumpWidget(wrap(ReceivingScreen(store: store)));

  await tester.tap(find.text('ยืนยันการรับ'));
  expect(repository.confirmed, isTrue);
});

หาสาเหตุที่เป็นไปได้ทั้งหมด

แนวคำตอบ

ห้าสาเหตุ เรียงจากที่พบบ่อยที่สุด

สาเหตุตรวจอย่างไรแก้
ปุ่มยังไม่ปรากฏเพราะข้อมูลยังโหลดไม่เสร็จเพิ่ม await tester.pumpAndSettle()ใส่ pump ก่อนหา
ปุ่มอยู่ล่างจอ ยังไม่ถูกเลื่อนมาถึงลอง scrollUntilVisibleเลื่อนก่อน หรือขยายจอในเทสต์
ข้อความไม่ตรง เช่นมีเว้นวรรคหรือขึ้นบรรทัดใหม่debugDumpApp() แล้วอ่านใช้ Key
แอปเป็นภาษาอังกฤษในเทสต์ดูว่า controller ตั้งภาษาอะไรตั้งภาษาใน setUp หรือใช้ Key
ข้อความถูกแบ่งเป็นหลาย TextSpanfind.text หาเฉพาะข้อความที่อยู่ใน widget เดียวใช้ find.textContaining หรือ Key

วิธีหาที่เร็วที่สุด คือดูว่าต้นไม้มีอะไรอยู่จริง

await tester.pumpWidget(wrap(ReceivingScreen(store: store)));
debugDumpApp();       // พิมพ์ต้นไม้ทั้งหมดออกมา

หรือถ้าต้นไม้ใหญ่เกินอ่าน ให้จำกัดขอบเขต

// ดูว่ามีข้อความอะไรอยู่ในจอบ้าง
final texts = tester.widgetList<Text>(find.byType(Text))
    .map((t) => t.data)
    .toList();
debugPrint(texts.join(' | '));

ข้อสรุปที่ควรจำ: สี่ในห้าสาเหตุนี้หายไปทันทีถ้าใช้ Key แทน find.text — ซึ่งเป็นเหตุผลที่ 14.7 แนะนำให้ใส่ Key กับสิ่งที่เทสต์ต้องหา

สาเหตุแรกยังอยู่ และเป็นเรื่องของจังหวะการ pump ซึ่ง 14.10 จะพูดถึง

10. สรุป

  • find.byKey เป็นตัวหาที่แนะนำ เพราะไม่พังตามคำและตามภาษา
  • tester.widget<T>() อ่านค่าที่ไม่ได้แสดงบนหน้าจอได้
  • tester.getSize ตรวจสิ่งที่ตามองแล้วบอกไม่ได้
  • find.descendant จำเป็นเมื่อข้อความซ้ำหลายแถว
  • widget ใน IndexedStack ต้องหาด้วย skipOffstage: false
  • widget ที่ยังไม่ถูกเลื่อนมาถึง ไม่มีอยู่ในต้นไม้เลย
