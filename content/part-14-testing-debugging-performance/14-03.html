1. ทำไมต้องมีของปลอม

store ที่จะทดสอบ ต้องการ repository ซึ่งต้องการ ApiClient ซึ่งต้องการเซิร์ฟเวอร์ — ถ้าไม่ตัดตรงไหนสักที่ เทสต์หนึ่งข้อจะต้องมีเซิร์ฟเวอร์รันอยู่

Store  →  Repository  →  ApiClient  →  เซิร์ฟเวอร์
   ↑          ↑              ↑
   │          │              └── ตัดที่นี่ = MockClient (14.4)
   │          └────────────────── ตัดที่นี่ = repository ปลอม (บทนี้)
   └───────────────────────────── ทดสอบตัวนี้
ตัดที่ทดสอบอะไรได้ทดสอบอะไรไม่ได้
repositoryตรรกะของ store ทั้งหมด รวมถึงจังหวะเวลาการแปลง JSON, path, header
http.Clientทุกอย่างตั้งแต่ store ลงไปพฤติกรรมของเซิร์ฟเวอร์จริง

ทั้งสองจุดมีที่ใช้ และ lab ใช้ทั้งคู่ตามความเหมาะสม

2. ของปลอมที่ง่ายที่สุด

class FakeTaskRepository implements LiveTaskRepository {
  List<LiveTask> result = const [];
  int fetchCalls = 0;
  Object? throwOnFetch;

  @override
  Future<List<LiveTask>> fetchOpen() async {
    fetchCalls++;
    if (throwOnFetch case final error?) throw error;
    return result;
  }
}
ส่วนให้อะไรกับเทสต์
resultตั้งได้ว่าจะคืนอะไร
fetchCallsตรวจว่าถูกเรียกกี่ครั้ง
throwOnFetchจำลองความล้มเหลว

สามสิบบรรทัดนี้ทำให้เขียนเทสต์แบบนี้ได้

test('เหตุการณ์เปลี่ยนสถานะโดยไม่ต้องยิง HTTP', () async {
  repository.result = [task('T-1')];
  await store.start();
  final callsAfterLoad = repository.fetchCalls;

  inner.emit(const TaskDispatched(taskId: 'T-1', assignee: 'AGV-02', version: 2));

  expect(store.tasks.single.status, LiveTaskStatus.dispatched);
  // นี่คือเหตุผลทั้งหมดที่ยอมทำเรียลไทม์ — ไม่มีคำขอเพิ่มเลย
  expect(repository.fetchCalls, callsAfterLoad);
});

บรรทัดสุดท้ายคือสิ่งที่ทดสอบด้วย MockClient ก็ทำได้ แต่ด้วยของจริงทำไม่ได้เลย

3. ของปลอมที่คุมจังหวะเวลาได้

เมื่อต้องทดสอบเรื่อง “คำตอบมาถึงตอนไหน” ของปลอมที่ตอบทันทีใช้ไม่ได้

/// repository ปลอมที่คุมได้ว่าจะตอบเมื่อไหร่
///
/// สร้างได้เพราะมี interface ให้ implement — รุ่นรวมทุกอย่างไว้ในจอไม่มีที่ให้
/// ของปลอมเข้าไปนอกจากระดับ http (9.5)
class _ScriptedRepository implements PalletRepository {
  final List<Completer<List<Pallet>>> _fetches = [];

  @override
  Future<List<Pallet>> fetchInZone(String zone) {
    final completer = Completer<List<Pallet>>();
    _fetches.add(completer);
    return completer.future;      // ค้างไว้จนกว่าเทสต์จะสั่งให้ตอบ
  }

  void completeAt(int index, List<Pallet> pallets) =>
      _fetches[index].complete(pallets);
}

Completer คือสวิตช์ที่ทำให้ Future เสร็จเมื่อเราสั่ง ทำให้ทดสอบสถานการณ์ที่จำลองด้วยของจริงแทบไม่ได้

test('คำตอบของคำขอเก่าที่กลับมาช้า ต้องไม่ทับคำตอบใหม่', () async {
  final first = store.load('A1');
  final second = store.load('B2');

  // ตอบคำขอที่สองก่อน แล้วค่อยตอบคำขอแรก — สลับลำดับตั้งใจ
  repository.completeAt(1, [palletB2]);   await second;
  repository.completeAt(0, [palletA1]);   await first;

  expect(store.pallets.single.code, 'PLT-B2');
});

สิ่งที่ทำถูกทำไม

มีฟังก์ชันสร้างข้อมูลตัวอย่างเทสต์เขียนเฉพาะสิ่งที่ต่าง (14.2)

แถวสุดท้ายมีข้อยกเว้น — ของปลอมที่ใช้ตอนสาธิตหรือตอนพัฒนาโดยไม่มีเซิร์ฟเวอร์ ต้องอยู่ใน lib/ เช่น InMemorySettingsStorage กับ InMemoryRealtimeClient ของ lab ซึ่งใช้ได้ทั้งในเทสต์และตอนรันแอปแบบไม่ต่อเน็ต

5. ของปลอมสามแบบ

แบบทำอะไรใช้เมื่อ
stubคืนค่าที่ตั้งไว้ ไม่นับอะไรแค่ต้องการให้มีของส่งเข้าไป
spyคืนค่า และนับว่าถูกเรียกกี่ครั้งด้วยอะไรต้องตรวจว่าถูกเรียกจริง
fakeทำงานจริงแบบง่าย เช่นเก็บใน mapต้องการพฤติกรรมที่สมจริงกว่า
// fake — ทำงานจริงแต่เก็บในหน่วยความจำ
class InMemorySettingsStorage implements SettingsStorage {
  InMemorySettingsStorage({Map<String, String>? seed}) : _values = {...?seed};

  final Map<String, String> _values;

  /// จำลองเครื่องที่เขียนลงดิสก์ไม่ได้ เช่นพื้นที่เต็มหรือสิทธิ์ไม่พอ
  bool failOnWrite = false;

  @override
  Future<String?> read(String key) async => _values[key];

  @override
  Future<void> write(String key, String value) async {
    if (failOnWrite) throw Exception('เขียนลงเครื่องไม่ได้');
    _values[key] = value;
  }
}

ตัวนี้เป็น fake เพราะมันเก็บและอ่านได้จริง ทำให้ทดสอบเรื่อง “เขียนแล้วอ่านกลับได้” ได้โดยไม่ต้องตั้งค่าอะไร

test('เลือกแล้วจำได้ข้ามการปิดแอป', () async {
  final first = LanguageController(storage);
  await first.load();
  await first.setLanguage(AppLanguage.thai);

  // controller ตัวใหม่ อ่านจากที่เก็บเดียวกัน = จำลองการเปิดแอปใหม่
  final second = LanguageController(storage);
  await second.load();
  expect(second.language, AppLanguage.thai);
});

สังเกตว่า failOnWrite ทำให้จำลองเครื่องที่ดิสก์เต็มได้ ซึ่งเป็นกรณีที่เกิดจริงบน PDA เก่าและทดสอบด้วยมือแทบไม่ได้

6. อย่าให้ของปลอมฉลาดเกิน

สัญญาณอันตรายปัญหา
ของปลอมมีตรรกะทางธุรกิจต้องเขียนเทสต์ให้ของปลอมอีกที
ของปลอมยาวกว่าของจริงดูแลสองที่
เทสต์ผ่านเพราะของปลอมทำงานไม่เหมือนของจริงเขียวแต่ผิด

แถวสุดท้ายคือความเสี่ยงที่แท้จริงของการใช้ของปลอม — และวิธีลดคือ

ทำได้อะไร
ทดสอบของจริงด้วย MockClient แยกอีกชั้น (14.4)ยืนยันว่าตัวจริงทำงานถูก
ทดสอบกับเซิร์ฟเวอร์จริงเป็นครั้งคราวยืนยันว่าสัญญายังตรง
ให้ของปลอมกับของจริงใช้ interface เดียวกันคอมไพเลอร์บังคับให้เมธอดตรงกัน

แถวสุดท้ายทำงานให้อัตโนมัติ — วันที่เพิ่มเมธอดใน PalletRepository ของปลอมทุกตัวจะคอมไพล์ไม่ผ่านทันที ซึ่งเป็นการเตือนที่ถูกจังหวะที่สุด

7. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ทำของปลอมไม่ได้เลยไม่มี interfaceสร้าง interface ก่อน (9.5)
ของปลอมแต่ละไฟล์เขียนคนละแบบไม่มีที่รวมย้ายไป test/fakes/
เทสต์เขียวแต่ของจริงพังของปลอมทำงานไม่เหมือนทดสอบตัวจริงด้วย MockClient ด้วย
ของปลอมมีบั๊กฉลาดเกินไปทำให้โง่ที่สุดเท่าที่ยังใช้ได้
ทดสอบจังหวะเวลาไม่ได้ของปลอมตอบทันทีเสมอใช้ Completer
ของปลอมติดไปกับแอปจริงอยู่ใน lib/ โดยไม่มีเหตุผลย้ายไป test/

8. แบบฝึกหัด

ทีมอยากทดสอบว่า “เมื่อผู้ใช้ยืนยันการรับของ ระบบต้องบันทึกแล้วสร้างงานขนย้ายตามมา”

  1. ควรใช้ของปลอมกี่ตัว
  2. ควรตรวจอะไร
แนวคำตอบ

1. สองตัว ตามจำนวน repository ที่เกี่ยวข้อง

class FakeReceivingRepository implements ReceivingRepository {
  final List<ReceiveCommand> received = [];
  Object? throwOnReceive;

  @override
  Future<void> receive(ReceiveCommand command) async {
    if (throwOnReceive case final error?) throw error;
    received.add(command);
  }
}

class FakeTasksRepository implements TasksRepository {
  final List<String> dispatched = [];

  @override
  Future<String> dispatch({required String eventId, ...}) async {
    dispatched.add(eventId);
    return 'task-${dispatched.length}';
  }
}

2. ตรวจสามระดับ และแต่ละระดับตอบคำถามคนละข้อ

ตรวจตอบว่า
expect(receiving.received, hasLength(1))บันทึกแล้ว
expect(tasks.dispatched, hasLength(1))สร้างงานตามมาแล้ว
expect(store.inbound.first.received, isTrue)หน้าจอจะแสดงถูก

แต่เทสต์ที่มีค่าที่สุดคือกรณีที่ล้มเหลวกลางทาง

test('บันทึกสำเร็จแต่สร้างงานไม่สำเร็จ ต้องไม่ทำให้ผู้ใช้เข้าใจผิด', () async {
  tasks.throwOnDispatch = Exception('เชื่อมต่อไม่ได้');

  await store.confirmReceive();

  // ของรับเข้าไปแล้วจริง ห้ามบอกว่าล้มเหลวทั้งหมด
  expect(receiving.received, hasLength(1));
  // แต่ต้องบอกว่างานขนย้ายยังไม่ถูกสร้าง
  expect(store.message, contains('งานขนย้าย'));
});

นี่คือกรณีที่ 9.12 เรียกว่า “ลำดับหลายขั้นที่ต้องทำให้ครบ” และเป็นเหตุผลที่งานแบบนี้ควรมี use case ของตัวเอง

สิ่งที่ต้องคิดต่อและตอบเองไม่ได้ — ถ้าขั้นที่สองล้มเหลว ควรย้อนขั้นแรกไหม

คำตอบเหมาะเมื่อ
ย้อนของยังไม่ได้เข้าคลังจริง
ไม่ย้อน แต่แจ้งให้สร้างงานทีหลังของเข้าคลังไปแล้วจริง — ย้อนแปลว่าข้อมูลไม่ตรงกับความจริง

ในงานคลัง คำตอบมักเป็นข้อสอง เพราะระบบต้องสะท้อนความจริงที่เกิดขึ้นแล้ว ไม่ใช่ย้อนความจริงเพื่อให้ข้อมูลสวย — และเทสต์ที่เขียนไว้คือที่ที่บันทึกการตัดสินใจนี้ไว้ให้คนที่มาทีหลังอ่าน

9. สรุป

  • ของปลอมตัดสายพึ่งพา ทำให้ทดสอบทีละชั้นได้
  • ตัดที่ repository ได้ตรรกะของ store ตัดที่ http.Client ได้ทั้งสาย
  • Completer ทำให้คุมจังหวะเวลาได้
  • ของปลอมต้องโง่ที่สุดเท่าที่ยังใช้ได้
  • ให้ของปลอมกับของจริงใช้ interface เดียวกัน คอมไพเลอร์จะเตือนเมื่อไม่ตรง
  • รวมของปลอมไว้ที่เดียวใต้ test/

บทหน้า: ทดสอบตัวจริงที่คุยกับเซิร์ฟเวอร์ โดยไม่มีเซิร์ฟเวอร์
