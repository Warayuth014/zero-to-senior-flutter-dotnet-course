1. สองแบบที่ต่างกันมาก

/// รายการพาเลทที่สร้างลูกทุกตัวพร้อมกัน
///
/// `Column` ใน `SingleChildScrollView` สร้าง widget ครบทุกตัวตั้งแต่แรก
/// แม้จะเห็นแค่ห้าตัวบนจอ — 500 แถวคือ 500 widget ที่ถูกสร้างและวาด
class EagerPalletList extends StatelessWidget {
  @override
  Widget build(BuildContext context) => SingleChildScrollView(
    child: Column(
      children: [
        for (final code in codes)
          CountedBuilder(counter: counter, label: 'row', ...),
      ],
    ),
  );
}
/// รายการเดียวกันที่สร้างเฉพาะแถวที่กำลังจะเห็น
class LazyPalletList extends StatelessWidget {
  @override
  Widget build(BuildContext context) => ListView.builder(
    itemCount: codes.length,
    itemExtent: 56,
    itemBuilder: (context, index) => CountedBuilder(...),
  );
}

5. รูปในลิสต์

13.7 อธิบายเรื่องหน่วยความจำไปแล้ว ตรงนี้เพิ่มเรื่องที่เกี่ยวกับลิสต์โดยเฉพาะ

ต้องทำเพราะ
cacheWidth คูณ devicePixelRatioรูป 1024×1024 ที่แสดง 56×56 กิน 4 MB ต่อใบ
จองพื้นที่ด้วยขนาดตายตัวไม่งั้นลิสต์กระตุกทุกครั้งที่รูปโหลดเสร็จ
errorBuilderรูปโหลดไม่ขึ้นเป็นเรื่องปกติในคลัง
ขอรูปย่อจากเซิร์ฟเวอร์ได้ผลมากที่สุด
SizedBox(
  width: 56,
  height: 56,        // จองพื้นที่ไว้ก่อน ลิสต์จะไม่กระโดด
  child: Image.network(
    imageUrl(item.path),
    fit: BoxFit.cover,
    cacheWidth: (56 * MediaQuery.devicePixelRatioOf(context)).round(),
    errorBuilder: (_, _, _) => const Icon(Icons.broken_image),
  ),
)
แถวที่สองมีผลกับงานคลังมากกว่าที่คิด — ลิสต์ที่กระโดดตอนรูปโหลดเสร็จ ทำให้ผู้ใช้ที่กำลังจะกดแถวหนึ่ง กดโดนอีกแถว ซึ่งแปลว่าหยิบของผิด

บรรทัด tasks.where(...) วนทั้งลิสต์ต่อหนึ่งแถว — 500 แถวคือ 250,000 รอบ

// คำนวณครั้งเดียวก่อนสร้างลิสต์
final countByStatus = <TaskStatus, int>{};
for (final task in tasks) {
  countByStatus.update(task.status, (n) => n + 1, ifAbsent: () => 1);
}

itemBuilder: (context, index) {
  final task = tasks[index];
  return TaskRow(task: task, total: countByStatus[task.status] ?? 0);
}

10. สรุป

  • Column ในลิสต์ยาวสร้างทุกแถวตั้งแต่แรก ListView.builder สร้างเฉพาะที่จะเห็น
  • itemExtent ทำให้แถบเลื่อนแม่นและกระโดดไปแถวไกล ๆ ได้เร็ว
  • รูปในลิสต์ต้องมี cacheWidth และต้องจองพื้นที่ไว้
  • อย่าคำนวณที่วนทั้งลิสต์ใน itemBuilder
  • เกิน 1,000 แถว ให้ถามว่าผู้ใช้ต้องเห็นทั้งหมดจริงไหม
  • บนอุปกรณ์ที่มีเครื่องสแกน การสแกนชนะการค้นหาเสมอ
