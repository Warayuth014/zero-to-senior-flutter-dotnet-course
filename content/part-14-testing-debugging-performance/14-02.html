1. โครงของเทสต์หนึ่งไฟล์

import 'package:flutter_test/flutter_test.dart';
import 'package:flutter_wms_companion/settings_lab/server_profile.dart';

void main() {
  group('normalizeBasePath', () {
    test('รูปแบบที่ผู้ใช้พิมพ์ได้ทั้งหมด ต้องให้ผลเดียวกัน', () {
      for (final raw in ['wms', '/wms', 'wms/', '/wms/', '//wms//', ' /wms ']) {
        expect(normalizeBasePath(raw), '/wms', reason: 'ผิดที่ "$raw"');
      }
    });
  });
}
ส่วนหน้าที่
groupจัดกลุ่มเทสต์ที่เกี่ยวกัน ชื่อกลุ่มจะขึ้นนำหน้าชื่อเทสต์ตอนล้ม
testหนึ่งข้อ
expect(ค่าที่ได้, ค่าที่คาด)ลำดับสำคัญ — สลับแล้วข้อความตอนล้มจะกลับด้าน
reasonข้อความเพิ่มตอนล้ม สำคัญมากเมื่ออยู่ในลูป
reason ในลูปคือความต่างระหว่าง “เทสต์ล้ม” กับ “เทสต์ล้มที่ค่า //wms//” — อันหลังบอกได้ทันทีว่าต้องไปดูอะไร

2. ชื่อเทสต์คือเอกสาร

รันเทสต์แล้วอ่านรายชื่อ ควรได้รายการข้อกำหนดของระบบ

normalizeBasePath รูปแบบที่ผู้ใช้พิมพ์ได้ทั้งหมด ต้องให้ผลเดียวกัน
normalizeBasePath ว่างหรือมีแต่ทับ ต้องได้ข้อความว่าง ไม่ใช่ทับเดี่ยว
การตรวจฟอร์ม พิมพ์ที่อยู่เต็มลงช่อง host ต้องบอกว่าให้ไปเลือกที่ Protocol
ProfileStore สลับแล้วต้องลืมผลตรวจของเครื่องเก่า
ชื่อที่ไม่ดีชื่อที่ดี
test('normalizeBasePath works')test('ว่างหรือมีแต่ทับ ต้องได้ข้อความว่าง ไม่ใช่ทับเดี่ยว')
test('test1')test('เลือกภาษาเดิมซ้ำ ต้องไม่ประกาศ')
test('should return correct value')test('เหตุการณ์เก่าที่มาถึงช้า ต้องไม่ทับของใหม่')

รูปแบบที่ใช้ได้ทั่วไปคือ “เมื่อ<สถานการณ์> ต้อง<ผลที่คาด>” — และคำว่า “ต้อง” ทำให้เขียนชื่อที่คลุมเครือได้ยาก

4. ตัวเทียบที่ใช้บ่อย

เขียนตรวจว่า
expect(x, 5)เท่ากับ
expect(x, isNull) / isNotNullเป็น null หรือไม่
expect(list, isEmpty) / isNotEmptyว่างหรือไม่
expect(list, hasLength(3))จำนวนสมาชิก
expect(text, contains('Port'))มีข้อความย่อย
expect(x, isA<ImageRejected>())เป็นชนิดนั้น
expect(n, greaterThanOrEqualTo(4.5))เปรียบเทียบ
expect(list, containsAll(['a', 'b']))มีครบทุกตัว ไม่สนลำดับ
expect(() => f(), throwsA(isA<ContractException>()))โยนชนิดนั้น
expect(() => f(), returnsNormally)ไม่โยน

ตัวที่ทรงพลังแต่ใช้กันน้อยคือ having ซึ่งตรวจฟิลด์ข้างในของสิ่งที่โยนได้

await expectLater(
  repository.fetchInZone('A1'),
  throwsA(
    isA<ContractException>().having((e) => e.path, 'path', 'items[1].quantity'),
  ),
);

สามค่าใน having คือ ฟังก์ชันอ่านค่า, ชื่อที่จะแสดงตอนล้ม, และค่าที่คาด — ทำให้ข้อความตอนล้มบอกว่า path ได้อะไรมาแทน

5. เทสต์ที่เป็น async

test('เปิดครั้งแรกต้องมี profile เริ่มต้นให้ใช้ทันที', () async {
  final store = newStore();
  await store.load();

  expect(store.profiles, hasLength(1));
});
สิ่งที่ต้องระวังรายละเอียด
ใส่ async แล้วต้อง awaitลืม await แล้วเทสต์จะจบก่อนงานเสร็จ และผ่านทั้งที่โค้ดพัง
ใช้ expectLater กับ Futureexpect(future, throwsA(...)) ก็ทำงาน แต่ expectLater ชัดเจนกว่า
stream ต้องปล่อยให้ microtask ทำงานawait Future<void>.delayed(Duration.zero)

แถวแรกเป็นข้อผิดพลาดที่เจอบ่อยและอันตราย เพราะเทสต์จะเป็นสีเขียวทั้งที่ไม่ได้ตรวจอะไรเลย

// ผิด — เทสต์จบก่อน load() ทำงานเสร็จ
test('...', () async {
  store.load();                    // ← ลืม await
  expect(store.profiles, hasLength(1));   // ล้มเสมอ หรือผ่านโดยบังเอิญ
});
ตัววิเคราะห์ของ Dart เตือนเรื่องนี้ให้ถ้าเปิด lint unawaited_futures — คุ้มมากที่จะเปิดในโปรเจกต์ที่มีเทสต์เยอะ

7. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
เทสต์เขียว แต่โค้ดพังลืม awaitเปิด lint unawaited_futures
เทสต์ล้มแบบสุ่มมีสถานะค้างข้ามข้อสร้างใหม่ใน setUp
ล้มแล้วไม่รู้ว่าค่าไหนไม่ใส่ reason ในลูปใส่
ข้อความตอนล้มกลับด้านสลับลำดับ expectค่าที่ได้มาก่อน ค่าที่คาดตามหลัง
แก้โค้ดนิดเดียวแล้วเทสต์พังสิบข้อทดสอบวิธีที่ทำ ไม่ใช่สิ่งที่สัญญาทดสอบผ่าน API สาธารณะ
เทสต์ยาวจนอ่านไม่รู้เรื่องไม่มีฟังก์ชันช่วยสร้างข้อมูลทำ builder ที่มีค่าปริยาย

8. แบบฝึกหัด

เขียนเทสต์ให้ฟังก์ชันนี้

/// แปลงจำนวนที่สแกนได้เป็นข้อความสรุป
String pickSummary({required int scanned, required int required}) {
  if (scanned == required) return 'ครบแล้ว';
  if (scanned > required) return 'เกิน ${scanned - required} ชิ้น';
  return 'ขาดอีก ${required - scanned} ชิ้น';
}
แนวคำตอบ
group('pickSummary', () {
  test('ครบพอดี', () {
    expect(pickSummary(scanned: 5, required: 5), 'ครบแล้ว');
  });

  test('ขาด บอกว่าขาดเท่าไหร่', () {
    expect(pickSummary(scanned: 3, required: 5), 'ขาดอีก 2 ชิ้น');
  });

  test('เกิน บอกว่าเกินเท่าไหร่', () {
    expect(pickSummary(scanned: 7, required: 5), 'เกิน 2 ชิ้น');
  });

  test('ยังไม่ได้สแกนเลย', () {
    expect(pickSummary(scanned: 0, required: 5), 'ขาดอีก 5 ชิ้น');
  });
});

สี่ข้อนี้ครอบคลุมเส้นทางทั้งหมด แต่คำถามที่มีค่ากว่าคือค่าที่ยังไม่ได้ถาม

ค่าที่ควรถามควรเกิดอะไรใครตอบได้
required: 0งานที่ไม่ต้องหยิบอะไรเลยมีจริงไหมหัวหน้าคลัง
scanned: -1เป็นไปได้ไหม ถ้าไม่ ควรโยนทีมพัฒนา
จำนวนที่มีทศนิยมสินค้าที่ชั่งน้ำหนักใช้ int ไม่ได้ (9.5)หัวหน้าคลัง

สามแถวนี้คือสิ่งที่การเขียนเทสต์มอบให้นอกจากการจับบั๊ก — มันบังคับให้ถามคำถามที่ยังไม่มีใครถาม ตอนที่ยังแก้ได้ถูก

ถ้าคำตอบคือ “จำนวนติดลบเป็นไปไม่ได้” ก็ควรเขียนลงไปในโค้ด ไม่ใช่แค่รู้กันในหัว

String pickSummary({required int scanned, required int required}) {
  // ทั้งสองค่ามาจากการนับ จึงติดลบไม่ได้ — ถ้าติดลบแปลว่ามีบั๊กที่อื่น
  assert(scanned >= 0 && required >= 0, 'จำนวนติดลบ: $scanned/$required');
  ...
}

assert ทำงานเฉพาะตอนพัฒนาและตอนรันเทสต์ จึงไม่กระทบผู้ใช้ แต่จับบั๊กของเราได้ตั้งแต่ก่อนถึงมือใคร

9. สรุป

  • ชื่อเทสต์คือเอกสาร เขียนเป็น “เมื่อ…ต้อง…”
  • หนึ่งเทสต์ทดสอบเรื่องเดียว ยกเว้นการวนค่าที่ทดสอบกฎเดียวกัน
  • ใส่ reason ทุกครั้งที่อยู่ในลูป
  • ลืม await ทำให้เทสต์เขียวทั้งที่ไม่ได้ตรวจอะไร
  • สร้างของใหม่ใน setUp อย่าให้สถานะค้างข้ามข้อ
  • ฟังก์ชันช่วยสร้างข้อมูลทำให้เทสต์เขียนเฉพาะสิ่งที่ต่าง
  • การเขียนเทสต์บังคับให้ถามคำถามที่ยังไม่มีใครถาม
