ระดับ: กลางใช้เวลา: 50–65 นาทีลงมือทำ: เขียนเทสต์ที่ตรวจสิ่งที่ส่งออกไป
ต้องผ่านอะไรมาก่อน: 14.3, Part 7 (HTTP)
จบบทนี้แล้วจะทำได้: ทดสอบ repository และ ApiClient ได้โดยไม่ต้องมีเซิร์ฟเวอร์ และตรวจได้ว่าส่งอะไรออกไปจริง

1. MockClient คืออะไร

import 'package:http/testing.dart';

final client = MockClient((request) async {
  return http.Response('{"items": []}', 200);
});
MockClient เป็น http.Client ที่แทนที่การยิงเน็ตด้วยฟังก์ชันของเรา — ได้ request เข้ามา คืน response ออกไป

มันมากับแพ็กเกจ http อยู่แล้ว ไม่ต้องติดตั้งอะไรเพิ่ม

2. สองสิ่งที่ตรวจได้

ตรวจตอบคำถามว่า
คำขาที่ส่งออกไปเราส่งอะไรไปให้เซิร์ฟเวอร์
การจัดการคำตอบเราอ่านคำตอบถูกไหม
test('เข้าสู่ระบบแล้วต้องแนบ token และอ่านค่าล่าสุดทุกครั้ง', () async {
  final sent = <http.Request>[];
  final api = ApiClient(
    baseUrl: baseUrl,
    client: MockClient((request) async {
      sent.add(request);
      return http.Response('{}', 200);
    }),
  );

  api.authToken = 'tok-1';
  await api.getJson('/api/WMS/mobile_tasks');
  api.authToken = 'tok-2';
  await api.getJson('/api/WMS/mobile_tasks');

  expect(sent[0].headers['Authorization'], 'Bearer tok-1');
  expect(sent[1].headers['Authorization'], 'Bearer tok-2');
});

เทสต์นี้พิสูจน์ข้อกำหนดจาก 8.4 ว่า ApiClient อ่าน token ตอนที่จะยิง ไม่ใช่คัดลอกเก็บไว้ตอนสร้าง — ซึ่งเป็นสิ่งที่ทดสอบด้วยมือไม่มีทางเห็น

3. ตรวจสิ่งที่ส่งออกไป

อ่านจาก requestได้อะไร
request.methodGET, POST
request.urlUri เต็ม รวม query
request.headersทุก header ที่ส่งไป
request.bodyเนื้อคำขอเป็นข้อความ
request.bodyBytesเนื้อคำขอเป็นไบต์ — ใช้กับ multipart (13.3)
test('ยิงไปที่เส้นทาง health ของ WMS', () async {
  Uri? called;
  await probeServer(
    baseUrl: 'http://10.0.0.1:6191/wms',
    client: MockClient((request) async {
      called = request.url;
      return http.Response('{}', 200);
    }),
  );
  expect(called.toString(), 'http://10.0.0.1:6191/wms/api/WMS/mobile_health');
});

เทสต์นี้จับได้ทั้งเรื่องทับซ้อนใน URL และเรื่องเส้นทางที่ประกอบผิด — สองอย่างที่ 10.3 ใช้เวลาทั้งบทอธิบาย

4. จำลองคำตอบแบบต่าง ๆ

// สำเร็จ
MockClient((_) async => http.Response(
  jsonEncode({'url': '/uploads/a.jpg'}),
  200,
  headers: {'content-type': 'application/json; charset=utf-8'},
))

// เซิร์ฟเวอร์ปฏิเสธพร้อมข้อความ
MockClient((_) async => http.Response(
  jsonEncode({'detail': 'พาเลทนี้ถูกล็อกโดยสมชายอยู่'}),
  409,
  headers: {'content-type': 'application/json; charset=utf-8'},

// เน็ตหลุด
MockClient((_) async => throw http.ClientException('offline'))

// ตอบช้า
MockClient((_) => Future.delayed(const Duration(seconds: 20), () => ...))
อย่าลืม charset=utf-8 ในคำตอบที่มีภาษาไทย — ถ้าไม่ใส่ http จะถอดรหัสเป็น latin1 แล้วได้ข้อความเพี้ยน ซึ่งทำให้เทสต์ล้มด้วยเหตุผลที่ดูไม่เกี่ยวกับสิ่งที่ทดสอบ

5. ตอบต่างกันตามคำขอ

final client = MockClient((request) async {
  final body = request.method == 'POST'
      ? jsonEncode(palletJson('PLT-001', onHold: true))
      : zoneBody([palletJson('PLT-001')]);
  return http.Response(body, 200, headers: {...});
});

ทำให้ทดสอบเส้นทางที่มีหลายคำขอต่อกันได้ เช่นโหลดรายการแล้วกดล็อก

test('ล็อกพาเลทสำเร็จแล้วแถวนั้นเปลี่ยนเป็นล็อกแล้ว', (tester) async {
  await tester.pumpWidget(wrap(client));
  await tester.pumpAndSettle();

  await tester.tap(find.byKey(const Key('hold-PLT-001')));
  await tester.pumpAndSettle();

  expect(find.text('ล็อกแล้ว'), findsOneWidget);
});

อีกแบบที่ใช้บ่อยคือตอบต่างกันตามลำดับครั้งที่เรียก

var attempt = 0;
MockClient((_) async {
  attempt++;
  return attempt == 1
      ? http.Response('', 500)                              // ครั้งแรกล้ม
      : http.Response(jsonEncode({'url': '/a.jpg'}), 200);  // ครั้งที่สองผ่าน
})

ใช้ทดสอบการลองใหม่ และเป็นวิธีเดียวที่ทดสอบได้ว่ารหัสคำสั่งยังเป็นตัวเดิมตอนลองใหม่ (13.4)

6. ค้างคำตอบไว้ด้วย Completer

testWidgets('ระหว่างโหลดครั้งแรกเห็นวงหมุน', (tester) async {
  final gate = Completer<http.Response>();
  await tester.pumpWidget(wrap(MockClient((_) => gate.future)));
  await tester.pump();

  expect(find.byKey(const Key('initial-loading')), findsOneWidget);

  gate.complete(http.Response(zoneBody([palletJson('PLT-001')]), 200, ...));
  await tester.pumpAndSettle();
});

7. เมื่อไหร่ควรใช้ MockClient เมื่อไหร่ควรใช้ repository ปลอม

ใช้ MockClientใช้ repository ปลอม
ทดสอบว่า path, header, body ถูกทดสอบตรรกะของ store
ทดสอบการแปลงคำตอบเป็นโมเดลทดสอบจังหวะเวลาและลำดับ
ทดสอบการแปลรหัสตอบกลับทดสอบสถานะและการกันกดซ้ำ

8. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ภาษาไทยในคำตอบเพี้ยนไม่ได้ใส่ charset=utf-8ใส่ใน headers
Invalid UTF-8 byte ตอนอ่าน bodyคำขอมีไบต์ดิบlatin1.decode(request.bodyBytes) (13.3)
เทสต์ผ่านแต่ของจริงยิงผิด pathไม่ได้ตรวจ request.urlเก็บคำขอไว้แล้วตรวจ
เทสต์ค้างCompleter ไม่เคยถูก completecomplete ก่อนจบเทสต์เสมอ
ทดสอบสถานะระหว่างโหลดไม่ได้ใช้ pumpAndSettle ตั้งแต่แรกใช้ pump() ดูเฟรมแรก (14.10)
MockClient ไม่ถูกเรียกโค้ดสร้าง http.Client เองรับเข้ามาทาง constructor (9.8)

9. แบบฝึกหัด

ทีมเปลี่ยน endpoint จาก /api/WMS/mobile_tasks เป็น /api/WMS/v2/mobile_tasks โดยไม่ได้บอกฝั่งแอป

  1. เทสต์ที่มีอยู่จับได้ไหม
  2. จะทำอย่างไรให้จับได้
แนวคำตอบ

1. ไม่ได้ — และนี่คือข้อจำกัดที่สำคัญที่สุดของ MockClient

MockClient ตอบตามที่เราเขียนไว้ ไม่ว่าจะยิงไป path ไหน ดังนั้นเทสต์ทั้งหมดจะยังเขียว ส่วนแอปจริงจะได้ 404 ทุกคำขอ

สิ่งที่ MockClient จับได้สิ่งที่จับไม่ได้
ฝั่งเราเปลี่ยน path แล้วลืมแก้ที่อื่นฝั่งเซิร์ฟเวอร์เปลี่ยน path
ฝั่งเราส่ง header ไม่ครบฝั่งเซิร์ฟเวอร์เปลี่ยนชื่อฟิลด์
ฝั่งเราอ่านคำตอบผิดฝั่งเซิร์ฟเวอร์เปลี่ยนรหัสตอบกลับ

สังเกตว่าคอลัมน์ซ้ายทั้งหมดคือ “ความผิดพลาดที่มาจากฝั่งเรา” ซึ่งเป็นสาเหตุของปัญหาส่วนใหญ่ — แต่ไม่ใช่ทั้งหมด

2. มีสามทาง เรียงตามราคา

ทางราคาจับได้เมื่อ
เทสต์ที่ยิงเซิร์ฟเวอร์ทดสอบจริงใน CIสูง — ต้องมีเซิร์ฟเวอร์ที่รันได้ใน CIทุกครั้งที่ push
เทสต์ที่รันด้วยมือก่อนปล่อยรุ่นต่ำก่อนปล่อย
ให้เซิร์ฟเวอร์มี endpoint บอกรุ่นของสัญญากลาง — ต้องคุยกับทีมหลังบ้านตอนเปิดแอป

ทางที่สามน่าสนใจเพราะต่อยอดจาก health payload ที่ 10.9 ทำไว้แล้ว

{"service": "WMS", "ok": true, "apiVersion": 2}
if (health.apiVersion != supportedApiVersion) {
  // แอปรุ่นนี้คุยกับเซิร์ฟเวอร์รุ่นนั้นไม่ได้ บอกให้ชัดตั้งแต่ต้น
  // ดีกว่าปล่อยให้ทุกจอขึ้น "โหลดไม่สำเร็จ" โดยไม่มีใครรู้สาเหตุ
}

ข้อสรุปที่ควรจำ: MockClient ทดสอบว่าเราทำตามที่เราเข้าใจ ไม่ได้ทดสอบว่าความเข้าใจของเราตรงกับเซิร์ฟเวอร์ — อย่างหลังต้องมีอะไรที่คุยกับเซิร์ฟเวอร์จริง และ 14.5 จะพูดถึงทางที่ถูกที่สุดที่ทำได้

10. สรุป

  • MockClient แทนที่การยิงเน็ตด้วยฟังก์ชันของเรา
  • ตรวจได้ทั้งสิ่งที่ส่งออกไปและการจัดการคำตอบ
  • ใส่ charset=utf-8 เสมอเมื่อคำตอบมีภาษาไทย
  • Completer ทำให้ทดสอบสถานะระหว่างรอได้
  • MockClient สำหรับทั้งสาย repository ปลอมสำหรับตรรกะของ store
  • มันจับความผิดพลาดจากฝั่งเราได้ แต่จับการเปลี่ยนฝั่งเซิร์ฟเวอร์ไม่ได้
