1. ทำไม dialog ถึงต้องมีเทสต์

สิ่งที่ dialog ทำผลถ้าผิด
ยืนยันก่อนทำสิ่งที่ย้อนกลับยากทำไปแล้วโดยผู้ใช้ไม่ตั้งใจ
เตือนก่อนทิ้งงานที่กรอกค้างผู้ใช้เสียงานที่ทำมาสี่สิบนาที (10.11)
ให้เลือกที่มาของรูปเลือกไม่ได้ ทำงานไม่ได้ (13.1)

ทั้งสามอย่างเป็นจุดที่ผู้ใช้อยู่กลางงานพอดี — ผิดตรงนี้เสียหายกว่าผิดที่อื่น

2. โครงของเทสต์ dialog

testWidgets('ยืนยันแล้วจึงส่งคำสั่ง', (tester) async {
  await tester.pumpWidget(wrap(TaskScreen(store: store)));
  await tester.pumpAndSettle();

  // 1. กดสิ่งที่เปิด dialog
  await tester.tap(find.byKey(const Key('complete-T-1')));
  // 2. ต้อง settle เพื่อให้ dialog วาดเสร็จ
  await tester.pumpAndSettle();

  // 3. ตรวจว่า dialog โผล่จริง
  expect(find.text('ยืนยันปิด T-1?'), findsOneWidget);

  // 4. กดปุ่มใน dialog
  await tester.tap(find.text('ยืนยัน'));
  await tester.pumpAndSettle();

  // 5. ตรวจผล
  expect(repository.completed, contains('T-1'));
});
ขั้นที่ 2 ขาดไม่ได้ — showDialog มี animation การเปิด ถ้าไม่ pumpAndSettle จะหา widget ใน dialog ไม่เจอ

3. เทสต์ที่สำคัญกว่าคือการยกเลิก

testWidgets('กดยกเลิกแล้วต้องไม่ส่งคำสั่ง', (tester) async {
  await tester.pumpWidget(wrap(TaskScreen(store: store)));
  await tester.pumpAndSettle();

  await tester.tap(find.byKey(const Key('complete-T-1')));
  await tester.pumpAndSettle();
  await tester.tap(find.text('ยกเลิก'));
  await tester.pumpAndSettle();

  expect(repository.completed, isEmpty, reason: 'ยกเลิกแล้วต้องไม่ยิง');
  expect(find.byType(AlertDialog), findsNothing, reason: 'dialog ต้องปิด');
});

สองบรรทัดสุดท้ายทดสอบคนละเรื่อง และทั้งคู่เคยพังจริงในโปรเจกต์ต่าง ๆ

บั๊กที่เกิดได้อาการ
ยิงคำสั่งก่อนรอผลจาก dialogกดยกเลิกแล้วงานก็ถูกปิดอยู่ดี
Navigator.pop สองครั้งdialog ปิดพร้อมพาออกจากจอไปด้วย
Future<void> _confirmAndComplete(WmsTask task) async {
  final confirmed = await showDialog<bool>(...);
  // ต้องตรวจทั้งสองอย่าง — ผู้ใช้อาจปัดปิดโดยไม่กดปุ่ม (ได้ null)
  // และจออาจถูกทิ้งไปแล้วระหว่างที่ dialog เปิดอยู่
  if (confirmed != true || !mounted) return;
  await widget.store.complete(task);
}

4. ปัดปิดโดยไม่กดปุ่ม

testWidgets('ปิด dialog โดยไม่เลือกอะไร ต้องไม่ทำอะไร', (tester) async {
  await tester.tap(find.byKey(const Key('complete-T-1')));
  await tester.pumpAndSettle();

  // แตะนอกกล่อง = ปิดโดยไม่เลือก ได้ null กลับมา
  await tester.tapAt(const Offset(10, 10));
  await tester.pumpAndSettle();

  expect(repository.completed, isEmpty);
});

กรณีนี้คืนค่า null ซึ่งต่างจาก false — และเป็นเหตุผลที่ต้องเขียน if (confirmed != true) ไม่ใช่ if (confirmed == false)

ผู้ใช้ทำshowDialog คืน
กด “ยืนยัน”true
กด “ยกเลิก”false
แตะนอกกล่องnull
กดปุ่มย้อนกลับของเครื่องnull

สองแถวล่างคือทางที่ผู้ใช้ PDA ใช้บ่อยที่สุด เพราะกดปุ่มย้อนกลับเร็วกว่าเล็งกดปุ่มบนจอ

6. SnackBar

testWidgets('ปิดงานสำเร็จแล้วขึ้นข้อความ', (tester) async {
  await tester.tap(find.byKey(const Key('complete-T-1')));
  await tester.pumpAndSettle();
  await tester.tap(find.text('ยืนยัน'));
  await tester.pump();          // ให้ SnackBar เริ่มปรากฏ
  await tester.pump(const Duration(milliseconds: 750));   // ให้ animation จบ

  expect(find.text('ปิด T-1 สำเร็จ'), findsOneWidget);
});
อย่าใช้ pumpAndSettle กับ SnackBar — มันมีตัวจับเวลาที่จะปิดตัวเองในสี่วินาที pumpAndSettle จะรอจนกว่าจะปิด แล้วเทสต์จะหาไม่เจอ

ใช้ pump พร้อมระบุเวลาแทน ซึ่งเป็นรูปแบบเดียวกับที่ 14.10 จะอธิบาย

7. dialog ที่เปิดจากที่ที่ไม่ใช่การกดปุ่ม

ใน 12.7 มีกรณีที่เหตุการณ์จากเซิร์ฟเวอร์ทำให้ dialog เด้ง ซึ่งทดสอบได้เหมือนกัน

testWidgets('เหตุการณ์จุดคัดแยกเต็ม ต้องเด้งเตือน', (tester) async {
  await tester.pumpWidget(wrap(SortingScreen(client: shared)));
  await tester.pumpAndSettle();

  client.emit(const StationFull(stationId: 'ST-01'));
  await tester.pumpAndSettle();

  expect(find.text('จุดคัดแยก ST-01 เต็ม'), findsOneWidget);
});

testWidgets('จอที่ถูก push ทับ ต้องไม่เด้ง dialog', (tester) async {
  await tester.pumpWidget(wrap(SortingScreen(client: shared)));
  await tester.pumpAndSettle();

  // เปิดจอรายละเอียดทับ
  await tester.tap(find.byKey(const Key('station-ST-01')));
  await tester.pumpAndSettle();

  client.emit(const StationFull(stationId: 'ST-01'));
  await tester.pumpAndSettle();

  expect(find.byType(AlertDialog), findsNothing);
});

เทสต์คู่นี้บันทึกกฎจาก 12.7 §5 ไว้ — และเป็นสิ่งที่ทดสอบด้วยมือได้ยากมากเพราะต้องจับจังหวะให้เหตุการณ์มาถึงตอนที่จออื่นเปิดอยู่พอดี

8. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
หา widget ใน dialog ไม่เจอไม่ได้ pumpAndSettle หลังเปิดใส่
หา SnackBar ไม่เจอใช้ pumpAndSettle ซึ่งรอจนมันปิดใช้ pump พร้อมเวลา
เทสต์ผ่านทั้งที่กดยกเลิกยิงคำสั่งก่อนรอผลรอ await showDialog ก่อน
dialog ปิดแล้วออกจากจอด้วยNavigator.pop สองครั้งตรวจว่า pop ที่ถูก context
ทดสอบการปัดปิดไม่ได้ตั้ง barrierDismissible: falseถูกแล้วถ้าตั้งใจ — ทดสอบปุ่มย้อนกลับแทน
Looking up a deactivated widget's ancestorใช้ context หลังจอถูกทิ้งเช็ค mounted หลัง await

แถวสุดท้ายเป็นข้อผิดพลาดที่เจอบ่อยที่สุดกับ dialog เพราะระหว่างที่รอผู้ใช้ตัดสินใจ จออาจถูกทิ้งไปแล้ว

9. แบบฝึกหัด

10.11 ออกแบบกล่องเตือนก่อนทิ้งงานที่กรอกค้างไว้ เขียนเทสต์ให้ครบทุกเส้นทาง

แนวคำตอบ

ห้าเส้นทาง และแต่ละเส้นทางเป็นเทสต์แยก

เทสต์ข้อสุดท้ายคือข้อที่สำคัญที่สุด และเป็นข้อที่คนลืมบ่อยที่สุด

บั๊กที่มันจับคือ “เตือนหลังบันทึกสำเร็จแล้ว” ซึ่งเกิดเมื่อไม่ได้อัปเดตตัวที่ใช้เทียบหลังบันทึก (10.11 §7) — ผู้ใช้จะสับสนมากเพราะเขาเพิ่งกดบันทึกไป

ข้อที่สามก็มีรายละเอียดที่ควรสังเกต — ตรวจว่าค่าที่พิมพ์ไว้ยังอยู่ ไม่ใช่แค่ตรวจว่ายังอยู่ในฟอร์ม เพราะการกลับมาแล้วพบว่าต้องพิมพ์ใหม่ ก็เท่ากับเสียงานอยู่ดี

สังเกตว่าใช้ openForm และ fillValidForm เป็นฟังก์ชันช่วย ซึ่งทำให้แต่ละเทสต์เหลือแค่สิ่งที่ต่าง ตามหลักใน 14.2

10. สรุป

  • ต้อง pumpAndSettle หลังเปิด dialog ไม่งั้นหาไม่เจอ
  • เทสต์การยกเลิกสำคัญกว่าเทสต์การยืนยัน
  • ปัดปิดกับกดย้อนกลับได้ null ไม่ใช่ false
  • อย่าใช้ pumpAndSettle กับ SnackBar
  • ทดสอบว่าจอที่ถูก push ทับต้องไม่เด้ง dialog
  • เทสต์ที่คนลืมบ่อยที่สุดคือ “บันทึกแล้วต้องไม่เตือน”

บทหน้า: pump กับ pumpAndSettle ต่างกันอย่างไร และเมื่อไหร่ใช้อันไหน
