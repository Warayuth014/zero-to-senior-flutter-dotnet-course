Future<void> _openScanPo() async {

ผู้ใช้ทำผลกับตัวเลข

เป็นวิธีที่ตรงไปตรงมาและได้ผล — ไม่ต้องมีตัวจัดการ state กลาง ไม่ต้องมีการแจ้งเตือนข้ามจอ

เทียบกับทางเลือกอื่น

วิธีข้อดีข้อเสีย
โหลดใหม่ตอนกลับมา (วิธีนี้)ง่าย ไม่ต้องมีโครงเพิ่มยิงคำขอทุกครั้งที่กลับ แม้ไม่มีอะไรเปลี่ยน
ส่งค่ากลับมาทาง Navigator.pop(value)ไม่ต้องยิงซ้ำทุกทางออกของจอปลายทางต้องส่งค่าให้ครบ
store กลางที่ทุกจอฟังทุกจอตรงกันเสมอต้องมีโครงที่โปรเจกต์นี้ยังไม่มี

สำหรับจอที่ผู้ใช้เข้าออกไม่กี่ครั้งต่อกะ วิธีแรกคุ้มที่สุด

สังเกตว่าทำไมวิธีที่สองอันตราย — ScanPartScreen มีทางออกสามทาง (กดย้อนกลับ, ปิด session อัตโนมัติ, popUntil กลับหน้าแรก) ถ้าใช้การส่งค่ากลับ ต้องจำส่งให้ครบทั้งสามทาง ลืมทางเดียวตัวเลขก็ผิด

3. จุดที่รูปแบบนี้พลาด

ไล่ดูตัวอย่างจริง

ผู้ใช้อยู่ที่ ReceivingMenuScreen เห็น "มี 3 รายการรอผูก Pallet"
  → เพื่อนร่วมกะที่เครื่องอื่นผูกไป 2 รายการ
    → ตัวเลขบนเครื่องนี้ยังเป็น 3
      → ผู้ใช้กดเข้าไป เห็นเหลือ 1
        → งง แต่ไม่เสียหาย

กรณีนี้ไม่ร้ายแรงเพราะกดเข้าไปแล้วเห็นของจริง แต่มีอีกกรณีที่ร้ายกว่า

ผู้ใช้ค้างอยู่ที่จอนี้ 20 นาที (วางเครื่องไว้)
  → มีของค้างเพิ่มเป็น 12 รายการ
    → หน้าจอยังบอกว่า "ไม่มีรายการค้าง" และการ์ดเป็นสีเทา
      → ผู้ใช้เดินผ่านไปโดยไม่รู้ว่ามีงานรออยู่

ตรงกับที่ 12.10 เตือน — ข้อมูลที่ไม่มีการโหลดซ้ำจะค้างเก่าโดยไม่มีอะไรบอก

// แบบที่ 1 — โหลดใหม่ตอนแอปกลับมาจากพื้นหลัง
class _ReceivingMenuScreenState extends State<ReceivingMenuScreen>
    with WidgetsBindingObserver {

  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addObserver(this);
    _loadPendingCount();
  }

  @override
  void dispose() {
    WidgetsBinding.instance.removeObserver(this);   // ห้ามลืม
    super.dispose();
  }

  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    if (state == AppLifecycleState.resumed) _loadPendingCount();
  }
}

4. ความล้มเหลวที่เงียบสนิท

ถ้ายิงไม่สำเร็จ ตัวเลขจะค้างที่ค่าเดิม ซึ่งตอนเปิดจอครั้งแรกคือ 0

สถานการณ์ผู้ใช้เห็นความจริง
ยิงสำเร็จ ไม่มีของค้าง“ไม่มีรายการค้าง” สีเทาถูก
ยิงไม่สำเร็จ“ไม่มีรายการค้าง” สีเทาไม่รู้

สองสถานการณ์นี้หน้าตาเหมือนกันทุกพิกเซล ผู้ใช้แยกไม่ออก และไม่มีอะไรบอกว่าเชื่อมต่อไม่ได้

ทำไมถึงเลือกเงียบ — ถ้าเด้งกล่องข้อผิดพลาดทุกครั้งที่กลับมาแล้วเน็ตไม่ดี ผู้ใช้จะต้องกดปิดตลอดเวลา ซึ่งน่ารำคาญกว่า
การเลือกเงียบจึงไม่ผิด แต่ต้องแยก “ยังไม่รู้” ออกจาก “รู้ว่าไม่มี” ตามที่ 10.8 สอน

}

7. แบบฝึกหัด

ทีมอยากให้ตัวเลขบนเมนูทันสมัยขึ้น โดยไม่ยิงคำขอถี่เกินไป

  1. เสนอทางเลือกสามทาง พร้อมต้นทุน
  2. เลือกหนึ่งทางแล้วเขียนโค้ด

แนวคำตอบ

ข้อ 1

ทางคำขอต่อกะ (โดยประมาณ)ความสดงานที่ต้องทำ
ตอนกลับมาจากพื้นหลังเพิ่มไม่กี่ครั้งดีสำหรับกรณีวางเครื่องไว้เพิ่ม observer + ต้องลบใน dispose
ลากลงเพื่อรีเฟรชเท่าที่ผู้ใช้กดผู้ใช้ควบคุมเองครอบด้วย RefreshIndicator
ตัวจับเวลาทุก 2 นาที~240 ครั้งสดตลอดต้องหยุดเมื่อออกจากจอ

สองทางแรกรวมกันคุ้มที่สุด และทั้งคู่ใช้ _loadPendingCount() ตัวเดิม

ข้อ 2

class _ReceivingMenuScreenState extends State<ReceivingMenuScreen>
    with WidgetsBindingObserver {
  final _api = ApiService();
  int? _pendingPalletCount;          // null = ยังไม่รู้

  @override
  void initState() {
    super.initState();
    WidgetsBinding.instance.addObserver(this);
    _loadPendingCount();
  }

  @override
  void dispose() {
    // ลืมบรรทัดนี้ = observer ค้างอยู่หลังจอถูกปิด แล้ว setState บน State ที่ตายแล้ว
    WidgetsBinding.instance.removeObserver(this);
    super.dispose();
  }

  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    if (state == AppLifecycleState.resumed) _loadPendingCount();
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: ...,
      body: RefreshIndicator(
        onRefresh: _loadPendingCount,
        // RefreshIndicator ต้องมีลูกที่เลื่อนได้เสมอ ไม่งั้นลากไม่ติด
        child: ListView(
          padding: const EdgeInsets.all(16),
          children: [ ...การ์ดทั้งสองใบ... ],
        ),
      ),
    );
  }
}

สองรายละเอียดที่พลาดง่าย: removeObserver ใน dispose และ RefreshIndicator ต้องมีลูกที่เลื่อนได้ — จอเดิมใช้ Padding + Column ซึ่งลากลงไม่ติด ต้องเปลี่ยนเป็น ListView

ข้อ 3 — ทำไมไม่ใช้ตัวจับเวลาทุก 10 วินาที

เหตุผลรายละเอียด
ตัวเลขนี้ไม่ได้เปลี่ยนเร็วขนาดนั้นของค้างเกิดจากการผูกพาเลทล้มเหลว ซึ่งนาน ๆ ครั้ง
เปลืองแบตบน PDAปลุกวิทยุทุก 10 วินาทีตลอดกะ (12.1)
ผู้ใช้ไม่ได้จ้องจอนี้เป็นจอทางผ่าน อยู่ไม่กี่วินาที
เพิ่มโหลดเซิร์ฟเวอร์คูณจำนวนเครื่อง20 เครื่อง × 6 ครั้ง/นาที = 120 คำขอ/นาที เพื่อตัวเลขเดียว

หลักที่ใช้ตัดสิน: ความถี่ของการถามต้องมาจากความเร็วที่ข้อมูลเปลี่ยนจริง และความเสียหายถ้าเห็นข้อมูลเก่า ไม่ใช่จากความอยากให้สด

ที่นี่ข้อมูลเปลี่ยนช้าและเห็นเลขเก่าก็ไม่เสียหาย เพราะกดเข้าไปแล้วเห็นของจริงอยู่ดี — 12.1 ใช้หลักเดียวกันตัดสินว่าเรื่องไหนควรใช้เรียลไทม์และเรื่องไหนไม่ต้อง

8. สรุป

  • จอ 107 บรรทัด มี state สองตัว และทำหน้าที่เดียว
  • ใช้รูปแบบ await Navigator.push(...) แล้วโหลดใหม่ — ง่ายและได้ผลสำหรับจอที่เข้าออกไม่บ่อย
  • ปลอดภัยกว่าการส่งค่ากลับ เพราะจอปลายทางมีทางออกหลายทาง
  • ไม่มีการโหลดซ้ำระหว่างที่ค้างอยู่ที่จอ — ตัวเลขค้างเก่าได้
  • ยิงไม่สำเร็จแล้วเงียบ ทำให้ “ไม่มีของค้าง” กับ “ตรวจไม่ได้” หน้าตาเหมือนกัน
  • แก้ได้ด้วยการทำให้ตัวนับเป็น int? โดย null แปลว่ายังไม่รู้
  • สีแดงถูกใช้ทั้งกับ “มีงานรอ” และ “ผิดพลาด” ซึ่งควรแยก
  • ความถี่ของการถามต้องมาจากความเร็วที่ข้อมูลเปลี่ยนจริง
