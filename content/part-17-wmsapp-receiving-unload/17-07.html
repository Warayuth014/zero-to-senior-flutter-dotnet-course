  // แต่เหตุผลต้องอ่านได้ ไม่งั้นผู้ใช้ลองพาเลทเดิมซ้ำ
  await showDialog<void>(
    context: context,
    builder: (ctx) => AlertDialog(
      title: Row(children: [
        Icon(Icons.warning_amber_rounded, color: AppTheme.warning),
        const SizedBox(width: 8),
        const Expanded(child: Text('บันทึกแล้ว แต่ยังไม่ได้ผูก Pallet')),
      ]),
      content: Column(
        mainAxisSize: MainAxisSize.min,
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          InfoRow(label: 'สินค้า', value: line.partId),
          InfoRow(label: 'จำนวน', value: '${line.qtyReceived} ชิ้น'),
          const SizedBox(height: 12),
          // เหตุผลเป็นบรรทัดของตัวเอง ไม่ปนกับอย่างอื่น
          Text(line.palletError!, style: const TextStyle(height: 1.5)),
          const SizedBox(height: 12),
          Text(
            'สินค้าถูกบันทึกไว้แล้ว — ไปที่รายการ "รอผูก Pallet" '
            'แล้วลองพาเลทใบอื่น',
            style: TextStyle(fontSize: 13, color: AppTheme.textGrey(ctx)),
          ),
        ],
      ),
      actions: [
        ElevatedButton(
          onPressed: () => Navigator.pop(ctx),
          child: const Text('เข้าใจแล้ว'),
        ),
      ],
    ),
  );
}
สิ่งที่เปลี่ยนผล
ใช้กล่องแทนแถบผู้ใช้ต้องอ่านก่อนทำต่อ
เหตุผลอยู่บรรทัดของตัวเองไม่ถูกตัดท้าย
บอกว่า “ลองพาเลทใบอื่น”ตัดวงการลองพาเลทเดิมซ้ำ
ย้ำว่าของบันทึกแล้วผู้ใช้ไม่กลัวว่าของหาย

แต่กล่องมีต้นทุน — ถ้าเน็ตมีปัญหาและล้มเหลวติดกันสิบครั้ง ผู้ใช้จะต้องกดปิดสิบครั้ง

ทางที่สมดุลกว่าคือใช้แถบสำหรับครั้งแรก แล้วเปลี่ยนเป็นกล่องเมื่อล้มเหลวซ้ำด้วยพาเลทเดิม

String? _lastFailedPalletId;

if (line.palletError != null) {
  final sameAsLast = palletId == _lastFailedPalletId;
  _lastFailedPalletId = palletId;

  if (sameAsLast) {
    await _showPalletErrorDialog(line);      // ซ้ำแล้ว → บังคับให้อ่าน
  } else {
    showWarningSnackbar(context, '...');     // ครั้งแรก → ไม่ขวาง
  }
  setState(() => _resumedPendingLines.insert(0, line));
}

หลักที่ได้: ความรุนแรงของวิธีแจ้งควรขึ้นกับว่าผู้ใช้กำลังทำผิดซ้ำหรือไม่ ไม่ใช่แค่ชนิดของความล้มเหลว — ครั้งแรกอาจเป็นเรื่องบังเอิญ ครั้งที่สองแปลว่าผู้ใช้ไม่ได้รับข้อมูลที่ต้องการ

9. สรุป

  • scan-part ทำสองงานในคำขอเดียว ผลลัพธ์จึงมีสามแบบ ไม่ใช่สองแบบ
  • palletError แยก “บันทึกแล้วแต่ผูกไม่ได้” ออกจาก “ล้มเหลว” — เป็นการออกแบบที่ถูกและหายาก
  • สำเร็จครึ่งทางใช้แถบเหลือง ไม่ใช่กล่องแดง เพราะไม่มีอะไรเสียหาย
  • ล้มเหลวแล้วไม่ล้าง S/N — ตรงกับข้อ 20 ของ 15.11
  • insert(0, ...) ทำให้รายการที่เพิ่งค้างอยู่บนสุด
  • รายการค้างจาก session เดิมถูกกู้จาก POResponse.pendingLines
  • PopScope กันการออกตอนผูกพาเลทค้าง แต่ไม่กันตอนสแกน S/N ค้าง
  • ความล้มเหลวชนิดเดียวกันถูกแจ้งสองแบบในสองเส้นทาง
  • ความรุนแรงของการแจ้งควรขึ้นกับว่าผู้ใช้ทำผิดซ้ำหรือไม่
