ระดับ: เริ่มต้น–กลางใช้เวลา: 30–40 นาทีลงมือทำ: กันไม่ให้ยิงของชิ้นเดิมซ้ำ
ต้องผ่านอะไรมาก่อน: 3.6 (ค่ามาตรฐาน) และ D1.5 (Set กับการกันค่าซ้ำ)
จบบทนี้แล้วจะทำได้: ออกแบบการกันซ้ำที่ถูกต้องตามชนิดข้อมูล และรู้ว่าการยิงซ้ำบางกรณีเป็นเรื่องที่ถูกต้อง

1. ปัญหาที่บทนี้แก้

ในคลังจริง การยิงซ้ำเกิดขึ้นตลอดเวลาด้วยเหตุผลที่ไม่ใช่ความผิดของใคร

  • พนักงานกดปุ่มยิงค้างไว้ เครื่องยิงสองครั้ง
  • ไม่แน่ใจว่าเข้าหรือยัง เลยยิงซ้ำเพื่อความมั่นใจ
  • กล่องเดียวกันถูกวางไว้สองที่ แล้วยิงทั้งสองรอบ

ถ้าไม่กัน ผลคือรายงานบอกว่ารับของ 240 ชิ้นทั้งที่จริงมี 120 — ซึ่งเป็นความเสียหายที่แก้ยากเพราะไม่มีใครรู้ว่าตัวเลขไหนถูก

2. คำศัพท์ที่ต้องรู้ก่อน

คำความหมาย

sessionขอบเขตของ “รอบนี้” เช่นหนึ่งกะ หรือหนึ่งใบงาน

3. Mental model: ทะเบียนที่เคยเจอ

ทบทวนจากแบบฝึกหัด D1.5 — เราต้องการทั้งลำดับสำหรับแสดงผล และการตรวจซ้ำที่เร็ว จึงใช้สองโครงสร้างคู่กัน

final List<String> _accepted = <String>[];   // ลำดับสำหรับหน้าจอ
final Set<String> _seen = <String>{};        // ตรวจซ้ำแบบเร็ว
ประโยคเดียวที่ต้องจำ: ทะเบียนต้องเก็บ ค่ามาตรฐาน ที่ผ่านการแปลงจาก 3.6 แล้ว — ถ้าเก็บค่าดิบ การกันซ้ำจะไม่ทำงานเมื่อตัวพิมพ์หรือช่องว่างต่างกัน

4. ดูของจริงใน lab

ScanOutcome accept(String raw) {
  if (_busy) return ScanOutcome.busy;

  final code = normalizeBarcode(raw);
  if (code.isEmpty) return ScanOutcome.empty;
  if (!isValidPalletCode(code)) return ScanOutcome.badFormat;
  if (!_seen.add(code)) return ScanOutcome.duplicate;   // ← กันซ้ำตรงนี้

  _accepted.add(code);
  return ScanOutcome.accepted;
}

สังเกตว่า _seen.add(code) ทำสองอย่างในบรรทัดเดียว — มันเพิ่มเข้าไปและคืน false ถ้ามีอยู่แล้ว จึงไม่ต้องเรียก contains ก่อนแล้วค่อย add ซึ่งเป็นการอ่านสองรอบโดยไม่จำเป็น (เรื่องนี้อยู่ในเฉลย D1.5 แล้ว)

เทสต์ยืนยันพฤติกรรมสองข้อ

test('ยิงซ้ำถูกปฏิเสธ แม้ตัวพิมพ์และช่องว่างต่างกัน', () {
  session.accept('PAL-1001');

  expect(session.accept('  pal-1001'), ScanOutcome.duplicate);
  expect(session.count, 1);        // ไม่เพิ่มซ้ำ
});

5. ยิงซ้ำไม่ได้แปลว่าผิดเสมอไป

นี่คือจุดที่สำคัญที่สุดของบทนี้ และเป็นที่ที่คนมักออกแบบผิด — กฎการกันซ้ำขึ้นกับว่าเรากำลังยิงอะไร

ยิงอะไรยิงซ้ำหมายถึงควรทำอย่างไร
ซีเรียลของสินค้าชิ้นเดียวผิดแน่นอน — ของชิ้นเดียวมีซีเรียลเดียวปฏิเสธ
รหัสพาเลทในรอบรับเข้าผิด — พาเลทใบเดิมรับซ้ำไม่ได้ปฏิเสธ
บาร์โค้ดสินค้าตอนนับสต็อกถูกต้อง — ยิงหนึ่งครั้งต่อหนึ่งชิ้นเพิ่มจำนวน
รหัสตำแหน่งตอนเดินตรวจถูกต้อง — เดินผ่านซ้ำได้ยอมรับตามปกติ
อย่าสร้างกฎกลางแบบ “ห้ามยิงรหัสเดิมซ้ำภายใน 3 วินาที” แล้วใช้ทั้งแอป เพราะหน้าจอนับสต็อกจะพังทันที — พนักงานยิงสินค้าชนิดเดียวกันติด ๆ กัน 20 ชิ้นเป็นเรื่องปกติ

ให้กำหนดกฎ ต่อหน้าจอ ตามความหมายทางธุรกิจของสิ่งที่กำลังยิง

6. ขอบเขตของ “รอบนี้”

คำถามถัดมาคือทะเบียนควรเก็บไว้นานแค่ไหน

ขอบเขตเหมาะกับล้างเมื่อ
หนึ่งใบงานรับเข้าตามใบสั่งซื้อกดบันทึกสำเร็จ
หนึ่งกะตรวจนับทั้งโซนปิดกะ
ไม่กันเลยนับสต็อกทีละชิ้น—

ใน lab ทะเบียนอยู่ใน ScanSession ซึ่งมีอายุเท่ากับหน้าจอ — เมื่อออกจากหน้าจอแล้วเข้าใหม่ ทะเบียนจะว่าง ซึ่งเหมาะกับงานรับเข้าทีละใบ

สิ่งที่ต้องระวังคือ อย่าเก็บทะเบียนไว้นานกว่าที่ธุรกิจต้องการ เพราะพนักงานจะยิงพาเลทใบเดิมไม่ได้ในวันถัดไปทั้งที่ควรได้

7. สามชั้นที่ต้องมีครบ

การกันซ้ำที่แอปเป็นเพียงชั้นเดียวจากสามชั้น

ชั้นกันอะไรกันไม่ได้
ทะเบียนในหน้าจอยิงซ้ำในรอบเดียวกันเปิดแอปใหม่แล้วยิงซ้ำ
ล็อกระหว่างบันทึกกดบันทึกซ้อนกัน (3.8)ส่งซ้ำจากคนละเครื่อง
กฎที่ serverทุกกรณีที่เหลือ—

เหตุผลที่ชั้นที่สามขาดไม่ได้

  • พนักงานสองคนอาจยิงพาเลทใบเดียวกันจากคนละเครื่อง
  • เน็ตหลุดตอนบันทึก แอปลองใหม่ แต่ครั้งแรกอาจถึง server ไปแล้ว
  • ผู้ใช้ปิดแอปแล้วเปิดใหม่ ทะเบียนหายไป

ทางที่ server ใช้กันคือให้แต่ละคำสั่งมีรหัสประจำตัว ถ้าได้รับรหัสเดิมซ้ำก็คืนผลเดิมโดยไม่บันทึกใหม่ — เรื่องนี้จะลงรายละเอียดใน Part 7

8. debounce ไม่ใช่ทางแก้ของเรื่องนี้

debounceทะเบียนกันซ้ำ
ตัดสินจากเวลาค่าที่เคยเจอ
เหมาะกับช่องค้นหาที่พิมพ์ไปเรื่อย ๆการยิงบาร์โค้ด
ทดสอบซ้ำได้ไหมยาก ขึ้นกับจังหวะง่าย ตรวจค่าตรง ๆ

เหตุผลที่ debounce ไม่เหมาะกับการยิงบาร์โค้ดคือเรารู้ชัดอยู่แล้วว่าข้อมูลจบเมื่อไหร่ — ตอนที่เครื่องสแกนกด Enter ตามที่คุยใน 3.5 จึงไม่ต้องเดาจากเวลา

ถ้าเจอปัญหาเครื่องยิงซ้ำจากการกดค้าง ให้แก้ที่การตั้งค่าเครื่องก่อนตามหลักใน 3.5 แล้วให้ทะเบียนเป็นตาข่ายรองรับ

9. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
กันซ้ำไม่ทำงานเก็บค่าดิบในทะเบียนเก็บค่ามาตรฐานจาก 3.6
หน้าจอนับสต็อกยิงชิ้นที่สองไม่ได้ใช้กฎกันซ้ำผิดบริบทกำหนดกฎต่อหน้าจอตามความหมายทางธุรกิจ
วันถัดไปยิงพาเลทเดิมไม่ได้ทะเบียนมีอายุยาวเกินไปล้างเมื่อจบใบงาน
ข้อมูลซ้ำถึง server ได้กันแค่ที่แอปให้ server มีกฎกันซ้ำด้วย
ใช้ contains แล้ว add แยกกันอ่านสองรอบโดยไม่จำเป็นใช้ค่าที่ add คืนมา
เอา debounce มากันซ้ำสับสนระหว่างเวลากับตัวตนใช้ทะเบียนค่าที่เคยเจอ
ปฏิเสธเงียบ ๆ ผู้ใช้ไม่รู้ไม่ได้แจ้งผลบอกให้ชัดว่าเป็นการซ้ำ (3.11)

10. แบบฝึกหัด

แอปหนึ่งมีสามหน้าจอที่ใช้เครื่องสแกน จงกำหนดกฎการกันซ้ำของแต่ละหน้าพร้อมเหตุผล

หน้าจอสิ่งที่ยิงบริบท
รับเข้าตามใบสั่งซื้อรหัสพาเลทหนึ่งใบสั่งซื้อมีหลายพาเลท แต่ละใบรับครั้งเดียว
นับสต็อกในโซนบาร์โค้ดสินค้ายิงหนึ่งครั้งต่อหนึ่งชิ้นที่นับได้
ตรวจซีเรียลก่อนส่งออกซีเรียลสินค้าแต่ละชิ้นมีซีเรียลไม่ซ้ำ
  1. ระบุกฎของแต่ละหน้า
  2. เขียนโครงโค้ดของหน้านับสต็อก ซึ่งต่างจาก lab
  3. อธิบายว่าถ้าใช้กฎของหน้ารับเข้ากับหน้านับสต็อก จะเกิดอะไรขึ้น
ดูแนวคำตอบและเหตุผล
หน้าจอกฎผลเมื่อยิงซ้ำ
รับเข้าปฏิเสธแจ้ง “สแกนซ้ำ” จำนวนไม่เพิ่ม
นับสต็อกเพิ่มจำนวนตัวนับของสินค้านั้นเพิ่มขึ้นหนึ่ง
ตรวจซีเรียลปฏิเสธแจ้ง “ซีเรียลนี้ตรวจไปแล้ว”
/// หน้านับสต็อก — ยิงซ้ำคือเพิ่มจำนวน ไม่ใช่ความผิดพลาด
class CountSession {
  final Map<String, int> _counts = <String, int>{};

  Map<String, int> get counts => Map.unmodifiable(_counts);

  int get totalItems => _counts.values.fold(0, (sum, n) => sum + n);

  CountOutcome accept(String raw) {
    final code = normalizeBarcode(raw);
    if (code.isEmpty) return CountOutcome.empty;
    if (!isValidProductCode(code)) return CountOutcome.badFormat;

    // ต่างจาก ScanSession ตรงนี้ — เพิ่มแทนที่จะปฏิเสธ
    _counts[code] = (_counts[code] ?? 0) + 1;
    return CountOutcome.counted;
  }

  /// ต้องมี เพราะพนักงานนับเกินแล้วต้องแก้ได้
  bool undo(String raw) {
    final code = normalizeBarcode(raw);
    final current = _counts[code];
    if (current == null || current == 0) return false;
    if (current == 1) {
      _counts.remove(code);
    } else {
      _counts[code] = current - 1;
    }
    return true;
  }
}

สิ่งที่เหมือนกันกับ lab

  • แปลงค่าก่อนเสมอ (3.6)
  • ตรวจรูปแบบก่อนใช้
  • คืนผลเป็น enum ที่มีความหมาย ไม่ใช่ bool
  • คืน collection แบบแก้ไม่ได้ออกไป (D1.7)

สิ่งที่ต่างกัน — ใช้ Map<String, int> แทน Set เพราะสิ่งที่ต้องจำไม่ใช่ “เคยเจอไหม” แต่เป็น “เจอกี่ครั้ง” ซึ่งเป็นการเลือกโครงสร้างข้อมูลจากคำถามที่จะถาม ตามหลักใน D1.5

ข้อ 3 — ถ้าใช้กฎของหน้ารับเข้ากับหน้านับสต็อก

พนักงานจะนับสินค้าชนิดเดียวกันได้แค่ชิ้นเดียว ชิ้นที่สองเป็นต้นไปจะถูกปฏิเสธด้วยข้อความ “สแกนซ้ำ” ผลที่ตามมา

  1. ยอดนับผิดทั้งหมด — ถ้าโซนนั้นมีเทปกาว 240 ม้วน ระบบจะบันทึกว่ามี 1
  2. พนักงานจะหาทางเลี่ยง — เช่นออกจากหน้าจอแล้วเข้าใหม่ทุกครั้งเพื่อล้างทะเบียน ซึ่งช้ามากและอาจทำให้ข้อมูลที่นับไว้หาย
  3. ไม่มีใครรายงานว่าเป็นบั๊ก — เพราะข้อความ “สแกนซ้ำ” ดูเหมือนระบบทำงานถูกต้อง คนใช้จะคิดว่าตัวเองทำผิด

ข้อสามคือสิ่งที่อันตรายที่สุด — บั๊กที่ระบบแจ้งข้อความสมเหตุสมผลจะอยู่ในระบบได้นานมาก เพราะไม่มีใครสงสัย

ทำไมต้องมี undo

เพราะเมื่อการยิงซ้ำถือเป็นเรื่องปกติ ความผิดพลาดจากการยิงเกินก็เป็นเรื่องปกติตามไปด้วย — พนักงานเผลอยิงกล่องเดิมสองครั้งย่อมเกิดขึ้น ถ้าไม่มีทางแก้ เขาจะต้องเริ่มนับใหม่ทั้งโซน

ลองต่อ: เขียนเทสต์ว่ายิงบาร์โค้ดเดียวกันสามครั้งแล้ว counts ของรหัสนั้นเป็น 3 และ totalItems เป็น 3 จากนั้นเรียก undo หนึ่งครั้งแล้วตรวจว่าเหลือ 2 — เทสต์นี้จะจับได้ทันทีถ้ามีคนเผลอเปลี่ยนไปใช้ Set

11. สรุปท้ายบท

  • ทะเบียนกันซ้ำต้องเก็บค่ามาตรฐาน ไม่ใช่ค่าดิบ
  • ใช้ Set คู่กับ List เพื่อได้ทั้งความเร็วและลำดับ
  • set.add() คืน false เมื่อมีอยู่แล้ว ใช้ประโยชน์ได้เลย
  • กฎการกันซ้ำขึ้นกับความหมายทางธุรกิจของสิ่งที่ยิง
  • บางหน้าจอการยิงซ้ำคือเรื่องถูกต้อง เช่นการนับสต็อก
  • กำหนดขอบเขตของ “รอบนี้” ให้ตรงกับงานจริง
  • ต้องกันสามชั้น — ที่หน้าจอ ระหว่างบันทึก และที่ server
  • debounce แก้ปัญหาคนละเรื่อง อย่าเอามาใช้แทน
