ระดับ: เริ่มต้น–กลางใช้เวลา: 30–40 นาทีลงมือทำ: ตรวจข้อมูลก่อนบันทึกและแสดงข้อความผิดพลาด
ต้องผ่านอะไรมาก่อน: 3.1 (TextEditingController) และ D1.4 (ฟังก์ชันที่คืน String? เมื่อผิด)
จบบทนี้แล้วจะทำได้: ใช้ Form ตรวจหลายช่องพร้อมกัน เขียน validator ที่ทดสอบแยกได้ และรู้ว่าการตรวจฝั่งแอปไม่ได้แทนการตรวจฝั่ง server

1. ปัญหาที่บทนี้แก้

เมื่อมีหลายช่องที่ต้องถูกต้องพร้อมกันก่อนบันทึก การเขียน if ตรวจทีละช่องแล้วแสดงข้อความเองจะยาวและซ้ำ

Flutter ให้ Form มาแก้เรื่องนี้ — ตรวจทุกช่องพร้อมกันด้วยคำสั่งเดียว และแสดงข้อความผิดพลาดใต้ช่องที่ผิดให้อัตโนมัติ

2. คำศัพท์ที่ต้องรู้ก่อน

คำความหมาย
Formตัวห่อที่รวมช่องกรอกหลายช่องเข้าด้วยกัน

autovalidateModeตั้งว่าจะตรวจตอนไหน

3. Mental model: null แปลว่าผ่าน

ข้อตกลงของ validator ใน Flutter ตรงกับที่เราเขียนไว้ในแบบฝึกหัด D1.4 พอดี

String? validateQuantity(String? raw, {int max = 999}) {
  final text = raw?.trim() ?? '';
  if (text.isEmpty) return 'กรุณากรอกจำนวน';

  final value = int.tryParse(text);
  if (value == null) return 'จำนวนต้องเป็นตัวเลข';
  if (value <= 0) return 'จำนวนต้องมากกว่า 0';
  if (value > max) return 'จำนวนต้องไม่เกิน $max';
  return null;                              // ผ่าน
}
ประโยคเดียวที่ต้องจำ: null แปลว่าผ่าน — ดูแปลกตอนแรก แต่ทำให้เขียน if (error != null) ได้ตรงไปตรงมา และบอกเหตุผลของความผิดพลาดไปพร้อมกัน (ทบทวน D1.13)

4. ประกอบเข้าด้วยกัน

class _ReceiveScreenState extends State<ReceiveScreen> {

        ],

  Future<void> _submit() async {
    // ตรวจทุกช่องพร้อมกัน ถ้าไม่ผ่านจะแสดงข้อความใต้ช่องเอง
    if (!(_formKey.currentState?.validate() ?? false)) return;
    ...

ทบทวนจาก 1.8: GlobalKey ต้องสร้างเป็น field ของ State ห้ามสร้างใน build ไม่งั้นจะได้ตัวใหม่ทุกรอบและ currentState จะเป็น null

ส่วน ?? false จำเป็นเพราะ currentState เป็น nullable — ถ้าเป็น null แปลว่าฟอร์มยังไม่ถูกสร้าง ซึ่งควรถือว่าไม่ผ่าน

5. validator ต้องเป็นฟังก์ชันธรรมดา

ใน lab เราแยก validateQuantity ออกไปไว้ใน scan_session.dart ซึ่งไม่ import Flutter เลย ผลคือทดสอบได้โดยไม่ต้องประกอบหน้าจอ

test('ปฏิเสธค่าว่าง ตัวอักษร ศูนย์ ติดลบ และเกินเพดาน', () {
  expect(validateQuantity(''), 'กรุณากรอกจำนวน');
  expect(validateQuantity('มาก'), 'จำนวนต้องเป็นตัวเลข');
  expect(validateQuantity('0'), 'จำนวนต้องมากกว่า 0');
  expect(validateQuantity('-3'), 'จำนวนต้องมากกว่า 0');
  expect(validateQuantity('1000'), 'จำนวนต้องไม่เกิน 999');
});
validator ควรเป็นห้ามเป็น
ทำงานเสร็จทันทีงานที่ต้องรอ เช่นถาม server
ให้ผลเหมือนเดิมเสมอเมื่อค่าเหมือนเดิมขึ้นกับเวลาหรือสิ่งภายนอก
ไม่เปลี่ยนอะไรข้างนอกแก้ controller หรือเรียก setState
ห้ามยิง API ใน validator เพราะ validate() เรียก validator ทุกตัวทันทีและอาจถูกเรียกหลายครั้ง — จะกลายเป็นยิง API รัวโดยไม่ตั้งใจ และ validator ต้องคืนคำตอบทันที รอไม่ได้ การตรวจกับ server เป็นเรื่องของขั้นตอนบันทึก ไม่ใช่ของ validator

6. ตรวจตอนไหน

autovalidateModeตรวจเมื่อเหมาะกับ
disabled (ค่าเริ่มต้น)เมื่อเราสั่ง validate() เท่านั้นงานคลังส่วนใหญ่
onUserInteractionหลังผู้ใช้แตะช่องแล้วฟอร์มยาวที่อยากบอกเร็ว
alwaysทุกครั้งที่วาดหน้าจอแทบไม่ควรใช้

สำหรับหน้าจอสแกน ค่าเริ่มต้นดีที่สุด — เพราะการขึ้นข้อความสีแดงตั้งแต่ยังไม่ทันกรอกอะไรทำให้ผู้ใช้รู้สึกว่าทำผิดตลอดเวลา ให้ตรวจตอนกดบันทึกแล้วบอกทีเดียว

7. ตรวจฝั่งแอปไม่ได้แทนฝั่ง server

เรื่องนี้สำคัญและมักถูกมองข้าม การตรวจในแอปมีไว้ให้ผู้ใช้รู้เร็ว ไม่ใช่เพื่อความปลอดภัยหรือความถูกต้องของข้อมูล

ระดับตรวจอะไรตรวจที่ไหน
รูปแบบว่างไหม เป็นตัวเลขไหม ยาวเกินไหมvalidator ในแอป
กฎในหน้าจอจำนวนไม่เกินที่เหลือ ไม่ซ้ำในรอบนี้ตรรกะในแอป
ข้อมูลจริงพาเลทนี้มีอยู่จริงไหม ช่องนี้ว่างไหมserver
สิทธิ์และความถูกต้องสุดท้ายคนนี้มีสิทธิ์ไหม ของยังอยู่ไหมserver เท่านั้น

เหตุผลง่าย ๆ คือแอปทำงานบนเครื่องของผู้ใช้ซึ่งเราควบคุมไม่ได้ และระหว่างที่พนักงานกรอกอยู่ คนอื่นอาจหยิบของไปแล้ว — server เป็นแหล่งความจริงเดียว เรื่องนี้จะลงลึกใน Part 7

9. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
currentState เป็น null ตลอดสร้าง GlobalKey ใน buildย้ายไปเป็น field ของ State
ใส่ validator แล้วไม่ทำงานใช้ TextField แทน TextFormFieldเปลี่ยนเป็น TextFormField
validator ไม่ถูกเรียกเลยช่องไม่ได้อยู่ใต้ Formตรวจว่า Form ครอบอยู่จริง
ข้อความแดงขึ้นตั้งแต่เปิดหน้าจอตั้ง autovalidateMode: alwaysใช้ค่าเริ่มต้น
ยิง API รัวโดยไม่ตั้งใจเรียก API ใน validatorย้ายไปขั้นตอนบันทึก
ผ่าน validator แต่ server ปฏิเสธกฎสองฝั่งไม่ตรงกันยอมรับว่า server คือแหล่งความจริง แล้วแสดงข้อความที่ server ส่งมา
ใช้ GlobalKey ตัวเดียวกับสองฟอร์มเข้าใจว่าเป็นตัวแปรกลางหนึ่งฟอร์มหนึ่ง key

10. แบบฝึกหัด

ฟอร์มย้ายสินค้ามีสามช่อง

  • รหัสพาเลท — ต้องขึ้นต้นด้วย PAL- ตามด้วยตัวเลข 4–6 หลัก
  • ตำแหน่งปลายทาง — รูปแบบ A-01-02
  • จำนวน — 1 ถึงจำนวนที่มีอยู่จริง ซึ่งส่งเข้ามาเป็นค่า
  1. เขียน validator ทั้งสามตัวแยกเป็นฟังก์ชันธรรมดา
  2. เขียนเทสต์ของ validator โดยไม่ต้องประกอบหน้าจอ
  3. อธิบายว่าอะไรใน 3 ข้อนี้ที่ยัง ต้อง ตรวจซ้ำที่ server
ดูแนวคำตอบและเหตุผล
// ไฟล์ตรรกะล้วน — ไม่ import Flutter
String? validatePalletCode(String? raw) {
  final value = raw?.trim().toUpperCase() ?? '';
  if (value.isEmpty) return 'กรุณาสแกนรหัสพาเลท';
  if (!RegExp(r'^PAL-\d{4,6}$').hasMatch(value)) {
    return 'รูปแบบต้องเป็น PAL-1001';
  }
  return null;
}

String? validateLocation(String? raw) {
  final value = raw?.trim().toUpperCase() ?? '';
  if (value.isEmpty) return 'กรุณาระบุตำแหน่งปลายทาง';
  if (!RegExp(r'^[A-Z]-\d{2}-\d{2}$').hasMatch(value)) {
    return 'รูปแบบต้องเป็น A-01-02';
  }
  return null;
}

String? validateMoveQuantity(String? raw, {required int available}) {
  final text = raw?.trim() ?? '';
  if (text.isEmpty) return 'กรุณากรอกจำนวน';

  final value = int.tryParse(text);
  if (value == null) return 'จำนวนต้องเป็นตัวเลข';
  if (value <= 0) return 'จำนวนต้องมากกว่า 0';
  if (value > available) return 'มีอยู่เพียง $available ชิ้น';
  return null;
}
void main() {
  test('รหัสพาเลท', () {
    expect(validatePalletCode('pal-1001'), isNull);      // normalize ให้ก่อน
    expect(validatePalletCode(''), 'กรุณาสแกนรหัสพาเลท');
    expect(validatePalletCode('PAL-1'), 'รูปแบบต้องเป็น PAL-1001');
    expect(validatePalletCode('BOX-1001'), 'รูปแบบต้องเป็น PAL-1001');
  });

  test('จำนวนเทียบกับของที่มีจริง', () {
    expect(validateMoveQuantity('5', available: 10), isNull);
    expect(validateMoveQuantity('11', available: 10), 'มีอยู่เพียง 10 ชิ้น');
    expect(validateMoveQuantity('0', available: 10), 'จำนวนต้องมากกว่า 0');
  });
}

ทำไม available ส่งเข้ามาเป็น parameter

เพราะถ้า validator ไปอ่านค่านี้จากที่เก็บข้อมูลเอง มันจะเลิกเป็นฟังก์ชันธรรมดาทันที ทดสอบยากขึ้น และผลลัพธ์จะขึ้นกับสิ่งภายนอก — การรับเข้ามาทำให้เทสต์กำหนดสถานการณ์ได้ทุกแบบในบรรทัดเดียว

ข้อ 3 — อะไรต้องตรวจซ้ำที่ server

เรื่องตรวจซ้ำไหมเหตุผล
รูปแบบรหัสพาเลทควรตรวจป้องกันข้อมูลผิดรูปเข้าฐานข้อมูลจากช่องทางอื่น
พาเลทนี้มีอยู่จริงไหมต้องแอปไม่มีทางรู้ ต้องถาม server
ตำแหน่งปลายทางว่างไหมต้องอาจมีคนอื่นวางของไปแล้วระหว่างที่เรากรอก
จำนวนไม่เกินที่มีต้องค่า available ที่แอปถืออาจเก่าไปแล้ว
คนนี้มีสิทธิ์ย้ายของไหมต้องแอปเปลี่ยนแปลงได้ ไม่ใช่แหล่งความจริงเรื่องสิทธิ์

บทเรียนที่สำคัญที่สุด: validator ในแอปตรวจ รูปแบบ ได้ครบถ้วน แต่ตรวจ ความจริง ณ ขณะนั้น ไม่ได้เลย — ข้อมูลที่แอปถืออยู่เป็นภาพนิ่งจากตอนที่โหลดมา ซึ่งอาจล้าสมัยไปแล้วตอนที่พนักงานกดบันทึก

ลองต่อ: เพิ่มเทสต์กรณี available: 0 แล้วดูว่าข้อความที่ได้คือ “มีอยู่เพียง 0 ชิ้น” ซึ่งอ่านแล้วแปลก — ลองแก้ให้แสดงข้อความที่เหมาะกว่า เช่น “ไม่มีของให้ย้าย” นี่คือตัวอย่างว่าเทสต์ช่วยให้เห็นกรณีที่ลืมคิด

11. สรุปท้ายบท

  • validator คืน null เมื่อผ่าน และคืนข้อความเมื่อผิด
  • Form ตรวจทุกช่องพร้อมกันและแสดงข้อความให้เอง
  • สร้าง GlobalKey เป็น field ของ State ห้ามสร้างใน build
  • ใช้ TextFormField ไม่ใช่ TextField เมื่อต้องการ validator
  • เขียน validator เป็นฟังก์ชันธรรมดาที่ทดสอบแยกได้
  • ห้ามยิง API หรือเรียก setState ใน validator
  • การตรวจในแอปมีไว้ให้รู้เร็ว ไม่ได้แทนการตรวจที่ server
