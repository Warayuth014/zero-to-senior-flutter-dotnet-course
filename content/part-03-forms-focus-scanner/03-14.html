1. ปัญหาที่ lab นี้แก้

ตลอด Part นี้เราเพิ่มธงเข้าไปทีละตัว — _submitting, _busy, _lastOutcome, _pendingCode ซึ่งใช้ได้กับหน้าจอที่มีขั้นตอนเดียว

3. ขั้นที่ 1: กำหนดสถานะ

ใช้ sealed class จาก D1.12 เพราะแต่ละสถานะพกข้อมูลคนละชุด ซึ่ง enum ทำไม่ได้

sealed class ScanState {
  const ScanState();
}

/// พร้อมรับการยิงบาร์โค้ด
class AwaitingScan extends ScanState {
  const AwaitingScan();
}

/// กำลังถามข้อมูลจาก server ว่ารหัสนี้คืออะไร
class Validating extends ScanState {
  const Validating(this.code);
  final String code;                    // จำไว้ว่ากำลังรอใบไหน (3.13)
}

/// ได้ข้อมูลแล้ว รอผู้ใช้ยืนยัน
class Review extends ScanState {
  const Review(this.preview);
  final PalletPreview preview;
}

/// กำลังบันทึกจริง
class Committing extends ScanState {
  const Committing(this.preview);
  final PalletPreview preview;
}

/// ล้มเหลว พร้อมบอกว่าลองใหม่ได้ไหม
class Failed extends ScanState {
  const Failed(this.message, {required this.retryable});
  final String message;
  final bool retryable;
}
ประโยชน์ที่ได้ทันที: ไม่มีทางที่จะอยู่ในสถานะ Validating และ Committing พร้อมกัน เพราะตัวแปรเก็บได้ค่าเดียว — สถานะที่ขัดแย้งกันเองถูกกำจัดตั้งแต่ตอนออกแบบชนิดข้อมูล

4. ขั้นที่ 2: กำหนดว่าย้ายไปไหนได้บ้าง

AwaitingScan  ──ยิงรหัสที่ถูกรูปแบบ──→  Validating
AwaitingScan  ──ยิงรหัสผิดรูปแบบ────→  Failed

Validating    ──server ตอบว่ามีจริง──→  Review
Validating    ──server ปฏิเสธ────────→  Failed

Review        ──ผู้ใช้ยกเลิก─────────→  AwaitingScan
Review        ──ผู้ใช้ยืนยัน─────────→  Committing

Committing    ──บันทึกสำเร็จ─────────→  AwaitingScan (พร้อมใบถัดไป)
Committing    ──บันทึกไม่สำเร็จ──────→  Failed

Failed        ──ผู้ใช้รับทราบ────────→  AwaitingScan

แผนภาพนี้ตอบคำถามที่โค้ดแบบ bool หลายตัวตอบไม่ได้ — ยิงบาร์โค้ดตอนอยู่ในสถานะ Review ได้ไหม คำตอบคือไม่ได้ เพราะไม่มีลูกศรจาก Review ที่รับการยิง

5. ขั้นที่ 3: เขียน controller

class ScanController extends ChangeNotifier {
  ScanController({required this.repository});

  final PalletRepository repository;

  ScanState _state = const AwaitingScan();
  ScanState get state => _state;

  void _emit(ScanState next) {
    _state = next;
    notifyListeners();          // ลืมบรรทัดนี้ = หน้าจอไม่อัปเดต (1.13)
  }

  Future<void> submitRaw(String raw) async {
    // กฎการย้าย — ยิงได้เฉพาะตอนพร้อมรับ
    if (_state is! AwaitingScan) return;

    final code = normalizeBarcode(raw);              // 3.6
    if (code.isEmpty) {
      return _emit(const Failed('ไม่มีข้อมูล', retryable: false));
    }
    if (!isValidPalletCode(code)) {
      return _emit(const Failed('รูปแบบไม่ถูกต้อง', retryable: false));
    }

    _emit(Validating(code));
    try {
      final preview = await repository.preview(code);

      // ตรวจว่ายังรอใบนี้อยู่ไหม — กันผลลัพธ์เก่า (3.13)
      if (_state case Validating(code: final current) when current == code) {
        _emit(Review(preview));
      }
    } on Exception catch (error) {
      if (_state is Validating) {
        _emit(Failed('เชื่อมต่อไม่ได้', retryable: true));
        debugPrint('preview failed: $error');
      }
    }
  }
}

สังเกตสามอย่างที่มาจากบทก่อน ๆ

บรรทัดมาจากบท
if (_state is! AwaitingScan) return;3.8 — กันคำสั่งซ้อน แต่แม่นกว่าเพราะรู้ว่าอยู่สถานะไหน
normalizeBarcode(raw)3.6 — แปลงก่อนใช้เสมอ
if (_state case Validating(code: ...) when ...)3.13 — กันผลลัพธ์เก่า

บรรทัดสุดท้ายใช้ pattern matching จาก D1.12 — ตรวจทั้งชนิดของสถานะและค่าข้างในพร้อมกันในบรรทัดเดียว

6. ขั้นที่ 4: การบันทึกที่ทนต่อการส่งซ้ำ

String? _operationId;

Future<void> confirm() async {
  final current = _state;
  if (current is! Review) return;        // ยืนยันได้เฉพาะตอนอยู่ในขั้นทบทวน

  // สร้างรหัสประจำคำสั่งครั้งเดียว แล้วใช้ซ้ำเมื่อลองใหม่ (3.8)
  _operationId ??= _newOperationId();

  _emit(Committing(current.preview));
  try {
    await repository.commit(current.preview, operationId: _operationId!);
    _operationId = null;                 // สำเร็จแล้วจึงเริ่มรหัสใหม่ครั้งหน้า
    _emit(const AwaitingScan());
  } on ConflictException catch (error) {
    _operationId = null;                 // ขัดแย้งแล้ว ลองซ้ำไม่ช่วย
    _emit(Failed(error.message, retryable: false));
  } on Exception {
    // ไม่ล้างรหัส — ถ้าลองใหม่จะใช้รหัสเดิม server จึงไม่บันทึกซ้ำ
    _emit(const Failed('บันทึกไม่สำเร็จ', retryable: true));
  }
}
จุดที่ต้องคิดให้ดีที่สุดในทั้ง Part — เมื่อเน็ตหลุดตอนบันทึก เราไม่รู้ว่า server บันทึกไปแล้วหรือยัง ตามที่คุยใน 3.8 หัวข้อ 8 การเก็บรหัสประจำคำสั่งไว้แล้วส่งรหัสเดิมเมื่อลองใหม่ ทำให้ server รู้ว่าเป็นคำสั่งเดิม จึงคืนผลเดิมแทนที่จะบันทึกซ้ำ

7. ขั้นที่ 5: หน้าจออ่านสถานะแล้ววาด

Widget build(BuildContext context) {
  return ListenableBuilder(
    listenable: controller,
    builder: (context, child) => switch (controller.state) {
      AwaitingScan() => ScanField(
          enabled: true,
          onSubmitted: controller.submitRaw,
        ),
      Validating(code: final code) => BusyView(message: 'กำลังตรวจสอบ $code'),
      Review(preview: final preview) => ReviewView(
          preview: preview,
          onCancel: controller.cancel,
          onConfirm: controller.confirm,
        ),
      Committing() => const BusyView(message: 'กำลังบันทึก'),
      Failed(message: final message, retryable: final retryable) => FailedView(
          message: message,
          onRetry: retryable ? controller.retry : null,
          onDismiss: controller.acknowledge,
        ),
    },
  );
}

หน้าจอไม่มีตรรกะเลย — มันแค่แปลงสถานะเป็นหน้าตา ซึ่งเป็นความหมายของ “build คือการถ่ายรูป” จาก 1.9

และเพราะ ScanState เป็น sealed เมื่อเพิ่มสถานะใหม่ analyzer จะชี้ทันทีว่า switch นี้ยังไม่ครอบคลุม ตามที่เรียนใน D1.12

8. ขั้นที่ 6: แยกสิ่งที่เกิดครั้งเดียวออกจากการวาด

เสียง การสั่น และการคืนโฟกัส เป็นสิ่งที่ต้องเกิด ครั้งเดียวตอนสถานะเปลี่ยน ไม่ใช่ทุกครั้งที่วาดหน้าจอ

ScanState? _previous;

void _onStateChanged() {
  final current = controller.state;
  if (current == _previous) return;

  switch (current) {
    case Failed():
      HapticFeedback.heavyImpact();          // 3.11
    case AwaitingScan():
      HapticFeedback.lightImpact();
      _focusNode.requestFocus();             // 3.2
    default:
      break;
  }
  _previous = current;
}
อย่าเรียกการสั่นหรือเล่นเสียงใน build เพราะ build ถูกเรียกได้บ่อยและคาดเดาไม่ได้ตาม 1.9 — ผลคือเครื่องสั่นรัวทุกครั้งที่หน้าจอวาดใหม่

11. แบบทดสอบตัวเองก่อนไป Part 4

คุณพร้อมไปต่อเมื่อทำสิ่งเหล่านี้ได้โดยไม่เปิดดูเฉลย

  • อธิบายได้ว่าเครื่องสแกนของ PDA ส่งข้อมูลถึงแอปอย่างไร
  • บอกได้ว่าทำไมโฟกัสเป็นเรื่องคอขาดบาดตายของงานสแกน
  • เขียน validator ที่ทดสอบแยกได้และมีข้อความเฉพาะแต่ละกรณี
  • ระบุได้ว่าข้อมูลชนิดไหนแปลงตัวพิมพ์ได้และชนิดไหนห้าม
  • แยกได้ว่ากันยิงซ้ำกับกันส่งซ้อนต่างกันอย่างไร
  • บอกได้ว่าทำไมปิดปุ่มอย่างเดียวไม่พอ
  • ออกแบบสัญญาณตอบรับที่รับรู้ได้โดยไม่มองจอ
  • ไล่ตรวจได้ว่าหน้าจอหนึ่งมีอะไรต้องคืนบ้าง
  • อธิบายได้ว่าทำไม mounted ไม่ได้กันผลลัพธ์เก่า

12. สรุปปิด Part 3

  • เครื่องสแกนคือแป้นพิมพ์ — ทุกอย่างที่เรียนเรื่องช่องกรอกใช้ได้ตรง ๆ
  • ถ้าไม่มีช่องที่โฟกัสอยู่ ข้อมูลที่ยิงมาหายไปเงียบ ๆ
  • ข้อมูลดิบต้องผ่านการแปลงและตรวจก่อนเข้าสู่ตรรกะเสมอ
  • กฎการกันซ้ำขึ้นกับความหมายทางธุรกิจ ไม่ใช่กฎกลางทั้งแอป
  • การป้องกันจริงอยู่ในตรรกะ ส่วนการปิดปุ่มเป็นการสื่อสาร
  • พนักงานไม่ได้มองจอ ต้องมีสัญญาณที่รับรู้ได้โดยไม่มอง
  • ทุกอย่างที่สร้างต้องมีคนคืน
  • โลกเปลี่ยนไปได้ระหว่างที่เรารอ ต้องตรวจทั้งว่าหน้าจอยังอยู่และผลยังตรง
  • ตรรกะที่แยกออกจากหน้าจอได้ คือตรรกะที่ทดสอบได้
