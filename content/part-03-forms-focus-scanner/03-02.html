ระดับ: เริ่มต้น–กลางใช้เวลา: 30–40 นาทีลงมือทำ: ควบคุมว่าช่องไหนกำลังรับข้อมูล
ต้องผ่านอะไรมาก่อน: 3.1 (TextEditingController) และ 1.6 (วงจรชีวิตกับการคืนทรัพยากร)
จบบทนี้แล้วจะทำได้: ควบคุมโฟกัสได้ตามที่ต้องการ และเข้าใจว่าทำไมบางครั้งสั่งโฟกัสแล้วไม่ติด

1. ปัญหาที่บทนี้แก้

บนมือถือทั่วไป โฟกัสเป็นเรื่องเล็ก — ผู้ใช้แตะช่องที่ต้องการเอง แต่บน PDA มันเป็นเรื่องใหญ่ที่สุดของงานสแกน

เหตุผลคือเครื่องสแกนทำงานเหมือนแป้นพิมพ์ มันพิมพ์รหัสลงในช่องที่ กำลังโฟกัสอยู่ แล้วกด Enter ให้ ถ้าไม่มีช่องไหนโฟกัสอยู่ รหัสจะหายไปเฉย ๆ โดยไม่มี error อะไรเลย

อาการที่พนักงานจะรายงานคือ “ยิงแล้วไม่เข้า” ซึ่งหาสาเหตุยากมากถ้าไม่รู้เรื่องนี้

2. คำศัพท์ที่ต้องรู้ก่อน

คำความหมาย
FocusNodeตัวแทนของ “ความเป็นช่องที่รับข้อมูลอยู่”
autofocusขอโฟกัสทันทีที่แสดงครั้งแรก
requestFocus()สั่งให้โฟกัสมาที่ช่องนี้
unfocus()ปล่อยโฟกัส ทำให้แป้นพิมพ์ปิด
post-frame callbackโค้ดที่ทำงานหลังวาดหน้าจอเสร็จ

3. Mental model: มีได้ทีละหนึ่ง

ในหน้าจอหนึ่ง มีช่องที่โฟกัสอยู่ได้ทีละช่องเดียวเท่านั้น เหมือนไฟฉายที่ส่องได้ทีละจุด

เครื่องสแกนยิง "PAL-1001" + Enter
        ↓
ระบบส่งไปที่ช่องที่โฟกัสอยู่
        ↓
    ┌───────────┐
    │ ช่องบาร์โค้ด │  ← โฟกัสอยู่ ได้รับ
    └───────────┘
    ┌───────────┐
    │ ช่องจำนวน   │  ← ไม่ได้รับ
    └───────────┘
ประโยคเดียวที่ต้องจำ: ในงานสแกน หน้าที่ของเราคือ ทำให้ช่องบาร์โค้ดโฟกัสอยู่เสมอ — ทุกครั้งที่โฟกัสหลุดไปโดยไม่ตั้งใจ คือรหัสที่หายไปหนึ่งตัว

4. สร้างและคืน FocusNode

class _ReceiveScreenState extends State<ReceiveScreen> {
  final _barcodeFocus = FocusNode();

  @override
  void dispose() {
    _barcodeFocus.dispose();      // คู่ตรงข้ามของการสร้าง (1.6)
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return TextFormField(
      controller: _barcodeController,
      focusNode: _barcodeFocus,
      autofocus: true,            // โฟกัสทันทีที่เปิดหน้าจอ
      ...
    );
  }
}

กฎเดียวกับ controller ใน 3.1 ทุกประการ — สร้างเป็น field ของ State ห้ามสร้างใน build และคืนใน dispose

5. คืนโฟกัสหลังยิงเสร็จ

นี่คือบรรทัดที่สำคัญที่สุดของหน้าจอสแกนทั้งหน้า

void _handleScan() {
  final outcome = _session.accept(_barcodeController.text);

  if (outcome.shouldClearInput) {
    _barcodeController.clear();
  }
  _giveFeedback(outcome);
  setState(() => _lastOutcome = outcome);

  // คืนโฟกัสทันที เพื่อให้ยิงตัวถัดไปได้โดยไม่ต้องแตะจอ
  _barcodeFocus.requestFocus();
}

ถ้าไม่มีบรรทัดสุดท้าย พนักงานจะต้องแตะจอก่อนยิงทุกครั้ง ซึ่งเมื่อทำงาน 100 ครั้งต่อกะแปลว่าแตะจอเพิ่ม 100 ครั้งโดยไม่จำเป็น

เทสต์ใน lab ตรวจข้อนี้ไว้แล้ว

expect(field.controller?.text, isEmpty);      // ล้างแล้ว
expect(field.focusNode?.hasFocus, isTrue);    // และโฟกัสกลับมาแล้ว

6. ทำไมบางครั้งสั่งโฟกัสแล้วไม่ติด

อาการที่งงที่สุดของบทนี้ — สั่ง requestFocus() แล้วไม่มีอะไรเกิดขึ้น

// ไม่ติด — ช่องปลายทางยังไม่มีอยู่ในหน้าจอ
void _goToNextStep() {
  setState(() => _step = ScanStep.quantity);
  _quantityFocus.requestFocus();      // ช่องจำนวนยังไม่ถูกสร้าง
}

เหตุผลตรงกับสิ่งที่เรียนใน 1.5 — setState ไม่ได้วาดหน้าจอใหม่ทันที มันแค่ทำเครื่องหมายไว้ การวาดจริงเกิดในรอบถัดไป ดังนั้นตอนที่เราสั่งโฟกัส ช่องจำนวนยังไม่มีตัวตนอยู่ใน tree เลย

// ติด — รอให้วาดเสร็จก่อน
void _goToNextStep() {
  setState(() => _step = ScanStep.quantity);

  WidgetsBinding.instance.addPostFrameCallback((_) {
    if (!mounted) return;                  // ผู้ใช้อาจออกไปแล้ว (1.6)
    _quantityFocus.requestFocus();
  });
}

addPostFrameCallback แปลว่า “ทำหลังวาดหน้าจอรอบนี้เสร็จ” ซึ่งเป็นตอนที่ช่องใหม่มีตัวตนแล้ว

อย่าลืม if (!mounted) return; เพราะโค้ดนี้ทำงานทีหลัง และระหว่างนั้นผู้ใช้อาจกดย้อนกลับไปแล้ว — เป็นเรื่องเดียวกับที่ 1.6 หัวข้อ 8 เตือนไว้

7. กล่องโต้ตอบแย่งโฟกัสไป

เมื่อเปิดกล่องยืนยัน โฟกัสจะย้ายไปที่กล่องนั้น เมื่อปิดกล่องแล้วต้องคืนโฟกัสกลับมาเอง

Future<void> _submit() async {
  final confirmed = await _confirmSubmit();
  if (!mounted || confirmed != true) return;

  // ... บันทึก ...

  if (mounted) _barcodeFocus.requestFocus();   // คืนโฟกัสหลังกล่องปิด
}

ลำดับสำคัญ — ต้อง await ให้กล่องปิดก่อน แล้วค่อยขอโฟกัส ถ้าขอตอนกล่องยังเปิดอยู่ กล่องจะแย่งกลับไป และที่แย่กว่านั้นคือถ้าพนักงานยิงบาร์โค้ดตอนกล่องเปิดอยู่ รหัสจะไปโผล่ในที่ที่ไม่ควร

8. อย่าสั่งโฟกัสใน build

// ผิด — เป็นผลข้างเคียงใน build (1.9)
@override
Widget build(BuildContext context) {
  _barcodeFocus.requestFocus();     // ทำงานทุกครั้งที่วาดใหม่
  return TextField(...);
}
อาการที่จะเจอสาเหตุ
แป้นพิมพ์กระพริบขึ้นลงขอโฟกัสซ้ำ ๆ ทุกรอบ
ผู้ใช้แตะช่องอื่นแล้วเด้งกลับโค้ดแย่งโฟกัสคืนทันที
พิมพ์ในช่องจำนวนไม่ได้โฟกัสถูกดึงกลับไปช่องบาร์โค้ดตลอด

ให้ผูกการสั่งโฟกัสกับ เหตุการณ์ เสมอ — ยิงเสร็จ ปิดกล่อง เปลี่ยนขั้นตอน ไม่ใช่ผูกกับการวาดหน้าจอ

9. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ยิงบาร์โค้ดแล้วไม่มีอะไรเกิดขึ้นไม่มีช่องไหนโฟกัสอยู่ใส่ autofocus และคืนโฟกัสหลังยิงทุกครั้ง
สั่ง requestFocus แล้วไม่ติดช่องปลายทางยังไม่ถูกสร้างใช้ addPostFrameCallback
แป้นพิมพ์กระพริบสั่งโฟกัสใน buildผูกกับเหตุการณ์แทน
รหัสไปโผล่ในช่องจำนวนโฟกัสอยู่ผิดช่องคืนโฟกัสไปช่องบาร์โค้ดหลังจบทุกขั้นตอน
หลังปิดกล่องแล้วยิงไม่เข้าไม่ได้คืนโฟกัสขอโฟกัสหลัง await กล่องปิด
ทุกช่องตั้ง autofocus: true แล้วสับสนมีหลายช่องแย่งกันตั้งได้ช่องเดียวต่อหน้าจอ
แอปกินหน่วยความจำเพิ่มลืม dispose FocusNodeคืนทุกตัวใน dispose

10. แบบฝึกหัด

หน้าจอหยิบสินค้ามีสองขั้นตอน

  1. ยิงรหัสพาเลท เมื่อสำเร็จให้ไปขั้นที่ 2
  2. ยิงรหัสตำแหน่ง เมื่อสำเร็จให้กลับไปขั้นที่ 1 เพื่อรับใบถัดไป

จงเขียนโครงการจัดการโฟกัส และอธิบายว่าทำไมสั่งโฟกัสตรง ๆ หลัง setState ถึงไม่พอ

ดูแนวคำตอบและเหตุผล
enum PickStep { pallet, location }

class _PickScreenState extends State<PickScreen> {
  final _palletFocus = FocusNode(debugLabel: 'pallet');
  final _locationFocus = FocusNode(debugLabel: 'location');
  final _palletController = TextEditingController();
  final _locationController = TextEditingController();

  PickStep _step = PickStep.pallet;

  @override
  void dispose() {
    _palletFocus.dispose();
    _locationFocus.dispose();
    _palletController.dispose();
    _locationController.dispose();
    super.dispose();
  }

  /// จุดเดียวที่จัดการโฟกัส — อ่านจากสถานะปัจจุบันเสมอ
  void _focusCurrentStep() {
    WidgetsBinding.instance.addPostFrameCallback((_) {
      if (!mounted) return;
      switch (_step) {
        case PickStep.pallet:
          _palletFocus.requestFocus();
        case PickStep.location:
          _locationFocus.requestFocus();
      }
    });
  }

  void _onPalletScanned() {
    _palletController.clear();
    setState(() => _step = PickStep.location);
    _focusCurrentStep();
  }

  void _onLocationScanned() {
    _locationController.clear();
    setState(() => _step = PickStep.pallet);
    _focusCurrentStep();
  }
}

ทำไมสั่งโฟกัสตรง ๆ หลัง setState ไม่พอ

เพราะ setState เพียงทำเครื่องหมายว่าต้องวาดใหม่ ตามที่เรียนใน 1.5 หัวข้อ 5 — ณ บรรทัดถัดไป ช่องของขั้นตอนใหม่ยังไม่ถูกสร้าง FocusNode ของมันจึงยังไม่ได้ผูกกับอะไร การสั่ง requestFocus ตอนนั้นจึงไม่มีผล

ที่ทำให้บั๊กนี้ยากคือ บางครั้งมันติด — ถ้าช่องปลายทางบังเอิญมีอยู่แล้วในรอบก่อน (เช่นแค่ซ่อนไว้ไม่ได้ถอดออก) การสั่งตรง ๆ จะทำงาน ผู้พัฒนาจึงคิดว่าโค้ดถูก แล้วไปพังตอนที่โครงหน้าจอเปลี่ยน

ทำไมรวมการจัดการโฟกัสไว้ที่ _focusCurrentStep จุดเดียว

เพราะมันอ่านจาก _step ปัจจุบันเสมอ ไม่ใช่จำว่าใครเรียกมา ข้อดีคือเมื่อเพิ่มขั้นตอนที่สามในอนาคต เราแก้ที่เดียว และ switch บน enum จะให้ analyzer เตือนถ้าลืมกรณีใหม่ ตามที่เรียนใน D1.10

จุดที่คนมักพลาด: เขียนเป็น _locationFocus.requestFocus() ตรง ๆ ในแต่ละ method ซึ่งได้ผลเหมือนกันตอนนี้ แต่เมื่อมีเส้นทางที่สามที่พาไปขั้นตอนเดิม จะมีจุดที่ลืมสั่งโฟกัส และเกิดอาการ “ยิงแล้วไม่เข้า” เฉพาะบางเส้นทาง

ลองต่อ: ลบ addPostFrameCallback ออกแล้วเปลี่ยนเป็นสั่งตรง ๆ จากนั้นทดสอบสลับขั้นตอนไปมา — บนเครื่องที่เร็วอาจดูเหมือนทำงานได้ ให้ลองบน PDA จริงหรือเพิ่มความซับซ้อนของหน้าจอ จะเห็นอาการชัดขึ้น

11. สรุปท้ายบท

  • มีช่องที่โฟกัสอยู่ได้ทีละช่องเดียว และเครื่องสแกนส่งข้อมูลไปที่ช่องนั้น
  • ถ้าไม่มีช่องไหนโฟกัส รหัสที่ยิงมาจะหายไปเงียบ ๆ
  • สร้าง FocusNode เป็น field ของ State และคืนใน dispose
  • คืนโฟกัสหลังยิงเสร็จทุกครั้ง เพื่อไม่ให้ต้องแตะจอ
  • สั่งโฟกัสไปยังช่องที่เพิ่งถูกสร้าง ต้องใช้ addPostFrameCallback
  • กล่องโต้ตอบแย่งโฟกัส ต้อง await ให้ปิดก่อนแล้วคืน
  • อย่าสั่งโฟกัสใน build
