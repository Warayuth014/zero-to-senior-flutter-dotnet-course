1. ปัญหาที่บทนี้แก้

หน้าจอสแกนหนึ่งหน้ามีของที่ต้องคืนหลายชิ้น มากกว่าหน้าจอแสดงผลธรรมดามาก

final _barcodeController = TextEditingController();
final _quantityController = TextEditingController(text: '1');
final _barcodeFocus = FocusNode();

ยังไม่นับตัวจับเวลา การฟังเหตุการณ์ และตัวควบคุมการเลื่อน — ถ้าลืมคืนแม้ชิ้นเดียว จะเกิดอาการที่หาสาเหตุยากมาก เพราะมันไม่พังทันที แต่ค่อย ๆ สะสม

3. Mental model: บัญชีคู่

ทบทวนกฎจาก 1.6 — ทุกอย่างที่เริ่มไว้ต้องมีคู่ตรงข้ามที่ปิด

5. ลำดับใน dispose

@override
void dispose() {
  _timer?.cancel();                    // 1. หยุดสิ่งที่ยังทำงานอยู่ก่อน
  _scanner.removeListener(_onScan);    // 2. เลิกฟัง
  _controller.dispose();               // 3. คืนตัวควบคุม
  _focus.dispose();
  super.dispose();                     // 4. ท้ายสุดเสมอ
}

เหตุผลของลำดับ — ถ้าคืนตัวควบคุมก่อนหยุดตัวจับเวลา ตัวจับเวลาที่ยิงพอดีจะไปแตะตัวควบคุมที่ถูกคืนไปแล้ว แล้วได้ error

ส่วน super.dispose() ต้องอยู่ท้ายสุดเสมอ ตรงข้ามกับ super.initState() ที่อยู่บนสุด

6. ของที่ยืมมาจากภายนอก

เมื่อ widget รับตัวควบคุมมาจากแม่ผ่าน constructor ห้ามคืน เพราะแม่อาจยังใช้ต่อ

class ScanField extends StatefulWidget {
  const ScanField({super.key, this.controller});

  final TextEditingController? controller;   // อาจส่งมาหรือไม่ก็ได้
  ...
}

class _ScanFieldState extends State<ScanField> {
  TextEditingController? _owned;

  // ใช้ของแม่ถ้ามี ไม่งั้นใช้ของตัวเอง
  TextEditingController get _controller => widget.controller ?? _owned!;

  @override
  void initState() {
    super.initState();
    if (widget.controller == null) {
      _owned = TextEditingController();     // สร้างเองเมื่อไม่มีคนให้
    }
  }

  @override
  void dispose() {
    _owned?.dispose();                      // คืนเฉพาะที่สร้างเอง
    super.dispose();
  }
}

รูปแบบนี้ทำให้ widget ใช้ได้สองแบบ — แม่ที่ต้องการควบคุมค่าเองก็ส่งตัวควบคุมมา ส่วนแม่ที่ไม่สนใจก็ไม่ต้องส่ง

การคืนของที่ยืมมาอันตรายพอ ๆ กับการลืมคืน — จะได้ error ว่า A TextEditingController was used after being disposed ตอนที่แม่พยายามใช้ต่อ ซึ่งเกิดห่างจากจุดที่เขียนผิดมาก จึงหายาก

7. เมื่อมีของหลายชิ้นตามข้อมูล

หน้าจอนับสต็อกที่มีช่องกรอกหนึ่งช่องต่อสินค้าหนึ่งชนิด ต้องสร้างตัวควบคุมหลายตัว

final Map<String, TextEditingController> _controllers = {};

TextEditingController _controllerFor(String productCode) {
  return _controllers.putIfAbsent(productCode, TextEditingController.new);
}

@override
void dispose() {
  for (final controller in _controllers.values) {
    controller.dispose();               // คืนทุกตัว
  }
  _controllers.clear();
  super.dispose();
}

สองข้อที่ต้องระวังเป็นพิเศษ

  1. ใช้รหัสจริงเป็นกุญแจ ไม่ใช่ตำแหน่ง — เพราะเมื่อรายการเรียงใหม่ ตัวควบคุมจะย้ายไปอยู่กับสินค้าคนละตัว เป็นปัญหาเดียวกับ key ใน 1.8
  2. เมื่อรายการหายไป ต้องคืนตัวควบคุมของมัน ไม่งั้นจะสะสมไปเรื่อย ๆ ตลอดอายุหน้าจอ
void _removeProduct(String productCode) {
  _controllers.remove(productCode)?.dispose();   // เอาออกแล้วคืนทันที
  setState(() => _products.removeWhere((p) => p.code == productCode));
}

8. อาการที่บอกว่าลืมคืน

อาการลืมคืนอะไร
เข้าออกหน้าจอซ้ำ ๆ แล้วแอปช้าลงเรื่อย ๆตัวควบคุมหรือตัวจับเวลา
โค้ดของหน้าจอเก่ายังทำงานหลังออกไปแล้วตัวจับเวลาหรือการฟังเหตุการณ์
setState() called after dispose()งานที่รออยู่ยังทำงานต่อ (1.6)
ข้อมูลของหน้าจอเก่าโผล่มาทับหน้าจอใหม่การฟังเหตุการณ์ที่ไม่ได้ยกเลิก

อาการทั้งหมดนี้มีจุดร่วมคือ ไม่พังทันที — จึงผ่านการทดสอบด้วยมือได้สบาย และไปโผล่ตอนพนักงานใช้งานยาว ๆ ทั้งกะ

ทางที่ดีที่สุดคือเขียนเทสต์ที่ประกอบหน้าจอ ถอดออก แล้วตรวจว่าไม่มีอะไรทำงานต่อ ตามรูปแบบในแบบฝึกหัด 1.6

9. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
แอปช้าลงเมื่อใช้งานนานลืมคืนตัวควบคุมไล่ตรวจ field ทุกตัวว่ามีคู่ใน dispose
used after being disposedคืนของที่ยืมมาจากแม่คืนเฉพาะที่สร้างเอง
คืนสองครั้งทั้งแม่และลูกคืนตกลงกันให้ชัดว่าใครเป็นเจ้าของ
ตัวจับเวลาไปแตะของที่คืนแล้วลำดับใน dispose ผิดหยุดตัวจับเวลาก่อนคืนตัวควบคุม
ตัวควบคุมย้ายไปอยู่กับสินค้าคนละตัวใช้ตำแหน่งเป็นกุญแจใช้รหัสจริง
ตัวควบคุมสะสมเมื่อลบรายการลบข้อมูลแต่ไม่คืนตัวควบคุมคืนตอนที่ลบ
เรียก setState ใน disposeเข้าใจว่าต้องอัปเดตหน้าจอไม่ต้องทำ หน้าจอกำลังถูกทำลายอยู่แล้ว

10. แบบฝึกหัด

โค้ดนี้มีปัญหาเรื่องการคืนทรัพยากรสี่จุด จงหาให้ครบและแก้

class _CountScreenState extends State<CountScreen> {
  final Map<int, TextEditingController> _controllers = {};
  final _scrollController = ScrollController();
  Timer? _autoSaveTimer;

  @override
  void initState() {
    super.initState();
    _autoSaveTimer = Timer.periodic(
      const Duration(seconds: 30),
      (_) => _autoSave(),
    );
    widget.scanner.addListener(_onScan);
  }

  TextEditingController _controllerFor(int index) {
    return _controllers.putIfAbsent(index, TextEditingController.new);
  }

  void _removeLine(int index) {
    setState(() => _lines.removeAt(index));
  }

  @override
  void dispose() {
    _scrollController.dispose();
    super.dispose();
  }
}
ดูแนวคำตอบและเหตุผล

ปัญหาที่พบ

จุดปัญหาผลที่เกิด
1ไม่ได้คืนตัวควบคุมใน _controllersสะสมทุกครั้งที่เข้าหน้าจอ
2ไม่ได้หยุด _autoSaveTimerบันทึกอัตโนมัติทำงานต่อไปตลอดอายุแอป
3ไม่ได้เลิกฟัง widget.scannerโค้ดของหน้าจอเก่ายังทำงานเมื่อยิงบาร์โค้ด
4ใช้ index เป็นกุญแจ และลบรายการโดยไม่คืนตัวควบคุมตัวควบคุมย้ายไปอยู่กับรายการคนละตัวหลังลบ
class _CountScreenState extends State<CountScreen> {
  // แก้ปัญหาที่ 4 — ใช้รหัสจริงเป็นกุญแจ ไม่ใช่ตำแหน่ง
  final Map<String, TextEditingController> _controllers = {};
  final _scrollController = ScrollController();
  Timer? _autoSaveTimer;

  @override
  void initState() {
    super.initState();
    _autoSaveTimer = Timer.periodic(
      const Duration(seconds: 30),
      (_) => _autoSave(),
    );
    widget.scanner.addListener(_onScan);
  }

  TextEditingController _controllerFor(String productCode) {
    return _controllers.putIfAbsent(productCode, TextEditingController.new);
  }

  void _removeLine(String productCode) {
    // แก้ปัญหาที่ 4 — คืนตัวควบคุมของรายการที่ถูกลบทันที
    _controllers.remove(productCode)?.dispose();
    setState(() => _lines.removeWhere((line) => line.code == productCode));
  }

  @override
  void dispose() {
    _autoSaveTimer?.cancel();                    // แก้ที่ 2 — หยุดก่อน
    widget.scanner.removeListener(_onScan);      // แก้ที่ 3 — เลิกฟัง
    for (final controller in _controllers.values) {
      controller.dispose();                      // แก้ที่ 1 — คืนทุกตัว
    }
    _controllers.clear();
    _scrollController.dispose();
    super.dispose();
  }
}

ทำไมปัญหาที่ 3 ร้ายแรงที่สุด

เพราะ widget.scanner เป็นของที่อยู่ ยาวกว่าหน้าจอ — เมื่อเราเลิกฟังไม่ทัน มันจะยังถือ callback ของหน้าจอที่ตายไปแล้วไว้ ผลคือ

  • หน้าจอเก่าถูกยึดไว้ไม่ให้คืนหน่วยความจำ พร้อมทุกอย่างที่มันถืออยู่
  • เมื่อพนักงานยิงบาร์โค้ดในหน้าจอใหม่ โค้ดของหน้าจอเก่าจะทำงานด้วย
  • ถ้าหน้าจอเก่าเรียก setState จะได้ setState() called after dispose()

และถ้าเข้าออกหน้าจอสิบครั้ง จะมีสิบตัวที่ฟังอยู่พร้อมกัน — การยิงหนึ่งครั้งทำให้โค้ดทำงานสิบรอบ

ทำไมปัญหาที่ 4 ถึงตรวจไม่เจอด้วยการทดสอบทั่วไป

เพราะมันเกิดเฉพาะเมื่อ ลบรายการที่ไม่ใช่ตัวสุดท้าย — เมื่อลบรายการที่ตำแหน่ง 2 รายการที่เคยอยู่ตำแหน่ง 3 จะเลื่อนมาเป็น 2 แล้วได้ตัวควบคุมของรายการที่ถูกลบไป พร้อมตัวเลขที่พนักงานกรอกไว้

ผลคือจำนวนของสินค้าคนละชนิดสลับกัน โดยหน้าจอดูปกติทุกอย่าง — เป็นปัญหาเดียวกับที่ 1.8 อธิบายเรื่อง key ทุกประการ ต่างแค่เกิดกับตัวควบคุมแทนที่จะเกิดกับ widget

ลองต่อ: เขียนเทสต์ที่ประกอบหน้าจอ กรอกค่าลงรายการที่สอง ลบรายการแรก แล้วตรวจว่าค่าที่กรอกยังอยู่กับสินค้าเดิม — เทสต์นี้จะล้มเหลวทันทีถ้าใช้ index เป็นกุญแจ

11. สรุปท้ายบท

  • ทุกอย่างที่สร้างต้องมีคู่ใน dispose
  • ไล่อ่าน field ทุกตัวแล้วถามว่ามีคู่หรือยัง
  • ถ้ามี dispose หรือ cancel แปลว่าต้องเรียก
  • ลำดับ — หยุดสิ่งที่ทำงานอยู่ก่อน แล้วค่อยคืนตัวควบคุม
  • super.dispose() อยู่ท้ายสุดเสมอ
  • คืนเฉพาะของที่สร้างเอง อย่าคืนของที่ยืมมา
  • เมื่อมีของหลายชิ้นตามข้อมูล ให้ใช้รหัสจริงเป็นกุญแจ และคืนตอนลบ
  • อาการของการลืมคืนไม่พังทันที จึงต้องใช้เทสต์จับ
