ระดับ: กลางใช้เวลา: 30–40 นาทีลงมือทำ: ถามยืนยันให้ผู้ใช้อ่านจริง
ต้องผ่านอะไรมาก่อน: 5.1 (route และการ pop ให้ถูกตัว) และ 5.3 (ผลลัพธ์ที่ส่งกลับ)
จบบทนี้แล้วจะทำได้: เขียนกล่องยืนยันที่คืนค่าถูกต้องทุกทางออก และรู้ว่าเมื่อไหร่ไม่ควรถามเลย

Future<bool> showDiscardDialog(BuildContext context) async {
  final confirmed = await showDialog<bool>(
    context: context,
    builder: (dialogContext) => AlertDialog(

      content: const Text('จำนวนและหมายเหตุที่แก้ไว้จะหายไปทั้งหมด'),

          child: const Text('แก้ต่อ'),

          onPressed: () => Navigator.of(dialogContext).pop(true),
          child: const Text('ทิ้งไป'),
        ),

สี่จุดที่ทำให้ฟังก์ชันนี้ปลอดภัย

จุดทำไม
คืน Future<bool> ไม่ใช่ bool?ฝั่งที่เรียกไม่ต้องคิดเรื่อง null เลย

แยกเป็นฟังก์ชันของตัวเองเรียกจากหลายที่ได้ และเทสต์ได้

5. null คือกรณีที่ต้องเลือกทางที่ปลอดภัย

กล่องหนึ่งกล่องมีทางออกสี่ทาง แต่เราควบคุมได้แค่สองทาง

ทางออกค่าที่ได้เราควบคุมได้ไหม
กด “แก้ต่อ”falseได้
กด “ทิ้งไป”trueได้
แตะนอกกล่องnullไม่ได้
กดปุ่มย้อนกลับnullไม่ได้
// อันตรายมาก
if (await showDeleteDialog(context)!) { ... }   // แตะนอกกล่อง = พัง

// อันตรายกว่า
final result = await showDialog<bool>(...);
if (result != false) deleteEverything();        // null ก็ผ่าน — ลบทันที

บรรทัดที่สองคือบั๊กที่ร้ายจริง เพราะ null != false เป็นจริง การแตะนอกกล่องโดยไม่ตั้งใจจึงกลายเป็นการยืนยัน

กฎที่ใช้ได้ทุกครั้ง — null ต้องแปลว่า “ไม่ทำ” เสมอ เพราะมันหมายถึงผู้ใช้ยังไม่ได้ตัดสินใจ

6. barrierDismissible: ให้แตะนอกกล่องได้ไหม

สถานการณ์ตั้งเป็นเหตุผล
ถามยืนยันทั่วไปtrue (ค่าเริ่มต้น)ปิดง่าย และ null แปลว่าไม่ทำอยู่แล้ว
การกระทำที่ย้อนไม่ได้falseบังคับให้เลือกอย่างมีสติ
กำลังส่งข้อมูลอยู่falseปิดไปแล้วผู้ใช้จะไม่รู้ผล
showDialog<bool>(
  context: context,
  barrierDismissible: false,     // ต้องเลือกปุ่ม
  builder: (dialogContext) => AlertDialog(...),
);
barrierDismissible: false ไม่ได้กันปุ่มย้อนกลับของเครื่อง — ยังปิดได้และยังได้ null อยู่ดี ถ้าต้องกันจริงต้องครอบด้วย PopScope ตาม 5.5

7. ปุ่มต้องเป็นคำกริยา

ผู้ใช้ที่รีบจะอ่านปุ่มก่อน แล้วอ่านคำถามทีหลัง (หรือไม่อ่านเลย) ปุ่มจึงต้องบอกตัวเองได้

ยกเลิกใบรับเข้านี้?
[ ไม่ใช่ ]  [ ใช่ ]          ← ต้องอ่านคำถามก่อนถึงจะรู้

ยกเลิกใบรับเข้านี้?
[ ทำต่อ ]  [ ยกเลิกใบนี้ ]    ← อ่านปุ่มอย่างเดียวก็รู้
หลักตัวอย่าง
ปุ่มบอกว่าจะเกิดอะไร“ทิ้งไป” ไม่ใช่ “ตกลง”
ทางที่ปลอดภัยอยู่ซ้ายตำแหน่งที่นิ้วโดนน้อยกว่า
เนื้อหาบอกสิ่งที่จะเสีย“18 บรรทัดที่สแกนไว้จะหายไป”
ระบุตัวสิ่งของให้ชัด“ลบโปรไฟล์ WMS-PROD” ไม่ใช่ “ลบรายการนี้”

แถวสุดท้ายสำคัญเมื่อผู้ใช้เปิดหลายอย่างพร้อมกัน — ถ้ากล่องไม่บอกว่ากำลังจะลบอะไร เขาต้องปิดกล่องไปดูก่อนแล้วกลับมาใหม่

8. อย่าถามในสิ่งที่ควรทำให้ย้อนได้แทน

กล่องยืนยันเป็นเครื่องมือที่แพงที่สุดในการขอความสนใจ — มันหยุดทุกอย่างและบังคับให้ตอบ

การกระทำควรใช้เพราะ
ลบใบรับเข้าทั้งใบกล่องยืนยันย้อนไม่ได้ ผลกระทบสูง
ทิ้งร่างที่แก้ไว้กล่องยืนยันย้อนไม่ได้
ลบบรรทัดที่เพิ่งสแกนลบเลย + ปุ่มเลิกทำย้อนได้ง่าย และเกิดบ่อย
สแกนซ้ำเตือนด้วยสีและเสียงเกิดทุกไม่กี่วินาที ถ้าเด้งกล่องคืองานหยุด

สองแถวล่างสำคัญบน PDA เป็นพิเศษ เพราะงานสแกนคือการทำสิ่งเดิมซ้ำเร็ว ๆ หลายร้อยครั้ง

สแกน 200 ชิ้น × เด้งกล่องทุกครั้งที่ผิด = งานช้าลงเท่าตัว
                                          และผู้ใช้เริ่มกดปุ่มโดยไม่อ่าน

เมื่อผู้ใช้ถูกถามบ่อยเกินไป เขาจะเรียนรู้ที่จะกดปุ่มขวาโดยอัตโนมัติ — และวันที่กล่องนั้นถามเรื่องสำคัญจริง ๆ เขาก็จะกดผ่านไปเหมือนกัน

เกณฑ์ตัดสิน — ถ้าย้อนกลับได้ ให้ทำเลยแล้วมีปุ่มเลิกทำ ถ้าย้อนไม่ได้ ให้ถาม

9. กล่องกับเครื่องสแกน

ปัญหาเฉพาะของ PDA ที่ไม่มีบนแอปทั่วไป

กล่องยืนยันเปิดอยู่
   → ผู้ใช้ยิงบาร์โค้ดโดยไม่ได้ตั้งใจ
   → เครื่องสแกนส่งข้อมูลตามด้วยปุ่ม Enter
   → Enter ไปกดปุ่มที่กำลังโฟกัสอยู่ในกล่อง

ผลคือกล่องถูกตอบโดยที่ผู้ใช้ไม่ได้ตั้งใจ และถ้าปุ่มที่โฟกัสคือ “ยืนยันลบ” ก็เสียหายทันที

ทางกันรายละเอียด
ให้ปุ่มที่ปลอดภัยได้โฟกัสตั้งต้นEnter ที่หลุดเข้ามาจึงไม่ทำอันตราย
ปิดการรับข้อมูลจากเครื่องสแกนระหว่างเปิดกล่องตรงที่สุด แต่ต้องมีทางเปิดกลับให้ครบทุกเส้นทาง
คืนโฟกัสให้ช่องสแกนหลังปิดกล่องไม่งั้นผู้ใช้จะสแกนแล้วไม่มีอะไรเกิดขึ้น (3.6)

แถวสุดท้ายเป็นอาการที่ผู้ใช้รายงานว่า “สแกนไม่ติด” บ่อยที่สุด ทั้งที่เครื่องสแกนทำงานปกติ — แค่ไม่มีช่องไหนรับข้อมูลอยู่

12. แบบฝึกหัด

ผู้ใช้กดลบโปรไฟล์การเชื่อมต่อชื่อ WMS-PROD ซึ่งเป็นโปรไฟล์ที่กำลังใช้งานอยู่ และมีอีกสองโปรไฟล์ในระบบ

Future<void> _deleteProfile(ConnectionProfile profile) async {
  final nextProfile = store.profileAfterDeleting(profile.id);
  if (nextProfile == null) {
    // ไม่มีโปรไฟล์อื่นเหลือ — ไม่ใช่เรื่องที่ถามยืนยันแล้วจบ
    showResultBanner(
      ScaffoldMessenger.of(context),
      'ต้องมีอย่างน้อยหนึ่งโปรไฟล์ กรุณาเพิ่มโปรไฟล์ใหม่ก่อน',
      isError: true,
    );
    return;
  }

  final confirmed = await showDeleteProfileDialog(
    context,
    name: profile.name,
    isActive: profile.id == store.activeProfileId,
    nextName: nextProfile.name,
  );
  if (!confirmed || !context.mounted) return;    // ครอบทั้ง false และ null

  await store.deleteProfile(profile.id);         // busy/error อยู่ที่ store
}

ข้อ 3 — ควรเป็นกล่องยืนยันธรรมดาไหม

ขึ้นกับว่ามีโปรไฟล์อื่นเหลือหรือไม่ ซึ่งแยกเป็นสามกรณี

สถานการณ์ควรใช้เหตุผล
โปรไฟล์ที่ไม่ได้ใช้อยู่ลบเลย + ปุ่มเลิกทำย้อนได้ง่าย ไม่กระทบอะไร
โปรไฟล์ที่กำลังใช้ และมีอันอื่นเหลือกล่องยืนยันกระทบการเชื่อมต่อทันที ต้องรู้ว่าจะสลับไปไหน
โปรไฟล์สุดท้ายห้ามลบ + บอกเหตุผลลบแล้วแอปใช้งานไม่ได้เลย

โจทย์ระบุว่ามีอีกสองโปรไฟล์ จึงตกอยู่ในแถวกลาง — กล่องยืนยันเหมาะแล้ว

โปรไฟล์ที่ไม่ได้ใช้ ลบแล้วไม่มีอะไรเปลี่ยนทันที การถามทุกครั้งจะทำให้ผู้ใช้ที่กำลังจัดระเบียบโปรไฟล์ห้าอัน ต้องกดยืนยันห้ารอบ — ซึ่งเป็นการฝึกให้เขากดผ่านโดยไม่อ่าน ตามหัวข้อ 8

ทำไมแถวสุดท้ายไม่ควรถาม

เพราะคำตอบมีทางเดียวคือ “ทำไม่ได้” การเปิดกล่องถามว่า “แน่ใจไหม” แล้วพอกดยืนยันกลับบอกว่าทำไม่ได้ คือการหลอกให้ผู้ใช้เสียเวลา

หลักที่ได้จากข้อนี้ — คำถามที่ต้องตอบก่อนตัดสินใจใส่กล่องยืนยัน ไม่ใช่ “อันตรายไหม” แต่คือ

คำถามถ้าตอบทำอะไร
ผู้ใช้ทำได้ไหมไม่ได้บอกเหตุผล อย่าถาม
ย้อนกลับได้ไหมได้ทำเลย + ปุ่มเลิกทำ
ย้อนไม่ได้และมีผลที่ผู้ใช้อาจไม่รู้ใช่ถาม และบอกผลนั้นในกล่อง

ลำดับนี้ทำให้กล่องยืนยันเหลือน้อยลงมาก และกล่องที่เหลืออยู่ก็เป็นกล่องที่ผู้ใช้จะอ่านจริง

13. สรุปท้ายบท

  • กล่องโต้ตอบคือ route จึงมี context ของตัวเองและคืนค่าได้
  • ปิดกล่องด้วย dialogContext เสมอ
  • null ต้องแปลว่า “ไม่ทำ” และควรแปลงเป็นค่าที่ปลอดภัยตั้งแต่ในฟังก์ชัน
  • barrierDismissible: false ไม่กันปุ่มย้อนกลับ
  • ปุ่มต้องเป็นคำกริยาที่บอกตัวเองได้ ไม่ใช่ “ใช่/ไม่”
  • เนื้อหาต้องบอกว่าจะเกิดอะไรขึ้น และระบุตัวสิ่งของให้ชัด
  • ย้อนได้ให้ทำเลยพร้อมปุ่มเลิกทำ ย้อนไม่ได้จึงถาม
  • ถามบ่อยเกินไปทำให้ผู้ใช้เลิกอ่านทุกกล่อง
  • บน PDA ต้องระวัง Enter จากเครื่องสแกน และคืนโฟกัสหลังปิด
  • ให้กล่องตอบอย่างเดียว แล้วให้คนเรียกเป็นคนทำงาน
