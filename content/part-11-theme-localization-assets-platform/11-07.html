3. คลาส Message

/// คู่ข้อความอังกฤษ–ไทยหนึ่งก้อน
///
/// เป็นคลาสแทนที่จะเป็น `Map<String, String>` เพื่อให้ลืมภาษาใดภาษาหนึ่ง
/// ไม่ได้ — คอมไพเลอร์บังคับให้ใส่ครบทั้งสองตัวตั้งแต่ตอนสร้าง
class Message {
  const Message(this.en, this.th);

  final String en;
  final String th;

  String of(AppLanguage language) =>
      language == AppLanguage.thai ? th : en;
}

ถ้าใช้ Map<String, String> แทน จะเขียน {'en': 'Save'} โดยลืม th ได้ และจะไปพบตอนผู้ใช้สลับภาษาแล้วเจอข้อความว่าง

4. ข้อความที่ต้องเติมค่า

นี่คือจุดที่แยกระบบที่คิดมาแล้วออกจากระบบที่ต่อสตริงเอา

// วิธีที่ดูเหมือนใช้ได้
Text('${store.count} ${tr('pallets', 'พาเลท')}')

ได้ภาษาอังกฤษว่า “12 pallets” ถูกต้อง และภาษาไทยว่า “12 พาเลท” ซึ่งผิดหลักภาษา — ภาษาไทยพูดว่า “พาเลท 12 ใบ” โดยเอาคำนามขึ้นก่อนและมีลักษณนามต่อท้าย

class Message1<T> {
  const Message1(this._en, this._th);

  final String Function(T) _en;
  final String Function(T) _th;

  String of(AppLanguage language, T value) =>
      language == AppLanguage.thai ? _th(value) : _en(value);
}

// ภาษาอังกฤษต้องเลือกรูปเอกพจน์/พหูพจน์ ส่วนภาษาไทยไม่ต้อง
String _palletCountEn(int count) => count == 1 ? '1 pallet' : '$count pallets';
String _palletCountTh(int count) => 'พาเลท $count ใบ';

แต่ละภาษาสร้างประโยคของตัวเองทั้งประโยค จึงเรียงคำได้ตามหลักของภาษานั้น

test('ข้อความที่มีค่าแทรก เรียงคำตามภาษาของตัวเอง', () {
  expect(AppText.palletCount.of(AppLanguage.english, 12), '12 pallets');
  expect(AppText.palletCount.of(AppLanguage.thai, 12), 'พาเลท 12 ใบ');

  expect(AppText.heldBy.of(AppLanguage.english, 'สมชาย'), 'Held by สมชาย');
  expect(AppText.heldBy.of(AppLanguage.thai, 'สมชาย'), 'สมชาย ล็อกไว้อยู่');
});

คู่ที่สองแสดงปัญหาชัดกว่า — อังกฤษเอาชื่อไว้ท้าย ไทยเอาไว้หน้า การต่อสตริงแบบ tr('Held by ', '') + name จึงเป็นไปไม่ได้ตั้งแต่ต้น

6. ลำดับที่ถูกต้องตอนสลับ

Future<void> setLanguage(AppLanguage value) async {
  if (_language == value) return;
  _language = value;

  // ประกาศก่อนเขียนลงเครื่อง เพราะหน้าจอต้องเปลี่ยนทันทีที่กด
  // ไม่ใช่รอให้เขียนดิสก์เสร็จ
  notifyListeners();

  try {
    await storage.write(storageKey, value.code);
  } catch (_) {
    // เขียนไม่ได้ก็ยังสลับได้ในรอบนี้
  }
}
บรรทัดป้องกันอะไร
if (_language == value) return;ประกาศทั้งที่ไม่มีอะไรเปลี่ยน ทำให้ทั้งจอวาดใหม่ฟรี ๆ
notifyListeners() ก่อน awaitหน้าจอค้างครึ่งวินาทีบนเครื่องที่ดิสก์ช้า
catch ที่ไม่ทำอะไรสลับภาษาไม่ได้เพราะเขียนดิสก์ไม่ได้
test('เลือกภาษาเดิมซ้ำ ต้องไม่ประกาศ', () async {
  var notified = 0;
  controller.addListener(() => notified++);

  await controller.setLanguage(AppLanguage.english);
  expect(notified, 0);
});

  2. รู้ได้อย่างไรว่าย้ายครบ

1. ย้ายทีละส่วนโดยให้สองแบบอยู่ร่วมกันได้ ตามหลักเดียวกับ 9.15

10. สรุป

  • tr(en, th) เขียนเร็วกว่า แต่ตอบไม่ได้ว่าแอปมีข้อความกี่ก้อนและก้อนไหนยังไม่แปล
  • รายการข้อความคุ้มเมื่อเกินร้อยก้อน หรือมีคนนอกทีมต้องตรวจคำ
  • ข้อความที่มีค่าแทรก ต้องให้แต่ละภาษาสร้างประโยคเอง
  • ประกาศการเปลี่ยนก่อนเขียนลงเครื่อง
  • ปุ่มบนหัวจอแสดงภาษาที่ใช้อยู่ ส่วนผลของการกดไว้ใน tooltip
