2. จับข้อผิดพลาดสองชนิด

Future<void> main() async {
  WidgetsFlutterBinding.ensureInitialized();

  // ข้อผิดพลาดที่เกิดในโค้ดของ Flutter เช่นตอน build หรือ layout
  FlutterError.onError = (details) {
    FlutterError.presentError(details);   // ยังพิมพ์ออก console ตามเดิม
    unawaited(reporter.record(details.exception, details.stack ?? StackTrace.empty));
  };

  // ข้อผิดพลาดที่หลุดออกนอก Flutter เช่นใน Future ที่ไม่มีใครจับ
  PlatformDispatcher.instance.onError = (error, stack) {
    unawaited(reporter.record(error, stack));
    return true;   // จัดการแล้ว อย่าให้แอปปิดตัวเอง
  };

  runApp(const WmsApp());
}
ตัวจับจับอะไรตัวอย่างจากคอร์สนี้
FlutterError.onErrorข้อผิดพลาดใน widgetเนื้อหาล้น, setState หลัง dispose
PlatformDispatcher.instance.onErrorFuture ที่โยนแล้วไม่มีใครจับContractException จาก 6.5
การคืน true ทำให้แอปไม่ปิดตัวเอง ซึ่งดีกับผู้ใช้ในคลัง — แต่ต้องระวังว่าแอปที่ทำงานต่อในสถานะที่ผิด อาจบันทึกข้อมูลผิดได้ ให้คืน true เฉพาะเมื่อมั่นใจว่าจอนั้นยังใช้งานต่อได้

/// รายชื่อนี้ต้องตรงกับที่ 7.17 ใช้กับ log — ถ้าสองที่ไม่ตรงกัน จะมีทาง
/// หนึ่งที่ปิดไว้และอีกทางที่เปิดอยู่
const Set<String> sensitiveKeys = {
  'token', 'accesstoken', 'refreshtoken', 'authorization',

ความลับหลุดเข้ารายงานได้ทางไหน
ข้อความของ exception ที่มี URL พร้อม query string
stack trace ที่มีชื่อพารามิเตอร์และค่า
ข้อมูลประกอบที่ทีมใส่เอง

test('รายงานที่สร้างแล้ว ต้องไม่มีความลับหลงเหลือ', () {
  final report = buildReport(
    error: Exception('401 on /api/tasks?token=abc123'),
    stack: StackTrace.fromString('at login(password: s3cret)'),
    ...
    context: {'token': 'abc123'},
  );

  final json = report.toJson().toString();
  expect(json, isNot(contains('abc123')));
  expect(json, isNot(contains('s3cret')));
  // แต่ข้อมูลที่ใช้หาสาเหตุต้องยังอยู่ครบ
  expect(json, contains('401'));
  expect(json, contains('tasks'));
  expect(json, contains('3.1.0+42'));
});

สามบรรทัดสุดท้ายสำคัญพอกับสองบรรทัดแรก — ตัวปิดค่าที่ปิดจนอ่านอะไรไม่ได้เลย ก็ไร้ประโยชน์เหมือนกัน

5. บั๊กที่เทสต์จับได้ตอนเขียน lab นี้

เทสต์ข้อนี้ล้มในการรันครั้งแรก

test('Bearer token ต้องถูกปิด', () {
  const text = 'headers: {Authorization: Bearer eyJhbGciOi.J9.abc}';
  expect(redact(text), isNot(contains('eyJhbGciOi')));
});
Expected: not contains 'eyJhbGciOi'
  Actual: 'headers: {Authorization: REDACTED eyJhbGciOi.J9.abc}'

สาเหตุคือลำดับ — ลูปที่ปิดค่าตามชื่อคีย์ทำงานก่อน เจอ Authorization: Bearer แล้วถือว่า Bearer คือค่าที่ต้องปิด จึงปิดคำนั้นแล้วปล่อย token จริงไว้ทั้งอัน

String redact(String text) {
  // Bearer ต้องมาก่อนเสมอ — ถ้าปล่อยให้ลูปข้างล่างเจอ `Authorization: Bearer`
  // ก่อน มันจะปิดคำว่า "Bearer" แล้วปล่อย token จริงไว้ ซึ่งเป็นการปิดที่
  // ดูเหมือนทำงานแต่ไม่ได้ปิดอะไรเลย
  var result = text.replaceAll(
    RegExp(r'Bearer\s+[A-Za-z0-9._\-]+', caseSensitive: false),
    'Bearer $_masked',
  );

  for (final key in sensitiveKeys) { ... }
  return result;
}
อาการของบั๊กนี้คือเห็นคำว่า REDACTED ในรายงาน ซึ่งทำให้ทุกคนสบายใจว่าการปิดค่าทำงานอยู่ — เป็นความล้มเหลวที่แย่กว่าการไม่มีตัวปิดค่าเลย เพราะไม่มีใครไปตรวจซ้ำ

บทเรียนที่ใช้ได้ทั่วไป — ทุกตัวปิดค่าต้องมีเทสต์ที่ตรวจว่าค่าจริงหายไป ไม่ใช่แค่ตรวจว่ามีคำว่า REDACTED โผล่ขึ้นมา

8. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ไม่มีรายงานเข้ามาเลยตั้งตัวจับแค่ตัวเดียวตั้งทั้ง FlutterError และ PlatformDispatcher
token โผล่ในรายงานตัวปิดค่าไม่ได้ปิดจริงเขียนเทสต์ยืนยันว่าค่าจริงหาย
stack trace อ่านไม่ออกobfuscate โดยไม่เก็บ symbolsเก็บทุกรุ่น
รู้ว่าพังแต่ไม่รู้ว่าทำอะไรอยู่ไม่ได้เก็บจอปัจจุบันเก็บ screen
แอปวนพังไม่หยุดการส่งรายงานโยนซ้ำครอบ try/catch ที่ไม่ทำอะไร
รายงานเยอะจนไม่มีใครดูรายงานทุกข้อผิดพลาดรวมถึงที่จัดการแล้วรายงานเฉพาะที่ไม่ได้จัดการ

แถวสุดท้ายสำคัญ — ความล้มเหลวที่แอปจัดการแล้ว เช่น ApiFailureKind.network ที่แสดงข้อความให้ผู้ใช้ลองใหม่ (9.13) ไม่ใช่ crash และไม่ควรส่งเข้าระบบเดียวกัน

9. แบบฝึกหัด

ทีมเปิดระบบเก็บรายงานแล้ว สัปดาห์แรกมีรายงานเข้ามา 4,000 รายการ

  1. เป็นสัญญาณอะไร
  2. จะทำอะไรต่อ
แนวคำตอบ

1. ตัวเลขนี้บอกอะไรไม่ได้เลยจนกว่าจะจัดกลุ่ม — 4,000 รายการอาจเป็นบั๊กเดียวที่เกิด 4,000 ครั้ง หรือ 4,000 บั๊กที่ต่างกัน ซึ่งพาไปคนละทาง

ถ้าเป็นแปลว่า
บั๊กเดียว 4,000 ครั้งดีข่าวหนึ่ง — แก้ที่เดียวจบ
รายงานสิ่งที่จัดการแล้วปนมาตั้งค่าผิด ไม่ใช่บั๊ก
บั๊กกระจาย 50 แบบมีปัญหาเชิงโครงสร้าง

2. ทำสามอย่างตามลำดับ

ขั้นทำ
1จัดกลุ่มตามข้อความและบรรทัดแรกของ stack
2เรียงตามจำนวนเครื่องที่เจอ ไม่ใช่จำนวนครั้ง
3ดูสามอันดับแรก

ขั้นที่สองสำคัญกว่าที่คิด — บั๊กที่เกิด 3,000 ครั้งบนเครื่องเดียว อาจเป็นเครื่องที่มีปัญหาเฉพาะตัว ส่วนบั๊กที่เกิด 20 ครั้งบน 20 เครื่อง กระทบทุกคน

-- จัดกลุ่มแล้วนับเครื่องที่ไม่ซ้ำ
SELECT error_signature, COUNT(*) AS total, COUNT(DISTINCT device_id) AS devices
FROM crash_reports
WHERE received_at > NOW() - INTERVAL '7 days'
GROUP BY error_signature
ORDER BY devices DESC;

สิ่งที่ต้องระวัง — ก่อนไล่แก้ทั้งหมด ให้ตรวจว่ารายงานที่เข้ามาเป็น crash จริงไหม

ถ้าเจอทำ
ApiException(network) จำนวนมากไม่ใช่ crash — เป็นเน็ตหลุดที่แอปจัดการแล้ว ให้ตัดออกจากระบบนี้
ContractExceptionเป็น crash จริง และเป็นสัญญาณว่าสัญญากับเซิร์ฟเวอร์เปลี่ยน (14.5)
A ChangeNotifier was used after being disposedการรั่ว (12.8)
RenderFlex overflowedเนื้อหาล้นบนจอบางขนาด (14.13)

แถวที่หนึ่งกับแถวที่สองต่างกันตรงที่ใครแก้ได้ — ซึ่งเป็นเส้นแบ่งเดียวกับที่ 9.13 ใช้ตัดสินว่าความล้มเหลวไหนควรแปลให้ผู้ใช้และไหนควรปล่อยให้ดัง

ถ้าตัดสองแถวแรกออกแล้วเหลือรายงานไม่กี่สิบรายการ นั่นคือรายการงานที่แท้จริง — และมักจะสั้นกว่าที่ตัวเลข 4,000 ทำให้กลัวมาก

10. สรุป

  • ตั้งตัวจับสองตัว — FlutterError.onError และ PlatformDispatcher.onError
  • เก็บจอที่ผู้ใช้อยู่ตอนเกิด เพราะมันบอกวิธีทำซ้ำ
  • ตัวปิดค่าต้องมีเทสต์ที่ตรวจว่าค่าจริงหายไป ไม่ใช่แค่มี REDACTED
  • ตัวปิดค่าที่ปิดไม่จริง แย่กว่าไม่มีเลย
  • อย่าให้การส่งรายงานทำให้แอปพังซ้ำ
  • ความล้มเหลวที่แอปจัดการแล้ว ไม่ใช่ crash
  • จัดกลุ่มแล้วเรียงตามจำนวนเครื่อง ไม่ใช่จำนวนครั้ง

บทหน้า: ทำให้แอปทนกับเครือข่ายที่ไม่ดีของคลัง
