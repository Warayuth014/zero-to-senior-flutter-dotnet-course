ต้องผ่านอะไรมาก่อน: 15.1, 13.1 (สิทธิ์กล้อง)
จบบทนี้แล้วจะทำได้: ประกาศสิทธิ์ให้ครบสำหรับ build ที่ปล่อยจริง และรู้จักบั๊กที่มองไม่เห็นตอนพัฒนา

1. Flutter ใส่สิทธิ์ให้ — แต่เฉพาะบางโหมด

โปรเจกต์ Flutter มี manifest สามไฟล์

android/app/src/
  main/AndroidManifest.xml       ← ใช้กับทุกโหมด รวมถึง release
  debug/AndroidManifest.xml      ← เพิ่มเฉพาะตอน debug
  profile/AndroidManifest.xml    ← เพิ่มเฉพาะตอน profile
<!-- debug/AndroidManifest.xml ที่ Flutter สร้างให้ -->
<manifest xmlns:android="http://schemas.android.com/apk/res/android">
    <!-- The INTERNET permission is required for development. Specifically,
         the Flutter tool needs it to communicate with the running application
         to allow setting breakpoints, to provide hot reload, etc.
    -->
    <uses-permission android:name="android.permission.INTERNET"/>
</manifest>
อ่านคอมเมนต์ให้ดี — สิทธิ์นี้ถูกใส่ไว้เพื่อให้เครื่องมือของ Flutter ทำงาน ไม่ใช่เพื่อให้แอปของเราใช้ ดังนั้นมันจึงมีเฉพาะสองโหมดนั้น

2. บั๊กที่ตามมา

คำสั่งมีสิทธิ์ INTERNET ไหมแอปต่อเน็ตได้ไหม
flutter runมี — จาก debug manifestได้
flutter run --profileมี — จาก profile manifestได้
flutter build apk --releaseเฉพาะที่ประกาศใน mainขึ้นกับว่าประกาศไว้ไหม

ถ้า main/AndroidManifest.xml ไม่มีบรรทัดนั้น แอปที่ปล่อยจริงจะต่อเน็ตไม่ได้เลย และอาการที่ผู้ใช้เห็นคือทุกจอขึ้นว่า “โหลดไม่สำเร็จ” — ซึ่งแยกไม่ออกจากปัญหา Wi-Fi ของคลัง

SocketException: Failed host lookup / Permission denied

ทีมที่ทดสอบด้วย flutter run มาตลอด จะไม่มีทางเจอปัญหานี้จนกว่าจะ build release แล้วเอาไปลงเครื่องจริง

3. เทียบสองโปรเจกต์จริง

<!-- wms_absolute_mobile/android/app/src/main/AndroidManifest.xml -->
<manifest xmlns:android="http://schemas.android.com/apk/res/android">
    <!-- Talks to the WMS backend over the local network. -->
    <uses-permission android:name="android.permission.INTERNET"/>
    <application
        android:label="ABS PDA V3"
        android:usesCleartextTraffic="true">
<!-- wmsapp/android/app/src/main/AndroidManifest.xml -->
<manifest xmlns:android="http://schemas.android.com/apk/res/android">
    <application
        android:label="TKS"
        android:enableOnBackInvokedCallback="false">
wms_absolute_mobilewmsapp
สิทธิ์ INTERNET ใน mainมีไม่มี
อนุญาต httpมีไม่มี
ผลกับ wmsapp: APK ที่ build เป็น release จะขาดสิทธิ์ INTERNET และถูกบล็อกไม่ให้เรียก http:// ด้วย — สองอย่างนี้แยกกัน และแต่ละอย่างพอที่จะทำให้ทุกคำขอล้มเหลว ทั้งคู่มองไม่เห็นเลยตอนรันด้วย flutter run

สิ่งที่ต้องเพิ่มมีสองบรรทัด

<manifest xmlns:android="http://schemas.android.com/apk/res/android">
    <uses-permission android:name="android.permission.INTERNET"/>
    <application
        android:label="TKS"
        android:networkSecurityConfig="@xml/network_security_config">

4. เปลี่ยน checklist ให้เป็นเทสต์

ปัญหาของ checklist ที่เป็นเอกสารคือไม่มีใครอ่านตอนรีบ ทางแก้คือให้เครื่องตรวจแทน

List<ReleaseFinding> checkAndroidManifest(String manifest) {
  final findings = <ReleaseFinding>[];

  // Flutter ใส่สิทธิ์นี้ให้เฉพาะใน debug และ profile manifest — ถ้า main
  // ไม่มี แอปที่ build เป็น release จะต่อเน็ตไม่ได้เลย และอาการที่เห็นคือ
  // "โหลดไม่สำเร็จ" ทุกจอ โดยที่ทดสอบตอนพัฒนาแล้วปกติทุกอย่าง
  if (!manifest.contains('android.permission.INTERNET')) {
    findings.add(const ReleaseFinding(
      Severity.blocker,
      'missing-internet-permission',
      'main manifest ไม่มีสิทธิ์ INTERNET',
      'เพิ่ม <uses-permission android:name="android.permission.INTERNET"/> '
          'ใน android/app/src/main/AndroidManifest.xml',
    ));
  }
  ...
}
test('ไม่มีสิทธิ์ INTERNET ต้องเป็นตัวขวางการปล่อย', () {
  final manifest = goodManifest.replaceAll(
    '<uses-permission android:name="android.permission.INTERNET"/>',
    '',
  );

  final findings = checkAndroidManifest(manifest);
  final blocker = findings.firstWhere((f) => f.isBlocker);
  expect(blocker.code, 'missing-internet-permission');
  expect(blocker.fix, contains('uses-permission'));
});

สังเกตว่าทุก finding มีช่อง fix — เพราะการบอกว่า “ผิด” โดยไม่บอกว่าแก้อย่างไร ทำให้คนที่เจอต้องไปหาต่อ ซึ่งเป็นหลักเดียวกับข้อความผิดพลาดใน 10.7

5. สิทธิ์อื่นที่แอปคลังต้องใช้

สิทธิ์ใช้ตอนต้องขอตอนรันไหม
INTERNETทุกคำขอไม่ — ให้อัตโนมัติเมื่อประกาศ
CAMERAถ่ายรูปหลักฐาน (13.1)ต้อง
ACCESS_NETWORK_STATEตรวจสถานะเครือข่าย (10.12)ไม่
POST_NOTIFICATIONSแจ้งเตือน Android 13 ขึ้นไปต้อง
VIBRATEสั่นเมื่อสแกนสำเร็จไม่
ขอเท่าที่ใช้ — สิทธิ์ที่ประกาศไว้แต่ไม่ได้ใช้ ทำให้ผู้ใช้และฝ่ายไอทีขององค์กรตั้งคำถาม และบางองค์กรมีนโยบายตรวจสิทธิ์ของแอปก่อนอนุญาตให้ติดตั้ง

สิทธิ์ที่ไม่ควรขอในแอปคลัง

สิทธิ์ทำไมไม่ควร
READ_EXTERNAL_STORAGEAndroid รุ่นใหม่ใช้ตัวเลือกไฟล์ของระบบแทน (13.1)
ACCESS_FINE_LOCATIONถ้าไม่ได้ใช้ตำแหน่งจริง ๆ จะดูเหมือนติดตามพนักงาน
READ_PHONE_STATEมักถูกขอเพื่อหารหัสเครื่อง ซึ่งมีวิธีอื่นที่ไม่ต้องขอสิทธิ์

6. ตรวจว่า APK ที่ build ออกมามีอะไรบ้าง

# ดูสิทธิ์ทั้งหมดใน APK ที่ build แล้ว
aapt dump permissions build/app/outputs/flutter-apk/app-release.apk

# หรือดู manifest ที่ถูกรวมแล้ว
build/app/intermediates/merged_manifests/release/AndroidManifest.xml

ไฟล์ที่สองสำคัญ — มันคือ manifest ที่รวมของเราเข้ากับของปลั๊กอินทุกตัวแล้ว ปลั๊กอินบางตัวเพิ่มสิทธิ์เข้ามาเองโดยไม่ได้บอก

ปลั๊กอินอาจเพิ่มสิทธิ์
image_pickerการเข้าถึงกล้องและสื่อ
ไลบรารีวิเคราะห์การใช้งานสถานะเครือข่าย รหัสโฆษณา

ควรดูไฟล์นี้อย่างน้อยหนึ่งครั้งก่อนปล่อยรุ่นแรก และดูซ้ำทุกครั้งที่เพิ่มปลั๊กอินใหม่

<!-- ลบสิทธิ์ที่ปลั๊กอินเพิ่มมาแต่เราไม่ได้ใช้ -->
<uses-permission
    android:name="android.permission.ACCESS_FINE_LOCATION"
    tools:node="remove" />

7. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ทุกจอขึ้น “โหลดไม่สำเร็จ” เฉพาะ APK ที่ปล่อยไม่มี INTERNET ใน main manifestเพิ่ม
ทดสอบแล้วปกติทุกอย่างทดสอบด้วย flutter run เท่านั้นทดสอบ APK ที่ build จริง (15.1)
ฝ่ายไอทีไม่อนุญาตให้ติดตั้งขอสิทธิ์เกินจำเป็นตรวจ manifest ที่รวมแล้ว
สิทธิ์โผล่มาโดยไม่ได้ประกาศปลั๊กอินเพิ่มให้ลบด้วย tools:node="remove"
กล้องใช้ไม่ได้บนเครื่องบางรุ่นประกาศสิทธิ์แล้วแต่ไม่ได้ขอตอนรันขอตอนใช้ครั้งแรก (13.1)

8. แบบฝึกหัด

ทีมกำลังจะปล่อย wmsapp รุ่นแรกให้คลังใช้จริง โดยที่ผ่านมาทดสอบด้วย flutter run มาตลอด

  1. ทำอะไรก่อนเป็นอันดับแรก
  2. จะป้องกันไม่ให้เกิดซ้ำอย่างไร
แนวคำตอบ

1. build APK แล้วลงเครื่องจริง ก่อนทำอย่างอื่นทั้งหมด

flutter build apk --release
adb install -r build/app/outputs/flutter-apk/app-release.apk

แล้วทดสอบเส้นทางหลักหนึ่งรอบ — เข้าสู่ระบบ โหลดรายการ สแกน ยืนยัน

ในกรณีของ wmsapp จะติดตั้งแต่ขั้นเข้าสู่ระบบ เพราะขาดทั้งสิทธิ์ INTERNET และการอนุญาต http สองบรรทัดที่ต้องเพิ่มคือ

<uses-permission android:name="android.permission.INTERNET"/>
<application android:networkSecurityConfig="@xml/network_security_config" ...>

พร้อมไฟล์ config ที่จำกัดเฉพาะเซิร์ฟเวอร์ของคลัง (15.4)

2. ป้องกันสามชั้น

ชั้นทำอะไรจับได้เมื่อ
เทสต์รัน checkRelease ในชุดเทสต์ปกติทุกครั้งที่รันเทสต์
สคริปต์ buildให้ tool/release.sh ตรวจก่อน buildก่อนสร้าง APK
กระบวนการทุกงานที่ปิด ต้องทดสอบบน APK ที่ build จริงก่อนบอกว่าเสร็จ
// test/release_check_test.dart — รันกับไฟล์จริงของโปรเจกต์
test('โปรเจกต์นี้พร้อมปล่อย', () {
  final report = checkRelease(
    androidManifest: File('android/app/src/main/AndroidManifest.xml')
        .readAsStringSync(),
    androidGradle: File('android/app/build.gradle.kts').readAsStringSync(),
    pubspec: File('pubspec.yaml').readAsStringSync(),
  );

  expect(report.blockers, isEmpty, reason: report.toString());
});

เทสต์นี้จะล้มพร้อมข้อความที่บอกวิธีแก้ทุกข้อ ซึ่งดีกว่าการค้นพบตอนที่ APK อยู่ในมือผู้ใช้แล้ว

ข้อสังเกตที่สำคัญที่สุด: บั๊กนี้ไม่ได้เกิดจากความไม่รู้ แต่เกิดจากความแตกต่างระหว่างสิ่งที่ทดสอบกับสิ่งที่ปล่อย — และนั่นคือช่องว่างที่ Part นี้ทั้ง Part พยายามปิด

9. สรุป

  • Flutter ใส่สิทธิ์ INTERNET ให้เฉพาะ debug และ profile
  • build ที่ปล่อยจริงใช้เฉพาะสิ่งที่ประกาศใน main manifest
  • อาการคือทุกจอโหลดไม่สำเร็จ ซึ่งแยกไม่ออกจากปัญหา Wi-Fi
  • ขอสิทธิ์เท่าที่ใช้ และตรวจ manifest ที่รวมกับปลั๊กอินแล้ว
  • เปลี่ยน checklist ให้เป็นเทสต์ที่รันได้
  • ทดสอบบน APK ที่ build จริงก่อนบอกว่าเสร็จ

บทหน้า: เซิร์ฟเวอร์ที่เป็น http และการอนุญาตที่แคบที่สุดเท่าที่ทำได้
