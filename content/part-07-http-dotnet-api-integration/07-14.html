ระดับ: กลาง–สูงใช้เวลา: 30–40 นาทีลงมือทำ: ลองใหม่เองโดยไม่ทำให้สถานการณ์แย่ลง
ต้องผ่านอะไรมาก่อน: 7.9 (ชนิดความล้มเหลว) และ 4.11 (การยืดระยะและ jitter)
จบบทนี้แล้วจะทำได้: ตัดสินได้ว่าคำขอไหนลองใหม่เองได้ และตั้งจังหวะไม่ให้ซ้ำเติมเซิร์ฟเวอร์

1. ปัญหาที่บทนี้แก้

เครือข่ายในคลังไม่เสถียร — คำขอที่ล้มเหลวเพราะสัญญาณแวบเดียว มักสำเร็จถ้าลองใหม่ในอีกวินาทีเดียว

แต่การลองใหม่แบบไม่คิด สร้างปัญหาที่ใหญ่กว่าเดิม

ลองใหม่แบบผิดผลที่ตามมา
ลองใหม่กับทุกความล้มเหลวยิงคำขอที่รู้อยู่แล้วว่าจะถูกปฏิเสธ
ลองใหม่กับคำสั่งที่เปลี่ยนข้อมูลงานถูกบันทึกหลายครั้ง
ลองใหม่ทันทีไม่มีการรอยิงสามครั้งในเสี้ยววินาที ได้ผลเดิมทั้งสามครั้ง
ทุกเครื่องลองใหม่พร้อมกันเซิร์ฟเวอร์ที่เพิ่งฟื้นล้มซ้ำทันที

แถวสุดท้ายเป็นเรื่องที่มองไม่เห็นจากเครื่องเดียว — PDA 40 เครื่องในคลังหลุดพร้อมกัน แล้วกลับมายิงพร้อมกัน

2. คำศัพท์ที่ต้องรู้ก่อน

คำความหมาย
retryยิงคำขอเดิมซ้ำ
backoffยืดระยะรอออกไปทุกครั้งที่ล้มเหลว
jitterสุ่มเวลาเล็กน้อย ไม่ให้ทุกเครื่องยิงพร้อมกัน
idempotentทำซ้ำแล้วผลเท่ากับทำครั้งเดียว
attemptครั้งที่ยิง รวมครั้งแรก

3. Mental model: สองคำถามก่อนลองใหม่

1. ลองใหม่แล้วมีโอกาสได้ผลต่างไหม
     ไม่มี → อย่าลอง เสียเวลาเปล่า

2. ยิงซ้ำแล้วปลอดภัยไหม
     ไม่ปลอดภัย → อย่าลองเอง ให้ผู้ใช้ตัดสินใจ
ประโยคเดียวที่ต้องจำ: ต้องตอบว่า “ได้” ทั้งสองข้อจึงจะลองใหม่เองได้ — ข้อเดียวไม่พอ

4. คำถามที่หนึ่ง: มีโอกาสได้ผลต่างไหม

เปิด lib/core/api_failure.dart

/// ลองใหม่ด้วยคำขอเดิมแล้วมีโอกาสได้ผลต่างไหม
///
/// เขียนเป็นรายชื่อที่อนุญาต ไม่ใช่รายชื่อที่ห้าม — ชนิดใหม่ที่เพิ่มมา
/// จะได้ไม่กลายเป็น "ลองใหม่ได้" โดยอัตโนมัติ
bool get isRetryable => switch (this) {
  ApiFailureKind.network || ApiFailureKind.timeout || ApiFailureKind.server =>
    true,
  ApiFailureKind.unauthorized ||
  ApiFailureKind.forbidden ||
  ApiFailureKind.notFound ||
  ApiFailureKind.conflict ||
  ApiFailureKind.validation ||
  ApiFailureKind.contract => false,
};
ลองใหม่ได้เพราะ
networkสัญญาณอาจกลับมาแล้ว
timeoutเซิร์ฟเวอร์อาจว่างลงแล้ว
serverปัญหาภายในอาจเป็นชั่วคราว
ลองใหม่ไม่ได้เพราะ
unauthorizedtoken เดิมยังใช้ไม่ได้อยู่ดี
forbiddenสิทธิ์ไม่เปลี่ยนในสามวินาที
validationข้อมูลเดิมยังผิดอยู่
conflictคนอื่นทำไปแล้ว ยิงอีกกี่ครั้งก็ 409
contractสัญญายังไม่ตรงอยู่ดี

การเขียนเป็นรายชื่อที่อนุญาตสำคัญกว่าที่เห็น — ถ้าเขียนแบบ kind != unauthorized && kind != forbidden วันที่เพิ่มชนิดใหม่ มันจะกลายเป็น “ลองใหม่ได้” เงียบ ๆ

5. คำถามที่สอง: ยิงซ้ำปลอดภัยไหม

เปิด lib/core/api_retry.dart

/// [isIdempotent] ต้องเป็น true เท่านั้นจึงจะลองใหม่ — คำสั่งที่ยังไม่รู้ผล
/// ห้ามยิงซ้ำโดยอัตโนมัติ ถ้าไม่มีรหัสคำสั่งกำกับ (4.14)
Future<T> run<T>(
  Future<T> Function() action, {
  required bool isIdempotent,
}) async {
  var attempt = 1;
  while (true) {
    try {
      return await action();
    } on ApiException catch (error) {
      final canRetry =
          isIdempotent && error.kind.isRetryable && attempt < maxAttempts;
      if (!canRetry) rethrow;
      await _sleep(delayForAttempt(attempt));
      attempt++;
    }
  }
}

สังเกตว่า isIdempotent เป็น required — ไม่มีค่าตั้งต้น

ถ้ามีค่าตั้งต้นเป็น trueถ้ามีค่าตั้งต้นเป็น false
ลืมระบุแล้วคำสั่งถูกยิงซ้ำ — อันตรายลืมระบุแล้วไม่ลองใหม่ — ปลอดภัยแต่ไม่มีใครรู้ว่าลืม

การบังคับให้ระบุทำให้ ทุกจุดที่ใช้ต้องคิดเรื่องนี้ ซึ่งเป็นสิ่งที่ต้องการ

test('คำสั่งที่ไม่ปลอดภัยต่อการยิงซ้ำ ห้ามลองใหม่เอง', () async {
  ...
  await expectLater(
    policy.run(
      () async {
        attempts++;
        throw const ApiException('server ไม่ตอบหลังส่งคำสั่ง',
            kind: ApiFailureKind.timeout, outcomeUnknown: true);
      },
      isIdempotent: false,
    ),
    throwsA(isA<ApiException>()),
  );

  expect(attempts, 1);
});

สังเกตว่าชนิดคือ timeout ซึ่ง isRetryable เป็น true — แต่ยังยิงครั้งเดียว เพราะคำถามที่สองตอบว่าไม่ปลอดภัย

6. ยืดระยะและสุ่ม

/// ยืดเป็นทวีคูณ มีเพดาน และสุ่มเล็กน้อย
///
/// jitter จำเป็นเพราะ PDA ทั้งคลังมักหลุดพร้อมกัน ถ้าไม่สุ่มก็จะ
/// กลับมายิงพร้อมกันจนเซิร์ฟเวอร์ล้มซ้ำ
Duration delayForAttempt(int attempt) {
  final growth = baseDelay.inMilliseconds * (1 << (attempt - 1));
  final capped = min(growth, maxDelay.inMilliseconds);
  final jitter = _random.nextInt(capped ~/ 4 + 1);
  return Duration(milliseconds: capped + jitter);
}
ส่วนประกอบทำอะไรถ้าไม่มี
1 << (attempt - 1)คูณสองทุกครั้งยิงถี่เท่าเดิม ซ้ำเติมเซิร์ฟเวอร์
min(..., maxDelay)ไม่ให้เกินเพดานครั้งที่สิบต้องรอเป็นนาที
nextIntสุ่มเพิ่มเล็กน้อยทุกเครื่องยิงพร้อมกัน

7. ต้องมีจุดจบเสมอ

/// รวมครั้งแรกด้วย — 3 แปลว่ายิงจริงได้มากที่สุด 3 ครั้ง
final int maxAttempts;

คอมเมนต์บรรทัดนี้จำเป็น เพราะคำว่า “ลองใหม่ 3 ครั้ง” ตีความได้สองแบบ

ตีความว่ายิงจริงกี่ครั้ง
ครั้งแรก + ลองใหม่ 34
รวมทั้งหมด 33

ความต่างนี้สำคัญเมื่อคำนวณเวลารอรวม — ถ้าตั้งเวลารอ 30 วินาทีต่อครั้ง การยิง 4 ครั้งแปลว่าผู้ใช้อาจรอถึงสองนาที

เวลารอรวมที่แย่ที่สุด = (timeout × maxAttempts) + ผลรวมของเวลาที่ยืด
                     = (30 × 3) + (0.4 + 0.8)
                     ≈ 91 วินาที

ตัวเลขนี้ต้องคำนวณและยอมรับได้ — ไม่ใช่ปล่อยให้เกิดขึ้นแล้วค่อยรู้

อย่าลองใหม่ไม่จำกัดครั้ง แม้จะยืดระยะแล้ว — เพราะถ้าเซิร์ฟเวอร์ล่มยาว แอปจะยิงต่อไปเรื่อย ๆ กินแบตของ PDA ทั้งคลัง

8. บอกผู้ใช้ว่ากำลังลองใหม่

การลองใหม่เงียบ ๆ ทำให้ผู้ใช้เห็นแค่ว่า “ค้างนาน”

ควรบอกเมื่อ
ไม่ต้องบอกลองใหม่ครั้งแรกที่เร็วมาก
“กำลังลองใหม่...”เริ่มลองครั้งที่สองขึ้นไป
“ลองแล้ว 3 ครั้ง ยังไม่สำเร็จ”หมดโควตาแล้ว
final policy = RetryPolicy(
  sleep: (duration) async {
    onRetrying?.call();          // แจ้งชั้นบนว่ากำลังจะลองใหม่
    await Future<void>.delayed(duration);
  },
);

และเมื่อหมดโควตา ข้อความควรบอกด้วยว่าลองไปแล้วกี่ครั้ง — เพราะ “ลองแล้ว 3 ครั้ง” บอกผู้ใช้ว่าไม่ต้องกดลองใหม่เองอีก ปัญหาอยู่ที่อื่น

9. การลองใหม่ที่ดีที่สุดคือไม่ต้องลอง

แทนที่จะลองใหม่ทำอะไรได้
เก็บคำสั่งไว้ส่งทีหลังผู้ใช้ทำงานต่อได้เลย ไม่ต้องรอ (Part 12)
ให้ผู้ใช้กดลองใหม่เองเขารู้ดีกว่าว่าตอนนี้สัญญาณดีหรือยัง
โหลดข้อมูลใหม่แทนการส่งซ้ำสำหรับกรณีที่ไม่รู้ผล (4.13)

แถวกลางสำคัญบน PDA — คนหน้างานรู้ว่าตัวเองอยู่มุมอับสัญญาณ และเดินออกมาสองก้าวแล้วกดใหม่ได้ผลกว่าให้แอปลองเองสามครั้งอยู่ที่เดิม

ดังนั้นการลองใหม่อัตโนมัติควรจำกัดอยู่แค่ สองสามครั้งแรกที่เร็วมาก แล้วปล่อยให้ผู้ใช้ตัดสินใจ ไม่ใช่พยายามลองให้ครบทุกทางเอง

10. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ข้อมูลถูกบันทึกหลายครั้งลองใหม่กับคำสั่งที่เปลี่ยนข้อมูลต้องมีรหัสคำสั่งก่อน (7.15)
ยิงคำขอที่รู้ว่าจะถูกปฏิเสธลองใหม่กับทุกชนิดตรวจ isRetryable
เซิร์ฟเวอร์ล้มซ้ำทันทีที่ฟื้นทุกเครื่องยิงพร้อมกันใส่ jitter
รอนานมากในครั้งที่หลัง ๆไม่มีเพดานใส่ maxDelay
แบตหมดเร็วตอนเน็ตมีปัญหาลองใหม่ไม่จำกัดครั้งจำกัดจำนวนครั้ง
ผู้ใช้เห็นแค่ว่าค้างนานลองใหม่เงียบ ๆแจ้งเมื่อลองครั้งที่สองขึ้นไป
ชนิดใหม่กลายเป็นลองใหม่ได้เองเขียนเป็นรายชื่อที่ห้ามเขียนเป็นรายชื่อที่อนุญาต

11. แบบฝึกหัด

ทีมจะเพิ่มการลองใหม่อัตโนมัติให้สี่คำขอ

2ปิดงาน พร้อมรหัสคำสั่งPOST

ข้อ 1 — อันไหนลองใหม่ได้

#ลองใหม่เองได้ไหมเหตุผล
1ได้GET ไม่เปลี่ยนอะไร ยิงกี่ครั้งก็ปลอดภัย
2ได้มีรหัสคำสั่ง เซิร์ฟเวอร์กันซ้ำให้
3ไม่ได้ไม่มีอะไรกันการบันทึกซ้ำ
4ได้ แต่ต้องระวังดูข้อ 3

ข้อ 2 กับ 3 เป็นคำขอเดียวกันทุกอย่าง ต่างกันแค่มีรหัสคำสั่งหรือไม่ — ซึ่งเปลี่ยนคำตอบจากได้เป็นไม่ได้

นี่แสดงว่าความปลอดภัยของการลองใหม่ ไม่ได้ขึ้นกับ method อย่างเดียว

ข้อ 2 — โค้ด

@override
Future<List<WmsTask>> fetchOpen() => _retry.run(
  () async {
    final json = await api.getJson('/api/WMS/mobile_tasks');
    return _parseTasks(json);
  },

// 2. ปิดงาน — ปลอดภัยเพราะรหัสคำสั่งเดียวกันถูกส่งทุกครั้ง
@override
Future<TaskCompletion> complete(String id, {required String commandId}) =>
    _retry.run(
      () async {
        final json = await api.postJson(
          '/api/WMS/mobile_tasks/$id/complete',
          const {},
          headers: {'Idempotency-Key': commandId},

        return _parseCompletion(json, id);
      },
      // ปลอดภัยเพราะ commandId ถูกสร้างนอก closure นี้

      isIdempotent: true,

บั๊กนี้มองไม่เห็นจากการอ่านเผิน ๆ และจะเจอเฉพาะตอนที่เน็ตไม่ดีจนเกิดการลองใหม่จริง

ข้อ 3 — จุดที่ต้องระวังของอันที่ 4

การค้นหาผ่าน POST มีสองเรื่องซ้อนกัน

เรื่องที่หนึ่ง — ธง outcomeUnknown ตั้งผิด

ตามที่ 7.12 หัวข้อ 7 บอกไว้ postJson ตั้งว่าไม่รู้ผลเสมอ ทั้งที่การค้นหาไม่ได้เปลี่ยนอะไร

final json = await api.postJson(
  '/api/WMS/mobile_tasks/query',
  { ... },
  mutates: false,          // บอกว่าไม่ได้เปลี่ยนข้อมูล
);

เรื่องที่สอง — การลองใหม่ทับกับตัวนับรุ่น

นี่คือจุดที่ระวังยากกว่า

0.0s  ผู้ใช้พิมพ์ "PAL" → search() เริ่ม generation 5
0.2s  ล้มเหลว → เริ่มรอ 400ms เพื่อลองใหม่
0.3s  ผู้ใช้พิมพ์ "PALL" → search() เริ่ม generation 6
0.6s  การลองใหม่ของ generation 5 ยิงออกไป      ← ยังยิงอยู่ทั้งที่ล้าสมัยแล้ว
1.1s  ผลของ generation 5 กลับมา → ถูกทิ้งเพราะรุ่นไม่ตรง ✓
ผลที่ได้
ความถูกต้องปลอดภัย — ตัวนับรุ่นทิ้งผลให้
ความสิ้นเปลืองเสีย — ยิงคำขอที่ไม่มีใครต้องการแล้ว

ทางแก้คือให้การลองใหม่รู้ว่าควรหยุด

  isIdempotent: true,
  // หยุดลองใหม่ถ้าไม่มีใครรอผลนี้แล้ว
  shouldContinue: () => generation == _generation,
);
final canRetry = isIdempotent &&
    error.kind.isRetryable &&
    attempt < maxAttempts &&
    (shouldContinue?.call() ?? true);

บทเรียนที่ควรจำ: การลองใหม่กับตัวนับรุ่นแก้คนละปัญหา — ตัวนับรุ่นกันผลเก่าเขียนทับ ส่วนการหยุดลองใหม่กันการยิงเปล่า และเมื่อใช้ทั้งสองอย่างพร้อมกัน ต้องให้ตัวหนึ่งรู้จักอีกตัว ไม่งั้นจะปลอดภัยแต่สิ้นเปลือง

12. สรุปท้ายบท

  • ต้องตอบได้ทั้งสองข้อ — มีโอกาสได้ผลต่างไหม และยิงซ้ำปลอดภัยไหม
  • เขียนรายชื่อที่ลองใหม่ได้แบบอนุญาต ไม่ใช่แบบห้าม
  • บังคับให้ระบุว่าปลอดภัยต่อการยิงซ้ำหรือไม่ ไม่มีค่าตั้งต้น
  • สร้างรหัสคำสั่งนอกส่วนที่ถูกลองใหม่ ไม่ใช่ข้างใน
  • ยืดระยะเป็นทวีคูณ มีเพดาน และใส่ jitter
  • จำกัดจำนวนครั้ง และคำนวณเวลารอรวมที่แย่ที่สุดไว้
  • บอกผู้ใช้เมื่อลองครั้งที่สองขึ้นไป
  • การลองใหม่ที่ดีที่สุดคือให้ผู้ใช้ตัดสินใจ หรือเก็บไว้ส่งทีหลัง
