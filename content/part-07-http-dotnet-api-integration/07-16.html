ระดับ: กลางใช้เวลา: 25–35 นาทีลงมือทำ: ตรวจว่าเซิร์ฟเวอร์ยังคุยได้ไหม
ต้องผ่านอะไรมาก่อน: 7.3 (ที่อยู่ตามอุปกรณ์) และ 7.10 (โยนหรือคืนค่า)
จบบทนี้แล้วจะทำได้: ทำปุ่มทดสอบการเชื่อมต่อที่บอกสาเหตุได้จริง และรู้ว่าไม่ควรใช้มันทำอะไร

1. ปัญหาที่บทนี้แก้

7.3 จบด้วยรายการตรวจก่อนติดตั้ง ซึ่งทั้งหมดต้องทำด้วยมือ — บทนี้ทำให้แอปตรวจเองได้

ปัญหาที่แก้คือ ช่องว่างระหว่างตอนติดตั้งกับตอนใช้งานจริง

ทีมติดตั้งตั้งค่าที่อยู่เซิร์ฟเวอร์ → กดบันทึก → กลับบ้าน
วันรุ่งขึ้น คนหน้างานเปิดแอป → กดปุ่มโหลดงาน → "เชื่อมต่อไม่ได้"
  → ไม่รู้ว่าที่อยู่ผิด เน็ตมีปัญหา หรือแอปพัง
  → ต้องรอทีมมาดู ซึ่งอาจเป็นวันถัดไป

ปุ่มทดสอบการเชื่อมต่อย้ายการค้นพบปัญหามาไว้ตอนติดตั้ง ซึ่งเป็นตอนที่คนที่แก้ได้ยังอยู่ตรงนั้น

โหลดรายการงาน  →  ล้มเหลว = ผิดปกติ = โยน

ประโยคเดียวที่ต้องจำ: ปุ่มทดสอบมีไว้เพื่อ รู้ว่าต่อไม่ได้ — การโยนความล้มเหลวออกไปจึงผิดความหมาย ตามที่ 7.10 บอกไว้

4. ผลลัพธ์ที่บอกสาเหตุได้

sealed class HealthResult {
  const HealthResult();
}

class HealthOk extends HealthResult {
  const HealthOk(this.elapsed);
  final Duration elapsed;      // ช้าหรือเร็ว ก็เป็นข้อมูล
}

class HealthUnreachable extends HealthResult {
  const HealthUnreachable(this.message);
  final String message;
}

class HealthUnauthorized extends HealthResult {
  const HealthUnauthorized();
}

class HealthWrongServer extends HealthResult {
  const HealthWrongServer(this.detail);
  final String detail;
}

สี่กรณีนี้ทำให้คนติดตั้งรู้ว่าต้องไปแก้ที่ไหน

ผลลัพธ์บอกอะไรไปแก้ที่
HealthOkทุกอย่างพร้อม—
HealthUnreachableต่อไม่ถึงเลยที่อยู่ เครือข่าย firewall
HealthUnauthorizedต่อถึงแล้ว แต่ยังไม่ได้เข้าสู่ระบบไม่ใช่ปัญหา — ที่อยู่ถูกแล้ว
HealthWrongServerต่อถึงแต่ไม่ใช่เซิร์ฟเวอร์ที่ต้องการที่อยู่ชี้ผิดระบบ

แถวที่สามน่าสนใจ — การได้ 401 จากปุ่มทดสอบเป็นข่าวดี เพราะแปลว่าคุยกับเซิร์ฟเวอร์ที่ถูกต้องได้แล้ว เหลือแค่เข้าสู่ระบบ

5. เขียนตัวตรวจ

Future<HealthResult> checkHealth(Uri baseUrl) async {
  final stopwatch = Stopwatch()..start();
  try {
    final response = await client
        .get(baseUrl.replace(path: '${baseUrl.path}/health'))
        .timeout(const Duration(seconds: 5));   // สั้นกว่าปกติ (7.7)
    stopwatch.stop();

    if (response.statusCode == 401) return const HealthUnauthorized();
    if (response.statusCode >= 500) {
      return HealthUnreachable('เซิร์ฟเวอร์ตอบว่ามีปัญหาภายใน '
          '(${response.statusCode})');
    }
    if (response.statusCode != 200) {
      return HealthWrongServer('ได้รหัส ${response.statusCode} '
          'จากที่อยู่นี้ อาจไม่ใช่เซิร์ฟเวอร์ WMS');
    }

    // ตรวจว่าเป็นเซิร์ฟเวอร์ที่ถูกต้องจริง ไม่ใช่แค่มีอะไรตอบมา
    final contentType = response.headers['content-type'] ?? '';
    if (!contentType.contains('application/json')) {
      return HealthWrongServer('ที่อยู่นี้ตอบเป็น $contentType '
          'ซึ่งไม่ใช่รูปแบบของ API');
    }

    return HealthOk(stopwatch.elapsed);
  } on SocketException catch (error) {
    return HealthUnreachable('ติดต่อเซิร์ฟเวอร์ไม่ได้: ${error.message}');
  } on TimeoutException {
    return const HealthUnreachable('เซิร์ฟเวอร์ไม่ตอบภายใน 5 วินาที');
  }
}

การตรวจ content-type จับกรณีที่พบบ่อยแต่ไม่มีใครคิดถึง — ผู้ใช้พิมพ์ที่อยู่ของเว็บ WMS แทนที่จะเป็นที่อยู่ของ API ซึ่งจะได้ 200 พร้อมหน้า HTML

10. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ปุ่มทดสอบบอกแค่ว่าได้หรือไม่ได้คืน boolคืนชนิดที่แยกสาเหตุได้
ผู้ใช้รอนานกว่าจะรู้ว่าที่อยู่ผิดใช้เวลารอเดียวกับคำขอปกติตั้งให้สั้นกว่า
ทดสอบผ่านแต่ใช้งานจริงไม่ได้ต่อไปโดนเว็บอื่นที่ตอบ 200ตรวจ content-type
คลังหยุดทำงานเพราะ /health ถูกปิดใช้เป็นด่านกั้นให้เป็นข้อมูล ไม่ใช่ด่าน
แบตหมดเร็วตรวจถี่เกินไปใช้ผลของคำขอจริงแทน
ได้ 401 แล้วคิดว่าที่อยู่ผิดไม่แยก 401 ออกจากต่อไม่ได้401 แปลว่าที่อยู่ถูกแล้ว

11. แบบฝึกหัด

ทีมติดตั้งที่สาขาใหม่ กดปุ่มทดสอบแล้วได้ผลต่างกันสี่ครั้ง

ครั้งที่อยู่ที่กรอกผลที่ได้
1192.168.1.50:5000ติดต่อไม่ได้
2http://192.168.1.50ที่อยู่นี้ตอบเป็น text/html
3http://192.168.1.50:5000เชื่อมต่อได้ กรุณาเข้าสู่ระบบ
4http://192.168.1.50:5000เชื่อมต่อได้ แต่ตอบช้า (4.8 วินาที)
  1. อธิบายว่าแต่ละครั้งเกิดอะไร
  2. ตอบว่าครั้งไหนถือว่าตั้งค่าสำเร็จแล้ว
  3. ครั้งที่ 4 ควรทำอย่างไรต่อ
ดูแนวคำตอบและเหตุผล

ข้อ 1 — แต่ละครั้งเกิดอะไร

ครั้งสาเหตุ
1ไม่มี http:// — ตัวแยกที่อยู่ตีความว่า 192.168.1.50 เป็น scheme (7.2)
2ไม่มี port จึงไปที่ port 80 ซึ่งเป็นเว็บ WMS ไม่ใช่ API
3ถูกต้องแล้ว — API ตอบและบอกว่าต้องเข้าสู่ระบบ
4ที่อยู่ถูก แต่เครือข่ายหรือเซิร์ฟเวอร์ช้า

ครั้งที่ 2 เป็นกรณีที่อันตรายที่สุดถ้าไม่มีการตรวจ content-type — เพราะเว็บ WMS ตอบ 200 กลับมา ถ้าตรวจแค่รหัสจะบอกว่า “เชื่อมต่อได้” ทั้งที่คุยผิดระบบ

แล้วปัญหาจะไปโผล่ตอนที่คนหน้างานกดโหลดงาน แล้วได้ HTML มาแทน JSON ซึ่งเป็นอาการที่ 7.8 อธิบายไว้ว่าไล่หาสาเหตุยาก

ข้อ 2 — ครั้งไหนถือว่าสำเร็จ

ครั้งที่ 3 — และเป็นจุดที่คนติดตั้งมักเข้าใจผิด

ความเข้าใจผิดความจริง
“ขึ้นว่าต้องเข้าสู่ระบบ แปลว่ายังไม่สำเร็จ”สำเร็จแล้ว — 401 มาจาก API ที่ถูกต้อง

สิ่งที่ 401 พิสูจน์ให้เห็นแล้ว

ที่อยู่ถูกต้อง        ✓  ไม่งั้นคงต่อไม่ถึง
เครือข่ายถึงกัน      ✓  ไม่งั้นคงเป็น "ติดต่อไม่ได้"
firewall เปิดแล้ว    ✓  เหตุผลเดียวกัน
เป็น API จริง        ✓  ไม่งั้นคงไม่ตอบ 401 แบบ JSON
เซิร์ฟเวอร์ทำงานอยู่  ✓

ข้อความบนหน้าจอจึงควรสื่อว่าเป็นความสำเร็จ ไม่ใช่ข้อผิดพลาด

✓ เชื่อมต่อได้ กรุณาเข้าสู่ระบบเพื่อเริ่มใช้งาน

ถ้าใช้สีแดงหรือคำว่า “ไม่สำเร็จ” คนติดตั้งจะนั่งแก้ที่อยู่ต่อไปเรื่อย ๆ ทั้งที่ถูกแล้ว

ข้อ 3 — ครั้งที่ 4 ควรทำอย่างไรต่อ

4.8 วินาทีสำหรับ endpoint ที่แทบไม่ทำอะไรเลย เป็นสัญญาณที่ต้องหาสาเหตุ ไม่ใช่ปล่อยผ่าน

สาเหตุที่เป็นไปได้ตรวจอย่างไร
สัญญาณ Wi-Fi อ่อนตรงจุดที่ยืนอยู่เดินไปใกล้จุดกระจายสัญญาณแล้วลองใหม่
เซิร์ฟเวอร์ทำงานหนักลองจากคอมพิวเตอร์เครื่องอื่นเทียบ
การแปลงชื่อเป็นที่อยู่ช้าไม่น่าใช่ เพราะกรอกเป็นตัวเลขอยู่แล้ว
เครือข่ายมีอุปกรณ์คั่นกลางที่ช้าให้ทีมไอทีตรวจเส้นทาง

บันทึกค่าที่วัดได้ไว้ในเอกสารของสาขา

แถวสุดท้ายมีค่ามากในระยะยาว — เมื่อสาขานั้นรายงานว่า “ช้าลง” อีกหกเดือนข้างหน้า จะมีตัวเลขให้เทียบว่าช้าลงจริงหรือช้ามาตั้งแต่แรก

บทเรียนที่ควรจำ: ปุ่มทดสอบที่ดีไม่ได้ตอบแค่ “ได้หรือไม่ได้” — มันย้ายการค้นพบปัญหามาไว้ตอนที่คนที่แก้ได้ยังอยู่ตรงนั้น และตัวเลขเวลาที่มันวัดได้ คือข้อมูลพื้นฐานที่ใช้เทียบในอนาคต

12. สรุปท้ายบท

  • ปุ่มทดสอบย้ายการค้นพบปัญหามาไว้ตอนติดตั้ง
  • ความล้มเหลวคือคำตอบ จึงคืนค่าไม่ใช่โยน
  • คืนชนิดที่แยกสาเหตุได้ ไม่ใช่ bool
  • ตั้งเวลารอสั้นกว่าคำขอปกติ
  • ตรวจ content-type เพื่อจับกรณีที่ต่อไปโดนเว็บอื่น
  • ได้ 401 แปลว่าที่อยู่ถูกแล้ว ซึ่งเป็นความสำเร็จ
  • อย่าใช้เป็นด่านกั้นการทำงาน
  • สถานะการเชื่อมต่อควรมาจากผลของคำขอจริง
  • เวลาที่วัดได้เป็นข้อมูลพื้นฐานที่ใช้เทียบในอนาคต

บทถัดไป: 7.17 จะทำให้ทุกอย่างที่ทำมาไล่หาสาเหตุได้ โดยไม่ทำให้ข้อมูลของลูกค้ารั่วออกไป
