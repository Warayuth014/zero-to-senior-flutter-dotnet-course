ระดับ: กลางใช้เวลา: 25–35 นาทีลงมือทำ: ใช้ endpoint ที่ออกแบบมาให้เว็บ โดยไม่ให้รูปแบบนั้นลามเข้าแอป
ต้องผ่านอะไรมาก่อน: 6.10 (ซองที่ห่อรายการ) และ 7.1 (การแยกชั้น)
จบบทนี้แล้วจะทำได้: ต่อกับ endpoint ที่ถูกออกแบบมาเพื่ออย่างอื่น โดยที่แอปยังสะอาด

1. ปัญหาที่บทนี้แก้

ในระบบ WMS จริง เว็บมักถูกสร้างก่อนแอปมือถือ และ endpoint ก็ถูกออกแบบให้ตารางบนเว็บใช้

// สิ่งที่ตารางบนเว็บส่งไป
{
  "pageIndex": 1,
  "pageItems": 20,
  "sortName": "createdAt",
  "sortOrder": "Desc",
  "filters": [
    {"fieldKey": "status", "fieldValue": "PENDING", "filterAction": "Equal"}
  ]
}
// สิ่งที่ได้กลับมา
{
  "items": [ ... ],
  "totalCount": 137
}

แอปมือถือแค่อยากได้ “งานที่ยังไม่เสร็จ 20 รายการแรก” แต่ต้องประกอบโครงทั้งหมดนี้ขึ้นมา

ถ้าปล่อยให้รูปแบบนี้ลามเข้าแอปผลที่ตามมา
store ต้องรู้จักคำว่า filterActionชั้นบนผูกกับรายละเอียดของเว็บ
หน้าจอต้องส่ง pageIndexเปลี่ยนวิธีแบ่งหน้าแล้วกระทบทุกที่
เพิ่มเงื่อนไขค้นหาต้องแก้หลายชั้นขัดกับหลักใน 7.1

2. คำศัพท์ที่ต้องรู้ก่อน

คำความหมาย
gridตารางข้อมูลบนเว็บที่มีการแบ่งหน้าและกรอง
query optionsก้อนที่บอกว่าอยากได้ข้อมูลแบบไหน
filterเงื่อนไขกรองข้อมูล
adapterตัวแปลงระหว่างสองรูปแบบที่ไม่ตรงกัน
leaky abstractionรายละเอียดของชั้นล่างที่รั่วขึ้นมาชั้นบน

3. Mental model: แปลที่ชั้นเดียว

หน้าจอ / store        พูดภาษาของงาน
       │              "งานที่ยังไม่เสร็จ หน้าแรก 20 รายการ"
       ▼
repository            ← แปลตรงนี้ที่เดียว
       │
       ▼
endpoint ของเว็บ      พูดภาษาของตาราง
                      pageIndex, filters, filterAction
ประโยคเดียวที่ต้องจำ: endpoint ที่ออกแบบมาเพื่ออย่างอื่น ไม่ใช่ปัญหา — ปัญหาคือปล่อยให้รูปแบบของมันเดินทางเข้ามาถึงชั้นที่ไม่ควรรู้จัก

4. ทำให้ชั้นบนพูดภาษาของงาน

// สิ่งที่ store เห็น — ไม่มีคำว่า pageIndex หรือ filters เลย
abstract interface class TaskRepository {
  Future<TaskPage> fetchOpen({
    String? search,
    int skip = 0,
    int take = 20,
  });
}
class TaskPage {
  const TaskPage({required this.items, required this.totalCount});
  final List<WmsTask> items;
  final int totalCount;
  bool get hasMore => items.length < totalCount;
}

สังเกตว่าใช้ skip และ take ไม่ใช่ pageIndex

pageIndexskip / take
เหมาะกับตารางที่มีปุ่มเลขหน้ารายการที่โหลดเพิ่มเมื่อเลื่อนถึงท้าย
คำนวณจากอะไรหมายเลขหน้าจำนวนที่โหลดมาแล้ว

บนมือถือแทบไม่มีปุ่มเลขหน้า มีแต่การเลื่อนลงแล้วโหลดเพิ่ม — skip จึงตรงกับสิ่งที่หน้าจอต้องการมากกว่า

5. แปลงที่ repository

@override
Future<TaskPage> fetchOpen({
  String? search,
  int skip = 0,
  int take = 20,
}) async {
  // แปลงจากภาษาของงาน เป็นภาษาของตารางบนเว็บ
  final json = await api.postJson('/api/WMS/mobile_tasks/query', {
    'pageIndex': (skip ~/ take) + 1,       // ตารางนับหน้าจาก 1
    'pageItems': take,
    'sortName': 'createdAt',
    'sortOrder': 'Desc',
    'filters': [
      {'fieldKey': 'status', 'fieldValue': 'PENDING', 'filterAction': 'Equal'},
      if (search != null && search.isNotEmpty)
        {
          'fieldKey': 'palletCode',
          'fieldValue': search,
          'filterAction': 'Contains',
        },
    ],
  });

  final page = TaskPageDto.fromJson(json);
  return TaskPage(
    items: [for (final dto in page.items) toTask(dto)],
    totalCount: page.totalCount,
  );
}

ความยุ่งทั้งหมดอยู่ในเมธอดเดียว และไม่มีอะไรรั่วออกไป

บรรทัด (skip ~/ take) + 1 มีข้อจำกัดที่ต้องรู้ตัว — มันใช้ได้เฉพาะเมื่อ skip หารด้วย take ลงตัวเสมอ ถ้าหน้าจอส่ง skip: 15, take: 20 มา จะได้หน้าที่ 1 ซึ่งเป็นข้อมูลซ้ำกับที่โหลดไปแล้ว

ทางแก้คือให้ repository บังคับเงื่อนไขนั้นไว้

assert(skip % take == 0, 'endpoint นี้แบ่งเป็นหน้า จึงข้ามได้ทีละหน้าเท่านั้น');

หรือถ้าคุมไม่ได้ ต้องขอให้ฝั่งหลังบ้านเพิ่ม endpoint ที่รับ skip ตรง ๆ

6. เมื่อ endpoint ให้ข้อมูลเกินที่ต้องการ

ตารางบนเว็บมักต้องการ field เยอะกว่ามือถือมาก

{
  "id": "T-001",
  "palletCode": "PAL-0101",
  "quantity": 12,
  "status": "PENDING",
  "createdAt": "...",
  "createdByName": "สมชาย ใจดี",
  "createdByDepartment": "คลังสินค้า",
  "lastModifiedAt": "...",
  "lastModifiedByName": "...",
  "remarks": "...",
  "attachmentCount": 0,
  "approvalHistory": [ ... ]
}
สิ่งที่ควรทำเพราะ
อ่านเฉพาะ field ที่ใช้ตามหลักใน 6.2 — ผ่อนกับสิ่งที่ไม่ได้ใช้
ไม่ต้องทำ DTO ให้ครบทุก fieldfield ที่ไม่ได้อ่านจะถูกข้ามไปเอง
บอกฝั่งหลังบ้านถ้าข้อมูลใหญ่เกินไปPDA บนเครือข่ายในคลังรับข้อมูลใหญ่ช้า

ข้อสุดท้ายเป็นเรื่องจริงที่วัดได้ — รายการ 20 รายการที่มี approvalHistory ติดมาด้วย อาจใหญ่กว่าที่จำเป็นสิบเท่า

ถ้าแก้ที่ต้นทางไม่ได้ ให้วัดก่อนว่าช้าจริงไหมตามหลักใน 2.10 — อย่าเพิ่งขอ endpoint ใหม่โดยที่ยังไม่รู้ว่าเป็นปัญหา

7. เมื่อ endpoint ใช้ POST สำหรับการอ่าน

สังเกตว่าตัวอย่างข้างบนใช้ postJson ทั้งที่เป็นการอ่านข้อมูล

final json = await api.postJson('/api/WMS/mobile_tasks/query', { ... });

เป็นเรื่องปกติสำหรับ endpoint แบบตาราง เพราะเงื่อนไขกรองยาวเกินกว่าจะใส่ใน URL ได้ — แต่ต้องรู้ตัวว่าเสียอะไรไป

สิ่งที่เสียไปผลกระทบ
ตัวกลางลองใหม่ให้เองไม่ได้ไม่กระทบ — เราจัดการเองอยู่แล้ว (7.14)
ผลไม่ถูกเก็บไว้ใช้ซ้ำไม่กระทบมาก
ธง outcomeUnknown จะเป็น trueกระทบ — ดูข้างล่าง

แถวสุดท้ายเป็นปัญหาจริง เพราะ postJson ใน lab ตั้งธงว่าไม่รู้ผลเสมอตาม 7.4

} on SocketException {
  // POST เปลี่ยนข้อมูลได้ จึงไม่รู้ว่า server ทำไปแล้วหรือยัง
  throw const ApiException(..., outcomeUnknown: true, ...);
}

แต่การอ่านข้อมูลไม่ได้เปลี่ยนอะไร — การบอกว่า “ไม่รู้ผล” จึงผิด และจะทำให้ชั้นบนไปเรียกการตรวจสอบที่ไม่จำเป็นตาม 4.13

ทางแก้คือให้ผู้เรียกบอกได้ว่าคำขอนี้เปลี่ยนข้อมูลหรือไม่

Future<Map<String, dynamic>> postJson(
  String path,
  Map<String, dynamic> body, {
  Map<String, String> headers = const {},
  bool mutates = true,        // ค่าตั้งต้นปลอดภัยไว้ก่อน
}) async {
  ...
  } on SocketException {
    throw ApiException(
      mutates ? 'การเชื่อมต่อขาดระหว่างส่งคำสั่ง' : 'เชื่อมต่อ server ไม่ได้',
      outcomeUnknown: mutates,
      kind: ApiFailureKind.network,
    );
  }
}
// การอ่านผ่าน POST — บอกว่าไม่ได้เปลี่ยนอะไร
final json = await api.postJson(
  '/api/WMS/mobile_tasks/query',
  { ... },
  mutates: false,
);

ค่าตั้งต้นเป็น true เพราะถ้าลืมระบุ จะได้ผลที่ปลอดภัยกว่า — คือถือว่าไม่รู้ผลไว้ก่อน

8. อย่าเพิ่งขอ endpoint ใหม่ทันที

ขั้นทำอะไร
1ใช้ endpoint ที่มีอยู่ แล้วแปลงที่ repository
2วัดว่าช้าจริงไหม หรือข้อมูลใหญ่เกินไปจริงไหม
3ถ้าเป็นปัญหาจริง จึงขอ endpoint ที่เหมาะกับมือถือ

เหตุผลที่ไม่ควรข้ามไปขั้น 3 เลย

ต้นทุนของ endpoint ใหม่
ฝั่งหลังบ้านต้องดูแลสองทางที่ทำเรื่องเดียวกัน
กฎธุรกิจอาจไม่ตรงกันเมื่อแก้ที่เดียว
ต้องทดสอบสองชุด

แต่เมื่อถึงขั้น 3 จริง ๆ สิ่งที่ควรขอคือ endpoint ที่ คืนเฉพาะสิ่งที่มือถือใช้ ไม่ใช่แค่เปลี่ยนรูปแบบพารามิเตอร์

9. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
store ต้องรู้จัก filterActionปล่อยรูปแบบของเว็บลามขึ้นมาแปลที่ repository
โหลดหน้าถัดไปแล้วได้ข้อมูลซ้ำคำนวณหมายเลขหน้าจาก skip ที่หารไม่ลงตัวบังคับเงื่อนไข หรือขอ endpoint ที่รับ skip
รายการโหลดช้ามากบน PDAข้อมูลติดมาเกินที่ใช้วัดก่อน แล้วค่อยขอลดข้อมูล
อ่านข้อมูลแล้วแอปบอกว่า “ไม่รู้ผล”ใช้ POST อ่านข้อมูลแต่ธงตั้งผิดให้ระบุได้ว่าเปลี่ยนข้อมูลหรือไม่
มี endpoint สองทางที่ทำเรื่องเดียวกันแต่ผลไม่ตรงขอ endpoint ใหม่เร็วเกินไปใช้ของเดิมก่อน วัดก่อน

10. แบบฝึกหัด

ทีมต้องทำหน้าค้นหางานบนมือถือ โดยใช้ endpoint ของตารางบนเว็บที่มีอยู่แล้ว

สิ่งที่หน้าจอต้องการ
ค้นด้วยรหัสพาเลท แบบพิมพ์ไปค้นไป
แสดงเฉพาะงานที่ยังไม่เสร็จ
เรียงจากใหม่ไปเก่า
โหลดเพิ่มเมื่อเลื่อนถึงท้าย ครั้งละ 20
  1. ออกแบบลายเซ็นของ repository ที่ store จะเรียก
  2. ระบุว่าอะไรที่ store ไม่ควรรู้เลย
  3. ตอบว่าถ้าฝั่งหลังบ้านเปลี่ยนจาก filters เป็นรูปแบบใหม่ จะกระทบกี่ไฟล์ และทำไม
ดูแนวคำตอบและเหตุผล

ข้อ 1 — ลายเซ็น

abstract interface class TaskRepository {
  /// ค้นงานที่ยังไม่เสร็จ
  ///
  /// [skip] คือจำนวนที่โหลดมาแล้ว ไม่ใช่หมายเลขหน้า
  Future<TaskPage> searchOpen({
    String? palletCode,
    int skip = 0,
    int take = 20,
  });
}

สังเกตสามอย่าง

สิ่งที่ทำเพราะ
ชื่อ palletCode ไม่ใช่ searchบอกว่าค้นด้วยอะไร ไม่ใช่ค้นลอย ๆ
ไม่มีพารามิเตอร์เรียงลำดับหน้าจอนี้เรียงแบบเดียวเสมอ — ไม่ต้องเปิดให้เลือก
ไม่มีพารามิเตอร์สถานะชื่อเมธอดบอกแล้วว่าเอาเฉพาะที่ยังไม่เสร็จ

สองข้อล่างเป็นการตัดสินใจที่สำคัญ — อย่าเปิดพารามิเตอร์ที่ยังไม่มีใครใช้ เพราะทุกพารามิเตอร์คือสิ่งที่ต้องทดสอบและดูแลต่อไป

ถ้าวันหนึ่งต้องให้เลือกเรียงลำดับได้ ค่อยเพิ่มตอนนั้น ตามหลักที่ไม่เขียนโค้ดเผื่ออนาคต

ข้อ 2 — สิ่งที่ store ไม่ควรรู้

store ไม่ควรรู้เพราะ
pageIndex, pageItemsเป็นวิธีนับของตารางบนเว็บ
filters, fieldKey, filterActionเป็นโครงของตัวกรองบนเว็บ
sortName, sortOrderเหมือนกัน
ค่า 'PENDING'เป็นค่าบนสาย — store ควรรู้จัก TaskStatus เท่านั้น (6.9)
ว่าเป็น GET หรือ POSTเป็นเรื่องของการส่ง
เส้นทาง /api/WMS/...เป็นเรื่องของ repository (7.1)

แถวที่สี่มักถูกมองข้าม — ถ้า store เขียน 'PENDING' ตรง ๆ แปลว่าค่าบนสายรั่วขึ้นมาแล้ว ซึ่งเป็นปัญหาเดียวกับที่ 6.9 แก้ไว้

ข้อ 3 — ถ้ารูปแบบตัวกรองเปลี่ยน

ไฟล์เดียว คือไฟล์ที่มี RemoteTaskRepository

// เดิม
'filters': [
  {'fieldKey': 'status', 'fieldValue': 'PENDING', 'filterAction': 'Equal'},
],

// ใหม่ — สมมติว่าเปลี่ยนเป็นรูปแบบอื่น
'where': {'status': {'eq': 'PENDING'}},
ไฟล์ต้องแก้ไหม
หน้าจอค้นหาไม่
task_store.dartไม่
task_repository.dartใช่
api_client.dartไม่
เทสต์ของ repositoryใช่

ทำไมถึงเป็นแค่ไฟล์เดียว

เพราะลายเซ็น searchOpen({String? palletCode, int skip, int take}) ไม่มีอะไรที่สะท้อนรูปแบบของเซิร์ฟเวอร์เลย — มันพูดแต่เรื่องที่หน้าจอสนใจ

store พูดว่า      "ค้น PAL-01 ข้ามไป 20 เอามา 20"
repository แปลเป็น  รูปแบบอะไรก็ได้ที่ endpoint ต้องการ ณ ตอนนั้น

ถ้าลายเซ็นเป็นแบบนี้แทน ผลจะต่างออกไปทันที

// ลายเซ็นที่รั่ว
Future<TaskPage> query(Map<String, dynamic> queryOptions);

แบบนี้ store ต้องประกอบ queryOptions เอง ซึ่งแปลว่ารูปแบบของเซิร์ฟเวอร์อยู่ใน store แล้ว และการเปลี่ยนจะกระทบทุกที่ที่เรียก

บทเรียนที่ควรจำ: ลายเซ็นของ repository คือด่านกันการเปลี่ยนแปลง — ถ้ามันพูดภาษาของงาน การเปลี่ยนฝั่งเซิร์ฟเวอร์จะหยุดอยู่ที่ชั้นเดียว แต่ถ้ามันพูดภาษาของเซิร์ฟเวอร์ ก็ไม่ได้กันอะไรเลย

11. สรุปท้ายบท

  • endpoint ที่ออกแบบมาเพื่ออย่างอื่นใช้ได้ ถ้าไม่ปล่อยให้รูปแบบลามเข้ามา
  • แปลจากภาษาของงานเป็นภาษาของ endpoint ที่ repository ที่เดียว
  • ใช้ skip และ take เพราะตรงกับการเลื่อนโหลดเพิ่มบนมือถือ
  • อ่านเฉพาะ field ที่ใช้ ไม่ต้องทำ DTO ให้ครบ
  • การอ่านผ่าน POST ต้องบอกได้ว่าไม่ได้เปลี่ยนข้อมูล
  • อย่าเปิดพารามิเตอร์ที่ยังไม่มีใครใช้
  • ใช้ของเดิมและวัดก่อน ค่อยขอ endpoint ใหม่
  • ลายเซ็นของ repository คือด่านกันการเปลี่ยนแปลง
