5. เกณฑ์ที่ต้องผ่าน

#เกณฑ์บท
1store ไม่รู้จัก URL, header หรือรหัสตอบกลับ7.1
2ประกอบ URL ด้วย Uri และ orderNo ถูก encode7.2
3การอ่านใช้ GET ส่วนการยืนยันใช้ POST7.4
4รหัสคำสั่งส่งทาง header และสร้างนอกส่วนที่ถูกลองใหม่7.5, 7.15
5เทสต์ตรวจคำขอที่ส่งไป ไม่ใช่แค่ผลลัพธ์7.6
6เน็ตหลุดตอน GET กับตอน POST ตั้ง outcomeUnknown ต่างกัน7.4, 7.7
7204 และ body ว่างไม่ทำให้พัง7.8
8แปลง errorCode เป็นผลลัพธ์ที่แอปทำต่างกันได้7.9
9การยืนยันคืนค่า ไม่ใช่โยน เพราะมีหลายผลลัพธ์7.10
10ข้อผิดพลาดรายช่องไปแสดงที่ช่องที่ถูกต้อง7.11
11รายการใช้ skip/take และกรองซ้ำก่อนต่อท้าย7.12, 7.13
12GET ลองใหม่เองได้ POST ลองใหม่ได้เพราะมีรหัสคำสั่ง7.14
13มีปุ่มทดสอบการเชื่อมต่อที่แยกสาเหตุได้7.16
14บันทึกคำขอโดยไม่มี token และไม่มี body7.17

เกณฑ์ข้อ 8 กับ 9 เป็นสองข้อที่ยากที่สุด และทั้งคู่เป็นเรื่องเดียวกัน — ทำให้ผลลัพธ์ที่ต่างกันถูกแยกตั้งแต่ชนิด ไม่ใช่ให้ชั้นบนไปอ่านข้อความเอง

6. คำใบ้สำหรับข้อที่ยาก

เกณฑ์ 8 และ 9 — ผลลัพธ์ของการยืนยัน

sealed class ConfirmOutcome {
  const ConfirmOutcome();
}

class ConfirmSucceeded extends ConfirmOutcome {
  const ConfirmSucceeded(this.receiptNo, {required this.wasAlreadyConfirmed});
  final String receiptNo;
  /// true เมื่อ server ตอบผลเดิมเพราะรหัสคำสั่งซ้ำ (7.15)
  final bool wasAlreadyConfirmed;
}

class ConfirmLocked extends ConfirmOutcome {
  const ConfirmLocked(this.message);
  final String message;
}

class ConfirmQuantityMismatch extends ConfirmOutcome {
  const ConfirmQuantityMismatch(this.fieldErrors);
  final Map<String, String> fieldErrors;    // ไปแสดงที่ช่อง (7.11)
}

class ConfirmFailed extends ConfirmOutcome {
  const ConfirmFailed(this.kind, this.message, {this.traceId});
  final ApiFailureKind kind;
  final String message;
  final String? traceId;
}

สังเกตว่า INBOUND_ALREADY_CONFIRMED กลายเป็น ConfirmSucceeded ไม่ใช่ความล้มเหลว — เพราะผลลัพธ์ที่ผู้ใช้ต้องการเกิดขึ้นแล้ว ตามข้อสรุปของ 7.9

เกณฑ์ 12 — การลองใหม่ของ POST

// รหัสสร้างที่นี่ ครั้งเดียว

7. เทสต์ที่ต้องเขียน

อย่างน้อย 15 ข้อ ตัวอย่างสามข้อที่คิดยากที่สุด

test('รหัสใบที่มีอักขระพิเศษต้องถูก encode ใน path', () async {
  http.Request? sent;
  final api = clientThatReturns(
    (_) => jsonResponse(const <String, Object>{}, 200),
    onRequest: (request) => sent = request,
  );

  await InboundRepository(api).confirm('IN/2026/0812', commandId: 'cmd-1');

  // ขีดทับต้องไม่กลายเป็นตัวแบ่งเส้นทาง (7.2)
  expect(sent!.url.path, contains('IN%2F2026%2F0812'));
});
test('ยืนยันซ้ำด้วยรหัสเดิม ต้องได้ผลสำเร็จที่บอกว่าทำไปแล้ว', () async {
  final api = clientThatReturns(
    (_) => jsonResponse({
      'orderNo': 'IN-2026-0812',
      'receiptNo': 'RC-2481',
      'alreadyConfirmed': true,
      'confirmedAt': '2026-08-16T03:12:00Z',
    }, 200),
  );

  final outcome = await InboundRepository(api)
      .confirm('IN-2026-0812', commandId: 'cmd-1');

  // เป็นความสำเร็จ ไม่ใช่ความล้มเหลว (7.9, 7.15)
  expect(outcome, isA<ConfirmSucceeded>());
  expect((outcome as ConfirmSucceeded).wasAlreadyConfirmed, isTrue);
  expect(outcome.receiptNo, 'RC-2481');
});
test('ลองใหม่ต้องส่งรหัสคำสั่งเดิมทุกครั้ง', () async {
  final commandIds = <String?>[];
  var attempts = 0;
  final api = clientThatReturns(
    (_) {
      attempts++;
      if (attempts < 3) return http.Response('', 503);
      return jsonResponse({...}, 200);
    },
    onRequest: (request) => commandIds.add(request.headers['Idempotency-Key']),
  );

  await InboundRepository(api, retry: fastRetry)
      .confirm('IN-2026-0812', commandId: 'cmd-1');

  // สามครั้ง รหัสเดียวกันทั้งหมด (7.15)
  expect(commandIds, ['cmd-1', 'cmd-1', 'cmd-1']);
});

เทสต์ข้อสุดท้ายจับบั๊กที่ 7.14 หัวข้อ 11 เตือนไว้ — ถ้าสร้างรหัสข้างใน closure จะได้ ['cmd-1', 'cmd-2', 'cmd-3'] และระบบกันซ้ำจะไม่ทำงานเลย

8. ข้อผิดพลาดที่พบบ่อยตอนทำ lab นี้

อาการสาเหตุวิธีแก้
เทสต์ผ่านแต่รหัสคำสั่งไม่ถูกส่งตรวจแค่ผลลัพธ์ตรวจ header ที่ส่งไป (7.6)
404 เมื่อรหัสใบมีขีดทับไม่ได้ encodeUri.encodeComponent (7.2)
ลองใหม่แล้วได้ใบรับซ้ำสร้างรหัสข้างใน closureย้ายออกมาข้างนอก (7.15)
“ยืนยันไปแล้ว” แสดงเป็นข้อผิดพลาดแปลง 409 เป็นความล้มเหลวทั้งหมดแยกตาม errorCode (7.9)
อ่านข้อมูลแล้วแอปบอกว่าไม่รู้ผลใช้ POST อ่านโดยไม่ระบุระบุว่าไม่ได้เปลี่ยนข้อมูล (7.12)
ข้อผิดพลาดรายช่องไม่ขึ้นที่ช่องชื่อ field จาก server เป็นตัวใหญ่เทียบส่วนท้ายแบบไม่สนตัวพิมพ์ (7.11)
token อยู่ในไฟล์บันทึกบันทึก header ทั้งชุดปิดค่าที่ตัวกลาง (7.17)

9. แบบทดสอบตัวเองก่อนจบ Part 7

  1. มีกี่ชั้น และแต่ละชั้นรู้เรื่องอะไรบ้าง (7.1)
  2. ทำไมไม่ควรต่อ URL ด้วยการต่อข้อความ (7.2)
  3. resolve ทำอะไรที่ทำให้พลาดง่าย (7.2)
  4. localhost บน emulator หมายถึงเครื่องไหน (7.3)
  5. ทำไมเซิร์ฟเวอร์ที่ bind ที่ localhost ถึงเข้าจากเครื่องอื่นไม่ได้ (7.3)
  6. เลือก GET หรือ POST ตัดสินจากอะไร (7.4)
  7. ทำไม GET กับ POST ตั้ง outcomeUnknown ต่างกัน (7.4)
  8. ทำไมต้องอ่าน token ตอนส่ง ไม่ใช่จำไว้ตอนสร้าง (7.5)
  9. 401 กับ 403 ต้องทำต่างกันอย่างไร (7.5, 7.9)
  10. ทำไมต้องตรวจคำขอที่ส่งไป ไม่ใช่แค่ผลลัพธ์ (7.6)
  11. หมดเวลารอแปลว่าเซิร์ฟเวอร์ไม่ทำหรือไม่ (7.7)
  12. ทำไมคำสั่งที่เปลี่ยนข้อมูลควรตั้งเวลารอนานกว่า (7.7)
  13. รหัส 200 รับประกันอะไรและไม่รับประกันอะไร (7.8)
  14. ทำไมห้ามใช้ข้อความจากเซิร์ฟเวอร์เป็นเงื่อนไข (7.9)
  15. เลือกโยนหรือคืนค่า ตัดสินจากอะไร (7.10)
  16. title กับ detail ต่างกันอย่างไร (7.11)
  17. คีย์ว่างใน errors แปลว่าอะไร (7.11)

  20. สองคำถามที่ต้องตอบก่อนลองใหม่คืออะไร (7.14)

  22. ได้ 401 จากปุ่มทดสอบ แปลว่าอะไร (7.16)

10. สรุปท้าย Part 7

สิ่งที่ทำได้แล้ว

  • วางชั้นให้การเปลี่ยนแปลงหยุดอยู่ที่ชั้นเดียว
  • ประกอบที่อยู่ให้ถูกทุกกรณี รวมถึงค่าที่มีอักขระพิเศษ
  • ทำให้เครื่องจริงคุยกับเซิร์ฟเวอร์ได้ และไล่หาสาเหตุเมื่อไม่ได้
  • เลือกวิธีส่งคำขอตามผลที่เกิดกับเซิร์ฟเวอร์
  • ส่งข้อมูลกำกับคำขอให้ครบและไม่รั่ว
  • ทดสอบทุกกรณีโดยไม่ต้องมีเซิร์ฟเวอร์
  • แยกความล้มเหลวเป็นชนิดที่ผู้ใช้ทำต่อได้
  • แยกความสำเร็จของการส่ง ออกจากความถูกต้องของข้อมูล
  • อ่านรายละเอียดข้อผิดพลาดมาตรฐานและแสดงให้ตรงจุด
  • ลองใหม่โดยไม่ทำให้ข้อมูลซ้ำหรือซ้ำเติมเซิร์ฟเวอร์
  • ทำให้คำสั่งส่งซ้ำได้อย่างปลอดภัย
  • บันทึกให้พอไล่หาสาเหตุ โดยไม่ให้ข้อมูลรั่ว

สิ่งที่ยังไม่ได้ทำและจะทำที่ไหน
