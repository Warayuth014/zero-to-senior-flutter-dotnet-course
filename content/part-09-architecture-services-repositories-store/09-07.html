1. store คืออะไร

Store คือวัตถุที่เก็บสถานะของฟีเจอร์หนึ่ง และประกาศให้คนที่สนใจรู้เมื่อสถานะเปลี่ยน — เป็นคำตอบของคำถามว่า “ตอนนี้เรื่องนี้เป็นอย่างไรอยู่”

ต่างจาก repository ตรงที่ repository ไม่จำอะไรเลย ถามครั้งไหนก็ไปเอาใหม่ ส่วน store จำ ว่าเอามาแล้วได้อะไร กำลังโหลดอยู่ไหม และพังตรงไหน

2. ChangeNotifier ทำงานอย่างไร

class PalletStore extends ChangeNotifier {
  List<Pallet> _pallets = const [];

  List<Pallet> get pallets => List.unmodifiable(_pallets);

  Future<void> load(String zone) async {
    _loading = true;
    notifyListeners();      // ← บอกทุกคนที่ฟังอยู่ว่ามีอะไรเปลี่ยน
    ...
  }
}
ส่วนหน้าที่
_pallets (มีขีดล่าง)ของจริง แก้ได้เฉพาะข้างในคลาส
get palletsทางออกให้คนข้างนอกอ่าน
List.unmodifiableกันไม่ให้จอเผลอ .add() ใส่ลิสต์ตรง ๆ — ถ้าทำได้ store จะไม่รู้ว่ามีการเปลี่ยน แล้วหน้าจอกับความจริงจะไม่ตรงกัน
notifyListeners()สั่งให้ widget ที่ผูกไว้วาดใหม่

3. สถานะแยกเป็นสองกลุ่ม

String _zone = '';
List<Pallet> _pallets = const [];       // ข้อมูล
bool _loading = false;                   // สถานะของงาน
PalletFailure? _failure;                 // สถานะของงาน
final Set<String> _busyCodes = {};       // สถานะของงาน รายแถว
int _generation = 0;                     // กลไกภายใน ไม่เปิดให้ใครเห็น

สังเกตว่า _generation ไม่มี getter — เป็นกลไกภายในล้วน ๆ ที่ไม่มีใครนอกคลาสต้องรู้

4. getter ที่คำนวณ ไม่ใช่ field ที่เก็บ

bool get initialLoading => _loading && _pallets.isEmpty;
bool get refreshing => _loading && _pallets.isNotEmpty;

สองอย่างนี้คือสิ่งที่จอต้องรู้ คือ “โชว์วงหมุนเต็มจอ” กับ “โชว์แถบบาง ๆ ข้างบน” แต่ทั้งคู่คำนวณจาก _loading กับจำนวนข้อมูล ไม่ได้เก็บแยก

ถ้าเก็บเป็น field แยกจะมีสถานะที่เป็นไปไม่ได้เกิดขึ้นได้ทันที เช่น initialLoading == true พร้อมกับ refreshing == true ซึ่งไม่มีความหมาย และวันหนึ่งจะมีบั๊กที่มาจากตรงนี้

หลัก: เก็บให้น้อยที่สุดเท่าที่จำเป็น แล้วคำนวณที่เหลือ — สิ่งที่ไม่ได้เก็บ ไม่มีวันไม่ตรงกัน

5. งานที่ store ต้องทำที่ repository ทำแทนไม่ได้

Future<void> load(String zone) async {
  _zone = zone;
  final generation = ++_generation;       // จองเลขรุ่นของคำขอนี้
  _loading = true;
  _failure = null;
  notifyListeners();

  try {
    final result = await repository.fetchInZone(zone);
    if (generation != _generation) return;   // มีคำขอใหม่แซงไปแล้ว ทิ้งผลนี้
    _pallets = result;
  } on PalletFailure catch (failure) {
    if (generation != _generation) return;
    _failure = failure;
  } finally {
    if (generation == _generation) {
      _loading = false;
      notifyListeners();
    }
  }
}

สามอย่างที่เกิดขึ้นตรงนี้เป็นเรื่องของ “ลำดับเวลาในแอป” ซึ่ง repository ไม่มีทางรู้

บรรทัดป้องกันอะไร
++_generation แล้วเทียบทีหลังผู้ใช้กดโหลดใหม่รัว ๆ หรือเปลี่ยนโซนเร็ว ๆ แล้วคำตอบเก่าที่ช้ากลับมาทับคำตอบใหม่ (4.10)
_failure = null ตอนเริ่มข้อความผิดพลาดเก่าค้างอยู่ทั้งที่กำลังลองใหม่
finally + เช็ค generationวงหมุนค้างเมื่อคำขอเก่าจบหลังคำขอใหม่เริ่ม

เทสต์ที่พิสูจน์ข้อแรกอยู่ใน architecture_test.dart และมันจงใจตอบคำขอที่สองก่อนคำขอแรก

final first = store.load('A1');
final second = store.load('B2');

repository.completeAt(1, [palletB2]);   await second;   // ตอบตัวที่สองก่อน
repository.completeAt(0, [palletA1]);   await first;    // แล้วค่อยตอบตัวแรก

expect(store.pallets.single.code, 'PLT-B2');   // ของใหม่ต้องชนะ

6. กันกดซ้ำรายแถว

Future<void> hold(Pallet pallet) async {
  if (_busyCodes.contains(pallet.code)) return;   // กดซ้ำระหว่างรอ ไม่ยิงซ้ำ
  _busyCodes.add(pallet.code);
  ...
  } finally {
    _busyCodes.remove(pallet.code);
    notifyListeners();
  }
}

ใช้ Set ไม่ใช่ bool ตัวเดียว เพราะผู้ใช้ล็อกพาเลทหลายใบพร้อมกันได้ ถ้าใช้ bool busy ตัวเดียว การล็อกใบหนึ่งจะปิดปุ่มของทุกใบ

บน PDA ปุ่มโดนกดซ้ำเป็นเรื่องปกติ ไม่ใช่กรณีพิเศษ — หน้าจอไม่ตอบสนองทันทีเพราะเน็ตช้า ผู้ใช้ก็กดอีก

7. สิ่งที่ store ทำไม่ได้ และเป็นเรื่องดี

import 'package:flutter/foundation.dart';   // ไม่ใช่ material.dart

บรรทัดเดียวนี้ทำให้ store

  • เปิด dialog เองไม่ได้
  • push route เองไม่ได้
  • เรียก ScaffoldMessenger เองไม่ได้
  • ทดสอบได้โดยไม่ต้องสร้าง widget เลย

wms_absolute_mobile ใช้กฎเดียวกัน เปิด warehouse_store.dart บรรทัดที่ 2 จะเห็น foundation.dart ไม่ใช่ material.dart ทั้งที่ไฟล์ยาว 1,684 บรรทัด

แล้วเวลามันต้องบอกให้จอโชว์ข้อความล่ะ? ใช้ callback

void Function(String msg, {bool err})? onToast;
void Function(bool ok)? onFlash;

store บอกว่า “เกิดเรื่องนี้” ส่วนจอเป็นคนตัดสินใจว่าจะแสดงอย่างไร — แนวเดียวกับ onTokenChanged ใน 8.4

8. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
หน้าจอไม่อัปเดตทั้งที่ข้อมูลเปลี่ยนลืม notifyListeners()เรียกทุกครั้งที่ค่าที่คนอื่นเห็นเปลี่ยน
วาดใหม่รัวจนกระตุกเรียก notifyListeners() ในลูปเปลี่ยนค่าให้เสร็จก่อน แล้วค่อยเรียกครั้งเดียว
จอแก้ลิสต์ได้เองแล้วสถานะเพี้ยนคืน _pallets ตรง ๆList.unmodifiable
วงหมุนค้างไม่ได้ตั้ง _loading = false ใน finallyใช้ finally เสมอ
A ChangeNotifier was used after being disposedงาน async จบหลัง store ถูกทิ้งเช็คก่อน notify หรือยกเลิกงานตอน dispose (8.8)

9. แบบฝึกหัด

หัวหน้าคลังขอให้เพิ่ม “กรองเฉพาะพาเลทที่ยังไม่ถูกล็อก” ด้วยสวิตช์บนหัวจอ

  1. สถานะของสวิตช์ควรอยู่ที่ไหน — ในจอ หรือใน store
  2. ลิสต์ที่กรองแล้วควรเป็น field หรือ getter
แนวคำตอบ

1. ขึ้นกับว่ามันต้องอยู่รอดข้ามอะไรบ้าง ถ้าเปิดจอใหม่แล้วเริ่มใหม่ทุกครั้งก็อยู่ในจอได้ แต่ถ้าอยากให้ค่าคงอยู่ตอนกลับมาจากจออื่น หรืออยากเขียนเทสต์ว่ากรองถูกไหมโดยไม่สร้าง widget ให้อยู่ใน store

คำแนะนำจริง: เริ่มที่จอ ย้ายเมื่อมีเหตุผลข้อใดข้อหนึ่งข้างบนเกิดขึ้นจริง

2. getter เสมอ

List<Pallet> get visiblePallets =>
    _hideHeld ? [for (final p in _pallets) if (!p.onHold) p] : pallets;

ถ้าเก็บเป็น field จะต้องจำให้ได้ว่าต้องคำนวณใหม่ตอนไหนบ้าง — ตอนโหลดเสร็จ ตอนล็อกสำเร็จ ตอนสลับสวิตช์ และตอนที่ยังนึกไม่ออกในอนาคต ลืมที่เดียวก็เพี้ยน

10. สรุป

  • store จำสถานะของฟีเจอร์ ส่วน repository ไม่จำอะไรเลย
  • เก็บให้น้อย คำนวณให้มาก — สถานะที่ไม่ได้เก็บไม่มีวันไม่ตรงกัน
  • เลขรุ่นของคำขอและ Set ของแถวที่กำลังทำงาน เป็นหน้าที่ของ store
  • store ที่ไม่ import material.dart คือ store ที่ทดสอบได้และแยกหน้าที่ชัด

บทหน้า: ใครเป็นคนประกอบ repository กับ store เข้าด้วยกัน และประกอบที่ไหน
