สามช่องนี้คือแหล่งความจริงเรื่องผู้ใช้ทั้งแอป ไม่มีที่อื่นเก็บอีก — จอโดเมนทุกจอรับ userId กับ fullName ผ่านตัวสร้าง

2. โหลดผู้ใช้

@override
void initState() {
  super.initState();
  _loadUser();
}

Future<void> _loadUser() async {
  final session = await _sessionService.loadSession();
  setState(() {
    _userId = session.userId;
    _fullName = session.fullName;
    _role = session.role;
  });
}

เป็นรูปแบบเดียวกับที่ 4.11 สอน — initState เรียกงาน async แล้ว setState เมื่อเสร็จ

ขาด mounted — ระหว่างที่ loadSession() รออยู่ ถ้าจอถูกถอดออกไปก่อน setState จะถูกเรียกบน State ที่ตายแล้ว
ในทางปฏิบัติจอนี้เป็นจอแรกและไม่เคยถูกถอด จึงไม่เคยเกิด แต่ถ้าคัดลอกรูปแบบนี้ไปใช้กับจอย่อย จะเจอทันที ตรงกับที่ 4.13 เตือนไว้

3. การแข่งเวลาที่ซ่อนอยู่

สามฟังก์ชันนี้ทำงานต่อกันเป็นทอด

Future<void> _openLogin() async {
  await Navigator.push(
    context,
    MaterialPageRoute(
      builder: (_) => LoginScreen(
        onLoginSuccess: () {
          Navigator.pop(context);
          _loadUser();          // ← ไม่มี await และเรียกไม่ได้ด้วย
        },
      ),
    ),
  );
}

Future<bool> _requireLogin() async {
  if (_userId != null) return true;

  await _openLogin();
  return _userId != null;       // ← ตรงนี้ต้องหวังว่า _loadUser เสร็จแล้ว
}

ไล่ลำดับเวลาให้ครบ

ลำดับเกิดอะไร
1ผู้ใช้กดการ์ด → _requireLogin() เห็น _userId == null
2await _openLogin() → จอล็อกอินเปิด
3ผู้ใช้ล็อกอินสำเร็จ → onLoginSuccess ทำงาน
4Navigator.pop(context) — เริ่มปิดจอ
5_loadUser() เริ่มทำงาน แต่ไม่มีใครรอ
6await _openLogin() คืนค่าเมื่อจอปิดเสร็จ
7return _userId != null

ขั้นที่ 5 กับ 7 แข่งกัน ถ้าขั้น 5 ยังไม่เสร็จตอนถึงขั้น 7 ผู้ใช้จะล็อกอินสำเร็จแล้วแต่การ์ดไม่เปิด

ทำไมถึงไม่เคยเห็นบั๊กนี้ — เพราะขั้นที่ 6 ต้องรอแอนิเมชันปิดจอจบก่อน ซึ่งใช้เวลาราว 300 มิลลิวินาที ส่วน SharedPreferences ที่โหลดมาแล้วใช้เวลาไม่ถึงมิลลิวินาที
โค้ดนี้ทำงานถูกเพราะแอนิเมชันช้ากว่าการอ่านค่า ไม่ใช่เพราะเขียนให้ถูกลำดับ

วิธีแก้ที่ตรงเหตุ — ให้ค่ากลับมาทางเดียวแทนที่จะหวังลำดับ

Future<bool> _requireLogin() async {
  if (_userId != null) return true;

  await Navigator.push<void>(
    context,
    MaterialPageRoute(
      builder: (_) => LoginScreen(
        onLoginSuccess: () => Navigator.pop(context),
      ),
    ),
  );
  if (!mounted) return false;

  await _loadUser();      // โหลดหลังจอปิดแล้ว ไม่ต้องแข่งกับใคร
  return _userId != null;
}

ตอนนี้ลำดับถูกบังคับด้วย await ไม่ใช่ด้วยความบังเอิญ

4. เครื่องหมายอัศเจรีย์ที่กระจายอยู่

ทุกการ์ดในเมนูเขียนแบบเดียวกัน

onTap: () async {
  if (!await _requireLogin()) return;
  if (!context.mounted) return;
  Navigator.push(
    context,
    MaterialPageRoute(
      builder: (_) => ReceivingMenuScreen(
        userId: _userId!,
        fullName: _fullName!,     // ← ตรงนี้
      ),
    ),
  );
},

_requireLogin() รับประกันว่า _userId ไม่เป็น null แต่ไม่ได้รับประกัน _fullName

ค่าทั้งสามมาจาก SharedPreferences คนละคีย์

return AuthSession(
  userId: prefs.getString('userId'),
  fullName: prefs.getString('fullName'),
  role: prefs.getString('role'),
);

ถ้ามี userId แต่ไม่มี fullName — เช่นเวอร์ชันเก่าเคยเก็บไม่ครบ หรือเขียนไม่จบ — แอปจะพังตรง _fullName!

และมีอีกจุดที่แรงกว่า

child: Text(
  (_fullName ?? '?')[0],    // ← ป้องกัน null แล้ว แต่ไม่ได้ป้องกันสตริงว่าง
  ...

ถ้า _fullName เป็น '' จะได้ RangeError แทน

ค่าผล
'สมชาย ใจดี'ได้ 'ส'
nullได้ '?'
''RangeError

เขียนให้ครอบทั้งสองกรณีได้ในบรรทัดเดียว

final initial = (_fullName ?? '').trim();
Text(initial.isEmpty ? '?' : initial[0]);
รูปแบบ “ป้องกัน null แต่ลืมสตริงว่าง” เจอบ่อยมากในโค้ดจริง เพราะ Dart บังคับให้คิดเรื่อง null แต่ไม่บังคับให้คิดเรื่องความยาว — ตรงกับที่ D1.9 สอนเรื่องความต่างของ “ไม่มีค่า” กับ “มีค่าแต่ว่าง”

6. การนำทาง

ทั้งจอนี้ใช้ Navigator.push(MaterialPageRoute(builder: ...)) อย่างเดียว ไม่มี named route

วิธีนี้named route (5.4)
ส่งค่าเข้าจอผ่านตัวสร้าง มีชนิดกำกับผ่าน arguments เป็น Object?
คอมไพเลอร์ช่วยช่วยเต็มที่ไม่ช่วย ต้อง cast เอง

สำหรับแอปที่ทุกจอเปิดจากเมนูเดียวและไม่มีลิงก์จากภายนอก การส่งผ่านตัวสร้างชนะเรื่องความปลอดภัยของชนิด

ราคาที่จ่ายอยู่ในไฟล์นี้เอง — home_screen.dart ต้อง import ทุกจอปลายทาง รวม 12 บรรทัด import และเป็นเหตุผลหนึ่งที่ไฟล์ยาว 553 บรรทัด

ทีมอยากให้แอปจำการล็อกอินไว้ข้ามวัน แต่หมดอายุหลัง 12 ชั่วโมง

  1. ต้องแก้ไฟล์ไหนบ้าง

ข้อ 2 — ตรวจที่ไหน

จุดเหมาะไหม
ใน loadSession()ใช่ — ทุกคนที่ถามว่าใครล็อกอินอยู่ได้คำตอบที่ผ่านการตรวจแล้ว
ใน HomeScreen.initStateไม่ — ถ้ามีจอไหนโหลดเซสชันเองจะข้ามการตรวจ
ตอนกดการ์ดไม่ — การ์ดมี 8 ใบ ต้องเขียนซ้ำ 8 ที่
ตัวจับเวลาที่นับถอยหลังไม่ — แอปถูกปิดไปแล้วตัวจับเวลาก็หายไปด้วย

หลักการเดียวกับ 9.6: ตรวจที่ขอบเขตที่ทุกคนต้องผ่าน ไม่ใช่ที่ผู้เรียกแต่ละราย

ข้อ 3 — หมดอายุกลางงาน

นี่คือข้อที่สำคัญที่สุดและมักถูกมองข้าม

ทางเลือกผลกับผู้ใช้
เด้งกลับหน้าหลักทันทีเสียงานที่สแกนมาทั้งหมด — ผิดข้อ 20 ของ 15.11
ปล่อยให้ทำต่อจนจบงาน แล้วค่อยหมดอายุปลอดภัยกับผู้ใช้ แต่เซสชันยืดเกิน 12 ชั่วโมงจริง
ตรวจเฉพาะตอนกลับมาหน้าหลักงานที่ทำอยู่ไม่ถูกขัด

ทางที่สามตรงกับโครงของแอปนี้พอดี เพราะ _loadUser() ถูกเรียกจาก HomeScreen อยู่แล้ว และจอโดเมนรับ userId ไปเป็นค่าสำเนาตอนเปิด — เซสชันหมดอายุระหว่างทางจึงไม่กระทบจอที่เปิดค้างอยู่

ต้องเพิ่มอีกอย่างคือเรียก _loadUser() ตอนกลับมาที่หน้าหลัก ซึ่งตอนนี้ยังไม่ทำ

Navigator.push(...).then((_) {
  if (mounted) _loadUser();     // กลับจากจอโดเมน → ตรวจเซสชันใหม่
});

สิ่งที่ต้องบอกผู้ใช้: เมื่อเซสชันหมดอายุ ข้อความควรบอกว่า “หมดเวลาการใช้งาน กรุณาเข้าสู่ระบบใหม่” ไม่ใช่ปล่อยให้หน้าจอกลับไปเป็นสถานะยังไม่ล็อกอินเฉย ๆ ซึ่งผู้ใช้จะคิดว่าแอปพัง

9. สรุป

  • HomeScreen ถือ state ผู้ใช้ทั้งแอปไว้ในสามช่อง แล้วส่งต่อผ่านตัวสร้าง
  • _requireLogin() ทำงานถูกเพราะแอนิเมชันช้ากว่าการอ่านค่า ไม่ใช่เพราะบังคับลำดับไว้
  • _fullName! ไม่ได้ถูกรับประกันโดย _requireLogin()
  • (_fullName ?? '?')[0] ป้องกัน null แต่ไม่ป้องกันสตริงว่าง
  • เมนูซ่อนกดค้างที่รูปผู้ใช้ ทำงานได้ทั้งใน release
  • _role ถูกแสดงแต่ยังไม่ถูกใช้ตัดสินใจอะไร
  • ไม่มี named route — ปลอดภัยเรื่องชนิด แต่ทำให้ไฟล์นี้ต้อง import ทุกจอ

บทหน้า: จอล็อกอินที่ไม่ได้คุยกับเซิร์ฟเวอร์เลย และเซสชันที่ลบเกินความจำเป็น
