class PartThumbnail extends StatelessWidget {
  final String? imageUrl;
  final double size;

  const PartThumbnail({super.key, this.imageUrl, this.size = 40});

  @override
  Widget build(BuildContext context) {
    final hasImg = imageUrl != null && imageUrl!.isNotEmpty;

    return ClipRRect(
      borderRadius: BorderRadius.circular(8),
      child: Container(
        width: size,
        height: size,
        color: Colors.grey.shade100,
        child: hasImg
            ? FutureBuilder<String>(
                future: ApiService().getImageFullUrl(imageUrl!),   // ← ปัญหาที่ 1
                builder: (_, snap) {
                  if (!snap.hasData) {
                    return const Center(
                      child: SizedBox(
                        width: 16, height: 16,
                        child: CircularProgressIndicator(strokeWidth: 2),
                      ),
                    );
                  }
                  return Image.network(
                    snap.data!,                                    // ← ปัญหาที่ 2
                    fit: BoxFit.cover,
                    errorBuilder: (_, __, ___) => Icon(...),
                  );
                },
              )
            : Icon(MdiIcons.packageVariantClosed, ...),
      ),
    );
  }
}

3. ปัญหาที่ 1 — FutureBuilder อยู่ใน build

future: ApiService().getImageFullUrl(imageUrl!),

บรรทัดนี้สร้าง Future ใหม่ทุกครั้งที่ build() ทำงาน และ FutureBuilder ที่ได้ Future ตัวใหม่จะกลับไปเริ่มที่สถานะ “ยังไม่มีข้อมูล” เสมอ

ผลคือวงกลมหมุนกระพริบทุกครั้งที่ widget ถูกวาดใหม่

เมื่อbuild ทำงานไหมผู้ใช้เห็น
เลื่อนรายการขึ้นลงทำ ทุกครั้งที่แถวเข้าจอวงกลมแวบแล้วรูปโผล่
พิมพ์ค้นหาทีละตัวทำ ทุกตัวอักษรทั้งรายการกระพริบ
setState ของจอทำ ทั้งรายการกระพริบพร้อมกันหมด

ตรงกับกับดักที่ 13.10 อธิบายไว้ตรง ๆ

ที่ตลกร้ายคือ getImageFullUrl ไม่จำเป็นต้องเป็น async ด้วยซ้ำ

Future<String> getImageFullUrl(String relativePath) async {
  final base = await ApiService._resolveBase();
  final serverBase = base.replaceAll('/api', '');
  return '$serverBase$relativePath';
}

เป็น async เพราะ _resolveBase() เป็น async แต่หลังจากคำขอแรก _cachedBase มีค่าแล้ว (16.6) ฟังก์ชันนี้จึงคืนค่าทันทีเกือบทุกครั้ง — เสียค่าใช้จ่ายของ FutureBuilder ทั้งหมดเพื่อรอสิ่งที่มีอยู่แล้ว

แก้ได้โดยให้ผู้เรียกส่ง URL เต็มมาเลย

แต่แบบนี้ผู้เรียก 12 จุดต้องประกอบ URL เอง วิธีที่กระทบน้อยกว่าคือทำให้การประกอบเป็นแบบไม่ต้องรอ

// ใน ApiService — คืนทันทีถ้ารู้ base แล้ว
static String? get cachedServerBase =>
    _cachedBase?.replaceAll('/api', '');
// ใน PartThumbnail
final base = ApiService.cachedServerBase;
if (base == null) return Icon(...);          // ยังไม่รู้ base — แสดงไอคอนไปก่อน
return Image.network('$base$imageUrl', ...);

ไม่มี FutureBuilder ไม่มีการกระพริบ และกรณีที่ยังไม่รู้ base เกิดได้เฉพาะก่อนคำขอแรกซึ่งตอนนั้นยังไม่มีรายการให้แสดงอยู่แล้ว

ข้อควรระวังของ replaceAll('/api', '') — แทนที่ทุกที่ที่เจอ ไม่ใช่แค่ท้ายสุด ถ้าวันหนึ่ง base เป็น http://api-server:5000/api จะได้ http://-server:5000 ซึ่งผิด
ที่ถูกคือตัดเฉพาะท้าย: base.substring(0, base.length - 4) หรือใช้ Uri.resolve ตามที่ 13.6 สอน

8. สรุป

  • PartThumbnail 61 บรรทัด ถูกใช้ 12 จุด ทุกที่ขอขนาด 36–56 พิกเซล
  • ความถูกต้องครบ — ตรวจสตริงว่าง มี errorBuilder มีไอคอนสำรอง
  • FutureBuilder ใน build สร้าง Future ใหม่ทุกครั้ง → รูปกระพริบตอนเลื่อนและตอนพิมพ์ค้นหา
  • getImageFullUrl ไม่จำเป็นต้อง async หลังจากรู้ base แล้ว
  • ไม่มี cacheWidth → รูป 1024×1024 กิน 4.2 MB ต่อรูป ต่างจากที่ควรเป็น 84 เท่า
  • 50 แถวคือ 210 MB ซึ่งเป็นจุดที่ระบบเริ่มปิดแอปทิ้ง
  • cacheWidth ต้องคูณด้วยความละเอียดจอ ไม่งั้นรูปเบลอ
  • จอ part_image_screen โหลดสินค้าทั้งหมดแล้วกรองในเครื่อง — ดีจนถึงหลักพัน
