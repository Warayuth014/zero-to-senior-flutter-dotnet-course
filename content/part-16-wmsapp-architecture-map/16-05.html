Future<String?> getCustomBase() async {
  final prefs = await SharedPreferences.getInstance();
  final host = prefs.getString(_keyHost);
  if (host == null || host.trim().isEmpty) return null;
  final port = prefs.getInt(_keyPort) ?? defaultPort;
  final protocol = prefs.getString(_keyProtocol) ?? defaultProtocol;
  return '$protocol://$host:$port';
}

null แปลว่า “ผู้ใช้ยังไม่ได้ตั้งค่า” ไม่ใช่ “เกิดข้อผิดพลาด” — และตรวจทั้ง null และสตริงว่าง ต่างจาก _fullName ใน 16.3 ที่ตรวจแค่ null

3. ตัวตรวจค่าที่กรอก — ฟังก์ชันบริสุทธิ์

String? validateHost(String host) {
  final trimmed = host.trim();
  if (trimmed.isEmpty) return 'กรุณาใส่ IP หรือชื่อเครื่อง';
  if (trimmed.contains('://') || trimmed.toLowerCase().startsWith('http')) {
    return 'ใส่เฉพาะ IP/hostname ไม่ต้องมี http:// (เลือก Protocol ด้านล่างแทน)';
  }
  if (trimmed.contains(' ')) return 'ห้ามมีช่องว่าง';
  if (trimmed.contains(':')) return 'พอร์ตให้ใส่ในช่อง Port แยกต่างหาก';
  if (trimmed.contains('/')) return 'ห้ามมี / ในช่องนี้';
  return null;
}

สองฟังก์ชันนี้อยู่นอกคลาส ไม่แตะ SharedPreferences ไม่แตะเครือข่าย รับสตริงคืนสตริง — เทสต์ได้ทันที

test('ใส่ http:// มาในช่อง host ต้องบอกให้ไปเลือก protocol แทน', () {
  expect(validateHost('http://192.168.1.50'), contains('Protocol'));
});

test('host ที่ถูกต้องผ่าน', () {
  expect(validateHost('192.168.1.50'), isNull);
  expect(validateHost('  wms-server  '), isNull);
});

ที่สำคัญกว่าคือข้อความบอกว่าค่าควรไปอยู่ที่ไหน ไม่ได้บอกแค่ว่าผิด

ผู้ใช้พิมพ์ข้อความที่ได้ผู้ใช้รู้ว่าต้องทำอะไร
http://192.168.1.50ไปเลือก Protocol ด้านล่างแทนรู้
192.168.1.50:5000พอร์ตให้ใส่ในช่อง Port แยกต่างหากรู้
192.168.1.50/apiห้ามมี / ในช่องนี้รู้

สามเคสนี้คือสิ่งที่ผู้ใช้พิมพ์จริงเพราะเคยเห็น URL เต็ม ๆ มาก่อน — คนเขียนดักไว้ตรงจุด ตรงกับหลักของ 10.7

ราว 10–12 ข้อ ครอบสี่กลุ่ม: ค่าที่ถูก, ค่าว่าง, ค่าที่ผู้ใช้พิมพ์ผิดแบบที่เจอจริง, และขอบเขตตัวเลข

  • ชื่อคีย์เป็นค่าคงที่ ค่าเริ่มต้นอยู่ที่เดียว คืน record
  • ตัวตรวจค่าเป็นฟังก์ชันบริสุทธิ์ และข้อความบอกว่าค่าควรไปอยู่ช่องไหน
  • probeConnection อ่าน payload ไม่ใช่แค่รหัสสถานะ — กัน fallback ที่ตอบ 200 ให้ทุก path
  • รับ client เข้ามาได้ และปิดเฉพาะตัวที่ตัวเองสร้าง
  • _save() เรียก resetBaseUrl() ทำให้ค่าใหม่มีผลทันที
  • ยังขาด: หลายเซิร์ฟเวอร์, คำเตือนตอนสลับ, เทสต์, และการแสดงที่อยู่ในจออื่น

บทหน้า: ApiService — singleton ที่ถูกสร้างขึ้นใหม่ 26 ครั้งในโปรเจกต์ และ cache ที่ทำให้เกิดผลข้างเคียงที่คาดไม่ถึง
