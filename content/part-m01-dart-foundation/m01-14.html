1. ปัญหาที่บทนี้แก้

ใน D1.13 เราค้างคำถามไว้ว่า เมื่องานล้มเหลวควรบอกผู้เรียกอย่างไร การคืน null บอกได้แค่ “ไม่มีค่า” แต่บอกไม่ได้ว่า

  • บาร์โค้ดว่างเปล่า
  • ไม่พบพาเลทนี้ในระบบ
  • ของในสต็อกไม่พอ
  • เชื่อมต่อ server ไม่ได้

ทั้งสี่กรณีต้องการการตอบสนองคนละแบบจากผู้ใช้ ถ้าเราตอบเหมือนกันหมด พนักงานจะไม่รู้ว่าควรสแกนใหม่ เดินไปหาของ หรือเรียกช่างไอที

2. คำศัพท์ที่ต้องรู้ก่อน

คำความหมาย
exceptionวัตถุที่บอกว่า “งานนี้ทำไม่สำเร็จ เพราะเหตุนี้”

finallyบล็อกที่ทำงานเสมอ ไม่ว่าจะสำเร็จหรือพัง
rethrowจัดการบางส่วนแล้วส่งต่อให้ชั้นบน
stack traceรายการว่าโปรแกรมเดินผ่านฟังก์ชันใดมาบ้างก่อนพัง

3. Mental model: สายด่วนที่ข้ามหัวหน้าไปหาคนที่แก้ได้

เมื่อฟังก์ชันชั้นในเจอปัญหาที่ตัวเองแก้ไม่ได้ มันไม่ต้องคืนค่าประหลาด ๆ กลับไปให้คนเรียกเดา แต่ “โยน” ปัญหาขึ้นไปตามสายบังคับบัญชา จนกว่าจะเจอชั้นที่รู้ว่าต้องทำอะไร

ชั้นรู้อะไรทำอะไรกับข้อผิดพลาด
ฟังก์ชันที่ตรวจข้อมูลข้อมูลผิดตรงไหนโยนพร้อมเหตุผลที่ชัดเจน
ฟังก์ชันกลางกำลังทำงานอะไรอยู่ส่งผ่านไป หรือเพิ่มบริบทแล้วส่งต่อ
หน้าจอผู้ใช้กำลังทำอะไรดักไว้ แล้วแสดงข้อความที่ผู้ใช้ทำอะไรต่อได้
ประโยคเดียวที่ต้องจำ: จับข้อผิดพลาดเฉพาะในชั้นที่ “ตัดสินใจอะไรบางอย่างได้จริง” — ถ้าจับแล้วไม่รู้จะทำอะไรต่อ แสดงว่ายังไม่ใช่ชั้นของคุณ

4. สร้าง exception ของตัวเอง

class ScanException implements Exception {
  ScanException(this.message);
  final String message;
  @override
  String toString() => 'ScanException: $message';
}

class OutOfStockException implements Exception {
  OutOfStockException(this.code, this.available);
  final String code;
  final int available;   // พกข้อมูลที่คนแก้ปัญหาต้องใช้ไปด้วย
}

สังเกตว่าเราใช้ implements Exception ตามที่เรียนใน D1.8 — Exception ใน Dart เป็นเพียงสัญญาที่ไม่มีโค้ดอะไรมาให้ หน้าที่ของมันคือบอกคนอ่านว่า “class นี้มีไว้แทนความล้มเหลว”

จุดสำคัญคือ OutOfStockException พก available ติดตัวมาด้วย ทำให้หน้าจอบอกได้ว่า “เหลือ 3 ชิ้น” แทนที่จะบอกแค่ “ของไม่พอ” นี่คือความต่างระหว่าง exception ที่มีประโยชน์กับที่ไม่มี

6. finally: ทำความสะอาดที่ต้องเกิดขึ้นเสมอ

void runWithCleanup({required bool shouldFail}) {
  print('เปิดรอบทำงาน');
  try {
    if (shouldFail) throw ScanException('จำลองความผิดพลาด');
    print('  ทำงานสำเร็จ');
  } on ScanException catch (error) {
    print('  จับได้: ${error.message}');
  } finally {
    print('  ปิดรอบทำงาน (finally ทำงานเสมอ)');
  }
}

finally ใช้กับสิ่งที่ต้องเกิดขึ้นไม่ว่าอะไรจะเกิด เช่น ปิดสถานะกำลังโหลด ปลดล็อกปุ่ม หรือคืนทรัพยากร ถ้าเขียนโค้ดเหล่านี้ไว้ท้าย try เฉย ๆ มันจะถูกข้ามเมื่อเกิดข้อผิดพลาด แล้วปุ่มจะค้างอยู่ในสถานะกดไม่ได้ตลอดไป

7. Stack trace และ rethrow

try {
  pick('PAL-9999', 1);
} catch (error, stackTrace) {
  print('error = $error');
  print(stackTrace.toString().split('\n').first);
}

catch รับได้สองค่า ค่าที่สองคือ stack trace ซึ่งบอกว่าโปรแกรมเดินผ่านฟังก์ชันไหนมาบ้าง — บรรทัดแรกสุดคือจุดที่พังจริง ให้อ่านจากบนลงล่างเสมอ

void logThenRethrow() {
  try {
    pick('', 1);
  } on ScanException catch (error) {
    print('บันทึก log ไว้ก่อน: ${error.message}');
    rethrow;   // ส่งต่อพร้อม stack trace เดิม
  }
}
ใช้ rethrow ไม่ใช่ throw error — การเขียน throw error จะรีเซ็ต stack trace ทำให้เห็นแค่ว่าพังตรงจุดที่โยนใหม่ ไม่เห็นต้นตอจริง ซึ่งทำให้หาสาเหตุยากขึ้นมาก

มีสองแนวทางในการรายงานความล้มเหลว และทั้งสองถูกต้อง แต่ต้องเลือกอย่างมีกติกา

เหมาะกับความผิดปกติที่ไม่ควรเกิดความล้มเหลวที่คาดไว้แล้วและเกิดบ่อย

คำแนะนำที่ใช้ได้ทั่วไป: ความผิดปกติของระบบให้โยน ความล้มเหลวที่เป็นส่วนหนึ่งของงานให้คืนเป็นข้อมูล เช่น “เน็ตหลุด” ควรโยน ส่วน “ของไม่พอ” เป็นเรื่องปกติในคลังจึงควรคืนเป็นข้อมูลให้หน้าจอตัดสินใจ

สิ่งที่ห้ามทำคือผสมสองแบบโดยไม่มีกติกา จนผู้เรียกต้องทั้ง try/catch และตรวจสถานะไปพร้อมกัน

9. อย่ากลืนข้อผิดพลาด

// แย่ที่สุด: ปัญหาหายไปโดยไม่มีใครรู้
try {
  await loadTasks();
} catch (_) {}

// แย่เช่นกัน: ผู้ใช้แยกไม่ออกว่า "ไม่มีงาน" หรือ "server ล่ม"
try {
  return await loadTasks();
} catch (_) {
  return <Task>[];
}

รูปแบบที่สองอันตรายเป็นพิเศษ เพราะหน้าจอจะแสดง “ไม่มีงานในกะนี้” อย่างมั่นใจทั้งที่ระบบล่ม พนักงานก็จะกลับบ้าน ทั้งที่มีของรออยู่ — เป็นตัวอย่างชัดเจนของบั๊กที่โค้ดไม่พังแต่ธุรกิจเสียหาย

อีกเรื่องที่ต้องระวังคือ อย่าแสดงรายละเอียดทางเทคนิคให้ผู้ใช้เห็น ข้อความอย่าง token, คำสั่งฐานข้อมูล หรือ stack trace ควรอยู่ใน log เท่านั้น ส่วนผู้ใช้ควรเห็นข้อความที่บอกว่าต้องทำอะไรต่อ

10. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
บล็อก on X ไม่เคยทำงานมี catch แบบกว้างอยู่ก่อนหน้าเรียงจากเจาะจงไปกว้างเสมอ
ปุ่มค้างอยู่ในสถานะกดไม่ได้รีเซ็ตสถานะไว้ท้าย try ซึ่งถูกข้ามเมื่อพังย้ายไปไว้ใน finally
stack trace ชี้ไปที่ catch ของเราเองใช้ throw error แทน rethrowเปลี่ยนเป็น rethrow
ผู้ใช้เห็นข้อความอย่าง SocketException: Failed host lookupเอา error.toString() ไปแสดงตรง ๆแปลงเป็นข้อความที่ผู้ใช้เข้าใจก่อนแสดง
หน้าจอบอกว่าไม่มีข้อมูล ทั้งที่ระบบล่มดักข้อผิดพลาดแล้วคืนรายการว่างแยกสถานะ “ว่าง” ออกจาก “ล้มเหลว” (ทบทวน D1.10 ข้อ 9)
ดัก Exception แล้วแอปยังพังอยู่ดีบางความผิดพลาดเป็น Error ไม่ใช่ ExceptionError คือบั๊กของโปรแกรมเมอร์ ให้แก้โค้ด ไม่ใช่ดักไว้

ความต่างข้อสุดท้ายสำคัญ: Exception คือสิ่งที่คาดว่าจะเกิดได้ในสภาพแวดล้อมจริง เช่นเน็ตหลุด ส่วน Error เช่น TypeError หรือ RangeError แปลว่าโค้ดเราเขียนผิด ให้ไปแก้ต้นเหตุ ไม่ใช่ดักไว้แล้วทำเป็นไม่มีอะไรเกิดขึ้น

11. แบบฝึกหัด

ระบบต้องโหลดรายการงานจาก server และมีสี่กรณีที่ต้องแยกให้ออก

  1. โหลดสำเร็จแต่ไม่มีงาน
  2. ผู้ใช้หมดสิทธิ์ (session หมดอายุ)
  3. เชื่อมต่อไม่ได้ภายในเวลาที่กำหนด
  4. server ตอบมาแต่ข้อมูลผิดรูปแบบ

จงออกแบบว่าแต่ละกรณีควรโยนหรือคืนเป็นข้อมูล และหน้าจอควรทำอะไร

ดูแนวคำตอบและเหตุผล
กรณีวิธีรายงานหน้าจอทำอะไรเหตุผล
ไม่มีงานคืนรายการว่างตามปกติแสดง “ยังไม่มีงานในกะนี้”ไม่ใช่ความล้มเหลว เป็นผลลัพธ์ที่ถูกต้อง
หมดสิทธิ์โยน UnauthorizedExceptionล้าง session แล้วพากลับหน้าล็อกอินต้องการการกระทำเฉพาะที่ต่างจากทุกกรณี
เชื่อมต่อไม่ได้โยน NetworkExceptionแสดงข้อความพร้อมปุ่มลองใหม่แก้ได้ด้วยการลองใหม่ ไม่ต้องเปลี่ยนอะไร
ข้อมูลผิดรูปโยน ContractExceptionแสดงข้อความทั่วไป และบันทึก log รายละเอียดผู้ใช้แก้เองไม่ได้ ต้องให้ทีมพัฒนารู้
class UnauthorizedException implements Exception {}
class NetworkException implements Exception {}
class ContractException implements Exception {
  ContractException(this.detail);
  final String detail;
}

String handleLoad(void Function() load) {
  try {
    load();
    return 'โหลดสำเร็จ';
  } on UnauthorizedException {
    return 'เซสชันหมดอายุ กรุณาเข้าสู่ระบบใหม่';
  } on NetworkException {
    return 'เชื่อมต่อไม่ได้ กรุณาลองใหม่อีกครั้ง';
  } on ContractException catch (error) {
    // รายละเอียดเข้า log เท่านั้น ไม่แสดงให้ผู้ใช้
    print('[log] ข้อมูลผิดรูปแบบ: ${error.detail}');
    return 'ระบบขัดข้องชั่วคราว กรุณาแจ้งผู้ดูแล';
  }
}

จุดที่คนมักตอบผิด: หลายคนรวมกรณีที่ 1 เข้ากับความล้มเหลว โดยคิดว่า “ไม่มีข้อมูล = ผิดพลาด” แต่ในความจริง “กะนี้ยังไม่มีงาน” เป็นคำตอบที่ถูกต้องสมบูรณ์ ถ้าปฏิบัติกับมันเหมือนข้อผิดพลาด ผู้ใช้จะเห็นข้อความเตือนสีแดงทุกเช้าก่อนงานเข้า

จุดที่คนมักตอบผิดอีกข้อ: การเอาข้อความจาก ContractException ไปแสดงตรง ๆ ผู้ใช้ในคลังไม่ได้ประโยชน์อะไรจากคำว่า “field palletCode expected String but got int” — เขาต้องการรู้แค่ว่าควรทำอะไรต่อ

ลองต่อ: เพิ่มกรณีที่ห้า “ผู้ใช้กดยกเลิกระหว่างโหลด” แล้วคิดว่าควรจัดอยู่กลุ่มไหน — คำใบ้: นี่ไม่ใช่ความล้มเหลวเลย และไม่ควรแสดงข้อความอะไรทั้งสิ้น

12. สรุปท้ายบท

  • สร้าง exception ของตัวเองที่พกข้อมูลซึ่งคนแก้ปัญหาต้องใช้
  • จับด้วย on เรียงจากเจาะจงไปกว้าง
  • finally ใช้กับสิ่งที่ต้องเกิดขึ้นเสมอ เช่นปลดสถานะกำลังโหลด
  • ใช้ rethrow เพื่อรักษา stack trace เดิม
  • จับข้อผิดพลาดเฉพาะในชั้นที่ตัดสินใจอะไรได้จริง
  • ห้ามกลืนข้อผิดพลาดเป็นรายการว่าง เพราะผู้ใช้จะแยกไม่ออกจากกรณีไม่มีข้อมูล
  • Error คือบั๊กของเรา ให้แก้โค้ด ส่วน Exception คือสิ่งที่ต้องรับมือ
