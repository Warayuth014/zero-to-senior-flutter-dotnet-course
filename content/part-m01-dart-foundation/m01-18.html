1. ปัญหาที่บทนี้แก้

ทุกอย่างที่เขียนมาตั้งแต่ D1.1 อยู่ในไฟล์เดียว ซึ่งใช้ได้ดีกับโปรแกรมร้อยบรรทัด แต่แอปจริงอย่าง wms_absolute_mobile มีไฟล์นับร้อย ถ้าอยู่ไฟล์เดียวจะเกิดสามปัญหาทันที: หาโค้ดไม่เจอ, แก้พร้อมกันหลายคนไม่ได้, และทุกอย่างเข้าถึงกันได้หมดจนไม่มีขอบเขต

3. Mental model: หนึ่งไฟล์คือหนึ่งห้องที่มีประตู

แต่ละไฟล์เป็นห้องที่มีของอยู่ข้างใน ของที่ชื่อขึ้นต้นด้วย _ คือของที่ล็อกไว้ในห้อง ส่วนของอื่นวางไว้หน้าประตูให้ใครมาหยิบก็ได้ — แต่ต้องเดินมาที่ห้องนี้เอง ซึ่งก็คือการ import

ทบทวนจาก D1.7: _ ทำให้ private ต่อ ไฟล์ ไม่ใช่ต่อ class — บทนี้คือจุดที่ความจริงข้อนั้นเริ่มมีประโยชน์จริง เพราะเมื่อแยกไฟล์แล้ว _ จึงกลายเป็นกำแพงที่แท้จริง

5. import สี่รูปแบบที่ต้องใช้เป็น

// 1) ไฟล์ในโปรเจกต์เดียวกัน ใช้เส้นทางสัมพัทธ์
import 'warehouse/warehouse.dart';

// 2) ตั้งชื่อ namespace ด้วย as กันชื่อชนและบอกที่มา
import 'dart:math' as math;

// 3) เอามาเฉพาะที่ใช้ ด้วย show
import 'dart:convert' show jsonEncode;

// 4) เอามาทั้งหมดยกเว้นบางตัว ด้วย hide
import 'dart:convert' hide jsonDecode;
รูปแบบใช้เมื่อ
ธรรมดากรณีทั่วไป
asชื่อชนกัน หรืออยากให้อ่านออกว่าของมาจากไหน เช่น math.max
showต้องการแค่หนึ่งสองตัวจาก library ใหญ่
hideต้องการเกือบทั้งหมด ยกเว้นตัวที่ชนกับของเรา

นอกจากนี้ยังมีการ import จาก package ซึ่งขึ้นต้นด้วย package: เช่น package:http/http.dart สำหรับโค้ดที่ติดตั้งเพิ่ม และ dart: สำหรับสิ่งที่มากับภาษาเอง

6. private กลายเป็นกำแพงจริงเมื่อแยกไฟล์

// pallet.dart
class Pallet {
  Pallet._(this.code, this.quantity);     // constructor private

  factory Pallet.parse(String raw) {      // ทางเข้าเดียวที่เปิดไว้
    final normalized = _normalize(raw);
    if (!_isValidCode(normalized)) {
      throw FormatException('รหัสพาเลทไม่ถูกรูปแบบ: $raw');
    }
    return Pallet._(normalized, 0);
  }
  ...
}

String _normalize(String raw) => raw.trim().toUpperCase();   // ซ่อนสนิท
bool _isValidCode(String code) => code.startsWith('PAL-') && code.length == 8;

จากไฟล์อื่น Pallet._(...) และ _normalize(...) มองไม่เห็นเลยแม้จะ import แล้ว ผลคือ ไม่มีทางสร้าง Pallet ที่รหัสผิดรูปแบบได้เลย เพราะทางเข้าเดียวคือ parse ซึ่งตรวจให้เสมอ

นี่คือการยกระดับหลักการจาก D1.7 ขึ้นไปอีกขั้น — ตอนนั้นเราปกป้องข้อมูลจากคนที่อยู่ใน class อื่น ตอนนี้เราปกป้องมันจากทั้งโปรเจกต์

7. export: หน้าร้านของโฟลเดอร์

// warehouse/warehouse.dart
export 'location.dart';
export 'pallet.dart';

ไฟล์นี้ไม่มีโค้ดของตัวเองเลย มีหน้าที่เดียวคือรวมของจากไฟล์ข้างในแล้วเปิดให้คนนอกใช้ ผู้ใช้จึงเขียน import 'warehouse/warehouse.dart' บรรทัดเดียวแทนที่จะต้อง import ทีละไฟล์

ข้อดีข้อเสีย
ผู้ใช้เขียน import สั้นลงมองไม่เห็นว่าจริง ๆ แล้วใช้ของจากไฟล์ไหน
เราคุมได้ว่าจะเปิดอะไรบ้างถ้า export ทุกอย่าง ชื่อจะชนกันง่ายขึ้น
ย้ายไฟล์ข้างในได้โดยผู้ใช้ไม่กระทบเสี่ยงดึงโค้ดที่ไม่ได้ใช้เข้ามา

คำแนะนำ: ใช้ export เมื่อโฟลเดอร์นั้นเป็น “หน่วยที่สมบูรณ์” ที่คนนอกควรมองเป็นก้อนเดียว ไม่ใช่ใช้กับทุกโฟลเดอร์โดยอัตโนมัติ

8. ทิศทางการพึ่งพา

เรื่องนี้สำคัญกว่าไวยากรณ์ทั้งหมดในบทนี้ กฎมีข้อเดียว: การพึ่งพาต้องไหลไปทางเดียว ห้ามวนกลับ

d1_18_library.dart  ──import──>  warehouse/warehouse.dart
                                        │
                                        ├──> pallet.dart
                                        └──> location.dart

สังเกตว่า warehouse/ ไม่รู้จักไฟล์ที่เรียกใช้มันเลย ถ้าวันหนึ่ง pallet.dart ต้อง import กลับมาที่ไฟล์บนสุด นั่นคือสัญญาณว่าแบ่งหน้าที่ผิด

อาการแปลว่าทำอย่างไร
สองไฟล์ import กันไปมาหน้าที่ปนกันแยกส่วนที่ใช้ร่วมกันออกเป็นไฟล์ที่สาม
ไฟล์ model ต้อง import ไฟล์หน้าจอทิศทางกลับด้านให้หน้าจอเป็นฝ่ายรู้จัก model ไม่ใช่กลับกัน
ต้อง import สิบไฟล์เพื่อใช้ของชิ้นเดียวของชิ้นนั้นผูกกับทุกอย่างเกินไปลดสิ่งที่มันต้องรู้จัก

เมื่อถึง Part 9 เรื่องสถาปัตยกรรม กฎข้อนี้จะกลายเป็นเครื่องมือหลักในการตัดสินว่าโค้ดควรอยู่โฟลเดอร์ไหน

9. part: เมื่อหลายไฟล์ต้องเป็น library เดียวกัน

มีอีกกลไกหนึ่งที่จะเจอในโค้ดจริง คือการรวมหลายไฟล์เป็น library เดียว

// ไฟล์หลัก
part 'model.g.dart';

// ไฟล์ที่ถูกรวม
part of 'model.dart';

ผลคือทุกไฟล์ที่เป็น part ของกันและกันจะเห็นของ private ของกันหมด เหมือนอยู่ไฟล์เดียวกัน

อย่าใช้ part เพียงเพื่อเข้าถึงของ private ของไฟล์อื่น เพราะมันขยายขอบเขตที่ทุกอย่างเข้าถึงกันได้ ทำให้กำแพงที่เราสร้างไว้หายไป — การใช้งานที่ถูกต้องส่วนใหญ่คือให้เครื่องมือสร้างโค้ดอัตโนมัติเขียนไฟล์ .g.dart ซึ่งจะเจอจริงใน Part 30

10. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
Target of URI doesn't existเส้นทางไฟล์ผิดตรวจเส้นทางเทียบกับไฟล์ปัจจุบัน
The name 'X' is defined in the libraries ... and ...สอง library มีชื่อชนกันใช้ as หรือ show/hide
เรียก method ที่มีอยู่แน่ ๆ ไม่ได้ชื่อขึ้นต้นด้วย _ จึงมองไม่เห็นข้ามไฟล์ถ้าตั้งใจให้ใช้ได้ ให้เอา _ ออก
แก้ไฟล์ .g.dart แล้วการแก้หายไปไฟล์นั้นถูกสร้างอัตโนมัติแก้ไฟล์ต้นทางแล้วสั่งสร้างใหม่
import ไม่ได้เพราะวนกันทิศทางการพึ่งพาผิดแยกส่วนกลางออกเป็นไฟล์ที่สาม
โฟลเดอร์มี barrel file แต่ยังต้อง import ทีละไฟล์ลืมใส่ export ให้ครบเพิ่มบรรทัด export ที่ขาด

11. แบบฝึกหัด

จัดระเบียบโค้ดจากบทก่อน ๆ ใหม่ตามโครงนี้

my_wms/
├── main.dart
└── domain/
    ├── domain.dart        <- barrel
    ├── task_type.dart     <- enum จาก D1.10
    └── warehouse_task.dart
  1. warehouse_task.dart ต้องสร้าง object ได้ทางเดียวคือผ่านการตรวจข้อมูล
  2. ฟังก์ชันช่วยตรวจข้อมูลต้องมองไม่เห็นจากไฟล์อื่น
  3. main.dart ต้อง import แค่ไฟล์เดียว
  4. ไฟล์ใน domain/ ห้าม import main.dart
ดูแนวคำตอบและเหตุผล
// domain/task_type.dart
enum TaskType {
  putaway('PUTAWAY', 'จัดเก็บ'),
  retrieve('RETRIEVE', 'หยิบออก');

  const TaskType(this.wireValue, this.label);
  final String wireValue;
  final String label;

  static TaskType? tryFromWire(String raw) {
    for (final type in values) {
      if (type.wireValue == raw.toUpperCase()) return type;
    }
    return null;
  }
}
// domain/warehouse_task.dart
import 'task_type.dart';

class WarehouseTask {
  WarehouseTask._(this.id, this.type, this.quantity);

  factory WarehouseTask.create({
    required String id,
    required String type,
    required int quantity,
  }) {
    final parsedType = TaskType.tryFromWire(type);
    if (parsedType == null) {
      throw FormatException('ไม่รู้จักงานชนิด "$type"');
    }
    if (!_isValidId(id)) {
      throw FormatException('รหัสงานไม่ถูกรูปแบบ: $id');
    }
    if (quantity <= 0) {
      throw FormatException('จำนวนต้องมากกว่า 0');
    }
    return WarehouseTask._(id, parsedType, quantity);
  }

  final String id;
  final TaskType type;
  final int quantity;

  @override
  String toString() => '[$id] ${type.label} x$quantity';
}

bool _isValidId(String id) => id.startsWith('T-') && id.length >= 4;
// domain/domain.dart
export 'task_type.dart';
export 'warehouse_task.dart';
// main.dart
import 'domain/domain.dart';

void main() {
  print(WarehouseTask.create(id: 'T-01', type: 'putaway', quantity: 5));
  try {
    WarehouseTask.create(id: 'X1', type: 'putaway', quantity: 5);
  } on FormatException catch (error) {
    print('ถูกปฏิเสธ: ${error.message}');
  }
}

ทำไม _isValidId ต้องอยู่นอก class: เพราะมันไม่ต้องใช้ข้อมูลของ object ใด ๆ การวางไว้ระดับไฟล์ทำให้เห็นชัดว่าเป็นเครื่องมือภายในของไฟล์นี้ล้วน ๆ และ _ ทำให้ไฟล์อื่นเรียกไม่ได้เลย

ทำไม tryFromWire คืน TaskType? ไม่โยน error: เพราะ task_type.dart ไม่ควรตัดสินว่าค่าที่ไม่รู้จักคือความผิดพลาดร้ายแรงแค่ไหน — หน้าที่นั้นเป็นของผู้ใช้ ในที่นี้คือ WarehouseTask.create ซึ่งเลือกโยน FormatException ตามหลักที่เรียนใน D1.14

ทิศทางการพึ่งพาในคำตอบนี้: main.dart → domain.dart → warehouse_task.dart → task_type.dart ไหลไปทางเดียวตลอด ไม่มีลูกศรย้อนกลับ ถ้าวันหนึ่ง task_type.dart ต้องรู้จัก WarehouseTask แสดงว่าโครงเริ่มผิด และคำตอบมักคือย้ายสิ่งที่ทั้งคู่ต้องใช้ออกมาเป็นไฟล์ที่สาม

ลองต่อ: ลองเรียก WarehouseTask._('T-01', TaskType.putaway, 5) จาก main.dart — analyzer จะปฏิเสธ นั่นคือหลักฐานว่ากำแพงทำงานจริง

12. สรุปท้ายบท

  • หนึ่งไฟล์คือหนึ่ง library ซึ่งเป็นหน่วยของขอบเขตใน Dart
  • _ กลายเป็นกำแพงจริงเมื่อแยกไฟล์แล้ว
  • import มีสี่รูปแบบ: ธรรมดา, as, show, hide
  • export ทำให้โฟลเดอร์มีหน้าร้านเดียว แต่อย่าใช้กับทุกโฟลเดอร์
  • การพึ่งพาต้องไหลทางเดียว ห้ามวนกลับ
  • อย่าใช้ part เพียงเพื่อทะลุกำแพง private
