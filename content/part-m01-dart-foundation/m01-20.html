1. สิ่งที่เราจะสร้าง

โปรแกรมรับใบงานจากคลัง ซึ่งทำหกขั้นตอน

  4. สรุปจำนวนตามชนิดงาน
  5. ส่งงานขึ้น server ทีละใบแบบ async
  6. พิสูจน์ทุกอย่างด้วยเทสต์

ยังไม่มี Flutter ไม่มีหน้าจอ ไม่มีเครือข่ายจริง — แต่โครงสร้างเหมือนที่แอปจริงใช้ทุกประการ

4. ด่านตรวจ: แปลงข้อมูลดิบเป็น model

factory WarehouseTask.fromJson(Map<String, dynamic> json) {
  final id = json['id'];
  final type = json['type'];
  final palletCode = json['palletCode'];
  if (id is! String || type is! String || palletCode is! String) {
    throw TaskFormatException('ฟิลด์ id/type/palletCode ต้องเป็นข้อความ');
  }

  final rawQuantity = json['quantity'];
  final quantity = switch (rawQuantity) {
    final int value => value,
    final String value when int.tryParse(value) != null => int.parse(value),
    _ => throw TaskFormatException('quantity ไม่ใช่จำนวนเต็ม: $rawQuantity'),
  };
  if (quantity <= 0) {
    throw TaskFormatException('quantity ต้องมากกว่า 0 (ได้ $quantity)');
  }

  return WarehouseTask(
    id: id,
    type: TaskType.fromWire(type),
    palletCode: palletCode.trim().toUpperCase(),
    quantity: quantity,
    location: json['location'] as String?,
  );
}

สังเกตว่า switch ทำงานสามอย่างในก้อนเดียว: ยอมรับ int, ยอมรับข้อความที่แปลงได้, และปฏิเสธที่เหลือ — ตรงกับหลัก “ยืดหยุ่นกับความต่างที่ไม่สำคัญ เข้มงวดกับข้อมูลที่ผิดจริง” จาก D1.19

และสังเกตว่าการตรวจ quantity <= 0 อยู่ แยกออกมา จาก switch เพราะเป็นกฎทางธุรกิจ ไม่ใช่การตรวจรูปร่าง — ตรงกับที่เตือนไว้ใน D1.12 ว่า pattern ไม่ได้ตรวจกฎธุรกิจให้

5. นโยบายแถวเสีย

typedef ParseReport = ({List<WarehouseTask> tasks, List<String> errors});

ParseReport parseTasks(String rawJson) {
  final tasks = <WarehouseTask>[];
  final errors = <String>[];
  // ... วนทีละแถว: สำเร็จใส่ tasks, ล้มเหลวใส่ errors พร้อมเลขแถว
  return (tasks: tasks, errors: errors);

นี่คือทางเลือกที่สามจากตารางใน D1.19 — ข้ามแถวเสีย แต่รายงานเสมอ ผลคือผู้ใช้ทำงานต่อได้ และทีมพัฒนายังรู้ว่ามีอะไรผิดพลาด

ส่วน JSON ที่เสียทั้งก้อนถือเป็นคนละเรื่อง จึงคืนรายงานที่มี error เดียวและไม่มีงานเลย

6. ส่งงานแบบ async พร้อม callback

Future<int> dispatchAll(
  List<WarehouseTask> tasks, {
  required Future<bool> Function(WarehouseTask task) send,
  void Function(WarehouseTask task, bool accepted)? onResult,
}) async {
  var accepted = 0;
  for (final task in tasks) {
    final ok = await send(task);
    if (ok) accepted++;
    onResult?.call(task, ok);
  }
  return accepted;
}

จุดออกแบบที่สำคัญที่สุดของฟังก์ชันนี้คือ มันไม่รู้จักเครือข่ายเลย — วิธีส่งถูกส่งเข้ามาเป็น callback ตามหลัก D1.11 ผลคือ

  • ตอนใช้งานจริง ส่ง callback ที่ยิง HTTP
  • ตอนเขียนเทสต์ ส่ง callback ที่คืนคำตอบที่เรากำหนดเอง

และสังเกตว่าใช้ for กับ await ทีละใบ ไม่ใช่ Future.wait — ตั้งใจ เพราะนี่เป็นคำสั่งที่เปลี่ยนข้อมูลบน server ตามที่เตือนไว้ใน D1.15

9. ข้อผิดพลาดที่พบบ่อยตอนทำโปรเจกต์นี้

อาการสาเหตุวิธีแก้
แถวเสียแถวเดียวทำให้ไม่ได้งานเลยไม่ได้ครอบ try/catch รอบการแปลงแต่ละแถวย้าย try เข้าไปในลูป
ข้อมูลเสียหายไปเงียบ ๆดักข้อผิดพลาดแล้วไม่เก็บเหตุผลเก็บลง errors พร้อมเลขแถว

งานหลายใบถูกส่งซ้ำใช้ Future.wait กับคำสั่งที่เปลี่ยนข้อมูลวนส่งทีละใบด้วย for + await

10. แบบฝึกหัดปิด Part

ต่อยอดโปรเจกต์นี้ด้วยสามงาน ทำทีละข้อและรัน flutter analyze กับ flutter test ทุกครั้ง

  1. เพิ่มงานชนิดใหม่ COUNT (ตรวจนับ) ที่ไม่ต้องระบุช่องเก็บ
  2. เพิ่มฟังก์ชัน tasksNeedingAttention ที่คืนงานซึ่งจำนวนเกิน 50 ชิ้น หรือยังไม่ระบุช่องเก็บ
  3. เขียนเทสต์อย่างน้อยสองข้อสำหรับแต่ละงานข้างต้น รวมทั้งเทสต์กรณีข้อมูลเสีย
ดูแนวคำตอบและเหตุผล
// 1) เพิ่มค่าใน enum — สิ่งเดียวที่ต้องแก้ในไฟล์นี้
enum TaskType {
  putaway('PUTAWAY', 'จัดเก็บ'),
  retrieve('RETRIEVE', 'หยิบออก'),
  count('COUNT', 'ตรวจนับ');       // <- เพิ่มบรรทัดเดียว

  const TaskType(this.wireValue, this.label);
  final String wireValue;
  final String label;
  ...
}

// 2) ฟังก์ชันใหม่
List<WarehouseTask> tasksNeedingAttention(List<WarehouseTask> tasks) => tasks
    .where((task) => task.quantity > 50 || task.needsLocation)
    .toList();
// 3) เทสต์
test('รองรับงานชนิด COUNT และไม่บังคับช่องเก็บ', () {
  final report = parseTasks(
    '[{ "id": "T-10", "type": "count", "palletCode": "PAL-2001", "quantity": 30 }]',
  );

  expect(report.errors, isEmpty);
  expect(report.tasks.single.type, TaskType.count);
  expect(report.tasks.single.needsLocation, isFalse);
});

test('tasksNeedingAttention จับทั้งจำนวนมากและช่องเก็บที่ยังว่าง', () {
  final report = parseTasks('''
    [
      { "id": "T-01", "type": "PUTAWAY",  "palletCode": "PAL-1", "quantity": 99, "location": "A-01-02" },
      { "id": "T-02", "type": "PUTAWAY",  "palletCode": "PAL-2", "quantity": 5 },
      { "id": "T-03", "type": "RETRIEVE", "palletCode": "PAL-3", "quantity": 5 }
    ]
  ''');

  expect(
    tasksNeedingAttention(report.tasks).map((task) => task.id),
    <String>['T-01', 'T-02'],
  );
});

สิ่งที่ควรสังเกตมากที่สุดในข้อ 1: การเพิ่มงานชนิดใหม่แก้ไฟล์เดียวบรรทัดเดียว เพราะ fromWire วนอ่านจาก values อยู่แล้ว และ needsLocation คำนวณจาก type == TaskType.putaway — ถ้าตอนออกแบบเราเขียนเป็นชุด if กระจายไปตามไฟล์ต่าง ๆ การเพิ่มชนิดใหม่จะต้องไล่แก้หลายจุดและมีโอกาสลืม นี่คือผลตอบแทนของ enhanced enum จาก D1.10

ทำไมข้อ 2 ต้องคืน list ใหม่ ไม่แก้ของเดิม: ตามหลัก D1.5 — where(...).toList() สร้างก้อนใหม่ ทำให้ผู้เรียกมั่นใจว่า list ต้นทางไม่ถูกแตะต้อง

ทำไมเทสต์ข้อ 3 ต้องเช็คทั้งสองเงื่อนไขในเทสต์เดียว: เพราะสิ่งที่เราต้องการพิสูจน์คือ ตรรกะ “หรือ” ทำงานถูก ไม่ใช่แค่แต่ละเงื่อนไขทำงานได้ — สังเกตว่า T-03 ที่ไม่เข้าเงื่อนไขใดเลยต้องไม่ติดมาด้วย ซึ่งเป็นการพิสูจน์ฝั่งลบที่มักถูกลืม

ลองต่อ: เพิ่มกฎว่างาน COUNT ห้ามมีจำนวนเกิน 500 แล้วสังเกตว่าคุณควรใส่กฎนี้ตรงไหน — ใน fromJson (ปฏิเสธตั้งแต่ด่าน) หรือใน tasksNeedingAttention (เตือนแต่ยังรับ) คำตอบขึ้นกับว่าธุรกิจมองว่ามันคือข้อมูลผิดหรือข้อมูลที่ต้องตรวจสอบ

11. แบบทดสอบตัวเองก่อนไป Part 0

คุณพร้อมเข้าสู่ Flutter เมื่อทำสิ่งเหล่านี้ได้โดยไม่ต้องเปิดดูเฉลย

  • อ่านบรรทัดประกาศฟังก์ชันแล้วบอกได้ว่ารับอะไร คืนอะไร และอาจเป็น null ตรงไหน
  • อธิบายได้ว่าทำไม final list ยัง add ได้
  • บอกได้ว่าเมื่อไหร่ใช้ extends เมื่อไหร่ใช้ implements
  • เขียน switch บน enum ให้ analyzer ไม่บ่นได้
  • แยกได้ว่าปัญหาไหนแก้ด้วย await และปัญหาไหนต้องใช้ isolate
  • เขียน parser ที่แปลง JSON เป็น model และปฏิเสธข้อมูลเสียได้
  • รัน analyze กับ test แล้วแก้ต้นเหตุของคำเตือนได้

ถ้าข้อไหนยังไม่มั่นใจ ให้กลับไปรัน lab ของบทนั้นแล้วแก้โค้ดทดลองดู — การอ่านซ้ำช่วยได้น้อยกว่าการลองผิดแล้วอ่าน error

12. สรุปท้าย Dart Foundation

  • โปรแกรมที่ดีมีด่านตรวจชัดเจนระหว่างข้อมูลดิบกับข้อมูลที่เชื่อถือได้
  • ความล้มเหลวต้องมีเหตุผลติดตัวเสมอ และต้องไม่หายไปเงียบ ๆ
  • ตรรกะที่ไม่ผูกกับหน้าจอหรือเครือข่าย คือตรรกะที่เขียนเทสต์ได้
  • ชนิดข้อมูลที่ออกแบบดีทำให้สถานะที่ผิดเขียนออกมาไม่ได้เลย
  • เครื่องมือ format, analyze, test คือเพื่อนร่วมงานที่ตรวจให้ตลอดเวลา

ต่อจากนี้: Part 0 จะพาไปติดตั้งเครื่องมือและสร้างแอป Flutter ตัวแรก จากนั้น Part 1 จะเริ่มเรื่อง widget — และคุณจะพบว่าทุกอย่างที่เห็นในนั้น ไม่ว่าจะเป็น named parameter, callback, generic หรือ const ล้วนเป็นสิ่งที่คุณเรียนมาแล้วทั้งหมดใน 20 บทนี้
