3. Mental model: ด่านตรวจชายแดน

ให้คิดว่า parser คือด่านตรวจที่ชายแดน ข้อมูลทุกชิ้นที่เข้าประเทศต้องผ่านด่านนี้และได้รับเอกสารรับรอง (ชนิดที่ถูกต้อง) ก่อนจะเดินทางต่อเข้าไปในแอปได้

ข้อความ JSON  ──jsonDecode──>  Map/List ที่เป็น dynamic
                                        │
                                   [ ด่านตรวจ ]  <- ที่นี่ที่เดียวที่แตะ dynamic
                                        │
                                  model ที่มีชนิด  ──> หน้าจอ, ตรรกะ, ฐานข้อมูล
ประโยคเดียวที่ต้องจำ: dynamic ควรมีอายุสั้นที่สุดเท่าที่จะเป็นไปได้ — เกิดที่ jsonDecode และตายที่ parser ถ้ามันไหลไปถึงหน้าจอ แปลว่าด่านตรวจล้มเหลว

5. โครงสร้าง parser ที่ใช้ได้จริง

class Pallet {
  Pallet({
    required this.palletCode,
    required this.quantity,
    this.location,
    this.receivedAt,
  });

  // จุดเดียวในระบบที่ยอมแตะ dynamic
  factory Pallet.fromJson(Map<String, dynamic> json) {
    return Pallet(
      palletCode: _asString(json['palletCode']),
      quantity: _asInt(json['quantity']),
      location: json['location'] as String?,
      receivedAt: _asDate(json['receivedAt']),
    );
  }

  final String palletCode;    // บังคับ ต้องมี
  final int quantity;         // บังคับ ต้องมี
  final String? location;     // ไม่มีได้จริง
  final DateTime? receivedAt; // ไม่มีได้จริง
}

จุดที่ต้องคิดมากที่สุดคือ field ไหนบังคับ field ไหนว่างได้ ซึ่งเป็นการตัดสินใจเชิงธุรกิจ ไม่ใช่เชิงเทคนิค

คำถามที่ต้องตอบถ้าตอบว่าใช่
“พาเลทที่ไม่มีรหัส มีอยู่จริงไหม”ไม่มี → String บังคับ
“พาเลทที่ยังไม่ถูกจัดเก็บ มีอยู่จริงไหม”มี → String?
กับดักที่พบบ่อยที่สุด: การทำให้ทุก field เป็น nullable เพราะ “กลัว API ส่งมาไม่ครบ” ผลคือหน้าจอทุกหน้าต้องเขียน ?. และ ?? '' ทุกบรรทัด และไม่มีใครรู้ว่าค่าไหนหายจริงหรือหายเพราะบั๊ก — ตรวจให้จบที่ด่าน แล้วให้ส่วนที่เหลือของแอปทำงานกับข้อมูลที่เชื่อถือได้

6. ตัวช่วยที่ทนความต่างเล็กน้อย แต่ไม่กลืนข้อมูลเสีย

int _asInt(Object? value) {
  if (value is int) return value;
  if (value is num) return value.toInt();
  if (value is String) {
    final parsed = int.tryParse(value);
    if (parsed != null) return parsed;
  }
  throw FormatException('แปลงเป็นจำนวนเต็มไม่ได้: $value');
}

ฟังก์ชันนี้ทำสองอย่างที่ดูขัดกันแต่ถูกต้องทั้งคู่

  • ยืดหยุ่นกับความต่างที่ไม่สำคัญ — 12, 12.0 และ "12" ล้วนหมายถึงสิบสอง
  • เข้มงวดกับข้อมูลที่ผิดจริง — "มาก" หรือ null ต้องถูกปฏิเสธ ไม่ใช่แปลงเป็น 0 เงียบ ๆ

ความต่างระหว่างสองอย่างนี้สำคัญมาก ถ้าเขียน int.tryParse(...) ?? 0 ข้อมูลที่เสียจะกลายเป็นเลขศูนย์ที่ดูปกติ แล้วไหลเข้าไปในรายงานสต็อกโดยไม่มีใครรู้

10. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
type '_Map<String, dynamic>' is not a subtype of type 'List'cast ก่อนตรวจว่าเป็นอะไรตรวจด้วย is ก่อน แล้วโยน FormatException ที่อ่านรู้เรื่อง
ค่าเป็น null ทั้งที่ API ส่งมาแน่สะกดชื่อ key ผิด หรือตัวพิมพ์ไม่ตรงพิมพ์ json.keys ออกมาดูของจริง
ยอดรวมเพี้ยนโดยไม่มีข้อผิดพลาดใช้ ?? 0 กลบข้อมูลที่แปลงไม่ได้ให้ปฏิเสธและรายงานแทนที่จะแทนด้วยศูนย์
แอปพังใน build methodปล่อย dynamic ไหลไปถึงหน้าจอแปลงให้เสร็จที่ parser
เวลาที่แสดงคลาดไป 7 ชั่วโมงไม่ได้ตกลงเรื่องเขตเวลากำหนดนโยบายแล้วแปลงที่ด่านจุดเดียว
model มี field nullable ทุกตัวไม่ได้ตัดสินใจว่าอะไรบังคับไล่ถามทีละ field ว่า “ไม่มีค่านี้ แล้วยังมีความหมายไหม”

11. แบบฝึกหัด

API ส่งข้อมูลรายการงานมาในรูปแบบนี้

{
  "items": [
    { "taskId": "T-01", "status": "PENDING", "quantity": 5 },
    { "taskId": "T-02", "status": "pending", "quantity": "3" },
    { "taskId": "T-03", "status": "TELEPORT", "quantity": 1 }
  ],
  "totalCount": "3"
}

จงเขียน parser ที่

  1. ปฏิเสธทั้งชุดถ้า items ไม่ใช่ list
  2. รองรับ totalCount ที่เป็นทั้งตัวเลขและข้อความ
  3. แปลง status เป็น enum โดยไม่สนใจตัวพิมพ์เล็กใหญ่
  4. ข้ามแถวที่เสีย แต่รายงานว่าข้ามไปกี่แถวและเพราะอะไร
ดูแนวคำตอบและเหตุผล
import 'dart:convert';

enum TaskStatus {
  pending('PENDING'),
  done('DONE');

  const TaskStatus(this.wireValue);
  final String wireValue;

  static TaskStatus? tryFromWire(String raw) {
    for (final status in values) {
      if (status.wireValue == raw.trim().toUpperCase()) return status;
    }
    return null;
  }
}

class WmsTask {
  WmsTask({required this.taskId, required this.status, required this.quantity});

  factory WmsTask.fromJson(Map<String, Object?> json) {
    final id = json['taskId'];
    if (id is! String || id.isEmpty) {
      throw const FormatException('taskId ต้องเป็นข้อความที่ไม่ว่าง');
    }
    final rawStatus = json['status'];
    if (rawStatus is! String) {
      throw const FormatException('status ต้องเป็นข้อความ');
    }
    final status = TaskStatus.tryFromWire(rawStatus);
    if (status == null) {
      throw FormatException('ไม่รู้จักสถานะ "$rawStatus"');
    }
    return WmsTask(
      taskId: id,
      status: status,
      quantity: _asInt(json['quantity']),
    );
  }

  final String taskId;
  final TaskStatus status;
  final int quantity;

  @override
  String toString() => '$taskId ${status.name} x$quantity';
}

int _asInt(Object? value) {
  if (value is int) return value;
  if (value is num) return value.toInt();
  if (value is String) {
    final parsed = int.tryParse(value.trim());
    if (parsed != null) return parsed;
  }
  throw FormatException('แปลงเป็นจำนวนเต็มไม่ได้: $value');
}

typedef TaskPage = ({List<WmsTask> tasks, int totalCount, List<String> skipped});

TaskPage parseTaskPage(String body) {
  final root = jsonDecode(body);
  if (root is! Map<String, Object?>) {
    throw const FormatException('คาดว่าเป็น JSON object');
  }
  final items = root['items'];
  if (items is! List) {
    throw const FormatException('items ต้องเป็น list');   // ข้อ 1: ล้มทั้งชุด
  }

  final tasks = <WmsTask>[];
  final skipped = <String>[];

  for (var i = 0; i < items.length; i++) {
    final row = items[i];
    if (row is! Map<String, Object?>) {
      skipped.add('แถวที่ $i ไม่ใช่ object');
      continue;
    }
    try {
      tasks.add(WmsTask.fromJson(row));
    } on FormatException catch (error) {
      skipped.add('แถวที่ $i: ${error.message}');      // ข้อ 4: ข้ามแต่รายงาน
    }
  }

  return (tasks: tasks, totalCount: _asInt(root['totalCount']), skipped: skipped);
}

  T-01 pending x5
  T-02 pending x3
ข้ามไป 1 แถว:
  - แถวที่ 2: ไม่รู้จักสถานะ "TELEPORT"

ทำไม items ผิดชนิดจึงล้มทั้งชุด แต่แถวเสียกลับข้ามได้: เพราะทั้งสองบอกคนละเรื่อง — items ที่ไม่ใช่ list แปลว่าเราเข้าใจ contract ผิดทั้งหมด ไม่มีทางกู้ได้ ส่วนแถวเดียวที่มีสถานะแปลกแปลว่าข้อมูลแถวนั้นมีปัญหา ที่เหลือยังใช้งานได้ — นี่คือการใช้ตารางในหัวข้อ 8 จริง

ทำไมต้องเก็บ skipped ไว้ด้วย: เพราะถ้าข้ามเงียบ ๆ วันหนึ่งฝั่ง .NET เพิ่มสถานะใหม่แล้วงานครึ่งหนึ่งหายไปจากหน้าจอ โดยไม่มีใครรู้จนกว่าลูกค้าจะโทรมา การเก็บเหตุผลไว้ทำให้เราแสดงคำเตือนหรือส่ง log ได้

ทำไมใช้ record เป็นผลลัพธ์: เพราะเป็นค่าสามตัวที่คืนพร้อมกันและใช้เฉพาะจุดนี้ ตรงกับเกณฑ์ใน D1.12 พอดี ถ้าผลลัพธ์นี้ต้องเดินทางไปหลายไฟล์และมีกฎของตัวเอง จึงค่อยทำเป็น class

จุดที่คนมักตอบผิด: การเขียน quantity: json['quantity'] as int ตรง ๆ ซึ่งจะพังทั้งชุดเมื่อเจอ "3" ที่เป็นข้อความ — ความยืดหยุ่นตรงนี้ไม่ใช่การตามใจ API แต่เป็นการยอมรับความจริงว่าระบบสองฝั่งพัฒนาไม่พร้อมกัน

ลองต่อ: เพิ่มกรณีที่ totalCount หายไปทั้ง key แล้วตัดสินใจว่าควรล้มทั้งชุด หรือใช้ tasks.length แทน — และเขียนเหตุผลกำกับไว้ในโค้ด

12. สรุปท้ายบท

  • parser คือด่านตรวจระหว่างโลก JSON กับโลกที่มีชนิด
  • dynamic ต้องเกิดที่ jsonDecode และตายที่ parser
  • ตัดสินใจให้ชัดว่า field ไหนบังคับ field ไหนว่างได้จริง
