Future<T> คือผลลัพธ์หนึ่งค่าที่จะสำเร็จหรือผิดพลาดภายหลัง await หยุดเฉพาะ async function ปัจจุบัน ไม่ block UI isolate แต่โค้ดหลัง await จะรันต่อในอนาคตเมื่อโลกอาจเปลี่ยนแล้ว

1. async เปลี่ยน return type

Future<Task> loadTask(String id) async {
  final json = await client.get('/tasks/$id');
  return Task.fromJson(json);
}

function async คืน Future ทันที แล้ว complete ด้วย Task หรือ error ภายหลัง Error ก่อน/หลัง await จะกลายเป็น failed Future ให้ caller await/catch

2. Sequential กับ concurrent

// ต่อกัน: B ต้องรอ A
final profile = await loadProfile();
final tasks = await loadTasks(profile.siteId);

// อิสระ: เริ่มทั้งคู่ก่อนรอ
final results = await Future.wait([loadBadges(), loadAnnouncements()]);

อย่าใช้ Future.wait กับ command ที่ต้องรักษาลำดับหรือมี side effect ชนกัน เช่น validate → confirm stock ต้องต่อกัน

3. Event loop และงาน CPU

HTTP/timer ปล่อย event loop ไปทำ frame อื่นได้ จึงไม่ต้องใช้ isolate เพียงเพราะมี await แต่งาน parse/loop CPU หนักยังบล็อก UI แม้อยู่ใน async function—async ไม่สร้าง thread ใหม่

4. Fire-and-forget อย่างมีเจตนา

unawaited(store.loadAll());

warehouse_shell.dart ใช้ unawaited เพื่อประกาศชัดว่าไม่รอ bootstrap ก่อน render แต่ method ที่ถูกปล่อยต้องจัด error ภายใน มิฉะนั้นเกิด unhandled async error อย่าละ await เงียบ ๆ

5. Lifecycle หลัง await

final result = await repository.submit(code);
if (!mounted) return;
setState(() => message = result.message);

ระหว่างรอผู้ใช้อาจ back ออกจากหน้า Context/State จึงหมดอายุ นอกจากนี้ response เก่าอาจกลับหลัง query ใหม่ ต้องมี request token/cancellation ในบท state

6. Timeout และ duplicate command

Timeout หมายถึง client เลิกรอ ไม่ได้พิสูจน์ว่า server ไม่ทำงาน ถ้า retry command เช่น confirm pick อาจทำซ้ำ ต้องออกแบบ idempotency key/backend rule และ disable submit ระหว่าง request

7. แบบฝึกหัด

เขียน submit ที่กันกดซ้ำ คืน loading ใน finally และไม่แตะ widget หลัง dispose

เฉลยตัวอย่าง
Future<void> submit() async {
  if (_submitting) return;
  setState(() => _submitting = true);
  try {
    final result = await repository.submit(controller.text.trim());
    if (!mounted) return;
    setState(() => _message = result.message);
  } catch (e) {
    if (mounted) setState(() => _message = readableError(e));
  } finally {
    if (mounted) setState(() => _submitting = false);
  }
}

8. จุดพลาด

await ไม่ทำให้ operation ถูกยกเลิกเมื่อ widget dispose และ async ไม่ทำให้งาน CPU หนักลื่นขึ้น ต้องออกแบบ lifecycle, stale response และ cancellation แยกต่างหาก
