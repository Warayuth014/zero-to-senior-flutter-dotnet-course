<!-- android/app/src/main/AndroidManifest.xml -->
<uses-permission android:name="android.permission.CAMERA" />
สิ่งที่ต้องรู้รายละเอียด
การเลือกจากแกลเลอรีไม่ต้องขอสิทธิ์แล้วAndroid 13 ขึ้นไปใช้ตัวเลือกรูปของระบบ ซึ่งผู้ใช้เลือกเองว่าจะให้ไฟล์ไหน

ข้อความเหตุผลบน iOS ผู้ใช้อ่านจริง — เขียนว่า “เพื่อถ่ายรูปหลักฐานการรับของ” ดีกว่า “แอปต้องการใช้กล้อง” เพราะข้อความที่บอกประโยชน์ทำให้คนกดอนุญาตมากกว่า

5. ผู้ใช้ปฏิเสธสิทธิ์

นี่คือกรณีที่โค้ดจริงยังไม่ได้จัดการ และเป็นกรณีที่เกิดจริงบน PDA ที่เพิ่งตั้งเครื่องใหม่

/// ผู้ใช้ปฏิเสธไม่ให้สิทธิ์
///
/// แยกเป็นชนิดของตัวเองเพราะต้องจัดการต่างจากความผิดพลาดอื่น — ลองใหม่
/// เฉย ๆ ไม่ช่วย ต้องพาผู้ใช้ไปหน้าตั้งค่าของเครื่อง
class PermissionDeniedException implements Exception { ... }
try {
  picked = await picker.pick(source);
} on PermissionDeniedException {
  _stage = UploadStage.idle;
  // ข้อความบอกทางแก้ ไม่ใช่แค่บอกว่าถูกปฏิเสธ — ลองใหม่ในแอปไม่ช่วย
  // ต้องไปเปิดสิทธิ์ในหน้าตั้งค่าของเครื่อง
  _message = source == PickSource.camera
      ? 'แอปยังไม่ได้รับอนุญาตให้ใช้กล้อง เปิดสิทธิ์ในหน้าตั้งค่าของเครื่องก่อน'
      : 'แอปยังไม่ได้รับอนุญาตให้เข้าถึงรูป เปิดสิทธิ์ในหน้าตั้งค่าของเครื่องก่อน';
  notifyListeners();
  return;
}

ความต่างระหว่างการปฏิเสธสองแบบ

แบบเกิดอะไรถ้าขอใหม่ควรทำ
ปฏิเสธครั้งแรกระบบถามอีกครั้งขอใหม่ได้ แต่ควรอธิบายก่อนว่าทำไมต้องใช้
ปฏิเสธถาวร (กด “ไม่ถามอีก”)ระบบไม่ถามเลย คืนค่าปฏิเสธทันทีพาไปหน้าตั้งค่า

ถ้าไม่แยกสองกรณีนี้ ผู้ใช้ที่ปฏิเสธถาวรจะกดปุ่มถ่ายรูปแล้วไม่มีอะไรเกิดขึ้น ซ้ำแล้วซ้ำเล่า โดยไม่มีทางรู้ว่าทำไม

// เมื่อรู้ว่าปฏิเสธถาวร ให้ปุ่มที่พาไปได้จริง
FilledButton(
  onPressed: openAppSettings,   // จากแพ็กเกจ permission_handler
  child: const Text('ไปหน้าตั้งค่า'),
)

test('ไม่ได้รับสิทธิ์ ต้องบอกทางแก้ ไม่ใช่แค่บอกว่าถูกปฏิเสธ', () async {
  picker.denyPermission = true;

  await store.pick(PickSource.camera);

  expect(store.stage, UploadStage.idle);
  expect(store.message, contains('ตั้งค่า'));
  expect(store.message, contains('กล้อง'));
});

7. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
แอปปิดตัวเองตอนเปิดกล้องบน iOSไม่มี NSCameraUsageDescriptionเพิ่มใน Info.plist พร้อมเหตุผลจริง
กดถ่ายรูปแล้วไม่มีอะไรเกิดขึ้นปฏิเสธสิทธิ์ถาวรตรวจแล้วพาไปหน้าตั้งค่า
อัปโหลดนานมากไม่ได้ย่อตอนเลือกตั้ง maxWidth และ imageQuality
แอปค้างตอนแสดงตัวอย่างรูปเต็มความละเอียดย่อตั้งแต่ต้นทาง (13.7)
ทดสอบฟีเจอร์นี้ไม่ได้เลยเรียกปลั๊กอินตรงจากในจอห่อด้วย interface
ผู้ใช้กดยกเลิกแล้วเห็นข้อความผิดพลาดถือว่า null คือความล้มเหลวแยกการยกเลิกออกจากความผิดพลาด

8. แบบฝึกหัด

หัวหน้าขอให้เพิ่มฟีเจอร์ “ถ่ายรูปหลักฐานตอนรับของ” โดยบังคับให้ถ่ายอย่างน้อยหนึ่งรูปก่อนยืนยันการรับ

  1. ควรบังคับไหม
  2. ถ้าบังคับ ต้องจัดการอะไรบ้าง
แนวคำตอบ

1. ต้องถามก่อนว่าบังคับเพื่ออะไร

เหตุผลที่อ้างการบังคับแก้ปัญหาไหม
“จะได้มีหลักฐานตอนเคลม”ไม่ — พนักงานจะถ่ายพื้นหรือถ่ายเพดานเพื่อผ่านด่าน
“ของบุบแล้วไม่มีใครถ่ายไว้”บังคับเฉพาะตอนที่ผู้ใช้ระบุว่าของมีปัญหา — ตรงกว่ามาก

การบังคับทุกครั้งจะได้รูปที่ไร้ประโยชน์เก็บสะสมเป็นแสนรูป ซึ่งกินพื้นที่และทำให้รูปที่มีประโยชน์จริงหายไปในกอง

ข้อเสนอที่ดีกว่า: บังคับเฉพาะเมื่อผู้ใช้เลือกว่า “ของมีปัญหา” และทำให้การถ่ายรูปตอนปกติง่ายมากจนคนอยากถ่ายเอง

2. ถ้าหัวหน้ายืนยันว่าต้องบังคับ ต้องจัดการห้าอย่าง

กรณีต้องทำ
ผู้ใช้ไม่ให้สิทธิ์กล้องต้องมีทางออก ไม่งั้นเขาทำงานไม่ได้เลยทั้งกะ
กล้องเสียเหมือนกัน — ต้องมีทางข้ามพร้อมบันทึกเหตุผล
อัปโหลดไม่ผ่านเพราะเน็ตหลุดห้ามบล็อกการรับของ ให้เก็บรูปไว้ส่งทีหลัง
ถ่ายแล้วแบตหมดรูปที่ถ่ายไปแล้วต้องไม่หาย
ผู้ใช้ถ่ายผิดต้องถ่ายใหม่ได้ก่อนยืนยัน

แถวที่สามสำคัญที่สุดและเป็นหลักที่ใช้ได้ทั่วไป — อย่าให้ฟีเจอร์เสริมบล็อกงานหลัก การรับของเข้าคลังคือสิ่งที่ธุรกิจต้องการ ส่วนรูปคือของแถมที่มีประโยชน์ ถ้ารูปส่งไม่ได้แล้วรับของไม่ได้ด้วย แสดงว่าเราทำให้ระบบเปราะกว่าเดิมเพื่อฟีเจอร์ที่ตั้งใจจะช่วย

ทางที่ถูกคือเก็บรูปไว้ในเครื่องแล้วส่งเมื่อเน็ตกลับมา ซึ่งเป็นงานของ Part 29 — และถ้ายังทำไม่ได้ในรุ่นนี้ ให้ยอมรับว่ารูปบางรูปจะหาย ดีกว่าให้ของค้างอยู่หน้าท่ารับ

9. สรุป

  • ย่อรูปตั้งแต่ตอนเลือก ประหยัดทั้งเวลา แบต และหน่วยความจำ
  • ขนาดที่เลือกขึ้นกับว่ารูปนั้นเอาไปทำอะไร
  • การกดยกเลิกไม่ใช่ความผิดพลาด
  • การปฏิเสธสิทธิ์ถาวรต้องพาไปหน้าตั้งค่า ไม่ใช่ให้กดใหม่
  • ห่อปลั๊กอินไว้หลัง interface ไม่งั้นทดสอบสี่กรณีสำคัญไม่ได้เลย
  • อย่าให้ฟีเจอร์เสริมบล็อกงานหลัก
