จบบทนี้แล้วจะทำได้: ใช้ FutureBuilder ให้ถูก และมองออกทันทีเมื่อเห็นการใช้ผิด

1. โค้ดจริง

// wmsapp/lib/screens/supervisor/part_image_screen.dart
Widget _buildPartCard(Map<String, dynamic> part) {
  final imageUrl = part['imageUrl'] as String?;
  final hasImg = imageUrl != null && imageUrl.isNotEmpty;

  return Card(
    child: Row(
      children: [
        ClipRRect(
          child: Container(
            width: 56,
            height: 56,
            child: hasImg
                ? FutureBuilder<String>(
                    future: _api.getImageFullUrl(imageUrl),   // ← ตรงนี้
                    builder: (_, snap) {
                      if (!snap.hasData) {
                        return const Center(child: CircularProgressIndicator());
                      }
                      return Image.network(snap.data!, fit: BoxFit.cover, ...);
                    },
                  )
                : Icon(MdiIcons.imageOff),
          ),
        ),
        ...

โค้ดนี้ทำงานถูกต้องและแสดงรูปได้ แต่มีปัญหาที่ผู้ใช้เห็นได้ — วงหมุนกะพริบทุกครั้งที่เลื่อนลิสต์และทุกครั้งที่พิมพ์ในช่องค้นหา

2. ทำไม

FutureBuilder เริ่มงานใหม่ทุกครั้งที่ได้รับ Future ตัวใหม่ และการเรียกฟังก์ชัน async ใน build สร้าง Future ตัวใหม่ทุกครั้งเสมอ
_buildPartCard(...) ถูกเรียก
  → getImageFullUrl(imageUrl) ทำงาน → ได้ Future ตัวใหม่
    → FutureBuilder เห็นว่า future ไม่ใช่ตัวเดิม
      → รีเซ็ตกลับไปที่ ConnectionState.waiting
        → snap.hasData เป็น false
          → แสดงวงหมุน
            → Future เสร็จ → แสดงรูป

แล้ว _buildPartCard ถูกเรียกบ่อยแค่ไหน

เมื่อไหร่ความถี่
เลื่อนลิสต์ListView.builder สร้างการ์ดใหม่เมื่อเลื่อนเข้ามาในจอ
พิมพ์ในช่องค้นหา_filterParts เรียก setState ทุกตัวอักษร → การ์ดทุกใบวาดใหม่
อัปโหลดเสร็จsetState → วาดใหม่ทั้งลิสต์

แถวที่สองแรงที่สุด — ผู้ใช้พิมพ์รหัสสินค้าหกตัว รูปทุกใบในจอจะกะพริบหกครั้ง

3. ผลที่ตามมานอกจากกะพริบ

ผลรายละเอียด
งานซ้ำ_resolveBase() ถูกเรียกซ้ำ แม้จะมีแคชอยู่ก็ยังเสียเวลาเข้า Future
Image.network ถูกสร้างใหม่โชคดีที่ Flutter มีแคชรูป จึงไม่ได้โหลดใหม่จากเน็ต แต่ยังต้องประกอบ widget ใหม่
วัดประสิทธิภาพยากเวลาที่หายไปกระจายอยู่ในทุกเฟรม ไม่ใช่ก้อนเดียวที่เห็นชัด
อาการนี้อยู่รอดในโค้ดได้นานเพราะไม่มีอะไรผิด — รูปขึ้นถูก ข้อมูลถูก ไม่มีข้อผิดพลาด มีแค่การกะพริบที่คนเขียนโค้ดบนคอมที่เร็วอาจไม่ทันสังเกต แต่บน PDA ที่ช้ากว่า มันเห็นชัด

4. ทางแก้ เรียงจากดีที่สุด

ทางที่หนึ่ง — ทำให้ไม่ต้องใช้ FutureBuilder เลย

// base URL อ่านได้แบบ sync หลังโหลดเสร็จตอนเปิดแอป (13.5)
String imageUrl(String relativePath) =>
    Uri.parse(AppConfig.apiBaseUrl).resolve(relativePath).toString();

// ในจอ
Image.network(imageUrl(part.imageUrl), fit: BoxFit.cover)

ดีที่สุดเพราะกำจัดปัญหาที่ต้นเหตุ — ไม่มี Future ก็ไม่มีอะไรให้รีเซ็ต และโค้ดสั้นลงมาก

ทางที่สอง — เก็บ Future ไว้ อย่าสร้างใหม่

class _PartCardState extends State<PartCard> {
  late Future<String> _urlFuture;

  @override
  void initState() {
    super.initState();
    _urlFuture = widget.api.getImageFullUrl(widget.imageUrl);   // ครั้งเดียว
  }

  @override
  void didUpdateWidget(PartCard old) {
    super.didUpdateWidget(old);
    // สร้างใหม่เฉพาะเมื่อรูปเปลี่ยนจริง
    if (old.imageUrl != widget.imageUrl) {
      _urlFuture = widget.api.getImageFullUrl(widget.imageUrl);
    }
  }

  @override
  Widget build(BuildContext context) =>
      FutureBuilder<String>(future: _urlFuture, builder: ...);
}

ใช้เมื่อ Future นั้นเลี่ยงไม่ได้จริง — สังเกตว่าต้องมี didUpdateWidget ด้วย ไม่งั้นการ์ดที่ถูกใช้ซ้ำใน ListView จะแสดงรูปของสินค้าตัวเก่า

ทางที่สาม — แคชผลลัพธ์

final _urlCache = <String, String>{};

String? cachedUrl(String path) => _urlCache[path];

ช่วยได้ แต่ไม่ได้แก้การกะพริบ เพราะ FutureBuilder ยังรีเซ็ตอยู่ดี — ต้องทำคู่กับทางที่สอง

5. กฎที่จำง่าย

อย่าเรียกฟังก์ชัน async ใน build — ถ้าเห็น future: ตามด้วยการเรียกฟังก์ชัน ให้สงสัยไว้ก่อน
เขียนแบบนี้ = ผิดเขียนแบบนี้ = ถูก
future: api.load()future: _loadFuture
future: Future.delayed(...)เก็บใน initState
stream: repository.watch()stream: _stream

กฎเดียวกันใช้กับ StreamBuilder ทุกประการ และผลของการทำผิดหนักกว่า — Stream ตัวใหม่ทุกเฟรมแปลว่าสมัครใหม่และยกเลิกของเก่าทุกเฟรม ซึ่งอาจเปิดการเชื่อมต่อใหม่ทุกครั้ง

6. เมื่อไหร่ FutureBuilder ถึงเหมาะจริง

เหมาะไม่เหมาะ
งาน async ที่ทำครั้งเดียวตอนเปิดจองานที่ผูกกับข้อมูลในลิสต์ที่วาดซ้ำบ่อย
ผลลัพธ์ไม่เปลี่ยนหลังได้มาแล้วข้อมูลที่ต้องโหลดใหม่ตามการกระทำของผู้ใช้
ไม่มีสถานะอื่นที่ต้องจัดการมีการโหลดใหม่ ลองใหม่ หรือแก้ไข

สำหรับกรณีที่ไม่เหมาะ ให้ใช้ store ที่ Part 9 สอน

ListenableBuilder(
  listenable: store,
  builder: (context, _) {
    if (store.loading) return const CircularProgressIndicator();
    if (store.error case final error?) return ErrorView(error);
    return PartList(parts: store.parts);
  },
)
ได้อะไรเพิ่มจาก store
โหลดใหม่ได้โดยไม่ต้องสร้าง widget ใหม่
หลายจอเห็นข้อมูลชุดเดียวกัน (9.9)
ทดสอบได้โดยไม่ต้องสร้าง widget (9.7)
จัดการสถานะกำลังโหลดกับข้อมูลเก่าพร้อมกันได้

7. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
วงหมุนกะพริบตอนเลื่อนลิสต์สร้าง Future ใน buildเก็บไว้ หรือทำให้เป็น sync
การ์ดแสดงรูปของรายการอื่นเก็บ Future แต่ลืม didUpdateWidgetสร้างใหม่เมื่อข้อมูลเปลี่ยน
คำขอถูกยิงซ้ำหลายรอบเหมือนข้อแรก แต่ฟังก์ชันยิงเน็ตจริงย้ายไป store
snapshot.data เป็น null ตอนมีข้อผิดพลาดไม่ได้เช็ค hasErrorเช็คทั้งสามสถานะ
โหลดใหม่ไม่ได้FutureBuilder ไม่มีทางสั่งให้ทำซ้ำใช้ store

แถวสุดท้ายคือข้อจำกัดพื้นฐาน — FutureBuilder ไม่มี API สำหรับ “ทำใหม่” วิธีเดียวคือใส่ Future ตัวใหม่เข้าไป ซึ่งต้องมี setState อยู่ดี

8. แบบฝึกหัด

อ่านโค้ดนี้แล้วบอกว่ามีปัญหาอะไร

@override
Widget build(BuildContext context) {
  return StreamBuilder<List<WmsTask>>(
    stream: repository.watchOpenTasks(),
    builder: (context, snapshot) {
      if (!snapshot.hasData) return const CircularProgressIndicator();
      return TaskList(tasks: snapshot.data!);
    },
  );
}
แนวคำตอบ

ปัญหาเดียวกับ FutureBuilder แต่หนักกว่า

ปัญหาผล
watchOpenTasks() สร้าง stream ใหม่ทุกเฟรมสมัครใหม่และยกเลิกของเก่าทุกเฟรม
ถ้า stream นั้นเปิดการเชื่อมต่อต่อและตัดทุกเฟรม — อาจถึงหกสิบครั้งต่อวินาที
วงหมุนกลับมาทุกเฟรมหน้าจอไม่มีทางแสดงข้อมูลได้เลย เพราะ hasData เป็น false ตลอด

ข้อสุดท้ายทำให้บั๊กนี้เห็นชัดกว่ากรณี FutureBuilder — หน้าจอจะค้างที่วงหมุนตลอดกาล ซึ่งอย่างน้อยก็ทำให้เจอเร็ว

แก้แบบเร็ว

late final _stream = repository.watchOpenTasks();   // สร้างครั้งเดียว

@override
Widget build(BuildContext context) => StreamBuilder(stream: _stream, ...);

แก้แบบที่ควรทำ — ย้ายไป store ตามที่ Part 12 ทำ

class LiveTaskStore extends ChangeNotifier {
  Future<void> start() async {
    _eventSub = client.events.listen(_onEvent);   // สมัครที่เดียว
    ...
  }

  @override

    _eventSub?.cancel();      // และยกเลิกที่เดียว (12.8)
    super.dispose();
  }

ข้อสังเกตที่สำคัญกว่าตัวโค้ด: ทั้ง FutureBuilder และ StreamBuilder ดูเหมือนเป็นทางลัดที่ทำให้ไม่ต้องมี store แต่จริง ๆ แล้วมันแค่ย้ายการจัดการวงจรชีวิตไปไว้ในที่ที่ควบคุมยากกว่า — ใน build ที่ถูกเรียกเมื่อไหร่ก็ได้และกี่ครั้งก็ได้

ใช้มันกับงานที่ทำครั้งเดียวและจบ ส่วนงานที่มีวงจรชีวิต ให้มีคนถือชัดเจน

9. สรุป

  • FutureBuilder รีเซ็ตทุกครั้งที่ได้ Future ตัวใหม่
  • เรียกฟังก์ชัน async ใน build สร้างตัวใหม่ทุกครั้ง
  • ทางที่ดีที่สุดคือทำให้ไม่ต้องมี Future ตั้งแต่แรก
  • ถ้าเลี่ยงไม่ได้ ให้เก็บไว้ พร้อม didUpdateWidget
  • กฎเดียวกันใช้กับ StreamBuilder และผลหนักกว่า
  • งานที่มีวงจรชีวิต ให้ store ถือ
