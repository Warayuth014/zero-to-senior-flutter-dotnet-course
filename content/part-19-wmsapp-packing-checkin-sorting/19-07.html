Future<void> _scanCarton() async {
  final packingId = _scanCtrl.text.trim().toUpperCase();
  if (packingId.isEmpty) return;

  setState(() => _loading = true);
  final res = await _api.previewCheckIn(packingId);       // ← 1. preview
  if (!mounted) return;
  setState(() => _loading = false);

  if (!res.success) {
    showErrorDialog(context, message: res.error ?? 'สแกนไม่สำเร็จ');
    _scanCtrl.clear();
    _scanFocus.requestFocus();
    return;
  }

  final preview = res.data!;
  _scanCtrl.clear();

  // popup แจ้งปลายทาง
  await _showDestinationDialog(preview);                   // ← 2. บังคับให้อ่าน
  if (!mounted) return;

  setState(() {
    _pendingPack = preview;
    _deliveryNotePack = null;
    _deliveryNoteCreatedAt = null;
    if (preview.isNewSlot) {
      _slotDetail = _syntheticSlotFromPreview(preview);    // ← 3a. สร้างเอง
    }
  });
  if (!preview.isNewSlot) {
    await _openSlot(preview.slotId);                       // ← 3b. โหลดจริง
  }
  _scanFocus.requestFocus();
}

/// สถานะของจอ Check-in — ถือค่าได้ทีละแบบ
sealed class CheckInState {
  const CheckInState();
}

class CheckInScanning extends CheckInState {

}

class CheckInPending extends CheckInState {
  const CheckInPending({required this.pack, required this.slot});

/// ยืนยันแล้ว — แสดงใบส่งสินค้า
class CheckInConfirmed extends CheckInState {

    required this.pack,
    required this.confirmedAt,
    required this.slot,
  });
  final PreviewCheckInResponse pack;
  final DateTime confirmedAt;
  final CheckInSlotDetail slot;
}

/// โหลดข้อมูลช่อง — ถ้าโหลดไม่ได้ ใช้ข้อมูลจาก preview แทน
/// เพื่อให้ผู้ใช้ยังทำงานต่อได้ ไม่ต้องเริ่มสแกนใหม่
Future<CheckInSlotDetail> _loadSlotOrSynthetic(PreviewCheckInResponse p) async {
  final res = await _api.getCheckInSlot(p.slotId);
  if (res.success) return res.data!;

  if (mounted) {
    showWarningSnackbar(context, 'โหลดข้อมูลช่องไม่สำเร็จ — แสดงข้อมูลเท่าที่มี');
  }
  return _syntheticSlotFromPreview(p);
}

9. สรุป

  • เจ็ดฟิลด์ แต่สามตัวเป็น nullable ที่ประกอบกันเป็นโหมด
  • ชนิดข้อมูลยอมให้เกิดแปดชุดค่า ทั้งที่ตั้งใจให้มีสามสถานะ
  • คอมเมนต์สี่บรรทัดจำเป็นเพราะชื่อฟิลด์บอกความหมายไม่ได้
  • onSlotPage รวมสามเงื่อนไขเป็นบรรทัดเดียวเพื่อแบ่งหน้า
  • หน้าที่สองประกอบจากสามการ์ดที่แสดงตามฟิลด์ของตัวเอง
  • preview เป็นขั้นบังคับที่มี endpoint แยก ไม่แตะฐานข้อมูล
  • ช่องใหม่สร้างข้อมูลขึ้นเองจาก preview — ชื่อ synthetic บอกชัด
  • popup ปิดไม่ได้ บังคับให้อ่านปลายทาง และมีตาข่ายชั้นสองตอนกดยืนยัน
  • ค้นจุดที่เปลี่ยนสถานะต้องค้นสามคำ ไม่ใช่คำเดียว
