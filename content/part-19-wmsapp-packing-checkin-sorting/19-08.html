ต้องผ่านอะไรมาก่อน: 19.7, 7.10 (คำสั่งที่ไม่รู้ผล), 19.5 (การวนเรียก API)

1. สามคำสั่งจากการกดปุ่มเดียว

Future<void> _confirmCheckIn() async {
  final pack = _pendingPack;
  if (pack == null) return;

  if (pack.isAlreadyCheckedIn) { ...ปฏิเสธ... }

  setState(() => _loading = true);
  final res = await _api.scanCheckIn(                    // ← 1
    packingId: pack.packingId,
    operatorId: widget.userId,
  );
  if (!mounted) return;
  setState(() => _loading = false);

  if (!res.success) {
    showErrorDialog(context, message: res.error ?? 'Check-IN ไม่สำเร็จ');
    return;
  }

  final scanResult = res.data!;

  // ถ้าครบทุกกล่อง → auto complete + auto dispatch เงียบ ๆ
  if (scanResult.isReadyToComplete) {
    await _autoCompleteSlot(scanResult.slotId);          // ← 2
    if (!mounted) return;
    await _autoDispatchSlot(scanResult.slotId);          // ← 3
    if (!mounted) return;
  }

  setState(() {
    _pendingPack = null;
    _deliveryNotePack = pack;
    _deliveryNoteCreatedAt = DateTime.now();
  });
  await _openSlot(scanResult.slotId);                    // ← 4
  _scanFocus.requestFocus();
}

3. สองเมธอดที่ล้มเหลวแล้วไม่หยุด

Future<void> _autoCompleteSlot(String slotId) async {
  final res = await _api.completeCheckIn(slotId: slotId, operatorId: widget.userId);
  if (!mounted) return;
  if (!res.success) {
    showErrorDialog(context, message: res.error ?? 'Complete ไม่สำเร็จ');
  }

Future<void> _autoDispatchSlot(String slotId) async {
  final res = await _api.dispatchCheckIn(slotId: slotId, operatorId: widget.userId);
  if (!mounted) return;
  if (!res.success) {
    showErrorDialog(context, message: res.error ?? 'Dispatch ไม่สำเร็จ');
  }
}

ทั้งสองเมธอดไม่คืนค่าอะไร — ผู้เรียกไม่รู้ว่าสำเร็จหรือไม่

if (scanResult.isReadyToComplete) {
  await _autoCompleteSlot(scanResult.slotId);
  if (!mounted) return;
  await _autoDispatchSlot(scanResult.slotId);    // ← ยิงต่อแม้ขั้นก่อนล้มเหลว
  if (!mounted) return;
}
ถ้า complete ล้มเหลวเกิดอะไร
ผู้ใช้เห็นกล่อง errorกดปิด
dispatch ถูกยิงต่อช่องยังไม่ถูกปิด แต่สั่งส่งขึ้นรถ
เซิร์ฟเวอร์น่าจะปฏิเสธผู้ใช้เห็นกล่อง error อีกครั้ง
ผู้ใช้ต้องกดปิดกล่องสองครั้งติดกัน — และไม่มีข้อความไหนบอกว่าเกิดอะไรขึ้นจริง
ปัญหาเดียวกับที่ 19.5 พบใน _confirmCollectedPack ที่วนต่อทั้งที่ล้มเหลว
ต่างกันตรงที่ที่นี่คำสั่งที่สองขึ้นกับคำสั่งแรก — ถ้าปิดช่องไม่สำเร็จ ส่งขึ้นรถก็ไม่ควรทำ
Future<bool> _autoCompleteSlot(String slotId) async {
  final res = await _api.completeCheckIn(slotId: slotId, operatorId: widget.userId);
  if (!mounted) return false;
  if (!res.success) {
    showErrorDialog(context, message: res.error ?? 'ปิดช่องไม่สำเร็จ');
    return false;
  }
  return true;
}
if (scanResult.isReadyToComplete) {
  final completed = await _autoCompleteSlot(scanResult.slotId);
  if (!mounted) return;

  // ปิดช่องไม่สำเร็จ → ยังไม่ควรสั่งส่งขึ้นรถ
  if (completed) {
    await _autoDispatchSlot(scanResult.slotId);
    if (!mounted) return;
  }
}

4. คำสั่งที่กลับไม่ได้และไม่รู้ผล

dispatchCheckIn ทำให้ของขึ้นรถ — และเมธอดนี้ไม่มีอะไรบอกผู้ใช้เมื่อไม่รู้ผล

if (!res.success) {
  showErrorDialog(context, message: res.error ?? 'Dispatch ไม่สำเร็จ');
}
เน็ตหลุดหลังเซิร์ฟเวอร์รับคำสั่งผู้ใช้เห็นความจริง
ใช่“Dispatch ไม่สำเร็จ”ของขึ้นรถไปแล้ว

ผู้ใช้ที่เชื่อข้อความจะไปตามหาของ หรือสั่งซ้ำ

if (!res.success) {
  showErrorDialog(
    context,
    title: 'ไม่ทราบผล',
    message: '${res.error ?? "ส่งขึ้นรถไม่สำเร็จ"}\n\n'
        'คำสั่งอาจส่งไปแล้ว — ตรวจสถานะช่อง ${slotId} ก่อนสั่งซ้ำ',
  );
}

และหลังจากนั้นควรโหลดสถานะช่องใหม่ให้เลย

await _openSlot(scanResult.slotId);      // ← มีอยู่แล้วในโค้ด
บรรทัดนี้ช่วยไว้โดยบังเอิญ — ผู้ใช้ที่เห็นข้อความล้มเหลวจะเห็นสถานะช่องที่โหลดใหม่ทันที
ถ้าช่องเป็น SHIPPED แปลว่าคำสั่งไปถึงจริง
แต่ไม่มีอะไรบอกผู้ใช้ให้ดูตรงนั้น — ข้อความควรชี้ไป

7. ข้อผิดพลาดที่พบบ่อย

เข้าใจผิดที่จริง
ปุ่มเดียวยิงคำสั่งเดียวที่นี่ยิงได้ถึงสี่
เมธอดที่ไม่คืนค่าแปลว่าไม่มีอะไรผิดพลาดได้ทั้งสองเมธอด auto ล้มเหลวได้แต่ไม่บอกผู้เรียก
คำสั่งที่ต่อกันจะหยุดเองเมื่ออันแรกล้มต้องเขียนให้หยุดเอง
“Dispatch ไม่สำเร็จ” แปลว่าของยังไม่ขึ้นรถอาจขึ้นไปแล้ว
เลขที่เอกสารจากแอปใช้อ้างอิงได้เป็นของจำลอง ไม่มีในระบบ

8. แบบฝึกหัด

ผู้ใช้แจ้งว่า “สแกนกล่องสุดท้าย กดยืนยัน แล้วขึ้น error สองรอบ พอกดปิดหมดแล้วดูสถานะช่อง กลายเป็น SHIPPED ทั้งที่บอกว่าไม่สำเร็จ”

  1. อธิบายว่าเกิดอะไรขึ้น
  2. ข้อความที่ผู้ใช้ควรได้คืออะไร
  3. ออกแบบทางแก้ที่ครบ
แนวคำตอบ

ข้อ 1 — เกิดอะไรขึ้น

มีสองสถานการณ์ที่อธิบายได้

สถานการณ์ A — เน็ตสะดุดตอน complete แต่คำสั่งไปถึง
  1. scanCheckIn สำเร็จ → isReadyToComplete = true
  2. completeCheckIn → คำสั่งไปถึง แต่คำตอบไม่กลับมา → error #1
  3. dispatchCheckIn → ยิงต่อ (ไม่ได้หยุด) → สำเร็จจริง
                       แต่คำตอบไม่กลับมา → error #2
  4. _openSlot → โหลดใหม่ → เห็น SHIPPED
สถานการณ์ B — complete ล้มเหลวจริง แต่ dispatch ทำงานได้
  1. scanCheckIn สำเร็จ
  2. completeCheckIn ล้มเหลวจริง → error #1
  3. dispatchCheckIn ยิงต่อ → เซิร์ฟเวอร์อาจ complete ให้เองแล้ว dispatch
                             → error #2 จากอย่างอื่น
  4. _openSlot → SHIPPED

ทั้งสองกรณีมีสาเหตุร่วมกัน — ยิงคำสั่งที่สองต่อทั้งที่คำสั่งแรกล้มเหลว

และผู้ใช้เห็น error สองครั้งเพราะสองเมธอดแสดงกล่องเอง โดยไม่มีใครประสานกัน

ข้อ 2 — ข้อความที่ควรได้

ผู้ใช้ต้องรู้ปัจจุบันบอกไหม
กล่องถูกบันทึกแล้วหรือยังไม่
ช่องถูกปิดแล้วหรือยังไม่
ของขึ้นรถแล้วหรือยังบอกผิด
ต้องทำอะไรต่อไม่

สี่คำถาม และข้อความปัจจุบันตอบไม่ได้เลยสักข้อ

ข้อ 3 — ทางแก้

Future<void> _confirmCheckIn() async {
  final pack = _pendingPack;
  if (pack == null) return;
  if (pack.isAlreadyCheckedIn) { ...ปฏิเสธ... }

  setState(() => _loading = true);
  final res = await _api.scanCheckIn(
    packingId: pack.packingId,
    operatorId: widget.userId,
  );
  if (!mounted) return;
  setState(() => _loading = false);

  if (!res.success) {
    showErrorDialog(
      context,
      title: 'ไม่ทราบผล',
      message: '${res.error ?? "Check-IN ไม่สำเร็จ"}\n\n'
          'กล่องอาจถูกบันทึกไปแล้ว — สแกนซ้ำเพื่อตรวจก่อนกดใหม่',
    );
    return;
  }

  final scanResult = res.data!;

  // กล่องบันทึกแล้วแน่นอน — สลับสถานะก่อน ไม่ให้หายถ้าขั้นถัดไปล้ม
  setState(() {
    _pendingPack = null;
    _deliveryNotePack = pack;
    _deliveryNoteCreatedAt = DateTime.now();
  });

  if (scanResult.isReadyToComplete) {
    await _finishSlot(scanResult.slotId);
    if (!mounted) return;
  }

  await _openSlot(scanResult.slotId);
  _scanFocus.requestFocus();
}
/// ปิดช่องแล้วส่งขึ้นรถ — สองคำสั่งที่ต้องทำตามลำดับ
/// ล้มเหลวขั้นไหนก็หยุด แล้วสรุปให้ผู้ใช้ครั้งเดียว
Future<void> _finishSlot(String slotId) async {
  final completed = await _completeSlot(slotId);
  if (!mounted) return;

  if (!completed) {
    await _showSlotIncompleteDialog(
      slotId: slotId,
      stage: 'ปิดช่อง',
      hint: 'กล่องถูกบันทึกแล้ว แต่ช่องยังไม่ถูกปิด',
    );
    return;
  }

  final dispatched = await _dispatchSlot(slotId);
  if (!mounted) return;

  if (!dispatched) {
    await _showSlotIncompleteDialog(
      slotId: slotId,
      stage: 'ส่งขึ้นรถ',
      hint: 'ช่องถูกปิดแล้ว แต่ยังไม่ได้สั่งขึ้นรถ',
    );
  }
}
Future<void> _showSlotIncompleteDialog({
  required String slotId,
  required String stage,
  required String hint,
}) async {
  await showDialog<void>(
    context: context,
    builder: (ctx) => AlertDialog(
      title: Row(children: [
        Icon(Icons.warning_amber_rounded, color: AppTheme.warning),
        const SizedBox(width: 8),
        Expanded(child: Text('$stage ไม่สำเร็จ')),
      ]),
      content: Column(
        mainAxisSize: MainAxisSize.min,
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Text(hint),
          const SizedBox(height: 12),
          Text(
            'คำสั่งอาจส่งไปแล้ว — ดูสถานะช่อง $slotId ด้านล่างก่อนทำซ้ำ',
            style: TextStyle(fontSize: 13, color: AppTheme.textGrey(ctx)),
          ),
        ],
      ),
      actions: [
        ElevatedButton(
          onPressed: () => Navigator.pop(ctx),
          child: const Text('เข้าใจแล้ว'),
        ),
      ],
    ),
  );
}

สลับสถานะก่อนขั้นที่ 2–3“กล่องถูกบันทึกแล้วหรือยัง”

  2. ถ้าสั่ง dispatch ซ้ำกับช่องที่ SHIPPED แล้ว จะเกิดอะไร
  3. มี endpoint ให้ถามสถานะช่องแบบเบา ๆ ไหม
ข้อสองสำคัญที่สุด — ถ้าเซิร์ฟเวอร์ปฏิเสธอย่างสุภาพ ข้อความสามารถบอกว่า “กดซ้ำได้ปลอดภัย” ซึ่งช่วยผู้ใช้มาก
ถ้าไม่รู้ ต้องเขียนข้อความแบบระวังไว้ก่อน ซึ่งทำให้ผู้ใช้ทำงานช้าลงโดยไม่จำเป็น

ข้อสรุปที่ใหญ่กว่า: คำสั่งที่ทำงานเงียบ ๆ แทนผู้ใช้ ประหยัดการกดได้จริง — แต่เมื่อล้มเหลว ผู้ใช้จะไม่มีบริบทเลยว่าอะไรเกิดขึ้น เพราะเขาไม่ได้เป็นคนสั่ง

ยิ่งทำเงียบมาก ยิ่งต้องสรุปให้ชัดเมื่อผิดพลาด

9. สรุป

  • ปุ่มเดียวยิงได้ถึงสี่คำสั่ง สามในนั้นเปลี่ยนข้อมูล
  • เซิร์ฟเวอร์ตัดสินว่าครบทุกกล่องแล้วหรือยัง แอปไม่เดา
  • _autoCompleteSlot และ _autoDispatchSlot ไม่คืนค่า ผู้เรียกไม่รู้ผล
  • คำสั่งที่สองถูกยิงต่อแม้คำสั่งแรกล้มเหลว
  • ผู้ใช้เห็นกล่อง error สองครั้งติดโดยไม่มีใครประสาน
  • dispatchCheckIn ทำให้ของขึ้นรถ — ล้มเหลวแล้วไม่บอกว่าอาจไปถึงแล้ว
  • สลับ _pendingPack เป็น _deliveryNotePack ในก้อนเดียว
  • ใบส่งสินค้าเป็นของจำลอง และชื่อเมธอดบอกไว้ชัด
  • ช่องสแกนถูกล้างครบทุกทาง ต่างจากจออื่นใน Part 18
  • คำสั่งที่ทำเงียบแทนผู้ใช้ ต้องสรุปให้ชัดเมื่อผิดพลาด
