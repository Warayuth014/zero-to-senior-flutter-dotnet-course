Future<void> _openStation(SortingStationView s) async {
  if (s.status == 'DISABLED') {
    _showDisabledMenu(s);       // ← เมนูเปิดใช้งาน
    return;
  }
  if (s.status == 'AVAILABLE') {

  }
  // BUSY → open detail sheet
  await showModalBottomSheet<void>(
    context: context,
    isScrollControlled: true,
    useSafeArea: true,
    backgroundColor: Colors.transparent,
    builder: (_) => _StationDetailSheet(
      stationId: s.stationId,
      userId: widget.userId,
      onChanged: _loadStations,
    ),
  );
}

3. การยืนยันที่บอกผลลัพธ์

Future<void> _confirmToggle(SortingStationView s, {required bool enable}) async {
  final stationLabel = _formatSortingStationId(s.stationId);
  final confirm = await showConfirmDialog(
    context,
    title: enable ? 'เปิดใช้งาน Station' : 'ปิดใช้งาน Station',
    message: '$stationLabel จะ ${enable ? "พร้อมรับ batch" : "หยุดรับ batch ใหม่"}',
    confirmLabel: enable ? 'เปิดใช้งาน' : 'ปิดใช้งาน',
    isDanger: !enable,
  );
  if (!confirm) return;

  setState(() => _loading = true);
  final res = await _api.toggleSortingStation(
    stationId: s.stationId,
    enable: enable,
    operatorId: widget.userId,
  );
  if (!mounted) return;
  setState(() => _loading = false);
  if (!res.success) {
    showErrorDialog(context, message: res.error ?? 'Toggle ไม่สำเร็จ');
    return;
  }
  _loadStations();
}
รายละเอียดทำไมดี
ข้อความบอกว่าจะเกิดอะไร“หยุดรับ batch ใหม่” ไม่ใช่ “ยืนยันหรือไม่”
confirmLabel เขียนการกระทำผู้ใช้ที่ไม่อ่านเนื้อหาก็เห็นจากปุ่ม
isDanger: !enableปุ่มแดงเมื่อจะปิด — การกระทำที่กระทบงาน
โหลดใหม่หลังสำเร็จสถานะบนจอตรงกับความจริง

แถวที่สามใช้ isDanger ที่ showConfirmDialog มีอยู่แล้ว (16.9) — ตรงกับที่ 18.6 แบบฝึกหัดเสนอ

คำว่า “หยุดรับ batch ใหม่” สำคัญกว่าที่เห็น — บอกว่าปิดสถานีแล้วงานที่อยู่บนสถานีตอนนี้ไม่หายไป
ถ้าเขียนว่า “ปิดสถานี” ผู้ใช้อาจกลัวว่าของที่ค้างอยู่จะหาย
เป็นหลักเดียวกับ “AMR จะมารับทันที” ใน 18.3

แต่ยังขาดสิ่งที่ 7.10 สอน

8. แบบฝึกหัด

เทียบ SortingScreen กับ PutawayScreen (18.2) แล้วตอบ

  1. ปัญหาที่ 18.2 และ 18.4 พบ ข้อไหนไม่เกิดที่นี่ และเพราะอะไร
  2. ปัญหาข้อไหนที่ยังเกิดทั้งสองจอ
  3. ถ้าจะแก้ PutawayScreen ให้เหมือนจอนี้ ต้องทำอะไร
แนวคำตอบ

ข้อ 1 — ปัญหาที่ไม่เกิด

ปัญหาใน 18.2/18.4เกิดที่นี่ไหมเพราะ
แตะสถานีที่ไม่ว่างแล้ว return; เงียบไม่ทุกสถานะมีเมนู
Row ที่ล้นเมื่อเกินสามสถานีไม่ใช้ Column
สถานีฝังในโค้ด เพิ่มต้องปล่อยแอปใหม่ไม่มาจากเซิร์ฟเวอร์
ข้อความ “Supported stations” พิมพ์รายชื่อซ้ำไม่ไม่มีช่องสแกนสถานี
สถานะสถานีเป็น Map ดิบไม่มี SortingStationView
ล้มเหลวแล้วเงียบ สถานีดูว่างทั้งที่ไม่ว่างไม่_loadStations แสดงกล่อง error

หกข้อจากรายการของสองบทนั้น — และทั้งหมดเป็นผลของการที่จอนี้เขียนทีหลัง

สังเกตว่าข้อที่สามกับสี่หายไปพร้อมกัน เพราะเป็นผลของการตัดสินใจเดียว: ให้เซิร์ฟเวอร์เป็นเจ้าของรายชื่อสถานี

18.4 แบบฝึกหัดถามว่าคุ้มไหมที่จะย้ายรายชื่อสถานีไปเซิร์ฟเวอร์ — จอนี้คือคำตอบว่าทำแล้วได้อะไร

ข้อ 2 — ปัญหาที่ยังเกิดทั้งสองจอ

ปัญหาทั้งสองจอราก
ต่อ SignalR ไม่ได้แล้วเงียบเกิดบริการทั้งสองตัวเหมือนกัน (18.8)
ไม่มี onreconnectedเกิดเหมือนกัน
ล้มเหลวแล้วไม่บอกว่าอาจสำเร็จแล้วเกิดApiResult ไม่มี outcomeUnknown (16.6)
แอนิเมชันหรือสถานะที่ไม่ผูกกับความจริงputaway เท่านั้น—

สามข้อแรกมีรากอยู่นอกจอ — ตรงกับที่ 18.11 สรุปไว้ว่าปัญหาที่รากอยู่ชั้นล่างแก้ที่จอไม่ได้

ข้อ 3 — แก้ PutawayScreen อย่างไร

ลำดับทำอะไรต้องแตะเซิร์ฟเวอร์ไหม
1เปลี่ยน Row เป็น Wrap หรือ Columnไม่
2ให้ทุกสถานะมีสิ่งที่ทำได้ หรืออย่างน้อยบอกว่าทำไมกดไม่ได้ไม่
3สร้าง StationStatus แทน Map ดิบ (18.7)ไม่
4ย้ายรายชื่อสถานีไปเซิร์ฟเวอร์ต้อง

สามข้อแรกทำได้ทันที และแก้ปัญหาสี่ข้อจากหกข้อในตารางแรก

นี่คือแถวใหม่ที่ต้องเติมเข้าตารางกฎที่ไม่ตรงกันใน 18.16 — และเป็นสิ่งที่ Part 23 จะแก้

9. สรุป

  • ห้าฟิลด์ ไม่มีตัวควบคุมช่องกรอก และไม่มีสถานะของ flow เลย
  • _stations เป็นสำเนาของสิ่งที่เซิร์ฟเวอร์รู้ ไม่ใช่สถานะที่แอปเป็นเจ้าของ
  • สามสถานะสถานีแตะได้ทั้งหมด — ไม่มี return; เงียบสักจุด
  • เมนูปิดก่อนแล้วค่อยเปิดกล่องยืนยัน
  • ข้อความยืนยันบอกผลลัพธ์ และใช้ isDanger กับการกระทำที่กระทบงาน
  • ล้มเหลวแล้วไม่โหลดใหม่ และไม่บอกว่าคำสั่งอาจไปถึงแล้ว
  • _buildGrid เป็น Column ที่รองรับกี่สถานีก็ได้
  • ตัวเลขสรุปสองตัวคำนวณเอง สองตัวมาจากเซิร์ฟเวอร์ — แบ่งถูก
  • _loadStations ถูกเรียกจากห้าทาง ครบกว่าจอไหนในโปรเจกต์
  • หกปัญหาของ PutawayScreen ไม่เกิดที่นี่ เพราะเขียนทีหลัง
