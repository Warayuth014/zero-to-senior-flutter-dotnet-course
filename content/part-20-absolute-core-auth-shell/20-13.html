String _uniqueName(String base) {
  final taken = _controller.profiles.map((p) => p.name.trim().toLowerCase()).toSet();
  for (var i = 2; i < 100; i++) {
    final candidate = '$base ($i)';
    if (!taken.contains(candidate.toLowerCase())) return candidate;
  }
  return '$base ${DateTime.now().millisecondsSinceEpoch}';
}

เป็นตัวอย่างของข้อจำกัดในชั้น widget ที่ทำให้จอเขียนสิ่งที่ถูกไม่ได้ — ตรงกับสิ่งที่ 18.16 สรุปไว้

10. ข้อผิดพลาดที่พบบ่อย

ผิดอาการถูก
เก็บ _isDirty เป็นธงเพิ่มช่องแล้วลืมตั้ง งานหายคำนวณจากลายนิ้วมือของฟอร์ม
ปล่อยลากปิดได้ข้ามด่านการยืนยันenableDrag: false
ไม่ล้างผลทดสอบเมื่อค่าเปลี่ยนเขียวของค่าชุดก่อนล้างใน listener
แสดงข้อความผิดพลาดตั้งแต่เปิดฟอร์มเปล่าเต็มไปด้วยสีแดง_showErrors หลังกดครั้งแรก
สลับไปเซิร์ฟเวอร์ที่ต่อไม่ได้เงียบ ๆแอปตายทั้งเครื่องตรวจก่อน แล้วให้ผู้ใช้เลือก
ห้ามบันทึกที่อยู่ที่ยังต่อไม่ได้ตั้งค่าล่วงหน้าไม่ได้เตือนแล้วให้ตัดสินใจ
เลือกตัวแทนให้เองเมื่อลบตัวที่ใช้ไปโผล่เซิร์ฟเวอร์ที่ไม่ได้ตั้งใจให้ผู้ใช้เลือก
ปุ่มยืนยันอยู่ในส่วนที่เลื่อนแป้นพิมพ์ขึ้นแล้วหาปุ่มไม่เจอหัว/กลางเลื่อน/ท้ายคงที่

11. แบบฝึกหัด

  1. ผู้ใช้เปิดแผ่น แก้พอร์ต กดทดสอบได้เขียว แล้วแก้ host ต่อ แล้วกดปิด — ไล่ลำดับ state ทั้งหมด
  2. _saveChanges ถาม “ใช้ค่าใหม่ทันทีไหม” แต่ _saveAsNew ไม่ถาม — ถูกต้องหรือไม่
แนวคำตอบ

ข้อ 1

เปิดแผ่น
  initState → _loadProfile(active) → _afterLoad()
  _baseline = 'Office|192.168.1.50|6191||development|http'
  _isDirty = false

แก้พอร์ตเป็น 6192
  _onFormChanged → _testResult = null
  (_showErrors ยังเป็น false จึงไม่ตรวจ)
  _isDirty = true      ← ลายนิ้วมือต่างจาก baseline

กดทดสอบ
  _draft() → _showErrors = true, ผ่านการตรวจ
  _testing = true → probeConnection → _testResult = ok
  _testing = false

แก้ host เป็น 192.168.1.51
  _onFormChanged → _testResult = null      ← ผลเขียวหายทันที

จุดสำคัญคือบรรทัดที่ทำเครื่องหมายไว้ — ถ้าไม่มีการล้าง _testResult ผู้ใช้จะเห็นการ์ดเขียวของ 192.168.1.50 อยู่ใต้ฟอร์มที่ตอนนี้เขียนว่า .51

ข้อ 2

ถูกต้อง และเหตุผลมาจากความหมายของสองการกระทำ

_saveChanges_saveAsNew
แก้อะไรโปรไฟล์ที่มีอยู่สร้างอันใหม่
อาจกระทบตัวที่ใช้อยู่ไหมได้ ถ้าแก้ตัวที่กำลังใช้ไม่ — ของใหม่ไม่มีใครใช้
ต้องถามไหมต้องไม่ต้อง

กรณีที่ _saveChanges ต้องถามคือ: ผู้ใช้กำลังใช้ “Office” แล้วแก้ IP ของ “Office” — ค่าใหม่จะมีผลกับคำขอถัดไปทันทีอยู่แล้ว (เพราะ apiBaseUrl เป็น getter ที่อ่านสด, 20.3)

if (_controller.isActive(saved)) {
  final useNow = await _ask(title: 'ใช้ค่าใหม่ทันทีไหม?', ...);
  if (useNow == true && mounted) {
    await _controller.refreshStatus();
  }
}
แต่ตรงนี้มีสิ่งที่คำถามกับผลลัพธ์ไม่ตรงกัน
คำถามคือ “ให้ใช้ค่าที่แก้ทันทีเลยไหม”
สิ่งที่เกิดเมื่อตอบ “ใช้ทันที” คือ refreshStatus() — แค่ตรวจสถานะใหม่
สิ่งที่เกิดเมื่อตอบ “ยกเลิก” คือไม่ทำอะไร — แต่คำขอถัดไปก็ยังใช้ค่าใหม่อยู่ดี
เพราะค่าถูกบันทึกลงโปรไฟล์ที่กำลังใช้ไปแล้วก่อนถึงคำถามนี้

แปลว่าคำถามนี้ให้ทางเลือกที่ไม่มีอยู่จริง — ตอบอะไรก็ใช้ค่าใหม่เหมือนกัน ต่างแค่จุดสถานะจะถูกอัปเดตหรือไม่

ทางแก้มีสองแนว

แนวทำอะไร
แก้คำถามให้ตรงกับสิ่งที่ทำเปลี่ยนเป็น “ตรวจการเชื่อมต่อกับค่าใหม่เลยไหม” — เล็กและตรง
แก้พฤติกรรมให้ตรงกับคำถามต้องมีทางไม่ให้ค่าใหม่มีผล ซึ่งขัดกับการที่เพิ่งบันทึกทับไปแล้ว

แนวแรกถูก และควรทำ — คำถามที่ให้ทางเลือกลวงแย่กว่าไม่ถามเลย เพราะผู้ใช้จะเชื่อว่าตัวเองควบคุมบางอย่างอยู่

ข้อสังเกตของบทนี้ทั้งบท
จอ 1,016 บรรทัดนี้คิดครบเกือบทุกทางออก — สามทางเลือกเมื่อต่อไม่ติด, ชื่อซ้ำ, ตัวแทนเมื่อลบ, ด่านเดียวสำหรับสามทางปิด
สิ่งที่ยังไม่ตรงมีสองจุดเล็ก ๆ: ปุ่มที่ปิดแล้วไม่เป็นสีเทา และคำถามที่ให้ทางเลือกลวง
ทั้งสองไม่ใช่เรื่องของตรรกะ แต่เป็นเรื่องของสิ่งที่ผู้ใช้เชื่อว่ากำลังเกิดขึ้น — ซึ่งเป็นชั้นที่ตรวจด้วยการอ่านโค้ดอย่างเดียวไม่พอ ต้องลองใช้จริง

12. สรุป

  • แผ่นระดับ route เพราะต้องคลุมทั้งแถบบนและแถบล่างของเปลือก
  • enableDrag: false ทำให้ทุกทางออกผ่านด่านเดียวกัน
  • _isDirty คำนวณจากลายนิ้วมือของฟอร์ม ไม่ใช่ธงที่ต้องจำตั้ง
  • ผลการทดสอบถูกล้างทันทีที่ค่าเปลี่ยน — ผลเก่าต้องไม่พูดแทนค่าใหม่
  • ข้อความผิดพลาดโผล่หลังกดครั้งแรก แล้วอัปเดตสดหลังจากนั้น
  • สามปุ่มบันทึกทำคนละอย่าง และมีปุ่มเดียวที่สลับเซิร์ฟเวอร์
  • ต่อไม่ได้แล้วยังใช้ได้ ถ้าผู้ใช้เลือกเอง — เตือน ไม่ใช่ห้าม
  • ชื่อซ้ำเทียบแบบผ่อนปรน แต่เก็บตามที่พิมพ์
  • ด่านการลบซ้ำกับ controller โดยตั้งใจ เพราะคนละความรับผิดชอบ
  • โครงหัวคงที่/กลางเลื่อน/ท้ายคงที่ ทำให้ปุ่มไม่หายไปกับแป้นพิมพ์
  • ปุ่มที่ปิดยังไม่เป็นสีเทา เพราะ PdaButton ไม่รับ null
