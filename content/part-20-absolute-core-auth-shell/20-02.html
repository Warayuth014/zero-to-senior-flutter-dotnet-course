2. main() ทีละบรรทัด

Future<void> main() async {
  WidgetsFlutterBinding.ensureInitialized();          // ①
  await AppLanguageController.instance.load();        // ②
  // Must finish before the first request: AppConfig.apiBaseUrl falls back to the
  // built-in address until the operator's active profile is in memory.
  await ConnectionController.instance.load();         // ③

  // Match the system status bar to the PDA bar so they read as one strip.
  SystemChrome.setSystemUIOverlayStyle(_systemUiStyle);   // ④

  runApp(WmsApp(auth: AuthRepository(ApiClient())));      // ⑤
}

① WidgetsFlutterBinding.ensureInitialized()

Flutter มีตัวเชื่อมระหว่างโค้ด Dart กับระบบปฏิบัติการ — เรียกว่า binding

ปกติ runApp() จะสร้าง binding ให้เอง แต่ถ้าจะใช้ปลั๊กอิน (เช่น shared_preferences) ก่อน runApp() ต้องสร้างเองก่อน

ลืมบรรทัดนี้แล้วจะเจอ
Unhandled Exception: Binding has not yet been initialized.
เป็นข้อความที่อ่านไม่รู้เรื่องเลยถ้าไม่เคยเจอ — จำไว้ว่า async ใน main() + ปลั๊กอิน = ต้องมีบรรทัดนี้

② โหลดภาษา

Future<void> load() async {
  try {
    final saved = await SharedPreferencesAsync().getString(_preferenceKey);
    _language = saved == 'th' ? AppLanguage.thai : AppLanguage.english;
  } on StateError {
    // Widget tests and unsupported platforms may not register the plugin.
    _language = AppLanguage.english;
  }
}

อ่านจากที่เก็บของเครื่อง ถ้าอ่านไม่ได้ก็ใช้อังกฤษ

③ โหลดที่อยู่เซิร์ฟเวอร์

คอมเมนต์เหนือบรรทัดนี้บอกเหตุผลไว้ตรง ๆ

“ต้องเสร็จก่อนคำขอแรก เพราะ AppConfig.apiBaseUrl จะตกกลับไปใช้ที่อยู่ที่ฝังมากับตัวแอป จนกว่าโปรไฟล์ที่เลือกไว้จะอยู่ในหน่วยความจำ”

④ สีแถบสถานะ

ทำให้แถบสถานะของ Android เป็นสีเดียวกับแถบของแอป — จะขยายในบทที่ 5

⑤ สร้างของทุกอย่างแล้วเริ่ม

runApp(WmsApp(auth: AuthRepository(ApiClient())));

อ่านจากในออกนอก

ApiClient()                        ← ตัวส่ง HTTP
AuthRepository(ApiClient())        ← ตัวล็อกอิน ใช้ ApiClient
WmsApp(auth: ...)                  ← แอป ใช้ AuthRepository

3. ทำไมต้องรอ ไม่รอได้ไหม

สมมติเขียนแบบไม่รอ

void main() {
  WidgetsFlutterBinding.ensureInitialized();
  AppLanguageController.instance.load();      // ไม่มี await
  ConnectionController.instance.load();       // ไม่มี await
  runApp(...);
}

โค้ดนี้คอมไพล์ผ่าน และส่วนใหญ่ทำงานได้ — ซึ่งเป็นเหตุผลที่บั๊กแบบนี้หลุดไปถึงหน้างาน

5. WmsApp

class WmsApp extends StatelessWidget {
  const WmsApp({super.key, required this.auth});

  final AuthRepository auth;

  @override
  Widget build(BuildContext context) {
    return ListenableBuilder(
      listenable: AppLanguageController.instance,
      builder: (context, _) => MaterialApp(
        title: AppConfig.appTitle,
        debugShowCheckedModeBanner: false,
        theme: buildAppTheme(),
        locale: AppLanguageController.instance.locale,
        supportedLocales: const [Locale('en'), Locale('th')],
        localizationsDelegates: const [
          GlobalMaterialLocalizations.delegate,
          GlobalWidgetsLocalizations.delegate,
          GlobalCupertinoLocalizations.delegate,
        ],
        builder: (context, child) => ...,
        home: LoginScreen(auth: auth),
      ),
    );
  }
}

สังเกตสองอย่าง

ListenableBuilder ครอบ MaterialApp

ปกติจะเห็น ListenableBuilder ครอบส่วนเล็ก ๆ ในจอ — ที่นี่ครอบทั้งแอป

ListenableBuilder(
  listenable: AppLanguageController.instance,
  builder: (context, _) => MaterialApp(...),
)

แปลว่ากดสลับภาษาที่ไหนก็ได้ ทั้งแอปวาดใหม่ทันที โดยไม่ต้องส่งอะไรผ่านทุกจอ

ค่าใช้จ่ายของการวาดใหม่ทั้งแอปไม่สูงอย่างที่กลัว — Flutter เทียบ widget tree ใหม่กับของเดิม แล้ววาดเฉพาะที่ต่าง (4.4)
และการสลับภาษาเกิดวันละไม่กี่ครั้ง ไม่ใช่ทุกเฟรม

StatelessWidget แต่เก็บ auth

WmsApp ไม่มี state ของตัวเอง — auth เป็น final ที่รับมาแล้วส่งต่อลงไปให้ LoginScreen เท่านั้น

นี่คือรูปแบบที่ 9.10 เรียกว่า constructor injection — ของถูกสร้างข้างบนแล้วยื่นลงมา ไม่ใช่ให้ข้างล่างไปหยิบเอง

// ถ้าทำแบบ wmsapp
class LoginScreen extends StatefulWidget {
  final _api = ApiService();     // จอไปหยิบเอง — เทสต์เปลี่ยนไม่ได้
}

// ที่นี่
class LoginScreen extends StatefulWidget {
  const LoginScreen({super.key, required this.auth});
  final AuthRepository auth;     // รับมา — เทสต์ส่งของปลอมได้
}

6. ส่วน edge-to-edge

// Android edge-to-edge devices draw Flutter behind the native status
// bar. Keep every route below that inset while painting the exposed
// area with the same navy as the branded PDA strip.
builder: (context, child) => AnnotatedRegion<SystemUiOverlayStyle>(
  value: _systemUiStyle,
  child: ColoredBox(
    color: AppColors.statusBar,
    child: SafeArea(
      bottom: false,
      child: child ?? const SizedBox.shrink(),
    ),
  ),
),

สามชั้นซ้อนกัน แต่ละชั้นแก้ปัญหาคนละอย่าง

ชั้นแก้อะไร
AnnotatedRegionบังคับสี/ความสว่างไอคอนแถบสถานะซ้ำ เผื่อ route ใหม่มาเปลี่ยนทับ
ColoredBoxทาสีพื้นที่ใต้แถบสถานะให้เป็นน้ำเงินเข้มเดียวกับแถบแอป
SafeArea(bottom: false)ดันเนื้อหาลงมาใต้แถบสถานะ แต่ไม่ดันขึ้นจากขอบล่าง

bottom: false คือรายละเอียดที่คนมักพลาด — บนพีดีเอไม่มีแถบนำทางแบบท่าทางที่ขอบล่าง การเว้นขอบล่างจะกินพื้นที่ที่จอเล็กอยู่แล้วไปเปล่า ๆ

child ?? const SizedBox.shrink()

child ของ MaterialApp.builder เป็น Widget? — เป็น null ได้ในบางจังหวะ

ถ้าเขียน child! แล้วมันเป็น null จริง แอปจะล้ม — ที่นี่ใส่กล่องว่างขนาดศูนย์แทน

ทำไม const SizedBox.shrink() ไม่ใช่ Container()
SizedBox.shrink() คือกล่องขนาด 0×0 ที่วาดอะไรไม่เลย และเป็น const ได้ — สร้างครั้งเดียวใช้ตลอด
Container() ที่ไม่มีลูกจะกินพื้นที่เท่าที่พ่อให้ ซึ่งไม่ใช่สิ่งที่ต้องการ

8. แบบฝึกหัด

  1. สร้างโปรเจกต์เปล่า เขียน main() ที่เรียก SharedPreferencesAsync().getString('x') โดยไม่ใส่ ensureInitialized — อ่านข้อความผิดพลาดจริง
  2. ในโปรเจกต์นี้ ลบ await หน้า ConnectionController.instance.load() แล้วอธิบายว่าต้องทำอย่างไรถึงจะเห็นบั๊กจริงบนเครื่อง
แนวคำตอบ

ข้อ 1

Unhandled Exception: Binding has not yet been initialized.
The "instance" getter on the ServicesBinding binding mixin is only available
once that binding has been initialized.

เติม WidgetsFlutterBinding.ensureInitialized(); เป็นบรรทัดแรกแล้วหาย

ข้อ 2 — ทำไมบั๊กนี้จับยาก

ช่องว่างคือเวลาที่ใช้อ่านดิสก์ ซึ่งบนเครื่องพัฒนาเร็วมาก และคนกดล็อกอินไม่ทันอยู่แล้ว

วิธีทำให้เห็น

Future<void> load() async {
  await Future<void>.delayed(const Duration(seconds: 3));   // ชั่วคราว
  ...
}

แล้วรีบกดล็อกอินภายในสามวินาที — คำขอจะไปที่อยู่ที่ฝังมากับตัวแอป ไม่ใช่โปรไฟล์ที่เลือกไว้

ยืนยันได้จากบันทึกของเซิร์ฟเวอร์ หรือดักด้วย http.Client ปลอมที่พิมพ์ URI ออกมา

บทเรียนที่กว้างกว่าบั๊กนี้
บั๊กที่ขึ้นกับจังหวะเวลา จะไม่โผล่บนเครื่องพัฒนา เพราะเครื่องพัฒนาเร็วและคนทดสอบไม่รีบ
มันโผล่บนเครื่องจริงที่ช้ากว่า กับผู้ใช้จริงที่กดเร็วกว่าเพราะทำงานเดิมทุกวัน
วิธีตรวจคือถ่วงเวลาให้ช้าลงชั่วคราว แล้วดูว่ามีหน้าต่างที่กดผิดจังหวะได้หรือไม่ — เทคนิคนี้ใช้ได้กับทุกเรื่องที่มี async

ข้อสังเกตเพิ่ม — โปรเจกต์นี้ไม่มีจอ splash คั่น จอแรกที่ผู้ใช้เห็นคือจอล็อกอินเลย

แปลว่าไม่มีที่ให้ซ่อนการโหลด ถ้ามีอะไรต้องโหลดนานกว่านี้ในอนาคต ต้องเลือกระหว่างจอขาวนานขึ้น หรือเพิ่มจอคั่น — แต่ตอนนี้ยังไม่ต้อง

9. สรุป

  • main.dart 70 บรรทัด ไม่มีตรรกะธุรกิจเลย
  • ensureInitialized() จำเป็นเมื่อใช้ปลั๊กอินก่อน runApp()
  • รอสองอย่าง: ภาษา และที่อยู่เซิร์ฟเวอร์ — ตัวหลังมีคอมเมนต์บอกเหตุผลไว้
  • ไม่รอ = มีหน้าต่างที่คำขอแรกไปเซิร์ฟเวอร์ผิดตัว
  • เกณฑ์ว่าจะรออะไร: ผิดแล้วคำขอไปผิดที่ = รอ; แค่หน้าตา = โหลดทีหลัง
  • ListenableBuilder ครอบทั้ง MaterialApp ทำให้สลับภาษาแล้วทั้งแอปเปลี่ยน
  • runApp(WmsApp(auth: AuthRepository(ApiClient()))) คือ composition root ที่เดียวของโปรเจกต์
  • สามชั้น AnnotatedRegion/ColoredBox/SafeArea(bottom: false) จัดการ edge-to-edge
