3. load()

Future<void> load() async {
  String? raw;
  try {
    raw = await SharedPreferencesAsync().getString(_preferenceKey);
  } catch (_) {
    // Widget tests and unsupported platforms have no plugin registered.
    raw = null;
  }

  final snapshot = decodeConnectionSnapshot(raw);
  _profiles..clear()..addAll(snapshot.profiles);
  _activeId = snapshot.activeId;

  if (_profiles.isEmpty) {
    final seed = seedConnectionProfile();
    _profiles.add(seed);
    _activeId = seed.id;
    await _persist();
  }

  _loaded = true;
  notifyListeners();
}

5. refreshStatus

/// Called on shell mount, when the app returns to the foreground, and after
/// switching profiles — there is no polling timer, which would cost battery
/// and radio on a handheld all day for a dot nobody is watching.
Future<ConnectionStatus> refreshStatus({http.Client? client}) async {
  // Resolved the same way requests are, so the dot can never report on a
  // different server than the one the app is actually calling.
  final url = resolveBaseUrl(activeBaseUrl);

  _status = ConnectionStatus.connecting;
  notifyListeners();                          // ① แจ้งระหว่างทาง

  final result = await probeConnection(baseUrl: url, client: client);
  _lastResult = result;
  _status = result.ok
      ? ConnectionStatus.connected
      : ConnectionStatus.disconnected;
  notifyListeners();                          // ② แจ้งผล
  return _status;
}

แจ้งสองครั้ง

ครั้งแรกทำให้จุดกลายเป็นสถานะ “กำลังตรวจ” ทันที ผู้ใช้จึงเห็นว่ามีอะไรเกิดขึ้น — ไม่ใช่จุดที่ค้างอยู่จนกว่าผลจะมา

นี่คือเหตุผลที่ ConnectionStatus มีสี่ค่า ไม่ใช่ bool

enum ConnectionStatus { unknown, connecting, connected, disconnected }
ค่าแปลว่า
unknownยังไม่เคยตรวจ / เพิ่งสลับโปรไฟล์
connectingกำลังตรวจอยู่
connectedตรวจแล้วต่อได้
disconnectedตรวจแล้วต่อไม่ได้

ถ้าใช้ bool จะแยก “ยังไม่เคยตรวจ” ออกจาก “ตรวจแล้วไม่ผ่าน” ไม่ได้ — ซึ่งเป็นสองสถานะที่ควรแสดงต่างกัน (4.14)

ไม่มีตัวจับเวลา

คอมเมนต์ให้เหตุผลเชิงฮาร์ดแวร์ที่ตรงประเด็น
ยิงทุก 30 วินาทีตลอดกะแปดชั่วโมง = 960 คำขอ เพื่อรักษาจุดเล็ก ๆ ที่ไม่มีใครมองให้เป็นปัจจุบัน
บนพีดีเอที่ต้องอยู่ได้ทั้งกะโดยไม่ชาร์จ วิทยุ Wi-Fi คือตัวกินแบตหลัก
ทางที่เลือกคือตรวจในจังหวะที่คำตอบมีค่า — เข้าแอป กลับมาจากพัก และหลังสลับโปรไฟล์
เทียบกับ WarehouseStore ที่มีตัวจับเวลา 5 วินาที (20.14) — ความต่างคือสถานะของงานเปลี่ยนโดยที่ผู้ใช้ไม่ได้ทำอะไร ส่วนที่อยู่เซิร์ฟเวอร์ไม่เปลี่ยนเอง

6. @visibleForTesting สองตัว

/// Publishes a status without running a probe, so UI bound to the dot can be
/// tested for every state.
@visibleForTesting
void debugSetStatus(ConnectionStatus status) { ... }

/// Puts the controller into a known state for tests without touching disk.
@visibleForTesting
void debugReset({List<ConnectionProfile> profiles = const [], String? activeId}) {
  _profiles..clear()..addAll(profiles);
  _activeId = activeId ?? (profiles.isEmpty ? null : profiles.first.id);
  _status = ConnectionStatus.unknown;
  _lastResult = null;
  _loaded = profiles.isNotEmpty;
  notifyListeners();
}

@visibleForTesting ทำอะไร

เป็นคำสั่งถึงตัววิเคราะห์ ไม่ใช่คอมไพเลอร์ — ถ้ามีโค้ดใน lib/ เรียกเมธอดนี้ จะขึ้นคำเตือน แต่โค้ดใน test/ เรียกได้ปกติ

info: The member 'debugReset' can only be used within tests.
นี่คือคำตอบของปัญหาที่ 20.8 ทิ้งไว้
AppLanguageController เป็น singleton ที่ไม่มีทางรีเซ็ต เทสต์จึงต้อง addTearDown เองและอาจลืม
ConnectionController เป็น singleton เหมือนกัน แต่มีทางออกที่ประกาศไว้ชัด
setUp(() {
  ConnectionController.instance.debugReset(profiles: [profileA, profileB]);

7. _persist() ที่กลืนความล้มเหลว

Future<void> _persist() async {
  try {
    await SharedPreferencesAsync().setString(
      _preferenceKey,
      encodeConnectionSnapshot(_profiles, _activeId),
    );
  } catch (_) {
    // Settings still apply for this session even if the write fails.
  }
}

ทุกเมธอดที่แก้ข้อมูลเรียก _persist() แล้วค่อย notifyListeners()

_profiles.add(saved);
await _persist();          // เขียนก่อน
notifyListeners();         // แจ้งทีหลัง
ลำดับตรงข้ามกับ AppLanguageController.setLanguage ที่แจ้งก่อนบันทึก (20.8)
ต่างกันเพราะการเขียนที่นี่เร็ว (ไม่มีการรอเครือข่าย) และเป็นการกระทำที่ผู้ใช้กำลังรอผลอยู่แล้ว — เขากดปุ่ม “บันทึก”
ส่วนการสลับภาษาต้องเห็นผลทันที เพราะผู้ใช้คาดหวังว่าจอจะเปลี่ยนพร้อมนิ้ว
ทั้งสองลำดับถูกในบริบทของตัวเอง แต่ควรมีคอมเมนต์บอกว่าเลือกลำดับนี้เพราะอะไร ไม่งั้นคนอ่านจะคิดว่าเป็นความไม่สม่ำเสมอ

การกลืนความล้มเหลวถูกในบริบทนี้ — ถ้าเขียนดิสก์ไม่ได้ ผู้ใช้ยังทำงานต่อได้ในกะนี้ แค่ค่าจะไม่อยู่หลังปิดแอป

แต่ผู้ใช้ไม่ได้ถูกบอก ซึ่งเป็นข้อจำกัดที่ควรรู้ — เขาจะเจอเองตอนเปิดแอปครั้งหน้าแล้วค่าหาย

8. ข้อผิดพลาดที่พบบ่อย

ผิดอาการถูก
import วนกันระหว่างสอง singletonลำดับการสร้างเดาไม่ได้ให้ทิศทางเดียว + คอมเมนต์กำกับ
เก็บวัตถุที่เลือกแทน idแก้แล้วยังชี้ตัวเก่าเก็บ id แล้วค้น
บันทึกแล้วสลับให้เลยงานไปลงเครื่องผิดโดยไม่รู้แยกบันทึกกับสลับ
ไม่รีเซ็ตสถานะตอนสลับจุดเขียวของเซิร์ฟเวอร์ตัวก่อนตั้งเป็น unknown
ให้ลบโปรไฟล์สุดท้ายได้ไม่มีที่อยู่ให้ใช้และพิมพ์ใหม่ไม่ได้ปฏิเสธแล้วคืน false
ใช้ bool แทนสถานะสี่ค่า“ยังไม่ตรวจ” กับ “ตรวจแล้วไม่ผ่าน” เหมือนกันenum
ยิงตรวจสถานะเป็นรอบตลอดวันแบตหมดก่อนจบกะตรวจเมื่อคำตอบมีค่า
singleton ที่รีเซ็ตไม่ได้เทสต์ล้มตามลำดับการรัน@visibleForTesting + debugReset

9. แบบฝึกหัด

  1. ผู้ใช้มีสามโปรไฟล์ กำลังใช้ตัวที่สอง แล้วลบตัวที่สองโดยไม่เลือกตัวแทน — เดินตาม remove ทีละบรรทัด
  2. _persist() ล้มเงียบ ๆ — ออกแบบวิธีบอกผู้ใช้ โดยไม่ทำให้ทุกเมธอดยุ่งขึ้น
แนวคำตอบ

ข้อ 1

_profiles = [A, B, C], _activeId = 'B'
remove('B', nextActiveId: null)

length <= 1 ?              → 3 ไม่ใช่ ไปต่อ
indexWhere('B')            → 1
removeAt(1)                → [A, C]
_activeId == 'B' ?         → ใช่
  replacement = null ?? _profiles.first.id  → 'A'
  มี 'A' อยู่จริงไหม        → ใช่
  _activeId = 'A'
  _status = unknown
_persist()
notifyListeners()
return true

ผลคือ [A, C] ใช้ตัว A และจุดสถานะกลับไปเป็น “ยังไม่ทราบ”

สังเกตว่าโค้ดยอมให้ nextActiveId เป็น null ได้ ทั้งที่คอมเมนต์เขียนว่า “must be supplied by the caller when the active one is being removed”
คอมเมนต์บอกข้อตกลง แต่โค้ดไม่ได้บังคับ — มันมีทางถอยให้เงียบ ๆ
ในกรณีนี้ทางถอยสมเหตุสมผล (ใช้ตัวแรก) จึงไม่เป็นบั๊ก แต่คอมเมนต์กับโค้ดพูดไม่ตรงกัน คนอ่านที่เชื่อคอมเมนต์จะคิดว่าลืมส่งแล้วจะพัง
เขียนคอมเมนต์ให้ตรงจะดีกว่า: “ถ้าไม่ส่ง จะใช้โปรไฟล์แรกแทน — จอควรถามผู้ใช้ก่อน”

ข้อ 2

สิ่งที่ต้องเลี่ยงคือให้ทุกเมธอดคืน bool แล้วให้จอตรวจทีละที่

ทางที่พอดีคือเพิ่ม state ตัวเดียวใน controller แล้วให้ UI ที่มีอยู่แล้วแสดงผล

bool _persistFailed = false;
bool get persistFailed => _persistFailed;

Future<void> _persist() async {
  try {
    await SharedPreferencesAsync().setString(...);
    _persistFailed = false;
  } catch (_) {
    _persistFailed = true;    // ไม่ต้อง notify — ผู้เรียกทำอยู่แล้ว
  }
}

แล้วในแผ่นตั้งค่า (20.13) ซึ่งฟัง controller อยู่แล้ว

if (controller.persistFailed)
  const _Banner(text: 'บันทึกลงเครื่องไม่สำเร็จ — ค่าจะหายเมื่อปิดแอป'),
ได้ทำไม
ไม่ต้องแก้ลายเซ็นเมธอดไหนเลยข้อมูลไหลผ่านทางที่มีอยู่แล้ว
แสดงที่เดียวไม่กระจายการจัดการข้อผิดพลาดไปทุกจุดกด
ทดสอบได้เป็น getter ธรรมดา
รูปแบบทั่วไปที่ได้จากข้อนี้
เมื่อความล้มเหลวไม่ได้ขัดขวางการกระทำ แต่ผู้ใช้ควรรู้ อย่าเปลี่ยนให้การกระทำนั้นล้ม
ให้เก็บเป็นสถานะแล้วแสดงในที่ที่ผู้ใช้จะเห็นตอนที่มันสำคัญ
ต่างจากความล้มเหลวที่ขัดขวางจริง (ยืนยันงานไม่สำเร็จ) ซึ่งต้องหยุดและบอกทันที

10. สรุป

  • คอมเมนต์หัวคลาสห้าม import app_config.dart เพื่อไม่ให้เกิดวงจรที่ทำให้ลำดับเริ่มแอปเดาไม่ได้
  • เก็บ _activeId ไม่ใช่วัตถุ — ค้นใหม่ทุกครั้งจึงได้ตัวที่แก้แล้วเสมอ
  • load() เพาะโปรไฟล์แรกด้วยที่อยู่เดิม เพื่อให้คนที่อัปเกรดไม่ต้องทำอะไรเพิ่ม
  • บันทึกกับสลับเป็นคนละการกระทำโดยตั้งใจ
  • activate และ remove รีเซ็ตสถานะเป็น unknown เพื่อไม่ให้จุดพูดถึงเซิร์ฟเวอร์ตัวก่อน
  • ปฏิเสธการลบโปรไฟล์สุดท้าย และคืน bool ให้จอบอกผู้ใช้ได้
  • ConnectionStatus มีสี่ค่า แยก “ยังไม่ตรวจ” ออกจาก “ตรวจแล้วไม่ผ่าน”
  • ไม่มีตัวจับเวลา — ตรวจในจังหวะที่คำตอบมีค่า เพื่อประหยัดแบตของพีดีเอ
  • @visibleForTesting + debugReset() คือทางออกของ singleton ที่เทสต์ต้องรีเซ็ต
