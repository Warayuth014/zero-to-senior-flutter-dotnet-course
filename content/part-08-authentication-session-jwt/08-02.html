ระดับ: กลางใช้เวลา: 30–40 นาทีลงมือทำ: จัดโครงชั้นที่ดูแลการเข้าสู่ระบบ
ต้องผ่านอะไรมาก่อน: 8.1 (เส้นทางการเข้าสู่ระบบ) และ 7.1 (การแยกชั้น)
จบบทนี้แล้วจะทำได้: ตอบได้ว่าใครเป็นเจ้าของ token และวางโครงให้ทดสอบได้โดยไม่ต้องมีเซิร์ฟเวอร์

1. ปัญหาที่บทนี้แก้

คำถามที่ต้องตอบก่อนเขียนโค้ด — ใครถือ token

ทางเลือกปัญหาที่ตามมา
ตัวแปรระดับแอปที่ใครก็แก้ได้ไม่รู้ว่าใครแก้ และทดสอบไม่ได้
หน้าเข้าสู่ระบบถือไว้หน้านั้นถูกทำลายไปแล้วตอนใช้งานจริง
ApiClient ถือไว้อย่างเดียวไม่มีใครรู้ว่าตอนนี้เข้าสู่ระบบอยู่หรือไม่
แต่ละ store ถือของตัวเองออกจากระบบแล้วบางส่วนยังใช้ token เก่า

แถวสุดท้ายคืออาการที่ 7.5 แบบฝึกหัดข้อ 2 พูดถึง — และเป็นเหตุผลที่ต้องมีเจ้าของชัดเจนเพียงหนึ่งเดียว

2. คำศัพท์ที่ต้องรู้ก่อน

คำความหมาย
repositoryชั้นที่รู้ว่าคุยกับเซิร์ฟเวอร์อย่างไร
storeชั้นที่ถือสถานะและตัดสินใจ
source of truthที่เดียวที่ถือความจริง
callbackฟังก์ชันที่ส่งเข้าไปให้เรียกกลับเมื่อมีอะไรเปลี่ยน

3. Mental model: สองชั้นที่ทำคนละเรื่อง

AuthRepository   รู้ว่าเข้าสู่ระบบอย่างไร    →  ไม่จำอะไรเลย
SessionStore     รู้ว่าตอนนี้เป็นใครอยู่      →  จำ และบอกคนอื่น
ประโยคเดียวที่ต้องจำ: repository ตอบคำถาม “รหัสผ่านนี้ถูกไหม” ครั้งเดียวแล้วจบ ส่วน store ตอบคำถาม “ตอนนี้ใครใช้อยู่” ได้ตลอดเวลา

5. การแบ่งงานที่เห็นได้จากโครงสร้าง try

    'password': password,
  });
} on ApiException catch (error) {
  return switch (error.kind) {
    ApiFailureKind.unauthorized => const InvalidCredentials(),
    ApiFailureKind.forbidden => AccountNotAllowed(error.userMessage),
    _ => LoginFailed(error.kind, error.userMessage),
  };

return LoginSucceeded(

ตำแหน่งของวงเล็บปีกกาปิดเป็นการตัดสินใจ ไม่ใช่เรื่องรูปแบบ

อาการสาเหตุวิธีแก้
ออกจากระบบแล้วบางส่วนยังใช้ token เก่าtoken อยู่หลายที่ให้ store เดียวเป็นเจ้าของ
คำตอบผิดสัญญากลายเป็น “รหัสผ่านผิด”try ครอบกว้างเกินครอบเฉพาะการยิงคำขอ
ทดสอบ store ต้องเปิดเซิร์ฟเวอร์ผูกกับคลาสจริงรับสัญญาเข้ามา
ทดสอบเรื่องหมดอายุต้องรอเวลาจริงเรียก DateTime.now() ข้างในรับ clock เข้ามา

11. แบบฝึกหัด

ทีมต้องเพิ่มความสามารถให้ผู้ใช้เปลี่ยนรหัสผ่านจากในแอปได้

สัญญาที่ได้มา
POST /api/Authenticate/change-password ต้องมี token
ส่ง {"currentPassword": "...", "newPassword": "..."}
สำเร็จได้ 204 และ token เดิมถูกยกเลิกทันที
รหัสผ่านปัจจุบันผิดได้ 401
  1. ตัดสินว่าเมธอดนี้ควรอยู่ที่ AuthRepository หรือที่อื่น
  2. ออกแบบผลลัพธ์
  3. ข้อกำหนดที่ว่า token เดิมถูกยกเลิก สร้างปัญหาอะไร และแก้อย่างไร
ดูแนวคำตอบและเหตุผล

ข้อ 1 — อยู่ที่ไหน

อยู่ที่ AuthRepository เพราะเป็นเรื่องของการยืนยันตัวตน ไม่ใช่เรื่องของงานคลัง

abstract interface class AuthRepository {
  Future<LoginOutcome> login({...});
  Future<ChangePasswordOutcome> changePassword({
    required String currentPassword,
    required String newPassword,
  });
}

สังเกตว่าไม่ต้องส่ง username หรือ serverId — เพราะ token ที่แนบไปกับคำขอบอกอยู่แล้วว่าเป็นใครและคุยกับเซิร์ฟเวอร์ไหน

ข้อ 2 — ผลลัพธ์

sealed class ChangePasswordOutcome {
  const ChangePasswordOutcome();
}

/// สำเร็จ — และ token เดิมใช้ไม่ได้แล้ว
class PasswordChanged extends ChangePasswordOutcome {
  const PasswordChanged();
}

/// รหัสผ่านปัจจุบันผิด — token ยังใช้ได้อยู่
class CurrentPasswordWrong extends ChangePasswordOutcome {
  const CurrentPasswordWrong();
}

/// รหัสผ่านใหม่ไม่ผ่านกฎของระบบ
class NewPasswordRejected extends ChangePasswordOutcome {
  const NewPasswordRejected(this.reasons);
  final List<String> reasons;      // จาก errors ใน ProblemDetails (7.11)
}

class ChangePasswordFailed extends ChangePasswordOutcome {
  const ChangePasswordFailed(this.kind, this.message);
  final ApiFailureKind kind;
  final String message;
}

ปัญหาที่ซ่อนอยู่ในข้อนี้ — รหัส 401 มีสองความหมายที่ต้องแยก

401 มาจากแปลว่าต้องทำ
รหัสผ่านปัจจุบันผิดผู้ใช้พิมพ์ผิดให้พิมพ์ใหม่ — ห้ามออกจากระบบ
token หมดอายุsession จบแล้วออกจากระบบ (8.7)

ถ้าแยกไม่ออก ผู้ใช้ที่พิมพ์รหัสผ่านเดิมผิดจะถูกเด้งออกจากระบบทันที ซึ่งเป็นอาการเดียวกับที่ 7.5 เตือนไว้เรื่อง 401 กับ 403

ทางแก้คือขอ errorCode จากฝั่งหลังบ้าน ตามหลักใน 7.9

return switch (error.problem?.errorCode) {
  'CURRENT_PASSWORD_INVALID' => const CurrentPasswordWrong(),
  'TOKEN_EXPIRED' => ChangePasswordFailed(error.kind, error.userMessage),
  _ => ChangePasswordFailed(error.kind, error.userMessage),
};

ข้อ 3 — token เดิมถูกยกเลิก

นี่คือประเด็นหลักของแบบฝึกหัด — คำขอที่สำเร็จ ทำให้ token ที่แอปถืออยู่ใช้ไม่ได้ทันที

ผู้ใช้เปลี่ยนรหัสผ่านสำเร็จ
  → server ยกเลิก token เดิม
  → แอปยังถือ token เดิมอยู่ และคิดว่ายังเข้าสู่ระบบอยู่
  → คำขอถัดไปทุกคำขอได้ 401
  → ถ้ามี polling อยู่ (4.11) จะได้ 401 รัว ๆ ทันที

สิ่งที่ผู้ใช้เห็นคือ “เปลี่ยนรหัสผ่านสำเร็จ” แล้วทุกอย่างพังพร้อมกัน

ทางแก้ที่ผิด

ทำอะไรทำไมไม่ดี
ปล่อยให้ได้ 401 แล้วค่อยออกจากระบบผู้ใช้เห็นข้อความ “session หมดอายุ” ทั้งที่เพิ่งเปลี่ยนรหัสผ่านสำเร็จ — สับสน
เก็บรหัสผ่านใหม่ไว้แล้วเข้าสู่ระบบเงียบ ๆต้องเก็บรหัสผ่าน ซึ่งขัดกับ 8.1

ทางแก้ที่ถูก — ให้ store จัดการทันทีที่รู้ผล

Future<ChangePasswordOutcome> changePassword({...}) async {
  final outcome = await repository.changePassword(...);

  if (outcome is PasswordChanged) {
    // token เดิมใช้ไม่ได้แล้ว ต้องออกจากระบบก่อนที่คำขออื่นจะได้ 401
    await signOut(reason: SignOutReason.passwordChanged);
  }

  return outcome;
}

แล้วเพิ่มเหตุผลใหม่เข้าไป เพื่อให้หน้าจอบอกได้ตรง

enum SignOutReason {
  userRequested,
  sessionExpired,
  serverChanged,
  /// เปลี่ยนรหัสผ่านแล้ว ต้องเข้าสู่ระบบใหม่ด้วยรหัสใหม่
  passwordChanged,
}
// ข้อความที่ผู้ใช้ควรเห็น
'เปลี่ยนรหัสผ่านเรียบร้อย กรุณาเข้าสู่ระบบใหม่ด้วยรหัสผ่านใหม่'

ต่างจาก “session หมดอายุ” อย่างสิ้นเชิง ทั้งที่กลไกข้างล่างเหมือนกันทุกอย่าง

บทเรียนที่ควรจำ: คำขอที่ทำให้ token ของตัวเองใช้ไม่ได้ ต้องจัดการผลข้างเคียงนั้นทันทีที่รู้ผล ไม่ใช่รอให้คำขออื่นไปเจอเอง — และเหตุผลของการออกจากระบบต้องเก็บไว้ เพราะมันเปลี่ยนข้อความที่ผู้ใช้เห็นทั้งหมด

12. สรุปท้ายบท

  • repository ตอบครั้งเดียวแล้วจบ ส่วน store จำและบอกคนอื่น
  • token ต้องมีเจ้าของเดียวในแอป
  • try ครอบเฉพาะการยิงคำขอ ไม่ครอบการอ่านคำตอบ
  • ไม่มี logout ที่ repository เพราะออกจากระบบเป็นเรื่องฝั่งแอป
  • ใช้ callback บอก ApiClient แทนการถือมันไว้
  • สถานะต้องมี unknown สำหรับช่วงที่ยังอ่านข้อมูลไม่เสร็จ
  • กันการกดซ้ำด้วยการคืนค่าที่อยู่ในชนิดเดียวกัน
  • รับ clock เข้ามาเพื่อทดสอบเรื่องเวลาได้

บทถัดไป: 8.3 จะเปิดดูข้างใน token ว่ามีอะไร และตอบว่าทำไมฝั่งแอปอ่านได้แต่เชื่อไม่ได้
