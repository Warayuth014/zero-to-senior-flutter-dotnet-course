1. โจทย์

สร้างหน้าแรกของแอปคลังที่แสดงเมนูหลัก 6 รายการ โดยต้องผ่านเกณฑ์ทั้งหมดนี้

เกณฑ์มาจากบท
บนจอ 320×568 และ 400×800 ต้องไม่มีแถบสีเหลืองดำ2.14
จอแคบใช้ 2 คอลัมน์ จอกว้างเพิ่มคอลัมน์และจำกัดความกว้างเนื้อหา2.7, 2.10
เมนูที่เกินหน้าจอต้องเลื่อนได้ ด้วยพื้นที่เลื่อนชั้นเดียว2.6
ตัวเลขงานค้างต้องมีทั้งสี ไอคอน และข้อความ2.12
ชื่อเมนูยาวและตัวอักษร 200% ต้องยังใช้งานได้2.11, 2.9
พื้นที่กดของแต่ละช่องอย่างน้อย 482.13
แตะแล้วส่งรหัสเมนูกลับให้แม่ ไม่เรียก API เอง1.12

3. ขั้นที่ 2: รวมการตัดสินใจเรื่อง layout ไว้ที่เดียว

class HomeLayout {
  const HomeLayout({
    required this.columns,
    required this.compact,
    required this.gap,
  });

  factory HomeLayout.from(BoxConstraints constraints) {
    final compact = constraints.maxWidth < 480 || constraints.maxHeight < 650;
    final columns = switch (constraints.maxWidth) {
      >= 900 => 4,
      >= 600 => 3,
      _ => 2,
    };
    return HomeLayout(
      columns: columns,
      compact: compact,
      gap: compact ? 8 : 12,
    );
  }

  final int columns;
  final bool compact;
  final double gap;
}

ประโยชน์สองข้อตามที่คุยไว้ใน 2.10 — ตัวเลขทั้งหมดอยู่ที่เดียว และทดสอบได้โดยไม่ต้องประกอบหน้าจอ

test('จอ PDA ได้ 2 คอลัมน์และโหมด compact', () {
  final layout = HomeLayout.from(const BoxConstraints(maxWidth: 320, maxHeight: 568));
  expect(layout.columns, 2);
  expect(layout.compact, isTrue);
});

4. ขั้นที่ 3: ประกอบหน้าจอ

Scaffold(
  appBar: AppBar(title: const Text('เมนูหลัก')),
  body: SafeArea(
    child: Center(
      child: ConstrainedBox(
        // จำกัดความกว้างบนแท็บเล็ต ไม่ให้ยืดจนอ่านยาก (2.4)

              gridDelegate: SliverGridDelegateWithFixedCrossAxisCount(

                compact: layout.compact,
                onTap: () => onSelected(menus[index].id), // แจ้งกลับ ไม่ตัดสินใจเอง (1.12)
              ),
            );
          },
        ),
      ),
    ),
  ),
)

สังเกตลำดับของ wrapper ซึ่งมีเหตุผลทุกชั้น

ชั้นทำไมอยู่ตรงนี้
SafeAreaอยู่นอกสุด เพื่อหลบรอยบากก่อนคำนวณอย่างอื่น (2.9)
ConstrainedBoxจำกัดความกว้าง ก่อน LayoutBuilder เพื่อให้ค่าที่วัดได้เป็นค่าหลังจำกัดแล้ว
LayoutBuilderวัดพื้นที่ที่ได้จริง ไม่ใช่ขนาดจอ (2.10)
GridViewเป็นเจ้าของการเลื่อนคนเดียว (2.6)
ถ้าสลับ ConstrainedBox ไปไว้ข้างใน LayoutBuilder จะได้ค่าความกว้างเป็น 1400 บนจอกว้าง แล้วคำนวณเป็น 4 คอลัมน์ ทั้งที่พื้นที่จริงถูกจำกัดไว้ที่ 1100 — เป็นบั๊กแบบเดียวกับแบบฝึกหัดใน 2.10 ทุกประการ

6. ขั้นที่ 5: ตัวเลขงานค้างที่แยกออกได้โดยไม่ต้องใช้สี

Widget _icon(ThemeData theme) {
  if (menu.pending == 0) {
    return Icon(menu.icon, size: compact ? 28 : 36);
  }

  return Semantics(
    // โปรแกรมอ่านหน้าจอได้ยินข้อความเต็ม ไม่ใช่แค่ตัวเลข (2.13)
    label: '${menu.title} งานค้าง ${menu.pending} รายการ',
    child: Badge(
      label: Text(menu.pending > 99 ? '99+' : '${menu.pending}'),
      backgroundColor: theme.colorScheme.error,
      child: Icon(menu.icon, size: compact ? 28 : 36),
    ),
  );
}

ครบสามช่องทางตาม 2.12 — สีแดงสำหรับคนที่กวาดตาเร็ว ตัวเลขสำหรับทุกคน และข้อความเต็มสำหรับโปรแกรมอ่านหน้าจอ

7. ขั้นที่ 6: เขียนเทสต์ตามเกณฑ์

แปลงเกณฑ์ทุกข้อในหัวข้อ 1 ให้เป็นเทสต์ ใช้รูปแบบเดียวกับ test/layout_basics_test.dart

Future<void> pumpHome(
  WidgetTester tester, {
  required double width,
  double height = 800,
  double textScale = 1.0,
  ValueChanged<String>? onSelected,
}) async {
  tester.view.physicalSize = Size(width, height);
  tester.view.devicePixelRatio = 1.0;
  addTearDown(tester.view.reset);

  await tester.pumpWidget(
    MediaQuery(
      data: MediaQueryData(textScaler: TextScaler.linear(textScale)),
      child: MaterialApp(
        home: HomeScreen(menus: demoMenus, onSelected: onSelected ?? (_) {}),
      ),
    ),
  );
  await tester.pumpAndSettle();
}
testWidgets('จอ PDA 320x568 ไม่ล้น', (tester) async {
  await pumpHome(tester, width: 320, height: 568);
  expect(tester.takeException(), isNull);              // 2.14
});

testWidgets('ตัวอักษร 200% ยังไม่ล้น', (tester) async {
  await pumpHome(tester, width: 320, height: 568, textScale: 2.0);
  expect(tester.takeException(), isNull);              // 2.9
});

testWidgets('พื้นที่กดของแต่ละช่องอย่างน้อย 48', (tester) async {
  await pumpHome(tester, width: 320, height: 568);
  final size = tester.getSize(find.byKey(const ValueKey<String>('receive')));
  expect(size.height, greaterThanOrEqualTo(48));       // 2.13
});

testWidgets('แตะแล้วส่งรหัสเมนูกลับ ไม่ใช่ตำแหน่ง', (tester) async {
  String? selected;
  await pumpHome(tester, width: 320, onSelected: (id) => selected = id);

  await tester.tap(find.byKey(const ValueKey<String>('receive')));
  expect(selected, 'receive');                          // 1.12
});

testWidgets('เมนูที่เกินหน้าจอเลื่อนถึงได้', (tester) async {
  await pumpHome(tester, width: 320, height: 568);

  await tester.drag(find.byType(GridView), const Offset(0, -300));
  await tester.pumpAndSettle();

  expect(find.byKey(const ValueKey<String>('settings')), findsOneWidget);
  expect(tester.takeException(), isNull);               // 2.6
});

ข้อสังเกตสำคัญของเทสต์ตัวที่สี่ — มันตรวจว่าได้ 'receive' ไม่ใช่ 0 ซึ่งพิสูจน์ว่าโค้ดใช้รหัสจริงเป็นตัวตน ไม่ใช่ตำแหน่ง ตามที่เตือนไว้ใน 1.8 และ 2.7

8. ขั้นที่ 7: ตรวจด้วยตาตัวเอง

เทสต์จับได้หลายอย่างแต่ไม่ใช่ทุกอย่าง ให้รันจริงแล้วตรวจรายการนี้

ตรวจอะไรผ่านเมื่อ
ย่อหน้าต่างเบราว์เซอร์จากกว้างไปแคบช้า ๆจำนวนคอลัมน์เปลี่ยนอย่างราบรื่น ไม่มีแถบเหลืองแวบ
แปลงภาพหน้าจอเป็นขาวดำยังแยกเมนูที่มีงานค้างออกได้
ตั้งขนาดตัวอักษรในระบบเป็นใหญ่สุดอ่านได้ครบ ปุ่มยังกดได้
ใส่ชื่อเมนูยาว 60 ตัวอักษรตัดด้วยจุดไข่ปลา ไม่ล้น
เปิดโหมดมืดสีทุกอย่างยังอ่านออก

9. แบบทดสอบตัวเองก่อนไป Part 3

คุณพร้อมไปต่อเมื่อทำสิ่งเหล่านี้ได้โดยไม่เปิดดูเฉลย

  • อธิบายกฎสามข้อของ layout ด้วยภาษาของตัวเอง
  • บอกได้ว่า error เรื่องขอบเขตไม่จำกัดเกิดจากอะไรและแก้อย่างไร
  • เลือกระหว่าง Expanded กับ Flexible ได้พร้อมเหตุผล
  • เลือกระหว่าง SingleChildScrollView กับ ListView.builder ได้
  • บอกได้ว่าเมื่อไหร่ใช้ LayoutBuilder เมื่อไหร่ใช้ MediaQuery
  • ระบุได้ว่าข้อมูลชิ้นไหนห้ามตัดด้วยจุดไข่ปลา
  • ไล่หาสาเหตุของแถบสีเหลืองดำได้อย่างเป็นระบบ
  • เขียนเทสต์ที่ตรวจว่า layout ไม่ล้นที่ขนาดจอและขนาดตัวอักษรที่กำหนด

10. สรุปปิด Part 2

  • Layout ทั้งหมดอธิบายได้ด้วยกฎสามข้อ — ข้อจำกัดไหลลง ขนาดไหลขึ้น แม่วางตำแหน่ง
  • เนื้อหาล้นคือหลักฐาน ไม่ใช่ความรำคาญ — มันบอกว่าเรายังไม่ได้ตัดสินใจเรื่องลำดับความสำคัญ
  • ตัดสินใจล่วงหน้าเสมอว่าอะไรตรึง อะไรยืด อะไรเลื่อน
  • เมื่อพื้นที่ไม่พอ ให้เปลี่ยนโครงก่อน ย่อขนาดเป็นทางสุดท้าย
  • ข้อมูลที่ใช้ตัดสินใจทำงานห้ามถูกตัดหรือย่อ
  • PDA ไม่ใช่มือถือจอเล็ก — มันมีข้อกำหนดของตัวเอง
  • ข้อกำหนดด้านการใช้งานที่เขียนเป็นเทสต์ได้ จะไม่ถูกลืม

ต่อจากนี้: Part 3 จะเข้าสู่การรับข้อมูลจากผู้ใช้และจากเครื่องสแกน ซึ่งเป็นหัวใจของการทำงานในคลัง — และเป็นที่ที่เรื่องโฟกัส การกันยิงซ้ำ และการคืนทรัพยากรจาก Part 1 จะถูกใช้งานจริงทั้งหมด
