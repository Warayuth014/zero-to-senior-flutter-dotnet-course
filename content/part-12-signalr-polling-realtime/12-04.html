ระดับ: สูงใช้เวลา: 60–75 นาทีลงมือทำ: รันเทสต์การหลุดและการต่อกลับใน lab
ต้องผ่านอะไรมาก่อน: 12.3, 7.12 (ลองใหม่แบบถอยหลัง)
จบบทนี้แล้วจะทำได้: จัดการการหลุดและการต่อกลับ โดยไม่ปล่อยให้ข้อมูลค้างเก่าเงียบ ๆ

1. การเชื่อมต่อค้างจะหลุด ไม่ใช่ว่าอาจหลุด

สาเหตุเกิดบ่อยแค่ไหนในคลัง
เดินระหว่างจุด Wi-Fiทุกไม่กี่นาที
เข้าไปในห้องเย็นที่สัญญาณอ่อนทุกกะ
ระบบพักหน้าจอ แล้วระบบปฏิบัติการตัดเครือข่ายทุกครั้งที่วางเครื่องไว้
เซิร์ฟเวอร์ถูกรีสตาร์ทตอนอัปเดตทุกสัปดาห์
ตัวโหลดสมดุลตัดการเชื่อมต่อที่ค้างนานเกินทุกชั่วโมง ถ้าตั้งค่าไว้แบบนั้น

ดังนั้นการต่อกลับไม่ใช่กรณีพิเศษที่ต้องเผื่อไว้ แต่เป็นเส้นทางหลักของการทำงาน

2. สี่สถานะ ไม่ใช่สอง

enum RealtimeStatus { disconnected, connecting, connected, reconnecting }
สถานะหมายถึงผู้ใช้ต้องทำอะไร
connectingกำลังต่อครั้งแรกรอ
connectedต่ออยู่ไม่ต้องทำอะไร
reconnectingหลุดแล้ว แต่ไลบรารีกำลังพยายามต่อใหม่รอ — จะกลับมาเอง
disconnectedเลิกพยายามแล้วต้องทำอะไรสักอย่าง

สองสถานะล่างต่างกันมากในสายตาผู้ใช้ แต่มักถูกยุบเป็น “ออฟไลน์” เหมือนกัน ผลคือผู้ใช้เห็นข้อความ “เชื่อมต่อไม่ได้” แล้วไปรีสตาร์ทแอป ทั้งที่อีกสามวินาทีมันจะกลับมาเอง

3. withAutomaticReconnect ทำอะไรให้ และไม่ทำอะไร

HubConnectionBuilder().withUrl(hubUrl).withAutomaticReconnect().build();
ทำให้ไม่ทำให้
ลองต่อใหม่ตามช่วงเวลาที่ถอยห่างขึ้นเอาเหตุการณ์ที่พลาดไประหว่างนั้นกลับมา
ประกาศสถานะ reconnecting และ connectedเข้ากลุ่มเดิมให้อัตโนมัติ
ยอมแพ้หลังลองครบตามที่ตั้งไว้บอกผู้ใช้ว่าเกิดอะไรขึ้น

ค่าปริยายคือลองที่ 0, 2, 10 และ 30 วินาที แล้วเลิก — รวมประมาณ 42 วินาที ซึ่งสั้นเกินไปสำหรับคลังที่คนเดินเข้าห้องเย็นห้านาที

// ตั้งเองได้ — ลองต่อไปเรื่อย ๆ โดยห่างขึ้นแต่มีเพดาน (7.12)
.withAutomaticReconnect(retryDelays: [0, 2000, 5000, 10000, 30000, 30000])
มีเพดานเสมอ ไม่ปล่อยให้ห่างขึ้นเรื่อย ๆ ไม่จำกัด — เพราะผู้ใช้ที่เดินกลับเข้ามาในพื้นที่ที่มีสัญญาณ ไม่ควรต้องรอสิบนาทีกว่าจะต่อกลับ

4. ช่วงเวลาที่หายไป

นี่คือหัวใจของบทนี้

10:00:00  ต่ออยู่ ทุกอย่างปกติ
10:00:05  สายหลุด
10:00:07  เซิร์ฟเวอร์ส่ง TaskCompleted(T-1)   ← ไม่มีใครรับ หายไปเลย
10:00:12  เซิร์ฟเวอร์ส่ง TaskDispatched(T-9)  ← หายไปเลย
10:00:18  ต่อกลับได้ สถานะเป็น connected

ตอนนี้หน้าจอบอกว่า T-1 ยังค้างอยู่ และไม่รู้จัก T-9 เลย
และไม่มีอะไรบอกว่าข้อมูลผิด

SignalR ไม่เก็บข้อความไว้ให้ ไม่มีคิว ไม่มีการส่งซ้ำ (12.2 §5) ดังนั้นเมื่อต่อกลับได้ ทางเดียวที่จะกลับมาถูกคือถามใหม่ทั้งชุดผ่าน HTTP

case RealtimeStatus.connected:
  _stopPolling();
  // ต่อกลับมาได้หลังจากหลุดไป แปลว่าระหว่างนั้นมีเหตุการณ์ที่เรา
  // ไม่ได้รับ — ไม่มีทางรู้ว่ากี่อัน จึงต้องโหลดใหม่ทั้งชุด
  if (previous == RealtimeStatus.reconnecting ||
      previous == RealtimeStatus.disconnected) {
    unawaited(load());
  }

เงื่อนไข previous == ... สำคัญ — การต่อครั้งแรกก็เข้าสู่สถานะ connected เหมือนกัน แต่ start() เรียก load() อยู่แล้ว ถ้าไม่ตรวจว่าก่อนหน้านี้เป็นอะไร จะโหลดสองครั้งซ้อนทุกครั้งที่เปิดจอ

test('ต่อกลับได้หลังหลุด ต้องโหลดใหม่เพื่ออุดช่วงที่ขาดไป', () async {
  await store.start();
  final callsBefore = repository.fetchCalls;

  inner.simulateDrop();
  await Future<void>.delayed(Duration.zero);
  // ระหว่างนี้เซิร์ฟเวอร์ส่งเหตุการณ์ไปหลายอันที่เราไม่ได้รับ

  inner.simulateReconnected();
  await Future<void>.delayed(Duration.zero);

  expect(repository.fetchCalls, callsBefore + 1);
});

5. สิ่งที่โปรเจกต์จริงยังไม่ได้ทำ

เปิด signalr_service.dart ของ wmsapp แล้วมองหา onreconnected — ไม่มี

_connection = HubConnectionBuilder()
    .withUrl(hubUrl)
    .withAutomaticReconnect()
    .build();

_connection!.on('StationDispatched', _onStationDispatched);
_connection!.on('PalletArrived', _onPalletArrived);
_connection!.on('PalletReturned', _onPalletReturned);
_connection!.on('LabelingCompleted', _onLabelingCompleted);

await _connection!.start();

ผลที่ตามมาคือ หลังจากสายหลุดแล้วต่อกลับ หน้าจอจะแสดงสถานะของสถานีที่หยุดอยู่ ณ วินาทีที่หลุด และจะไม่มีวันถูกต้องอีกจนกว่าผู้ใช้จะออกจากจอแล้วเข้าใหม่ (ซึ่งเรียก _loadStationStatus() ใหม่)

โชคดีที่การแก้ไม่ยาก และไม่ต้องแตะจอเลย

// ใน SignalRService.connect() หลัง build()
_connection!.onreconnected(({connectionId}) {
  for (final cb in List.of(_reconnectedListeners)) {
    cb();
  }
});
// ในจอ — เรียกโหลดใหม่แบบเดียวกับตอนเปิดจอ
_signalR.addReconnectedListener(_loadStationStatus);
อาการนี้แยกยากมากจาก “ไม่มีอะไรเกิดขึ้นในคลังช่วงนี้” — ซึ่งเป็นเหตุผลที่มันอยู่รอดในโค้ดได้นาน และเป็นตัวอย่างชัดเจนของหลักใน 12.1 ว่าเรียลไทม์หลุดเงียบ

6. อีกอย่างที่หายไปพร้อมการเชื่อมต่อ

ถ้า hub ใช้กลุ่ม (12.2 §2) การต่อกลับจะได้ connection ใหม่ที่ไม่อยู่ในกลุ่มเดิม

_connection!.onreconnected(({connectionId}) async {
  // เข้ากลุ่มซ้ำก่อน แล้วค่อยโหลดข้อมูลใหม่
  await _connection!.invoke('JoinStation', args: [stationId]);
  _notifyReconnected();
});

ลำดับสำคัญ — ถ้าโหลดข้อมูลก่อนเข้ากลุ่ม จะมีช่องว่างเล็ก ๆ ที่เหตุการณ์ยังไม่ถึงเราอีกครั้ง

7. แสดงให้ผู้ใช้เห็น

สถานะควรแสดงไม่ควรแสดง
connectingไม่ต้องแสดงอะไรถ้าเร็วกล่องโหลดเต็มจอ
connectedจุดเขียวเล็ก ๆ หรือไม่แสดงเลยข้อความ “เชื่อมต่อสำเร็จ” ที่เด้งทุกครั้ง
reconnectingแถบบางบอกว่ากำลังเชื่อมต่อใหม่“เกิดข้อผิดพลาด” — ยังไม่ผิดอะไร
disconnectedบอกว่าข้อมูลอาจไม่ทันสมัย + ปุ่มลองใหม่บล็อกไม่ให้ทำงาน (10.12)

แถวที่สามคือจุดที่ทำผิดกันบ่อย — การขึ้นข้อความผิดพลาดตอน reconnecting ทำให้ผู้ใช้ตกใจกับสิ่งที่กำลังจะหายเองในสามวินาที และเมื่อเห็นบ่อยเข้า เขาจะเลิกสนใจข้อความนั้น รวมถึงตอนที่มันสำคัญจริง

8. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ข้อมูลค้างเก่าหลังต่อกลับไม่ได้โหลดใหม่ตอน onreconnectedโหลดใหม่ทั้งชุด
โหลดสองครั้งตอนเปิดจอไม่ได้ตรวจสถานะก่อนหน้าโหลดเฉพาะเมื่อมาจาก reconnecting
ต่อกลับแล้วเงียบไม่ได้เข้ากลุ่มซ้ำinvoke เข้ากลุ่มใน onreconnected
ผู้ใช้เห็นข้อความผิดพลาดบ่อยจนเลิกสนใจยุบ reconnecting รวมกับ disconnectedแยกสองสถานะ
เลิกพยายามต่อเร็วเกินไปใช้ค่าปริยาย 42 วินาทีตั้ง retryDelays ที่มีเพดาน
ต่อกลับแล้วได้ 401token หมดอายุระหว่างที่หลุดaccessTokenFactory (12.3)

9. แบบฝึกหัด

ผู้ใช้รายงานว่า “เอาเครื่องใส่กระเป๋าไปกินข้าว กลับมาแล้วหน้าจอค้างที่ตัวเลขเดิมทั้งบ่าย”

  1. เกิดอะไรขึ้น
  2. จะแก้อย่างไรให้ครบ
แนวคำตอบ

1. น่าจะเป็นสองอย่างซ้อนกัน

ชั้นที่ 1ระบบปฏิบัติการตัดเครือข่ายเมื่อแอปอยู่พื้นหลังนาน การเชื่อมต่อจึงหลุด
ชั้นที่ 2ลองต่อใหม่ครบ 42 วินาทีตามค่าปริยายแล้วเลิก ตอนนั้นแอปยังอยู่ในกระเป๋า
ชั้นที่ 3กลับมาเปิดหน้าจอ — ไม่มีอะไรสั่งให้ต่อใหม่ เพราะไลบรารีเลิกไปแล้ว

2. ต้องแก้ทั้งสามชั้น แก้ชั้นเดียวไม่พอ

class _ShellState extends State<Shell> with WidgetsBindingObserver {
  @override
  void didChangeAppLifecycleState(AppLifecycleState state) {
    if (state != AppLifecycleState.resumed) return;

    // กลับมาแล้ว — ถ้าเลิกพยายามไปแล้ว ให้เริ่มใหม่
    if (client.currentStatus == RealtimeStatus.disconnected) {
      unawaited(client.acquire());
    }
    // และโหลดข้อมูลใหม่เสมอ ไม่ว่าเรียลไทม์จะเป็นอย่างไร
    unawaited(store.load());
  }
}
  • ชั้นที่ 2 ตั้ง retryDelays ที่มีเพดานและยาวกว่าเดิม เพื่อให้กรณีพักสั้น ๆ กลับมาเองได้
  • ชั้นที่ 3 ต่อใหม่ตอนกลับมาที่หน้าจอ ซึ่งเป็นหลักเดียวกับ 10.8 — ตรวจตอนที่คำตอบทั้งใหม่และกำลังจะถูกมอง
  • ทางสำรอง ถามซ้ำระหว่างที่ต่อไม่ได้ (12.11) เพื่อให้ข้อมูลไม่ค้างแม้เรียลไทม์จะยังกลับมาไม่ได้

ข้อคิดที่ใหญ่กว่าโค้ด: ผู้ใช้บอกว่า “ค้างทั้งบ่าย” ซึ่งแปลว่าเขาไม่รู้ว่าข้อมูลไม่ทันสมัย และนั่นแย่กว่าตัวข้อมูลที่ผิดเสียอีก — ต่อให้แก้ไม่ทันในรุ่นนี้ อย่างน้อยต้องมีป้ายบอกว่า “ข้อมูลล่าสุดเมื่อ 12:30” เพื่อให้เขาตัดสินใจเองได้

10. สรุป

  • การหลุดเป็นเส้นทางหลัก ไม่ใช่กรณีพิเศษ
  • reconnecting ต่างจาก disconnected ในสายตาผู้ใช้
  • withAutomaticReconnect ต่อกลับให้ แต่ไม่เอาเหตุการณ์ที่หายไปกลับมา
  • ต่อกลับแล้วต้องโหลดใหม่ทั้งชุด และเข้ากลุ่มซ้ำถ้ามีกลุ่ม
  • ตรวจสถานะก่อนหน้า ไม่งั้นจะโหลดซ้ำตอนเปิดจอ
  • ต่อใหม่ตอนแอปกลับมาจากพื้นหลัง
