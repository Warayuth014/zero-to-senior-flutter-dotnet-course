1. แพ็กเกจที่ใช้

dependencies:
  signalr_netcore: ^1.3.7
ชื่อ signalr_netcore ไม่ใช่ของทีม Microsoft โดยตรง แต่เป็นแพ็กเกจของชุมชนที่ทำตามโปรโตคอลของ SignalR — ต่างจากฝั่ง JavaScript ที่ Microsoft ดูแลเอง จึงควรตรึงรุ่นไว้และทดสอบก่อนอัปเกรด

2. สร้างการเชื่อมต่อ

final serverUrl = await ApiService.resolveServerUrl();
final hubUrl = '$serverUrl/hubs/putaway';

_connection = HubConnectionBuilder()
    .withUrl(hubUrl)
    .withAutomaticReconnect()
    .build();

_connection!.on('StationDispatched', _onStationDispatched);
_connection!.on('PalletArrived', _onPalletArrived);

await _connection!.start();
บรรทัดทำอะไร
withUrl(hubUrl)บอกว่า hub อยู่ที่ไหน
withAutomaticReconnect()ให้ไลบรารีพยายามต่อใหม่เองเมื่อสายหลุด (12.4)
build()สร้างวัตถุ แต่ยังไม่ต่อ
.on(...)ลงทะเบียนฟัง — ต้องทำก่อน start()
start()ต่อจริง

ลำดับสำคัญ — ถ้าเรียก start() ก่อนแล้วค่อย .on(...) เหตุการณ์ที่มาถึงในช่วงระหว่างนั้นจะหายไปเงียบ ๆ ซึ่งเป็นช่วงที่อาจมีเหตุการณ์อยู่จริงเพราะเซิร์ฟเวอร์ส่งทันทีที่มีคนต่อเข้ามา

โค้ดจริงของ wmsapp เรียงถูกต้อง และ lab ใช้หลักเดียวกัน

// ฟังก่อนต่อ ไม่ใช่ต่อก่อนฟัง — เหตุการณ์ที่มาถึงระหว่างที่ยังไม่ได้
// ลงทะเบียนฟัง จะหายไปเงียบ ๆ (12.7)
_eventSub = client.events.listen(_onEvent);
_statusSub = client.status.listen(_onStatus);

await client.acquire();

3. ส่ง token ไปด้วยเมื่อ hub ต้องการ

_connection = HubConnectionBuilder()
    .withUrl(
      hubUrl,
      options: HttpConnectionOptions(
        // เป็นฟังก์ชัน ไม่ใช่ค่า เพราะถูกเรียกใหม่ทุกครั้งที่ต่อ —
        // token ที่หมดอายุระหว่างที่หลุดอยู่ จะได้ตัวใหม่ตอนต่อกลับ
        accessTokenFactory: () async => sessionStore.token ?? '',
      ),
    )
    .withAutomaticReconnect()
    .build();

คอมเมนต์บรรทัดนั้นคือเหตุผลที่ API ออกแบบเป็นฟังก์ชัน — ถ้าส่งเป็นค่าคงที่ การต่อกลับหลังจากหลุดไปยี่สิบนาทีจะใช้ token ที่หมดอายุแล้ว แล้วได้ 401 โดยไม่มีทางแก้นอกจากปิดแอป

ต่อกับสิ่งที่ Part 8 วางไว้ — SessionStore เป็นเจ้าของ token ที่เดียว และทุกคนที่ต้องใช้ก็อ่านจากที่นั่นตอนที่จะใช้ ไม่ใช่คัดลอกเก็บไว้

4. ห่อไว้ไม่ให้รั่วออกไป

ปัญหาของการเรียก HubConnection ตรง ๆ จากในจอ

ปัญหาผล
จอต้องรู้จักชนิดของแพ็กเกจภายนอกเปลี่ยนแพ็กเกจแล้วต้องแก้ทุกจอ
เทสต์ต้องสร้าง HubConnection จริงทดสอบการหลุดแทบเป็นไปไม่ได้
ข้อมูลที่ได้เป็น List<Object?>ทุกจอต้องแปลงเอง และแปลงไม่เหมือนกัน

lab จึงประกาศ interface ที่พูดภาษาของแอป ไม่ใช่ภาษาของแพ็กเกจ

abstract interface class RealtimeClient {
  Stream<HubEvent> get events;
  Stream<RealtimeStatus> get status;
  RealtimeStatus get currentStatus;

  Future<void> connect();
  Future<void> disconnect();
}

ตัวจริงห่อ HubConnection ไว้ข้างใน ส่วนตัวปลอมในเทสต์ไม่มีอะไรเกี่ยวกับเครือข่ายเลย

/// ตัวจริงจะห่อ `HubConnection` ของ `signalr_netcore` แล้วแปลง callback
/// ของมันมาเป็น stream สองเส้นนี้ ส่วนที่เหลือของแอปไม่ต้องรู้ว่าข้างใน
/// เป็นอะไร — สลับไปใช้ WebSocket ธรรมดาหรือ MQTT ก็แก้แค่ไฟล์เดียว
class InMemoryRealtimeClient implements RealtimeClient { ... }

5. จาก callback เป็น stream

SignalR ให้ callback มา ส่วนแอปเราอยากได้ stream แปลงกันตรงนี้

class SignalRRealtimeClient implements RealtimeClient {
  final _events = StreamController<HubEvent>.broadcast();
  final _status = StreamController<RealtimeStatus>.broadcast();
  HubConnection? _connection;

  @override
  Stream<HubEvent> get events => _events.stream;

  @override
  Future<void> connect() async {
    _connection = HubConnectionBuilder().withUrl(url)
        .withAutomaticReconnect().build();

    // แปลงทุกชื่อเหตุการณ์เป็น stream เส้นเดียว
    for (final name in PutawayEvents.all) {
      _connection!.on(name, (args) => _events.add(decodeHubEvent(name, args)));
    }

    await _connection!.start();
  }
}
ได้อะไรจากการเป็น streamรายละเอียด
ยกเลิกการฟังได้ด้วยตัวเดียวsubscription.cancel() แทนการจำว่าต้อง remove callback ตัวไหนบ้าง (12.8)
กรองและแปลงได้events.where((e) => e is TaskCompleted)
เข้ากับ StreamBuilderต่อเข้า UI ได้ตรง ๆ ถ้าต้องการ
.broadcast() จำเป็นเพราะมีผู้ฟังหลายคน — stream ธรรมดาใน Dart อนุญาตให้ฟังได้คนเดียว และจะโยน Bad state: Stream has already been listened to ทันทีที่จอที่สองมาฟัง

6. singleton หรือรับเข้ามา

// wmsapp — singleton
class SignalRService {
  static final SignalRService _instance = SignalRService._internal();
  factory SignalRService() => _instance;
  SignalRService._internal();
}

// ในจอ
final _signalR = SignalRService();

เหตุผลที่ใช้ singleton ตรงนี้แข็งแรงกว่ากรณี ApiService ใน 9.3 เพราะการเชื่อมต่อต้องมีตัวเดียวจริง ๆ ไม่ใช่แค่ควรมี — สามจอที่เปิดสามการเชื่อมต่อคือการสิ้นเปลืองที่เห็นได้ชัดทั้งฝั่งเครื่องและฝั่งเซิร์ฟเวอร์

แต่ปัญหาเดิมยังอยู่ คือจอหยิบเอง จึงใส่ตัวปลอมในเทสต์ไม่ได้ lab แยกสองเรื่องนี้ออกจากกัน

// มีตัวเดียว — จัดการด้วยตัวนับ
class SharedRealtimeClient {
  SharedRealtimeClient(this._inner);
  final RealtimeClient _inner;
  int _users = 0;
}

// แต่ยังส่งเข้ามาได้
LiveTaskStore({required this.repository, required this.client});

ผลคือได้ทั้งการเชื่อมต่อเดียวและความสามารถในการทดสอบ ซึ่งเป็นข้อสรุปเดียวกับ 9.8 — ปัญหาไม่ได้อยู่ที่มีตัวเดียว แต่อยู่ที่หยิบเอง

7. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
เหตุการณ์แรก ๆ หายไปลงทะเบียนฟังหลัง start()ฟังก่อนต่อเสมอ
Stream has already been listened toใช้ stream ธรรมดา.broadcast()
ต่อกลับแล้วได้ 401ส่ง token เป็นค่าคงที่ใช้ accessTokenFactory
เทสต์ต้องมีเซิร์ฟเวอร์จริงจอเรียก HubConnection ตรง ๆห่อด้วย interface
เปลี่ยนเซิร์ฟเวอร์แล้วยังต่อที่เก่าURL ถูกอ่านครั้งเดียวตอนสร้างสร้างการเชื่อมต่อใหม่เมื่อสลับ profile (10.6)

แถวสุดท้ายเป็นจุดที่ Part 10 กับ Part 12 ชนกัน — ApiClient อ่าน base URL ใหม่ทุกคำขอจึงสลับได้ทันที แต่การเชื่อมต่อค้างไม่ได้ทำงานแบบนั้น มันจำ URL ตั้งแต่ตอนต่อ ดังนั้นสลับ profile แล้วต้องตัดแล้วต่อใหม่

8. แบบฝึกหัด

เขียน SignalRRealtimeClient ที่ implement RealtimeClient ให้สมบูรณ์ โดยห่อ HubConnection

  1. สถานะสี่ตัวใน RealtimeStatus จะมาจากไหน
  2. dispose ต้องทำอะไรบ้าง
แนวคำตอบ

1. signalr_netcore มี callback ให้ผูก

_connection!.onclose(({error}) => _emit(RealtimeStatus.disconnected));
_connection!.onreconnecting(({error}) => _emit(RealtimeStatus.reconnecting));
_connection!.onreconnected(({connectionId}) {
  _emit(RealtimeStatus.connected);
  // ต้องเข้ากลุ่มซ้ำที่นี่ — connection ใหม่ไม่ได้อยู่ในกลุ่มเดิม (12.2)
  unawaited(_connection!.invoke('JoinStation', args: [stationId]));
});

ส่วน connecting กับ connected ประกาศเองรอบ ๆ start()

2. สี่อย่าง และลำดับมีความหมาย

Future<void> dispose() async {
  await _connection?.stop();      // 1. หยุดก่อน ไม่ให้มีเหตุการณ์เข้ามาอีก
  _connection = null;             // 2. ปล่อยอ้างอิง
  await _events.close();          // 3. ปิด controller
  await _status.close();
}

ถ้าปิด controller ก่อนหยุดการเชื่อมต่อ เหตุการณ์ที่มาถึงระหว่างนั้นจะทำให้เกิด Bad state: Cannot add event after closing ซึ่งเป็นข้อผิดพลาดที่โผล่แบบสุ่มและหาสาเหตุยากมาก

สิ่งที่ต้องคิดต่อคือ ใครเรียก dispose — ถ้าเป็น singleton ที่มีอายุเท่ากับแอป คำตอบอาจเป็น “ไม่มีใคร” ซึ่งยอมรับได้ แต่ต้องรู้ตัว และถ้าวันหนึ่งต้องสลับเซิร์ฟเวอร์ (§7 แถวสุดท้าย) จะต้องมีคนเรียกแล้ว

9. สรุป

  • ลงทะเบียนฟังก่อน start() เสมอ
  • ส่ง token เป็นฟังก์ชัน ไม่ใช่ค่า เพราะถูกเรียกใหม่ทุกครั้งที่ต่อ
  • ห่อแพ็กเกจไว้หลัง interface ที่พูดภาษาของแอป
  • แปลง callback เป็น stream แล้วยกเลิกการฟังได้ด้วยตัวเดียว
  • ใช้ .broadcast() เมื่อมีผู้ฟังหลายคน
  • มีการเชื่อมต่อตัวเดียวได้ โดยไม่ต้องให้จอหยิบเอง
