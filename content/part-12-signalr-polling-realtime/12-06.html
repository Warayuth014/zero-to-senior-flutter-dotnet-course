จบบทนี้แล้วจะทำได้: ส่งเหตุการณ์ถึงคนที่สนใจโดยไม่ต้องเขียนโค้ดซ้ำต่อเหตุการณ์หนึ่งตัว

1. แบบที่หนึ่ง: รายชื่อ callback ต่อเหตุการณ์

class SignalRService {
  final List<SignalRCallback> _stationDispatchedListeners = [];
  final List<SignalRCallback> _palletArrivedListeners = [];
  final List<SignalRCallback> _palletReturnedListeners = [];
  final List<SignalRCallback> _labelingCompletedListeners = [];

  void addStationDispatchedListener(SignalRCallback cb) =>
      _stationDispatchedListeners.add(cb);
  void removeStationDispatchedListener(SignalRCallback cb) =>
      _stationDispatchedListeners.remove(cb);
  // ...คูณสี่...

  void _onStationDispatched(List<Object?>? args) {
    final data = _parseArgs(args);
    for (final cb in List.of(_stationDispatchedListeners)) {
      cb(data);
    }
  }
  // ...คูณสี่...
}

อ่านง่ายและตรงไปตรงมา แต่มีสองอย่างที่โตตามจำนวนเหตุการณ์

ต่อเหตุการณ์หนึ่งตัว ต้องเขียนบรรทัด
ฟิลด์รายชื่อ1
เมธอด add / remove4
ตัวจัดการที่กระจายไปทุกคน6
บรรทัด .on(...)1

สิบสองบรรทัดต่อเหตุการณ์หนึ่งตัว และไฟล์จริงสองไฟล์รวมกัน 235 บรรทัดสำหรับ 10 เหตุการณ์ — โดยที่โครงของทั้งสองไฟล์เหมือนกันทุกประการ ต่างแค่ชื่อ

สังเกตว่า SortingSignalRService เห็นปัญหานี้แล้วและแก้ไปครึ่งทาง

// helper: build handler ที่ broadcast ไปทุก listener ใน list
void Function(List<Object?>?) _on(List<SortingCallback> listeners) => (args) {
  final data = _parseArgs(args);
  for (final cb in List.of(listeners)) {
    cb(data);
  }
};

ตัดส่วนที่ซ้ำที่สุดออกไปได้ เหลือ _connection!.on('StationFull', _on(_stationFull)); บรรทัดเดียวต่อเหตุการณ์ — เป็นการปรับปรุงที่ดีและเห็นได้ว่าคนเขียนรู้ตัว

2. รายละเอียดเล็ก ๆ ที่ถูกต้องมาก

for (final cb in List.of(_stationDispatchedListeners)) {
  cb(data);
}

List.of(...) คัดลอกรายชื่อก่อนวน แทนที่จะวนบนตัวจริง คำถามคือทำไม

เพราะ callback ที่ถูกเรียกอาจยกเลิกการฟังของตัวเองระหว่างนั้น

void _handleStationDispatched(Map<String, dynamic> data) {
  if (data['stationId'] == _myStation) {
    Navigator.pop(context);              // ปิดจอ
    // → dispose() ทำงาน → removeStationDispatchedListener(...)
    // → รายชื่อถูกแก้ระหว่างที่กำลังวนอยู่
  }
}

ถ้าวนบนตัวจริงจะได้ ConcurrentModificationError ซึ่งเป็นข้อผิดพลาดที่เกิดเฉพาะบางจังหวะ ทำซ้ำยาก และหาสาเหตุยากมาก

รายละเอียดแบบนี้คือสิ่งที่แยกโค้ดที่ผ่านการใช้งานจริง ออกจากโค้ดที่เพิ่งเขียนเสร็จ — มันไม่มีทางคิดออกตอนเขียนครั้งแรก แต่มาจากการที่เคยเจอมาแล้ว

3. แบบที่สอง: stream เส้นเดียวที่มีชนิด

abstract interface class RealtimeClient {
  Stream<HubEvent> get events;
  Stream<RealtimeStatus> get status;
  ...
}

เหตุการณ์ทุกชนิดไหลผ่านเส้นเดียว แล้วคนฟังแยกเองด้วย switch

_eventSub = client.events.listen(_onEvent);

void _onEvent(HubEvent event) {
  switch (event) {
    case TaskDispatched(:final taskId, :final assignee, :final version): ...
    case TaskCompleted(:final taskId, :final version): ...
    case StationCounterUpdated(): break;    // จอนี้ไม่สนใจ
    case UnknownHubEvent(): break;
    case MalformedHubEvent(): malformedEventCount++;
  }
}
ต่อเหตุการณ์หนึ่งตัว ต้องเขียนบรรทัด

สาขาใน decodeHubEventราว 4
โค้ดในตัวรับ0 — ไม่ต้องเขียนอะไรเพิ่มถ้าไม่สนใจ

4. เทียบสองแบบ

รายชื่อ callbackstream ที่มีชนิด
ข้อมูลที่คนฟังได้รับMap<String, dynamic>ชนิดที่คอมไพเลอร์ตรวจได้
ยกเลิกการฟังต้องจำว่า remove ตัวไหนบ้างsubscription.cancel() ตัวเดียว
ลืมยกเลิกแล้วเกิดอะไรรั่วเงียบ (12.8)รั่วเงียบเหมือนกัน แต่มีที่เดียวให้ลืม
เพิ่มเหตุการณ์ใหม่เขียน 12 บรรทัด ทุกคนที่สนใจต้องเพิ่ม listenerเพิ่มคลาส คอมไพเลอร์ชี้ให้ว่าต้องแก้ตรงไหน
คนฟังที่ไม่สนใจเหตุการณ์นั้นไม่ถูกเรียกเลยถูกเรียกแต่ตกในสาขาที่ไม่ทำอะไร
กรองก่อนถึงคนฟังทำไม่ได้events.where(...)

แถวรองสุดท้ายเป็นข้อเสียจริงของ stream — จอที่สนใจเหตุการณ์เดียวจะถูกปลุกทุกครั้งที่มีเหตุการณ์ใดก็ตาม ในระบบที่เหตุการณ์เยอะมากอาจมีผล แต่ในคลังที่มีเหตุการณ์ไม่กี่ร้อยครั้งต่อวัน ไม่มีผลที่วัดได้

ถ้ามีผลจริง กรองที่ต้นทางได้

_sub = client.events
    .where((event) => event is StationCounterUpdated)
    .cast<StationCounterUpdated>()
    .listen(_onCounter);

5. เมื่อไหร่ควรใช้แบบไหน

ใช้รายชื่อ callbackใช้ stream
มีเหตุการณ์ไม่กี่ตัวและจะไม่เพิ่มเหตุการณ์เพิ่มเรื่อย ๆ
ทีมยังไม่คุ้นกับ streamอยากได้ชนิดที่คอมไพเลอร์ตรวจ
ต้องการให้คนฟังถูกเรียกเฉพาะที่สนใจอยากกรองและแปลงก่อนถึงคนฟัง

ในทางปฏิบัติ ปัจจัยที่มีน้ำหนักที่สุดคือชนิดของข้อมูลที่คนฟังได้รับ — Map<String, dynamic> แปลว่าทุกจอที่ฟังต้องเขียนโค้ดอ่านฟิลด์เอง และเขียนไม่เหมือนกัน

// ทุกจอต้องเขียนแบบนี้เอง และแต่ละจอเขียนไม่เหมือนกัน
void _onCounterUpdated(Map<String, dynamic> data) {
  final stationId = data['stationId'] as int?;   // จอนี้คิดว่าเป็น int
  if (stationId == null || !mounted) return;
  ...
}

สังเกต as int? — ถ้าเซิร์ฟเวอร์ส่ง "01" มาเป็นสตริง บรรทัดนี้จะได้ null เงียบ ๆ แล้ว return ทันที ผลคือเหตุการณ์ถูกทิ้งโดยไม่มีร่องรอย ซึ่งเป็นปัญหาเดียวกับที่ 12.5 แก้ด้วย MalformedHubEvent

6. รูปแบบที่สาม: หนึ่ง stream ต่อหนึ่งเหตุการณ์

class RealtimeClient {
  Stream<TaskDispatched> get taskDispatched => ...;
  Stream<TaskCompleted> get taskCompleted => ...;
}

ได้ชนิดที่ตรงตัวและคนฟังถูกเรียกเฉพาะที่สนใจ แต่กลับไปมีปัญหาเดิม คือต้องเขียนเพิ่มต่อเหตุการณ์หนึ่งตัว และคนฟังหลายเหตุการณ์ต้องเก็บ subscription หลายตัว

ทางสายกลางที่ใช้ได้ดีคือมี stream เดียว แล้วให้ตัวช่วยกรอง

extension HubEventStream on Stream<HubEvent> {
  Stream<T> only<T extends HubEvent>() => where((e) => e is T).cast<T>();
}

// ในจอ
_sub = client.events.only<StationCounterUpdated>().listen(_onCounter);

ได้ทั้งชนิดที่ตรง การกรอง และ subscription ตัวเดียวต่อสิ่งที่สนใจ โดยที่ตัวไคลเอนต์ไม่ต้องเขียนเพิ่มเลยเมื่อมีเหตุการณ์ใหม่

7. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ConcurrentModificationError แบบสุ่มวนบนรายชื่อจริงขณะที่มีคน removeList.of(...) ก่อนวน
เหตุการณ์ถูกทิ้งเงียบ ๆas int? ที่ได้ null แล้ว returnแปลงที่ขอบให้มีชนิด (12.5)
เพิ่มเหตุการณ์แล้วบางจอไม่รู้เรื่องต้องไปเพิ่ม listener ทีละจอใช้ sealed + switch
คนฟังตัวเดิมถูกเรียกสองครั้งเพิ่ม listener ซ้ำโดยไม่ได้ลบตัวเก่าตรวจใน initState ให้เพิ่มครั้งเดียว (12.7)
เขียนโค้ดซ้ำสองไฟล์ที่เหมือนกันเป๊ะhub ที่สองคัดลอกจากตัวแรกทำตัวห่อที่รับรายชื่อเหตุการณ์เข้ามา

8. แบบฝึกหัด

โปรเจกต์มี SignalRService กับ SortingSignalRService ที่โครงเหมือนกันทุกประการ ต่างแค่ชื่อ hub และรายชื่อเหตุการณ์ ตอนนี้จะเพิ่ม hub ที่สามสำหรับงานหยิบของ

  1. ควรคัดลอกไฟล์ที่สามไหม
  2. ถ้าไม่ จะรวมอย่างไร
แนวคำตอบ

1. ไม่ควร — สามครั้งคือจุดที่รูปแบบชัดแล้ว และการคัดลอกครั้งที่สามแปลว่าบั๊กที่พบในไฟล์หนึ่งต้องไปแก้อีกสองที่ ซึ่งจะลืม

แต่ต้องบอกตรง ๆ ว่าการคัดลอกครั้งที่สอง (ตอนที่ทำ SortingSignalRService) เป็นการตัดสินใจที่สมเหตุสมผล — ตอนนั้นยังไม่รู้ว่ารูปแบบจะซ้ำอีก และการรวมของสองชิ้นบ่อยครั้งได้ของกลางที่ผิดรูป

2. ทำตัวห่อที่รับสิ่งที่ต่างเข้ามา

class HubClient implements RealtimeClient {

สามสิ่งที่ต่างกันระหว่าง hub กลายเป็นพารามิเตอร์ ส่วนสิ่งที่เหมือนกัน — การต่อ การต่อใหม่ การนับผู้ใช้ การแปลง callback เป็น stream — เขียนครั้งเดียว

วิธีย้ายที่ปลอดภัย ตามหลักใน 9.15

  1. เขียน HubClient ใหม่ แล้วใช้กับ hub ที่สามก่อน — ยังไม่แตะสองตัวเดิม
  2. ใช้จริงสักสองสัปดาห์ ดูว่าโครงรับมือกับของจริงได้ไหม
  3. ย้าย SortingSignalRService มาใช้ พร้อมเทสต์ที่คุมพฤติกรรมเดิม
  4. ย้าย SignalRService เป็นตัวสุดท้าย เพราะมีจอที่ใช้มากที่สุด

ข้อ 2 สำคัญ — ของกลางที่ออกแบบจากสองตัวอย่างมักจะพอดีกับสองตัวนั้นเกินไป การเอาไปใช้กับตัวที่สามก่อนจะเผยว่าโครงยืดหยุ่นพอจริงไหม

9. สรุป

  • รายชื่อ callback ต่อเหตุการณ์ ต้องเขียนราว 12 บรรทัดต่อเหตุการณ์หนึ่งตัว
  • List.of(...) ก่อนวน กัน ConcurrentModificationError เมื่อคนฟังยกเลิกตัวเอง
  • stream ที่มีชนิดให้คอมไพเลอร์ช่วยตอนเพิ่มเหตุการณ์ใหม่
  • ปัจจัยที่มีน้ำหนักที่สุดคือชนิดของข้อมูลที่คนฟังได้รับ
  • hub ที่สามคือจุดที่ควรรวมเป็นตัวห่อตัวเดียว
