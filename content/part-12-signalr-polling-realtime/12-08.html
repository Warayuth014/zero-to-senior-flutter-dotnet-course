ต้องผ่านอะไรมาก่อน: 12.7, 4.11 (ยกเลิกตัวจับเวลา), 8.8 (ล้างสถานะ)
จบบทนี้แล้วจะทำได้: เก็บของให้ครบตอนปิดจอ และเขียนเทสต์ที่จับการรั่วซึ่งมองไม่เห็นด้วยตา

1. การรั่วหน้าตาเป็นอย่างไร

ต่างจากบั๊กทั่วไปที่พังทันที การรั่วสะสมเงียบ ๆ แล้วโผล่ในรูปแบบที่โยงกลับหาสาเหตุยาก

สิ่งที่ลืมเก็บอาการที่ผู้ใช้เจอ
ไม่ยกเลิกการฟังเหตุการณ์เปิดปิดจอสิบครั้ง แล้วเหตุการณ์เดียวถูกจัดการสิบครั้ง
ไม่ยกเลิกตัวจับเวลายิงคำขอเพิ่มขึ้นเรื่อย ๆ จนแอปช้าและแบตหมดเร็ว
ไม่ปล่อยการเชื่อมต่อเซิร์ฟเวอร์ถือ connection ที่ไม่มีใครใช้
ไม่ปิด StreamControllerหน่วยความจำโตขึ้นเรื่อย ๆ
แถวแรกเป็นอาการที่หลอกที่สุด เพราะแอปยังทำงานถูกต้อง แค่ทำงานซ้ำ — ตัวเลขบนหน้าจอยังถูก และไม่มีข้อผิดพลาดใด ๆ จนกว่าจะมีการกระทำที่ทำซ้ำแล้วเห็น เช่นเสียงเตือนที่ดังสิบครั้ง

2. รายการที่ต้องเก็บ

/// หยุดทุกอย่างที่ยังทำงานอยู่
///
/// สามอย่างที่ต้องเก็บให้ครบ — สองการฟัง หนึ่งตัวจับเวลา และการปล่อย
/// การเชื่อมต่อที่ใช้ร่วมกัน ลืมอย่างใดอย่างหนึ่งแปลว่ารั่ว
@override
void dispose() {
  _eventSub?.cancel();
  _statusSub?.cancel();
  _stopPolling();
  unawaited(client.release());
  super.dispose();
}

เทียบกับฝั่งที่ใช้รายชื่อ callback

@override
void dispose() {
  _signalR.removeStationDispatchedListener(_handleStationDispatched);
  _signalR.removePalletArrivedListener(_handlePalletArrived);
  _signalR.removePalletReturnedListener(_handlePalletReturned);
  _signalR.removeLabelingCompletedListener(_handleLabelingCompleted);
  _signalR.disconnect();
  _stationController.dispose();
  super.dispose();
}

โค้ดนี้ถูกต้องครบทุกบรรทัด แต่สังเกตว่ามันเป็นรายการที่ต้องจำ — เพิ่ม listener ตัวที่ห้าใน initState แล้วลืมเพิ่มบรรทัดที่ห้าใน dispose ไม่มีอะไรฟ้อง

วิธีลดโอกาสลืมรายละเอียด
ใช้ stream ตัวเดียวsubscription ตัวเดียวต่อสิ่งที่ฟัง ไม่ว่าจะมีเหตุการณ์กี่ชนิด
เก็บ subscription ไว้ในลิสต์for (final s in _subs) s.cancel();
ให้ store เป็นคนถือ ไม่ใช่จอจอเรียก store.dispose() ที่เดียว

3. dispose ที่ต้องรอผล

unawaited(client.release());

dispose() ของ ChangeNotifier เป็น void ไม่ใช่ Future จึงรอ await ไม่ได้ ทางเลือกมีสองทาง

ทางผล
unawaited(...)สั่งแล้วปล่อย — ถูกสำหรับการปล่อยการเชื่อมต่อ เพราะไม่มีใครรอผล
เมธอด Future<void> close() แยกคนเรียกรอได้ แต่ต้องจำว่าต้องเรียกทั้ง close และ dispose

unawaited ไม่ได้แปลว่า “ไม่สนใจ” แต่แปลว่า “ตั้งใจไม่รอ” — ถ้าไม่เขียนคำนี้ ตัววิเคราะห์จะเตือนว่าลืม await ซึ่งเป็นการเตือนที่มีประโยชน์ในที่อื่น

4. เทสต์ที่จับการรั่ว

การรั่วมองไม่เห็นด้วยตา แต่เทสต์จับได้ — และนี่คือเหตุผลหลักที่ lab ออกแบบให้ทุกอย่างรับเข้ามาทาง constructor

test('ทิ้ง store แล้วตัวจับเวลาต้องหยุด', () {
  fakeAsync((async) {
    unawaited(store.start());
    async.flushMicrotasks();
    inner.simulateDrop();
    async.flushMicrotasks();
    expect(store.isPolling, isTrue);

    store.dispose();
    final callsAtDispose = repository.fetchCalls;

    async.elapse(const Duration(minutes: 10));
    expect(repository.fetchCalls, callsAtDispose);
  });
});

fakeAsync ทำให้เวลาสิบนาทีผ่านไปในเสี้ยววินาที และถ้าตัวจับเวลายังเดินอยู่ จำนวนคำขอจะเพิ่มขึ้น 40 ครั้ง ซึ่งเทสต์จับได้ทันที

test('ทิ้ง store แล้วเหตุการณ์ที่มาทีหลังต้องไม่ถูกจัดการ', () async {
  await store.start();
  store.dispose();

  // ถ้าไม่ยกเลิกการฟัง บรรทัดนี้จะทำให้ ChangeNotifier ที่ถูกทิ้งแล้ว
  // ถูกเรียก notifyListeners ซึ่งโยนข้อผิดพลาด
  inner.emit(const TaskCompleted(taskId: 'T-1', version: 9));
  await Future<void>.delayed(Duration.zero);

  expect(store.malformedEventCount, 0);
});
test('ทิ้ง store แล้วปล่อยการเชื่อมต่อที่ใช้ร่วมกัน', () async {
  await store.start();
  expect(shared.userCount, 1);

  store.dispose();
  await Future<void>.delayed(Duration.zero);

  expect(shared.userCount, 0);
  expect(inner.disconnectCalls, 1);
});
สามเทสต์นี้ตรวจสามอย่างที่ต่างกัน และเขียนแยกกันโดยตั้งใจ — ถ้ารวมเป็นเทสต์เดียว เมื่อมันล้มจะไม่รู้ว่าลืมเก็บอะไร

5. ChangeNotifier ที่ถูกทิ้งแล้วจะโยน

Flutter รุ่นใหม่ทำให้การเรียก notifyListeners() หลัง dispose() โยนข้อผิดพลาด

A ChangeNotifier was used after being disposed.
Once you have called dispose() on a ChangeNotifier, it can no longer be used.

ฟังดูน่ารำคาญ แต่เป็นของขวัญ — มันเปลี่ยนการรั่วที่เงียบให้กลายเป็นข้อผิดพลาดที่เห็นได้ระหว่างพัฒนา

อย่าแก้ด้วยให้แก้ด้วย
ใส่ธง _disposed แล้วเช็คก่อน notifyยกเลิกการฟังให้ครบ
ครอบด้วย try/catchหาว่าใครยังเรียกอยู่

ธง _disposed ทำให้ข้อผิดพลาดหายไปแต่การรั่วยังอยู่ — subscription ยังทำงาน ตัวจับเวลายังเดิน และการเชื่อมต่อยังถูกถือไว้ เปลี่ยนจากบั๊กที่เห็นได้เป็นบั๊กที่มองไม่เห็น ซึ่งแย่กว่าเดิม

7. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
เหตุการณ์ถูกจัดการหลายครั้งลืม remove listenerใช้ stream ตัวเดียว หรือเก็บใน _subs

8. แบบฝึกหัด

ทีมสงสัยว่าแอปรั่ว เพราะผู้ใช้บอกว่า “ใช้ไปสักพักแล้วช้า ต้องปิดเปิดใหม่” แต่ไม่มีใครทำซ้ำได้

  1. จะยืนยันว่ารั่วจริงอย่างไร
  2. จะหาว่ารั่วตรงไหน
แนวคำตอบ

1. เขียนเทสต์ที่จำลองการใช้งานยาว ๆ แทนที่จะรอผู้ใช้รายงาน

testWidgets('เปิดปิดจอ 20 ครั้ง แล้วต้องไม่มีผู้ฟังค้าง', (tester) async {
  final client = InMemoryRealtimeClient();
  final shared = SharedRealtimeClient(client);

  for (var i = 0; i < 20; i++) {
    await tester.pumpWidget(TaskScreen(client: shared, ...));
    await tester.pumpAndSettle();
    await tester.pumpWidget(const SizedBox());   // ทิ้งจอ
    await tester.pumpAndSettle();
  }

  expect(shared.userCount, 0);
  expect(client.disconnectCalls, 20);
});

ตัวนับที่เปิดไว้ใน SharedRealtimeClient มีไว้เพื่อสิ่งนี้โดยเฉพาะ

/// จำนวนผู้ใช้งานตอนนี้ เปิดให้เทสต์ตรวจได้ว่าปล่อยครบไหม
int get userCount => _users;

2. ไล่จากที่ตรวจง่ายไปยาก

วิธีจับอะไรได้
นับ listener ตอนรัน แล้ว log ทุกครั้งที่เพิ่ม/ลดlistener ที่ไม่ถูกลบ — ตรงและเร็วที่สุด
DevTools แท็บ Memory ถ่าย snapshot ก่อนและหลังเปิดปิดจอ 10 ครั้งวัตถุที่ควรถูกเก็บแต่ยังอยู่
ค้นหา Timer.periodic และ .listen( ทั้งโปรเจกต์ แล้วเทียบกับ cancel()ที่ที่ลืมทั้งหมด

วิธีที่สามหยาบแต่ได้ผลดีอย่างน่าประหลาดใจ

grep -rn "\.listen(\|Timer.periodic" lib --include=*.dart | wc -l
grep -rn "\.cancel()" lib --include=*.dart | wc -l

ตัวเลขไม่ต้องเท่ากันเป๊ะ แต่ถ้าต่างกันมาก ให้ไล่ดูทีละที่

ข้อคิดที่สำคัญกว่าวิธีหา: อาการ “ใช้ไปสักพักแล้วช้า” เป็นอาการที่ผู้ใช้ทนอยู่มานานก่อนจะบอก เพราะเขาแก้เองได้ด้วยการปิดเปิดแอป — แปลว่าตอนที่ได้ยิน มันอาจมีมานานแล้ว และการเพิ่มเทสต์แบบข้อ 1 เข้าไปในทุกจอสำคัญ คุ้มกว่าการไล่หาทีละครั้ง

9. สรุป

  • การรั่วเงียบ และแอปยังทำงานถูกอยู่ แค่ทำซ้ำ
  • เก็บให้ครบ — การฟัง ตัวจับเวลา การเชื่อมต่อ และ controller
  • ยิ่งมีของให้จำมาก ยิ่งลืม ให้ store ถือแทนจอ
  • อย่าใส่ธง _disposed กลบข้อผิดพลาด มันเปลี่ยนบั๊กที่เห็นเป็นบั๊กที่มองไม่เห็น
  • เทสต์ที่เปิดปิดจอหลายรอบแล้วนับ จับการรั่วได้ก่อนถึงมือผู้ใช้
