3. แต่ก็ไม่กลืนเงียบ ๆ

ทางง่ายคือคืน null แล้วข้ามไป ซึ่งแก้ปัญหาการหลุดได้ แต่สร้างปัญหาใหม่ — ไม่มีใครรู้ว่าสัญญาผิดตั้งแต่เมื่อไหร่

lab จึงแยกสองกรณีที่หน้าตาเหมือนกันแต่ความหมายต่างกันมาก

/// เหตุการณ์ที่แอปรุ่นนี้ยังไม่รู้จัก
///
/// ไม่ใช่ความผิดพลาด — เซิร์ฟเวอร์รุ่นใหม่ส่งเหตุการณ์ที่แอปรุ่นเก่า
/// ยังไม่รองรับเป็นเรื่องปกติ และต้องไม่ทำให้อะไรพัง
class UnknownHubEvent extends HubEvent { ... }

/// เหตุการณ์ที่รู้จักชื่อ แต่ข้อมูลข้างในไม่ตรงกับที่ตกลงไว้
///
/// ต่างจาก UnknownHubEvent ตรงที่อันนี้**เป็นปัญหา**ที่ทีมต้องรู้
class MalformedHubEvent extends HubEvent {
  final String reason;   // อธิบายว่าอะไรไม่ตรง
}
กรณีความหมายควรทำ
ชื่อที่ไม่รู้จักเซิร์ฟเวอร์ใหม่กว่าแอปข้ามไปเงียบ ๆ — เป็นเรื่องปกติ
ชื่อรู้จัก แต่ข้อมูลผิดสัญญาถูกละเมิดนับไว้และส่งเข้าระบบบันทึก

ถ้าไม่แยกสองอย่างนี้ จะเจอทางเลือกที่แย่ทั้งคู่ — เตือนทุกครั้งที่เซิร์ฟเวอร์ส่งอะไรใหม่มา (รกจนไม่มีใครดู) หรือเงียบทุกกรณี (ไม่มีวันรู้ว่าสัญญาพัง)

4. โครงของตัวแปลง

HubEvent decodeHubEvent(String name, List<Object?>? args) {
  final payload = _firstMap(args);
  if (payload == null) {
    return MalformedHubEvent(name, 'ข้อมูลที่ส่งมาไม่ใช่ object');
  }

  try {
    return switch (name) {
      'TaskDispatched' => TaskDispatched(
        taskId: _requireText(payload, 'taskId'),
        assignee: _requireText(payload, 'assignee'),
        version: _requireInt(payload, 'version'),
      ),
      'TaskCompleted' => TaskCompleted(
        taskId: _requireText(payload, 'taskId'),
        version: _requireInt(payload, 'version'),
      ),
      ...
      _ => UnknownHubEvent(name),
    };
  } on _FieldError catch (error) {
    return MalformedHubEvent(name, error.message);
  }
}

รูปแบบที่ใช้คือ ข้างในโยน ข้างนอกจับ — ตัวอ่านแต่ละฟิลด์โยนได้ตามปกติ ทำให้เขียนสั้นและอ่านง่าย ส่วนขอบนอกสุดจับทั้งหมดแล้วแปลงเป็นค่าคืน

ข้อดีคือการเพิ่มเหตุการณ์ใหม่ไม่ต้องเขียนโค้ดจัดการข้อผิดพลาดซ้ำ และเป็นไปไม่ได้ที่จะลืม

class _FieldError implements Exception {
  const _FieldError(this.message);
  final String message;
}

_FieldError เป็นชนิดส่วนตัวของไฟล์นี้โดยตั้งใจ — มันไม่ควรหลุดออกไปข้างนอก เพราะข้างนอกไม่ควรต้องจับอะไรเลย

5. รายละเอียดที่ทำให้ต่างกัน

/// SignalR ส่ง argument มาเป็นลิสต์ เพราะฝั่ง .NET เรียก
/// `SendAsync("EventName", arg1, arg2)` ได้หลายตัว
///
/// สัญญาของระบบนี้คือส่ง object เดียว จึงอ่านเฉพาะตัวแรก
Map<String, dynamic>? _firstMap(List<Object?>? args) {
  if (args == null || args.isEmpty) return null;
  final first = args.first;
  if (first is Map<String, dynamic>) return first;
  // มาจาก JSON ที่ถอดรหัสแล้วอาจเป็น Map<dynamic, dynamic>
  if (first is Map) return Map<String, dynamic>.from(first);
  return null;
}

บรรทัด if (first is Map) ตัวที่สองไม่ใช่ของฟุ่มเฟือย — ขึ้นกับว่า JSON ถูกถอดรหัสมาอย่างไร ค่าที่ได้อาจเป็น Map<dynamic, dynamic> ซึ่งไม่ผ่านการตรวจ is Map<String, dynamic> โค้ดจริงของ wmsapp ก็เขียนสองชั้นแบบนี้เหมือนกัน

int _requireInt(Map<String, dynamic> json, String key) {
  final value = json[key];
  // JSON ไม่แยก int กับ double — 3 กับ 3.0 คือเลขเดียวกัน (6.7)
  if (value is int) return value;
  if (value is double && value == value.roundToDouble()) return value.toInt();
  throw _FieldError('$key ต้องเป็นจำนวนเต็ม แต่ได้ ${value.runtimeType}');
}

กฎเดียวกับ Part 6 — รับ 3.0 เพราะเป็นจำนวนเต็มเท่ากัน แต่ปฏิเสธ 3.5

7. sealed ทำให้ลืมไม่ได้

sealed class HubEvent {
  const HubEvent();
  String get name;
}

sealed แปลว่าคลาสลูกทั้งหมดต้องอยู่ในไฟล์เดียวกัน ผลคือคอมไพเลอร์รู้รายชื่อครบ และ switch ที่จัดการไม่ครบจะคอมไพล์ไม่ผ่าน

void _onEvent(HubEvent event) {
  switch (event) {
    case TaskDispatched(:final taskId, :final assignee, :final version): ...
    case TaskCompleted(:final taskId, :final version): ...
    case StationCounterUpdated(): ...
    case UnknownHubEvent(): ...
    case MalformedHubEvent(): ...
    // ลบบรรทัดใดบรรทัดหนึ่งออก แล้วคอมไพล์ไม่ผ่านทันที
  }
}

วันที่เพิ่มเหตุการณ์ใหม่ คอมไพเลอร์จะชี้ให้เห็นทุกที่ที่ต้องตัดสินใจว่าจะทำอย่างไรกับมัน — ต่างจากการใช้ if (event.name == '...') ที่เพิ่มแล้วไม่มีใครรู้ว่าต้องไปแก้ที่ไหนบ้าง

สังเกตว่า case StationCounterUpdated(): break; ที่ไม่ทำอะไร ก็ยังต้องเขียน — และนั่นคือข้อดี เพราะมันคือการบันทึกว่า “จอนี้ตัดสินใจแล้วว่าไม่สนใจ” ไม่ใช่ “ลืม”

8. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
การเชื่อมต่อหลุดเป็นระยะโดยไม่มีสาเหตุตัวแปลงโยนเมื่อเจอข้อมูลแปลกห้ามโยน คืนค่าที่บอกว่าเสียแทน
สัญญาพังมาสามเดือนแล้วไม่มีใครรู้กลืนเงียบด้วยการคืน nullแยก unknown ออกจาก malformed
แปลงไม่ผ่านทั้งที่ข้อมูลถูกตรวจ is Map<String, dynamic> อย่างเดียวรองรับ Map ทั่วไปด้วย
เลขที่มาเป็น 3.0 แปลงไม่ผ่านตรวจ is int อย่างเดียวรับ double ที่เป็นจำนวนเต็ม
เพิ่มเหตุการณ์ใหม่แล้วลืมจัดการในบางจอใช้ if เทียบชื่อใช้ sealed + switch
แอปรุ่นเก่าพังเมื่อเซิร์ฟเวอร์อัปเดตไม่มีกรณี unknownให้ชื่อที่ไม่รู้จักตกไปที่ UnknownHubEvent

9. แบบฝึกหัด

ทีมหลังบ้านเปลี่ยน version จากจำนวนเต็มเป็นสตริงแบบ "3" โดยไม่ได้บอกล่วงหน้า

  1. แอปจะเป็นอย่างไร
  2. ควรรับสตริงด้วยไหม
แนวคำตอบ

1. ทุกเหตุการณ์กลายเป็น MalformedHubEvent ผลคือ

  • การเชื่อมต่อยังอยู่ ไม่หลุด — นี่คือสิ่งที่การออกแบบนี้ซื้อมา
  • หน้าจอไม่อัปเดตจากเหตุการณ์อีกต่อไป แต่ยังถูกต้องอยู่เพราะยังโหลดผ่าน HTTP ได้
  • malformedEventCount เพิ่มขึ้นเรื่อย ๆ ซึ่งเป็นสัญญาณที่ทีมเห็นได้

เทียบกับการออกแบบที่โยน — การเชื่อมต่อจะหลุดทันทีที่เหตุการณ์แรกมาถึง แล้วต่อใหม่ แล้วหลุดอีก วนไปทั้งวัน กินแบตและกินเซิร์ฟเวอร์ โดยที่อาการที่ผู้ใช้เห็นคือ “แอปช้าและร้อน”

2. คำถามนี้เป็นคำถามเดียวกับ 6.6 เรื่องการรองรับสองชื่อฟิลด์ และคำตอบเป็นแบบเดียวกัน

รับสตริงด้วยไม่รับ
แอปรุ่นเก่าที่ติดตั้งไปแล้วยังทำงานได้สัญญาชัด ใครผิดรู้ทันที
ต้องดูแลสองรูปแบบตลอดไปต้องอัปเดตแอปทุกเครื่องพร้อมเซิร์ฟเวอร์

คำตอบที่ใช้จริงคือรับ แต่บันทึกไว้

int _requireInt(Map<String, dynamic> json, String key) {
  final value = json[key];
  if (value is int) return value;
  if (value is double && value == value.roundToDouble()) return value.toInt();

  // ถ้าตัวนับนี้ขึ้น แปลว่าสัญญาถูกเปลี่ยนโดยไม่ได้ตกลงกัน
  if (value is String && int.tryParse(value) case final parsed?) {
    legacyStringVersionCount++;
    return parsed;
  }
  throw _FieldError('$key ต้องเป็นจำนวนเต็ม แต่ได้ ${value.runtimeType}');
}

เหตุผลที่เลือกทางนี้เฉพาะกับเรียลไทม์ ทั้งที่ Part 6 เข้มงวดกว่า คือความเสียหายไม่เท่ากัน — HTTP ที่ล้มเหลวผู้ใช้เห็นและลองใหม่ได้ ส่วนเรียลไทม์ที่พังจะเงียบและกินทรัพยากรไปเรื่อย ๆ

แต่ต้องมีวันหมดอายุ ให้เปิดตั๋วงานไว้ว่าจะลบบรรทัดนี้เมื่อเครื่องทุกเครื่องอัปเดตแล้ว ไม่งั้นมันจะอยู่ตลอดกาลและไม่มีใครกล้าลบ

10. สรุป

  • hub ส่ง List<Object?>? ซึ่งเป็นชนิดที่บอกอะไรไม่ได้เลย
  • ตัวแปลงห้ามโยน เพราะการโยนทำให้การเชื่อมต่อหลุดและเงียบทั้งวัน
  • แต่ต้องไม่กลืนเงียบ — แยก “ยังไม่รู้จัก” ออกจาก “ข้อมูลผิด”
  • ข้างในโยน ข้างนอกจับ ทำให้เพิ่มเหตุการณ์ใหม่ไม่ต้องเขียนซ้ำ
  • sealed ทำให้ลืมจัดการเหตุการณ์ใหม่ไม่ได้
