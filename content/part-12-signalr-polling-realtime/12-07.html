ระดับ: กลางใช้เวลา: 45–60 นาทีลงมือทำ: อ่านลำดับใน initState ของโปรเจกต์จริง
ต้องผ่านอะไรมาก่อน: 12.6, 3.12 (วงจรชีวิตของ State)
จบบทนี้แล้วจะทำได้: ลงทะเบียนฟังในจังหวะที่ถูก และรู้ว่าจังหวะไหนทำให้เหตุการณ์หาย

1. ลงทะเบียนที่ไหน

class _PutawayScreenState extends State<PutawayScreen> {
  @override
  void initState() {
    super.initState();
    _signalR.addStationDispatchedListener(_handleStationDispatched);
    _signalR.addPalletArrivedListener(_handlePalletArrived);
    _signalR.addPalletReturnedListener(_handlePalletReturned);
    _signalR.addLabelingCompletedListener(_handleLabelingCompleted);
    _initSignalR();
    _loadStationStatus();
  }
}
ที่ถูกไหมเพราะ
initStateถูกถูกเรียกครั้งเดียวตลอดอายุของ State
buildผิดถูกเรียกหลายครั้ง จะลงทะเบียนซ้ำเรื่อย ๆ
constructor ของ Stateผิดยังไม่มี widget ให้อ่าน
didChangeDependenciesได้ ถ้าต้องใช้ contextแต่ถูกเรียกได้หลายครั้ง ต้องกันซ้ำเอง
อาการของการลงทะเบียนใน build คือ เหตุการณ์เดียวถูกจัดการหลายครั้ง และจำนวนครั้งเพิ่มขึ้นเรื่อย ๆ ตามการวาดใหม่ — จอที่เปิดไว้นาน ๆ จะทำงานช้าลงและอาจเปิด dialog ซ้อนกันสิบอัน

2. ลำดับที่ถูกต้อง

_signalR.addStationDispatchedListener(...);   // 1. ลงทะเบียนฟัง
_initSignalR();                                // 2. ต่อ
_loadStationStatus();                          // 3. โหลดสถานะปัจจุบัน

สามบรรทัดนี้เรียงถูกแล้ว และเหตุผลของแต่ละลำดับต่างกัน

ถ้าสลับ 1 กับ 2เหตุการณ์ที่มาถึงระหว่างต่อจนถึงตอนลงทะเบียน จะหายไป
ถ้าสลับ 2 กับ 3ไม่มีผลมาก แต่ 3 ควรมาทีหลังเพื่อให้ข้อมูลใหม่ที่สุดเป็นตัวชนะ

lab เขียนกฎข้อแรกไว้เป็นคอมเมนต์

// ฟังก่อนต่อ ไม่ใช่ต่อก่อนฟัง — เหตุการณ์ที่มาถึงระหว่างที่ยังไม่ได้
// ลงทะเบียนฟัง จะหายไปเงียบ ๆ
_eventSub = client.events.listen(_onEvent);
_statusSub = client.status.listen(_onStatus);

await client.acquire();
await load();

3. ปัญหาการแข่งกันระหว่างโหลดกับเหตุการณ์

ลำดับที่ 2 กับ 3 เปิดช่องให้เกิดเรื่องนี้

10:00:00.000  ต่อสำเร็จ
10:00:00.010  เริ่มยิง GET /tasks
10:00:00.100  เหตุการณ์ TaskCompleted(T-1) มาถึง → ทำเครื่องหมายว่าเสร็จ
10:00:00.400  GET ตอบกลับ (ข้อมูลถ่ายไว้ตอน 10:00:00.010 — T-1 ยังค้าง)
              → เขียนทับ → T-1 กลับมาค้างอีก ทั้งที่เสร็จไปแล้ว

เหตุการณ์มาถึงก่อน แต่ผลของมันถูกลบโดยข้อมูลที่เก่ากว่าแต่มาถึงทีหลัง

นี่คือเหตุผลที่ทุกเหตุการณ์ต้องมีเลขรุ่น และทำไม lab ตรวจก่อนใช้เสมอ

if (version <= existing.version) return; // เก่ากว่าหรือซ้ำ ทิ้ง

เมื่อ load() เขียนทับทั้งชุด มันเขียนด้วยเลขรุ่นที่มากับข้อมูล ถ้าเซิร์ฟเวอร์ทำงานถูกต้อง ข้อมูลที่ถ่ายตอน 10:00:00.010 จะมีรุ่นเก่ากว่าเหตุการณ์ — แต่การเขียนทับทั้งชุดก็ยังทับอยู่ดี

ทางแก้ที่ครบกว่า คือให้ load() ผสานแทนที่จะเขียนทับ โดยเทียบเลขรุ่นทีละรายการ — แต่ต้องแลกกับความซับซ้อน และต้องตอบให้ได้ว่ารายการที่หายไปจากคำตอบของเซิร์ฟเวอร์ควรถูกลบไหม 12.10 จะพูดถึงการแลกนี้

4. mounted ก่อนแตะหน้าจอเสมอ

void _onCounterUpdated(Map<String, dynamic> data) {
  final stationId = data['stationId'] as int?;
  if (stationId == null || !mounted) return;
  ...
  setState(() => ...);
}

เหตุการณ์มาถึงได้ทุกเมื่อ รวมถึงเสี้ยววินาทีหลังจากผู้ใช้กดย้อนกลับ ถ้าเรียก setState บน State ที่ถูกทิ้งแล้วจะได้

setState() called after dispose(): _SortingScreenState#a1b2c(lifecycle state: defunct)

แต่สังเกตว่านี่เป็นทางแก้ชั้นที่สอง — ชั้นแรกคือยกเลิกการฟังใน dispose ซึ่งทำให้ callback ไม่ถูกเรียกตั้งแต่ต้น (12.8)

ชั้นทำอะไรพอไหม
ยกเลิกการฟังใน disposeไม่ถูกเรียกอีกเลยพอ ถ้าทำครบ
เช็ค mountedถูกเรียกแต่ไม่ทำอะไรเป็นตาข่ายกันพลาด

มีทั้งสองชั้นดีที่สุด เพราะชั้นแรกลืมง่ายเมื่อมี listener หลายตัว

5. จอที่ไม่ควรฟังตอนไม่ได้แสดงอยู่

จอที่ถูก push ทับ ยังมี State อยู่และยังได้รับเหตุการณ์ต่อไป ซึ่งบางครั้งถูก บางครั้งไม่

สถานการณ์ควรฟังต่อไหม
จอรายการถูก push ทับด้วยจอรายละเอียดควร — กลับมาแล้วข้อมูลจะทันสมัย
จอที่แสดง dialog เตือนเมื่อมีเหตุการณ์ไม่ควร — จะเด้ง dialog ทับจอที่อยู่ข้างบน
จอที่เล่นเสียงเมื่อมีเหตุการณ์ไม่ควร — ผู้ใช้จะได้ยินเสียงโดยไม่เห็นว่ามาจากไหน

สองแถวล่างแก้ด้วยการตรวจว่าเส้นทางนี้อยู่บนสุดหรือไม่

void _onEvent(HubEvent event) {
  if (!mounted) return;
  // จอนี้ไม่ได้อยู่บนสุด — เก็บไว้ก่อน อย่าเด้ง dialog ทับจอที่ผู้ใช้ดูอยู่
  if (ModalRoute.of(context)?.isCurrent != true) {
    _pendingAlerts.add(event);
    return;
  }
  _showAlert(event);
}

แล้วแสดงที่ค้างไว้ตอนกลับมา ซึ่งทำได้ใน didPopNext ของ RouteAware

6. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
เหตุการณ์เดียวถูกจัดการหลายครั้งลงทะเบียนใน buildย้ายไป initState
เหตุการณ์แรก ๆ หายไปต่อก่อนฟังฟังก่อนต่อ
งานที่เสร็จแล้วกลับมาค้างคำตอบของ HTTP ที่เก่ากว่ามาถึงทีหลังเทียบเลขรุ่น (12.10)
setState() called after dispose()ไม่ได้ยกเลิกการฟังยกเลิกใน dispose + เช็ค mounted
dialog เด้งทับจอที่ผู้ใช้กำลังดูจอที่ถูก push ทับยังฟังอยู่ตรวจ ModalRoute.of(context)?.isCurrent
ลงทะเบียนใน didChangeDependencies แล้วซ้ำเมธอดนั้นถูกเรียกได้หลายครั้งกันด้วยธง หรือย้ายไป initState

7. แบบฝึกหัด

จอ “จุดคัดแยก” ต้องส่งเสียงเตือนและสั่นเมื่อได้รับเหตุการณ์ StationFull ผู้ใช้รายงานสองเรื่อง

  1. “บางครั้งเสียงดังทั้งที่ปิดจอนั้นไปแล้ว”
  2. “บางครั้งเสียงดังสามครั้งติดกัน”
แนวคำตอบ

1. ไม่ได้ยกเลิกการฟังตอนออกจากจอ — State ถูกทิ้งแล้วแต่ callback ยังอยู่ในรายชื่อของ singleton

สังเกตว่าอาการนี้ การเช็ค mounted ไม่ช่วย ถ้าการเล่นเสียงไม่ได้ผ่าน setState

void _onStationFull(Map<String, dynamic> data) {
  HapticFeedback.heavyImpact();     // ← ทำงานได้แม้ State ถูกทิ้งแล้ว
  _player.play(alarmSound);         // ← ทำงานได้เช่นกัน
  if (!mounted) return;             // ← สายไปแล้ว
  setState(() => ...);
}

ทางแก้ที่ถูกคือยกเลิกการฟังใน dispose (12.8) และถ้าจะเช็ค mounted ต้องเช็คบรรทัดแรก ไม่ใช่ก่อน setState

2. เสียงดังสามครั้งมีสองสาเหตุที่เป็นไปได้ และต้องแยกให้ออก

สาเหตุวิธีตรวจ
ลงทะเบียนซ้ำสามครั้งนับจำนวนใน _stationFullListeners ตอนรัน
เซิร์ฟเวอร์ส่งเหตุการณ์ซ้ำจริงบันทึกทุกเหตุการณ์ที่ได้รับพร้อมเวลาและเลขรุ่น

ถ้าเป็นข้อแรก มักเกิดจากการที่จอถูกสร้างใหม่โดยที่ตัวเก่ายังไม่ถูกทิ้ง — เช่นการ push จอเดิมซ้ำ หรือ didChangeDependencies ที่ถูกเรียกหลายครั้ง

ถ้าเป็นข้อสอง ให้กันที่ฝั่งแอปด้วยเลขรุ่น

if (version <= _lastAlertedVersion[stationId]) return;   // เตือนไปแล้ว
_lastAlertedVersion[stationId] = version;
_alert();

ข้อคิดที่ใหญ่กว่าทั้งสองข้อ: การกระทำที่ผู้ใช้รับรู้ได้ — เสียง การสั่น การแจ้งเตือน — ต้องผ่านการกันซ้ำเสมอ เพราะมันคือสิ่งที่ผู้ใช้สังเกตเห็นทันทีเมื่อผิด ต่างจากการอัปเดตตัวเลขบนหน้าจอที่ทำซ้ำกี่ครั้งก็ได้ผลเหมือนเดิม

เป็นหลักเดียวกับ Idempotency-Key ใน 7.13 แต่อยู่ฝั่งขาเข้าแทนที่จะเป็นขาออก

8. สรุป

  • ลงทะเบียนใน initState ไม่ใช่ใน build
  • ฟังก่อนต่อ ไม่ใช่ต่อก่อนฟัง
  • คำตอบของ HTTP ที่มาทีหลังอาจเก่ากว่าเหตุการณ์ที่มาก่อน
  • เช็ค mounted เป็นบรรทัดแรก ไม่ใช่ก่อน setState
  • จอที่ถูก push ทับยังได้รับเหตุการณ์ — ตรวจก่อนเด้ง dialog หรือเล่นเสียง
  • การกระทำที่ผู้ใช้รับรู้ได้ ต้องกันซ้ำเสมอ

บทหน้า: เลิกฟังให้ครบ และการรั่วที่ไม่มีใครเห็น
