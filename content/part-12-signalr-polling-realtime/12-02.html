จบบทนี้แล้วจะทำได้: อ่านโค้ด hub ฝั่ง .NET พอที่จะรู้ว่าแอปต้องรออะไร และถามคำถามที่ถูกก่อนเริ่มเขียน

1. hub คืออะไร

Hub คือคลาสฝั่ง .NET ที่เป็นจุดนัดพบระหว่างเซิร์ฟเวอร์กับ client ที่ต่ออยู่ — เซิร์ฟเวอร์เรียกเมธอดบน client ได้ และ client เรียกเมธอดบนเซิร์ฟเวอร์ได้
// ฝั่ง server (แนวคิด — รายละเอียดอยู่ในคอร์ส .NET Part 23)
public class PutawayHub : Hub
{
    public async Task JoinStation(string stationId)
        => await Groups.AddToGroupAsync(Context.ConnectionId, stationId);
}
// Program.cs
app.MapHub<PutawayHub>("/hubs/putaway");
app.MapHub<SortingHub>("/hubs/sorting");

บรรทัด MapHub คือสิ่งที่แอปต้องรู้ — มันบอกว่า hub อยู่ที่ URL ไหน ซึ่งตรงกับที่ wmsapp เขียนไว้

final serverUrl = await ApiService.resolveServerUrl();
final hubUrl = '$serverUrl/hubs/putaway';

2. เซิร์ฟเวอร์ส่งเหตุการณ์อย่างไร

// ฝั่ง server (แนวคิด)
public class PutawayService
{
    private readonly IHubContext<PutawayHub> _hub;

    public async Task DispatchAsync(string stationId, string palletCode)
    {
        // ...ทำงานจริงในฐานข้อมูล...

        await _hub.Clients.All.SendAsync("StationDispatched", new {
            stationId,
            palletCode,
            version = task.Version,
        });
    }
}
ส่วนความหมายสำหรับฝั่งแอป
"StationDispatched"ชื่อที่แอปต้องใช้ตอนลงทะเบียนฟัง ต้องตรงตัวพิมพ์เป๊ะ
object ที่ส่งไปกลายเป็น argument ตัวแรกที่แอปได้รับ
Clients.Allทุกคนที่ต่ออยู่ได้รับ

มีอีกหลายแบบที่เลือกผู้รับได้ และแต่ละแบบมีผลกับแอป

เขียนฝั่ง .NETใครได้รับแอปต้องทำอะไรเพิ่ม
Clients.Allทุกเครื่องกรองเองว่าเหตุการณ์นี้เกี่ยวกับจอที่เปิดอยู่ไหม
Clients.Group("ST-01")เฉพาะที่เข้ากลุ่มนั้นต้องเรียก JoinStation ก่อน และเรียกใหม่ทุกครั้งที่ต่อกลับ
Clients.User(userId)เฉพาะผู้ใช้คนนั้นต้องส่ง token ตอนต่อ
Clients.Callerเฉพาะคนที่เพิ่งเรียกมาใช้เป็นคำตอบของคำสั่ง
แถวที่สองมีกับดักที่เจ็บ — การเข้ากลุ่มผูกกับ connection ไม่ใช่ผูกกับผู้ใช้ เมื่อสายหลุดแล้วต่อใหม่ จะได้ connection ใหม่ที่ไม่อยู่ในกลุ่มเดิม ถ้าไม่เข้ากลุ่มซ้ำ จอจะเงียบตลอดกาลโดยที่สถานะบอกว่า “เชื่อมต่ออยู่” (12.4)

3. สัญญาที่ต้องตกลงก่อนเขียนโค้ด

เหมือน endpoint ของ HTTP เหตุการณ์ก็เป็นสัญญา และต้องตกลงห้าข้อนี้ให้ชัด

ข้อตัวอย่างคำตอบถ้าไม่ถามแล้วเกิดอะไร
URL ของ hub/hubs/putawayต่อไม่ติดโดยไม่มีข้อความบอก
ชื่อเหตุการณ์ทั้งหมดStationDispatched, PalletArrivedลงทะเบียนผิดชื่อ แล้วไม่มีอะไรเกิดขึ้นเลย — ไม่มีข้อผิดพลาดใด ๆ
หน้าตาของข้อมูลแต่ละเหตุการณ์ชื่อฟิลด์และชนิดแปลงไม่ได้ตอนรัน
ต้องเข้าสู่ระบบไหมต้อง / ไม่ต้องต่อได้แต่ไม่ได้รับอะไร
มีกลุ่มไหมมี ต้องเรียก JoinStationเงียบตลอดกาล

แถวที่สองอันตรายที่สุด และเป็นเรื่องที่ต่างจาก HTTP อย่างชัดเจน

// พิมพ์ชื่อผิดตัวเดียว
_connection!.on('StationDispached', _onStationDispatched);
//                          ↑ ขาด t

// ผลลัพธ์: คอมไพล์ผ่าน ต่อสำเร็จ สถานะเขียว และไม่มีอะไรเกิดขึ้นตลอดไป
HTTP ที่ยิงผิด path จะได้ 404 กลับมาทันที แต่การลงทะเบียนฟังเหตุการณ์ที่ไม่มีอยู่จริง ไม่มีข้อผิดพลาดใด ๆ — SignalR ไม่รู้ว่าเซิร์ฟเวอร์จะส่งอะไรบ้าง

4. ป้องกันชื่อผิดด้วยค่าคงที่

/// ชื่อเหตุการณ์ทั้งหมดของ hub นี้ รวมไว้ที่เดียว
///
/// เขียนชื่อเป็นสตริงตรงในที่ที่ใช้ แปลว่าพิมพ์ผิดแล้วไม่มีใครรู้
class PutawayEvents {
  const PutawayEvents._();

  static const stationDispatched = 'StationDispatched';
  static const palletArrived = 'PalletArrived';
  static const palletReturned = 'PalletReturned';
  static const labelingCompleted = 'LabelingCompleted';

  /// ใช้ในเทสต์ที่ยิงเหตุการณ์ทุกตัวเพื่อดูว่าแอปรับได้ครบ
  static const all = [
    stationDispatched, palletArrived, palletReturned, labelingCompleted,
  ];
}

ยังไม่ได้ป้องกันชื่อที่ไม่ตรงกับฝั่งเซิร์ฟเวอร์ — มันแค่ทำให้ทั้งแอปพิมพ์ผิดพร้อมกัน ซึ่งอย่างน้อยก็หาเจอที่เดียว

วิธีที่ป้องกันได้จริงมีสองทาง

ทางได้อะไรต้องมีอะไร
เทสต์ที่ยิงเข้า hub จริงจับได้ทันทีเมื่อฝั่งเซิร์ฟเวอร์เปลี่ยนชื่อเซิร์ฟเวอร์ทดสอบที่รันได้ใน CI (Part 14)
ให้เซิร์ฟเวอร์ตอบรายชื่อเหตุการณ์ที่รองรับแอปตรวจเองได้ตอนต่อทีมหลังบ้านต้องทำ endpoint เพิ่ม

ทางที่สองน่าสนใจสำหรับระบบที่แอปกับเซิร์ฟเวอร์อัปเดตไม่พร้อมกัน — ต่อยอดจาก health payload ที่ 10.9 ทำไว้แล้ว

5. การส่งต่อกับความน่าเชื่อถือ

SignalR ไม่รับประกันว่าข้อความจะถึง — ถ้า client หลุดตอนที่เซิร์ฟเวอร์ส่ง ข้อความนั้นหายไปเลย ไม่มีการส่งซ้ำ ไม่มีคิวรอ

ข้อนี้เปลี่ยนวิธีออกแบบทั้งหมด และเป็นที่มาของกฎสำคัญที่สุดใน Part นี้

อย่าออกแบบให้ให้ออกแบบให้
สถานะของแอปสร้างจากลำดับเหตุการณ์ทั้งหมดเหตุการณ์เป็นแค่ตัวบอกว่า “มีอะไรเปลี่ยน”
พลาดเหตุการณ์เดียวแล้วผิดตลอดไปโหลดใหม่ผ่าน HTTP แล้วกลับมาถูกได้เสมอ

วิธีตรวจว่าออกแบบถูกคือถามคำถามเดียว — ถ้าปิดแอปแล้วเปิดใหม่ ข้อมูลจะถูกต้องไหม ถ้าตอบว่าใช่ แปลว่าไม่ได้พึ่งเหตุการณ์เกินไป

6. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
ต่อสำเร็จแต่ไม่ได้รับอะไรเลยชื่อเหตุการณ์ผิด หรือยังไม่ได้เข้ากลุ่มตรวจกับฝั่งเซิร์ฟเวอร์ทีละตัว
ต่อกลับได้แล้วเงียบconnection ใหม่ไม่ได้อยู่ในกลุ่มเดิมเข้ากลุ่มซ้ำทุกครั้งที่ต่อ (12.4)
ได้รับเหตุการณ์ของสถานีอื่นเซิร์ฟเวอร์ใช้ Clients.Allกรองฝั่งแอป หรือขอให้เปลี่ยนเป็นกลุ่ม
401 ตอนต่อhub ต้องการ token แต่ไม่ได้ส่งส่งผ่าน accessTokenFactory (12.3)
ข้อมูลผิดหลังใช้ไปนานสร้างสถานะจากเหตุการณ์อย่างเดียวโหลดใหม่เป็นระยะ (12.10)

7. แบบฝึกหัด

ทีมหลังบ้านจะทำ hub ใหม่สำหรับงานหยิบของ และถามว่าฝั่งแอปอยากได้เหตุการณ์อะไรบ้าง

  1. จะขออะไร
  2. มีอะไรที่ควรบอกเขาว่า “อย่าทำ”
แนวคำตอบ

1. ขอเหตุการณ์ที่ตอบคำถามว่า “มีอะไรเปลี่ยน” ไม่ใช่ “ข้อมูลทั้งหมดคืออะไร”

ขอข้อมูลที่ต้องมี
PickAssignedpickId, assignee, version
PickCompletedpickId, version
ConveyorFullconveyorNo, count

สิ่งที่ต้องขอให้มีทุกเหตุการณ์คือ version หรือเวลาที่เกิด — เพราะเหตุการณ์ที่มาถึงไม่เรียงลำดับ และถ้าไม่มีตัวเทียบ แอปจะบอกไม่ได้ว่าอันไหนใหม่กว่า (12.10)

2. สามอย่างที่ควรบอกว่าอย่าทำ

  • อย่าส่งข้อมูลทั้งลิสต์มาในเหตุการณ์ — เหตุการณ์ที่ส่ง 200 รายการมาทุกครั้งที่มีอะไรเปลี่ยน แพงกว่าการถามซ้ำเสียอีก และทำให้ประโยชน์ของเรียลไทม์หายไปหมด
  • อย่าใส่ข้อมูลอ่อนไหว — ถ้า hub ใช้ Clients.All ทุกเครื่องในคลังจะได้รับ รวมถึงเครื่องของคนที่ไม่ควรเห็น (8.9)
  • อย่าให้เหตุการณ์เป็นทางเดียวที่จะรู้ผล — คำสั่งที่ผู้ใช้กดต้องได้คำตอบผ่าน HTTP ด้วย ไม่ใช่รอเหตุการณ์อย่างเดียว เพราะถ้าเหตุการณ์นั้นหายไป ผู้ใช้จะรอตลอดกาลโดยไม่รู้ว่าคำสั่งสำเร็จหรือไม่ (7.10)

ข้อสุดท้ายเป็นข้อที่ผิดกันบ่อยที่สุดตอนเริ่มใช้เรียลไทม์ใหม่ ๆ เพราะมันดูสวยกว่า — กดปุ่ม แล้วรอให้เหตุการณ์มาอัปเดตหน้าจอ แต่ในความจริง คำสั่งกับการแจ้งเตือนเป็นคนละเรื่อง และต้องเชื่อถือได้คนละระดับ

8. สรุป

  • hub คือจุดนัดพบ และ MapHub บอกว่ามันอยู่ URL ไหน
  • ตกลงห้าข้อก่อนเขียน — URL, ชื่อเหตุการณ์, หน้าตาข้อมูล, การเข้าสู่ระบบ, กลุ่ม
  • ชื่อเหตุการณ์ที่ผิด ไม่มีข้อผิดพลาดใด ๆ ให้เห็น
  • กลุ่มผูกกับ connection ต่อใหม่แล้วต้องเข้ากลุ่มซ้ำ
  • SignalR ไม่รับประกันว่าข้อความจะถึง — ออกแบบให้กลับมาถูกได้เสมอ
