1. การ์ดข้อมูลพาเลท

class StationPalletInfoCard extends StatelessWidget {
  final PutawayPalletInfo pallet;
  final bool hasFixedDestination;
  final VoidCallback onViewItems;
  ...
}

รับสามอย่าง: ข้อมูล เงื่อนไขหนึ่งข้อ และ callback

InfoRow(label: 'สถานะ', value: pallet.status),
InfoRow(
  label: 'ปลายทาง',
  value: hasFixedDestination ? 'ASRS (convert PW->FG)' : 'เลือกด้านล่าง',
),
InfoRow(label: 'สินค้า', value: '${pallet.items.length} รายการ'),

ใช้ InfoRow จาก common_widgets.dart (16.9) — ป้ายกำกับกว้าง 110 เท่ากันทุกแถว ค่าชิดซ้ายตรงกัน

แต่แถวกลางมีปัญหา

value: hasFixedDestination ? 'ASRS (convert PW->FG)' : 'เลือกด้านล่าง',
ปัญหารายละเอียด
ค่า 'ASRS' ถูกฝังไว้ที่จริงมาจาก station.fixedDestination ซึ่งอาจไม่ใช่ ASRS
คำว่า convert PW->FG ถูกฝังเป็นผลลัพธ์ของสถานีฝั่งส่ง ไม่ใช่ของทุกสถานีที่มีปลายทางตายตัว
hasFixedDestination เป็น boolเสียข้อมูลว่าปลายทางคืออะไร

ที่ผ่านมาไม่พังเพราะสถานีเดียวที่มี fixedDestination คือฝั่งส่ง และปลายทางคือ ASRS — แต่ 18.3 แบบฝึกหัดแสดงว่าถ้าเพิ่มสถานี QC ที่ปลายทางเป็น QC การ์ดจะบอกผู้ใช้ว่าจะส่งไป ASRS

// รับค่าจริงแทน bool — ไม่มีทางบอกผิด
final String? fixedDestination;
...
InfoRow(
  label: 'ปลายทาง',
  value: fixedDestination ?? 'เลือกด้านล่าง',
),
รูปแบบที่เจอซ้ำ: ส่ง bool เข้ามาแทนค่าจริง แล้วประกอบข้อความเอง
เกิดจากการที่ตอนเขียนมีแค่กรณีเดียว จึงดูเหมือนไม่ต่างกัน — แต่ bool ตอบได้แค่ “มีไหม” ส่วนค่าจริงตอบได้ว่า “คืออะไร” ซึ่งเป็นสิ่งที่หน้าจอต้องแสดง

3. สวิตช์ที่กดได้สองที่

return GestureDetector(
  onTap: () => onChanged(!value),           // ← แตะทั้งแถว
  child: AnimatedContainer(
    ...
    child: Row(
      children: [
        Icon(...),
        Expanded(child: Column(...)),
        Switch(
          value: value,
          onChanged: onChanged,             // ← หรือแตะที่สวิตช์
          activeThumbColor: AppTheme.primary,
        ),
      ],
    ),
  ),
);

แตะที่ไหนก็ได้ในแถว หรือแตะที่สวิตช์โดยตรง

นี่คือสิ่งที่จำเป็นบน PDA — สวิตช์ Material มีขนาดจับราว 40×40 พิกเซล ซึ่งต่ำกว่าเกณฑ์ 48 ที่ AppTheme.touchTarget ตั้งไว้ (16.2)
การทำให้ทั้งแถวแตะได้ทำให้พื้นที่จับกลายเป็นเต็มความกว้าง ซึ่งกดโดนง่ายกว่ามากเมื่อใส่ถุงมือ

แต่มีกับดักที่ต้องรู้

onTap: () => onChanged(!value),      // สลับค่า
onChanged: onChanged,                 // Switch ส่งค่าใหม่มาเอง
ผู้ใช้แตะเรียกอะไรผลลัพธ์
ที่แถวonChanged(!value)สลับ
ที่สวิตช์onChanged(ค่าใหม่จาก Switch)สลับ

ทั้งสองทางได้ผลเดียวกันเพราะ Switch ส่งค่าตรงข้ามมาให้อยู่แล้ว — ถ้าเขียนผิดเป็น onChanged: (_) => onChanged(!value) ก็ยังทำงานได้ แต่จะพังถ้ามีวิธีสั่งค่าจากที่อื่น

สังเกตว่า GestureDetector ครอบ Switch อยู่ — การแตะที่สวิตช์จะถูก Switch จับก่อน ไม่ทะลุไปถึง GestureDetector จึงไม่สลับสองครั้ง

ข้อความในสวิตช์บอกทั้งสิ่งที่ทำและเหตุผล

Text('พันสินค้า (Wrapping Machine)')      // ทำอะไร
Text('พัน Pallet ก่อนนำเข้า ASRS')        // เมื่อไรและทำไม

บรรทัดที่สองอธิบายว่าทำไมสวิตช์นี้ถึงมีเฉพาะตอนเลือก ASRS — ผู้ใช้ไม่ต้องเดา

สองที่แรกตรงกัน ที่สามไม่ตรง — FG ได้สีน้ำเงินแทนเขียว

ผู้ใช้ที่สแกนพาเลท FG จะเห็นป้ายสีเขียวบนการ์ด แล้วหัวข้อตัวเลือกปลายทางเป็นสีน้ำเงินทันทีข้างล่าง

เทียบกับ StatusBadge ใน 16.9 ซึ่งเป็นตารางแปลงสถานะเป็นสีที่อยู่ที่เดียวและทุกจอใช้ร่วมกัน
ประเภทพาเลทไม่มีตารางแบบนั้น จึงถูกเขียนซ้ำสามที่และไม่ตรงกันหนึ่งที่

สังเกตว่า StatusBadge รองรับ 'FG' และ 'PW' อยู่แล้ว

Color get _color => switch (status.toUpperCase()) {
  'FG' => AppTheme.success,
  'PW' => AppTheme.warning,
  ...

ป้ายประเภทในการ์ดจึงใช้ StatusBadge(pallet.type) ได้เลย แทนที่จะสร้าง Container เอง — ได้สีที่ตรงกับทุกจอฟรี และโค้ดสั้นลง 15 บรรทัด

5. ข้อความจากเซิร์ฟเวอร์ที่แสดงเมื่อมี

if (pallet.message.isNotEmpty) ...[

message มาจากเซิร์ฟเวอร์และถูกกันไว้ในโมเดล

ค่าเริ่มต้นเป็นสตริงว่าง แล้วหน้าจอตรวจ isNotEmpty ก่อนแสดง — ผลคือถ้าเซิร์ฟเวอร์ไม่ส่งอะไรมา ไม่มีช่องว่างลอยบนจอ

วิธีเก็บวิธีตรวจผล
String + ?? ''isNotEmptyทำงานได้ทั้งกรณีไม่ส่งและส่งค่าว่าง
String?!= nullพลาดกรณีที่ส่ง '' มา

แบบแรกทนกว่าและใช้ที่นี่ถูกต้อง — ตรงกับที่ 16.5 พบใน getCustomBase() ที่ตรวจทั้ง null และว่าง

ช่องนี้เป็นทางที่เซิร์ฟเวอร์บอกอะไรเพิ่มได้โดยไม่ต้องแก้แอป เช่น “พาเลทนี้ค้างมา 3 วัน” หรือ “มีของหมดอายุใน 7 วัน”

เข้าใจผิดที่จริง

8. แบบฝึกหัด

ผู้ใช้แจ้งว่า “พาเลท FG กับ PW ดูคล้ายกันไปหมด บางทีสแกนผิดแล้วไม่รู้ตัวจนกดยืนยัน”

  1. ตอนนี้มีสัญญาณอะไรบอกประเภทบ้าง และทำไมยังไม่พอ
  2. แก้อย่างไรโดยไม่เพิ่มขั้นตอนให้ผู้ใช้
  3. ควรเพิ่มการยืนยันไหม
แนวคำตอบ

ข้อ 1 — สัญญาณที่มีอยู่

ที่สัญญาณเด่นแค่ไหน
การ์ดข้อมูลพาเลทป้ายเล็กมุมขวา สีเขียว/เหลือง ตัวอักษร 13เล็ก
หัวข้อตัวเลือกปลายทางข้อความ “Pallet ประเภท FG” + สีปานกลาง แต่สีไม่ตรงกับการ์ด
ตัวเลือกที่แสดงPW เห็น Prework, FG เห็น Replenishต้องรู้กฎถึงจะสังเกต
กล่องยืนยันไม่บอกประเภทเลยไม่มี

สาเหตุที่ยังไม่พอมีสามข้อ

  1. ป้ายประเภทเล็กและอยู่มุมขวาซึ่งไม่ใช่ที่ที่สายตาไป
  2. สีไม่สม่ำเสมอ ทำให้สมองไม่จำว่า “เขียว = FG”
  3. กล่องยืนยันซึ่งเป็นด่านสุดท้าย ไม่มีข้อมูลนี้เลย

ข้อ 2 — แก้โดยไม่เพิ่มขั้นตอน

เริ่มจากทำสีให้ตรงกันก่อน เพราะเป็นการแก้ที่ถูกที่สุดและได้ผลทันที

// putaway/shared/models/putaway_rules.dart
Color palletTypeColor(String type) => switch (type) {
  'FG' => AppTheme.success,
  'PW' => AppTheme.warning,
  _ => AppTheme.purple,
};

แล้วใช้ทั้งสามที่ — ตอนนี้ FG เป็นเขียวทุกจอ

ต่อมาทำให้ป้ายเด่นขึ้นโดยใช้ของที่มีอยู่

// แทนที่ Container ที่เขียนเอง 15 บรรทัด
StatusBadge(pallet.type, fontSize: 15),

StatusBadge มี fontSize ให้ปรับอยู่แล้ว (16.9) และรู้จัก FG/PW อยู่แล้ว

สุดท้ายเติมประเภทเข้ากล่องยืนยัน ซึ่งเป็นจุดที่ได้ผลมากที่สุด

// ใน _confirmPutaway ของ StationSheet
message = 'เก็บ Pallet ${_pallet!.palletId} (ประเภท ${_pallet!.type})\n'
    'ที่ ${widget.station.id} -> $destLabel$wrappingNote\n\n'
    '${_wrappingRequired ? "จะเข้า Wrapping Machine ก่อน" : "AMR จะมารับทันที"}';
แก้อะไรผู้ใช้ต้องทำเพิ่มไหม
สีตรงกันทุกจอไม่
ป้ายใหญ่ขึ้นไม่
กล่องยืนยันบอกประเภทไม่ — อ่านสิ่งที่มีอยู่แล้ว

ข้อ 3 — ควรเพิ่มการยืนยันไหม

มีกล่องยืนยันอยู่แล้ว คำถามคือควรทำให้เข้มขึ้นไหม เช่นให้พิมพ์ประเภทซ้ำ

เข้มขึ้นข้อดีข้อเสีย
ให้เลือกประเภทซ้ำในกล่องผิดยากทุกครั้งที่ทำถูกก็ต้องทำ
เตือนเฉพาะเมื่อสถานีมี allowedTypeตรงจุดสถานีปกติไม่ได้ประโยชน์
ไม่เพิ่ม แค่แสดงให้ชัดไม่ช้าลงยังผิดได้ถ้าผู้ใช้ไม่อ่าน

คำตอบขึ้นกับว่าสแกนผิดแล้วเสียหายแค่ไหน

ถ้าควร
สั่งผิดแล้วยกเลิกได้ง่ายแค่แสดงให้ชัด
สั่งผิดแล้วหุ่นยนต์ขนไปที่ผิดและเอากลับยากเพิ่มการยืนยัน

ในกรณีของ putaway ที่ AMR มารับพาเลททันทีหลังยืนยัน (ตามข้อความในกล่อง) การสั่งผิดแปลว่าของถูกขนไปผิดที่จริง ๆ

ทางที่สมดุลคือเตือนเฉพาะเมื่อผู้ใช้กำลังจะทำสิ่งที่ผิดปกติ

// เตือนเฉพาะกรณีที่ประเภทกับปลายทางไม่เข้าคู่กันตามปกติ
final isUnusual = _pallet!.type == 'PW' && dest == 'ASRS';

final confirm = await showConfirmDialog(
  context,
  title: title,
  message: message,
  confirmLabel: 'ยืนยัน',
  isDanger: isUnusual,      // ปุ่มเป็นสีแดง ทำให้ต้องอ่านก่อนกด
);

showConfirmDialog มี isDanger อยู่แล้ว (16.9) — เปลี่ยนปุ่มเป็นสีแดงโดยไม่เพิ่มขั้นตอน

ข้อสรุป: ก่อนจะเพิ่มขั้นตอนให้ผู้ใช้ ให้ตรวจก่อนว่าสิ่งที่มีอยู่ถูกใช้เต็มที่แล้วหรือยัง — ที่นี่มีป้ายประเภท มีสี มีกล่องยืนยัน และมี isDanger ครบทุกอย่าง แค่ไม่ได้ใช้ให้ตรงกัน

9. สรุป

  • จำนวนรายการแสดงสองที่ ไม่ผิดเพราะอ่านจากแหล่งเดียว แต่ซ้ำ
  • สวิตช์แตะได้ทั้งแถว เพราะ Switch เล็กกว่าเกณฑ์ 48 พิกเซล
  • สีของประเภทพาเลทเขียนซ้ำสามที่ และไม่ตรงกันหนึ่งที่
  • StatusBadge รู้จัก FG/PW อยู่แล้ว ใช้แทนได้เลย
  • message จากเซิร์ฟเวอร์แสดงเมื่อไม่ว่าง — ช่องที่เพิ่มข้อมูลได้โดยไม่แก้แอป
  • ListView.separated จัดการช่องว่างระหว่างแถวให้ ไม่มีช่องว่างเกินท้ายรายการ
