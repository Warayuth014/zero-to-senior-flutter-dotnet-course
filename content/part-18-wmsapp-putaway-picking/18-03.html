ข้อ 2 — แก้อะไรใน StationSheet

จุดต้องแก้ไหม

ทางที่ทนกว่าคือย้ายพฤติกรรมเข้าไปอยู่กับข้อมูล

class StationBehavior {
  const StationBehavior({
    required this.scanButtonLabel,
    required this.scanButtonIcon,
    required this.confirmTitle,
    required this.destinationLabel,
    this.allowedType,
    this.fixedDestination,
    this.allowWrapping = false,
    this.allowDestinationChoice = true,
  });
  ...
}
const _qcStation = StationBehavior(
  scanButtonLabel: 'ค้นหา Pallet',
  scanButtonIcon: Icons.search,
  confirmTitle: 'ยืนยันส่งตรวจสอบ',
  destinationLabel: 'สถานีตรวจสอบ',
  fixedDestination: 'QC',
  allowWrapping: false,
  allowDestinationChoice: false,
);

ตอนนี้เพิ่มสถานีแบบใหม่ = เพิ่มค่าคงที่หนึ่งตัว ไม่ต้องแตะ StationSheet เลย และตอบได้ทันทีว่าสถานีแบบไหนทำอะไรโดยอ่านที่เดียว

แต่อย่าเพิ่งทำตอนนี้ — ที่สามแบบ วิธีปัจจุบันอ่านง่ายกว่าและโค้ดน้อยกว่า การเพิ่มชั้นนามธรรมก่อนที่จะมีปัญหาจริงคือสิ่งที่ทำให้โค้ดอ่านยากขึ้นโดยไม่ได้อะไร
สิ่งที่ควรทำตอนนี้คือแก้ destLabel กับ isSendStation ให้ถูก แล้วจดไว้ว่าถ้าถึงห้าแบบเมื่อไรให้กลับมาที่นี่

9. สรุป

  • StationInfo 20 บรรทัด แต่สามฟิลด์ท้ายทำให้แผ่นเดียวทำงานได้สามแบบ
  • ปรับพฤติกรรมด้วยข้อมูล ไม่ใช่การสืบทอด — เลี่ยงโค้ดซ้ำ 80%
  • เงื่อนไขกระจายหกจุดในไฟล์ สองจุดถูกทำเป็น getter ที่อ่านง่าย
  • isSendStation ตั้งชื่อไม่ตรงกับสิ่งที่คำนวณ — ตรงกันโดยบังเอิญเท่านั้น
  • setState ถูกเรียกก่อนตรวจ mounted สองจุด — สลับกับทุกจอใน Part 17
  • ไม่มี PopScope กันการปิดแผ่นระหว่างที่คำขอค้าง
  • แอปตรวจประเภทพาเลทเอง กฎมีอยู่ที่เดียวคือฝั่งแอป
  • ข้อความยืนยันบอกผลลัพธ์ทางกายภาพ ซึ่งจำเป็นในคลังที่มีหุ่นยนต์
  • ปุ่มสแกนดูตัวหมุนจาก _loadingConfirm ผิดตัวเมื่อเป็นสถานีฝั่งรับ
