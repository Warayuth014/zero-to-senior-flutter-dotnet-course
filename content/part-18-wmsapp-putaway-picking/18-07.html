เหตุผลหนักแน่นไหม

เหตุผลที่ควรมี

class StationStatus {
  const StationStatus({
    required this.stationId,
    this.palletId,
    this.destination,
    this.items = const [],
    this.palletStatus,
    this.cutItems = const [],
  });

  final String stationId;
  final String? palletId;
  final String? destination;
  final List<UnloadItem> items;
  final String? palletStatus;                 // prework เท่านั้น
  final List<Map<String, dynamic>> cutItems;  // prework เท่านั้น

  factory StationStatus.fromJson(Map<String, dynamic> json) => StationStatus(
    stationId: json['stationId'],
    palletId: json['palletId'],
    destination: json['destination'],
    items: ((json['items'] as List?) ?? [])
        .map((e) => UnloadItem.fromJson(e as Map<String, dynamic>))
        .toList(),
    palletStatus: json['palletStatus'],
    cutItems: ((json['cutItems'] as List?) ?? []).cast<Map<String, dynamic>>(),
  );

  bool get isBusy => palletId != null;
  bool get isInTransit => palletStatus == 'IN_TRANSIT';

  StationStatus copyWith({String? palletId, String? destination}) => StationStatus(
    stationId: stationId,
    palletId: palletId ?? this.palletId,
    destination: destination ?? this.destination,
    items: items,                              // เก็บไว้เสมอ
    palletStatus: palletStatus,
    cutItems: cutItems,
  );
}
ได้อะไรตรงไหน
คีย์ทั้งหมดอยู่ที่เดียวอ่านไฟล์เดียวรู้ครบ
isBusy / isInTransitแทนที่จะเทียบสตริงในสามไฟล์
copyWith ที่เก็บ items ให้เสมอแทนโค้ดกระจายสามบรรทัดใน 18.4
เทสต์ได้fromJson เป็นฟังก์ชันบริสุทธิ์

4. PutawayResult ที่มีฟิลด์ชื่อ success

class PutawayResult {
  final bool success;         // ← ชื่อชนกับ ApiResult.success
  final String palletId;
  final String stationId;
  final String destination;
  final String message;

  factory PutawayResult.fromJson(Map<String, dynamic> json) => PutawayResult(

    stationId: json['stationId'],

ตอนใช้จะมี success สองชั้น

final result = await _api.confirmPutaway(...);
if (!result.success) { ... }              // ← ของ ApiResult
// result.data!.success                   // ← ของ PutawayResult
ชั้นsuccess หมายถึง
ApiResultคำขอสำเร็จไหม (HTTP + JSON แปลงได้)
PutawayResultเซิร์ฟเวอร์บอกว่างานสำเร็จไหม

สองชั้นนี้ต่างกันจริง — เหมือน palletError ใน 17.7 ที่บอกว่าคำขอสำเร็จแต่งานล้มเหลว

แต่ไม่มีใครอ่านชั้นที่สอง

// station_sheet.dart
final result = await _api.confirmPutaway(...);
...
if (!result.success) {
  showErrorDialog(context, message: result.error!);
  return;
}

if (!mounted) return;
Navigator.pop(context);
widget.onConfirmed();          // ← ไม่เคยดู result.data!.success

ถ้าเซิร์ฟเวอร์ตอบ 200 พร้อม {"success": false, "message": "สถานีถูกใช้งานอยู่"} แอปจะปิดแผ่นและถือว่าสำเร็จ

success: json['success'] ?? true,     // ← ไม่ส่งมา = ถือว่าสำเร็จ

ค่าเริ่มต้นเป็น true ซึ่งเป็นแบบผ่อนปรนตามที่ 17.9 อธิบาย — เดาผิดแล้วปล่อยผ่าน

ต้องยืนยันกับฝั่ง .NET ก่อนตัดสิน — ถ้า endpoint นี้ตอบ 4xx เมื่อล้มเหลวเสมอ ฟิลด์ success ก็ไม่จำเป็นและควรลบทิ้ง
ถ้าตอบ 200 พร้อม success: false ได้ นี่คือบั๊กที่ทำให้ผู้ใช้คิดว่าสั่งสำเร็จทั้งที่ไม่
ฟิลด์ที่ประกาศไว้แต่ไม่มีใครอ่านคือคำถามเสมอ — ไม่ใช่ว่าไม่จำเป็น แต่แปลว่ายังไม่รู้ว่าจำเป็นไหม

ถ้าจำเป็น ต้องอ่านทั้งสองชั้น

if (!result.success) {
  showErrorDialog(context, message: result.error!);
  return;
}

final putaway = result.data!;
if (!putaway.success) {
  showErrorDialog(context, message: putaway.message);
  return;
}

ฟิลด์ถ้าไม่มีร้ายแรงไหม
palletIdTypeErrorสมเหตุสมผล — ไม่มีรหัสก็ไร้ความหมาย

itemsTypeErrorควรกัน — พาเลทว่างเป็นไปได้

แถวสุดท้ายเป็นจุดที่เสี่ยงสุด

items: (json['items'] as List).map(...)      // ไม่มี ?? []

ถ้าเซิร์ฟเวอร์ไม่ส่ง items เมื่อพาเลทว่าง จะพังทันทีตอนสแกน — และข้อความที่ผู้ใช้เห็นคือ เกิดข้อผิดพลาด: type 'Null' is not a subtype of type 'List'

items: ((json['items'] as List?) ?? [])
    .map((i) => UnloadItem.fromJson(i as Map<String, dynamic>))
    .toList(),

สังเกตว่า UnloadItem.fromJson(i) ไม่ได้ cast i — เป็นจุดเดียวกับที่ 16.8 ชี้ว่าทำให้ข้อผิดพลาดไกลจากสาเหตุ

ส่วน suggestedDestination ?? 'ASRS' เป็นค่าเริ่มต้นที่ปลอดภัย

ถ้าเดาผิดผล
ค่าที่แนะนำผิดผู้ใช้เลือกใหม่เองได้ (ตัวเลือกยังแสดงอยู่)

ต่างจาก condition ?? 'FG' ใน 17.9 ที่เดาผิดแล้วของเสียผ่านไปเป็นของดี — ที่นี่แค่แนะนำผิด ผู้ใช้ยังตัดสินใจเอง

7. ข้อผิดพลาดที่พบบ่อย

เข้าใจผิดที่จริง

9. สรุป

  • putaway มีโมเดลน้อยที่สุด — 49 บรรทัด สองคลาส และยืม UnloadItem จาก unload
  • การยืมถูกต้อง เพราะของบนพาเลทเป็นสิ่งเดียวกันไม่ว่าดูจากมุมไหน
  • สถานะสถานีเป็น Map ดิบ อ่านจากสี่ไฟล์ และไม่มีที่ไหนเขียนรวมว่ามีคีย์อะไรบ้าง
  • ควรสร้างโมเดลเมื่อ Map ถูกอ่านจากมากกว่าสองไฟล์ หรือต้องประกอบใหม่ด้วยมือ
  • PutawayResult.success มีอยู่แต่ไม่มีใครอ่าน — ฟิลด์ที่ไม่ถูกใช้คือคำถามเสมอ
  • ค่าเริ่มต้น success ?? true เป็นแบบผ่อนปรน
  • items ไม่กัน null — พาเลทว่างจะทำให้สแกนพัง
  • path ต่อสตริงทั้งรหัสพาเลทและสถานี ทั้งที่รหัสพาเลทมาจากบาร์โค้ด
  • โค้ดที่ “ต้องจำเขียนให้ครบ” หลายที่ ควรย้ายเข้าไปอยู่ในสิ่งที่ลืมไม่ได้
