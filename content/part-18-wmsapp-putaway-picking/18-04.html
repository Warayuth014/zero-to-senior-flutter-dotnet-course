3. แอนิเมชันที่คุมด้วยตัวจับเวลาซ้อนกัน

Timer? _returnAnimMidTimer;
Timer? _returnAnimEndTimer;
String? _returnAnimStation;
String _returnAnimText = '';

void _playReturnAnimation(String stationId) {
  _cancelReturnAnimation();                    // ← ยกเลิกของเก่าก่อนเสมอ
  setState(() {
    _returnAnimStation = stationId;
    _returnAnimText = 'AMR is picking up the pallet...';
  });

  _returnAnimMidTimer = Timer(const Duration(seconds: 2), () {
    if (!mounted || _returnAnimStation != stationId) return;   // ← ตรวจสองอย่าง

    setState(() => _returnAnimText = 'AMR is returning the pallet...');

    _returnAnimEndTimer = Timer(const Duration(seconds: 2), () {
      if (!mounted || _returnAnimStation != stationId) return;

      setState(() {
        _returnAnimStation = null;
        _returnAnimText = '';
      });
      _loadAllStatus();                        // ← โหลดของจริงหลังจบ
    });
  });
}

สี่วินาที แบ่งเป็นสองช่วง ๆ ละสองวินาที พร้อมข้อความคนละแบบ

เวลาข้อความ
0–2 วินาทีAMR is picking up the pallet…
2–4 วินาทีAMR is returning the pallet…
4 วินาทีจบ แล้วโหลดสถานะจริง

การป้องกันสองชั้นในแต่ละตัวจับเวลาทำได้ครบ

เงื่อนไขป้องกันอะไร
!mountedจอถูกปิดไปแล้ว — setState จะพัง
_returnAnimStation != stationIdมีแอนิเมชันของสถานีอื่นเริ่มไปแล้ว

เงื่อนไขที่สองสำคัญมาก ลองไล่กรณี

t=0.0  PW-STN-1 คืนพาเลท → เริ่มแอนิเมชัน, ตั้ง MidTimer(2s)
t=1.0  PW-STN-3 คืนพาเลท → _cancelReturnAnimation() ยกเลิก MidTimer เก่า
                          → _returnAnimStation = 'PW-STN-3'
t=3.0  MidTimer ของ STN-3 ทำงาน → ตรวจ: 'PW-STN-3' == 'PW-STN-3' ✓ ผ่าน

ในกรณีนี้ _cancelReturnAnimation() จัดการให้แล้ว — แต่การตรวจซ้ำเป็นตาข่ายชั้นที่สองที่กันกรณีที่เผลอลืมยกเลิก

นี่คือรูปแบบที่ถูกที่สุดในเรื่องตัวจับเวลาที่เจอในทั้งโปรเจกต์ — ยกเลิกของเก่าก่อนสร้างใหม่, ตรวจ mounted, ตรวจว่ายังเป็นงานเดิมอยู่, และยกเลิกทั้งสองตัวใน dispose()

  ...
}

      _returnAnimText = '';

// จบเมื่อได้รับเหตุการณ์จริง
void _handlePalletArrived(Map<String, dynamic> data) {
  final stationId = data['stationId'] as String?;
  if (stationId != null && _returnAnimStation == stationId) {
    _cancelReturnAnimation();
    setState(() { _returnAnimStation = null; _returnAnimText = ''; });
  }
  _loadAllStatus();
}

5. การอัปเดตแบบทันตาก่อนของจริง

void _handleStationDispatched(Map<String, dynamic> data) {
  final stationId = data['stationId'] as String?;
  final destination = data['destination'] as String?;

  if (stationId != null &&
      _kPWStations.any((s) => s.id == stationId && s.pwRole == PWRole.send) &&
      mounted) {
    setState(() {
      _stationStatus = {
        ..._stationStatus,
        stationId: {
          ...?_stationStatus[stationId],       // เก็บของเดิมไว้
          'stationId': stationId,
          'palletId': data['palletId'],
          'destination': destination,
          'items': _stationStatus[stationId]?['items'] ?? const [],
        },
      };
    });
  }

  if (destination == 'PREWORK') {
    _loadPreworkStationStatus();
  }
  _loadStationStatus();                        // ← โหลดของจริงตามมา
}

อัปเดตสองรอบ: จากข้อมูลในเหตุการณ์ทันที แล้วโหลดของจริงตามมา

รอบได้อะไรเวลา
1 จากเหตุการณ์สถานีเปลี่ยนเป็นสีแดงทันที~0 ms
2 จากเซิร์ฟเวอร์ข้อมูลครบ รวมรายการของบนพาเลท~200–1000 ms

ผู้ใช้เห็นการตอบสนองทันที ไม่ต้องรอเน็ต — และข้อมูลที่ขาดถูกเติมทีหลัง

สังเกตสามตัวดำเนินการที่ใช้

{..._stationStatus, stationId: {...}}        // คัดลอก Map เดิม แล้วทับคีย์เดียว
{...?_stationStatus[stationId], ...}         // คัดลอกถ้ามี ไม่มีก็ข้าม
_stationStatus[stationId]?['items'] ?? const []   // เก็บ items เดิมไว้

...? เป็นตัวกระจายที่ยอมให้ค่าเป็น null — ถ้าเขียน ... เฉย ๆ จะพังเมื่อสถานีนั้นยังไม่มีข้อมูล

และการสร้าง Map ใหม่แทนการแก้ของเดิมสำคัญ

// ผิด — แก้ Map เดิม Flutter อาจไม่เห็นว่าเปลี่ยน
_stationStatus[stationId] = {...};
setState(() {});

// ถูก — สร้างใหม่
setState(() => _stationStatus = {..._stationStatus, stationId: {...}});
สังเกตว่า items ถูกเก็บไว้เป็นพิเศษ — เพราะเหตุการณ์ SignalR ไม่ได้ส่งรายการของบนพาเลทมา ถ้าไม่เก็บไว้ การ์ดจะแสดงว่าพาเลทไม่มีของ จนกว่าการโหลดจริงจะเสร็จ
รายละเอียดแบบนี้เกิดจากการเจอปัญหาจริงแล้วแก้ ไม่ใช่จากการออกแบบล่วงหน้า

void _openStationPopup(StationInfo station) {
  if (_returnAnimStation == station.id) return;       // กันตอนแอนิเมชันเล่นอยู่

  if (station.pwRole == PWRole.receive) {
    final info = _receiveStatus[station.id];
    final palletId = info?['palletId'] as String?;
    final palletStatus = info?['palletStatus'] as String?;

    if (palletStatus == 'IN_TRANSIT') return;         // กำลังขนมา กดไม่ได้

    final cutItems = (info?['cutItems'] as List?)?.cast<Map<String, dynamic>>() ?? [];

    showModalBottomSheet(
      ...
      builder: (_) => PreworkReceiveSheet(
        station: station, userId: widget.userId,
        palletId: palletId, palletStatus: palletStatus, cutItems: cutItems,
        onCompleted: () => _playReturnAnimation(station.id),
      ),
    );
    return;
  }

  final busy = _stationStatus[station.id];
  if (busy != null) return;                            // ฝั่งส่งกำลังทำงาน

  showModalBottomSheet(
    ...
    builder: (_) => StationSheet(station: station, userId: widget.userId,
        onConfirmed: () { _loadAllStatus(); }),
  );
}

8. แบบฝึกหัด

ผู้ใช้แจ้งว่า “กดสถานี PW-STN-1 แล้วไม่มีอะไรขึ้น กดหลายรอบก็ไม่ขึ้น สักพักถึงจะเข้าได้”

  1. สาเหตุที่เป็นไปได้มีกี่อย่าง
  2. จะแยกแยะอย่างไรว่าเป็นอันไหน
  3. แก้ให้ผู้ใช้ไม่ต้องเดา
แนวคำตอบ

ข้อ 1 — สาเหตุ

PW-STN-1 เป็นสถานีฝั่งรับ จึงมีสองเงื่อนไขที่ทำให้เปิดไม่ได้

สาเหตุเงื่อนไขในโค้ด“สักพักถึงเข้าได้” ตรงไหม
แอนิเมชันคืนพาเลทเล่นอยู่_returnAnimStation == station.idตรง — จบใน 4 วินาที
พาเลทกำลังถูกขนมาpalletStatus == 'IN_TRANSIT'ตรง — จบเมื่อพาเลทถึง

ทั้งสองอธิบาย “สักพักถึงเข้าได้” ได้ทั้งคู่ — ต้องแยกด้วยระยะเวลา

ข้อ 2 — แยกอย่างไร

ถาม/สังเกตถ้าใช่
รอไม่เกิน 5 วินาทีแล้วเข้าได้แอนิเมชัน
รอเป็นสิบวินาทีหรือนาทีIN_TRANSIT
มีข้อความ “AMR is…” บนการ์ดตอนนั้นไหมมี = แอนิเมชัน
เกิดหลังจากเพิ่งกดคืนพาเลทเองหรือเปล่าใช่ = แอนิเมชัน

คำถามที่สามตรงที่สุดเพราะดูได้จากหน้าจอทันที และไม่ต้องพึ่งความทรงจำของผู้ใช้

แต่มีเรื่องที่ต้องคิดต่อ — แถบข้อความที่โผล่ทุกครั้งที่กดจะน่ารำคาญถ้าผู้ใช้กดหลายครั้ง

ทางที่ดีกว่าคือทำให้การ์ดบอกตั้งแต่แรกว่ากดไม่ได้ แล้วแถบข้อความเป็นตาข่ายชั้นสอง

      : isInTransit

          : null,

)

ข้อสรุปที่ใหญ่กว่า: ทุกจุดที่โค้ดเขียน return; เงียบ ๆ คือจุดที่ผู้ใช้จะไม่รู้ว่าเกิดอะไรขึ้น

ค้นหา return; ที่ไม่มีการแสดงผลอะไรก่อนหน้า เป็นวิธีหาจุดแบบนี้ได้เร็ว — ในจอนี้เจอสามจุด และในทั้ง Part 17 เจออีกหลายจุด

9. สรุป

  • หกสถานีแบ่งสองบทบาทด้วย pwRole แล้ววาดเป็นสองคอลัมน์ที่ใช้สีอุ่น/เย็นแยกทิศทาง
  • สถานะสองชุดจากสอง endpoint โหลดพร้อมกันด้วย Future.wait
  • ทั้งสองเมธอดล้มเหลวแล้วเงียบ — สถานีจะดูว่างทั้งที่อาจไม่ว่าง
  • ตัวจับเวลาสองตัวซ้อนกัน ตรวจทั้ง mounted และว่ายังเป็นงานเดิม — รูปแบบที่ถูกที่สุดในโปรเจกต์
  • แต่แอนิเมชันจบตามเวลา ไม่ใช่ตามความจริงที่หุ่นยนต์ทำ
  • อัปเดตสองรอบ: จากเหตุการณ์ทันที แล้วโหลดของจริงตามมา
  • ...? จำเป็นเมื่อค่าที่จะกระจายอาจเป็น null
  • items ถูกเก็บไว้เป็นพิเศษเพราะเหตุการณ์ไม่ได้ส่งมา
  • สามเงื่อนไขที่ทำให้เปิดแผ่นไม่ได้ ทั้งหมด return เงียบ

บทหน้า: การสแกนพาเลทและตัวเลือกปลายทาง — สามตัวเลือกที่เปลี่ยนกันเองตามเงื่อนไข
