Future<void> _sendPallet() async {
  final palletId = _sendPalletCtrl.text.trim().toUpperCase();
  if (palletId.isEmpty) {
    showErrorDialog(context, message: 'กรุณาใส่ Pallet ID');
    _sendPalletFocus.requestFocus();
    return;
  }

  // ── 1) Preview ──
  setState(() => _loading = true);
  final preview = await _api.previewReturnPallet(palletId);
  if (!mounted) return;
  setState(() => _loading = false);

  if (!preview.success) { ... return; }

  final info = preview.data!;
  if (!info.canReturn) {
    showErrorDialog(context, message: info.reason);
    _sendPalletFocus.requestFocus();
    return;
  }

  // ── 2) Show popup confirm ──
  final dest = info.destination!;
  ...
  final confirmed = await showDialog<bool>(...);
  if (confirmed != true || !mounted) {
    _sendPalletFocus.requestFocus();
    return;
  }

  // ── 3) Execute ──
  setState(() => _loading = true);
  final ok = await _api.returnPallet(palletId: palletId, destination: dest);
  ...
}

คอมเมนต์แบ่งขั้นไว้ชัด และ endpoint ฝั่งเซิร์ฟเวอร์บอกไว้ว่าไม่มีผลข้างเคียง

/// Preview ก่อนคืน pallet — เช็คว่าสแกนได้ไหม + จะส่งไปไหน (ไม่มี side effect)

ขั้นได้อะไรปลอดภัยไหม
1 previewส่งได้ไหม และจะไปไหนอ่านอย่างเดียว ยิงซ้ำได้
2 ยืนยันผู้ใช้เห็นปลายทางก่อนตัดสินไม่ยิงอะไร
3 สั่งจริงพาเลทถูกส่งมีผลข้างเคียง
ทำไมรูปแบบนี้สำคัญ — ปลายทางถูกตัดสินโดยเซิร์ฟเวอร์ ไม่ใช่ผู้ใช้
/// - ถ้าไม่ระบุ → backend ตัดสินใจเอง:
///   - PACKED → ZONE_PACK
///   - อื่นๆ (PICKING/AVAILABLE) → ASRS
ถ้าไม่มี preview ผู้ใช้จะสแกนแล้วพาเลทหายไปที่ใดที่หนึ่งโดยไม่รู้ล่วงหน้า
preview ทำให้ผู้ใช้เห็นก่อนว่า “พาเลทนี้จะไป ZONE PACK” แล้วค่อยยืนยัน

ปุ่มในกล่องยืนยันปรับตามปลายทาง

final destLabel = dest == 'ZONE_PACK' ? 'ZONE PACK' : dest;
final destIcon = dest == 'ZONE_PACK' ? MdiIcons.truckDeliveryOutline : MdiIcons.warehouse;
final btnText = dest == 'ZONE_PACK' ? 'ส่งไป ZONE PACK' : 'ส่งกลับ ASRS';

ปุ่มบอกสิ่งที่จะเกิด ไม่ใช่คำว่า “ยืนยัน” — ผู้ใช้ที่กดเร็วโดยไม่อ่านเนื้อหาก็ยังเห็นจากปุ่ม

สามบรรทัดนี้เป็นเงื่อนไขสองทางอีกครั้ง — ทุกอย่างที่ไม่ใช่ ZONE_PACK จะถูกเรียกว่า ASRS แม้ปลายทางจะเป็นอย่างอื่น

สั่งไม่สำเร็จ → แสดงข้อความ → ให้ผู้ใช้ลองใหม่

แต่การส่งพาเลทเป็นคำสั่งที่ไม่รู้ผลเมื่อเน็ตหลุด — คำสั่งอาจไปถึงแล้ว หุ่นยนต์อาจกำลังมา แต่แอปบอกว่าล้มเหลว

ผู้ใช้ทำถ้าคำสั่งไปถึงแล้วจริง
สแกนซ้ำแล้วส่งอีกครั้งpreview จะบอกว่า canReturn = false เพราะพาเลทถูกส่งไปแล้ว

โชคดีที่ preview ป้องกันไว้โดยบังเอิญ — ผู้ใช้ที่ลองซ้ำจะถูกปฏิเสธที่ขั้น 1 แทนที่จะสั่งซ้ำจริง

แต่ข้อความที่ได้จะสับสน — ผู้ใช้เห็น “ส่งไม่สำเร็จ” แล้วสแกนซ้ำ แล้วเห็นข้อความจาก info.reason ที่บอกว่าพาเลทนี้ส่งไม่ได้
ทั้งที่ความจริงคือ “ส่งไปแล้ว” ซึ่งเป็นสิ่งที่ผู้ใช้ต้องการรู้
if (!ok.success) {

        'คำสั่งอาจส่งไปแล้ว — สแกน $palletId ซ้ำเพื่อตรวจสถานะก่อนสั่งใหม่',
  );
  _sendPalletFocus.requestFocus();

8. แบบฝึกหัด

หัวหน้ากะบอกว่า “คนงานยืนรอหน้าจอนี้ คอยดึงรีเฟรชทุกสิบวินาทีเพื่อดูว่างานพร้อมหรือยัง เสียเวลามาก”

  1. อาการนี้เกิดจากอะไร
  2. เสนอทางแก้สามระดับ
  3. ทำไมการโหลดถี่ขึ้นอย่างเดียวไม่พอ
แนวคำตอบ

ข้อ 1 — สาเหตุ

ข้อเท็จจริงผล
งานเริ่มที่สถานะ WAITINGผู้ใช้ทำอะไรไม่ได้
เปลี่ยนเป็น PICKING เมื่อหุ่นยนต์ขนพาเลทมาถึงเป็นเหตุการณ์ที่ผู้ใช้ไม่ได้ทำ
จอโหลดใหม่เฉพาะตอนเปิดและตอนดึงลงผู้ใช้ต้องถามเอง
ไม่มี SignalR ในจอนี้ไม่มีอะไรบอกเมื่อเปลี่ยน

ตรงกับหลักของ 12.1 พอดี — ข้อมูลที่เปลี่ยนโดยที่ผู้ใช้ไม่ได้กด คือกรณีที่ต้องผลักข้อมูลมาหา ไม่ใช่ให้ผู้ใช้ถาม

ฝั่ง putaway ตัดสินใจแบบนี้ถูกแล้ว (มี SignalR) ส่วนจอนี้มีความต้องการเดียวกันแต่ไม่ได้ทำ

ข้อ 2 — สามระดับ

ระดับทำอะไรต้องแตะเซิร์ฟเวอร์ไหมความสด
1โหลดใหม่ทุก 30 วินาที เฉพาะเมื่อมีงาน WAITINGไม่ช้าสุด 30 วิ
2เพิ่ม + แจ้งเตือนเมื่อมีงานเปลี่ยนเป็น PICKINGไม่เท่าเดิม แต่ผู้ใช้ไม่ต้องจ้อง
3ฟัง SignalRต้องทันที

ระดับ 2 คุ้มที่สุด เพราะแก้อาการที่หัวหน้าเล่ามาโดยตรง — คนงานยืนจ้องจอ

Timer? _pollTimer;
Set<String> _knownPickingIds = {};

@override
void initState() {
  super.initState();
  _load();
  _pollTimer = Timer.periodic(const Duration(seconds: 30), (_) {
    // ถามซ้ำเฉพาะเมื่อมีอะไรให้รอ — ไม่มีงานรอ ก็ไม่ต้องถาม
    if (_orders.any((o) => o.isWaiting)) _load();
  });
}

@override
void dispose() {
  _pollTimer?.cancel();
  _sendPalletCtrl.dispose();
  _sendPalletFocus.dispose();
  super.dispose();
}

Future<void> _load() async {
  ...
  if (!res.success) { ... return; }

  final orders = res.data!;
  final nowPicking = orders.where((o) => o.isPicking).map((o) => o.pickOrderId).toSet();
  final newlyReady = nowPicking.difference(_knownPickingIds);

  setState(() {
    _orders = orders;
    _knownPickingIds = nowPicking;
  });

  // งานที่เพิ่งพร้อม — บอกด้วยเสียงสั่น ผู้ใช้ไม่ต้องจ้องจอ
  if (newlyReady.isNotEmpty && mounted) {
    showSuccessSnackbar(
      context,
      newlyReady.length == 1
          ? 'งาน ${newlyReady.first} พร้อมหยิบแล้ว'
          : 'มีงานพร้อมหยิบ ${newlyReady.length} รายการ',
    );
  }
}

showSuccessSnackbar สั่นเบาให้ด้วย (16.9) — ผู้ใช้ที่ทำอย่างอื่นอยู่จะรู้สึกได้

ต้องตรวจอย่างหนึ่ง: ครั้งแรกที่โหลด _knownPickingIds ว่าง ทำให้ทุกงานที่ PICKING ถูกนับเป็น “เพิ่งพร้อม”

ข้อ 3 — ทำไมโหลดถี่ขึ้นอย่างเดียวไม่พอ

ถ้าโหลดทุก 5 วินาทีแต่ไม่แจ้งเตือน
ผู้ใช้ยังต้องจ้องจออยู่ดี เพราะไม่รู้ว่าเมื่อไรจะเปลี่ยน
กินแบตมากขึ้นหกเท่า
20 เครื่อง × 12 ครั้ง/นาที = 240 คำขอ/นาที

อาการที่หัวหน้าเล่าคือ “ยืนรอหน้าจอ” — ปัญหาไม่ใช่ข้อมูลช้า แต่คือผู้ใช้ไม่มีทางรู้โดยไม่มอง

การแจ้งเตือนแก้อาการนั้นได้แม้ข้อมูลจะช้า 30 วินาที เพราะคนงานเดินไปทำอย่างอื่นได้แล้ว

หลักที่ใช้ได้ทั่วไป: เมื่อผู้ใช้บ่นว่าต้องคอยรีเฟรช ให้ถามก่อนว่าเขาต้องการข้อมูลที่สดกว่า หรือไม่ต้องคอยดู
สองอย่างนี้แก้คนละวิธี และอย่างที่สองมักเป็นสิ่งที่ต้องการจริง

9. สรุป

  • จอเดียวทำสองเรื่องที่ไม่เกี่ยวกัน เพราะผู้ใช้ทำสลับกันตลอดกะ
  • Column + Expanded ตรงกลาง ทำให้ช่องสแกนตรึงอยู่ล่างเสมอ
  • ตัวเลขสรุปคำนวณจาก _orders ตัวเดียว แต่สีถูกเลือกให้ต่างกัน ไม่ได้เลือกตามความหมาย
  • งานที่ยัง WAITING ยังกดได้ แล้วอธิบายด้วยกล่อง — ดีกว่าที่ฝั่ง putaway ทำ
  • แต่ไม่มีอะไรบอกเมื่อสถานะเปลี่ยน ผู้ใช้ต้องดึงรีเฟรชเอง
  • การส่งพาเลทใช้รูปแบบ preview → ยืนยัน → สั่งจริง เพราะปลายทางตัดสินโดยเซิร์ฟเวอร์
  • ปุ่มยืนยันบอกสิ่งที่จะเกิด ไม่ใช่คำว่า “ยืนยัน”
  • สั่งไม่สำเร็จแล้วไม่บอกว่าอาจไปถึงแล้ว — preview ป้องกันไว้โดยบังเอิญ
  • ช่องสแกนล้างเฉพาะเมื่อสำเร็จ ทำให้ค่าต่อกันเมื่อล้มเหลว
