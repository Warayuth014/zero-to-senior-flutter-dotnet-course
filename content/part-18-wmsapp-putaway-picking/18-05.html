// putaway/shared/models/putaway_rules.dart
List<String> allowedDestinations({
  required String palletType,
  required int totalQty,
}) {
  if (palletType == 'PW') return const ['ASRS', 'PREWORK'];
  if (palletType != 'FG') return const ['ASRS'];

  // FG ที่หนักเกิน 50 ชิ้น ต้องเข้าคลังหลัก — Rack รับไม่ไหว
  if (totalQty > 50) return const ['ASRS'];
  return const ['ASRS', 'REPLENISH'];
}

void main() {
  group('allowedDestinations', () {
    test('PW ส่งได้แค่ ASRS กับ Prework', () {
      final dests = allowedDestinations(palletType: 'PW', totalQty: 10);
      expect(dests, containsAll(['ASRS', 'PREWORK']));
      expect(dests, isNot(contains('REPLENISH')));
    });

    test('FG น้อยกว่าหรือเท่ากับ 50 ส่ง Replenish ได้', () {
      expect(allowedDestinations(palletType: 'FG', totalQty: 50),
          contains('REPLENISH'));
    });

    test('FG เกิน 50 ส่ง Replenish ไม่ได้', () {
      expect(allowedDestinations(palletType: 'FG', totalQty: 51),
          isNot(contains('REPLENISH')));
    });

    test('ทุกกรณีส่งเข้า ASRS ได้เสมอ', () {
      for (final type in ['PW', 'FG', 'QC', '']) {
        for (final qty in [0, 50, 51, 9999]) {
          expect(allowedDestinations(palletType: type, totalQty: qty),
              contains('ASRS'),
              reason: 'type=$type qty=$qty');
        }
      }
    });

    test('ประเภทที่ไม่รู้จัก ส่งได้แค่ ASRS', () {
      expect(allowedDestinations(palletType: 'QC', totalQty: 10), ['ASRS']);
    });
  });
}

เทสต์ที่สองกับสามทดสอบขอบเขต — 50 ผ่าน 51 ไม่ผ่าน ซึ่งเป็นจุดที่โค้ดผิดได้ง่ายที่สุด (> กับ >=)

เทสต์ที่สี่ใช้ reason: เพื่อให้รู้ว่าล้มที่ชุดค่าไหน — สำคัญเมื่อวนหลายค่าในเทสต์เดียว

ข้อสรุปที่ใหญ่กว่าโจทย์: กฎธุรกิจที่อยู่ในเงื่อนไขของ widget เทสต์ได้เฉพาะผ่านการวาดหน้าจอ ซึ่งช้าและเปราะ
พอย้ายออกมาเป็นฟังก์ชันบริสุทธิ์ เทสต์ห้าข้อข้างบนรันในไม่กี่มิลลิวินาที และยังใช้ได้แม้จะเปลี่ยนหน้าตาทั้งหมด

  • สามไฟล์ซ้อนกัน: ตัวเลือก → ปุ่ม และมีสวิตช์อยู่ข้างใน
  • กฎ “PW ไป Prework ได้ FG ไป Replenish ได้” อยู่ในเงื่อนไขของ widget ไม่มีที่อื่นเขียนไว้
  • ย้ายออกมาเป็นฟังก์ชันบริสุทธิ์แล้วเทสต์ได้ทันที และใช้ตรวจซ้ำตอนยืนยันได้ด้วย
  • สวิตช์พันฟิล์มแสดงเฉพาะเมื่อปลายทางเป็น ASRS และค่าถูกล้างเมื่อเปลี่ยนปลายทาง
  • เมื่อ widget ถูกซ่อน ต้องถามเสมอว่าค่าที่มันคุมถูกจัดการหรือยัง
  • isPW ? A : B ทำให้ทุกอย่างที่ไม่ใช่ PW ได้ B — เจอรูปแบบนี้สามที่แล้ว
  • ปุ่มปลายทางบอกทั้งรหัส ผลลัพธ์ และไอคอน — ใช้ได้ทั้งคนใหม่และคนชำนาญ
  • PalletItemsPage เป็น StatelessWidget รับข้อมูลผ่านตัวสร้าง เปิดทันทีไม่มีตัวหมุน

บทหน้า: การพันฟิล์ม รายการของบนพาเลท และสิ่งที่เกิดขึ้นหลังกดยืนยัน
