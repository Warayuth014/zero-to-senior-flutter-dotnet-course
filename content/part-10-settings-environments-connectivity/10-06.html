ต้องผ่านอะไรมาก่อน: 10.5, 9.7 (store), 8.11 (session ผูกกับเซิร์ฟเวอร์)
จบบทนี้แล้วจะทำได้: จัดการ “ตัวที่ใช้งานอยู่” ให้มีหนึ่งตัวเสมอ และรู้ว่าการสลับกระทบอะไรบ้าง

1. กฎสามข้อที่ต้องเป็นจริงตลอดเวลา

กฎถ้าละเมิดจะเกิดอะไร
มี profile อย่างน้อยหนึ่งตัวเสมอไม่มีที่อยู่ให้ยิง และไม่มีทางพิมพ์ตัวใหม่เข้าไป
มีตัวที่ใช้งานอยู่หนึ่งตัวเสมอคำขอไม่รู้จะไปไหน
ตัวที่ใช้งานอยู่ต้องมีอยู่ในรายการจริงactiveBaseUrl เป็น null ทั้งที่มีรายการอยู่

กฎแบบนี้เรียกว่า invariant — สิ่งที่ต้องจริงก่อนและหลังทุกการกระทำ วิธีที่ดีที่สุดคือทำให้ละเมิดไม่ได้ ไม่ใช่ตรวจทีหลัง

2. กฎข้อที่หนึ่ง: มีอย่างน้อยหนึ่งตัว

บังคับที่สองจุด — ตอนเปิดแอปครั้งแรก และตอนลบ

Future<void> load() async {
  ...
  if (_profiles.isEmpty) {
    final seed = seedProfile(id: _createId());
    _profiles.add(seed);
    _activeId = seed.id;
    await _persist();          // บันทึกด้วย ไม่ใช่แค่สร้างในหน่วยความจำ
  }
  _loaded = true;
  notifyListeners();
}
Future<bool> remove(String id, {String? nextActiveId}) async {
  if (_profiles.length <= 1) return false;      // ลบตัวสุดท้ายไม่ได้
  ...
}

บรรทัด await _persist() ในตัวแรกสำคัญ ถ้าลืม จะสร้าง profile ใหม่ทุกครั้งที่เปิดแอป และ id ที่ผู้ใช้เห็นจะเปลี่ยนไปเรื่อย ๆ เทสต์จับข้อนี้ไว้

test('profile เริ่มต้นถูกบันทึกลงเครื่องด้วย ไม่ใช่สร้างใหม่ทุกครั้ง', () async {
  await newStore().load();

  final second = ProfileStore(storage, createId: () => 'ไม่ควรถูกเรียก');
  await second.load();
  expect(second.profiles.single.id, 'p1');
});
เทคนิคที่ใช้ในเทสต์นี้: ตั้งค่า createId ให้คืนข้อความที่ทำให้เทสต์ล้มเหลวอย่างชัดเจนถ้าถูกเรียก — ตรวจว่า “ต้องไม่เกิดขึ้น” ได้โดยไม่ต้องนับจำนวนครั้ง

3. กฎข้อที่สาม: ตัวที่ใช้งานอยู่ต้องมีจริง

Future<void> activate(String id) async {
  if (_profiles.every((p) => p.id != id)) return;   // ไม่มี ก็ไม่เปลี่ยน
  _activeId = id;
  ...
}
Future<bool> remove(String id, {String? nextActiveId}) async {
  ...
  _profiles.removeAt(index);
  if (_activeId == id) {
    final replacement = nextActiveId ?? _profiles.first.id;
    _activeId = _profiles.any((p) => p.id == replacement)
        ? replacement
        : _profiles.first.id;                  // แม้คนเรียกส่ง id มั่ว ก็ยังปลอดภัย
  }
  ...
}

nextActiveId เป็นพารามิเตอร์เพราะการเลือกว่าจะไปใช้ตัวไหนต่อ ควรเป็นของผู้ใช้ ไม่ใช่ให้แอปเลือกให้ แต่ถ้าคนเรียกไม่ส่งมา หรือส่งมาผิด store ก็ยังต้องรักษากฎไว้ได้เอง

4. บันทึก ≠ สลับ

/// เพิ่ม profile ใหม่ โดย**ไม่**เปลี่ยนตัวที่ใช้งานอยู่
///
/// บันทึกกับสลับเป็นคนละการกระทำโดยตั้งใจ — คนที่จดที่อยู่เครื่องทดสอบไว้
/// ตอนบ่าย ไม่ได้อยากให้ PDA เปลี่ยนไปยิงเครื่องทดสอบทันที
Future<ServerProfile> saveAsNew(ServerProfile draft) async { ... }

นี่เป็นการตัดสินใจด้าน UX ที่สะท้อนอยู่ใน API ของ store ผลของการเลือกอีกทางคือ

ถ้าบันทึกแล้วสลับให้เลยผลจริงที่หน้างาน
ผู้ใช้จดที่อยู่ไว้เฉย ๆPDA ย้ายไปยิงเครื่องทดสอบทันที งานที่ทำอยู่พัง
ผู้ใช้แก้ typo ในชื่อ profile ที่ไม่ได้ใช้สลับไปเครื่องนั้นโดยไม่ตั้งใจ
ผู้ใช้ต้องการสลับจริง ๆทำได้ในคลิกเดียว

สองแถวแรกแพงกว่าความสะดวกที่ได้ในแถวที่สาม จึงแยกเป็นสองปุ่ม

5. สลับแล้วเกิดอะไรขึ้นบ้าง

Future<void> activate(String id) async {
  if (_profiles.every((p) => p.id != id)) return;
  _activeId = id;
  _status = ServerStatus.unknown;      // ← สำคัญ
  _lastResult = null;
  await _persist();
  notifyListeners();
}

การล้างสถานะสำคัญกว่าที่ดู ถ้าไม่ล้าง จุดสีเขียวที่ค้างอยู่จะบอกผู้ใช้ว่า “เชื่อมต่อได้” ทั้งที่นั่นเป็นผลของการตรวจเซิร์ฟเวอร์ตัวเก่า เทสต์บังคับข้อนี้

test('สลับแล้วต้องลืมผลตรวจของเครื่องเก่า', () async {
  ...ตรวจจนได้ connected...
  await store.activate(saved.id);

  expect(store.status, ServerStatus.unknown);
  expect(store.lastResult, isNull);
});

แล้วสิ่งที่ไม่เกิดขึ้นล่ะ — ไม่มีการสร้าง ApiClient ใหม่

static String get apiBaseUrl =>
    resolveBaseUrl(ConnectionController.instance.activeBaseUrl);

ApiClient อ่าน base URL ใหม่ทุกครั้งที่ยิง ไม่ได้เก็บไว้ตอนสร้าง ผลคือการสลับมีผลทันทีทั้งแอป และ token ที่ติดอยู่กับ client ตัวเดิมก็ไม่หาย

6. แต่ token ที่ไม่หาย ก็ใช้กับเซิร์ฟเวอร์ใหม่ไม่ได้

นี่คือจุดที่ Part 10 ต่อกับ Part 8 พอดี

สลับ profile
  → ApiClient ยิงไปเซิร์ฟเวอร์ใหม่ พร้อม token เก่าที่ยังติดอยู่
    → เซิร์ฟเวอร์ใหม่ไม่รู้จัก token นี้ ตอบ 401
      → SessionStore.handleUnauthorized() (8.7)
        → ออกจากระบบ พร้อมเหตุผล serverChanged (8.11)

ทำงานถูกต้อง แต่ผู้ใช้ต้องรอให้คำขอแรกล้มเหลวก่อน ทางที่ดีกว่าคือให้ SessionStore ตรวจตั้งแต่ตอนสลับ

if (stored.serverId != currentServerId) {
  await _clearTo(AuthStatus.signedOut, reason: SignOutReason.serverChanged);
  return;
}

สิ่งที่ต้องระวังคืออย่าให้ ProfileStore เรียก SessionStore โดยตรง ไม่งั้นจะได้การพึ่งพาที่วนกลับตามที่ 9.15 เตือนไว้ ให้ shell ที่ถือทั้งสองตัวเป็นคนต่อ

profileStore.addListener(() {
  sessionStore.handleServerChanged(profileStore.active?.id);
});

7. ข้อผิดพลาดที่พบบ่อย

อาการสาเหตุวิธีแก้
สลับแล้วบางส่วนของแอปยังยิงที่เก่าเก็บ base URL ไว้ตอนสร้าง clientอ่านใหม่ทุกครั้งที่ยิง
จุดสถานะเขียวทั้งที่ยังไม่ได้ตรวจเครื่องใหม่ไม่ล้างสถานะตอนสลับตั้งเป็น unknown
ลบ profile แล้วแอปยิงไม่ได้เลยลบตัวที่ใช้งานอยู่โดยไม่มีตัวแทนบังคับให้มีตัวแทนเสมอ
บันทึกที่อยู่ใหม่แล้วงานที่ทำอยู่พังบันทึกแล้วสลับให้อัตโนมัติแยกเป็นสองการกระทำ
สลับแล้วยังเห็นข้อมูลของเซิร์ฟเวอร์เก่าstore ของฟีเจอร์อื่นไม่ได้ล้างล้างพร้อมออกจากระบบ (8.8)

8. แบบฝึกหัด

ผู้ใช้กำลังนับสต๊อกค้างอยู่ครึ่งทาง แล้วเปิดหน้าตั้งค่าไปสลับ profile

  1. ควรให้สลับได้ไหม
  2. ถ้าให้สลับ ต้องทำอะไรกับงานที่ค้างอยู่
แนวคำตอบ

1. ควรให้สลับได้ แต่ต้องเตือน การห้ามสลับตอนมีงานค้างฟังดูปลอดภัย แต่จะทำให้ผู้ใช้ที่สลับผิดเครื่องตั้งแต่แรกติดกับ — เขาจะนับต่อไม่ได้ (เพราะเป็นเครื่องผิด) และสลับก็ไม่ได้ (เพราะมีงานค้าง)

2. สามอย่างตามลำดับ

  • เตือนก่อน ว่างานที่ค้างอยู่จะหาย พร้อมบอกว่าค้างอยู่กี่รายการ — ตัวเลขทำให้ผู้ใช้ตัดสินใจได้จริง ต่างจากคำเตือนลอย ๆ
  • ล้างให้หมด ถ้ายืนยัน — รายการที่นับไปแล้วเป็นของเซิร์ฟเวอร์เก่า เอามาใช้กับเครื่องใหม่ไม่ได้ นี่คือเหตุผลเดียวกับที่ 8.8 ล้าง store ทั้งหมดตอนออกจากระบบ
  • อย่าพยายามย้ายงานข้ามเซิร์ฟเวอร์ — รหัสพาเลทและรหัสสินค้าอาจซ้ำกันโดยบังเอิญระหว่างสองระบบ การย้ายไปจะเขียนข้อมูลผิดที่แบบที่ตามแก้ยากมาก

ถ้าอยากทำให้ดีกว่านั้น: ให้เก็บงานที่ค้างไว้ผูกกับ serverId แล้วเมื่อสลับกลับมาที่เดิม ค่อยเสนอให้ทำต่อ — แต่เป็นงานของ Part 29 ไม่ใช่ของบทนี้

9. สรุป

  • เขียนกฎที่ต้องจริงเสมอออกมาก่อน แล้วทำให้ละเมิดไม่ได้
  • บันทึกกับสลับเป็นคนละการกระทำ
  • สลับแล้วต้องล้างผลตรวจของเครื่องเก่า
  • อ่าน base URL ใหม่ทุกครั้งที่ยิง ไม่ใช่เก็บไว้ตอนสร้าง client
  • สลับเซิร์ฟเวอร์กระทบ session ให้ shell เป็นคนต่อสองอย่างเข้าด้วยกัน
